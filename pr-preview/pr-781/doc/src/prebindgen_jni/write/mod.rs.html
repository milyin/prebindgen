18/// What writing produces, as it accumulates.
19pub(crate) struct Out {
20    pub rust: RustFile,
21    /// Kotlin declarations by package, in the order written.
22    kt: BTreeMap<String, Vec<String>>,
23    /// The `external fun` declarations of the native holder.
24    pub externs: Vec<String>,
25}
26
27impl Out {
28    pub(crate) fn kt(&mut self, pkg: &str, decl: String) {
29        self.kt.entry(pkg.to_string()).or_default().push(decl);
30    }
31}
32
33/// Write every item of the plan.
34pub(crate) fn write(plan: &Plan) -> Res<Generation> {
35    let mut out = Out {
36        rust: RustFile::new(),
37        kt: BTreeMap::new(),
38        externs: Vec::new(),
39    };
40    out.rust.push(rust_prelude(plan));
41    out.kt(&plan.base_pkg, support::PRELUDE.to_string());
42    for item in &plan.items {
43        match item {
44            Item::Class {
45                class,
46                methods,
47                ctors,
48            } => class::class(plan, &mut out, class, methods, ctors)?,
49            Item::Function { pkg, binding } => {
50                let w = function::function(plan, &mut out, binding)?;
51                out.kt(pkg, w.text);
52            }
53            Item::Constant { pkg, name, binding } => {
54                class::constant(plan, &mut out, pkg, name, binding)?
55            }
56        }
57    }
58    for s in &plan.supports {
59        let (pkg, text) = support::support(plan, s)?;
60        out.kt(&pkg, text);
61    }
62    let holder = harness(plan, &out.externs);
63    out.kt(&plan.base_pkg, holder);
64    out.rust.guards(plan.flat);
65    let kotlin = out
66        .kt
67        .into_iter()
68        .map(|(pkg, decls)| {
69            let text = format!(
70                "// Auto-generated by prebindgen-jni — do not edit by hand.\n@file:Suppress(\"RedundantVisibilityModifier\", \"unused\", \"UNCHECKED_CAST\", \"FunctionName\", \"ClassName\", \"PropertyName\", \"LocalVariableName\", \"RemoveRedundantQualifierName\")\n\npackage {pkg}\n\n{}",
71                decls.join("\n")
72            );
73            (pkg, text)
74        })
75        .collect();
76    Ok(Generation {
77        rust: out.rust,
78        kotlin,
79    })
80}

82/// The object holding the `external fun` declarations.
83fn harness(plan: &Plan, externs: &[String]) -> String {
84    let init = plan
85        .native_init
86        .as_ref()
87        .map(|s| format!("    init {{\n        {s}\n    }}\n\n"))
88        .unwrap_or_default();
89    format!(
90        "internal object {} {{\n{init}{}\n}}\n",
91        plan.harness,
92        externs.join("\n\n")
93    )
94}

96/// The Rust side's shared helper: reporting a binding error to Kotlin.
97fn rust_prelude(plan: &Plan) -> proc_macro2::TokenStream {
98    let je = format!("{}/JniErrorHandler", plan.base_pkg.replace('.', "/"));
99    quote! {
100        /// Report a binding error to the Kotlin error sink, unless a
101        /// Kotlin exception is already pending.
102        #[allow(dead_code)]
103        fn __jni_signal(
104            env: &mut ::prebindgen_jni_runtime::jni::JNIEnv,
105            sink: &::prebindgen_jni_runtime::jni::objects::JObject,
106            msg: &str,
107        ) {
108            static __M: ::prebindgen_jni_runtime::CachedIfaceMethod =
109                ::prebindgen_jni_runtime::CachedIfaceMethod::new();
110            if env.exception_check().unwrap_or(false) {
111                return;
112            }
113            let __s = match env.new_string(msg) {
114                ::core::result::Result::Ok(s) => s,
115                ::core::result::Result::Err(_) => return,
116            };
117            let _ = __M.call_object(
118                env,
119                #je,
120                "run",
121                "(Ljava/lang/String;)Ljava/lang/Object;",
122                sink,
123                &[::prebindgen_jni_runtime::jni::sys::jvalue { l: __s.as_raw() }],
124            );
125        }
126    }
127}

129/// JNI's escaping of one name component: `_` → `_1`, `;` → `_2`, `[` →
130/// `_3`, anything outside ASCII alphanumerics → `_0xxxx`.
131fn jni_escape(s: &str) -> String {
132    let mut out = String::new();
133    for c in s.chars() {
134        match c {
135            '_' => out.push_str("_1"),
136            ';' => out.push_str("_2"),
137            '[' => out.push_str("_3"),
138            c if c.is_ascii_alphanumeric() => out.push(c),
139            c => {
140                let mut buf = [0u16; 2];
141                for u in c.encode_utf16(&mut buf) {
142                    out.push_str(&format!("_0{u:04x}"));
143                }
144            }
145        }
146    }
147    out
148}

150/// The JNI symbol of a native method: `Java_<class>_<method>`.
151pub(crate) fn jni_symbol(class_fqn: &str, method: &str) -> String {
152    let class = class_fqn
153        .split('.')
154        .map(jni_escape)
155        .collect::<Vec<_>>()
156        .join("_");
157    format!("Java_{class}_{}", jni_escape(method))
158}
