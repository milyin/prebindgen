1//! Input selectors: a parameter supplied in one of several ways, chosen at
2//! run time (`expand_param!(T).variant(fun!(ctor)).variant_self()`).
3//!
4//! Kotlin passes a selector `Int` plus every variant's arguments side by
5//! side, the inactive ones `null`: a constructor's arguments, each nullable,
6//! or the handle itself. Rust builds the value with the constructor, or
7//! takes (or borrows) the handle. `split_on_param` adds one typed overload
8//! per variant.
9//!
10//! An expansion with a single constructor and no handle variant is
11//! *direct*: there is nothing to choose, so Kotlin passes the constructor's
12//! arguments as the parameter and no selector crosses. An optional
13//! parameter is direct only when its absence shows in the arguments — the
14//! constructor takes at least one, and none is itself optional; otherwise it
15//! keeps the selector, whose `-1` means `None`.
16

37/// A resolved selector parameter.
38#[derive(Clone)]
39pub(crate) struct Selector {
40    pub param: FlatParam,
41    pub class: Rc<Class>,
42    pub variants: Vec<SelVariant>,
43    /// `Option<..>`: selector `-1` is `None`.
44    pub optional: bool,
45    /// How the callee takes the value.
46    pub access: Access,
47}
48
49#[derive(Clone)]
50#[allow(clippy::large_enum_variant)] // one per selector variant
51pub(crate) enum SelVariant {
52    Build { func: Function, callee: TokenStream },
53    Handle,
54}
55
56impl Selector {
57    /// A single constructor and nothing else, and — for an optional
58    /// parameter — arguments that can say the value is absent: no selector
59    /// crosses.
60    pub(crate) fn is_direct(&self) -> bool {
61        let [SelVariant::Build { func, .. }] = self.variants.as_slice() else {
62            return false;
63        };
64        !self.optional
65            || (!func.params.is_empty()
66                && func
67                    .params
68                    .iter()
69                    .all(|p| !matches!(p.ty.kind(), TypeKind::Optional(_))))
70    }
71}
72
73/// The Kotlin name of argument `j` of variant `i` of selector parameter
74/// `p`: `p1` for a one-argument constructor, `p01`, `p02` otherwise.
75fn arg_name(p: &str, i: usize, j: usize, arity: usize) -> String {
76    if arity == 1 {
77        format!("{p}{i}")
78    } else {
79        format!("{p}{i}{j}")
80    }
81}

84    /// The selector for `param`, if an input expansion applies to it.
85    pub(crate) fn selector(
86        &self,
87        param: &FlatParam,
88        explicit: Option<&ExpandParamDecl>,
89    ) -> Res<Option<Selector>> {
90        let (core, optional) = match param.ty.kind() {
91            TypeKind::Optional(inner) => (&**inner, true),
92            _ => (&param.ty, false),
93        };
94        let (name, access) = match shape(core, |n| self.types.get(n)) {
95            Ok(Shape::Declared { name, access, .. }) => (name, access),
96            Ok(Shape::Undeclared(name)) => (name, Access::Owned),
97            _ => return Ok(None),
98        };
99        if access == Access::Exclusive {
100            return Ok(None);
101        }
102        let decl = match explicit {
103            Some(d) => d,
104            None => match self.param_exp.get(name) {
105                Some(d) => d,
106                None => return Ok(None),
107            },
108        };
109        if decl.variants.len() == 1 && matches!(decl.variants[0], ParamVariant::Handle) {
110            return Ok(None);
111        }
112        let Some(class) = self.class(name).cloned() else {
113            return err(format!(
114                "`{name}`: an input expansion needs a declared class"
115            ));
116        };
117        let mut variants = Vec::new();
118        for v in &decl.variants {
119            variants.push(match v {
120                ParamVariant::Build(f) => SelVariant::Build {
121                    func: f.fun.resolve(self.flat).map_err(crate::Error)?,
122                    callee: f.fun.callee(&self.q),
123                },
124                ParamVariant::Handle => SelVariant::Handle,
125            });
126        }
127        Ok(Some(Selector {
128            param: param.clone(),
129            class,
130            variants,
131            optional,
132            access,
133        }))
134    }

136    /// The Kotlin parameters of a selector: `pSel`, then each variant's.
137    pub(crate) fn selector_params(&self, s: &Selector) -> Res<Vec<(String, String)>> {
138        let p = kt_ident(&names::camel(&names::bare(&s.param.name)));
139        if let (true, [SelVariant::Build { func, .. }]) = (s.is_direct(), s.variants.as_slice()) {
140            let n = func.params.len();
141            return func
142                .params
143                .iter()
144                .enumerate()
145                .map(|(j, fp)| {
146                    let name = if n == 1 { p.clone() } else { format!("{p}{j}") };
147                    let t = self.kt_type(&fp.ty)?;
148                    Ok((name, if s.optional { nullable(t) } else { t }))
149                })
150                .collect();
151        }
152        let mut out = vec![(format!("{p}Sel"), "Int".to_string())];
153        for (i, v) in s.variants.iter().enumerate() {
154            match v {
155                SelVariant::Build { func, .. } => {
156                    let n = func.params.len();
157                    for (j, fp) in func.params.iter().enumerate() {
158                        let t = self.kt_type(&fp.ty)?;
159                        out.push((arg_name(&p, i, j, n), nullable(t)));
160                    }
161                }
162                SelVariant::Handle => out.push((format!("{p}{i}"), format!("{}?", s.class.fqn()))),
163            }
164        }
165        Ok(out)
166    }

168    /// The raw leaves of a selector.
169    pub(crate) fn selector_leaves(&self, s: &Selector) -> Res<Vec<Leaf>> {
170        let root = names::bare(&s.param.name);
171        let mut out = Vec::new();
172        if !s.is_direct() {
173            out.push(Leaf::new(LeafTy::Prim(Prim::I)).under(&format!("{root}_sel")));
174        }
175        for (i, v) in s.variants.iter().enumerate() {
176            match v {
177                SelVariant::Build { func, .. } => {
178                    for (j, fp) in func.params.iter().enumerate() {
179                        let opt = self.variant_arg_ty(s, &fp.ty);
180                        out.extend(
181                            self.leaves(&opt, Dir::In)?
182                                .into_iter()
183                                .map(|l| l.under(&format!("{root}_{i}{j}"))),
184                        );
185                    }
186                }
187                SelVariant::Handle => {
188                    out.push(Leaf::new(LeafTy::Prim(Prim::J)).under(&format!("{root}_{i}")))
189                }
190            }
191        }
192        Ok(out)
193    }

195    /// Kotlin leaf expressions for a selector, from its Kotlin parameters.
196    pub(crate) fn selector_encode(&self, s: &Selector, cx: &mut KtEnc) -> Res<Vec<String>> {
197        let p = kt_ident(&names::camel(&names::bare(&s.param.name)));
198        if s.is_direct() {
199            let names: Vec<String> = self
200                .selector_params(s)?
201                .into_iter()
202                .map(|(n, _)| n)
203                .collect();
204            let [SelVariant::Build { func, .. }] = s.variants.as_slice() else {
205                unreachable!("a direct selector");
206            };
207            let mut out = Vec::new();
208            for (fp, n) in func.params.iter().zip(&names) {
209                let ty = self.variant_arg_ty(s, &fp.ty);
210                out.extend(self.kt_encode(&ty, n, false, cx, true)?);
211            }
212            return Ok(out);
213        }
214        let mut out = vec![format!("{p}Sel")];
215        for (i, v) in s.variants.iter().enumerate() {
216            match v {
217                SelVariant::Build { func, .. } => {
218                    let n = func.params.len();
219                    for (j, fp) in func.params.iter().enumerate() {
220                        let opt = self.variant_arg_ty(s, &fp.ty);
221                        out.extend(self.kt_encode(
222                            &opt,
223                            &arg_name(&p, i, j, n),
224                            false,
225                            cx,
226                            true,
227                        )?);
228                    }
229                }
230                SelVariant::Handle => {
231                    let e = format!("{p}{i}");
232                    out.extend(self.kt_encode_handle(&s.class, &e, s.access == Access::Owned, cx));
233                }
234            }
235        }
236        Ok(out)
237    }

239    fn kt_encode_handle(
240        &self,
241        class: &Rc<Class>,
242        expr: &str,
243        consumed: bool,
244        cx: &mut KtEnc,
245    ) -> Vec<String> {
246        cx.handles.push(HandleSite {
247            expr: expr.to_string(),
248            nullable: true,
249            consumed,
250            class: class.clone(),
251        });
252        vec![format!("({expr}?.ptr ?: 0L)")]
253    }

255    /// How argument type `ty` of a variant crosses: optional, unless the
256    /// selector is direct and not optional itself.
257    fn variant_arg_ty(
258        &self,
259        s: &Selector,
260        ty: &prebindgen_tools::flat::flat::TypeRef,
261    ) -> prebindgen_tools::flat::flat::TypeRef {
262        let already = matches!(ty.kind(), TypeKind::Optional(_));
263        if already || (s.is_direct() && !s.optional) {
264            ty.clone()
265        } else {
266            ty.optional()
267        }
268    }

270    /// The Rust input of a direct selector: the constructor called on its
271    /// arguments (`None` when an optional parameter's are absent).
272    fn direct_input(&self, s: &Selector, func: &Function, callee: &TokenStream) -> Res<Input> {
273        let root = names::bare(&s.param.name);
274        let mut wires = Vec::new();
275        let mut args = Vec::new();
276        let mut binds = Vec::new();
277        for (j, fp) in func.params.iter().enumerate() {
278            let ty = self.variant_arg_ty(s, &fp.ty);
279            let input = self.rs_decode(&ty, &format!("{root}_0{j}"), 0)?;
280            wires.extend(input.wires.clone());
281            let e = input.result();
282            let a = quote::format_ident!("__a{}", j);
283            binds.push(quote!(let #a = #e?;));
284            args.push(a);
285        }
286        let call = |args: &[syn::Ident]| match func.ret.kind() {
287            TypeKind::Fallible { .. } => {
288                quote!(#callee(#(#args),*).map_err(|__e| ::std::string::ToString::to_string(&__e))?)
289            }
290            _ => quote!(#callee(#(#args),*)),
291        };
292        let built = call(&args);
293        let wrap = |b: TokenStream| match s.access {
294            Access::Owned => b,
295            _ => quote!(::prebindgen_jni_runtime::MaybeOwned::Owned(#b)),
296        };
297        let expr = if s.optional {
298            let some = wrap(built);
299            quote!({
300                #(#binds)*
301                match (#(#args,)*) {
302                    (#(::core::option::Option::Some(#args),)*) => ::core::option::Option::Some(#some),
303                    _ => ::core::option::Option::None,
304                }
305            })
306        } else {
307            let v = wrap(built);
308            quote!({ #(#binds)* #v })
309        };
310        let name = &s.param.name;
311        let input = Input::fallible(wires, expr);
312        Ok(match (s.access, s.optional) {
313            (Access::Owned, _) => input,
314            (_, false) => input.with_pass(quote!(&*#name)),
315            (_, true) => input.with_pass(quote!(#name.as_deref())),
316        })
317    }

319    /// The Rust input of a selector parameter.
320    pub(crate) fn selector_input(&self, s: &Selector) -> Res<Input> {
321        if let (true, [SelVariant::Build { func, callee }]) = (s.is_direct(), s.variants.as_slice())
322        {
323            return self.direct_input(s, func, callee);
324        }
325        let root = names::bare(&s.param.name);
326        let sel = leaf_ident(&root, "sel");
327        let t = self.q.path(&names::ident(&s.class.rust));
328        let mut wires = vec![Wire::new(sel.clone(), Prim::I.rs())];
329        let mut arms = Vec::new();
330        for (i, v) in s.variants.iter().enumerate() {
331            let body = match v {
332                SelVariant::Build { func, callee } => {
333                    let mut args = Vec::new();
334                    for (j, fp) in func.params.iter().enumerate() {
335                        let r = format!("{root}_{i}{j}");
336                        let input = self.rs_decode(&self.variant_arg_ty(s, &fp.ty), &r, 0)?;
337                        wires.extend(input.wires.clone());
338                        let e = input.result();
339                        // An optional argument's `None` is its value, not a
340                        // missing argument.
341                        if let TypeKind::Optional(_) = fp.ty.kind() {
342                            args.push(quote!(#e?));
343                            continue;
344                        }
345                        let pname = names::bare(&fp.name);
346                        let msg = format!("missing argument `{pname}` for `{root}` variant {i}");
347                        args.push(quote!(#e?.ok_or_else(|| ::std::string::String::from(#msg))?));
348                    }
349                    let built = match func.ret.kind() {
350                        TypeKind::Fallible { .. } => {
351                            quote!(#callee(#(#args),*).map_err(|__e| ::std::string::ToString::to_string(&__e))?)
352                        }
353                        _ => quote!(#callee(#(#args),*)),
354                    };
355                    match s.access {
356                        Access::Owned => built,
357                        _ => quote!(::prebindgen_jni_runtime::MaybeOwned::Owned(#built)),
358                    }
359                }
360                SelVariant::Handle => {
361                    let w = leaf_ident(&root, &i.to_string());
362                    wires.push(Wire::new(w.clone(), Prim::J.rs()));
363                    match s.access {
364                        Access::Owned => quote!(::prebindgen_jni_runtime::take_handle::<#t>(#w)?),
365                        _ => quote!(::prebindgen_jni_runtime::MaybeOwned::Borrowed(
366                            ::prebindgen_jni_runtime::borrow_handle::<#t>(#w)?
367                        )),
368                    }
369                }
370            };
371            let i = i as i32;
372            arms.push(if s.optional {
373                quote!(#i => ::core::option::Option::Some(#body))
374            } else {
375                quote!(#i => #body)
376            });
377        }
378        if s.optional {
379            arms.push(quote!(-1 => ::core::option::Option::None));
380        }
381        let msg = format!("invalid selector {{}} for `{root}`");
382        let name = &s.param.name;
383        let expr = quote!(match #sel {
384            #(#arms,)*
385            __s => return ::core::result::Result::Err(::std::format!(#msg, __s)),
386        });
387        let input = Input::fallible(wires, expr);
388        Ok(match (s.access, s.optional) {
389            (Access::Owned, _) => input,
390            (_, false) => input.with_pass(quote!(&*#name)),
391            (_, true) => input.with_pass(quote!(#name.as_deref())),
392        })
393    }

395    /// The typed overloads of a selector-form function, one per combination
396    /// of the split parameters' variants.
397    pub(crate) fn split_overloads(
398        &self,
399        head: &str,
400        name: &str,
401        params: &[SigParam],
402        tail: &[(String, String)],
403        ret: &str,
404        prefix_names: bool,
405    ) -> Res<Vec<String>> {
406        let mut combos: Vec<Vec<(usize, usize)>> = vec![Vec::new()];
407        for (pi, p) in params.iter().enumerate() {
408            if let SigParam::Split(s) = p {
409                let mut next = Vec::new();
410                for c in &combos {
411                    for vi in 0..s.variants.len() {
412                        let mut c2 = c.clone();
413                        c2.push((pi, vi));
414                        next.push(c2);
415                    }
416                }
417                combos = next;
418            }
419        }
420        let mut out = Vec::new();
421        for combo in combos {
422            let mut sig = Vec::new();
423            let mut args = Vec::new();
424            for (pi, p) in params.iter().enumerate() {
425                match p {
426                    SigParam::Plain(n, t) => {
427                        sig.push(format!("{n}: {t}"));
428                        args.push(n.clone());
429                    }
430                    SigParam::Split(s) => {
431                        let vi = combo.iter().find(|(i, _)| *i == pi).unwrap().1;
432                        let base = kt_ident(&names::camel(&names::bare(&s.param.name)));
433                        args.push(vi.to_string());
434                        for (i, v) in s.variants.iter().enumerate() {
435                            match v {
436                                SelVariant::Build { func, .. } => {
437                                    for fp in &func.params {
438                                        if i == vi {
439                                            let pn = names::bare(&fp.name);
440                                            let n = if prefix_names {
441                                                kt_ident(&names::camel(&format!(
442                                                    "{}_{pn}",
443                                                    names::bare(&s.param.name)
444                                                )))
445                                            } else {
446                                                kt_ident(&names::camel(&pn))
447                                            };
448                                            sig.push(format!("{n}: {}", self.kt_type(&fp.ty)?));
449                                            args.push(n);
450                                        } else {
451                                            args.push("null".to_string());
452                                        }
453                                    }
454                                }
455                                SelVariant::Handle => {
456                                    if i == vi {
457                                        sig.push(format!("{base}: {}", s.class.fqn()));
458                                        args.push(base.clone());
459                                    } else {
460                                        args.push("null".to_string());
461                                    }
462                                }
463                            }
464                        }
465                    }
466                    SigParam::Selector(s) => {
467                        for (n, t) in self.selector_params(s)? {
468                            sig.push(format!("{n}: {t}"));
469                            args.push(n);
470                        }
471                    }
472                }
473            }
474            for (n, t) in tail {
475                sig.push(format!("{n}: {t}"));
476                args.push(n.clone());
477            }
478            out.push(format!(
479                "{head}{name}({}): {ret} =\n    {name}({})\n",
480                sig.join(", "),
481                args.join(", ")
482            ));
483        }
484        Ok(out)
485    }
486}

488/// A Kotlin wrapper parameter: plain, a selector group, or a selector group
489/// that is split into overloads.
490pub(crate) enum SigParam {
491    Plain(String, String),
492    Selector(Selector),
493    Split(Selector),
494}
495
496fn nullable(t: String) -> String {
497    if t.ends_with('?') {
498        t
499    } else {
500        format!("{t}?")
501    }
