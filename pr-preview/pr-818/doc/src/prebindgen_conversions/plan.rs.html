1use crate::template_parameter;
2use prebindgen_flat::{WireKind, WireLayout};
3use proc_macro2::TokenStream;
4use quote::quote;
5/// An input conversion split into call-wide validation, acquisition and construction.
6///
7/// Optional storage for the staged request protocol in this crate.
8/// Generators may use this protocol or implement their own request types.
9#[derive(Clone, Debug)]
10pub struct InputPlan<W: WireKind, C = (), G = ()> {
11    /// Context type supplied by the language generator at emission.
12    pub context: std::marker::PhantomData<fn(&C)>,
13    /// Authoritative wire tree; its leaves determine boundary parameter order.
14    pub layout: WireLayout<W, G>,
15    /// Shared call acknowledgement, if this input needs one.
16    pub acknowledgement: Option<TokenStream>,
17    /// Validate and prepare every active leaf before any ownership transfer.
18    pub validation: TokenStream,
19    /// Infallible acquisition of all owned inputs, before constructors can run.
20    pub acquire: TokenStream,
21    /// Active foreign resources whose identities must be checked across all arguments.
22    pub resources: Vec<Resource>,
23    /// Argument construction after all acquisitions, including source constructor calls.
24    pub statements: TokenStream,
25    /// Prepared value expression assigned to the caller-supplied target after `statements`.
26    pub argument: TokenStream,
27}
28/// Output conversion whose owned temporaries survive until successful delivery.
29///
30/// Optional concrete storage implementing [`prebindgen_flat::Conversion`].
31/// Delivery and ownership commit are separate requests.
32#[derive(Clone, Debug)]
33pub struct OutputPlan<W: WireKind, C = (), G = ()> {
34    /// Context type supplied by the language generator at emission.
35    pub context: std::marker::PhantomData<fn(&C)>,
36    /// Delivery and return code prepared with this result slot.
37    pub delivery: Delivery,
38    /// Authoritative result tree; each outer wire (including a sequence carrier)
39    /// corresponds to one entry in `values`. Element fields have separate positions.
40    pub layout: WireLayout<W, G>,
41    /// Transfer ownership only after all foreign delivery operations succeed.
42    pub commit: TokenStream,
43    /// Code producing owned output temporaries and evaluating observations before moves.
44    pub statements: TokenStream,
45    /// Delivery expressions in wire order; they must remain valid until commit.
46    pub values: Vec<TokenStream>,
47}
48/// Access to one foreign resource during an exported call.
49#[derive(Clone, Copy, Debug, PartialEq, Eq)]
50pub enum Access {
51    /// Read-only borrow; multiple active shared accesses may alias.
52    Shared,
53    /// Mutable borrow; conflicts with every other active use of the resource.
54    Exclusive,
55    /// Ownership transfer; conflicts with every other active use of the resource.
56    Move,
57}
58/// One allocation address, or a runtime-sized set of addresses with the same access mode.
59#[derive(Clone, Debug)]
60pub enum ResourceIdentity {
61    /// Expression evaluating to one allocation address.
62    Single(TokenStream),
63    /// Expression implementing IntoIterator<Item = usize>.
64    Many(TokenStream),
65}
66/// The adapter supplies identity; flat enforces call-wide access conflicts.
67#[derive(Clone, Debug)]
68pub struct Resource {
69    /// Side-effect-free Rust expression identifying a live resource, such as an address.
70    /// All identities compared within a call must share an equality-compatible domain.
71    pub identity: ResourceIdentity,
72    /// Borrow or ownership requirement imposed by this input conversion.
73    pub access: Access,
74    /// Side-effect-free Rust boolean expression; false resources do not conflict.
75    pub active: TokenStream,
76}
77/// Result-slot glue executed after output preparation and before ownership commit.
78#[derive(Clone, Debug)]
79pub struct Delivery {
80    /// Deliver every output; failure must leave owned temporaries uncommitted.
81    pub statements: TokenStream,
82    /// Infallible wrapper result expression evaluated after commit.
83    pub result: TokenStream,
84}
85impl Default for Delivery {
86    fn default() -> Self {
87        Self {
88            statements: TokenStream::new(),
89            result: quote!(()),
90        }
91    }
92}
