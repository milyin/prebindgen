9pub(crate) fn input(
10    wire: syn::Ident,
11    length: syn::Ident,
12    index: syn::Ident,
13    element: &WireLayout<JniWire, JniGroup>,
14    active: &Condition,
15) -> Result<SequenceInput<JniWire>> {
16    let env = prebindgen_conversions::template_parameter("environment");
17    let row = format_ident!("{wire}_row");
18    let count = element.len();
19    let capacity = i32::try_from(count + 8).map_err(|_| "too many collection element fields")?;
20    let frame = format_ident!("{wire}_frame");
21    let mut bind = quote! {let #frame=unsafe {::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;let #row=::prebindgen_jni_runtime::row(&mut #env,&#wire,#index,#count)?;};
22    let mut validation_values = Vec::new();
23    let mut prepared_values = Vec::new();
24    for (position, field) in element.iter().enumerate() {
25        let name = &field.name;
26        let (reader, reference, boolean) = match &field.kind {
27            JniWire::Scalar { primitive, .. } => match primitive {
28                Primitive::Bool => ("row_bool", false, true),
29                Primitive::I8 => ("row_i8", false, false),
30                Primitive::I16 => ("row_i16", false, false),
31                Primitive::U8 | Primitive::U16 | Primitive::I32 | Primitive::Char => {
32                    ("row_i32", false, false)
33                }
34                Primitive::U32
35                | Primitive::U64
36                | Primitive::I64
37                | Primitive::Usize
38                | Primitive::Isize => ("row_i64", false, false),
39                Primitive::F32 => ("row_f32", false, false),
40                Primitive::F64 => ("row_f64", false, false),
41                Primitive::I128 | Primitive::U128 => ("row_bytes", true, false),
42            },
43            JniWire::Text { .. } => ("row_text", true, false),
44            JniWire::Bytes { .. } => ("row_bytes", true, false),
45            JniWire::Rows { .. } => ("row_rows", true, false),
46            JniWire::Handle { .. } => ("row_i64", false, false),
47            JniWire::Enum { .. } | JniWire::Selector { .. } => ("row_i32", false, false),
48        };
49        let reader = format_ident!("{reader}");
50        let convert = if boolean {
51            quote!(.map(u8::from))
52        } else {
53            quote!()
54        };
55        bind.extend(quote! {
56            let #name=::prebindgen_jni_runtime::#reader(&mut #env,&#row,#position)#convert;
57            if #name.is_err() && #env.exception_check().map_err(|e|e.to_string())? {return Err("JNI collection field decoding failed".into());}
58        });
59        let borrow = quote!(*);
60        validation_values.push(quote!(#borrow #name.as_ref().map_err(|e|e.clone())?));
61        prepared_values.push(if reference {
62            None
63        } else {
64            Some(quote!(#borrow #name.as_ref().expect("validated collection field")))
65        });
66    }
67    Ok(SequenceInput {
68        carrier: JniWire::Rows {
69            condition: kotlin_condition(active),
70            null_niche_available: true,
71        },
72        open: quote! {if #wire.is_null(){return Err("null collection".into());}let #length=#env.get_array_length(&#wire).map_err(|e|e.to_string())? as usize;},
73        bind,
74        validation_values,
75        prepared_values,
76        release: quote!(drop(#frame);),
77    })
78}
79pub(crate) fn output(
80    wire: syn::Ident,
81    length: syn::Ident,
82    index: syn::Ident,
83    element: &WireLayout<JniWire, JniGroup>,
84    values: &[TokenStream],
85) -> Result<SequenceOutput<JniWire>> {
86    let env = prebindgen_conversions::template_parameter("environment");
87    let row = format_ident!("{wire}_row");
88    let count = element.len();
89    let capacity = i32::try_from(count + 8).map_err(|_| "too many collection fields")?;
90    let encode = encode_array(element, values, &row, 0)?;
91    let frame = format_ident!("{wire}_frame");
92    Ok(SequenceOutput {
93        commit_each: false,
94        fallible: true,
95        carrier: JniWire::Rows {
96            condition: "true".into(),
97            null_niche_available: true,
98        },
99        begin: quote!(let #wire=::prebindgen_jni_runtime::array(&mut #env,#length)?;),
100        enter: quote!(let #frame=unsafe {::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;),
101        store: quote! {
102            let #row=::prebindgen_jni_runtime::array(&mut #env,#count)?;#encode
103            #env.set_object_array_element(&#wire,#index as i32,&#row).map_err(|e|e.to_string())?;
104            drop(#frame);
105        },
106        finish: quote!(let #wire=Some(#wire);),
107        value: quote!(#wire),
108        commit: quote!(),
109    })
110}

112pub(crate) fn input_list(
113    carrier: &Wire<JniWire>,
114    element: &WireLayout<JniWire, JniGroup>,
115    context: &KotlinContext<'_>,
116) -> Result<KotlinInput> {
117    let child = input_kotlin(element.nodes(), context)?;
118    if child.parameters.len() != 1 {
119        return Err("Kotlin collection elements must expose exactly one value; declare a data class for expanded fields".into());
120    }
121    let (parameter, ty) = child.parameters[0]
122        .split_once(": ")
123        .ok_or("invalid element parameter")?;
124    let name = carrier.name.to_string();
125    let state = format!("{name}_elements");
126    let JniWire::Rows { condition, .. } = &carrier.kind else {
127        return Err("sequence node requires an array carrier".into());
128    };
129    let mut code = String::new();
130    for binding in &child.setup {
131        code.push_str(&format!("val {} = {}\n", binding.name, binding.expression));
132    }
133    let mut locks = if child.locks.is_empty() {
134        "kotlin.collections.emptyList<Nothing>()".into()
135    } else {
136        format!(
137            "kotlin.collections.listOfNotNull({})",
138            child.locks.join(", ")
139        )
140    };
141    for group in &child.lock_groups {
142        locks = format!("({locks} + ({group}))");
143    }
144    let expression=format!("if ({condition}) {name}?.toList()?.map {{ {parameter} ->\n{code}kotlin.Triple({{ kotlin.arrayOf<kotlin.Any?>({}) }}, {{ {locks} }}, {{ {}; kotlin.Unit }})\n}} else null",child.arguments.join(", "),child.consumed.join("; "));
145    Ok(KotlinInput {
146        parameters: vec![format!(
147            "{name}: kotlin.collections.List<{ty}>{}",
148            if condition == "true" { "" } else { "?" }
149        )],
150        setup: vec![KotlinBinding {
151            name: state.clone(),
152            expression,
153            absent: "null".into(),
154        }],
155        arguments: vec![format!(
156            "{state}?.map {{ it.first() }}?.toTypedArray<kotlin.Any?>()"
157        )],
158        locks: Vec::new(),
159        lock_groups: if child.locks.is_empty() && child.lock_groups.is_empty() {
160            vec![]
161        } else {
162            vec![format!("{state}.orEmpty().flatMap {{ it.second() }}")]
163        },
164        consumed: if child.consumed.is_empty() {
165            vec![]
166        } else {
167            vec![format!("{state}.orEmpty().forEach {{ it.third() }}")]
168        },
169    })
170}

172pub(crate) fn output_list(
173    element: &WireLayout<JniWire, JniGroup>,
174    context: &KotlinContext<'_>,
175) -> Result<KotlinOutput> {
176    let row = format!("{}_row", context.output_prefix);
177    let prefix = format!("{}item_", context.output_prefix);
178    let failure = format!("{}failure", context.output_prefix);
179    let ctx = KotlinContext {
180        failure: &failure,
181        cleanup_failure: None,
182        result: &row,
183        output_prefix: &prefix,
184        ..*context
185    };
186    let child = output_kotlin(element.nodes(), &ctx)?;
187    if child.delivered.len() != 1 {
188        return Err("Kotlin collection elements must deliver exactly one value; declare a data class for expanded outputs".into());
189    }
190    let ty = &child.result_types[0];
191    let value = &child.delivered[0];
192    let raw = format!("({}[0] as kotlin.Array<*>)", context.result);
193    let output = format!("{}list", context.output_prefix);
194    let actions = format!("{}cleanups", context.output_prefix);
195    let (failure_value, finish_cleanup) = match context.cleanup_failure {
196        Some(target) => (
197            format!("{target} ?: kotlin.RuntimeException(\"collection cleanup failed\")"),
198            format!("if ({target} == null && {failure}.suppressed.isNotEmpty()) {{ {target} = {failure}.suppressed.first(); {failure}.suppressed.drop(1).forEach {{ {target}!!.addSuppressed(it) }} }}"),
199        ),
200        None => (context.failure.to_owned(), String::new()),
201    };
202    Ok(KotlinOutput {
203        delivered:vec![format!("requireNotNull({output})")],result_types:vec![format!("kotlin.collections.List<{ty}>")],
204        output_locals:format!("var {output}: kotlin.collections.MutableList<{ty}>? = null\nvar {actions}: kotlin.collections.MutableList<(kotlin.Throwable) -> kotlin.Unit>? = null\n"),
205        materialize:format!("{actions} = kotlin.collections.mutableListOf()\n{output} = kotlin.collections.mutableListOf()\nfor (rawRow in {raw}) {{\nval {row} = rawRow as kotlin.Array<*>\n{}\n{actions}.add {{ {failure} -> {} }}\n{}\n{output}.add({value})\n}}\n",child.output_locals,child.cleanup,child.materialize),
206        cleanup:format!("val {failure} = {failure_value}\n{actions}?.forEach {{ it({failure}) }}\nfor (index in ({actions}?.size ?: 0) until {raw}.size) {{\nval {row} = {raw}[index] as kotlin.Array<*>\n{}\n{}\n}}\n{finish_cleanup}\n",child.output_locals,child.cleanup),
207    })
