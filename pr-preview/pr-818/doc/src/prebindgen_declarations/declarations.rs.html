9/// An ordered list of input representations for one source type.
10///
11/// One variant becomes a direct recipe; multiple variants become a selector.
12/// Constructor arguments recursively use the registered type defaults.
13#[derive(Clone)]
14pub struct ExpandParamDecl {
15    /// Source type to which this declaration applies, resolved against the flat model.
16    pub ty: syn::Type,
17    variants: Vec<Option<FunctionDecl>>,
18}
19impl ExpandParamDecl {
20    /// Start an input expansion with no variants. Add at least one before use.
21    pub fn new(ty: syn::Type) -> Self {
22        Self {
23            ty,
24            variants: Vec::new(),
25        }
26    }
27    /// Append a source constructor as an input alternative.
28    ///
29    /// Only the constructor name is supported here; renaming and per-function
30    /// overrides on this reference are rejected when resolving declarations.
31    pub fn variant(mut self, f: FunctionDecl) -> Self {
32        self.variants.push(Some(f));
33        self
34    }
35    /// Append the type itself as a leaf, converted by the adapter's codec.
36    ///
37    /// This bypasses recursive expansion of the same type.
38    pub fn variant_self(mut self) -> Self {
39        self.variants.push(None);
40        self
41    }
42}
43/// An ordered list of simultaneous output views for one source type.
44///
45/// One view becomes a direct recipe; multiple views become [`Output::Combine`].
46#[derive(Clone)]
47pub struct ExpandReturnDecl {
48    /// Source type whose result occurrences use this expansion.
49    pub ty: syn::Type,
50    fields: Vec<ReturnField>,
51}
52#[derive(Clone)]
53enum ReturnField {
54    Identity,
55    Projected(Projection),
56    IntoFields(FieldsDecl),
57    Getter(Box<FunctionDecl>),
58    Fields(FieldsDecl),
59}
60/// Project the named fields of a source getter's returned record.
61///
62/// The getter must accept a shared reference to the converted type. Fields use
63/// their type defaults unless overridden with [`Self::field`].
64#[derive(Clone)]
65pub struct FieldsDecl {
66    function: String,
67    overrides: BTreeMap<String, ExpandReturnDecl>,
68}
69impl FieldsDecl {
70    /// Select the source getter whose returned record will be projected.
71    ///
72    /// `function` is the getter's unqualified name.
73    pub fn new(function: syn::Ident) -> Self {
74        Self {
75            function: function.to_string(),
76            overrides: BTreeMap::new(),
77        }
78    }
79    /// Replace the output expansion for one named field of the returned record.
80    ///
81    /// # Arguments
82    ///
83    /// * `name` — the field's name in the source record.
84    /// * `decl` — how to deliver that field instead of its type's default.
85    ///
86    /// # Panics
87    /// Panics if the same field is overridden twice. Unknown fields and mismatched
88    /// source types are rejected during declaration resolution.
89    pub fn field(mut self, name: impl AsRef<str>, decl: ExpandReturnDecl) -> Self {
90        assert!(
91            self.overrides.insert(name.as_ref().into(), decl).is_none(),
92            "duplicate field override"
93        );
94        self
95    }
96}
97impl ExpandReturnDecl {
98    /// Append the fields of a getter's returned record as output views.
99    pub fn fields(mut self, decl: FieldsDecl) -> Self {
100        self.fields.push(ReturnField::Fields(decl));
101        self
102    }
103    /// Consume the source accessor's receiver, then move its returned record fields.
104    /// The accessor must take `T`, not `&T`. Per-field expansions still override
105    /// type defaults. Nested data classes retain their borrowed default unless an
106    /// explicit field expansion or type default selects owned projection.
107    /// Observing sibling views run before this consuming view.
108    pub fn fields_into(mut self, decl: FieldsDecl) -> Self {
109        self.fields.push(ReturnField::IntoFields(decl));
110        self
111    }
112    /// Deprecated alias for [`Self::fields_into`].
113    #[deprecated(note = "use fields_into instead")]
114    pub fn fields_self_into(self, decl: FieldsDecl) -> Self {
115        self.fields_into(decl)
116    }
117    /// Append a leaf converted with explicit ownership and optional field policies.
118    pub fn project_self(mut self, projection: Projection) -> Self {
119        self.fields.push(ReturnField::Projected(projection));
120        self
121    }
122    /// Append a shared-reference view; record fields inherit this policy.
123    pub fn field_self_ref(self) -> Self {
124        self.project_self(Projection::borrowed())
125    }
126    /// Append an owned view; record fields inherit this policy.
127    pub fn field_self_into(self) -> Self {
128        self.project_self(Projection::owned())
129    }
130    /// Start an output expansion with no views. Add at least one before use.
131    pub fn new(ty: syn::Type) -> Self {
132        Self {
133            ty,
134            fields: Vec::new(),
135        }
136    }
137    /// Append a shared-borrow getter, recursively expanding its owned result.
138    ///
139    /// The reference supports only the source name, without renaming or overrides.
140    pub fn field(mut self, f: FunctionDecl) -> Self {
141        self.fields.push(ReturnField::Getter(Box::new(f)));
142        self
143    }
144    /// Append the value itself as a leaf, converted by the adapter's codec.
145    ///
146    /// Ownership follows the adapter default: JniGen data classes borrow, while
147    /// owning leaf conversions consume after observing other views.
148    pub fn field_self(mut self) -> Self {
149        self.fields.push(ReturnField::Identity);
150        self
151    }
152}
153/// Explicit default error-message mapper for one source error type.
154pub struct ErrorDecl {
155    /// Source error type, resolved against the flat model.
156    pub ty: syn::Type,
157    /// Source helper taking E or &E and returning String.
158    pub mapper: FunctionDecl,
159}
160impl ErrorDecl {
161    /// Declare a type-wide mapper; per-function `on_error` takes precedence.
162    pub fn new(ty: syn::Type, mapper: FunctionDecl) -> Self {
163        Self { ty, mapper }
164    }
165}
166impl From<ErrorDecl> for ExpandDecl {
167    fn from(value: ErrorDecl) -> Self {
168        Self::Error(Box::new(value))
169    }
170}
171/// A generator-wide conversion or error-mapping default, registered by source type.
172pub enum ExpandDecl {
173    /// Default error-message mapper for Result returns with this error type.
174    Error(Box<ErrorDecl>),
175    /// Default input alternatives.
176    Param(ExpandParamDecl),
177    /// Default simultaneous output views.
178    Return(ExpandReturnDecl),
179}
180impl From<ExpandParamDecl> for ExpandDecl {
181    fn from(d: ExpandParamDecl) -> Self {
182        Self::Param(d)
183    }
184}
185impl From<ExpandReturnDecl> for ExpandDecl {
186    fn from(d: ExpandReturnDecl) -> Self {
187        Self::Return(d)
188    }
189}

203/// Resolved type defaults used to build boundary converters.
204///
205/// Per-function overrides replace type defaults. Existing converters are preserved
206/// unless explicitly overridden. By default unit has no wires and every other
207/// type is a leaf, converted by the adapter's [`Codec`]. Recursive expansion
208/// cycles are rejected. Recipes are local to [`Self::fill`]; only compiled conversion
209/// data is attached to Flat. See [`Codec`] for a buildable example of constructor
210/// alternatives, combined outputs and manual overrides.
211pub struct Declarations {
212    inputs: BTreeMap<String, ExpandParamDecl>,
213    outputs: BTreeMap<String, ExpandReturnDecl>,
214    errors: BTreeMap<String, FunctionDecl>,
215}
216impl Declarations {
217    /// Index expansions by resolved source type.
218    ///
219    /// Returns an error for an unresolved type or duplicate declarations for the
220    /// same type and direction. Recipes are resolved later by [`Self::fill`].
221    ///
222    /// # Arguments
223    ///
224    /// * `flat` — the model in which each declaration's type is resolved.
225    /// * `decls` — type-wide defaults, at most one per type and direction.
226    pub fn new<I: prebindgen_flat::Conversion, O: prebindgen_flat::Conversion>(
227        flat: &Flat<I, O>,
228        decls: Vec<ExpandDecl>,
229    ) -> Result<Self> {
230        let mut out = Self {
231            inputs: BTreeMap::new(),
232            outputs: BTreeMap::new(),
233            errors: BTreeMap::new(),
234        };
235        for d in decls {
236            match d {
237                ExpandDecl::Error(d) => {
238                    let key = type_key(&flat.resolve_type(&d.ty)?);
239                    let f = &d.mapper;
240                    if !f.params.is_empty()
241                        || f.ret.is_some()
242                        || f.foreign.is_some()
243                        || f.error.is_some()
244                        || f.callback_return.is_some()
245                    {
246                        return Err("error mapper references only support their source name".into());
247                    }
248                    if out.errors.insert(key, d.mapper).is_some() {
249                        return Err("duplicate error mapper for source type".into());
250                    }
251                }
252                ExpandDecl::Param(d) => {
253                    if crate::contains_result(&d.ty) {
254                        return Err("Result expansions are unsupported; configure the success type and an error mapper".into());
255                    }
256                    let key = type_key(&flat.resolve_type(&d.ty)?);
257                    if out.inputs.insert(key, d).is_some() {
258                        return Err("duplicate input expansion".into());
259                    }
260                }
261                ExpandDecl::Return(d) => {
262                    if crate::contains_result(&d.ty) {
263                        return Err("Result expansions are unsupported; configure the success type and an error mapper".into());
264                    }
265                    let key = type_key(&flat.resolve_type(&d.ty)?);
266                    if out.outputs.insert(key, d).is_some() {
267                        return Err("duplicate output expansion".into());
268                    }
269                }
270            }
271        }
272        Ok(out)
273    }
274    /// Clone a declared sequence function for an additional delivery protocol.
275    ///
276    /// The original path, source types and input slots are preserved. The new
277    /// return slot is empty, so the adapter can fill it using its callback
278    /// transport while retaining the same expansion declarations. The returned
279    /// identifier selects the outer sequence carrier, never a nested collection.
280    pub fn callback_variant<I, O>(
281        &self,
282        flat: &mut Flat<I, O>,
283        decl: &FunctionDecl,
284        name: &str,
285    ) -> Result<(FunctionDecl, syn::Ident)>
286    where
287        I: prebindgen_flat::Conversion + Clone,
288        O: prebindgen_flat::Conversion + Clone,
289    {
290        let source = flat.function(&decl.name)?;
291        let (carrier, _) = crate::sequence_return(
292            source
293                .ret
294                .slot
295                .as_ref()
296                .ok_or("unbound sequence return")?
297                .layout(),
298        )
299        .ok_or("callback_return requires a sequence output recipe")?;
300        let layout = source.ret.slot.as_ref().unwrap().layout();
301        fn count<W: prebindgen_flat::WireKind, G>(
302            layout: &prebindgen_flat::WireLayout<W, G>,
303        ) -> usize {
304            layout
305                .nodes()
306                .iter()
307                .map(|node| match node {
308                    prebindgen_flat::WireNode::Sequence { .. } => 1,
309                    prebindgen_flat::WireNode::Group { children, .. } => count(children),
310                    _ => 0,
311                })
312                .sum()
313        }
314        if count(layout) != 1 || layout.len() > 2 {
315            return Err(
316                "callback_return permits only one sequence and its optional presence flag".into(),
317            );
318        }
319        let carrier = carrier.name.clone();
320        let mut function = source.clone();
321        function.name = crate::ident(name)?;
322        function.ret.slot = None;
323        if flat.functions.contains_key(name) {
324            return Err("callback return function name collision".into());
325        }
326        flat.functions.insert(name.into(), function);
327        let mut variant = decl.clone();
328        variant.name = name.into();
329        variant.foreign = None;
330        variant.callback_return = None;
331        Ok((variant, carrier))
332    }
333    /// Build one function's input and output converters from its declarations.
334    ///
335    /// Existing converters are retained unless explicitly overridden. `extra` is
336    /// used for every converter built by this call; retained converters are untouched.
337    /// For different data at individual occurrences, use [`crate::bind_input`] and
338    /// [`crate::bind_output`] before filling defaults, or after applying overrides.
339    /// No construction data is retained for later overrides.
340    ///
341    /// Resolves and compiles all required conversions before modifying any slot.
342    /// An unknown parameter, recursive expansion, incompatible type or invalid
343    /// constructor/getter leaves the entire function unchanged.
344    ///
345    /// # Arguments
346    ///
347    /// * `flat` — the model holding the function; its slots receive the converters.
348    /// * `codec` — the adapter's leaf conversions.
349    /// * `decl` — the function to export, with its per-parameter and result overrides.
350    /// * `extra` — occurrence data passed to every codec call made by this fill.
351    pub fn fill<C: Codec>(
352        &self,
353        flat: &mut prebindgen_flat::Flat<C::Input, C::Output>,
354        codec: &C,
355        decl: &FunctionDecl,
356        extra: &C::Extra,
357    ) -> Result<()> {
358        let source = flat.function(&decl.name)?;
359        if decl.callback_return.is_some()
360            && source.ret.slot.is_some()
361            && decl.ret.is_none()
362            && decl.error.is_none()
363        {
364            return Err("callback_return cannot reconstruct a prebound output slot; supply an explicit expand_return override".into());
365        }
366
367        for name in decl.params.keys() {
368            if !source.params.iter().any(|p| p.name == name.as_str()) {
369                return Err(format!("`{}`: unknown parameter `{name}`", decl.name));
370            }
371        }
372        let mut inputs = Vec::new();
373        for (index, p) in source.params.iter().enumerate() {
374            let explicit = decl.params.get(&p.name.to_string());
375            inputs.push(if explicit.is_some() || p.ty.slot.is_none() {
376                let recipe = self.input(flat, &p.ty.ty, explicit, &mut Vec::new())?;
377                Some(Compiler::new(flat, codec, format!("p{index}")).input(
378                    &p.ty.ty,
379                    &recipe,
380                    extra,
381                    &Condition::default(),
382                )?)
383            } else {
384                None
385            });
386        }
387        if decl.error.is_some() && result_parts(&source.ret.ty).is_none() {
388            return Err("on_error requires a top-level Result<T, E> return".into());
389        }
390        let mut output = if decl.ret.is_some() || decl.error.is_some() || source.ret.slot.is_none()
391        {
392            let recipe = if let Some((ok, error_ty)) = result_parts(&source.ret.ty) {
393                let mapper = self.error_mapper_for(decl, error_ty);
394                if mapper.is_some_and(|mapper| {
395                    !mapper.params.is_empty()
396                        || mapper.ret.is_some()
397                        || mapper.foreign.is_some()
398                        || mapper.error.is_some()
399                        || mapper.callback_return.is_some()
400                }) {
401                    return Err("error mapper references only support their source name".into());
402                }
403                Output::Result {
404                    ok: Box::new(self.output(flat, ok, decl.ret.as_ref(), &mut Vec::new())?),
405                    error: mapper.map(|mapper| mapper.name.clone()),
406                }
407            } else {
408                self.output(flat, &source.ret.ty, decl.ret.as_ref(), &mut Vec::new())?
409            };
410            Some(Compiler::new(flat, codec, "ret".into()).output_root(
411                &source.ret.ty,
412                &recipe,
413                extra,
414                prebindgen_conversions::source_result_expr(),
415                false,
416            )?)
417        } else {
418            None
419        };
420        for input in inputs.iter_mut().flatten() {
421            codec.finish_input(input, extra)?;
422            input.layout.validate()?;
423        }
424        if let Some(output) = &mut output {
425            codec.finish_output(output, extra)?;
426            output.validate_value_count()?;
427            output.layout.validate()?;
428        }
429        // Install slots only after every conversion has compiled successfully.
430        let dest = flat.function_mut(&decl.name)?;
431        for (p, input) in dest.params.iter_mut().zip(inputs) {
432            if let Some(input) = input {
433                p.ty.slot = Some(input.into());
434            }
435        }
436        if let Some(output) = output {
437            dest.ret.slot = Some(output.into());
438        }
439        Ok(())
440    }
441    fn input<I: prebindgen_flat::Conversion, O: prebindgen_flat::Conversion>(
442        &self,
443        flat: &Flat<I, O>,
444        ty: &syn::Type,
445        explicit: Option<&ExpandParamDecl>,
446        stack: &mut Vec<String>,
447    ) -> Result<Input> {
448        let key = type_key(ty);
449        if let (Some(inner), Some(decl)) = (option_inner(ty), explicit) {
450            if type_key(&flat.resolve_type(&decl.ty)?) != key {
451                return Ok(Input::Optional(Box::new(
452                    self.input(flat, inner, explicit, stack)?,
453                )));
454            }
455        }
456        if let (Some((element, _)), Some(decl)) = (sequence_parts(ty), explicit) {
457            if type_key(&flat.resolve_type(&decl.ty)?) != key {
458                return Ok(Input::Sequence(Box::new(
459                    self.input(flat, element, explicit, stack)?,
460                )));
461            }
462        }
463        let d = explicit.or_else(|| self.inputs.get(&key));
464        if let Some(d) = d {
465            if type_key(&flat.resolve_type(&d.ty)?) != key {
466                return Err("input expansion type does not match occurrence".into());
467            }
468            if stack.contains(&key) {
469                return Err(format!("recursive input expansion `{key}`"));
470            }
471            stack.push(key);
472            let mut arms = Vec::new();
473            for variant in &d.variants {
474                arms.push(match variant {
475                    None => if let Some((element,_))=sequence_parts(ty) {
476                        Input::Sequence(Box::new(self.input(flat,element,None,stack)?))
477                    } else { Input::Leaf },
478                    Some(f) => {
479                        if !f.params.is_empty() || f.ret.is_some() || f.foreign.is_some() || f.error.is_some() || f.callback_return.is_some() {
480                            return Err("constructor references only support their source name in this subset".into());
481                        }
482                        let func = flat.function(&f.name)?;
483                        let args = func.params.iter()
484                            .map(|p| self.input(flat, &p.ty.ty, None, stack))
485                            .collect::<Result<Vec<_>>>()?;
486                        Input::constructor(&f.name, args)
487                    }
488                });
489            }
490            stack.pop();
491            return match arms.len() {
492                0 => Err("empty input expansion".into()),
493                1 => Ok(arms.remove(0)),
494                _ => Ok(Input::selector(arms)),
495            };
496        }
497        if let Some(inner) = option_inner(ty) {
498            return Ok(Input::Optional(Box::new(
499                self.input(flat, inner, None, stack)?,
500            )));
501        }
502        if let Some((element, _)) = sequence_parts(ty) {
503            return Ok(Input::Sequence(Box::new(
504                self.input(flat, element, None, stack)?,
505            )));
506        }
507        let key = match ty {
508            syn::Type::Reference(r) => type_key(&r.elem),
509            _ => key,
510        };
511        Ok(if key == "()" {
512            Input::fields([])
513        } else {
514            Input::Leaf
515        })
516    }

553    fn output<I: prebindgen_flat::Conversion, O: prebindgen_flat::Conversion>(
554        &self,
555        flat: &Flat<I, O>,
556        ty: &syn::Type,
557        explicit: Option<&ExpandReturnDecl>,
558        stack: &mut Vec<String>,
559    ) -> Result<Output> {
560        let key = type_key(ty);
561        if let (Some(inner), Some(decl)) = (option_inner(ty), explicit) {
562            if type_key(&flat.resolve_type(&decl.ty)?) != key {
563                return Ok(Output::Optional(Box::new(
564                    self.output(flat, inner, explicit, stack)?,
565                )));
566            }
567        }
568        if let (Some((element, _view)), Some(decl)) = (sequence_parts(ty), explicit) {
569            if type_key(&flat.resolve_type(&decl.ty)?) != key {
570                return Ok(Output::Sequence {
571                    element: Box::new(self.output(flat, element, explicit, stack)?),
572                    mode: None,
573                });
574            }
575        }
576        let d = explicit.or_else(|| self.outputs.get(&key));
577        if let Some(d) = d {
578            if type_key(&flat.resolve_type(&d.ty)?) != key {
579                return Err("output expansion type does not match occurrence".into());
580            }
581            if stack.contains(&key) {
582                return Err(format!("recursive output expansion `{key}`"));
583            }
584            stack.push(key);
585            let mut parts = Vec::new();
586            for field in &d.fields {
587                parts.push(match field {
588                    ReturnField::Identity => {
589                        if let Some((element, _view)) = sequence_parts(ty) {
590                            Output::Sequence {
591                                element: Box::new(self.output(flat, element, None, stack)?),
592                                mode: None,
593                            }
594                        } else {
595                            Output::Leaf
596                        }
597                    }
598                    ReturnField::Projected(p) => {
599                        if let Some((element, _)) = sequence_parts(ty) {
600                            if !p.fields.is_empty() {
601                                return Err(
602                                    "collection field projections belong on the element type"
603                                        .into(),
604                                );
605                            }
606                            Output::Sequence {
607                                element: Box::new(self.output(flat, element, None, stack)?),
608                                mode: Some(p.mode),
609                            }
610                        } else {
611                            Output::Projected(p.clone())
612                        }
613                    }
614                    ReturnField::Getter(f) => {
615                        if !f.params.is_empty()
616                            || f.ret.is_some()
617                            || f.foreign.is_some()
618                            || f.error.is_some()
619                            || f.callback_return.is_some()
620                        {
621                            return Err(
622                                "getter references only support their source name in this subset"
623                                    .into(),
624                            );
625                        }
626                        let func = flat.function(&f.name)?;
627                        Output::getter(&f.name, self.output(flat, &func.ret.ty, None, stack)?)
628                    }
629                    ReturnField::Fields(d) | ReturnField::IntoFields(d) => {
630                        let f = flat.function(&d.function)?;
631                        let record = flat.record(&f.ret.ty)?;
632                        for name in d.overrides.keys() {
633                            if !record.fields.iter().any(|p| p.name == name.as_str()) {
634                                return Err(format!("unknown output field `{name}`"));
635                            }
636                        }
637                        let children = record
638                            .fields
639                            .iter()
640                            .map(|p| {
641                                self.output(
642                                    flat,
643                                    &p.ty,
644                                    d.overrides.get(&p.name.to_string()),
645                                    stack,
646                                )
647                            })
648                            .collect::<Result<Vec<_>>>()?;
649                        if matches!(field, ReturnField::IntoFields(_)) {
650                            Output::Into {
651                                function: d.function.clone(),
652                                output: Box::new(Output::OwnedFields(children)),
653                            }
654                        } else {
655                            Output::getter(&d.function, Output::fields(children))
656                        }
657                    }
658                });
659            }
660            stack.pop();
661            return match parts.len() {
662                0 => Err("empty output expansion is outside this subset".into()),
663                1 => Ok(parts.remove(0)),
664                _ => Ok(Output::combine(parts)),
665            };
666        }
667        if let Some(inner) = option_inner(ty) {
668            return Ok(Output::Optional(Box::new(
669                self.output(flat, inner, None, stack)?,
670            )));
671        }
672        if let Some((element, _view)) = sequence_parts(ty) {
673            return Ok(Output::Sequence {
674                element: Box::new(self.output(flat, element, None, stack)?),
675                mode: None,
676            });
677        }
678        Ok(if key == "()" {
679            Output::fields([])
680        } else {
681            Output::Leaf
682        })
683    }
684}

713/// Which sequence return variants an adapter should expose.
714#[derive(Clone, Copy, Debug, PartialEq, Eq)]
715pub enum CallbackReturn {
716    /// Keep the ordinary return and add a callback variant.
717    Additional,
718    /// Expose only the callback variant, including multi-value Kotlin elements
719    /// that have no ordinary List representation.
720    Only,
721}
722
723/// A source callable to export, with optional destination name and occurrence overrides.
724#[derive(Clone)]
725pub struct FunctionDecl {
726    /// Also emit an explicit callback-delivery variant for a sequence return.
727    pub callback_return: Option<CallbackReturn>,
728    /// Unqualified source function name in the flat namespace.
729    pub name: String,
730    /// Explicit destination name; supported by JNI, currently rejected by C.
731    pub foreign: Option<String>,
732    /// Input expansion overrides keyed by source parameter name.
733    pub params: BTreeMap<String, ExpandParamDecl>,
734    /// Result expansion override, replacing the type default.
735    pub ret: Option<ExpandReturnDecl>,
736    /// Source mapper for a top-level Result error; it takes E or &E and returns String.
737    pub error: Option<Box<FunctionDecl>>,
738}
739impl FunctionDecl {
740    /// Reference a source function without overriding names or conversions.
741    ///
742    /// `name` is the function's unqualified source name.
743    pub fn new(name: syn::Ident) -> Self {
744        Self {
745            callback_return: None,
746            name: name.to_string(),
747            foreign: None,
748            params: BTreeMap::new(),
749            ret: None,
750            error: None,
751        }
752    }
753    /// Add callback delivery alongside the ordinary sequence return.
754    ///
755    /// C emits `<function>_each`, whose visitor receives each element's converted
756    /// fields. Kotlin emits `<function>Fold(initial, ..., fold)`, whose folder
757    /// receives an accumulator and each element's converted Kotlin values.
758    /// Supports one projected sequence, optionally inside Option and an outer Result.
759    /// Existing buffer/list returns are preserved. Adapter builders validate the
760    /// return shape and generated-name collisions.
761    pub fn callback_return(mut self) -> Self {
762        self.callback_return = Some(CallbackReturn::Additional);
763        self
764    }
765    /// Expose only callback delivery, without generating an ordinary return.
766    ///
767    /// This explicitly permits decomposed multi-value Kotlin elements without
768    /// requiring a Kotlin data class. Generated names are the same as for
769    /// `callback_return`: C `_each` and Kotlin `Fold`.
770    pub fn callback_return_only(mut self) -> Self {
771        self.callback_return = Some(CallbackReturn::Only);
772        self
773    }
774    /// Map the error of a top-level `Result<T, E>` return with a source function.
775    ///
776    /// The helper must take exactly `E` or `&E` and return `String`. Only its
777    /// source name is accepted; it is selected for capture but is not exported.
778    /// The success value uses normal return defaults and `expand_return` overrides.
779    /// Replaces any previously selected mapper. Non-Result returns reject this option.
780    pub fn on_error(mut self, mapper: FunctionDecl) -> Self {
781        self.error = Some(Box::new(mapper));
782        self
783    }
784    /// Set a literal destination name, subject to adapter validation.
785    ///
786    /// The JNI adapter uses it instead of its default camel-case name.
787    pub fn name(mut self, name: impl Into<String>) -> Self {
788        self.foreign = Some(name.into());
789        self
790    }
791    /// Replace the expansion for one source parameter.
792    ///
793    /// # Arguments
794    ///
795    /// * `name` — the parameter's name, as written in the source.
796    /// * `decl` — how to build that argument instead of its type's default.
797    ///
798    /// # Panics
799    /// Panics if that parameter already has an override. Resolution checks that the
800    /// parameter exists and the expansion type matches.
801    pub fn expand_param(mut self, name: impl AsRef<str>, decl: ExpandParamDecl) -> Self {
802        assert!(
803            self.params.insert(name.as_ref().into(), decl).is_none(),
804            "duplicate parameter override"
805        );
806        self
807    }
808    /// Replace the result expansion, including any previously assigned override.
809    pub fn expand_return(mut self, decl: ExpandReturnDecl) -> Self {
810        self.ret = Some(decl);
811        self
812    }
813}

815/// Collect source callable names referenced by bindings and nested expansions.
816///
817/// Used when selecting captured items before constructing a flat model. The
818/// returned list can contain duplicates; collecting names does not resolve or
819/// validate the recipes or export their helper functions.
820///
821/// # Arguments
822///
823/// * `expansions` — type-wide defaults; constructors, getters and error mappers are collected.
824/// * `functions` — exported functions; their names and those referenced by their
825///   overrides are collected.
826pub fn source_functions<'a>(
827    expansions: &[ExpandDecl],
828    functions: impl IntoIterator<Item = &'a FunctionDecl>,
829) -> Vec<String> {
830    fn input(d: &ExpandParamDecl, names: &mut Vec<String>) {
831        for f in d.variants.iter().flatten() {
832            function(f, names);
833        }
834    }
835    fn output(d: &ExpandReturnDecl, names: &mut Vec<String>) {
836        for field in &d.fields {
837            match field {
838                ReturnField::Identity | ReturnField::Projected(_) => {}
839                ReturnField::Getter(f) => function(f, names),
840                ReturnField::Fields(d) | ReturnField::IntoFields(d) => {
841                    names.push(d.function.clone());
842                    for d in d.overrides.values() {
843                        output(d, names);
844                    }
845                }
846            }
847        }
848    }
849    fn function(f: &FunctionDecl, names: &mut Vec<String>) {
850        names.push(f.name.clone());
851        if let Some(mapper) = &f.error {
852            function(mapper, names);
853        }
854        for d in f.params.values() {
855            input(d, names);
856        }
857        if let Some(d) = &f.ret {
858            output(d, names);
859        }
860    }
861    let mut names = Vec::new();
862    for f in functions {
863        function(f, &mut names);
864    }
865    for d in expansions {
866        match d {
867            ExpandDecl::Error(d) => function(&d.mapper, &mut names),
868            ExpandDecl::Param(d) => input(d, &mut names),
869            ExpandDecl::Return(d) => output(d, &mut names),
870        }
871    }
872    names
873}
