3/// Fully qualify source types using the binding's final module policy.
4pub(super) fn qualify_source_type(ty: &syn::Type, source_module: Option<&syn::Path>) -> syn::Type {
5    use syn::visit_mut::VisitMut;
6
7    struct Qualifier<'a>(Option<&'a syn::Path>);
8    impl VisitMut for Qualifier<'_> {
9        fn visit_type_path_mut(&mut self, ty: &mut syn::TypePath) {
10            if ty.qself.is_none() {
11                let as_type = syn::Type::Path(ty.clone());
12                if is_scalar(&as_type) {
13                    // Primitive scalars have no module.
14                } else if is_string(&as_type) {
15                    ty.path = syn::parse_quote!(::std::string::String);
16                } else if ty.path.leading_colon.is_none() && ty.path.segments.len() == 1 {
17                    let segment = ty.path.segments[0].clone();
18                    if let Some(full) = prelude_path(&segment.ident) {
19                        let mut path: syn::Path =
20                            syn::parse_str(full).expect("a path literal in `prelude_path`");
21                        path.segments.last_mut().expect("non-empty").arguments = segment.arguments;
22                        ty.path = path;
23                    } else if let Some(module) = self.0 {
24                        let mut path = module.clone();
25                        path.segments.push(segment);
26                        ty.path = path;
27                    }
28                }
29            }
30            // Visit generic arguments after qualifying the current path, so
31            // `Option<Handle>` resolves the two names against different roots.
32            syn::visit_mut::visit_type_path_mut(self, ty);
33        }
34    }
35
36    // Trait bounds are paths rather than `TypePath`s. This deliberately leaves
37    // `impl Fn(..) + Send` alone while still reaching its argument types.
38    let mut result = ty.clone();
39    Qualifier(source_module).visit_type_mut(&mut result);
40    result
41}

264    /// Declare an **inline-opaque by-value, plain-data** type: the Rust value

266    /// to/from `opaque_ty`, an opaque `#[repr(C, align(_))]` counterpart of

291    /// Shared body of [`Self::opaque_data_struct`] / [`Self::opaque_owned_struct`].
292    pub(super) fn declare_opaque(
293        mut self,
294        rust_ty: syn::Type,
295        opaque_ty: syn::Type,
296        kind: OpaqueKind,
297    ) -> Self {
298        let method = match kind {
299            OpaqueKind::Data => "opaque_data_struct",
300            OpaqueKind::Owned => "opaque_owned_struct",

321    /// Declare a **`#[repr(C)]`, FFI-safe value struct** crossed **by direct
322    /// reinterpret** (zero-copy) — the C struct's memory *is* the Rust struct's
323    /// memory. Unlike [`Self::data_struct`] (which copies each field, lowering a
324    /// `String` to `char*`), this passes the whole struct by value via
325    /// `prebindgen_c_runtime::Transmute` and a `&T` borrow / `impl Fn(&T)` callback as a
326    /// zero-copy `*const` pointer cast — the value-opaque machinery, but with an
327    /// **auto-generated visible-field** `#[repr(C)]` C mirror (so C reads the
328    /// fields directly) instead of an opaque blob.
329    ///
330    /// Every field must be FFI-safe: a primitive, a declared
331    /// [`Self::enum_type`], or an **opaque pointer** `Option<Box<T>>` / `Box<T>`
332    /// where `T` is a declared [`Self::opaque_ptr`] (rendered `*mut t_t`; this is
333    /// how a heap `String` rides along — `Option<Box<String>>` → `string_t *`). The
334    /// source type **must** be `#[repr(C)]`; a fail-closed `size_of`/`align_of`
335    /// assert against the generated mirror proves the reinterpret sound at compile
336    /// time. Call after the manglers are configured (the mirror name is resolved
337    /// through them). A `<base>_drop` is generated.
338    ///
339    /// **Owned-ness is inferred** from the fields: a struct with an opaque-pointer field
340    /// owns external resources, so a by-value consume cleans the moved-from slot (nulls
341    /// the owned pointers) to keep the caller's later `_drop` a no-op — no `.owned()`
342    /// modifier and no double-free footgun. A struct with only scalar/enum fields is
343    /// plain data (a by-value crossing is a bitwise copy with no write-back). The source
344    /// type needs `Default` **only** if it has a bare `Box<T>` field (whose gravestone
345    /// can't be a NULL pointer); `Option<Box<T>>` fields are nulled in place.
346    ///
347    /// # Why the spelling is load-bearing here
348    ///
349    /// This is the one position that is **exempt** from the rule stated on
350    /// [`prebindgen_registry::Prebindgen`] — *same `kind` ⇒ same
351    /// destination-language type* — and the exemption is structural rather than
352    /// a concession. A mirror is not converted; it is **reinterpreted from the
353    /// source struct's bytes**, so its field types are a *layout* fact. `Box<T>`
354    /// is a pointer and `T` is inline: to C they really are different types, and
355    /// the size/align assert above would reject a mirror that pretended
356    /// otherwise. So this path reads the wrapper the model erases —
357    /// [`kind`](prebindgen_registry::flat::TypeRef::kind) rather than
358    /// [`unwrapped`](prebindgen_registry::flat::TypeRef::unwrapped) — on
359    /// purpose. It is the one place the usual "classify off `kind`, spell off
360    /// the syntax" split inverts, and it inverts because the contract is layout
361    /// rather than surface.
362    ///
363    /// That is also why a wrapper the model erases must **not** be refused here
364    /// (prebindgen#230). `Option<Box<String>>` is how a source crate says "this
365    /// field is a nullable pointer" — a layout statement a zero-copy mirror is
366    /// entitled to read, not the source naming a C type. There is no competing
367    /// spelling to prefer: `Option<String>` is a 24-byte niche-optimised value
368    /// with no C representation at all, and declaring one is a hard error naming
369    /// the field. Rejecting the `Box` would leave a nullable-pointer field
370    /// inexpressible.
371    pub fn repr_c_struct(mut self, ty: syn::Type) -> Self {
372        let key = TypeKey::from_type(&ty);
373        assert!(
374            !self.ignored_types.contains(&key),
375            "Cbindgen::repr_c_struct cannot declare `{}` because it is already ignored",
376            key
377        );
378        let mirror = self.c_type_ident(&key);
379        self.value_opaque.insert(
380            key.clone(),
381            ValueOpaqueCfg {
382                opaque: syn::parse_quote!(#mirror),
383                kind: OpaqueKind::Data,
384                generate_mirror: true,
385                assume_c_field_validity: false,
386                cfg: TypeCfg::new(ty),
387            },
388        );
389        self.current = Some(CurrentDecl::ValueOpaque(key));
390        self
391    }

393    /// Accept a [`Self::repr_c_struct`] whose mirror has **restricted-validity**
394    /// fields, taking responsibility for their bytes.
395    ///
396    /// A `repr_c_struct` crosses IN by one whole-struct reinterpret, so there is
397    /// no per-field hook where a C-supplied byte could be normalised or checked
398    /// before the source struct exists. A field whose Rust type accepts only
399    /// *some* bit patterns — `bool` (`0`/`1`) or a declared [`Self::enum_type`]
400    /// (the declared discriminants) — is therefore undefined behaviour the
401    /// moment C writes anything else into the mirror and hands it back. The
402    /// generator rejects such a declaration by default (#170 instance 3, #158
403    /// instance 3); the real fix is a raw-wire lowering, which does not exist
404    /// yet.
405    ///
406    /// This modifier is the acknowledgement, not a fix: it says the C side of
407    /// this binding is trusted to write only in-domain bytes into those fields.
408    /// It exists so that the audit rejects **silently unsound new declarations**
409    /// without removing bindings that already ship. Chain it directly after the
410    /// [`Self::repr_c_struct`] it applies to; the panic message names every
411    /// field it would cover.
412    ///
413    /// Prefer, in order: move the field into a [`Self::data_struct`] (per-field
414    /// wires, so `bool` normalises), pass it as a separate scalar parameter, or
415    /// widen it to an integer the whole domain of which is valid.
416    pub fn assume_c_field_validity(mut self) -> Self {
417        match self.current.clone() {
418            Some(CurrentDecl::ValueOpaque(key)) => {
419                self.value_opaque
420                    .get_mut(&key)
421                    .expect("entry vanished")
422                    .assume_c_field_validity = true;
423            }
424            other => panic!(
425                "Cbindgen::assume_c_field_validity must follow a `repr_c_struct` declaration, \
426                 not {}",
427                describe_current(&other)
428            ),
429        }
430        self
431    }

452    /// Set the **base name** token of the **current declaration** (universal
453    /// modifier): the per-declaration base fed to the name manglers, replacing the
454    /// auto-derived one. For a type it replaces the `mangle_rust_type` base (so
455    /// `mangle_type_name`/`mangle_destructor`/`mangle_take` all see it); for a
456    /// function it replaces the ident fed to `mangle_function`; for a callback it
457    /// is the sole base fed to `mangle_callback` (replacing the args' bases —
458    /// useful to disambiguate e.g. `&T` from `T` closures). E.g.
459    /// `.callback(...).base_name("sample_ref")` with a `|bases| "z_closure_{…}_t"`
460    /// mangler → `z_closure_sample_ref_t`. Panics if not chained directly after a
461    /// declaration.
462    pub fn base_name(mut self, base: impl Into<String>) -> Self {
463        let base = base.into();
464        match self.current.clone() {
465            Some(CurrentDecl::Ptr(key)) => {
466                self.opaque.get_mut(&key).expect("entry vanished").base = Some(base);
467            }
468            Some(CurrentDecl::Data(key)) => {
469                self.data.get_mut(&key).expect("entry vanished").base = Some(base);
470            }
471            Some(CurrentDecl::ValueOpaque(key)) => {
472                self.value_opaque
473                    .get_mut(&key)
474                    .expect("entry vanished")
475                    .cfg
476                    .base = Some(base);
477            }
478            Some(CurrentDecl::Enum(key)) => {
479                self.enums.get_mut(&key).expect("entry vanished").base = Some(base);
480            }
481            Some(CurrentDecl::TaggedUnion(key)) => {
482                self.tagged_unions
483                    .get_mut(&key)
484                    .expect("entry vanished")
485                    .base = Some(base);
486            }
487            Some(CurrentDecl::Callback(key)) => {
488                self.callbacks.get_mut(&key).expect("entry vanished").base = Some(base);
489            }
490            Some(CurrentDecl::Function(ident)) => {
491                self.functions.get_mut(&ident).expect("entry vanished").base = Some(base);
492            }
493            Some(CurrentDecl::Convert(key)) => {
494                self.convert_bases.insert(key, base);
495            }
496            None => panic!(
497                "Cbindgen::base_name must be chained directly after a declaration \
498                 (`opaque_ptr` / `data_struct` / `enum_type` / `tagged_union` / `callback` / \
499                 `function` / `convert`)"
500            ),
501        }
502        self
503    }

524    /// `Result<_, E>` but is *not* a by-value [`Self::data_struct`] (e.g.
525    /// `ZError = Box<dyn Error + Send + Sync>`). Such an error is marshalled to C
526    /// as a `char*` message obtained by calling `message_fn(&err) -> String`
527    /// (e.g. `z_error_message`); the generated wrapper's error out-param becomes
528    /// `char **e`. The type must implement `From<String>` (so a fallible input's
529    /// internal message can be lifted into it). Root-level modifier (resets the
530    /// current declaration).
531    pub fn opaque_error(mut self, error_ty: syn::Type, message_fn: syn::Ident) -> Self {
532        let key = TypeKey::from_type(&error_ty);
533        self.error.insert(key.clone());
534        self.opaque_errors.insert(key, message_fn);
535        self.clear_current();
536        self
537    }

539    /// Declare a C-like (fieldless) enum type to convert. (Mirrors `JniExt`'s
540    /// `enum_class`.)
541    ///
542    /// Crossing **into** Rust, the caller's discriminant is validated before any
543    /// Rust enum is built — an out-of-range one is a fallible-input error, so a
544    /// non-`Result` function taking this enum by value needs [`Self::panic`].
545    /// See the module docs for why the wire is `MaybeUninit<mirror>`.
546    pub fn enum_type(mut self, ty: syn::Type) -> Self {
547        let key = TypeKey::from_type(&ty);
548        assert!(
549            !self.ignored_types.contains(&key),
550            "Cbindgen::enum_type cannot declare `{}` because it is already ignored",
551            key
552        );
553        self.enums.insert(key.clone(), TypeCfg::new(ty));
554        self.current = Some(CurrentDecl::Enum(key));
555        self
556    }

558    /// Declare a **data-carrying** enum: it crosses by value as a `#[repr(C)]`
559    /// enum with payload variants, which cbindgen renders as the idiomatic C
560    /// tagged union (a tag enum plus a `union` of the variant bodies). The
561    /// counterpart of [`Self::enum_type`], which is for the unit-variant-only
562    /// case a plain C `enum` can hold. (Mirrors `JniExt`'s `sealed_class`.)
563    ///
564    /// Each payload field crosses as its own wire, chosen by the same policy a
565    /// [`Self::repr_c_struct`] field uses, extended with `String` → `char *`:
566    /// a scalar passes through, a declared [`Self::enum_type`] becomes its C
567    /// enum, a `String` becomes a malloc'd `char *`, and an opaque pointer
568    /// `Option<Box<T>>` / `Box<T>` (with `T` a declared [`Self::opaque_ptr`])
569    /// becomes `*mut t_t`. Anything else is a generation error.
570    ///
571    /// **Ownership.** The union crosses by value, so when any variant's
572    /// payload wire owns memory (`char *`, an opaque pointer) a typed
573    /// `<base>_drop(t_t *)` is generated that frees the **active arm** —
574    /// consistent with the existing typed per-pointer drops. A union whose
575    /// payloads are all plain data needs no drop and gets none.
576    ///
577    /// **Validity.** Crossing **into** Rust, the tag a C caller supplied is
578    /// range-checked before any Rust enum is built — so, exactly like
579    /// [`Self::enum_type`], a union taken by value is a fallible input, and a
580    /// function taking one without a `Result` return needs [`Self::panic`].
581    /// A data struct carrying a union field inherits that fallibility. See the
582    /// module docs for the wire this uses and why.
583    pub fn tagged_union(mut self, ty: syn::Type) -> Self {
584        let key = TypeKey::from_type(&ty);
585        assert!(
586            !self.ignored_types.contains(&key),
587            "Cbindgen::tagged_union cannot declare `{}` because it is already ignored",
588            key
589        );
590        self.tagged_unions.insert(key.clone(), TypeCfg::new(ty));
591        self.current = Some(CurrentDecl::TaggedUnion(key));
592        self
593    }

595    /// Declare a callback signature so its `impl Fn(...)` parameters resolve and
596    /// a `#[repr(C)]` closure struct (`{ void *context; call; drop }`) is
597    /// emitted for it. `ty` must be `impl Fn(Args...) + Send + Sync + 'static`.
598    /// Identical signatures share one struct. Sets the declaration cursor, so a
599    /// following `.base_name("...")` sets the base fed to
600    /// [`mangle_callback`](Self::mangle_callback) (else the args' bases drive
601    /// the generated name).
602    pub fn callback(mut self, ty: syn::Type) -> Self {
603        let args = extract_fn_trait_args(&ty).unwrap_or_else(|| {
604            panic!(
605                "Cbindgen::callback expects `impl Fn(Args...) + Send + Sync + 'static`, got `{}`",
606                ty.to_token_stream()
607            )
608        });
609        let key: CallbackKey = args.iter().map(TypeKey::from_type).collect();
610        self.callbacks.insert(key.clone(), CbCfg::new());
611        self.current = Some(CurrentDecl::Callback(key));
612        self
613    }

615    /// Mark argument `idx` of the **current callback declaration** as a *takeable
616    /// owned pointer*: the C `call` receives `*mut z_x_t` (not by value); the
617    /// callee may take the value (`z_x_take` moves it out, leaving a gravestone) or
618    /// just read it; the trampoline drops it after the call (no-op if taken). The
619    /// arg type must be an inline-opaque type ([`Self::opaque_owned_struct`] /
620    /// [`Self::opaque_data_struct`]). Chain after `.callback(...)` (and any `.base_name(...)`).
621    pub fn takeable_param(mut self, idx: usize) -> Self {
622        match &self.current {
623            Some(CurrentDecl::Callback(key)) => {
624                let key = key.clone();
625                self.callbacks
626                    .get_mut(&key)
627                    .expect("entry vanished")
628                    .takeable
629                    .insert(idx);
630            }
631            other => panic!(
632                "Cbindgen::takeable_param must be chained after a `callback(...)` call, not after {}",
633                describe_current(other)
634            ),
635        }
636        self
637    }

641    /// Fully-qualify every bare single-segment source type inside `ty` against
642    /// [`Self::source_module`] (e.g. `ZKeyExpr` → `zenoh_flat::ZKeyExpr`).
643    /// Anything already qualified, or with no `source_module` set, is returned
644    /// unchanged.
645    ///
646    /// **Every** path node in the type, not the outermost one: a source type
647    /// can sit under any accepted shape — `Option<&Handle>` in a callback
648    /// argument, `&[Payload]`, `(Rec, Rec)` — and a name left bare there does
649    /// not resolve in the consumer crate the generated file is included into
650    /// (#414). The walk is what makes one rule apply everywhere; the rule per
651    /// node is below.
652    pub(super) fn src_ty(&self, ty: &syn::Type) -> syn::Type {
653        qualify_source_type(ty, self.source_module.as_ref())
654    }

656    /// Path to a source function (e.g. `zenoh_flat::z_keyexpr_try_from`).
657    pub(super) fn src_fn(&self, ident: &syn::Ident) -> syn::Path {
658        match &self.source_module {
659            Some(m) => {
660                let mut p = m.clone();
661                p.segments.push(syn::PathSegment::from(ident.clone()));
662                p
663            }
664            None => syn::Path::from(ident.clone()),
665        }

679    /// Config of a declared type (across the opaque/data/enum maps), by key.

694    /// Type keys used as a takeable callback parameter (any `.takeable_param(idx)`
695    /// across all declared callbacks). These value_opaque types get a public
696    /// `<base>_take(dst, src)` move function.
697    pub(super) fn takeable_type_keys(&self) -> HashSet<TypeKey> {
698        let mut s = HashSet::new();
699        for (key, cfg) in &self.callbacks {
700            for &idx in &cfg.takeable {
701                if let Some(tk) = key.get(idx) {
702                    s.insert(tk.clone());
703                }
704            }
705        }
706        s

709    /// Public "take" (move) symbol for a takeable value_opaque type:
710    /// [`Self::mangle_take`] over the base, else `<base>_take` (e.g.
711    /// `z_sample_take`). Symmetric with [`Self::destructor_symbol`].
712    pub(super) fn take_symbol(&self, key: &TypeKey) -> syn::Ident {
713        if let Some(f) = &self.mangle_take {
714            return format_ident!("{}", f(&self.rust_base(key)));
715        }
716        format_ident!("{}_take", self.rust_base(key))
717    }

719    /// Base token for a Rust type: [`Self::mangle_rust_type`] applied to the Rust
720    /// short name, or the short name verbatim when unset. Feeds the type-name,
721    /// destructor and callback manglers.
722    pub(super) fn rust_base(&self, key: &TypeKey) -> String {
723        if let Some(b) = self.type_cfg(key).and_then(|c| c.base.clone()) {
724            return b;
725        }
726        let short = type_short(key);
727        match &self.mangle_rust_type {
728            Some(f) => f(&short),
729            // No mangler: a C-like `snake_case` default (so destructors/take/type
730            // names read e.g. `sample_drop`, not `Sample_drop`).
731            None => snake_case(&short),
732        }
733    }

735    /// Emitted C type name of a declared type: [`Self::mangle_type_name`] over the
736    /// base, else the base (which is the `mangle_rust_type`/`.base_name` token).
737    pub(super) fn c_type_name(&self, key: &TypeKey) -> String {
738        let base = self.rust_base(key);
739        match &self.mangle_type_name {
740            Some(f) => f(&base),
741            None => base,
742        }
743    }

760    /// Emitted C type name of a callback's closure struct: [`Self::mangle_callback`]
761    /// over the bases — a `.base_name(...)` override (as the sole base) when set,
762    /// else the args' derived bases — or, with no mangler, a generic default
763    /// (`closure` for zero bases, `closure_<base0>_<base1>…` otherwise). The
764    /// adapter's own default carries no target-language naming convention.
765    pub(super) fn callback_c_name(&self, key: &CallbackKey) -> String {
766        let base_override = self.callbacks.get(key).and_then(|c| c.base.clone());
767        if let Some(f) = &self.mangle_callback {
768            // The override (when set) is the sole base; otherwise the args' bases.
769            let bases: Vec<String> = match &base_override {
770                Some(b) => vec![b.clone()],
771                None => key.iter().map(|k| self.rust_base(k)).collect(),
772            };
773            return f(&bases);
774        }
775        // No mangler: an explicit base is the name as-is; otherwise compose from
776        // the args' bases.
777        if let Some(b) = base_override {
778            return b;
779        }
780        if key.is_empty() {
781            "closure".to_string()
782        } else {
783            let parts: Vec<String> = key.iter().map(|k| self.rust_base(k)).collect();
784            format!("closure_{}", parts.join("_"))
785        }
786    }
787
788    /// C struct identifier for a callback's closure type (see
789    /// [`Self::callback_c_name`]).
790    pub(super) fn callback_c_ident(&self, key: &CallbackKey) -> syn::Ident {
791        format_ident!("{}", self.callback_c_name(key))
792    }

795/// Human-readable description of the current declaration, for panic messages.
796fn describe_current(current: &Option<CurrentDecl>) -> String {
797    match current {
798        None => "no declaration".to_string(),
799        Some(CurrentDecl::Ptr(k)) => format!("opaque_ptr `{}`", k.as_str()),
800        Some(CurrentDecl::Data(k)) => format!("data_struct `{}`", k.as_str()),
801        Some(CurrentDecl::ValueOpaque(k)) => format!("value_opaque `{}`", k.as_str()),
802        Some(CurrentDecl::Enum(k)) => format!("enum_type `{}`", k.as_str()),
803        Some(CurrentDecl::TaggedUnion(k)) => format!("tagged_union `{}`", k.as_str()),
804        Some(CurrentDecl::Callback(k)) => {
805            let args: Vec<&str> = k.iter().map(|t| t.as_str()).collect();
806            format!("callback `impl Fn({})`", args.join(", "))
807        }
808        Some(CurrentDecl::Function(i)) => format!("function `{i}`"),
809        Some(CurrentDecl::Convert(k)) => format!("convert `{k}`"),
810    }
