1//! Hand the demand over, and grade the answers.
2//!
3//! The two halves of the exchange with a generator: [`Registry::crossings`]
4//! sorts the derived set inner-first, and `RegistryBuilder::build` grades what
5//! came back — every crossing answered, or a report naming the ones that were
6//! not.
7
8use std::collections::HashSet;
9
10use super::*;
11
12impl Registry {
13    /// Every crossing this binding needs a conversion for, **inner types
14    /// first**.
15    ///
16    /// The order is the whole point: a generator walking this list has already
17    /// built everything a given crossing can compose from, so it can work from
18    /// a flat list instead of being called back per type. Derived from
19    /// [`Self::immediate_edges`], which is structural — generic arguments,
20    /// tuple/reference/slice targets, declared struct fields, and `impl Fn`
21    /// arguments with the direction flipped — so no generator is consulted to
22    /// produce it.
23    ///
24    /// **Cycles.** A self-referential type (`struct Node { next:
25    /// Option<Box<Node>> }`) has no topological order. The walk breaks such a
26    /// cycle at its entry, so exactly one member is handed out before an inner
27    /// it contains; a generator that cannot build it supplies nothing, and
28    /// `RegistryBuilder::build` reports it like any other gap.
29    pub(crate) fn crossings(&self) -> Vec<Crossing> {
30        // Post-order DFS: a node is emitted only after everything it reaches,
31        // which IS inner-first. `visiting` breaks cycles — the back edge is
32        // simply not followed, so the node it points at lands later than its
33        // dependent, and that is the one documented exception above.
34        let mut order: Vec<Crossing> = Vec::new();
35        let mut done: HashSet<Crossing> = HashSet::new();
36        let mut visiting: HashSet<Crossing> = HashSet::new();
37
38        // Deterministic roots: same list every build, so a generator's output
39        // cannot depend on hash order.
40        let mut roots: Vec<Crossing> = Vec::new();
41        for dir in [Direction::Construct, Direction::Deconstruct] {
42            let mut keys: Vec<&TypeKey> = self.type_table(dir).keys().collect();
43            keys.sort_by(|a, b| a.as_str().cmp(b.as_str()));
44            roots.extend(keys.into_iter().map(|k| (dir, k.clone())));
45        }
46
47        for root in roots {
48            self.visit_crossing(root, &mut order, &mut done, &mut visiting);
49        }
50        order
51    }
52
53    pub(super) fn visit_crossing(
54        &self,
55        node: Crossing,
56        order: &mut Vec<Crossing>,
57        done: &mut HashSet<Crossing>,
58        visiting: &mut HashSet<Crossing>,
59    ) {
60        if done.contains(&node) || !visiting.insert(node.clone()) {
61            return;
62        }
63        let (dir, key) = node.clone();
64        // Every node this walk reaches has a cell: the roots are the table's own
65        // keys, and each edge below is filtered by `contains_key`. So what
66        // `plan_edges` needs is the reading the registry already stored — not
67        // one re-derived from the key (#291).
68        let plan_edges = self
69            .type_table(dir)
70            .get(&key)
71            .map(|cell| self.plan_edges(dir, &cell.subject))
72            .unwrap_or_default();
73        let mut edges: Vec<Crossing> = self
74            .immediate_edges(dir, &key)
75            .into_iter()
76            // The structural edges arrive as readings, so the key is the
77            // model's own answer rather than one re-derived from a spelling.
78            .map(|(d, t)| (d, t.key()))
79            .chain(plan_edges)
80            .chain(
81                self.declared
82                    .edges
83                    .iter()
84                    .filter(|(from, _)| *from == node)
85                    .map(|(_, on)| (on.0, on.1.clone())),
86            )
87            // Only crossings the scan actually registered: a structural edge to
88            // a type nothing asked for is not a crossing.
89            .filter(|c| self.type_table(c.0).contains_key(&c.1))
90            .collect();
91        edges.sort_by(|a, b| (a.0 as u8, a.1.as_str()).cmp(&(b.0 as u8, b.1.as_str())));
92        for edge in edges {
93            self.visit_crossing(edge, order, done, visiting);
94        }
95        visiting.remove(&node);
96        if done.insert(node.clone()) {
97            order.push(node);
98        }
99    }
100
101    /// Dependencies a **decomposition** adds, which the structural walk cannot
102    /// see.
103    ///
104    /// A callback argument delivered as leaves needs each leaf's own conversion
105    /// before the callback's can be built — and a leaf is named by a
106    /// decomposition, not by the argument's syntax. Without this the order would
107    /// be structurally correct and still wrong, which is exactly the kind of gap
108    /// the old fixed-point loop papered over by retrying.
109    ///
110    /// The leaves are the adapter's answer, declared as
111    /// [`Decompositions::callback_arg_leaves`](crate::Decompositions::callback_arg_leaves):
112    /// which readings a decomposition delivers is the adapter's to say, while
113    /// what that means for the order is the registry's.
114    /// Crossings, not spellings: every edge here is a table lookup, and the
115    /// model names both ends. `callback_args` is the classification
116    /// `extract_fn_trait_args` re-derived from the parameter's bounds, and a
117    /// leaf reading's key is its own.
118    pub(super) fn plan_edges(
119        &self,
120        dir: Direction,
121        ty: &prebindgen_flat::flat::TypeRef,
122    ) -> Vec<Crossing> {
123        if dir != Direction::Construct {
124            return Vec::new();
125        }
126        let Some(args) = ty.callback_args() else {
127            return Vec::new();
128        };
129        let mut out = Vec::new();
130        for arg in args {
131            if let Some(leaves) = self
132                .declared
133                .decompositions
134                .callback_arg_leaves
135                .get(&arg.key())
136            {
137                for leaf in leaves {
138                    out.push((Direction::Deconstruct, leaf.key()));
139                }
140            }
141        }
142        out
143    }
