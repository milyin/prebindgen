1//! The return side, read off the rows rather than off the decomposition.
2//!
3//! `unfold` still builds every plan the emitters use. This reads the same
4//! decompositions off the `parts` rows the recipe table already carries, and
5//! holds the two to each other — a build fails on any difference (#701 step 3).
6//!
7//! The comparison goes with the older path, once every decomposition this
8//! binding declares has been through it. Until then it is the oracle a fixture
9//! cannot be: it runs over `covertest-kotlin`, `perftest-kotlin`, `emitcheck`
10//! and zenoh-flat-jni, which is the whole declared surface.
11

14    fold::{Folding, UnfoldPolicy},
15    leaf::{Hoist, LeafSource, UnfoldLeaf},
16    recipe::{Reach, RecipeName, Recipes},
17};

21/// What the JVM calls the values a decomposition delivers.
22///
23/// Every answer is one the decomposition writes into a leaf today, and the
24/// comparison below is what holds them to it.
25pub(crate) struct JniUnfold<'a> {
26    decls: &'a Declarations,
27    /// The registry, for the one name lookup that needs a signature.
28    registry: &'a dyn prebindgen_registry::Conversions,
29    /// The type whose declaration names these parts. A part name is the
30    /// declaration's answer, and a declaration is per type.
31    key: prebindgen_registry::TypeKey,
32}
33
34impl UnfoldPolicy for JniUnfold<'_> {
35    fn selector(&self, source: &TypeRef) -> UnfoldLeaf {
36        UnfoldLeaf {
37            name: "tag".to_string(),
38            path: Vec::new(),
39            // The sum, not the `i32` it crosses as: this is how the emitter
40            // finds the enum to match on. Registered and not required — a sum
41            // has no whole-value output converter (#282).
42            out_ty: source.clone(),
43            identity: false,
44            nullable: false,
45            source: LeafSource::SumTag,
46            groups: Vec::new(),
47        }
48    }
49
50    fn presence(&self, name: &str, source: &TypeRef) -> UnfoldLeaf {
51        UnfoldLeaf {
52            name: format!("{name}__present"),
53            path: Vec::new(),
54            // The OPTIONAL as written: the emitter tests that value to set the
55            // flag, so the reading it names is the one it tests. It is not a
56            // `bool` on this side — what crosses is the presence of a value the
57            // JVM sees as nullable.
58            out_ty: source.clone(),
59            identity: false,
60            nullable: false,
61            source: LeafSource::Presence,
62            groups: Vec::new(),
63        }
64    }
65
66    fn part_name(&self, reach: &Reach, index: usize, field: Option<&syn::Ident>) -> String {
67        match reach {
68            Reach::Accessor(func) => self.decls.leaf_name_of(&self.key, func, index),
69            // A synthesized `data_class` decomposition names its slots after
70            // the struct's own fields — as the KOTLIN property is named, which
71            // camel-cases and escapes a name that is a keyword there. A
72            // declared decomposition names its parts itself.
73            _ => match field {
74                Some(name) => crate::jni::mangle_kotlin_ident(&crate::jni::kt_snake_to_camel(
75                    &name.to_string(),
76                )),
77                None => self.decls.leaf_name_at(&self.key, index),
78            },
79        }
80    }
81
82    fn arm_part_name(
83        &self,
84        sum: &TypeRef,
85        variant: &syn::Ident,
86        member: &syn::Member,
87        _index: usize,
88    ) -> String {
89        // The Kotlin variant class's name and the Kotlin property's, which is
90        // what the slot is called on the far side — neither is the Rust name.
91        // Read off the SUM's own declaration: a sum reached as a part of
92        // something else renames its variants for itself.
93        let kotlin = match self
94            .decls
95            .types
96            .get(&sum.stripped_key())
97            .and_then(|c| c.sum())
98        {
99            Some(cfg) => self.decls.sum_variant_class_name(cfg, variant),
100            None => variant.to_string(),
101        };
102        crate::jni::struct_plan::sum_slot_fragment(
103            &kotlin,
104            &crate::jni::struct_plan::sum_field_prop_name(member),
105        )
106    }
107
108    fn value_form_part(&self, index: usize) -> Option<String> {
109        // A value form's parts are named by the declaration that lists them,
110        // which is the one record `field_list` holds for such a declaration.
111        self.decls
112            .value_form_part_name(self.registry, &self.key, index)
113    }
114
115    fn identity_name(&self) -> String {
116        // A root identity leaf is `handle`; a nested one takes the name of the
117        // part it was reached through, which the view supplies instead.
118        "handle".to_string()
119    }
120
121    fn nest(&self, outer: &str, inner: &str) -> String {
122        format!("{outer}__{inner}")
123    }
124}

126impl Declarations {
127    /// Read every decomposition back off its row and hold the older path to it.
128    ///
129    /// Temporary, and deleted with `unfold::apply`. A decomposition whose row
130    /// the view does not read yet is skipped rather than failing: the forms it
131    /// refuses are the ones step 3 is still replacing, and the point of this is
132    /// what the two paths say about the forms it DOES read.
133    pub(crate) fn check_unfold_parity(
134        &self,
135        model: &Flat,
136        registry: &dyn prebindgen_registry::Conversions,
137        recipes: &Recipes,
138        bindings: &prebindgen_registry::recipe::Bindings,
139    ) -> Result<(), String> {
140        let folding = Folding::new(recipes, model);
141        // What was NOT compared, and why. Reported rather than discarded: a
142        // skip is work #701's step 3 still owes, and a list of them is how a
143        // reader sees the differential's reach shrink or grow.
144        let mut skipped: Vec<String> = Vec::new();
145        let mut compared = 0usize;
146        let unfolded = self.unfolded();
147        let mut plans: Vec<(String, &crate::unfold::UnfoldPlan)> = unfolded
148            .unfold_plans
149            .iter()
150            .map(|(f, p)| (format!("`{f}`'s return"), p))
151            .chain(
152                unfolded
153                    .error_plans
154                    .iter()
155                    .map(|(f, p)| (format!("`{f}`'s error"), p)),
156            )
157            .chain(
158                unfolded
159                    .callback_arg_plans
160                    .iter()
161                    .map(|(k, p)| (format!("the callback argument `{k}`"), p)),
162            )
163            .collect();
164        // Sorted, so a binding with two differences reports the same one every
165        // time rather than whichever the hash order reached first.
166        plans.sort_by(|(a, _), (b, _)| a.cmp(b));
167
168        for (what, plan) in plans {
169            // EVERY decomposition lands in exactly one bucket: compared, or
170            // named in the skip list. A `continue` before the list is a
171            // decomposition leaving the comparison with nothing to say so, and
172            // the pinned set then covers a part of the surface rather than all
173            // of it — which is this check disabling itself one level up from
174            // where it looks.
175            //
176            // A whole-element fold takes nothing apart: each element crosses
177            // through its own converter, so there is no row and nothing to
178            // compare.
179            let Some(decon) = &plan.decon else {
180                skipped.push(format!("{what}: whole-element-fold"));
181                continue;
182            };
183            // A per-function `.expand_return(...)` states a decomposition of
184            // its own, which is a row of its own under a name of its own —
185            // the shape step 2 gave the parameter side, and step 3's remaining
186            // work here. The type's row does not describe it, so comparing the
187            // two would compare two different decompositions.
188            if !matches!(decon, prebindgen_registry::leaf::DeconId::Default(_)) {
189                skipped.push(format!("{what}: per-function-expand-return"));
190                continue;
191            }
192            // A `sealed_class` that ALSO carries an `expand_return!` states two
193            // decompositions under one row name: the sum's arms, declared by
194            // the class, and the fields, declared by the author. The row table
195            // keeps the first and the decomposition used the second, so the two
196            // describe different things and comparing them says nothing. One
197            // row name for two meanings is step 3's to resolve.
198            if self.is_sum(&plan.source) && self.declares_return_expand(&plan.source) {
199                skipped.push(format!("{what}: sum-and-expand-return"));
200                continue;
201            }
202            let policy = JniUnfold {
203                decls: self,
204                registry,
205                key: plan.source.stripped_key(),
206            };
207            let row = RecipeName::new("parts");
208            // The reading as the decomposition holds it: a plan reached by
209            // reference is lent the value, and its root hands out a borrow
210            // rather than the value itself. `plan.source` is the owned core, so
211            // the borrow has to be put back for the walk to see it.
212            let reading = match plan.by_ref {
213                true => plan.source.borrowed(),
214                false => plan.source.clone(),
215            };
216            let (leaves, hoists, coverage) = match folding.unfold(&policy, bindings, &reading, &row)
217            {
218                Ok(read) => read,
219                // A shape the view does not represent yet — no row states
220                // this decomposition, or it states one in a form still
221                // being replaced. Step 3's remaining work rather than a
222                // disagreement.
223                Err(e) if e.is_not_yet_readable() => {
224                    skipped.push(format!("{what}: {}", e.code()));
225                    continue;
226                }
227                // Anything else is a row that is WRONG: two leaves of one
228                // name, two identities, a cycle, an accessor or an
229                // alternative the model does not have. Skipping those would
230                // let a bad row disable the check that found it.
231                Err(e) => return Err(format!("{what} has a row that cannot be read: {e}")),
232            };
233            // The walk says where it stopped; nothing is inferred from the
234            // SIZE of what came back. A measure that skips whenever one side is
235            // smaller cannot tell a shape the rows do not state yet from a
236            // defect that made the read smaller — and a defect of exactly that
237            // kind is what the previous commit fixed, invisible because it made
238            // one side smaller.
239            //
240            // So: an incomplete read is skipped and named, and a complete one
241            // is compared in full, with no exemptions.
242            if !coverage.is_complete() {
243                skipped.push(format!("{what}: {}", coverage.unread().join(", ")));
244                continue;
245            }
246            // What THIS adapter has not lowered to a binding yet. The walk read
247            // what the rows say; whether the older statement says more is a
248            // question about the declarations, which only this side can answer
249            // — and answering it here names the exact declaration rather than
250            // inferring from what came back.
251            if let Some(unlowered) = self.unlowered_parts(model, registry, recipes, &plan.source) {
252                skipped.push(format!("{what}: {unlowered}"));
253                continue;
254            }
255            compared += 1;
256            let from_row = describe(&leaves, &hoists);
257            let from_decl = describe(&plan.leaves, &plan.hoists);
258            if from_row != from_decl {
259                return Err(format!(
260                    "{what} reads back differently from its row.\nfrom the row:\n{from_row}\n\
261                     from the declarations:\n{from_decl}"
262                ));
263            }
264        }
265        // A differential that quietly compares less than it did is the failure
266        // this whole check exists to avoid, so what it did NOT compare is
267        // pinned rather than printed. The binding states the set it expects,
268        // and any growth fails the build naming what appeared.
269        //
270        // Shrinking fails too, and deliberately: each entry is a binding #701's
271        // step 3 still owes, and removing one from the expectation is how that
272        // work is recorded as done.
273        // How many decompositions the differential actually compared. The skip
274        // set names what left the comparison; this names what stayed, and one
275        // leaving the POPULATION shows up in neither.
276        if let Some(expected) = self.parity_compared {
277            if compared != expected {
278                return Err(format!(
279                    "the differential compared {compared} decomposition(s), and this binding \
280                     states {expected}. State the current count with \
281                     `.expect_parity_compared({compared})`."
282                ));
283            }
284        }
285        skipped.sort();
286        // Only where the binding stated one: a fixture exercising a single
287        // shape has no reach worth holding, and the four bindings this
288        // workspace builds all state theirs.
289        let Some(expected) = self.parity_skips.clone() else {
290            return Ok(());
291        };
292        if skipped != expected {
293            let only = |a: &[String], b: &[String]| {
294                a.iter()
295                    .filter(|x| !b.contains(x))
296                    .cloned()
297                    .collect::<Vec<_>>()
298                    .join("\n  ")
299            };
300            return Err(format!(
301                "the decompositions NOT compared against their rows have changed.\n                 newly skipped:\n  {}\nno longer skipped:\n  {}\n\n                 state the current set with `.expect_parity_skips([..])`:\n  {}",
302                only(&skipped, &expected),
303                only(&expected, &skipped),
304                skipped.join("\n  ")
305            ));
306        }
307        Ok(())
308    }
309}

311impl Declarations {
312    /// Which part of this type's decomposition declaration has no binding yet,
313    /// if any.
314    ///
315    /// `Declarations::bindings` writes a part binding for an accessor part
316    /// whose type carries a declaration of its own, and only where that part is
317    /// the value itself. Three shapes are left, each waiting on work #701's
318    /// step 3 still has: a part reached through an `Option`, which is the
319    /// `Optional` part of its decision 3; a part of a value form, whose fields
320    /// the row states as the returned struct's; and a value form that CONSUMES
321    /// its receiver, whose parts may be handed over rather than read.
322    ///
323    /// Named here rather than guessed at from the leaves, because this is where
324    /// the declaration is.
325    fn unlowered_parts(
326        &self,
327        model: &Flat,
328        registry: &dyn prebindgen_registry::Conversions,
329        recipes: &Recipes,
330        source: &prebindgen_registry::flat::TypeRef,
331    ) -> Option<&'static str> {
332        let key = source.stripped_key();
333        let decl = self.return_expand_decls.iter().find(|d| *d.key() == key)?;
334        if let [crate::jni::LocalField::Fields(form)] = decl.field_list() {
335            // A form that was handed the value may move a part out rather than
336            // read it, and which parts those are is a fact about their types
337            // that no row states. A BORROWING form has no such fact, and
338            // `Deconstruct::ValueForm` represents it completely.
339            let consuming = model
340                .function(&form.func())
341                .and_then(|f| f.params.first())
342                .is_some_and(|p| p.ty.borrow_target().is_none());
343            // Per FIELD, not per form. A form of ordinary fields is represented
344            // completely by `Deconstruct::ValueForm` — the call, the hoist, each
345            // field's reach and leaf — whether or not it consumes. Only a field
346            // that is waiting on something is.
347            for record in self.lower_value_form(registry, decl.key(), form) {
348                let core = record.ty.optional_inner().unwrap_or(&record.ty);
349                let core = core.borrow_target().unwrap_or(core);
350                let key = core.stripped_key();
351                if self.return_expand_decls.iter().any(|d| *d.key() == key) {
352                    return Some("value-form-field-with-parts");
353                }
354                // A CONSUMING form hands a handle field over rather than
355                // reading it, which the decomposition records as an identity
356                // leaf. Whether a field is a handle is this adapter's answer
357                // about its type, and the row states the same reach either way.
358                // An ordinary field has no such fact and is compared.
359                if consuming && self.types.get(&key).is_some_and(|c| c.is_opaque()) {
360                    return Some("consuming-value-form-handle-field");
361                }
362            }
363            return None;
364        }
365        for field in decl.field_list() {
366            let crate::jni::LocalField::Named(func, _) = field else {
367                continue;
368            };
369            let Some(ret) = model.function(func).map(|f| f.ret.clone()) else {
370                continue;
371            };
372            let core = ret.optional_inner().unwrap_or(&ret);
373            let core = core.borrow_target().unwrap_or(core);
374            let Some(_child) = self
375                .return_expand_decls
376                .iter()
377                .find(|d| *d.key() == core.stripped_key())
378            else {
379                continue;
380            };
381            if ret.optional_inner().is_some() {
382                return Some("optional-part");
383            }
384            // `bindings` writes a binding only where the row it would name
385            // exists. A declared type whose `parts` row this table does not
386            // state yet is the same unfinished work, seen from the other side.
387            let part = prebindgen_registry::recipe::Crossing::new(
388                core.clone(),
389                prebindgen_registry::recipe::Direction::Deconstruct,
390            );
391            if recipes
392                .key_of(&part.key(), &crate::jni::recipes::parts())
393                .is_none()
394            {
395                return Some("part-without-parts-row");
396            }
397        }
398        None
399    }
400}

402/// One decomposition, flattened far enough that two can be compared.
403fn describe(leaves: &[UnfoldLeaf], hoists: &[Hoist]) -> String {
404    use std::fmt::Write;
405    let path_of = |steps: &[prebindgen_registry::leaf::PathStep]| {
406        steps
407            .iter()
408            .map(|step| {
409                format!(
410                    "{}{}",
411                    step.ident(),
412                    if step.is_optional() { "?" } else { "" }
413                )
414            })
415            .collect::<Vec<_>>()
416            .join(".")
417    };
418    let mut out = String::new();
419    for leaf in leaves {
420        let _ = writeln!(
421            out,
422            "  {} : {} path=[{}] identity={} nullable={} source={} groups={:?}",
423            leaf.name,
424            leaf.out_ty.key(),
425            path_of(&leaf.path),
426            leaf.identity,
427            leaf.nullable,
428            source_of(&leaf.source),
429            leaf.groups
430        );
431    }
432    for hoist in hoists {
433        let _ = writeln!(
434            out,
435            "  hoist [{}] consuming={}",
436            path_of(&hoist.prefix),
437            hoist.consuming
438        );
439    }
440    out
441}

443/// How a leaf is reached, for the comparison.
444///
445/// A nested sum's payload is deliberately not distinguished from a plain reach.
446/// The two paths record it differently — `synth_value_struct_leaves` maps a
447/// rebased payload wire to `Reach`, dropping the alternative, where the view
448/// keeps `VariantField` — and which of the two is right is not this check's
449/// question. What the emitter reads off such a leaf is its `groups`, which both
450/// paths agree on and this compares.
451fn source_of(source: &LeafSource) -> String {
452    match source {
453        LeafSource::VariantField { .. } | LeafSource::Reach => "reach".to_string(),
454        LeafSource::SumTag => "tag".to_string(),
455        LeafSource::Presence => "present".to_string(),
456    }
