1//! A data class's own Kotlin **property** declarations: each field's property
2//! type, and whether holding it makes the class destructible.
3//!
4//! Not the leaf list. This module was the second derivation of that — both
5//! emitters walked the [`StructPlan`] to enumerate the slots of the
6//! `fromParts` bridge, and a standing check said the walk agreed with the
7//! registry-facing decomposition. #620 deleted it: the Rust encode and the
8//! Kotlin factory both render from the decomposition now, so the leaves have
9//! one derivation and nothing needs to agree with anything.
10//!
11//! What is left is a different question, and one plan answering it is not
12//! duplication: a property is per FIELD, where a leaf is per slot, and a field
13//! whose type is a nested class or a sum is one property over many slots.
14//! [`build_struct_plan`] classifies every field once, in one fixed priority
15//! order (projection → enum → `Option<enum>` → nested data-class → simple
16//! leaf), and [`crate::jni::render`] reads the property type and destructor
17//! off the result.
18

29/// One classified field of a [`StructPlan`]: what the Kotlin property for it
30/// is. The slots that carry it across are the decomposition's (#620).
31pub(crate) struct PlanField {
32    pub kind: PlanFieldKind,
33}
34
35/// What kind of Kotlin property a field declares. Each variant carries only
36/// what naming that property needs — how the value CROSSES is the
37/// decomposition's answer, not this one.
38pub(crate) enum PlanFieldKind {
39    /// Opaque handle / `ULong`: the property is the typed class `fqn` names,
40    /// and holding it is what can make the class destructible.
41    Projection { proj: Projection, fqn: String },
42    /// Bare `enum_class`: the property is the Kotlin enum.
43    Enum { kotlin: KtType },
44    /// `Option<enum_class>`: the same enum, nullable.
45    OptionEnum { kotlin: KtType },
46    /// Nested plain data-class: the property is the child class. `optional`
47    /// makes it nullable, and `plan` is the child's own properties, which the
48    /// emitter needs to know whether holding it is destructible.
49    Nested {
50        optional: bool,
51        /// The child's registered Kotlin FQN. `None` for an undeclared struct,
52        /// which has no class to name and so no property to declare.
53        child_fqn: Option<String>,
54        plan: StructPlan,
55    },
56    /// Data-carrying enum (`sealed_class`): the property is the sealed
57    /// interface, nullable when `optional`. `variants` are its alternatives'
58    /// own properties, for the same destructibility question.
59    Sum {
60        /// Kotlin FQN of the sealed interface.
61        kotlin_fqn: String,
62        optional: bool,
63        /// Variants in declaration order.
64        variants: Vec<SumPlanVariant>,
65    },
66    /// Anything else: the property is the Kotlin type its output conversion
67    /// names.
68    Leaf {
69        kotlin: KtType,
70        /// Kotlin-side `?` (an `Option` field whose wire is object-shaped).
71        nullable: bool,
72    },
73}

75/// One alternative of a [`PlanFieldKind::Sum`].
76pub(crate) struct SumPlanVariant {
77    /// This variant's payload, in declaration order. Empty for a unit
78    /// variant — the group that contributes nothing but its tag.

89/// Classify `s`'s fields into their Kotlin property declarations. `None` aborts
90/// the whole-value bridge — an unresolved field converter, or a missing Kotlin
91/// name — which is the same answer the decomposition gives for such a field,
92/// so the class and its leaves decline together.
93pub(crate) fn build_struct_plan(
94    ext: &Declarations,
95    registry: &impl Conversions,
96    s: &prebindgen_registry::flat::Struct,
97    depth: usize,
98) -> Option<StructPlan> {
99    assert!(
100        depth <= 16,
101        "struct fromParts plan: recursion too deep at struct `{}` (cyclic data_class?)",
102        s.name
103    );
104    let mut fields: Vec<PlanField> = Vec::new();
105    for field in &s.fields {
106        // A tuple struct is an `Extern` in the model, never a `Struct`, so a
107        // nameless field cannot reach here.
108        let fname = field.name.as_ref()?.clone();
109        let owner = format!("{}.{}", s.name, fname);
110        let kind = classify_field(ext, registry, &field.ty, &owner, depth)?;
111        fields.push(PlanField { kind });
112    }
113    Some(StructPlan { fields })
114}

116impl Declarations {
117    /// One data class's Kotlin property declarations, for the class emitter —
118    /// the only production reader left, now that the encode and the factory
119    /// take their leaves from the decomposition. During resolution it is
120    /// memoized; after resolution the lookup is served exclusively by
121    /// [`crate::jni::generation::JniGenerationPlan`].
122    pub(crate) fn struct_plan(
123        &self,
124        registry: &impl Conversions,
125        s: &prebindgen_registry::flat::Struct,
126        depth: usize,
127    ) -> Option<std::rc::Rc<StructPlan>> {
128        let key = s.type_ref().key();
129        if let Some(generation) = &self.generation {
130            return generation.struct_plan(&key);
131        }
132        if let Some(hit) = self.struct_plans.borrow().get(&key) {
133            return hit.clone();
134        }
135        let plan = build_struct_plan(self, registry, s, depth).map(std::rc::Rc::new);
136        self.struct_plans.borrow_mut().insert(key, plan.clone());
137        plan
138    }
139}

141/// Classify ONE value position — a struct field or a sum's variant payload —
142/// into the Kotlin property it declares. Both callers go through here so a
143/// payload and a struct field of the same Rust type declare the same type; a
144/// second classification walk beside it is the drift this exists to prevent.
145///
146/// `owner` is the dotted path used in diagnostics (`Config.mode`,
147/// `Reading::Exact.v0`).
148pub(crate) fn classify_field(
149    ext: &Declarations,
150    registry: &impl Conversions,
151    reading: &prebindgen_registry::flat::TypeRef,
152    owner: &str,
153    depth: usize,
154) -> Option<PlanFieldKind> {
155    // The **reading**, not a spelling. Every layer question below is answered
156    // from `kind` and cannot fail: holding a `TypeRef` is proof the model
157    // classified this type. Taking a `syn::Type` meant asking the registry per
158    // question, and a type it had never seen answered "no layer" rather than
159    // saying so — which is the missing `?` of #273 waiting to happen again.
160
161    // A sum is classified FIRST, because it is the one kind with no converter
162    // of its own: it crosses as a tag plus one leaf group per variant, never
163    // as a single wire, which is why `sealed_class` types are declared
164    // boundary-only. Demanding an output entry before this point would send
165    // every sum-typed field down the `None` path and fail the whole parent's
166    // plan with an unresolved-converter error naming the wrong thing.
167    // `Vec` is peeled alongside `Option` here purely to CLASSIFY: `type_kind`
168    // answers about a bare ident, so it reports `Vec<Reading>` as `Other` and
169    // a rejection guarded on the unpeeled type could never fire. Peeling first
170    // is what makes the `Vec<sum>` error reachable at all.
171    // Every layer question below is the MODEL's, asked once: a field spelled
172    // `Box<Option<T>>` is `Optional` and must classify, nest and render exactly
173    // as `Option<T>` does. Peeling by path segment answered "not optional" for
174    // it, and the seven peels in this function would then disagree with each
175    // other about the same field (#273).
176    let optional_inner = reading.optional_inner();
177    let bare_ref = optional_inner.unwrap_or(reading);
178    let seq_elem = bare_ref.sequence_elem();
179    let core = seq_elem.unwrap_or(bare_ref);
180    if matches!(ext.type_kind(registry, &core.key()), TypeKind::Sum) {
181        // A `Vec` of tag-gated groups has variable arity, exactly like a `Vec`
182        // of nested data classes — the flattened bridge is fixed-layout by
183        // construction.
184        if seq_elem.is_some() {
185            panic!(
186                "fromParts bridge: `Vec<{}>` sealed-class field (`{owner}`) is not supported \
187                 (variable arity)",
188                core,
189            );
190        }
191        return sum_plan_kind(
192            ext,
193            registry,
194            bare_ref,
195            owner,
196            optional_inner.is_some(),
197            depth,
198        );
199    }
200
201    let field_entry = ext.out_frag(reading)?;
202    field_entry.activate();
203
204    {
205        // Projection leaf (opaque handle / `ULong`).
206        if let Some(proj) = field_entry.metadata.projection.clone() {
207            if matches!(proj.strategy, FoldStrategy::Iterable(_)) {
208                panic!(
209                    "fromParts bridge: collection (`Vec<projection>`) field `{owner}` is not \
210                     supported — add array codegen to lift this guard"
211                );
212            }
213            let fqn = projection_leaf_kt(ext, &proj)?.to_string();
214            return Some(PlanFieldKind::Projection { proj, fqn });
215        }
216        // Enum leaf, bare or under `Option` — asked ONCE, of the model, and of
217        // the already-peeled reading beside us.
218        //
219        // It used to ask `is_kotlin_enum` twice, of two spellings. That answers
220        // about the WRAPPER: `builder.rs` documents `Box<Priority>` as `false`
221        // for it and `true` for the reading form, so a wrapped enum field fell
222        // through to the plain-leaf arm and rendered as its wire instead of the
223        // Kotlin enum class — the #273 family, output-side. `flat_input.rs` had
224        // already moved to the reading; this is the other half of the same
225        // question finally giving the same answer.
226        //
227        // Optionality stays the CALLER's fact rather than the probe's:
228        // `enum_probe` peels `Option` as well as borrows, so asking it about
229        // the unpeeled reading would make `Priority` and `Option<Priority>`
230        // indistinguishable and collapse the two arms into one.
231        if ext.is_kotlin_enum_reading(bare_ref) {
232            return match optional_inner {
233                None => {
234                    let kotlin = field_entry.metadata.kotlin_name.clone()?;
235                    Some(PlanFieldKind::Enum { kotlin })
236                }
237                Some(inner) => {
238                    let kotlin = ext.out_frag(inner)?.metadata.kotlin_name.clone()?;
239                    Some(PlanFieldKind::OptionEnum { kotlin })
240                }
241            };
242        }
243        // Nested plain data-class (optionally under `Option`).
244        //
245        // A `Vec<data class>` does NOT arrive here and needs no guard of its
246        // own (#217). `type_kind` answers `DataStruct` only for a key that is a
247        // single identifier, which a `Vec<_>` key never is — so this branch
248        // cannot be entered with a sequence in hand, and the field falls
249        // through to the simple-leaf arm below. That is the right answer rather
250        // than a missed one: the field stays ONE slot whose own converter is
251        // the element's fixed folder, so the bridge keeps its fixed slot count
252        // and the elements still cross as raw leaves. #217 expected this to
253        // need array codegen — a count slot plus a per-element sub-plan in all
254        // three producers — and it does not, because the sequence never has to
255        // enter the fixed layout at all.
256        //
257        // The `Vec<sum>` refusal above is a different question and stays: a sum
258        // has no converter of its own, so there is no single slot to fall
259        // through to.
260        let inner_ty = bare_ref;
261        if let TypeKind::DataStruct { st, cfg } = ext.type_kind(registry, &inner_ty.key()) {
262            let child_fqn = cfg
263                .and_then(|c| c.name_spec.as_ref())
264                .map(|s| ext.fqn_of(s));
265            let plan = build_struct_plan(ext, registry, st, depth + 1)?;
266            return Some(PlanFieldKind::Nested {
267                optional: optional_inner.is_some(),
268                child_fqn,
269                plan,
270            });
271        }
272        // Simple leaf: its Kotlin property type, and whether an `Option` over
273        // an object-shaped wire makes that property nullable. The slot's own
274        // descriptor and binding form are the encoder's, and the encoder reads
275        // them off the decomposition now (#619).
276        let wire = field_entry.wire.clone();
277        let kotlin = field_entry.metadata.kotlin_name.clone()?;
278        let nullable = optional_inner.is_some() && !is_jni_primitive(&wire);
279        Some(PlanFieldKind::Leaf { kotlin, nullable })
280    }
281}

283impl PlanFieldKind {
284    /// The Kotlin type of the `data class` **constructor property** this field
285    /// becomes.
286    ///
287    /// The class declaration reads it from here, and the `fromParts` factory
288    /// and the Rust encoder read the matching leaf types from the
289    /// decomposition — one derivation each, for two different questions. #156
290    /// is why the declaration is not a third: deriving a property's type
291    /// separately is what let it disagree with its own factory parameter.
292    ///
293    /// `owner` is the dotted path used in diagnostics.
294    pub(crate) fn property_type(&self, owner: &str) -> KtType {
295        match self {
296            // A projection's typed surface is its folded shape over the leaf
297            // class (`ZKeyExpr?`, `List<ZKeyExpr>`, `ULong`), which the plan
298            // already resolved into `fqn`.
299            PlanFieldKind::Projection { proj, fqn, .. } => {
300                handle_kt_type(&proj.strategy, &KtType::cls(fqn))
301            }
302            PlanFieldKind::Enum { kotlin, .. } => kotlin.clone(),
303            PlanFieldKind::OptionEnum { kotlin, .. } => kotlin.clone().nullable(),
304            PlanFieldKind::Nested {
305                optional,
306                child_fqn,
307                ..
308            } => {
309                let fqn = child_fqn.as_ref().unwrap_or_else(|| {
310                    panic!(
311                        "data class property `{owner}`: nested data-class field has no \
312                         registered Kotlin class — declare the child type in a package"
313                    )
314                });
315                let t = KtType::cls(fqn);
316                if *optional {
317                    t.nullable()
318                } else {
319                    t
320                }
321            }
322            PlanFieldKind::Sum {
323                kotlin_fqn,
324                optional,
325                ..
326            } => {
327                let t = KtType::cls(kotlin_fqn);
328                if *optional {
329                    t.nullable()
330                } else {
331                    t
332                }
333            }
334            // `nullable` is the plan's own rule — an `Option` field whose wire
335            // is object-shaped. An `Option` over a PRIMITIVE wire stays
336            // non-null, because the encoder passes the bare primitive with a
337            // sentinel and the JVM slot must match (`J`, not `Ljava/lang/Long;`).
338            PlanFieldKind::Leaf {
339                kotlin, nullable, ..
340            } => {
341                if *nullable {
342                    kotlin.clone().nullable()
343                } else {
344                    kotlin.clone()
345                }
346            }
347        }
348    }
349
350    /// The close strategy when this field **reaches** an owned native handle,
351    /// so the class implements `AutoCloseable` and `close()` walks it.
352    ///
353    /// Only an **owned** `Handle` projection is a handle: a `ULong` owns
354    /// nothing, and a borrowed handle is not this object's to release. But
355    /// reaching one is not the same as *being* one — a handle held inside a sum
356    /// payload or a nested data class is just as much this object's to release,
357    /// and used to fall through to `None` (#218). That made ownership depend on
358    /// how the field happened to be spelled: swapping a handle field for an
359    /// enum carrying that handle silently moved the free onto the consumer,
360    /// with nothing in the generated Kotlin saying so.
361    ///
362    /// The wrapped cases fold as `Base`/`Optional` over the field itself rather
363    /// than over the handle inside it, because the field's own generated type
364    /// is `AutoCloseable` too — a sum's `close()` is the `when` over its
365    /// alternatives (`Declarations::build_sealed_class`), a nested data
366    /// class's is this same cascade one level down. So every
367    /// container emits the same plain `field.close()`, and the walk into the
368    /// wrapper lives once, in the wrapped type, instead of at each use site.
369    pub(crate) fn destructible(&self) -> Option<FoldStrategy> {
370        match self {
371            PlanFieldKind::Projection { proj, .. }
372                if matches!(proj.kind, ProjectionKind::Handle) && proj.owned =>
373            {
374                Some(proj.strategy.clone())
375            }
376            // `sequence: false` is `classify_field`'s guarantee, not an
377            // assumption: it rejects `Vec<sum>` and `Vec<data class>` outright,
378            // so a classified field of either kind is never a sequence.
379            PlanFieldKind::Sum {
380                optional, variants, ..
381            } if variants.iter().any(SumPlanVariant::destructible) => {
382                Some(whole_value_close(*optional, false))
383            }
384            PlanFieldKind::Nested { optional, plan, .. } if plan.destructible() => {
385                Some(whole_value_close(*optional, false))
386            }
387            _ => None,
388        }
389    }
390}

409/// The fold of a wrapper that closes itself: the value is closed as a whole,
410/// `?.`-guarded when it is optional and `forEach`-ed when it is a sequence.
411/// Shared by the two forms of the reaches-a-handle question below so they
412/// cannot answer differently.
413///
414/// [`NullableKind::Boxed`] is not a guess here. The receiver is always a
415/// generated Kotlin *reference* — a sum or a data class — whose absent form is
416/// a JVM null; a niche encoding is a wire fact of a handle projection, and
417/// those come back carrying their own strategy without passing through this.
418fn whole_value_close(optional: bool, sequence: bool) -> FoldStrategy {
419    let mut fold = FoldStrategy::Base;
420    if sequence {
421        fold = FoldStrategy::Iterable(Box::new(fold));
422    }
423    if optional {
424        fold = FoldStrategy::Optional(NullableKind::Boxed, Box::new(fold));
425    }
426    fold
427}

429/// How to close a value of this type, or `None` when it **reaches** no owned
430/// native handle and so has nothing to release —
431/// [`PlanFieldKind::destructible`]'s question asked of a type rather than of an
432/// already-classified plan field.
433///
434/// Two callers hold a [`TypeRef`](prebindgen_registry::flat::TypeRef) and no
435/// plan: the sealed-interface emitter, deciding whether a sum is
436/// `AutoCloseable` and what each variant class's `close()` body does; and the
437/// callback interface builder, deciding whether a reassembled whole value is
438/// the proxy's to close after `run`.
439///
440/// The two forms must **agree wherever both answer**, and that is a tested
441/// invariant, not a structural one: `a_types_close_answer_matches_its_plans`
442/// asserts `type_close_strategy(ty).is_some() == plan(ty).destructible()` over
443/// every field of every declared shape in a set covering all the ways one can
444/// reach a handle. It is worth pinning because the walks are not identical.
445/// Two places they differ:
446///
447/// * **Order.** [`classify_field`] classifies a `Sum` *before* it consults
448///   `output_entry` (deliberately — see its comment); this asks `output_entry`
449///   first. Safe only while sums carry no converter of their own, which is a
450///   precondition rather than a construction.
451/// * **Totality.** A field [`classify_field`] refuses collapses the whole
452///   struct's plan to `None`, while this refuses nothing. So on a subtree the
453///   bridge rejects, the two answer differently *by design* — the plan
454///   builders are what diagnose those, with the path, and "is there anything
455///   to close" still has a defensible answer for every type.
456///
457/// A disagreement costs a Kotlin compile error in one direction and a silent
458/// leak — #218 again, at this seam — in the other.
459///
460/// `depth` bounds the same recursion `build_struct_plan` and `sum_plan_kind`
461/// bound, and asserts on the same bound rather than answering: a cycle deep
462/// enough to trip this is a declaration the plan builders already refuse
463/// loudly, and returning `None` for it would report "nothing to close" — the
464/// leak direction — for a shape nobody can compile anyway.
465pub(crate) fn type_close_strategy(
466    ext: &Declarations,
467    registry: &impl Conversions,
468    ty: &prebindgen_registry::flat::TypeRef,
469    depth: usize,
470) -> Option<FoldStrategy> {
471    assert!(
472        depth <= 16,
473        "close-strategy walk: recursion too deep at type `{}` (cyclic data_class?)",
474        ty
475    );
476    // An owned `Handle` projection is the one thing that actually owns
477    // something: a `ULong` owns nothing, and a borrowed handle is not ours to
478    // release. Asked of the whole reading, so the `Option`/`Vec` folds the
479    // projection carries come back in its own strategy.
480    if let Some(proj) = ext.out_frag(ty).and_then(|e| e.metadata.projection.clone()) {
481        return (matches!(proj.kind, ProjectionKind::Handle) && proj.owned)
482            .then(|| proj.strategy.clone());
483    }
484    // Peel the layers the model names, exactly as `classify_field` does: what
485    // a `Box<Option<T>>` reaches is what `T` reaches.
486    let bare = ty.optional_inner().unwrap_or(ty);
487    let core = bare.sequence_elem().unwrap_or(bare);
488    let reaches = match ext.type_kind(registry, &core.key()) {
489        TypeKind::Sum => core
490            .key()
491            .ident()
492            .and_then(|ident| registry.flat().declared_type(&ident))
493            .is_some_and(|ty| match ty {
494                prebindgen_registry::flat::Type::Variant(sum) => {
495                    sum.alternatives.iter().any(|alt| {
496                        alt.fields
497                            .iter()
498                            .any(|f| type_close_strategy(ext, registry, &f.ty, depth + 1).is_some())
499                    })
500                }
501                _ => false,
502            }),
503        TypeKind::DataStruct { st, .. } => st
504            .fields
505            .iter()
506            .any(|f| type_close_strategy(ext, registry, &f.ty, depth + 1).is_some()),
507        TypeKind::Handle | TypeKind::Enum | TypeKind::Other => false,
508    };
509    // Put back exactly the layers peeled above. `reaches` was answered about
510    // the ELEMENT, so a `Vec<sum-that-reaches-a-handle>` must close each
511    // element — `close()` on the `List` itself would not compile. The field
512    // bridge rejects that shape, but this predicate exists precisely for the
513    // callers that hold no plan and so meet no rejection.
514    reaches.then(|| {
515        whole_value_close(
516            ty.optional_inner().is_some(),
517            bare.sequence_elem().is_some(),
518        )
519    })
520}

522/// Build the [`PlanFieldKind::Sum`] for a `sealed_class`-declared enum: one
523/// leaf group per variant, each payload classified through
524/// [`classify_field`].
525///
526/// Recursion is bounded by this function's own depth guard (see below) — a
527/// sum reaching its own type has no `jobject_input`-style escape hatch, since
528/// the flatten plan is finite by construction.
529///
530/// `None` propagates the resolver's **deferral** protocol: a payload whose
531/// converter has not resolved *yet* means "retry on the next fixed-point
532/// iteration", exactly as [`classify_field`] signals it for a struct field.
533/// Panicking there instead would turn a transient state into a build failure
534/// whenever a payload's converter happened to resolve later than this plan
535/// was first attempted.
536fn sum_plan_kind(
537    ext: &Declarations,
538    registry: &impl Conversions,
539    ty: &prebindgen_registry::flat::TypeRef,
540    owner: &str,
541    optional: bool,
542    depth: usize,
543) -> Option<PlanFieldKind> {
544    // Sum expansion needs its OWN depth guard. A sum whose payload is a sum
545    // never passes through `build_struct_plan`, so that function's assert —
546    // the only one on this recursion before now — cannot see a chain made
547    // purely of sums. Rust's sizedness rules make an unindirected cycle
548    // impossible to declare, and every indirection either classifies as
549    // `Other` (`Box<E>` is not a bare ident) or is already rejected
550    // (`Vec<E>`), so this is defence in depth rather than a reachable path
551    // today. It costs one comparison and makes the bound true for every
552    // future shape instead of true-by-accident.
553    assert!(
554        depth <= 16,
555        "fromParts bridge: sealed-class expansion too deep at `{owner}` (recursive sum?)"
556    );
557    // The key's ident: a sum is a declared name, and the key is that name when
558    // the key is one identifier.
559    let ident = ty.key().ident().unwrap_or_else(|| {
560        panic!("fromParts bridge: sealed-class field `{owner}` is not a path type")
561    });
562    // The sum as the MODEL holds it: its alternatives' payloads are `TypeRef`s
563    // already, so classifying one asks nothing and cannot be asked about a type
564    // the model never saw. One lookup, not two — the `enum_item` that used to
565    // sit beside this only fed a `SumSpec` of what the element already says.
566    let Some(prebindgen_registry::flat::Type::Variant(sum)) = registry.flat().declared_type(&ident)
567    else {
568        panic!("fromParts bridge: sealed-class field `{owner}`: `{ident}` is not an indexed sum")
569    };
570    let key = TypeKey::from_ident(&ident);
571    let cfg = ext
572        .types
573        .get(&key)
574        .unwrap_or_else(|| panic!("fromParts bridge: `{ident}` is not declared"));
575    // Asked for the diagnostic, not for a value: a `sealed_class!` declaration
576    // is what makes the alternatives' properties meaningful, and a field typed
577    // by an enum that has none is a build error rather than a silent decline.
578    cfg.sum()
579        .unwrap_or_else(|| panic!("fromParts bridge: `{ident}` is not a sealed class"));
580    let kotlin_fqn = cfg
581        .name_spec
582        .as_ref()
583        .map(|s| ext.fqn_of(s))
584        .unwrap_or_else(|| panic!("fromParts bridge: sealed class `{ident}` has no Kotlin name"));
585
586    let mut variants: Vec<SumPlanVariant> = Vec::new();
587    for alt in &sum.alternatives {
588        let mut fields: Vec<SumPlanField> = Vec::new();
589        for field in &alt.fields {
590            let member = field.member();
591            let prop = sum_field_prop_name(&member);
592            let owner = format!("{ident}::{}.{prop}", alt.name);
593            // `?` — a payload whose converter has not resolved yet defers the
594            // whole plan to the next iteration, it does not fail the build.
595            let kind = classify_field(ext, registry, &field.ty, &owner, depth + 1)?;
596            fields.push(SumPlanField { kind });
597        }
598        variants.push(SumPlanVariant { fields });
599    }
600
601    Some(PlanFieldKind::Sum {
602        kotlin_fqn,
603        optional,
604        variants,
605    })
606}

608/// Kotlin property name of one sum payload field — a named field keeps its
609/// camelCased name, a tuple field becomes `v0`, `v1`. Must agree with the
610/// sealed-interface emitter, which is why both call this.
611///
612/// Takes the **member**, which is the whole of what the name depends on: every
613/// caller holds a `flat::Field` and asks `Field::member()`. It took a
614/// `types_util::SumField` when a second description of a sum still existed
615/// beside the model's (#289).
616pub(crate) fn sum_field_prop_name(member: &syn::Member) -> String {
617    match member {
618        syn::Member::Named(id) => mangle_kotlin_ident(&kt_snake_to_camel(&id.to_string())),
619        syn::Member::Unnamed(i) => format!("v{}", i.index),
620    }
621}

623/// The wire tag of one alternative: its declaration-order index, as the `jint`
624/// the selector leaf carries.
625///
626/// One place, because the tag has to agree in three: the leaf's `group`, the
627/// Kotlin `when` arm, and the Rust `match` arm. Three separate `as i32` casts
628/// agreed by coincidence rather than by construction.
629///
630/// Deliberately **not** a checked conversion. `usize` → `i32` can truncate in
631/// general, but not here: the index counts alternatives of one enum, and an
632/// enum with `i32::MAX` variants is not a thing rustc can be handed. A
633/// `try_from(..).expect(..)` would put a panic in the working path for a state
634/// the compiler cannot produce, which is the shape this crate avoids.
635pub(crate) fn sum_tag(alt: &prebindgen_registry::flat::Alternative) -> i32 {
636    alt.index as i32
637}

639/// Slot-name fragment for one variant field: `<variantCamel>_<prop>`. Keyed
640/// on the **Kotlin** variant name so a `variant!(V).name(...)` rename carries
641/// through to the slots.
642pub(crate) fn sum_slot_fragment(kotlin_variant: &str, prop: &str) -> String {
643    let mut chars = kotlin_variant.chars();
644    let head: String = match chars.next() {
645        Some(c) => c.to_lowercase().collect(),
646        None => String::new(),
647    };
648    format!("{head}{}_{prop}", chars.as_str())
649}
