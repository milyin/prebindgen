1//! Constructor expansion — fold a value's construction into the wire
2//! signature of the function that consumes it, so the foreign side builds
3//! the value and calls the function in a single FFI crossing.
4//!
5//! A *constructor* is any `#[prebindgen]` function `f(p0, …) -> T` (or
6//! `-> Result<T, E>`) that builds a target type `T`. A type's input flatten
7//! (a type-level `expand_param!` `.variant*` list, or the per-fn
8//! `.expand_param(param, …)` override) replaces a parameter of that type —
9//! in the generated foreign signature only — with the constructor's inputs,
10//! flattened. The generated wrapper decodes those inputs, runs the
11//! constructor Rust-side (the **fold**), and passes the built value to the
12//! underlying call.
13//!
14//! * **One `Ctor` variant** (no identity): the parameter becomes `f`'s
15//!   parameters directly (no selector) — the plain "single" form.
16//! * **Two or more variants** (or an identity arm): the parameter becomes a
17//!   runtime selector (`i32`) plus one `Option`-wrapped input group per variant.
18//!   The identity variant passes an already-built `T` straight through.
19//!
20//! Everything here is **language-agnostic**: the fold is pure Rust and the
21//! per-leaf wire encode/decode is delegated to the adapter's existing
22//! converters. Resolution turns the declarations into [`FoldPlan`]s (stored on
23//! the registry, keyed by `(fn, param)`) and registers each leaf type as a
24//! required input so the resolver produces its converter. [`emit_fold`]
25//! emits the dispatch expression at the parameter-emission site.
26

39/// Emit the fold expression for an expanded parameter. `leaf_locals` are the
40/// already-decoded Rust locals (1:1 with `plan.leaves`); `qualify` maps a
41/// constructor ident to its call path (e.g. prefixing the source module).
42///
43/// The returned expression has type `Result<<shaped> plan.target, String>`
44/// (`Result<Target>`, `Result<Option<Target>>`, …). The adapter routes its
45/// `Err(String)` through its own error channel. Folds the [`FoldShape`] layers
46/// top-down over one shared core construct — the value
47/// analog of how `Option<_>`/`Vec<_>` wrappers compose at the wire.
48pub fn emit_fold(
49    plan: &FoldPlan,
50    leaf_locals: &[syn::Ident],
51    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
52) -> syn::Expr {
53    fold_shape(&plan.shape, plan, leaf_locals, None, qualify)
54}
55
56/// Recurse over one [`FoldShape`] layer. `bound` is `Some(var)` when an
57/// enclosing `Optional`/`Iterable` layer has unwrapped the structured leaf and
58/// bound its element to `var` — the inner construct then builds from `var`
59/// instead of reading `leaf_locals`.
60fn fold_shape(
61    shape: &FoldShape,
62    plan: &FoldPlan,
63    leaf_locals: &[syn::Ident],
64    bound: Option<&syn::Ident>,
65    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
66) -> syn::Expr {
67    match shape {
68        FoldShape::Base => emit_core_construct(plan, leaf_locals, bound, qualify),
69        FoldShape::Optional((), inner) => {
70            if let Some(sidx) = plan.selector {
71                // Combined-selector dispatch under `Optional`: the selector
72                // ALSO encodes absence — `-1` = `None`, `0..n-1` = the taken
73                // arm (dispatched by the shared construct core; an out-of-range
74                // selector still hits its `Err` default arm).
75                let sel_local = &leaf_locals[sidx];
76                let inner_expr = emit_core_construct(plan, leaf_locals, None, qualify);
77                syn::parse_quote!(if #sel_local < 0 {
78                    ::core::result::Result::Ok(::core::option::Option::None)
79                } else {
80                    (#inner_expr).map(::core::option::Option::Some)
81                })
82            } else if let Some(pidx) = plan.present {
83                // Multi-arg: an explicit `present: bool` flag decides presence;
84                // the construct reads its plain arg leaves directly (`bound =
85                // None`), the flag leaf is consumed only by this `if`.
86                let present_local = &leaf_locals[pidx];
87                let inner_expr = emit_core_construct(plan, leaf_locals, None, qualify);
88                syn::parse_quote!(if #present_local {
89                    (#inner_expr).map(::core::option::Option::Some)
90                } else {
91                    ::core::result::Result::Ok(::core::option::Option::None)
92                })
93            } else {
94                // Single-arg: presence rides the sole shaped leaf's `Option`.
95                // The structured value is the enclosing bound var, or — at the
96                // top — that leaf's decoded local (`leaf_locals[0]`).
97                let value = bound.unwrap_or(&leaf_locals[0]);
98                let inner_ident = ident("__inner");
99                let inner_expr = fold_shape(inner, plan, leaf_locals, Some(&inner_ident), qualify);
100                syn::parse_quote!(match #value {
101                    ::core::option::Option::Some(#inner_ident) => {
102                        (#inner_expr).map(::core::option::Option::Some)
103                    }
104                    ::core::option::Option::None => {
105                        ::core::result::Result::Ok(::core::option::Option::None)
106                    }
107                })
108            }
109        }
110        FoldShape::Iterable(inner) => {
111            let value = bound.unwrap_or(&leaf_locals[0]);
112            let elem_ident = ident("__elem");
113            let inner_expr = fold_shape(inner, plan, leaf_locals, Some(&elem_ident), qualify);
114            syn::parse_quote!(
115                #value
116                    .into_iter()
117                    .map(|#elem_ident| #inner_expr)
118                    .collect::<::core::result::Result<::std::vec::Vec<_>, _>>()
119            )
120        }
121    }
122}

124/// Emit the innermost construct → `Result<Target, String>`. With `bound =
125/// Some(v)` (under an `Optional`/`Iterable` layer ⇒ single, single-arg ctor)
126/// the ctor is applied to `v`; with `bound = None` (top level) it reads the
127/// leaves — a single constructor (any arity) or a combined-selector dispatch.
128fn emit_core_construct(
129    plan: &FoldPlan,
130    leaf_locals: &[syn::Ident],
131    bound: Option<&syn::Ident>,
132    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
133) -> syn::Expr {
134    if let Some(v) = bound {
135        // Shaped construct: a single, single-arg constructor applied to the
136        // unwrapped element. (`apply` guarantees this shape — never identity,
137        // never combined, never multi-arg under a shape layer.)
138        let var = &plan.variants[0];
139        let func = var
140            .ctor
141            .as_ref()
142            .expect("shaped expansion is single-constructor (never identity)");
143        return ctor_call_result(&qualify(func), std::slice::from_ref(v), var.fallible);
144    }
145    emit_dispatch(plan.selector, &plan.variants, leaf_locals, qualify)
146}

148/// Emit a construct dispatch → `Result<Target, String>`: a single variant
149/// applied directly (no selector), or a `match` over the selector leaf. Shared
150/// by the top-level [`emit_core_construct`] and each nested [`emit_build`].
151fn emit_dispatch(
152    selector: Option<usize>,
153    variants: &[FoldVariant],
154    leaf_locals: &[syn::Ident],
155    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
156) -> syn::Expr {
157    match selector {
158        None => variant_result_expr(
159            &variants[0],
160            leaf_locals,
161            qualify,
162            /*dispatched=*/ false,
163        ),
164        Some(si) => {
165            let sel = &leaf_locals[si];
166            let arms: Vec<TokenStream> = variants
167                .iter()
168                .enumerate()
169                .map(|(vi, v)| {
170                    let lit = vi as i32;
171                    let body =
172                        variant_result_expr(v, leaf_locals, qualify, /*dispatched=*/ true);
173                    quote!(#lit => #body,)
174                })
175                .collect();
176            syn::parse_quote!({
177                match #sel {
178                    #(#arms)*
179                    __sel => ::core::result::Result::Err(::std::format!(
180                        "invalid constructor selector: {}",
181                        __sel
182                    )),
183                }
184            })
185        }
186    }
187}

189/// Emit a nested recursive-input build → `Result<SubTarget, String>` (the dual
190/// of [`emit_core_construct`] for a [`FoldArg::Build`] parameter).
191fn emit_build(
192    b: &FoldBuild,
193    leaf_locals: &[syn::Ident],
194    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
195) -> syn::Expr {
196    emit_dispatch(b.selector, &b.variants, leaf_locals, qualify)
197}

199/// Build a `Result<Target, String>` expression for one core variant. When
200/// `dispatched` (a combined-selector arm), the variant's input leaves are
201/// `Option<_>` — only the selected arm's inputs are present — so they are
202/// unwrapped (a missing input yields `Err`); otherwise they are passed
203/// directly. (This `Option`-ness is *selector presence*, distinct from
204/// [`FoldShape::Optional`], which is whole-param presence handled by the
205/// enclosing fold.)
206fn variant_result_expr(
207    v: &FoldVariant,
208    leaf_locals: &[syn::Ident],
209    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
210    dispatched: bool,
211) -> syn::Expr {
212    // A Leaf arg's decoded local. Identity arms and combined-dispatched arms are
213    // Leaf-only (recursive `Build` args appear only in a non-dispatched single
214    // constructor — `build_arg` rejects nesting under a dispatched variant).
215    let leaf = |a: &FoldArg| -> &syn::Ident {
216        match a {
217            FoldArg::Leaf(i, _) => &leaf_locals[*i],
218            FoldArg::Build(_) => {
219                unreachable!("recursive Build arg only in a non-dispatched single constructor")
220            }
221        }
222    };
223
224    match &v.ctor {
225        None => {
226            // Identity: the sole input is the target value (or a borrow of it
227            // that we clone, for `&T` consumers — preserving the caller's handle).
228            let loc = leaf(&v.inputs[0]);
229            // `&*__v` derefs through whatever the borrow leaf decoded to (a
230            // plain `&T`, or an adapter smart-pointer like jnigen's
231            // `OwnedObject<T>`) down to `T`, then clones — keeping the caller's
232            // handle alive without the core knowing the adapter's borrow type.
233            let some_val: syn::Expr = if v.clone {
234                syn::parse_quote!(::core::result::Result::Ok(::core::clone::Clone::clone(
235                    &*__v
236                )))
237            } else {
238                syn::parse_quote!(::core::result::Result::Ok(__v))
239            };
240            if dispatched {
241                syn::parse_quote!(match #loc {
242                    ::core::option::Option::Some(__v) => #some_val,
243                    ::core::option::Option::None => ::core::result::Result::Err(
244                        ::std::string::String::from("identity variant value missing")
245                    ),
246                })
247            } else if v.clone {
248                syn::parse_quote!(::core::result::Result::Ok(::core::clone::Clone::clone(&*#loc)))
249            } else {
250                syn::parse_quote!(::core::result::Result::Ok(#loc))
251            }
252        }
253        Some(func) => {
254            let path = qualify(func);
255            if dispatched {
256                // Combined arm — Leaf-only inputs. Selector-presence-wrapped
257                // inputs are unwrapped (missing ⇒ `Err`); **passthrough**
258                // inputs (constructor args that are themselves `Option<…>`)
259                // pass their decoded local directly — `None` is a legitimate
260                // value for the taken arm.
261                let mut wrapped_locals: Vec<&syn::Ident> = Vec::new();
262                let mut wrapped_binds: Vec<syn::Ident> = Vec::new();
263                let mut call_args: Vec<syn::Expr> = Vec::new();
264                for (i, a) in v.inputs.iter().enumerate() {
265                    let loc = leaf(a);
266                    if matches!(a, FoldArg::Leaf(_, true)) {
267                        call_args.push(syn::parse_quote!(#loc));
268                    } else {
269                        let b = ident(&format!("__p{}", i));
270                        wrapped_locals.push(loc);
271                        wrapped_binds.push(b.clone());
272                        call_args.push(syn::parse_quote!(#b));
273                    }
274                }
275                let call = ctor_call_result(&path, &call_args, v.fallible);
276                let missing = quote!(::core::result::Result::Err(::std::string::String::from(
277                    "constructor variant input missing"
278                )));
279                match wrapped_locals.len() {
280                    // All-passthrough arm: the selector alone decides; call directly.
281                    0 => call,
282                    1 => {
283                        // `match a { Some(p0) => <call>, None => Err }`
284                        let loc = wrapped_locals[0];
285                        let p0 = &wrapped_binds[0];
286                        syn::parse_quote!(match #loc {
287                            ::core::option::Option::Some(#p0) => #call,
288                            ::core::option::Option::None => #missing,
289                        })
290                    }
291                    _ => {
292                        // `match (a, b, …) { (Some(p0), Some(p1), …) => <call>, _ => Err }`
293                        let some_pats: Vec<TokenStream> = wrapped_binds
294                            .iter()
295                            .map(|b| quote!(::core::option::Option::Some(#b)))
296                            .collect();
297                        syn::parse_quote!(match ( #(#wrapped_locals),* ) {
298                            ( #(#some_pats),* ) => #call,
299                            _ => #missing,
300                        })
301                    }
302                }
303            } else if v.inputs.iter().all(|a| matches!(a, FoldArg::Leaf(..))) {
304                // Non-dispatched, flat (no recursion): call directly — identical
305                // to the pre-recursion form.
306                let args: Vec<&syn::Ident> = v.inputs.iter().map(&leaf).collect();
307                ctor_call_result(&path, &args, v.fallible)
308            } else {
309                // Non-dispatched with ≥1 recursive `Build` arg: bind each arg
310                // (Leaf = the decoded value; Build = the nested construct,
311                // `?`-unwrapped) in an IIFE that provides the `Result` context.
312                let mut stmts: Vec<TokenStream> = Vec::new();
313                let mut args: Vec<TokenStream> = Vec::new();
314                for (i, a) in v.inputs.iter().enumerate() {
315                    let ai = ident(&format!("__a{}", i));
316                    match a {
317                        FoldArg::Leaf(li, _) => {
318                            let loc = &leaf_locals[*li];
319                            stmts.push(quote!(let #ai = #loc;));
320                            args.push(quote!(#ai));
321                        }
322                        FoldArg::Build(b) => {
323                            // Pin the nested build's error type to `String` so a
324                            // non-fallible inner ctor's bare `Ok(..)` infers `E`.
325                            let be = emit_build(b, leaf_locals, qualify);
326                            stmts.push(quote!(
327                                let #ai = {
328                                    let __r: ::core::result::Result<_, ::std::string::String> = #be;
329                                    __r?
330                                };
331                            ));
332                            if b.by_ref {
333                                args.push(quote!(&#ai));
334                            } else {
335                                args.push(quote!(#ai));
336                            }
337                        }
338                    }
339                }
340                let call = ctor_call_result(&path, &args, v.fallible);
341                syn::parse_quote!({
342                    (|| -> ::core::result::Result<_, ::std::string::String> {
343                        #(#stmts)*
344                        #call
345                    })()
346                })
347            }
348        }
349    }
350}

352/// `path(args…)` lifted to `Result<Target, String>` (mapping a fallible
353/// constructor's error via `Display`).
354fn ctor_call_result<I: quote::ToTokens>(path: &syn::Path, args: &[I], fallible: bool) -> syn::Expr {
355    if fallible {
356        syn::parse_quote!(#path( #(#args),* ).map_err(|__e| ::std::format!("{}", __e)))
357    } else {
358        syn::parse_quote!(::core::result::Result::Ok(#path( #(#args),* )))
359    }
360}
