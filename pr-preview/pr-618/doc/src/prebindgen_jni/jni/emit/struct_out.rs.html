8/// Resolve the typed-handle Kotlin FQN for a handle-bearing struct field
9/// and assert its folded strategy is one the struct encode/decode bridge
10/// supports. Today only scalar handle slots (`Direct`, optionally wrapped
11/// in `Nullable`) are encodable as a single `L<FQN>;` ctor arg; a
12/// collection layer (`Iterable`, i.e. `Vec<Handle>`) would need array
13/// codegen and is a loud build-time error until implemented.
14pub(crate) fn handle_field_fqn(ext: &Declarations, h: &Projection) -> String {
15    fn assert_scalar(s: &FoldStrategy) {
16        match s {
17            FoldStrategy::Base => {}
18            FoldStrategy::Optional(_, inner) => assert_scalar(inner),
19            FoldStrategy::Iterable(_) => panic!(
20                "struct handle field: collection (Vec<Handle>) layers are not yet \
21                 supported by the struct encode/decode bridge — add array codegen \
22                 to the late struct output/input plans to lift this guard"
23            ),
24        }
25    }
26    assert_scalar(&h.strategy);
27    ext.kotlin_fqn(&h.leaf_key)
28        .map(|v| v.to_string())
29        .unwrap_or_else(|| {
30            panic!(
31                "struct handle field: leaf `{}` has no Kotlin FQN registered \
32                 (ptr_class)",
33                h.leaf_key
34            )
35        })
36}

38/// One flattened leaf wire slot of a struct's recursive `fromParts` encode
39/// (see [`flatten_struct_encode`]). `ident` holds the encoded wire after the
40/// preludes run; `default` is the value used for this slot when it sits under
41/// an absent `Option<nested>` parent.
42pub(crate) struct EncSlot {
43    ident: proc_macro2::Ident,
44    /// How deeply nested the field this slot came from is: 0 for the struct's
45    /// own field, one more per inlined `data_class`.
46    ///
47    /// Read only by [`encode_leaves`], which
48    /// `JniGen::assert_leaf_derivations_agree` compares against the
49    /// registry-facing decomposition's own nesting — that one spells it with
50    /// the reserved `__` separator, this one counts it. The encode itself
51    /// needs the number while it recurses, not after, which is why nothing
52    /// else reads the field.
53    #[allow(dead_code)]
54    depth: usize,
55    wire_ty: TokenStream,
56    descriptor: String,
57    is_object: bool,
58    default: TokenStream,
59}

61/// Zero/null wire value for a JVM descriptor — used to fill an absent
62/// `Option<nested>`'s leaf slots (the Kotlin `present` flag tells the factory
63/// to ignore them).
64pub(crate) fn primitive_default_for_descriptor(sig: &str) -> TokenStream {
65    match sig {
66        "Z" => quote!(0u8),
67        "B" => quote!(0i8),
68        "C" => quote!(0u16),
69        "S" => quote!(0i16),
70        "I" => quote!(0i32),
71        "J" => quote!(0i64),
72        "F" => quote!(0.0f32),
73        "D" => quote!(0.0f64),
74        _ => quote!(jni::objects::JObject::null()),
75    }
76}

78/// The [`LeafSource::Reach`](prebindgen_registry::unfold::LeafSource) leaves of
79/// a by-value `data_class`, as the fixed-builder output and callback paths want
80/// them.
81///
82/// The list is [`Declarations::struct_out_wires`]', mapped. Runs BEFORE
83/// `resolve`, which is exactly why it shares that composition rather than
84/// walking the struct a second time: the leaf names reach the foreign
85/// `fromParts` parameters and the recipe identically, and two walks agreeing was a
86/// property nothing checked.
87///
88/// `None` — the type keeps the whole-value `fromParts` path — when a field is
89/// one the decomposition cannot state at all. Those are structural: a repeated
90/// nested class, a field the model does not hold as a named one, nesting past
91/// 16. Not a missing transform — a handle, an `enum_class`, a sum, an optional
92/// nested class and a selector inside a gated group all state fine, each
93/// carrying its own conversion (#602). See [`Declarations::struct_out_wires`]
94/// for the list and why one such field declines the whole value.
95pub(crate) fn synth_value_struct_leaves(
96    ext: &Declarations,
97    registry: &impl Conversions,
98    s: &prebindgen_registry::flat::Struct,
99) -> Option<Vec<prebindgen_registry::unfold::UnfoldLeaf>> {
100    use prebindgen_registry::unfold::{LeafSource, UnfoldLeaf};
101    Some(
102        ext.struct_out_wires_of(registry, &s.name)?
103            .into_iter()
104            .map(|w| UnfoldLeaf {
105                name: w.name,
106                path: w.reach,
107                out_ty: w.out_ty,
108                identity: false,
109                nullable: w.nullable,
110                // A selector is not read off a place: the emitter assigns the
111                // live alternative's number in each arm of its `match`, which
112                // is what `LeafSource::SumTag` says and what keeps
113                // `OutWire::from_leaf` round-tripping it back to a tag rather
114                // than trying to compile the sum type as a crossing.
115                source: match w.from {
116                    crate::jni::compile::OutFrom::Tag => LeafSource::SumTag,
117                    crate::jni::compile::OutFrom::Present => LeafSource::Presence,
118                    _ => LeafSource::Reach,
119                },
120                groups: w.groups,
121            })
122            .collect(),
123    )
124}

126/// Walk a [`StructPlan`] emitting the Rust-side wire encode: per leaf a
127/// prelude statement binding `__<prefix>_<field>` to the converted wire and
128/// an [`EncSlot`] describing its `JValue` slot. `access` is the Rust
129/// expression yielding the current struct value (`v`, `v.field`, or the
130/// matched `__cN` under an Option); `prefix` namespaces the generated idents.
131/// Walk the struct's fields and encode each leaf into a `fromParts` argument.
132///
133/// This is a source-value walk that the registry does not perform, and what
134/// keeps it here is now only that nothing has moved it. The coverage reason is
135/// gone: #616, #617 and #618 taught the registry-facing decomposition
136/// (`Declarations::struct_out_wires_at`) every gated shape this encode
137/// supports — a sum field as a tag over one group per variant, an optional
138/// nested class as a presence flag over a defaulted group, and one selector
139/// inside another — so the two now cover the same shapes rather than differing.
140///
141/// Which makes this the duplication #613 step 3 exists to remove: two walks
142/// deriving one leaf list. `JniGen::assert_leaf_derivations_agree` is what says
143/// they still agree, on every binding a test writes, and #619 is where this
144/// walk goes and that check with it.
145/// The leaves this encode emits, flattened — the `fromParts` argument list,
146/// in order, each with how deeply nested the field it came from is.
147///
148/// Test support: the registry-facing decomposition of the same struct must
149/// agree with this wherever it exists, and #603 recorded that agreement as
150/// measured rather than checked. `JniGen::write_rust` checks it now.
151#[cfg(test)]
152pub(crate) fn encode_leaves(
153    plan: &StructPlan,
154    emit: &prebindgen_registry::RustWriter,
155) -> Vec<(String, usize)> {
156    let (_, slots) = encode_plan(plan, &quote!(v), "", 0, &quote!(env), emit);
157    slots
158        .iter()
159        .map(|slot| {
160            (
161                slot.ident.to_string().trim_start_matches('_').to_string(),
162                slot.depth,
163            )
164        })
165        .collect()
166}

168/// The prefix an arm-local binding carries, and the outer slot it feeds.
169///
170/// A leaf's name is the same on both sides — it is one leaf — so the two would
171/// collide in one function. The arm-local name is the outer name with a
172/// **prefix** this module adds, which is why recovering the outer one is
173/// stripping a known prefix at a known position rather than searching for a
174/// marker: `base` and every slot fragment come from source identifiers, and
175/// any sentinel spelled inside one of them can appear in the middle of a name
176/// too (#616 review).
177/// The one "arm" an optional value has: present. Absence contributes no
178/// bindings of its own, only the defaults its group carries.
179pub(crate) const PRESENT_ARM: i32 = 0;
180
181fn arm_local_base(tag: usize, base: &str) -> String {
182    format!("arm{tag}_{base}")
183}
184
185/// The outer slot ident for an arm-local binding produced under
186/// [`arm_local_base`].
187fn outer_of(tag: usize, inner: &proc_macro2::Ident) -> proc_macro2::Ident {
188    let name = inner.to_string();
189    let prefix = format!("__arm{tag}_");
190    let rest = name.strip_prefix(&prefix).unwrap_or_else(|| {
191        panic!("an arm-local binding starts with the prefix its outer slot drops: `{name}`")
192    });
193    format_ident!("__{rest}")
194}

196fn encode_plan(
197    plan: &StructPlan,
198    access: &TokenStream,
199    prefix: &str,
200    depth: usize,
201    env_expr: &TokenStream,
202    emit: &prebindgen_registry::RustWriter,
203) -> (TokenStream, Vec<EncSlot>) {
204    let mut preludes = TokenStream::new();
205    let mut slots: Vec<EncSlot> = Vec::new();
206
207    for f in &plan.fields {
208        let fname = &f.fname;
209        let base = format!("{}_{}", prefix, fname);
210        let value = quote! { #access.#fname };
211        let (pre, sl) = encode_field(&f.kind, &value, &base, depth, env_expr, emit);
212        preludes.extend(pre);
213        slots.extend(sl);
214    }
215    (preludes, slots)
216}

218/// Emit the Rust-side wire encode of ONE value position — a struct field or a
219/// sum's variant payload. `value` is the Rust expression yielding it (`v.mode`
220/// at a struct field, the bound pattern variable inside a variant arm), which
221/// is what lets a sum reuse this for its payloads: a payload is encoded by the
222/// same code as a field of the same type, not by a parallel walk.
223fn encode_field(
224    kind: &PlanFieldKind,
225    value: &TokenStream,
226    base: &str,
227    depth: usize,
228    env_expr: &TokenStream,
229    emit: &prebindgen_registry::RustWriter,
230) -> (TokenStream, Vec<EncSlot>) {
231    let mut preludes = TokenStream::new();
232    let mut slots: Vec<EncSlot> = Vec::new();
233    {
234        let id = format_ident!("__{}", base);
235        // The leaf's COMPLETE chain, not just its wire-facing converter: a
236        // `convert!`-declared type (`Duration`) reaches the wire through its
237        // rust-side stages first (`Duration → u64 → jlong`).
238        let conv_value =
239            |conv: &ConvChain| -> TokenStream { conv.call(env_expr, value, base, emit) };
240        match kind {
241            // Projection leaf (opaque handle → jlong, `ULong` → jlong).
242            PlanFieldKind::Projection { conv, proj, .. } => {
243                let value_expr = conv_value(conv);
244                match proj.kind {
245                    ProjectionKind::Handle => {
246                        preludes.extend(quote! { let #id: jni::sys::jlong = #value_expr; });
247                        slots.push(EncSlot {
248                            depth,
249                            ident: id,
250                            wire_ty: quote!(jni::sys::jlong),
251                            descriptor: "J".to_string(),
252                            is_object: false,
253                            default: quote!(0i64),
254                        });
255                    }
256                    ProjectionKind::Unsigned64 => match proj.strategy {
257                        FoldStrategy::Base => {
258                            preludes.extend(quote! { let #id: jni::sys::jlong = #value_expr; });
259                            slots.push(EncSlot {
260                                depth,
261                                ident: id,
262                                wire_ty: quote!(jni::sys::jlong),
263                                descriptor: "J".to_string(),
264                                is_object: false,
265                                default: quote!(0i64),
266                            });
267                        }
268                        FoldStrategy::Optional(NullableKind::Niche, _) => {
269                            preludes.extend(quote! { let #id: jni::sys::jlong = #value_expr; });
270                            slots.push(EncSlot {
271                                depth,
272                                ident: id,
273                                wire_ty: quote!(jni::sys::jlong),
274                                descriptor: "J".to_string(),
275                                is_object: false,
276                                default: quote!(0i64),
277                            });
278                        }
279                        FoldStrategy::Optional(NullableKind::Boxed, _) => {
280                            preludes
281                                .extend(quote! { let #id: jni::objects::JObject = #value_expr; });
282                            slots.push(EncSlot {
283                                depth,
284                                ident: id,
285                                wire_ty: quote!(jni::objects::JObject),
286                                descriptor: "Ljava/lang/Long;".to_string(),
287                                is_object: true,
288                                default: quote!(jni::objects::JObject::null()),
289                            });
290                        }
291                        FoldStrategy::Iterable(_) => unreachable!(
292                            "projection collection fields are rejected by build_struct_plan"
293                        ),
294                    },
295                }
296            }
297            // Enum leaf → jint discriminant (Kotlin `fromParts` calls `fromInt`).
298            PlanFieldKind::Enum { conv, .. } => {
299                let value_expr = conv_value(conv);
300                preludes.extend(quote! { let #id: jni::sys::jint = #value_expr; });
301                slots.push(EncSlot {
302                    depth,
303                    ident: id,
304                    wire_ty: quote!(jni::sys::jint),
305                    descriptor: "I".to_string(),
306                    is_object: false,
307                    default: quote!(0i32),
308                });
309            }
310            // `Option<enum>` uses the enum's frozen primitive niche. Keep the
311            // boxed Integer branch for plans produced without niche metadata.
312            PlanFieldKind::OptionEnum { conv, niche, .. } => {
313                let value_expr = conv_value(conv);
314                if niche.is_some() {
315                    preludes.extend(quote! { let #id: jni::sys::jint = #value_expr; });
316                    slots.push(EncSlot {
317                        depth,
318                        ident: id,
319                        wire_ty: quote!(jni::sys::jint),
320                        descriptor: "I".to_string(),
321                        is_object: false,
322                        default: quote!(0i32),
323                    });
324                } else {
325                    preludes.extend(quote! { let #id: jni::objects::JObject = #value_expr; });
326                    slots.push(EncSlot {
327                        depth,
328                        ident: id,
329                        wire_ty: quote!(jni::objects::JObject),
330                        descriptor: "Ljava/lang/Integer;".to_string(),
331                        is_object: true,
332                        default: quote!(jni::objects::JObject::null()),
333                    });
334                }
335            }
336            // Nested data-class: inline the child's leaves; under `Option` add
337            // a `present` flag and default the child slots in the `None` arm.
338            PlanFieldKind::Nested {
339                optional,
340                plan: child,
341                ..
342            } => {
343                if !*optional {
344                    let (child_pre, child_slots) =
345                        encode_plan(child, value, base, depth + 1, env_expr, emit);
346                    preludes.extend(child_pre);
347                    slots.extend(child_slots);
348                } else {
349                    let cbind = format_ident!("__c{}", depth);
350                    let child_access = quote! { #cbind };
351                    // The child encodes under an arm-local base, and the outer
352                    // slots drop that prefix: the two are the same leaf, named
353                    // the same way the decomposition names it, and one scope
354                    // assigns from the other.
355                    let (child_pre, child_slots) = encode_plan(
356                        child,
357                        &child_access,
358                        &arm_local_base(PRESENT_ARM as usize, base),
359                        depth + 1,
360                        env_expr,
361                        emit,
362                    );
363                    let flag_id = format_ident!("__{}_present", base);
364                    let outer_ids: Vec<proc_macro2::Ident> = child_slots
365                        .iter()
366                        .map(|slot| outer_of(PRESENT_ARM as usize, &slot.ident))
367                        .collect();
368                    let outer_tys: Vec<TokenStream> =
369                        child_slots.iter().map(|sl| sl.wire_ty.clone()).collect();
370                    let inner_ids: Vec<proc_macro2::Ident> =
371                        child_slots.iter().map(|sl| sl.ident.clone()).collect();
372                    let defaults: Vec<TokenStream> =
373                        child_slots.iter().map(|sl| sl.default.clone()).collect();
374                    // Destructured through a coercion site: `kind` says this
375                    // field is optional, and how Rust spells that is the
376                    // source's business (#268).
377                    let obind = format_ident!("__on{}", depth);
378                    let coerce =
379                        prebindgen_registry::unfold::bind_as_option(&quote!(&#value), &obind);
380                    preludes.extend(quote! {
381                        let #flag_id: jni::sys::jboolean;
382                        #( let #outer_ids: #outer_tys; )*
383                        #coerce
384                        match #obind {
385                            ::core::option::Option::Some(#cbind) => {
386                                #child_pre
387                                #flag_id = 1u8;
388                                #( #outer_ids = #inner_ids; )*
389                            }
390                            ::core::option::Option::None => {
391                                #flag_id = 0u8;
392                                #( #outer_ids = #defaults; )*
393                            }
394                        }
395                    });
396                    // The flag sits one level inside the struct, with the
397                    // group it gates: it is reached THROUGH this field, which
398                    // is what the decomposition's `__` join says.
399                    slots.push(EncSlot {
400                        depth: depth + 1,
401                        ident: flag_id,
402                        wire_ty: quote!(jni::sys::jboolean),
403                        descriptor: "Z".to_string(),
404                        is_object: false,
405                        default: quote!(0u8),
406                    });
407                    for (i, sl) in child_slots.iter().enumerate() {
408                        slots.push(EncSlot {
409                            depth: sl.depth,
410                            ident: outer_ids[i].clone(),
411                            wire_ty: sl.wire_ty.clone(),
412                            descriptor: sl.descriptor.clone(),
413                            is_object: sl.is_object,
414                            default: sl.default.clone(),
415                        });
416                    }
417                }
418            }
419            // Data-carrying enum: one `match` binds the tag and EVERY group's
420            // slots — the live group from its payload, the rest from the same
421            // defaults an absent `Option<nested>` uses. One crossing, no JVM
422            // object built for the sum.
423            PlanFieldKind::Sum {
424                source,
425                optional,
426                variants,
427                ..
428            } => {
429                let tag_id = format_ident!("__{}__tag", base);
430
431                // Encode each variant's group once, against its own pattern
432                // bindings. `arms` keeps them aligned with the flat slot list.
433                struct Arm {
434                    pattern: TokenStream,
435                    preludes: TokenStream,
436                    slots: Vec<EncSlot>,
437                }
438                let mut arms: Vec<Arm> = Vec::new();
439                for (tag, v) in variants.iter().enumerate() {
440                    let vident = &v.rust_ident;
441                    let binds: Vec<syn::Ident> = (0..v.fields.len())
442                        .map(|i| format_ident!("__s{}_{}", depth, i))
443                        .collect();
444                    let mut vpre = TokenStream::new();
445                    let mut vslots: Vec<EncSlot> = Vec::new();
446                    for (f, bind) in v.fields.iter().zip(&binds) {
447                        // The arm's own bindings carry a marker segment the
448                        // outer slots drop: the two are the same leaf and want
449                        // the same name, and one scope assigns from the other.
450                        let fbase = arm_local_base(tag, &format!("{base}_{}", f.slot));
451                        let bind_expr = quote!(#bind);
452                        let (p, s) =
453                            encode_field(&f.kind, &bind_expr, &fbase, depth + 1, env_expr, emit);
454                        vpre.extend(p);
455                        vslots.extend(s);
456                    }
457                    // Bind every payload field, shaped like the variant.
458                    let pattern = match v.fields.first().map(|f| &f.member) {
459                        None => quote!(#source::#vident),
460                        Some(syn::Member::Named(_)) => {
461                            let pairs = v.fields.iter().zip(&binds).map(|(f, b)| {
462                                let syn::Member::Named(n) = &f.member else {
463                                    unreachable!("variant field shapes are uniform")
464                                };
465                                quote!(#n: #b)
466                            });
467                            quote!(#source::#vident { #(#pairs),* })
468                        }
469                        Some(syn::Member::Unnamed(_)) => quote!(#source::#vident(#(#binds),*)),
470                    };
471                    arms.push(Arm {
472                        pattern,
473                        preludes: vpre,
474                        slots: vslots,
475                    });
476                }
477
478                // Outer bindings: the tag plus every group's slots, side by
479                // side in variant order. Each arm assigns its own group from
480                // the values it just computed and defaults all the others.
481                let all: Vec<&EncSlot> = arms.iter().flat_map(|a| a.slots.iter()).collect();
482                // Named after the leaf each slot carries — `__outcome_found_v0`,
483                // not `__outcome_g0`. The inner bindings already use that
484                // naming, and it is what the registry-facing decomposition
485                // calls the same leaf, so the two derivations agree by
486                // construction rather than by a positional coincidence. The
487                // inner binding of the same name lives inside its match arm
488                // and shadows nothing the outer tuple reads.
489                let outer_ids: Vec<proc_macro2::Ident> = arms
490                    .iter()
491                    .enumerate()
492                    .flat_map(|(tag, arm)| {
493                        arm.slots.iter().map(move |slot| outer_of(tag, &slot.ident))
494                    })
495                    .collect();
496                let outer_tys: Vec<TokenStream> = all.iter().map(|s| s.wire_ty.clone()).collect();
497                let defaults: Vec<TokenStream> = all.iter().map(|s| s.default.clone()).collect();
498
499                let mut offset = 0usize;
500                let arm_code: Vec<TokenStream> = arms
501                    .iter()
502                    .enumerate()
503                    .map(|(tag, a)| {
504                        let n = a.slots.len();
505                        let live_outer = &outer_ids[offset..offset + n];
506                        let live_inner: Vec<proc_macro2::Ident> =
507                            a.slots.iter().map(|s| s.ident.clone()).collect();
508                        // Every slot outside this arm's own group is inert.
509                        let inert_outer: Vec<proc_macro2::Ident> = outer_ids
510                            .iter()
511                            .enumerate()
512                            .filter(|(i, _)| *i < offset || *i >= offset + n)
513                            .map(|(_, id)| id.clone())
514                            .collect();
515                        let inert_defaults: Vec<TokenStream> = defaults
516                            .iter()
517                            .enumerate()
518                            .filter(|(i, _)| *i < offset || *i >= offset + n)
519                            .map(|(_, d)| d.clone())
520                            .collect();
521                        offset += n;
522                        let pattern = &a.pattern;
523                        let pre = &a.preludes;
524                        let tag_lit = proc_macro2::Literal::i32_unsuffixed(tag as i32);
525                        quote! {
526                            #pattern => {
527                                #pre
528                                #tag_id = #tag_lit;
529                                #( #live_outer = #live_inner; )*
530                                #( #inert_outer = #inert_defaults; )*
531                            }
532                        }
533                    })
534                    .collect();
535
536                let decls = quote! {
537                    let #tag_id: jni::sys::jint;
538                    #( let #outer_ids: #outer_tys; )*
539                };
540                if !*optional {
541                    preludes.extend(quote! {
542                        #decls
543                        match &#value { #(#arm_code)* }
544                    });
545                } else {
546                    // `Option<sum>` keeps its own present flag ahead of the
547                    // tag: optionality and choice are independent facts.
548                    let flag_id = format_ident!("__{}_present", base);
549                    let sbind = format_ident!("__o{}", depth);
550                    let inner_arms: Vec<TokenStream> =
551                        arm_code.iter().map(|a| quote! { #a }).collect();
552                    // Destructured through a coercion site: `kind` says this
553                    // field is optional, and how Rust spells that is the
554                    // source's business (#268).
555                    let obind = format_ident!("__oc{}", depth);
556                    let coerce =
557                        prebindgen_registry::unfold::bind_as_option(&quote!(&#value), &obind);
558                    preludes.extend(quote! {
559                        let #flag_id: jni::sys::jboolean;
560                        #decls
561                        #coerce
562                        match #obind {
563                            ::core::option::Option::Some(#sbind) => {
564                                #flag_id = 1u8;
565                                match #sbind { #(#inner_arms)* }
566                            }
567                            ::core::option::Option::None => {
568                                #flag_id = 0u8;
569                                #tag_id = 0i32;
570                                #( #outer_ids = #defaults; )*
571                            }
572                        }
573                    });
574                    // One level inside the struct, like the tag it gates and
575                    // like an optional nested class's own flag: it is reached
576                    // THROUGH this field. The decomposition says so by joining
577                    // the names with `__`, and this said `depth` — a
578                    // disagreement no shape reached while `Option<sum>` was
579                    // refused a decomposition to disagree with.
580                    slots.push(EncSlot {
581                        depth: depth + 1,
582                        ident: flag_id,
583                        wire_ty: quote!(jni::sys::jboolean),
584                        descriptor: "Z".to_string(),
585                        is_object: false,
586                        default: quote!(0u8),
587                    });
588                }
589                // A sum field's slots sit one level inside the struct, the
590                // same as an inlined nested class's: the tag and the groups
591                // are reached THROUGH the field. The decomposition says so by
592                // joining the names with `__`, and depth is that count.
593                slots.push(EncSlot {
594                    depth: depth + 1,
595                    ident: tag_id,
596                    wire_ty: quote!(jni::sys::jint),
597                    descriptor: "I".to_string(),
598                    is_object: false,
599                    default: quote!(0i32),
600                });
601                for (i, sl) in all.iter().enumerate() {
602                    slots.push(EncSlot {
603                        depth: sl.depth,
604                        ident: outer_ids[i].clone(),
605                        wire_ty: sl.wire_ty.clone(),
606                        descriptor: sl.descriptor.clone(),
607                        is_object: sl.is_object,
608                        default: sl.default.clone(),
609                    });
610                }
611            }
612            // Simple leaf: bind per the plan's wire form.
613            PlanFieldKind::Leaf {
614                conv,
615                wire,
616                form,
617                descriptor,
618                ..
619            } => {
620                let value_expr = conv_value(conv);
621                match form {
622                    LeafForm::Prim => {
623                        preludes.extend(quote! { let #id: #wire = #value_expr; });
624                        slots.push(EncSlot {
625                            depth,
626                            ident: id,
627                            wire_ty: quote!(#wire),
628                            descriptor: descriptor.clone(),
629                            is_object: false,
630                            default: primitive_default_for_descriptor(descriptor),
631                        });
632                    }
633                    LeafForm::IntoObject => {
634                        preludes.extend(
635                            quote! { let #id: jni::objects::JObject = #value_expr.into(); },
636                        );
637                        slots.push(EncSlot {
638                            depth,
639                            ident: id,
640                            wire_ty: quote!(jni::objects::JObject),
641                            descriptor: descriptor.clone(),
642                            is_object: true,
643                            default: quote!(jni::objects::JObject::null()),
644                        });
645                    }
646                    LeafForm::Object => {
647                        preludes.extend(quote! { let #id: jni::objects::JObject = #value_expr; });
648                        slots.push(EncSlot {
649                            depth,
650                            ident: id,
651                            wire_ty: quote!(jni::objects::JObject),
652                            descriptor: descriptor.clone(),
653                            is_object: true,
654                            default: quote!(jni::objects::JObject::null()),
655                        });
656                    }
657                }
658            }
659        }
660    }
661    (preludes, slots)
662}

664/// Assemble the Rust→JVM half of a frozen whole-struct codec.
665///
666/// Classification, child selection, JVM descriptors, and the target Kotlin
667/// class are all frozen before this point. This function only turns that plan
668/// into the final converter body; it performs no registry or declaration
669/// lookup and cannot rediscover source-type facts.
670pub(crate) fn render_struct_output_body(
671    plan: &StructPlan,
672    java_class_name: &str,
673    emit: &prebindgen_registry::RustWriter,
674) -> syn::Expr {
675    // Recursively flatten the whole object graph into leaf wires, then build it
676    // with ONE `call_static_method("fromParts", …)` — no per-nested-struct JNI
677    // crossing. The Kotlin `fromParts` factory (recursively flattened the same
678    // way in `render_data_class_source`) reassembles the graph in bytecode.
679    let access = quote!(v);
680    let (preludes, slots) = encode_plan(plan, &access, "", 0, &quote!(env), emit);
681
682    let mut sig = String::from("(");
683    let mut args: Vec<TokenStream> = Vec::new();
684    for sl in &slots {
685        sig.push_str(&sl.descriptor);
686        let id = &sl.ident;
687        if sl.is_object {
688            args.push(quote!(jni::objects::JValue::Object(&#id)));
689        } else {
690            args.push(quote!(jni::objects::JValue::from(#id)));
691        }
692    }
693    sig.push_str(&format!(")L{};", java_class_name));
694    let factory_sig_lit = syn::LitStr::new(&sig, Span::call_site());
695
696    let body: syn::Expr = syn::parse_quote!({
697        #preludes
698        let __obj = env.call_static_method(
699            #java_class_name,
700            "fromParts",
701            #factory_sig_lit,
702            &[#(#args),*],
703        )
704        .and_then(|__v| __v.l())
705        .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!("encode struct via fromParts: {}", e)))?;
706        __obj
707    });
708    body
709}

711pub(crate) fn struct_module_path(
712    ext: &Declarations,
713    registry: &impl Conversions,
714    name: &syn::Ident,
715) -> syn::Path {
716    // The module the struct is reachable under from the generated file: its
717    // origin crate (multi-source registries) or the default module. Takes the
718    // NAME, which is all it needs — so it serves a caller holding the element
719    // and one still holding the item.
720    ext.fn_module(registry, name)
721}
