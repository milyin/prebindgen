6pub(crate) fn input(
7    wire: syn::Ident,
8    length: syn::Ident,
9    index: syn::Ident,
10    element: &WireLayout<CWire, CGroup>,
11) -> Result<SequenceInput<CWire>> {
12    let row = prebindgen_conversions::template_parameter(&format!("row_{wire}"));
13    let mut bind = TokenStream::new();
14    let mut values = Vec::new();
15    for (position, field) in element.iter().enumerate() {
16        let name = &field.name;
17        let member = format_ident!("v{position}");
18        bind.extend(
19            quote!(let #name=unsafe {(*((#wire.data as *const #row).add(#index))).#member};),
20        );
21        values.push(quote!(#name));
22    }
23    Ok(SequenceInput {
24        carrier: CWire::Rows,
25        open: quote!(let #length=#wire.len;::prebindgen_runtime::validate_slice(#wire.data as *const #row,#length)?;),
26        bind,
27        validation_values: values.clone(),
28        prepared_values: values.into_iter().map(Some).collect(),
29        release: quote!(),
30    })
31}
32pub(crate) fn output(
33    wire: syn::Ident,
34    length: syn::Ident,
35    element: &WireLayout<CWire, CGroup>,
36    values: &[TokenStream],
37) -> Result<SequenceOutput<CWire>> {
38    let row = prebindgen_conversions::template_parameter(&format!("row_{wire}"));
39    if element.len() != values.len() {
40        return Err("collection row/value count differs".into());
41    }
42    let fields = values.iter().enumerate().map(|(i, value)| {
43        let field = format_ident!("v{i}");
44        quote!(#field:#value)
45    });
46    Ok(SequenceOutput {
47        commit_each: false,
48        fallible: false,
49        carrier: CWire::Rows,
50        begin: quote!(let mut #wire:Vec<#row>=Vec::with_capacity(#length);),
51        enter: quote!(),
52        store: quote!(#wire.push(#row {#(#fields),*});),
53        finish: quote!(let #wire=#wire.into_boxed_slice();),
54        value: quote!(__PbRows {data:if #wire.is_empty(){::core::ptr::null()}else{#wire.as_ptr().cast()},len:#wire.len()}),
55        commit: quote!(let _=Box::into_raw(#wire);),
56    })
57}
58pub(crate) fn schemas<'a>(
59    layout: &'a WireLayout<CWire, CGroup>,
60    out: &mut Vec<(&'a Wire, &'a WireLayout<CWire, CGroup>)>,
61) {
62    for node in layout.nodes() {
63        match node {
64            WireNode::Wire(_) => {}
65            WireNode::Group { children, .. } => schemas(children, out),
66            WireNode::Optional(c) => schemas(&c.element, out),
67            WireNode::Sequence(c) => {
68                schemas(&c.element, out);
69                if let Some(carrier) = c.carrier() {
70                    out.push((carrier, &c.element));
71                }
72            }
73        }
74    }
75}

77#[allow(clippy::too_many_arguments)]
78pub(crate) fn emit_schemas(
79    schemas: Vec<(&Wire, &WireLayout<CWire, CGroup>)>,
80    output_carriers: &std::collections::BTreeSet<String>,
81    context: &CContext,
82    prefix: &str,
83    pointers: &[(String, syn::Type)],
84    symbols: &mut std::collections::BTreeSet<String>,
85    tokens: &mut TokenStream,
86    header: &mut String,
87) -> Result<()> {
88    for (carrier, element) in schemas {
89        if !matches!(carrier.kind, CWire::Rows) {
90            continue;
91        }
92        let row = &context.row_types[&carrier.name.to_string()];
93        let base = row.to_string();
94        let base = base.trim_start_matches("__PbRow_");
95        let c_row = format!("{prefix}_{base}_row");
96        let c_rows = format!("{prefix}_{base}_rows");
97        let free_rows = format_ident!("{prefix}_{base}_rows_free");
98        let is_output = output_carriers.contains(&carrier.name.to_string());
99        let mut row_symbols = vec![c_row.clone(), c_rows.clone()];
100        if is_output {
101            row_symbols.push(free_rows.to_string());
102        }
103        for symbol in row_symbols {
104            if !symbols.insert(symbol) {
105                return Err("duplicate C collection symbol".into());
106            }
107        }
108        let fields = element
109            .iter()
110            .enumerate()
111            .map(|(i, w)| {
112                let name = format_ident!("v{i}");
113                let ty = w.rust_type();
114                quote!(pub #name:#ty)
115            })
116            .collect::<Vec<_>>();
117        tokens.extend(quote!(#[repr(C)] #[derive(Clone,Copy)] pub struct #row {#(#fields),*}));
118        header.push_str("typedef struct {\n");
119        for (i, w) in element.iter().enumerate() {
120            header.push_str(&format!(
121                "    {} v{i};\n",
122                c_ty(w, prefix, pointers, context)
123            ));
124        }
125        header.push_str(&format!(
126            "}} {c_row};\ntypedef struct {{ {c_row} *data; size_t len; }} {c_rows};\n"
127        ));
128        if is_output {
129            header.push_str(&format!("/* Release only the row allocation. Free or transfer owned element fields separately. Only pass buffers returned by this library. */\nvoid {free_rows}({c_rows} rows);\n"));
130            tokens.extend(quote! {
131                    #[doc="# Safety\nThe rows must be an unchanged allocation returned with this exact element layout. This frees only row storage, not element-owned fields."]
132                    #[no_mangle]pub unsafe extern "C" fn #free_rows(rows:__PbRows){if rows.len!=0{drop(Box::from_raw(std::ptr::slice_from_raw_parts_mut(rows.data as *mut #row,rows.len)));}}
133                });
134        }
135    }
136    Ok(())
137}

139/// Reclaim a committed output that will not reach its caller, including nested rows.
140pub(crate) fn release(
141    layout: &WireLayout<CWire, CGroup>,
142    values: &[TokenStream],
143    context: &CContext,
144    catch_panics: bool,
145) -> TokenStream {
146    let mut schemas = Vec::new();
147    self::schemas(layout, &mut schemas);
148    let mut code = TokenStream::new();
149    for (wire, value) in layout.iter().zip(values) {
150        match &wire.kind {
151            CWire::Buffer => code.extend(quote! {if !(#value).data.is_null(){drop(Box::from_raw(std::ptr::slice_from_raw_parts_mut((#value).data as *mut u8,(#value).len)));}}),
152            CWire::Pointer { pointer, .. } => {
153                let ty = &pointer.ty;
154                let drop = quote!(drop(Box::from_raw((#value) as *mut #ty)););
155                let drop = if catch_panics {
156                    quote!(let _ = ::prebindgen_runtime::boundary(|| { #drop Ok(()) });)
157                } else { drop };
158                code.extend(quote! { if !(#value).is_null() { #drop } });
159            }
160            CWire::Rows => {
161                let row=&context.row_types[&wire.name.to_string()];
162                let element=schemas.iter().find(|(carrier,_)|carrier.name==wire.name).unwrap().1;
163                let fields=element.iter().enumerate().map(|(i,_)|{let field=format_ident!("v{i}");quote!(row.#field)}).collect::<Vec<_>>();
164                let child=release(element,&fields,context,catch_panics);
165                code.extend(quote! {if (#value).len != 0 {let rows=Box::from_raw(std::ptr::slice_from_raw_parts_mut((#value).data as *mut #row,(#value).len));for row in rows.iter(){#child}}});
166            }
167            _=>{}
168        }
169    }
170    code
171}

173pub(crate) fn callback_output(values: &[TokenStream], length: syn::Ident) -> SequenceOutput<CWire> {
174    let names = (0..values.len())
175        .map(|i| format_ident!("__pb_callback_value_{i}"))
176        .collect::<Vec<_>>();
177    let prepare = names
178        .iter()
179        .zip(values)
180        .map(|(name, value)| quote!(let #name=#value;));
181    let commit = prebindgen_conversions::template_parameter("commit_element");
182    SequenceOutput {
183        commit_each: true,
184        fallible: true, // A callback pointer must be provided; checked before source invocation.
185        carrier: CWire::scalar(Primitive::Usize),
186        begin: quote!(),
187        enter: quote!(),
188        store: quote!(#(#prepare)* #commit unsafe { __pb_callback.expect("validated callback")(__pb_callback_context, #(#names),*); }),
189        finish: quote!(),
190        value: quote!(#length),
191        commit: quote!(),
192    }
193}

195/// A visitor may replace exactly one collection, optionally with its presence flag.
196pub(crate) fn callback_shape(layout: &WireLayout<CWire, CGroup>) -> bool {
197    fn one(nodes: &[WireNode<CWire, CGroup>]) -> bool {
198        match nodes {
199            [WireNode::Sequence(c)] => c.carrier().is_some_and(|carrier| {
200                matches!(
201                    carrier.kind,
202                    CWire::Scalar {
203                        primitive: Primitive::Usize,
204                        ..
205                    }
206                )
207            }),
208            [WireNode::Optional(c)] => match (&c.representation, &c.metadata) {
209                (
210                    prebindgen_flat::WireRepresentation::Inline { controls },
211                    Some(CGroup::Optional(o)),
212                ) if o.niche.is_none() => {
213                    matches!(controls.as_slice(),[flag] if flag.name==o.discriminator)
214                        && one(c.element.nodes())
215                }
216                _ => false,
217            },
218            [WireNode::Group { children, .. }] => one(children.nodes()),
219            _ => false,
220        }
221    }
222    one(layout.nodes())
223}
