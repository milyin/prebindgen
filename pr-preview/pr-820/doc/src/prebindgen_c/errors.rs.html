1//! Explicit Result error outputs. Message buffers and structured errors are caller-owned.
2use super::*;
3use prebindgen_conversions::OutputFragment;
4use prebindgen_flat::Conversion;
5use prebindgen_flat::{Function, TypeUse};

7pub(crate) struct ErrorBinding {
8    pub name: String,
9    pub ty: syn::Type,
10    pub plan: OutputPlan<CWire>,
11    pub fields: Vec<String>,
12}
13pub(crate) fn helper(name: &str) -> syn::Ident {
14    format_ident!("__pb_report_{name}")
15}
16pub(crate) fn prepare(
17    flat: &mut Bindings,
18    codec: &enums::EnumCodec,
19    declarations: &prebindgen_declarations::Declarations,
20    configured: Vec<(String, syn::Type)>,
21) -> Result<Vec<ErrorBinding>> {
22    let mut out = Vec::new();
23    for (name, ty) in configured {
24        let ty = flat.resolve_type(&ty)?;
25        let recipe = declarations.owned_record_output(flat, &ty)?;
26        let id = helper(&name);
27        if flat.functions.contains_key(&id.to_string()) {
28            return Err("C error helper name collision".into());
29        }
30        flat.functions.insert(
31            id.to_string(),
32            Function {
33                name: id.clone(),
34                path: syn::parse_quote!(#id),
35                params: vec![],
36                ret: TypeUse {
37                    ty: ty.clone(),
38                    slot: None,
39                },
40                location: Default::default(),
41            },
42        );
43        let result =
44            prebindgen_declarations::bind_output(flat, codec, &id.to_string(), &recipe, &());
45        let function = flat.functions.remove(&id.to_string()).unwrap();
46        result?;
47        let plan = function.ret.slot.unwrap();
48        let fields = field_names(&plan.nodes);
49        out.push(ErrorBinding {
50            name,
51            ty,
52            plan,
53            fields,
54        });
55    }
56    Ok(out)
57}

59fn field_names(layout: &[prebindgen_flat::WireNode<CWire, CGroup>]) -> Vec<String> {
60    use prebindgen_flat::WireNode;
61    let layout = match layout {
62        [prebindgen_flat::WireNode::Outcomes { success, .. }] => success.as_slice(),
63        nodes => nodes,
64    };
65    let mut names = Vec::new();
66    if let Some(WireNode::Group {
67        metadata: CGroup::Fields,
68        children,
69    }) = layout.first()
70    {
71        for node in children.as_slice() {
72            if let WireNode::Group {
73                metadata: CGroup::Field(name),
74                children,
75            } = node
76            {
77                for _ in crate::group::record_fields(children) {
78                    names.push(if crate::group::record_fields(children).len() == 1 {
79                        name.to_string().trim_start_matches("r#").to_owned()
80                    } else {
81                        format!("v{}", names.len())
82                    });
83                }
84            }
85        }
86    }
87    let keywords="auto break case char const continue default do double else enum extern float for goto if inline int long register restrict return short signed sizeof static struct switch typedef union unsigned void volatile while _Alignas _Alignof _Atomic _Bool _Complex _Generic _Imaginary _Noreturn _Static_assert _Thread_local bool true false NULL";
88    let mut unique = std::collections::BTreeSet::new();
89    if names.len() != crate::group::result_packet_fields(layout).len()
90        || names.iter().any(|name| {
91            !name.is_ascii()
92                || keywords.split_whitespace().any(|word| word == name)
93                || !unique.insert(name.clone())
94        })
95    {
96        return (0..crate::group::result_packet_fields(layout).len())
97            .map(|i| format!("v{i}"))
98            .collect();
99    }
100    names
101}

103pub(crate) fn emit(
104    errors: &[ErrorBinding],
105    prefix: &str,
106    pointers: &[(String, syn::Type)],
107    symbols: &mut std::collections::BTreeSet<String>,
108    tokens: &mut TokenStream,
109    header: &mut String,
110) -> Result<()> {
111    for error in errors {
112        let name = &error.name;
113        let c_name = format!("{prefix}_{name}");
114        let drop_name = format_ident!("{prefix}_drop_{name}");
115        for symbol in [c_name.clone(), drop_name.to_string()] {
116            if !symbols.insert(symbol) {
117                return Err("C error type/destructor symbol collision".into());
118            }
119        }
120        let rust_name = format_ident!("__PbError_{name}");
121        let source_ty = &error.ty;
122        let helper = helper(name);
123        let input = format_ident!("source_error");
124        let context = CContext {
125            row_types: {
126                let mut schemas = Vec::new();
127                sequence::schemas(&error.plan.nodes, &mut schemas);
128                schemas
129                    .into_iter()
130                    .map(|(wire, _)| {
131                        (
132                            wire.name.to_string(),
133                            format_ident!("__PbRow_error_{}_{}", name, wire.name),
134                        )
135                    })
136                    .collect()
137            },
138            acknowledgement: format_ident!("unused_acknowledgement"),
139            output_pointer: format_ident!("error_output"),
140            output_type: syn::parse_quote!(#rust_name),
141            error_pointer: format_ident!("unused_error"),
142            error_reported: format_ident!("unused_reported"),
143            output_committed: TokenStream::new(),
144            buffer_type: syn::parse_quote!(__PbBuffer),
145            wide_integer_type: syn::parse_quote!(__Pb128),
146            read_buffer: syn::parse_quote!(__pb_read),
147        };
148        let mut schemas = Vec::new();
149        sequence::schemas(&error.plan.nodes, &mut schemas);
150        let carriers = schemas
151            .iter()
152            .map(|(wire, _)| wire.name.to_string())
153            .collect();
154        sequence::emit_schemas(
155            schemas, &carriers, &context, prefix, pointers, symbols, tokens, header,
156        )?;
157        let commit = error.plan.code_fragment_slot(
158            OutputFragment::Commit {
159                context: &context,
160                source: &input,
161            },
162            source_ty,
163        )?;
164        let commit: syn::ExprBlock = syn::parse2(commit).map_err(|e| e.to_string())?;
165        let commit = commit.block.stmts;
166        let deliver = error.plan.code_fragment_slot(
167            OutputFragment::Deliver {
168                context: &context,
169                source: &input,
170                next: quote!(#(#commit)* *reported=true; Ok(())),
171            },
172            source_ty,
173        )?;
174        let prepare = error.plan.code_fragment_slot(
175            OutputFragment::Prepare {
176                context: &context,
177                source: &input,
178                next: deliver,
179            },
180            source_ty,
181        )?;
182        let mut fields = Vec::new();
183        let mut drops = TokenStream::new();
184        header.push_str("typedef struct {\n");
185        for (i, wire) in crate::group::result_packet_fields(&error.plan.nodes).enumerate() {
186            let field = format_ident!("v{i}");
187            let ty = wire.rust_type();
188            fields.push(quote!(pub #field:#ty));
189            let c_field = if error.fields.len()
190                == crate::group::result_packet_fields(&error.plan.nodes).len()
191            {
192                error.fields[i].trim_start_matches("r#").to_owned()
193            } else {
194                format!("v{i}")
195            };
196            prebindgen_declarations::ident(&c_field)?;
197            header.push_str(&format!(
198                "    {} {c_field};\n",
199                c_ty(wire, prefix, pointers, &context)
200            ));
201        }
202        let values = crate::group::result_packet_fields(&error.plan.nodes)
203            .enumerate()
204            .map(|(i, wire)| {
205                let field = format_ident!("v{i}");
206                if !matches!(
207                    wire.kind,
208                    CWire::Buffer | CWire::Pointer { .. } | CWire::Rows
209                ) {
210                    return (wire, quote!((*error).#field));
211                }
212                let empty = match wire.kind {
213                    CWire::Buffer => quote!(__PbBuffer {
214                        data: std::ptr::null(),
215                        len: 0
216                    }),
217                    CWire::Rows => quote!(__PbRows {
218                        data: std::ptr::null(),
219                        len: 0
220                    }),
221                    CWire::Pointer { .. } => quote!(std::ptr::null_mut()),
222                    _ => unreachable!(),
223                };
224                let local = format_ident!("owned_{i}");
225                drops.extend(quote!(let #local=std::mem::replace(&mut (*error).#field,#empty);));
226                (wire, quote!(#local))
227            })
228            .collect::<Vec<_>>();
229        drops.extend(sequence::release(
230            &error.plan.nodes,
231            &values,
232            &context,
233            false,
234        ));
235        if fields.is_empty() {
236            fields.push(quote!(pub unused:u8));
237            header.push_str("    uint8_t unused;\n");
238        }
239        header.push_str(&format!("}} {c_name};\n/* Frees owned error fields and clears them; repeated calls are harmless. */\nvoid {drop_name}({c_name} *error);\n"));
240        let wires = crate::group::result_packet_fields(&error.plan.nodes).collect::<Vec<_>>();
241        let mut comments = describe_groups(&error.plan.nodes, &wires, true, &mut 0, "error");
242        for i in (0..wires.len()).rev() {
243            let field = if error.fields.len() == wires.len() {
244                error.fields[i].trim_start_matches("r#").to_owned()
245            } else {
246                format!("v{i}")
247            };
248            comments = comments.replace(&format!("output->v{i}"), &format!("error->{field}"));
249        }
250        header.push_str(&comments);
251        tokens.extend(quote! {
252            #[repr(C)] pub struct #rust_name {#(#fields),*}
253            #[allow(clippy::needless_borrow,clippy::let_unit_value,clippy::unnecessary_cast,clippy::useless_conversion)]
254            unsafe fn #helper(#input:#source_ty,error_output:*mut #rust_name,reported:&mut bool)->Result<(),String>{
255                if error_output.is_null(){return Ok(());}
256                #prepare
257            }
258            #[doc="# Safety\nThe pointer must be null or writable storage containing a live error produced by this library."]
259            #[no_mangle]pub unsafe extern "C" fn #drop_name(error:*mut #rust_name) {
260                if let Err(message)=::prebindgen_runtime::boundary(|| {if !error.is_null(){#drops}Ok(())}) {panic!("C error destructor failed: {}",message);}
261            }
262        });
263    }
264    Ok(())
265}

267#[allow(clippy::too_many_arguments)]
268pub(crate) fn result_function(
269    source_name: &syn::Ident,
270    error_ty: &syn::Type,
271    outputs: &[&Wire],
272    function_output_layout: &[prebindgen_flat::WireNode<CWire, CGroup>],
273    mut params: Vec<TokenStream>,
274    mut cparams: Vec<String>,
275    body: &TokenStream,
276    context: &CContext,
277    prefix: &str,
278    pointers: &[(String, syn::Type)],
279    errors: &[ErrorBinding],
280    consumes: bool,
281    comments: &str,
282    tokens: &mut TokenStream,
283    header: &mut String,
284) -> Result<()> {
285    let name = format_ident!("{prefix}_{source_name}");
286    let internal = &context.output_type;
287    let out = &context.output_pointer;
288    let accepted = &context.acknowledgement;
289    let error = &context.error_pointer;
290    let reported = &context.error_reported;
291    let typed = errors
292        .iter()
293        .find(|e| prebindgen_flat::type_key(&e.ty) == prebindgen_flat::type_key(error_ty));
294    let (rust_error, c_error, report_binding) = if let Some(typed) = typed {
295        let rust = format_ident!("__PbError_{}", typed.name);
296        let report = helper(&typed.name);
297        (
298            quote!(#rust),
299            format!("{prefix}_{}", typed.name),
300            quote! {
301                let mut delivered=false;
302                if ::prebindgen_runtime::boundary(|| unsafe {#report(<#error_ty as From<String>>::from(message),#error,&mut delivered)}).is_err() && !delivered {
303                // All CWire carriers have a valid zero representation with no
304                // owned allocations. Never expose uninitialized error storage.
305                std::ptr::write(#error,std::mem::zeroed());
306            }
307            },
308        )
309    } else {
310        (
311            quote!(*mut std::ffi::c_char),
312            "char *".into(),
313            quote!(unsafe {__pb_report_message(message,#error)};),
314        )
315    };
316    let niche = outputs
317        .iter()
318        .enumerate()
319        .find_map(|(i, w)| w.kind.available_niche().map(|n| (i, n)));
320    let return_position = niche.as_ref().map(|(i, _)| *i);
321    let (rust_return, c_return, failure, success) = if let Some((i, niche)) = &niche {
322        let wire = outputs[*i];
323        let ty = wire.rust_type();
324        let field = format_ident!("v{i}");
325        let failure = CCodec.absent_value(&wire.kind, Some(niche), false)?;
326        (
327            quote!(#ty),
328            c_ty(wire, prefix, pointers, context),
329            failure,
330            quote!(value.#field),
331        )
332    } else {
333        (quote!(bool), "bool".into(), quote!(false), quote!(true))
334    };
335    let mut checks = TokenStream::new();
336    let mut writes = TokenStream::new();
337    let mut display = Vec::new();
338    for (i, wire) in outputs.iter().enumerate() {
339        if return_position == Some(i) {
340            display.push("the return value".to_owned());
341            continue;
342        }
343        let id = format_ident!("out{i}");
344        let field = format_ident!("v{i}");
345        let ty = wire.rust_type();
346        params.push(quote!(#id:*mut #ty));
347        cparams.push(format!("{} *{id}", c_ty(wire, prefix, pointers, context)));
348        checks.extend(quote!(if #id.is_null(){return Err("null success-output pointer".into());}));
349        writes.extend(quote!(std::ptr::write(#id,value.#field);));
350        display.push(format!("*{id}"));
351    }
352    let acknowledgement = if consumes {
353        params.push(quote!(ownership:*mut bool));
354        cparams.push("bool *ownership".into());
355        quote!(if !ownership.is_null(){std::ptr::write(ownership,#accepted);})
356    } else {
357        quote!()
358    };
359    let preflight = if consumes {
360        quote!(if ownership.is_null() {
361            return Err("null ownership-output pointer".into());
362        })
363    } else {
364        quote!()
365    };
366    params.push(quote!(#error:*mut #rust_error));
367    cparams.push(format!("{c_error} *{error}"));
368    let mut comments = comments.to_owned();
369    for (i, path) in display.iter().enumerate().rev() {
370        comments = comments.replace(&format!("output->v{i}"), path);
371    }
372    header.push_str(&comments);
373    if let Some((i, niche)) = &niche {
374        let absent = match niche {
375            CNiche::Null => "NULL".to_owned(),
376            CNiche::Integer(value) => value.to_string(),
377        };
378        header.push_str(&format!(
379            "/* Result: success field v{i} is returned directly; {absent} means failure. */\n"
380        ));
381    } else {
382        header.push_str("/* Result: true = success; false = failure. Success outputs are unchanged on failure. */\n");
383    }
384    if consumes {
385        header.push_str("/* ownership is required: false before acceptance; true after active owned inputs transfer, including on failure. */\n");
386    }
387    header.push_str("/* error may be NULL to discard details. Failure initializes nonnull error with caller-owned data (or an empty record if error conversion fails). Success leaves error unchanged. Free a prior error before reusing its storage. All output/ownership/error storage must be disjoint. */\n");
388    header.push_str(&format!("{c_return} {name}({});\n", cparams.join(", ")));
389    let rollback = sequence::release(
390        function_output_layout,
391        &outputs
392            .iter()
393            .enumerate()
394            .map(|(i, wire)| {
395                let field = format_ident!("v{i}");
396                (*wire, quote!(value.#field))
397            })
398            .collect::<Vec<_>>(),
399        context,
400        true,
401    );
402    tokens.extend(quote! {
403        #[doc="# Safety\nSuccess, ownership and nonnull error outputs must be writable and pairwise disjoint, with no unreleased prior error. Input allocations must satisfy their declared provenance, aliasing and ownership contract."]
404        #[allow(clippy::needless_borrow,clippy::let_unit_value,clippy::unnecessary_cast,clippy::useless_conversion)]
405        #[no_mangle]pub unsafe extern "C" fn #name(#(#params),*)->#rust_return {
406            #[allow(unused_mut)]let mut #accepted=false;
407            #[allow(unused_mut)]let mut #reported=false;
408            let mut success_committed=false;
409            let mut storage=std::mem::MaybeUninit::<#internal>::uninit();
410            let #out=storage.as_mut_ptr();
411            let outcome=::prebindgen_runtime::boundary(|| {#preflight #checks #body});
412            #acknowledgement
413            match outcome {
414                Ok(())=>{let value=storage.assume_init();#writes #success}
415                Err(message)=>{
416                    if success_committed {let value=storage.assume_init();#rollback}
417                    if !#reported && !#error.is_null(){#report_binding}#failure
418                }
419            }
420        }
421    });
422    Ok(())
