1//! Explicit enum declarations and the enum-aware C conversion codec.
2use super::*;
3use prebindgen_declarations::{Condition, EnumMapping};
4
5pub(crate) struct EnumCodec {
6    pub(crate) error_layouts:
7        std::collections::BTreeMap<String, prebindgen_flat::WireLayout<CWire, CGroup>>,
8    pub(crate) callback_wire: Option<syn::Ident>,
9    pub(crate) errors: std::collections::BTreeMap<String, String>,
10    pub(crate) enums: std::collections::BTreeMap<String, (String, EnumMapping)>,
11}
12impl EnumCodec {
13    fn mapping(&self, ty: &syn::Type) -> Option<(&str, &EnumMapping)> {
14        let ty = if let syn::Type::Reference(r) = ty {
15            &r.elem
16        } else {
17            ty
18        };
19        self.enums
20            .get(&prebindgen_flat::type_key(ty))
21            .map(|(name, m)| (name.as_str(), m))
22    }
23    fn wire(name: &str, mapping: &EnumMapping) -> CWire {
24        CWire::enumeration(name, mapping.clone())
25    }
26}
27impl Codec for EnumCodec {
28    type Wire = CWire;
29    type Niche = CNiche;
30    type Group = CGroup;
31
32    type Extra = ();
33    type Context = CContext;
34    type Input = InputPlan<CWire>;
35    type Output = OutputPlan<CWire>;
36    fn binding_failure_layout(
37        &self,
38        e: &(),
39    ) -> Result<Option<prebindgen_flat::WireLayout<CWire, CGroup>>> {
40        CCodec.binding_failure_layout(e)
41    }
42    fn source_error_layout(
43        &self,
44        ty: &syn::Type,
45        mapped: bool,
46        e: &(),
47    ) -> Result<prebindgen_flat::WireLayout<CWire, CGroup>> {
48        if let Some(layout) = self.error_layouts.get(&prebindgen_flat::type_key(ty)) {
49            return Ok(layout.clone());
50        }
51        CCodec.source_error_layout(ty, mapped, e)
52    }
53    fn optional_metadata(
54        &self,
55        p: syn::Ident,
56        d: syn::Ident,
57        n: Option<CNiche>,
58        controls: Vec<prebindgen_flat::Wire<CWire>>,
59    ) -> CGroup {
60        CCodec.optional_metadata(p, d, n, controls)
61    }
62    fn alternatives_metadata(&self, selector: prebindgen_flat::Wire<CWire>) -> CGroup {
63        CCodec.alternatives_metadata(selector)
64    }
65    fn field_group(&self, n: syn::Ident) -> CGroup {
66        CCodec.field_group(n)
67    }
68    fn fields_group(&self) -> CGroup {
69        CCodec.fields_group()
70    }
71    fn finish_input(&self, p: &mut InputPlan<CWire>, e: &()) -> Result<()> {
72        CCodec.finish_input(p, e)
73    }
74    fn finish_output(&self, p: &mut OutputPlan<CWire>, e: &()) -> Result<()> {
75        CCodec.finish_output(p, e)
76    }
77    fn sequence_input(
78        &self,
79        e: &(),
80        w: syn::Ident,
81        n: syn::Ident,
82        i: syn::Ident,
83        element: &prebindgen_flat::WireLayout<CWire, CGroup>,
84        active: &Condition,
85    ) -> Result<prebindgen_declarations::SequenceInput<CGroup>> {
86        CCodec.sequence_input(e, w, n, i, element, active)
87    }
88    fn sequence_output(
89        &self,
90        e: &(),
91        w: syn::Ident,
92        n: syn::Ident,
93        i: syn::Ident,
94        element: &prebindgen_flat::WireLayout<CWire, CGroup>,
95        values: &[(syn::Ident, TokenStream)],
96    ) -> Result<prebindgen_declarations::SequenceOutput<CGroup>> {
97        if self.callback_wire.as_ref() == Some(&w) {
98            sequence::callback_output(w, values, element, n)
99        } else {
100            CCodec.sequence_output(e, w, n, i, element, values)
101        }
102    }
103    fn result_error_value(
104        &self,
105        ty: &syn::Type,
106        value: TokenStream,
107        message: Option<TokenStream>,
108        extra: &(),
109    ) -> Result<TokenStream> {
110        if let Some(name) = self.errors.get(&prebindgen_flat::type_key(ty)) {
111            if message.is_some() {
112                return Err("structured C errors cannot also select a message mapper".into());
113            }
114            let helper = errors::helper(name);
115            let pointer = prebindgen_conversions::template_parameter("error_pointer");
116            let reported = prebindgen_conversions::template_parameter("error_reported");
117            Ok(
118                quote!({unsafe {#helper(#value,#pointer,&mut #reported)}?;#reported=true;String::new()}),
119            )
120        } else {
121            CCodec.result_error_value(ty, value, message, extra)
122        }
123    }
124    fn result_error(
125        &self,
126        ty: &syn::Type,
127        message: TokenStream,
128        extra: &(),
129    ) -> Result<TokenStream> {
130        CCodec.result_error(ty, message, extra)
131    }
132    fn niche(&self, w: &CWire) -> Option<CNiche> {
133        CCodec.niche(w)
134    }
135    fn set_niche(&self, w: &mut CWire, n: Option<CNiche>) {
136        CCodec.set_niche(w, n)
137    }
138    fn remaining_niche(&self, w: &CWire, n: &CNiche) -> Option<CNiche> {
139        CCodec.remaining_niche(w, n)
140    }
141    fn is_present(&self, w: &CWire, n: &CNiche, v: TokenStream) -> Result<TokenStream> {
142        CCodec.is_present(w, n, v)
143    }
144    fn absent_value(&self, w: &CWire, n: Option<&CNiche>, input: bool) -> Result<TokenStream> {
145        CCodec.absent_value(w, n, input)
146    }
147    fn selector(&self, e: &()) -> CWire {
148        CCodec.selector(e)
149    }
150    fn consumes(&self, ty: &syn::Type, e: &()) -> bool {
151        if let Some(inner) = prebindgen_declarations::option_inner(ty) {
152            return self.consumes(inner, e);
153        }
154        self.mapping(ty).is_none() && CCodec.consumes(ty, e)
155    }
156    fn input_leaf(
157        &self,
158        ty: &syn::Type,
159        e: &(),
160        w: syn::Ident,
161        v: syn::Ident,
162        active: &Condition,
163    ) -> Result<InputPlan<CWire>> {
164        if let Some(inner) = prebindgen_declarations::option_inner(ty) {
165            let tag = format_ident!("{w}_present");
166            let child = self.input_leaf(
167                inner,
168                e,
169                format_ident!("{w}_some"),
170                format_ident!("{v}_some"),
171                &active.arm(tag.clone(), 1),
172            )?;
173            return prebindgen_declarations::optional_input(self, e, tag, v, active, child);
174        }
175        if let Some((name, m)) = self.mapping(ty) {
176            return prebindgen_declarations::enum_input(m, ty, w, v, active, Self::wire(name, m));
177        }
178        CCodec.input_leaf(ty, e, w, v, active)
179    }
180    fn output_leaf(
181        &self,
182        ty: &syn::Type,
183        e: &(),
184        w: syn::Ident,
185        v: TokenStream,
186        borrowed: bool,
187    ) -> Result<OutputPlan<CWire>> {
188        if let Some(inner) = prebindgen_declarations::option_inner(ty) {
189            let tag = format_ident!("{w}_present");
190            let payload = format_ident!("{w}_payload");
191            let child = self.output_leaf(
192                inner,
193                e,
194                format_ident!("{w}_some"),
195                quote!(#payload),
196                borrowed,
197            )?;
198            return prebindgen_declarations::optional_output(
199                self, e, tag, payload, v, borrowed, child,
200            );
201        }
202        if let Some((name, m)) = self.mapping(ty) {
203            if matches!(ty,syn::Type::Reference(r) if r.mutability.is_some()) {
204                return Err("mutable-reference outputs are unsupported".into());
205            }
206            return Ok(prebindgen_declarations::enum_output(
207                m,
208                w,
209                v,
210                Self::wire(name, m),
211            ));
212        }
213        CCodec.output_leaf(ty, e, w, v, borrowed)
214    }
215
216    fn optional_niche(
217        &self,
218        payload: &prebindgen_flat::WireLayout<Self::Wire, Self::Group>,
219    ) -> Option<prebindgen_declarations::NicheReservation<Self::Niche>> {
220        CCodec.optional_niche(payload)
221    }
222    fn optional_presence(
223        &self,
224        payload: &prebindgen_flat::WireLayout<Self::Wire, Self::Group>,
225        reservation: &prebindgen_declarations::NicheReservation<Self::Niche>,
226    ) -> Result<TokenStream> {
227        CCodec.optional_presence(payload, reservation)
228    }
229    fn absent_output_values(
230        &self,
231        payload: &prebindgen_flat::WireLayout<Self::Wire, Self::Group>,
232        reservation: Option<&prebindgen_declarations::NicheReservation<Self::Niche>>,
233    ) -> Result<Vec<(syn::Ident, TokenStream)>> {
234        CCodec.absent_output_values(payload, reservation)
235    }
236    fn retire_niches(
237        &self,
238        payload: &mut prebindgen_flat::WireLayout<Self::Wire, Self::Group>,
239        reservation: Option<&prebindgen_declarations::NicheReservation<Self::Niche>>,
240    ) -> Result<()> {
241        CCodec.retire_niches(payload, reservation)
242    }
243    fn validate_input(&self, plan: &prebindgen_declarations::CodecInput<Self>) -> Result<()> {
244        CCodec.validate_input(plan)
245    }
246    fn validate_output(&self, plan: &prebindgen_declarations::CodecOutput<Self>) -> Result<()> {
247        CCodec.validate_output(plan)
248    }
249}

264    #[test]
265    fn enum_declarations_emit_fixed_width_types_and_preserve_discriminants() {
266        let output = Cbindgen::new(flat(
267            "pub enum E { A = -2, B, C = 7, D, Min = -2147483648 } pub fn echo(x:E)->E{}",
268        ))
269        .enum_type("code", syn::parse_quote!(E))
270        .function(prebindgen_declarations::fun!(echo))
271        .panic()
272        .build()
273        .unwrap();
274        assert!(output.header.contains("typedef int32_t pb_code;"));
275        assert!(output.header.contains("#define pb_code_B ((pb_code)(-1))"));
276        assert!(output.header.contains("#define pb_code_D ((pb_code)(8))"));
277        assert!(output
278            .header
279            .contains("#define pb_code_Min ((pb_code)(INT32_MIN))"));
280        syn::parse_file(&output.rust).unwrap();
281    }

336    #[test]
337    fn enum_niches_skip_valid_values_and_survive_optional_composition() {
338        let source = flat("pub enum E { A = -2, B, C = 0, D = 7 }");
339        let mapping = EnumMapping::new(source.enums.values().next().unwrap()).unwrap();
340        let codec = EnumCodec {
341            error_layouts: Default::default(),
342            callback_wire: None,
343            errors: Default::default(),
344            enums: std::collections::BTreeMap::from([(
345                "crate :: E".into(),
346                ("code".into(), mapping.clone()),
347            )]),
348        };
349        let wire = EnumCodec::wire("code", &mapping);
350        assert!(matches!(wire.available_niche(), Some(CNiche::Integer(-3))));
351        assert!(matches!(
352            codec.remaining_niche(&wire, &CNiche::Integer(8)),
353            Some(CNiche::Integer(6))
354        ));
355        let mut ty: syn::Type = syn::parse_quote!(crate::E);
356        for depth in 1..=6 {
357            ty = syn::parse_quote!(Option<#ty>);
358            let input = codec
359                .input_leaf(
360                    &ty,
361                    &(),
362                    syn::parse_quote!(wire),
363                    syn::parse_quote!(value),
364                    &Condition::default(),
365                )
366                .unwrap();
367            let output = codec
368                .output_leaf(&ty, &(), syn::parse_quote!(wire), quote!(value), false)
369                .unwrap();
370            assert_eq!(input.layout.len(), 1);
371            assert_eq!(crate::group::result_packet_fields(&output.layout).len(), 1);
372            assert!(
373                matches!(input.layout.iter().next().unwrap().kind.available_niche(),Some(CNiche::Integer(n)) if n == -3-depth)
374            );
375            super::super::group::validate_layout(&input.layout).unwrap();
376            super::super::group::validate_layout(&output.layout).unwrap();
377        }
378    }
379    #[test]
380    fn optional_enum_reservations_cannot_reuse_available_or_reserved_values() {
381        let source = flat("pub enum E { A = -1, B = 0 }");
382        let mapping = EnumMapping::new(source.enums.values().next().unwrap()).unwrap();
383        let wire = EnumCodec::wire("code", &mapping);
384        let mut tree: prebindgen_flat::WireLayout<CWire, CGroup> = vec![GenericWire {
385            name: syn::parse_quote!(wire),
386            kind: wire,
387        }]
388        .into();
389        let group = |presence, value| {
390            CGroup::Optional(COptional {
391                controls: vec![],
392                presence: syn::parse_str(presence).unwrap(),
393                discriminator: syn::parse_quote!(wire),
394                niche: Some(CNiche::Integer(value)),
395            })
396        };
397        tree.enclose_optional(Some(group("inner", -2)));
398        assert!(super::super::group::validate_layout(&tree)
399            .unwrap_err()
400            .contains("already reserved"));
401        tree[0].kind.clear_niche();
402        super::super::group::validate_layout(&tree).unwrap();
403        tree.enclose_optional(Some(group("outer", -2)));
404        assert!(super::super::group::validate_layout(&tree)
405            .unwrap_err()
406            .contains("duplicate"));
407    }
