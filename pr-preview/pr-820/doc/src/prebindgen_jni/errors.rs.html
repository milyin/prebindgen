1//! JNI error channels reuse ordinary output conversions. User callbacks run in Kotlin.
2use super::*;
3use prebindgen_conversions::NativeConversionExt;
4use prebindgen_conversions::OutputFragment;
5use prebindgen_flat::{Conversion, Function, TypeUse};
6
7#[derive(Clone)]
8pub(crate) struct ErrorBinding {
9    helper: syn::Ident,
10    ty: syn::Type,
11    pub output: JniOutput,
12    pub handler: String,
13    exception: Option<String>,
14}
15impl ErrorBinding {
16    pub fn throws(&self) -> String {
17        self.exception.as_ref().map_or_else(String::new, |class| {
18            format!("@kotlin.jvm.Throws({}::class)\n", class.replace('$', "."))
19        })
20    }
21    pub fn parameter(&self, ret: &str) -> String {
22        let default = self.exception.as_ref().map_or_else(String::new, |class| {
23            format!(
24                " = {} {{ message -> throw {}(message) }}",
25                self.handler,
26                class.replace('$', ".")
27            )
28        });
29        format!("onError: {}<{ret}>{default}", self.handler)
30    }
31}

33pub(crate) fn prepare(
34    flat: &mut Bindings,
35    codec: &mut data::DataCodec,
36    declarations: &prebindgen_declarations::Declarations,
37    functions: &[FunctionDecl],
38    classes: &BTreeMap<String, Class>,
39    prefix: &str,
40    seen: &mut BTreeSet<String>,
41) -> Result<BTreeMap<String, ErrorBinding>> {
42    let mut out = BTreeMap::new();
43    let mut cache: BTreeMap<(String, bool), ErrorBinding> = BTreeMap::new();
44    let mut handlers: BTreeMap<(String, bool), String> = BTreeMap::new();
45    for function in functions {
46        let ret = flat.function(&function.name)?.ret.ty.clone();
47        let Some((_, error)) = prebindgen_flat::result_parts(&ret) else {
48            continue;
49        };
50        let mapped = declarations.error_mapper_for(function, error).is_some();
51        let key = (type_key(error), mapped);
52        if let Some(binding) = cache.get(&key) {
53            out.insert(function.name.clone(), binding.clone());
54            continue;
55        }
56        let exception = codec.error_classes.get(&key.0).cloned();
57        if exception.is_some() && !mapped {
58            return Err(
59                "error_class requires an explicit message mapper for its default Kotlin handler"
60                    .into(),
61            );
62        }
63        let (ty, recipe) = if mapped {
64            (
65                syn::parse_quote!(String),
66                prebindgen_declarations::Output::Leaf,
67            )
68        } else {
69            (error.clone(), declarations.output_recipe(flat, error)?)
70        };
71        let helper = format_ident!("__pb_jni_error_{}", out.len());
72        if flat.functions.contains_key(&helper.to_string()) {
73            return Err("JNI error helper name collision".into());
74        }
75        flat.functions.insert(
76            helper.to_string(),
77            Function {
78                name: helper.clone(),
79                path: syn::parse_quote!(#helper),
80                params: vec![],
81                ret: TypeUse {
82                    ty: ty.clone(),
83                    slot: None,
84                },
85                location: Default::default(),
86            },
87        );
88        let result =
89            prebindgen_declarations::bind_output(flat, codec, &helper.to_string(), &recipe, &());
90        let output = flat.functions.remove(&helper.to_string()).unwrap().ret.slot;
91        result.map_err(|e| format!("JNI error `{}` in `{}`: {e}", key.0, function.name))?;
92        let output = output.unwrap();
93        codec.error_layouts.insert(
94            key.clone(),
95            output
96                .layout()
97                .outcome(prebindgen_flat::OutcomeKind::Success)
98                .expect("error payload")
99                .clone(),
100        );
101        let handler = if let Some(handler) = handlers.get(&key) {
102            handler.clone()
103        } else {
104            let short = if let syn::Type::Path(path) = error {
105                path.path.segments.last().unwrap().ident.to_string()
106            } else {
107                let mut name = camel(&function.name);
108                if let Some(first) = name.get_mut(..1) {
109                    first.make_ascii_uppercase();
110                }
111                format!("{name}Error")
112            };
113            let pkg = classes
114                .get(&key.0)
115                .map_or(prefix, |class| class.pkg.as_str());
116            let suffix = if mapped { "MessageHandler" } else { "Handler" };
117            let handler = format!("{pkg}.{short}{suffix}");
118            if !seen.insert(handler.clone()) {
119                return Err(format!("duplicate foreign name `{handler}`"));
120            }
121            handlers.insert(key.clone(), handler.clone());
122            handler
123        };
124        // Each function can have its own mapper; the helper converts the already mapped String.
125        codec.error_helpers.insert(key.clone(), helper.clone());
126        let binding = ErrorBinding {
127            helper,
128            ty,
129            output,
130            handler,
131            exception,
132        };
133        cache.insert(key, binding.clone());
134        out.insert(function.name.clone(), binding);
135    }
136    Ok(out)
137}

139pub(crate) fn emit(errors: &BTreeMap<String, ErrorBinding>) -> Result<TokenStream> {
140    let mut code = TokenStream::new();
141    let mut emitted = BTreeSet::new();
142    for error in errors.values() {
143        if !emitted.insert(error.helper.to_string()) {
144            continue;
145        }
146        let helper = &error.helper;
147        let ty = &error.ty;
148        let source = format_ident!("source");
149        let context = JniContext {
150            environment: format_ident!("env"),
151            acknowledgement: format_ident!("unused_ack"),
152            domain_output: format_ident!("reported"),
153            is_domain_error: true,
154            output_committed: quote!(*reported = Some(result.as_raw());),
155        };
156        let commit = error.output.code_fragment_slot(
157            OutputFragment::Commit {
158                context: &context,
159                source: &source,
160            },
161            &syn::parse_quote!(()),
162        )?;
163        let body = error.output.code_fragment_slot(
164            OutputFragment::Deliver {
165                context: &context,
166                source: &source,
167                next: quote!({let _: () = #commit; Ok(())}),
168            },
169            &syn::parse_quote!(::prebindgen_runtime::Result<()>),
170        )?;
171        let body = error.output.code_fragment_slot(
172            OutputFragment::Prepare {
173                context: &context,
174                source: &source,
175                next: body,
176            },
177            &syn::parse_quote!(::prebindgen_runtime::Result<()>),
178        )?;
179        code.extend(quote! {
180            #[allow(clippy::needless_borrow, clippy::let_unit_value, clippy::unnecessary_cast, clippy::useless_conversion)]
181            fn #helper(mut env: &mut ::prebindgen_jni_runtime::jni::JNIEnv<'_>, source: #ty, reported: &mut Option<::prebindgen_jni_runtime::jni::sys::jobjectArray>) -> ::prebindgen_runtime::Result<()> { #body }
182        });
183    }
184    Ok(code)
185}

187pub(crate) fn interfaces(prefix: &str, files: &mut BTreeMap<String, String>) -> Result<()> {
188    let binding_name = format!("{}/JniErrorHandler.kt", prefix.replace('.', "/"));
189    if files.insert(binding_name, format!("package {prefix}\n\n/** Binding failure callback, invoked after the native call and handle locks finish. */\npublic fun interface JniErrorHandler<out R> {{ public fun run(message: String?): R }}\n")).is_some() {
190        return Err("Kotlin error interface collision".into());
191    }
192    Ok(())
193}

195pub(crate) fn handler_files(
196    errors: &BTreeMap<String, ErrorBinding>,
197    context: &KotlinContext<'_>,
198    files: &mut BTreeMap<String, String>,
199) -> Result<()> {
200    let mut seen = BTreeSet::new();
201    for error in errors.values() {
202        if !seen.insert(error.handler.clone()) {
203            continue;
204        }
205        let (pkg, name) = error.handler.rsplit_once('.').unwrap();
206        let output = error.output.kotlin(context)?;
207        let slots: usize = output
208            .result_types
209            .iter()
210            .map(|ty| {
211                if matches!(ty.as_str(), "Long" | "Double" | "ULong") {
212                    2
213                } else {
214                    1
215                }
216            })
217            .sum();
218        if slots > 254 {
219            return Err(format!(
220                "domain handler `{}` exceeds the JVM parameter limit",
221                error.handler
222            ));
223        }
224        let params = output
225            .result_types
226            .iter()
227            .enumerate()
228            .map(|(i, ty)| format!("v{i}: {ty}"))
229            .collect::<Vec<_>>()
230            .join(", ");
231        let text = format!("package {pkg}\n\n/** Domain-error callback. Called after JNI and handle locks finish. Handle arguments are closed after run; call take() to retain an owner. */\npublic fun interface {name}<out R> {{ public fun run({params}): R }}\n");
232        if files
233            .insert(format!("{}/{}.kt", pkg.replace('.', "/"), name), text)
234            .is_some()
235        {
236            return Err("Kotlin error interface collision".into());
237        }
238    }
239    Ok(())
240}

242pub(crate) fn delivery(error: &ErrorBinding, context: &KotlinContext<'_>) -> Result<String> {
243    let context = KotlinContext {
244        cleanup_failure: Some("cleanupFailure"),
245        ..*context
246    };
247    let mut output = error.output.kotlin(&context)?;
248    kotlin::remap_output(
249        &mut output,
250        context.result,
251        context.result,
252        1,
253        error.output.native_wires().len(),
254    );
255    let call = format!("onError.run({})", output.delivered.join(", "));
256    if output.cleanup.is_empty() {
257        return Ok(format!("{}\n{call}", output.materialize));
258    }
259    let cleanup = output.cleanup;
260    Ok(format!("{}\nvar cleanupFailure: Throwable? = null\nvar callbackFailed = false\ntry {{\n{}\n{call}\n}} catch (failure: Throwable) {{ callbackFailed = true; cleanupFailure = failure; throw failure }} finally {{\n{cleanup}\nif (!callbackFailed) cleanupFailure?.let {{ throw it }}\n}}", output.output_locals, output.materialize))
261}
