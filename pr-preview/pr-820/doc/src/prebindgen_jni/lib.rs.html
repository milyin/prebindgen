78/// A Kotlin owning handle class for one nonzero-sized source type.
79///
80/// Generated code requires `Send` because the handle can be closed on any JVM
81/// thread. Foreign output observations also require `Clone`. Inputs may own,
82/// share or exclusively borrow the source value.
83#[derive(Clone)]
84pub struct PtrClassDecl {
85    /// Source type, resolved against the selected flat model.
86    pub ty: syn::Type,
87    /// Literal Kotlin class name; absent means the last segment of the source type.
88    pub name: Option<String>,
89}
90impl PtrClassDecl {
91    /// Declare a handle class with a name derived from the source type.
92    ///
93    /// `ty` is the source type the class owns, as written in the source crate.
94    pub fn new(ty: syn::Type) -> Self {
95        Self { ty, name: None }
96    }
97    /// Override the Kotlin short class name with `name`, validated during build.
98    pub fn name(mut self, name: impl Into<String>) -> Self {
99        self.name = Some(name.into());
100        self
101    }
102}
103/// A Kotlin package containing handle/data classes and top-level functions.
104///
105/// Its name is relative to [`JniGenBuilder::set_package_prefix`].
106#[derive(Clone, Default)]
107pub struct PackageDecl {
108    /// Relative dotted package name; empty means the root package prefix.
109    pub name: String,
110    /// Handle and data classes to emit in this package.
111    pub classes: Vec<ClassDecl>,
112    /// Source functions to export as top-level Kotlin functions.
113    pub functions: Vec<FunctionDecl>,
114}
115impl PackageDecl {
116    /// Create an empty package relative to the configured root prefix.
117    ///
118    /// `name` is a dotted package name relative to that prefix, or empty for the
119    /// prefix itself.
120    pub fn new(name: impl Into<String>) -> Self {
121        Self {
122            name: name.into(),
123            ..Default::default()
124        }
125    }
126    /// Add a handle or data class. Duplicate source types or names fail during build.
127    pub fn class(mut self, class: impl Into<ClassDecl>) -> Self {
128        self.classes.push(class.into());
129        self
130    }
131    /// Add a top-level Kotlin function with optional occurrence overrides.
132    ///
133    /// Names default to camel case; [`FunctionDecl::name`] overrides that spelling.
134    pub fn fun(mut self, f: FunctionDecl) -> Self {
135        self.functions.push(f);
136        self
137    }

160/// Generated Rust/JNI bindings and their Kotlin source files.
161///
162/// Start with [`Self::builder`]. Source capture, expansion declarations and package
163/// placement are resolved before wrapper emission. Data classes map supported
164/// record fields to Kotlin properties, including owning handle members in returns.
165/// Handle classes own native allocations.
166/// Selectors keep all alternatives
167/// in one native signature; multiple outputs use a Kotlin builder callback.
168///
169/// ```
170/// use prebindgen_jni::{JniGen, package, ptr_class, fun};
171/// let source = syn::parse_file(r#"
172///     pub struct Counter { pub value: i64 }
173///     pub fn new_counter(value: i64) -> Counter { Counter { value } }
174/// "#)?;
175/// let generated = JniGen::builder()
176///     .items(source.items.into_iter().map(|item| (item, Default::default())))
177///     .set_package_prefix("org.example")
178///     .package(package!()
179///         .class(ptr_class!(Counter))
180///         .fun(fun!(new_counter)))
181///     .build()?;
182/// assert!(generated.kotlin.contains_key("org/example/Counter.kt"));
183/// assert!(generated.kotlin["org/example/Functions.kt"].contains("newCounter"));
184/// syn::parse_file(&generated.rust)?;
185/// # Ok::<(), Box<dyn std::error::Error>>(())
186/// ```
187///
188/// In `build.rs`, replace `.items(...)` with `.source(flat_crate::PREBINDGEN_OUT_DIR)`,
189/// then write both outputs. Add the source crate as both a build dependency and
190/// a normal dependency. Include the Rust file in a binding crate depending on
191/// `prebindgen-runtime`, `prebindgen-jni-runtime` and `konst` (used by captured
192/// source feature guards). Compile all generated Kotlin
193/// files together. Calls must go through those wrappers: their locks, consumed
194/// state and ownership acknowledgement are part of the native safety contract.
195pub struct JniGen {
196    /// Formatted native Rust source, including conversion code and flat-model guards.
197    pub rust: String,
198    /// Kotlin sources keyed by package-relative paths such as `org/example/Counter.kt`.
199    pub kotlin: BTreeMap<String, String>,
200}

204        JniGenBuilder::default()
205    }

208/// Collect declarations before resolving a JNI/Kotlin binding.
209///
210/// Supply exactly one of [`Self::source`], [`Self::items`] or [`Self::flat`], plus
211/// a nonempty package prefix. Type-wide expansions are order-independent. Custom
212/// configuration callbacks run in registration order before declarations fill slots.
213#[derive(Default)]
214pub struct JniGenBuilder {
215    configure: Vec<Configure>,
216    flat: Option<Bindings>,
217    source: Option<std::path::PathBuf>,
218    items: Option<Vec<(syn::Item, prebindgen::SourceLocation)>>,
219    expansions: Vec<ExpandDecl>,
220    prefix: String,
221    init: String,
222    packages: Vec<PackageDecl>,
223    error_classes: Vec<(syn::Type, String)>,
224}
225impl JniGenBuilder {
226    /// Create an empty builder; equivalent to [`JniGen::builder`].
227    pub fn new() -> Self {
228        Self::default()
229    }
230    /// Adjust the selected flat model before declaration defaults are applied.
231    ///
232    /// `f` receives the model and may attach converters to it.
233    ///
234    /// Use this to attach converters with [`prebindgen_declarations::bind_input`] and
235    /// [`prebindgen_declarations::bind_output`], using [`JniCodec`]. Callbacks run in
236    /// registration order; errors stop generation. Source selection has already happened, so references
237    /// introduced here must already exist in the model. Per-function declarations
238    /// can explicitly override the converters assigned by these callbacks.
239    ///
240    /// ```
241    /// use prebindgen_jni::{Bindings, JniGen, JniCodec, fun, package};
242    /// use prebindgen_declarations::{bind_input, bind_output, Input, Output};
243    /// let source = syn::parse_file("pub fn echo(x: i64) -> i64 { x }")?;
244    /// let flat = Bindings::parse(
245    ///     source.items.into_iter().map(|item| (item, Default::default())),
246    /// )?;
247    /// let generated = JniGen::builder().flat(flat)
248    ///     .set_package_prefix("example")
249    ///     .configure(|flat| {
250    ///         bind_input(flat, &JniCodec, "echo", "x", &Input::Leaf, &())?;
251    ///         bind_output(flat, &JniCodec, "echo", &Output::Leaf, &())
252    ///     })
253    ///     .package(package!().fun(fun!(echo)))
254    ///     .build()?;
255    /// syn::parse_file(&generated.rust)?;
256    /// # Ok::<(), Box<dyn std::error::Error>>(())
257    /// ```
258    pub fn configure(mut self, f: impl FnOnce(&mut Bindings) -> Result<()> + 'static) -> Self {
259        self.configure.push(Box::new(f));
260        self
261    }
262    /// Append captured items, each paired with its source location.
263    ///
264    /// `items` usually comes from `prebindgen::Source::items_all`.
265    ///
266    /// Calls accumulate items. Only declared functions, expansion helpers and required
267    /// types are selected for validation. Automatic type discovery follows
268    /// unqualified source names (see [`Flat::parse_selected`]); types referenced
269    /// only by qualified paths can be selected with an unqualified `ptr_class!`
270    /// or `data_class!` declaration, or included through [`Self::flat`]. Mutually exclusive with
271    /// source and flat input.
272    pub fn items(
273        mut self,
274        items: impl IntoIterator<Item = (syn::Item, prebindgen::SourceLocation)>,
275    ) -> Self {
276        self.items.get_or_insert_with(Vec::new).extend(items);
277        self
278    }
279    /// Read captures from a source crate's `PREBINDGEN_OUT_DIR`, passed as `path`.
280    ///
281    /// Uses the same selection as [`Self::items`]. A repeated call replaces the path;
282    /// combining it with items or a flat model is rejected during build.
283    pub fn source(mut self, path: impl AsRef<std::path::Path>) -> Self {
284        self.source = Some(path.as_ref().to_owned());
285        self
286    }
287    /// Register a type-wide input or output default, built with `expand_param!`
288    /// or `expand_return!`.
289    ///
290    /// Per-function overrides replace defaults. Duplicate declarations for the same
291    /// type and direction are rejected during build.
292    pub fn expand(mut self, decl: impl Into<ExpandDecl>) -> Self {
293        self.expansions.push(decl.into());
294        self
295    }
296    /// Select a default source error-message mapper for Result returns using `ty`.
297    /// The helper takes exactly E or &E and returns String. A function's
298    /// `on_error(...)` overrides this default. Duplicate resolved types are rejected.
299    pub fn error_mapper(
300        mut self,
301        ty: syn::Type,
302        mapper: prebindgen_declarations::FunctionDecl,
303    ) -> Self {
304        self.expansions
305            .push(prebindgen_declarations::ErrorDecl::new(ty, mapper).into());
306        self
307    }

309    /// Supply an already parsed model with input and output conversion slots.
310    ///
311    /// Bound functions still need a Kotlin package declaration. Mutually
312    /// exclusive with source and item input.
313    pub fn flat(mut self, flat: Bindings) -> Self {
314        self.flat = Some(flat);
315        self
316    }
317    /// Set the required nonempty root Kotlin package, such as `org.example.bindings`.
318    ///
319    /// All package declarations are relative to this prefix. The last setting wins.
320    pub fn set_package_prefix(mut self, prefix: impl Into<String>) -> Self {
321        self.prefix = prefix.into();
322        self
323    }
324    /// Set Kotlin statements for the generated `JNINative` object's initialization.
325    ///
326    /// Typically loads the native library, e.g. `System.loadLibrary("bindings")`.
327    /// The text is emitted verbatim and checked by the Kotlin compiler.
328    pub fn set_jni_native_init(mut self, init: impl Into<String>) -> Self {
329        self.init = init.into();
330        self
331    }
332    /// Supply an optional throwing default for the Kotlin domain-error callback.
333    ///
334    /// `class` is a fully qualified dotted JVM binary name; use `$` between
335    /// nested classes (for example `example.Errors$Rejected`). The class must exist at
336    /// Kotlin compile time, extend Throwable, and provide a public String constructor.
337    /// Result-returning functions also require `error_mapper` or
338    /// `FunctionDecl::on_error` to select their message mapper. Kotlin invokes this
339    /// default after JNI and handle locks finish; callers can replace `onError`
340    /// with a recovering callback. Without this setting `onError` is required.
341    /// Binding failures use the separate `onBindingError` callback.
342    /// Duplicate registrations for the same resolved source type are rejected.
343    pub fn error_class(mut self, ty: syn::Type, class: impl Into<String>) -> Self {
344        self.error_classes.push((ty, class.into()));
345        self
346    }
347    /// Append a package's class and function declarations.
348    ///
349    /// Multiple declarations may share a package, but their exported names must be unique.
350    pub fn package(mut self, package: PackageDecl) -> Self {
351        self.packages.push(package);
352        self
353    }
354    /// Resolve declarations and generate Rust and Kotlin texts without writing files.
355    ///
356    /// Rejects ambiguous source selection, invalid or duplicate names, unsupported
357    /// recipes, undeclared handle types and signatures exceeding JVM parameter limits.
358    /// Layout, Rust trait bounds and injected Kotlin syntax are checked by the
359    /// respective compilers when consuming the generated output.
360    pub fn build(self) -> Result<JniGen> {
361        let references = prebindgen_declarations::declarations::source_functions(
362            &self.expansions,
363            self.packages.iter().flat_map(|p| &p.functions),
364        );
365        let types: Vec<_> = self
366            .packages
367            .iter()
368            .flat_map(|p| p.classes.iter().map(|c| c.ty().clone()))
369            .chain(self.error_classes.iter().map(|(ty, _)| ty.clone()))
370            .chain(self.expansions.iter().filter_map(|d| match d {
371                prebindgen_declarations::ExpandDecl::Error(d) => Some(d.ty.clone()),
372                _ => None,
373            }))
374            .collect();
375        let mut flat = match (self.flat, self.source, self.items) {
376            (Some(flat), None, None) => flat,
377            (None, Some(path), None) => Flat::parse_selected(
378                prebindgen::Source::new(path).items_all(),
379                &references,
380                &types,
381            )?,
382            (None, None, Some(items)) => Flat::parse_selected(items, &references, &types)?,
383            _ => return Err("supply exactly one source, item stream or flat model".into()),
384        };
385        for configure in self.configure {
386            configure(&mut flat)?;
387        }
388        let defaults = prebindgen_declarations::Declarations::new(&flat, self.expansions)?;
389        validate_package(&self.prefix)?;
390        let mut classes = BTreeMap::new();
391        let mut functions = Vec::new();
392        let mut declarations = Vec::new();
393        let mut seen = BTreeSet::new();
394        let mut bound = BTreeSet::new();
395        for p in self.packages {
396            let pkg = if p.name.is_empty() {
397                self.prefix.clone()
398            } else {
399                format!("{}.{}", self.prefix, p.name)
400            };
401            validate_package(&pkg)?;
402            for c in p.classes {
403                let ty = flat.resolve_type(c.ty())?;
404                let name = match c.name() {
405                    Some(n) => n.clone(),
406                    None => match c.ty() {
407                        syn::Type::Path(p) => p.path.segments.last().unwrap().ident.to_string(),
408                        _ => return Err("class needs a name".into()),
409                    },
410                };
411                validate_name(&name)?;
412                if [
413                    "JNINative",
414                    "Functions",
415                    "NativeHandle",
416                    "JniErrorHandler",
417                    "HandleValidationFailure",
418                ]
419                .contains(&name.as_str())
420                {
421                    return Err("reserved generated class name".into());
422                }
423                let fqn = format!("{pkg}.{name}");
424                if !seen.insert(fqn.clone()) {
425                    return Err(format!("duplicate foreign name `{fqn}`"));
426                }
427                if classes
428                    .insert(
429                        type_key(&ty),
430                        Class {
431                            pkg: pkg.clone(),
432                            name,
433                            ty,
434                            data: matches!(c, ClassDecl::Data(_)),
435                            enumeration: matches!(c, ClassDecl::Enum(_)),
436                            variant_names: match c {
437                                ClassDecl::Enum(c) => c.variant_names,
438                                _ => Vec::new(),
439                            },
440                        },
441                    )
442                    .is_some()
443                {
444                    return Err("duplicate class for source type".into());
445                }
446            }
447            for f in p.functions {
448                declarations.push(f.clone());
449                let name = f.foreign.unwrap_or_else(|| camel(&f.name));
450                validate_name(&name)?;
451                if !seen.insert(format!("{pkg}.{name}")) {
452                    return Err("duplicate foreign function name".into());
453                }
454                if !bound.insert(f.name.clone()) {
455                    return Err("binding one source function twice is outside this subset".into());
456                }
457                functions.push((f.name, pkg.clone(), name));
458            }
459        }
460        let mut codec = data::DataCodec::new(&flat, &classes)?;
461        for (ty, class) in self.error_classes {
462            let mut parts = class.split('$');
463            validate_package(parts.next().unwrap())?;
464            for nested in parts {
465                validate_name(nested)?;
466            }
467            let ty = flat.resolve_type(&ty)?;
468            if codec.error_classes.insert(type_key(&ty), class).is_some() {
469                return Err("duplicate exception class for source error type".into());
470            }
471        }
472        for enumeration in codec.enums.values() {
473            flat.guards.extend(enumeration.mapping.guards());
474        }
475        let mut errors = errors::prepare(
476            &mut flat,
477            &mut codec,
478            &defaults,
479            &declarations,
480            &classes,
481            &self.prefix,
482            &mut seen,
483        )?;
484        for declaration in declarations {
485            defaults.fill(&mut flat, &codec, &declaration, &())?;
486            if declaration.callback_return.is_some() {
487                let variant_name = format!("{}_fold", declaration.name);
488                let wire = sequence::callback_target(
489                    &(flat.function(&declaration.name)?.output_conversion()?)
490                        .output_outcomes()
491                        .success,
492                )?;
493                let variant = defaults.callback_variant(&mut flat, &declaration, &variant_name)?;
494                codec.callback_wire = Some(wire);
495                defaults.fill(&mut flat, &codec, &variant, &())?;
496                codec.callback_wire = None;
497                if let Some(error) = errors.get(&declaration.name).cloned() {
498                    errors.insert(variant_name.clone(), error);
499                }
500                let (_, pkg, name) = functions
501                    .iter()
502                    .find(|(source, _, _)| source == &declaration.name)
503                    .unwrap();
504                let (pkg, name) = (pkg.clone(), format!("{name}Fold"));
505                if !seen.insert(format!("{pkg}.{name}")) {
506                    return Err("callback return foreign name collision".into());
507                }
508                functions.push((variant_name.clone(), pkg, name));
509                bound.insert(variant_name.clone());
510                if declaration.callback_return
511                    == Some(prebindgen_declarations::CallbackReturn::Only)
512                {
513                    functions.retain(|(source, _, _)| source != &declaration.name);
514                    let original = flat.functions.get_mut(&declaration.name).unwrap();
515                    original.ret.slot = None;
516                    for param in &mut original.params {
517                        param.ty.slot = None;
518                    }
519                }
520            }
521        }
522        prebindgen_flat::validate_bindings(&flat)?;
523        for (name, f) in &flat.functions {
524            if f.ret.slot.is_some() && !bound.contains(name) {
525                return Err(format!(
526                    "configured function `{name}` has no Kotlin declaration"
527                ));
528            }
529        }
530        let mut data_files = BTreeMap::new();
531        for class in classes.values().filter(|class| class.data) {
532            data_files.insert(
533                format!("{}/{}.kt", class.pkg.replace('.', "/"), class.name),
534                codec.render(class)?,
535            );
536        }
537        for class in classes.values().filter(|class| class.enumeration) {
538            data_files.insert(
539                format!("{}/{}.kt", class.pkg.replace('.', "/"), class.name),
540                codec.enums[&type_key(&class.ty)].render(class)?,
541            );
542        }
543        classes.retain(|_, class| !class.data && !class.enumeration);
544        generate(
545            flat,
546            &self.prefix,
547            &self.init,
548            classes,
549            functions,
550            data_files,
551            errors,
552        )
553    }
554}

568/// Conversions and slot glue used by [`JniGen`].
569/// Prepared code explicitly references the emission context's JNI environment
570/// and ownership acknowledgement parameter. Output delivery owns its result local.
571/// The writer supplies the error boundary and matching Kotlin protocol.
572/// Use with [`prebindgen_declarations::bind_input`] and [`prebindgen_declarations::bind_output`]
573/// to fill [`Bindings`] manually.
574pub struct JniCodec;
575impl Codec for JniCodec {
576    type Niche = crate::optional::JniNiche;
577    type Optional = JniOptional;
578    type Sequence = JniSequence;
579    type Group = JniGroup;
580    type Alternatives = JniSelector;
581    fn group_metadata(&self) -> JniGroup {
582        JniGroup::Values
583    }
584
585    fn binding_failure_layout(
586        &self,
587        _: &(),
588    ) -> Result<
589        Option<
590            prebindgen_flat::WireNode<
591                Self::Wire,
592                Self::Optional,
593                Self::Sequence,
594                Self::Group,
595                Self::Alternatives,
596            >,
597        >,
598    > {
599        Ok(Some(prebindgen_flat::WireNode::Wire(vec![
600            prebindgen_flat::Wire {
601                name: syn::parse_quote!(message),
602                kind: JniWire::text(),
603            },
604        ])))
605    }
606    fn optional_metadata(
607        &self,
608        presence: syn::Ident,
609        discriminator: syn::Ident,
610        niche: Option<JniNiche>,
611        controls: Vec<prebindgen_flat::Wire<JniWire>>,
612    ) -> Result<JniOptional> {
613        JniOptional::from_parts(presence, discriminator, niche, controls)
614    }
615    fn alternatives_metadata(
616        &self,
617        selector: prebindgen_flat::Wire<JniWire>,
618    ) -> Result<JniSelector> {
619        JniSelector::try_from(selector)
620    }

632    fn finish_input(&self, plan: &mut InputPlan<JniWire>, _: &()) -> Result<()> {
633        group::validate_layout(&plan.node)?;
634        let env = prebindgen_conversions::template_parameter("environment");
635        let ownership = prebindgen_conversions::template_parameter("acknowledgement");
636        if plan.resources.iter().any(|r| r.access == Access::Move) {
637            plan.acknowledgement = Some(quote!(
638                if #env.get_array_length(&#ownership).map_err(|e|e.to_string())? != 1 {
639                    return Err("invalid ownership acknowledgement".into());
640                }
641                #env.set_boolean_array_region(&#ownership, 0, &[1]).map_err(|e|e.to_string())?;
642            ));
643        }
644        Ok(())
645    }
646    fn finish_output(&self, plan: &mut OutputPlan<JniWire>, _: &()) -> Result<()> {
647        plan.fallible = true;
648        group::validate_layout(&plan.outcomes.success)?;
649        let env = prebindgen_conversions::template_parameter("environment");
650        let count = crate::group::result_packet_fields(&plan.outcomes.success).len() + 1;
651        let encode = encode_array(
652            crate::group::result_packet(&plan.outcomes.success, &plan.values)?,
653            &syn::parse_quote!(result),
654            1,
655        );
656        let mark = prebindgen_conversions::template_parameter("mark_output");
657        plan.commit
658            .extend(prebindgen_conversions::template_parameter(
659                "output_committed",
660            ));
661        plan.delivery = prebindgen_conversions::Delivery {
662            statements: quote!(let result = ::prebindgen_jni_runtime::array(&mut #env, #count)?; #encode
663                #mark),
664            result: quote!(result.into_raw()),
665        };
666        Ok(())
667    }

737    fn input_leaf(
738        &self,
739        ty: &syn::Type,
740        _: &(),
741        w: syn::Ident,
742        v: syn::Ident,
743        active: &prebindgen_declarations::Condition,
744    ) -> Result<InputPlan<JniWire>> {
745        if prebindgen_flat::sequence_parts(ty).is_some() {
746            return Err("collections require a Sequence recipe, not an opaque leaf; use an alias for opaque storage".into());
747        }
748        if let Some(inner) = prebindgen_declarations::option_inner(ty) {
749            let tag = format_ident!("{w}_present");
750            let child = self.input_leaf(
751                inner,
752                &(),
753                format_ident!("{w}_some"),
754                format_ident!("{v}_some"),
755                &active.arm(tag.clone(), 1),
756            )?;
757            return prebindgen_declarations::optional_input(self, &(), tag, v, active, child);
758        }
759        let env = prebindgen_conversions::template_parameter("environment");
760        let (decode, kind) = match Value::of(ty) {
761            Some(Value::Scalar(p)) => (scalar::decode(p, &w), JniWire::scalar(p)),
762            Some(Value::Text) => (
763                quote!(::prebindgen_jni_runtime::text(&mut #env,&#w)?),
764                JniWire::text(),
765            ),
766            Some(Value::Bytes) => (
767                quote!(::prebindgen_jni_runtime::bytes(&mut #env,&#w)?),
768                JniWire::bytes(),
769            ),
770            None => {
771                return prebindgen_conversions::boxed::input(
772                    &storage(),
773                    ty,
774                    w,
775                    v,
776                    &active.rust(),
777                    |b| JniWire::handle(b, kotlin_condition(active)),
778                )
779            }
780        };
781        prebindgen_conversions::value::input(ty, w, v, &active.rust(), decode, kind)
782    }
783    fn output_leaf(
784        &self,
785        ty: &syn::Type,
786        _: &(),
787        w: syn::Ident,
788        v: TokenStream,
789        borrowed: bool,
790    ) -> Result<OutputPlan<JniWire>> {
791        if prebindgen_flat::sequence_parts(ty).is_some() {
792            return Err("collections require a Sequence recipe, not an opaque leaf; use an alias for opaque storage".into());
793        }
794        if let syn::Type::Reference(reference) = ty {
795            if reference.mutability.is_some() {
796                return Err("mutable-reference outputs are unsupported".into());
797            }
798            match type_key(&reference.elem).as_str() {
799                "str" => {
800                    return self.output_leaf(
801                        &syn::parse_quote!(String),
802                        &(),
803                        w,
804                        quote!((#v).to_string()),
805                        false,
806                    )
807                }
808                "[u8]" => {
809                    return self.output_leaf(
810                        &syn::parse_quote!(Vec<u8>),
811                        &(),
812                        w,
813                        quote!((#v).to_vec()),
814                        false,
815                    )
816                }
817                _ => {}
818            }
819            let value = if borrowed { quote!(*#v) } else { v };
820            return self.output_leaf(&reference.elem, &(), w, value, true);
821        }
822        if let Some(inner) = prebindgen_declarations::option_inner(ty) {
823            let tag = format_ident!("{w}_present");
824            let payload = format_ident!("{w}_payload");
825            let child = self.output_leaf(
826                inner,
827                &(),
828                format_ident!("{w}_some"),
829                quote!(#payload),
830                borrowed,
831            )?;
832            return prebindgen_declarations::optional_output(
833                self,
834                &(),
835                tag,
836                payload,
837                v,
838                borrowed,
839                child,
840            );
841        }
842        // Text and bytes stay optional Rust payloads until delivery. This keeps
843        // a null reference distinct from a present empty string or byte array.
844        let (encode, kind): (fn(TokenStream) -> TokenStream, _) = match Value::of(ty) {
845            Some(Value::Scalar(p)) => {
846                return prebindgen_conversions::value::output(
847                    ty,
848                    w,
849                    v,
850                    borrowed,
851                    |v| scalar::encode(p, v),
852                    JniWire::scalar(p),
853                )
854            }
855            Some(Value::Text) => (|v| quote!(Some(#v)), JniWire::text()),
856            Some(Value::Bytes) => (|v| quote!(Some(#v)), JniWire::bytes()),
857            None => {
858                return prebindgen_conversions::boxed::output(&storage(), ty, w, v, borrowed, |b| {
859                    JniWire::handle(b, "true")
860                })
861            }
862        };
863        prebindgen_conversions::value::output(ty, w, v, borrowed, encode, kind)
864    }

941pub(crate) fn encode_array(
942    fields: Vec<(&prebindgen_flat::Wire<JniWire>, &TokenStream)>,
943    array: &syn::Ident,
944    offset: usize,
945) -> TokenStream {
946    let env = prebindgen_conversions::template_parameter("environment");
947    let mut encode = TokenStream::new();
948    for (i, (wire, value)) in fields.into_iter().enumerate() {
949        let expr = wire.kind.output_object(value);
950        let i = (i + offset) as i32;
951        encode.extend(quote! {
952            let object = #expr;
953            #env.set_object_array_element(&#array, #i, &object).map_err(|e|e.to_string())?;
954            #env.delete_local_ref(object).map_err(|e|e.to_string())?;
955        });
956    }
957
958    encode
959}

965/// Another name for the generated Rust and Kotlin sources in [`JniGen`].
966pub type Generation = JniGen;
967impl JniGen {
968    /// Write native Rust source, returning the destination path.
969    ///
970    /// Relative paths resolve under Cargo's `OUT_DIR`; absolute paths are used
971    /// directly. Creates parent directories and overwrites an existing file.
972    /// Returns an error if `OUT_DIR` is needed but absent, or filesystem access fails.
973    pub fn write_rust(&self, path: impl AsRef<std::path::Path>) -> Result<std::path::PathBuf> {
974        let path = path.as_ref();
975        let path = if path.is_absolute() {
976            path.to_owned()
977        } else {
978            std::path::PathBuf::from(std::env::var_os("OUT_DIR").ok_or("OUT_DIR is not set")?)
979                .join(path)
980        };
981        if let Some(parent) = path.parent() {
982            std::fs::create_dir_all(parent).map_err(|e| e.to_string())?;
983        }
984        std::fs::write(&path, &self.rust).map_err(|e| e.to_string())?;
985        Ok(path)
986    }
987    /// Write all Kotlin files beneath `root`, creating package directories.
988    ///
989    /// Relative roots use the current directory, not `OUT_DIR`. Existing generated
990    /// paths are overwritten; stale files are not removed. Returns the written paths
991    /// or a filesystem error. Earlier writes remain if a later write fails.
992    pub fn write_kotlin(&self, root: &std::path::Path) -> Result<Vec<std::path::PathBuf>> {
993        let mut written = Vec::new();
994        for (path, text) in &self.kotlin {
995            let path = root.join(path);
996            std::fs::create_dir_all(path.parent().unwrap()).map_err(|e| e.to_string())?;
997            std::fs::write(&path, text).map_err(|e| e.to_string())?;
998            written.push(path);
999        }
1000        Ok(written)
1001    }
1002}

1009            "class",

1052fn camel(s: &str) -> String {
1053    let mut out = String::new();
1054    let mut upper = false;
1055    for c in s.chars() {
1056        if c == '_' {
1057            upper = true;
1058        } else if upper {
1059            out.extend(c.to_uppercase());
1060            upper = false;
1061        } else {
1062            out.push(c);
1063        }
1064    }
1065    out
1066}
1067/// JNI escaping is defined over UTF-16, including underscores in identifiers.
1068fn escape(s: &str) -> String {
1069    let mut out = String::new();
1070    for c in s.encode_utf16() {
1071        match c {
1072            46 | 47 => out.push('_'),
1073            95 => out.push_str("_1"),
1074            59 => out.push_str("_2"),
1075            91 => out.push_str("_3"),
1076            c if (c as u8).is_ascii_alphanumeric() && c < 128 => out.push(char::from(c as u8)),
1077            c => out.push_str(&format!("_0{c:04x}")),
1078        }
1079    }
1080    out
1081}

1085        escape(&format!("{pkg}.JNINative")),
1086        escape(method)
1087    )
1088}

1090fn generate(
1091    flat: Bindings,
1092    prefix: &str,
1093    init: &str,
1094    classes: BTreeMap<String, Class>,
1095    functions: Vec<(String, String, String)>,
1096    mut files: BTreeMap<String, String>,
1097    errors: BTreeMap<String, errors::ErrorBinding>,
1098) -> Result<JniGen> {
1099    let mut rust = errors::emit(&errors)?;
1100    let mut packages: BTreeMap<String, String> = BTreeMap::new();
1101    errors::interfaces(prefix, &mut files)?;
1102    let mut native = format!("package {prefix}\n\ninternal object JNINative {{\n");
1103    if !init.is_empty() {
1104        native.push_str(&format!("    init {{ {init} }}\n"));
1105    }
1106    files.insert(
1107        format!("{}/NativeHandle.kt", prefix.replace('.', "/")),
1108        native_handle(prefix),
1109    );
1110    for (index, c) in classes.values().enumerate() {
1111        let method = format!("drop{index}");
1112        let sym = symbol(prefix, &method);
1113        let ty = &c.ty;
1114        rust.extend(quote!{#[doc = "# Safety\nOnly generated Kotlin wrappers may call this native entry point, with their handle locks held and live pointers of the declared types."]
1115#[no_mangle]pub unsafe extern "system" fn #sym(mut env: ::prebindgen_jni_runtime::jni::JNIEnv<'_>,_class: ::prebindgen_jni_runtime::jni::objects::JClass<'_>,id:i64){if let Err(e)=::prebindgen_runtime::boundary(||unsafe {::prebindgen_jni_runtime::drop_pointer::<#ty>(id)}){::prebindgen_jni_runtime::fail(&mut env,&e);}}});
1116        native.push_str(&format!("    @JvmStatic external fun {method}(id: Long)\n"));
1117        let text=format!("package {}\n\npublic class {} internal constructor(raw: Long) : {prefix}.NativeHandle(raw) {{\n    public fun take(): {} = lock.run {{ lock(); try {{ check(!isClosed) {{ \"closed or consumed handle\" }}; val owner = {}(raw); markConsumed(); owner }} finally {{ unlock() }} }}\n    override fun freeNative(raw: Long) {{ {prefix}.JNINative.{method}(raw) }}\n}}\n",c.pkg,c.name,c.name,c.name);
1118        files.insert(format!("{}/{}.kt", c.pkg.replace('.', "/"), c.name), text);
1119    }
1120    let kotlin_classes = classes
1121        .iter()
1122        .enumerate()
1123        .map(|(index, (ty, class))| {
1124            (
1125                ty.clone(),
1126                KotlinClass {
1127                    name: class.fqn(),
1128                    drop_function: format!("{prefix}.JNINative.drop{index}"),
1129                },
1130            )
1131        })
1132        .collect();
1133    errors::handler_files(
1134        &errors,
1135        &KotlinContext {
1136            classes: &kotlin_classes,
1137            result: "result",
1138            failure: "failure",
1139            cleanup_failure: None,
1140            output_prefix: "errorOutput",
1141        },
1142        &mut files,
1143    )?;
1144    for (index, (source, pkg, name)) in functions.into_iter().enumerate() {
1145        let f = flat.function(&source)?;
1146        let callback =
1147            sequence::callback_delivery(&f.output_conversion()?.output_outcomes().success)?;
1148        let error = errors.get(&source);
1149        let outcomes = f.output_conversion()?.output_outcomes();
1150        let logical = &outcomes.success;
1151        for node in std::iter::once(logical)
1152            .chain(outcomes.source_error.iter())
1153            .chain(outcomes.binding_failure.iter())
1154        {
1155            group::validate_layout(node)?;
1156        }
1157        for error in outcomes
1158            .source_error
1159            .iter()
1160            .chain(outcomes.binding_failure.iter())
1161        {
1162            sequence::validate_buffered(error)?;
1163        }
1164        let has_source_error = outcomes.source_error.is_some();
1165        let has_binding_failure = outcomes.binding_failure.is_some();
1166        if has_source_error != error.is_some() {
1167            return Err("JNI source-error schema must match its configured error converter".into());
1168        }
1169        // Every JNI wrapper has a separate binding-error callback, including
1170        // functions without a source Result. Manual schemas must declare it.
1171        if !has_binding_failure {
1172            return Err("JNI native delivery requires a declared binding-failure schema".into());
1173        }
1174
1175        let throws = error.map_or_else(String::new, |error| error.throws());
1176        let env = format_ident!("jni_environment");
1177        let ownership = format_ident!("ownership_accepted");
1178        let context = JniContext {
1179            environment: env.clone(),
1180            acknowledgement: ownership.clone(),
1181            domain_output: format_ident!("domain_output"),
1182            is_domain_error: false,
1183            output_committed: quote!(),
1184        };
1185        let source = format_ident!("source_value");
1186        let method = format!("call{index}");
1187        let sym = symbol(prefix, &method);
1188        let mut params = Vec::new();
1189        let mut raw_params = Vec::new();
1190        let kotlin_context = KotlinContext {
1191            classes: &kotlin_classes,
1192            result: "result",
1193            failure: "failure",
1194            cleanup_failure: None,
1195            output_prefix: "output",
1196        };
1197        let mut kt_params = Vec::new();
1198        let mut kt_args = Vec::new();
1199        let mut locks = Vec::new();
1200        let mut lock_groups = Vec::new();
1201        let mut setup = String::new();
1202        let mut consumed = Vec::new();
1203        for w in crate::group::function_parameters(f)? {
1204            let n = &w.name;
1205            let rt = w.kind.rust_type();
1206            params.push(quote!(#n:#rt));
1207            raw_params.push(format!("{n}: {}", w.kind.kotlin_type()));
1208        }
1209        for input in f.input_conversions()? {
1210            let code = input.kotlin(&kotlin_context)?;
1211            kt_params.extend(code.parameters);
1212            kt_args.extend(code.arguments);
1213            locks.extend(code.locks);
1214            lock_groups.extend(code.lock_groups);
1215            for binding in code.setup {
1216                setup.push_str(&format!("val {} = {}\n", binding.name, binding.expression));
1217            }
1218            consumed.extend(code.consumed);
1219        }
1220        if kt_args.len() != crate::group::function_parameters(f)?.len() {
1221            return Err("Kotlin input argument/parameter count differs".into());
1222        }
1223        let has_consumes = !consumed.is_empty();
1224        let has_acknowledgement = f
1225            .input_conversions()?
1226            .iter()
1227            .map(|i| {
1228                i.code_fragment_slot(
1229                    prebindgen_conversions::InputFragment::Acknowledge { context: &context },
1230                    &syn::parse_quote!(()),
1231                )
1232                .map(|code| code.to_string() != quote!(()).to_string())
1233            })
1234            .collect::<Result<Vec<_>>>()?
1235            .into_iter()
1236            .any(|present| present);
1237        if has_acknowledgement != has_consumes {
1238            return Err("JNI acknowledgement does not match slot ownership updates".into());
1239        }
1240        if has_consumes {
1241            params.push(
1242                quote!(#ownership: ::prebindgen_jni_runtime::jni::objects::JBooleanArray<'local>),
1243            );
1244            raw_params.push("ownership: BooleanArray".into());
1245            kt_args.push("ownership".into());
1246        }
1247        if callback {
1248            let (_, element) =
1249                sequence::return_sequence(&f.output_conversion()?.output_outcomes().success)
1250                    .ok_or("missing callback sequence")?;
1251            let fields = crate::group::row_fields(element)
1252                .enumerate()
1253                .map(|(i, w)| format!("v{i}: {}", w.kind.kotlin_type()))
1254                .collect::<Vec<_>>();
1255            native.push_str(&format!(
1256                "    internal fun interface SequenceSink{index} {{ fun run({}) }}\n",
1257                fields.join(", ")
1258            ));
1259            params.push(
1260                quote!(__pb_callback: ::prebindgen_jni_runtime::jni::objects::JObject<'local>),
1261            );
1262            raw_params.push(format!("callback: SequenceSink{index}"));
1263            let names = (0..fields.len())
1264                .map(|i| format!("v{i}"))
1265                .collect::<Vec<_>>()
1266                .join(", ");
1267            setup.push_str(&format!("val __pb_rows = kotlin.collections.mutableListOf<kotlin.Array<kotlin.Any?>>()\nval __pb_sink = {prefix}.JNINative.SequenceSink{index} {{ {names} -> __pb_rows.add(kotlin.arrayOf<kotlin.Any?>({names})); kotlin.Unit }}\n"));
1268            kt_args.push("__pb_sink".into());
1269            params.push(quote!(__pb_output_accepted: ::prebindgen_jni_runtime::jni::objects::JBooleanArray<'local>));
1270            raw_params.push("outputAccepted: BooleanArray".into());
1271            setup.push_str("val __pb_outputAccepted = BooleanArray(1)\n");
1272            kt_args.push("__pb_outputAccepted".into());
1273        }
1274        let slots: usize = usize::from(has_consumes)
1275            + 2 * usize::from(callback)
1276            + crate::group::function_parameters(f)?
1277                .map(|w| w.kind.parameter_slots())
1278                .sum::<usize>();
1279        if slots > 255 {
1280            return Err(format!(
1281                "`{source}` needs {slots} JVM parameter slots; the limit is 255"
1282            ));
1283        }
1284        let body = prebindgen_conversions::staged::emit_call(
1285            f,
1286            &context,
1287            &source,
1288            &syn::parse_quote!(::prebindgen_jni_runtime::jni::sys::jobjectArray),
1289            &crate::group::function_parameters(f)?
1290                .map(|w| w.name.clone())
1291                .collect::<Vec<_>>(),
1292        )?;
1293        let body = if callback {
1294            quote!(if #env.get_array_length(&__pb_output_accepted).map_err(|e|e.to_string())? != 1 {return Err("invalid sequence output acknowledgement".into());} if __pb_callback.is_null(){return Err("null sequence callback".into());} #body)
1295        } else {
1296            body
1297        };
1298        let fold = if callback {
1299            Some(sequence_callback::fold(
1300                &f.output_conversion()?.output_outcomes().success,
1301                &kotlin_context,
1302            )?)
1303        } else {
1304            None
1305        };
1306        if let Some(fold) = &fold {
1307            setup.push_str(&fold.discard);
1308        }
1309        let mut kotlin_output = match &fold {
1310            Some(fold) => fold.output.clone(),
1311            None => f.output_conversion()?.kotlin(&kotlin_context)?,
1312        };
1313        kotlin::remap_output(
1314            &mut kotlin_output,
1315            kotlin_context.result,
1316            kotlin_context.result,
1317            1,
1318            crate::group::result_packet_fields(&f.output_conversion()?.output_outcomes().success)
1319                .len(),
1320        );
1321        let KotlinOutput {
1322            delivered,
1323            result_types,
1324            output_locals,
1325            materialize,
1326            cleanup,
1327        } = kotlin_output;
1328        if delivered.len() != result_types.len() {
1329            return Err("Kotlin output expression/type count differs".into());
1330        }
1331        rust.extend(quote!{
1332   #[allow(clippy::needless_borrow, clippy::borrow_deref_ref, clippy::let_unit_value, clippy::unnecessary_cast, clippy::useless_conversion)]
1333   #[doc = "# Safety\nOnly generated Kotlin wrappers may call this native entry point, with their handle locks held and live pointers of the declared types."]
1334#[no_mangle]pub unsafe extern "system" fn #sym<'local>(mut #env: ::prebindgen_jni_runtime::jni::JNIEnv<'local>,_class: ::prebindgen_jni_runtime::jni::objects::JClass<'local>,#(#params),*)->::prebindgen_jni_runtime::jni::sys::jobjectArray{
1335    #[allow(unused_mut)]
1336    let mut domain_output = None;
1337    match ::prebindgen_runtime::boundary(||{
1338     #body
1339    }){Ok(result)=>result,Err(e)=>{
1340        if let Some(result) = domain_output { result } else {
1341            match ::prebindgen_jni_runtime::binding_packet(&mut #env, &e) {
1342                Ok(result) => result.into_raw(),
1343                Err(message) => { ::prebindgen_jni_runtime::fail(&mut #env, &message); std::ptr::null_mut() }
1344            }
1345        }
1346    }}
1347   }
1348  });
1349        native.push_str(&format!(
1350            "    @JvmStatic external fun {method}({}): Array<Any?>\n",
1351            raw_params.join(", ")
1352        ));
1353        let (mut generic, ret, end) = match delivered.len() {
1354            0 => ("", "Unit".into(), "Unit".into()),
1355            1 => ("", result_types[0].clone(), delivered[0].clone()),
1356            _ => {
1357                kt_params.push(format!("build: ({}) -> R", result_types.join(", ")));
1358                (
1359                    "<R> ",
1360                    "R".into(),
1361                    format!("build({})", delivered.join(", ")),
1362                )
1363            }
1364        };
1365        if fold.is_some() {
1366            generic = "<A> ";
1367            kt_params.push("initial: A".into());
1368        }
1369        let failure = kotlin_context.failure;
1370        let result = kotlin_context.result;
1371        let end = if cleanup.is_empty() {
1372            end
1373        } else {
1374            format!("{output_locals}        try {{\n{materialize}            {end}\n        }} catch ({failure}: Throwable) {{\n{cleanup}            throw {failure}\n        }}")
1375        };
1376        let argument_locals = kt_args.iter().enumerate().map(|(i, arg)| format!("val nativeArgument{i} = try {{ {arg} }} catch (failure: IllegalArgumentException) {{ throw {prefix}.HandleValidationFailure(failure.message ?: \"invalid input\") }}")).collect::<Vec<_>>().join("; ");
1377        let argument_names = (0..kt_args.len())
1378            .map(|i| format!("nativeArgument{i}"))
1379            .collect::<Vec<_>>()
1380            .join(", ");
1381        let call =
1382            format!("run {{ {argument_locals}; {prefix}.JNINative.{method}({argument_names}) }}");
1383        let call = if has_consumes {
1384            format!("val ownership = BooleanArray(1)\n        try {{ {call} }} finally {{ if (ownership[0]) {{ {} }} }}", consumed.join("; "))
1385        } else {
1386            call
1387        };
1388        let binding_name = if error.is_some() {
1389            "onBindingError"
1390        } else {
1391            "onError"
1392        };
1393        let build = if delivered.len() > 1 {
1394            kt_params.pop()
1395        } else {
1396            None
1397        };
1398        kt_params.push(format!("{binding_name}: {prefix}.JniErrorHandler<{ret}> = {prefix}.JniErrorHandler {{ message -> throw IllegalArgumentException(message) }}"));
1399        if let Some(error) = error {
1400            kt_params.push(error.parameter(&ret));
1401        }
1402        if let Some(build) = build {
1403            kt_params.push(build);
1404        }
1405        if let Some(fold) = &fold {
1406            kt_params.push(fold.parameter.clone());
1407        }
1408        let call = if locks.is_empty() && lock_groups.is_empty() {
1409            call
1410        } else {
1411            let mut all_locks = if locks.is_empty() {
1412                "kotlin.collections.emptyList<Nothing>()".to_string()
1413            } else {
1414                format!("listOfNotNull({})", locks.join(", "))
1415            };
1416            for group in lock_groups {
1417                all_locks = format!("({all_locks} + ({group}))");
1418            }
1419            format!("run {{ val handles = try {{ {all_locks} }} catch (failure: IllegalArgumentException) {{ throw {prefix}.HandleValidationFailure(failure.message ?: \"invalid handle\") }}; {prefix}.withHandleLocks(handles) {{\n        {call}\n    }} }}")
1420        };
1421        let call = if callback {
1422            format!("try {{ {call} }} catch (failure: Throwable) {{ if (__pb_outputAccepted[0]) __pb_discard(failure); throw failure }}")
1423        } else {
1424            call
1425        };
1426        // Catch local handle validation separately from the native invocation: pending JVM
1427        // exceptions and exceptions from user callbacks must retain their original identity.
1428        let call = format!("try {{ {call} }} catch (failure: {prefix}.HandleValidationFailure) {{ return@run {binding_name}.run(failure.message) }}");
1429        let domain = if let Some(error) = error {
1430            let delivery = errors::delivery(error, &kotlin_context)?;
1431            format!("true -> {{ {delivery} }}")
1432        } else {
1433            String::new()
1434        };
1435        let callback_failure = if callback {
1436            format!("if (__pb_outputAccepted[0]) __pb_discard(IllegalArgumentException({result}[0] as String)); ")
1437        } else {
1438            String::new()
1439        };
1440        let native_call = format!("val {result} = {call}\n    when ({result}[0]) {{\n        null -> {{ {end} }}\n        {domain}\n        is String -> {{ {callback_failure}{binding_name}.run({result}[0] as String) }}\n        else -> error(\"invalid native result channel\")\n    }}");
1441        let text = packages.entry(pkg).or_default();
1442        text.push_str(&format!(
1443            "{throws}public fun {generic}`{name}`({}): {ret} = run {{\n    {setup}{native_call}\n}}\n\n",
1444            kt_params.join(", ")
1445        ));
1446    }
1447    native.push_str("}\n");
1448    files.insert(format!("{}/JNINative.kt", prefix.replace('.', "/")), native);
1449    for (pkg, body) in packages {
1450        let path = format!("{}/Functions.kt", pkg.replace('.', "/"));
1451        if files
1452            .insert(
1453                path,
1454                format!("@file:Suppress(\"UNUSED_VARIABLE\", \"UNCHECKED_CAST\", \"UNNECESSARY_NOT_NULL_ASSERTION\")\npackage {pkg}\n\n{body}"),
1455            )
1456            .is_some()
1457        {
1458            return Err("Kotlin file collision".into());
1459        }
1460    }
1461    let guards = &flat.guards;
1462    rust.extend(quote!(#(#guards)*));
1463    Ok(JniGen {
1464        rust: prebindgen_flat::rust_source(rust)?,
1465        kotlin: files,
1466    })
1467}
