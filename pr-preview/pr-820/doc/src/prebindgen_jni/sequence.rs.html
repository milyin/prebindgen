8/// Selected delivery of a logical sequence.
9#[derive(Clone, Copy, Debug, PartialEq, Eq)]
10pub enum JniSequence {
11    /// All rows travel in one collection carrier.
12    Buffered,
13    /// Each row is delivered through the function's sequence callback.
14    Callback,
15}

17pub(crate) fn input(
18    wire: syn::Ident,
19    length: syn::Ident,
20    index: syn::Ident,
21    element: &prebindgen_flat::WireNode<
22        JniWire,
23        crate::JniOptional,
24        crate::JniSequence,
25        JniGroup,
26        (),
27    >,
28    active: &Condition,
29) -> Result<SequenceInput<JniWire, crate::JniSequence>> {
30    let env = prebindgen_conversions::template_parameter("environment");
31    let row = format_ident!("{wire}_row");
32    let count = crate::group::row_fields(element).len();
33    let capacity = i32::try_from(count + 8).map_err(|_| "too many collection element fields")?;
34    let frame = format_ident!("{wire}_frame");
35    let mut bind = quote! {let #frame=unsafe {::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;let #row=::prebindgen_jni_runtime::row(&mut #env,&#wire,#index,#count)?;};
36    let mut validation_values = Vec::new();
37    let mut prepared_values = Vec::new();
38    for (position, field) in crate::group::row_fields(element).enumerate() {
39        let name = &field.name;
40        let read = field.kind.read_array_element(&quote!(#row), position);
41        bind.extend(quote! {
42            let #name=#read;
43            if #name.is_err() && #env.exception_check().map_err(|e|e.to_string())? {return Err("JNI collection field decoding failed".into());}
44        });
45        let borrow = quote!(*);
46        validation_values.push((
47            field.name.clone(),
48            quote!(#borrow #name.as_ref().map_err(|e|e.clone())?),
49        ));
50        prepared_values.push((
51            field.name.clone(),
52            if field.kind.is_reference() {
53                None
54            } else {
55                Some(quote!(#borrow #name.as_ref().expect("validated collection field")))
56            },
57        ));
58    }
59    Ok(SequenceInput {
60        controls: vec![prebindgen_flat::Wire {
61            name: wire.clone(),
62            kind: JniWire::Rows {
63                condition: kotlin_condition(active),
64                null_niche_available: true,
65            },
66        }],
67        metadata: JniSequence::Buffered,
68        open: quote! {if #wire.is_null(){return Err("null collection".into());}let #length=#env.get_array_length(&#wire).map_err(|e|e.to_string())? as usize;},
69        bind,
70        validation_values,
71        prepared_values,
72        release: quote!(drop(#frame);),
73    })
74}
75pub(crate) fn output(
76    wire: syn::Ident,
77    length: syn::Ident,
78    index: syn::Ident,
79    element: &prebindgen_flat::WireNode<
80        JniWire,
81        crate::JniOptional,
82        crate::JniSequence,
83        JniGroup,
84        (),
85    >,
86    values: &[(syn::Ident, TokenStream)],
87) -> Result<SequenceOutput<JniWire, crate::JniSequence>> {
88    let env = prebindgen_conversions::template_parameter("environment");
89    let row = format_ident!("{wire}_row");
90    let count = crate::group::row_fields(element).len();
91    let capacity = i32::try_from(count + 8).map_err(|_| "too many collection fields")?;
92    let encode = encode_array(crate::group::row_values(element, values)?, &row, 0);
93    let frame = format_ident!("{wire}_frame");
94    Ok(SequenceOutput {
95        controls: vec![prebindgen_flat::Wire {
96            name: wire.clone(),
97            kind: JniWire::Rows {
98                condition: ("true").into(),
99                null_niche_available: true,
100            },
101        }],
102        commit_each: false,
103        fallible: true,
104        metadata: JniSequence::Buffered,
105        begin: quote!(let #wire=::prebindgen_jni_runtime::array(&mut #env,#length)?;),
106        enter: quote!(let #frame=unsafe {::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;),
107        store: quote! {
108            let #row=::prebindgen_jni_runtime::array(&mut #env,#count)?;#encode
109            #env.set_object_array_element(&#wire,#index as i32,&#row).map_err(|e|e.to_string())?;
110            drop(#frame);
111        },
112        finish: quote!(let #wire=Some(#wire);),
113        values: vec![(wire.clone(), quote!(#wire))],
114        commit: quote!(),
115    })
116}

118pub(crate) fn input_list(
119    carrier: &prebindgen_flat::Wire<JniWire>,
120    sequence: &JniSequence,
121    element: &prebindgen_flat::WireNode<
122        JniWire,
123        crate::JniOptional,
124        crate::JniSequence,
125        JniGroup,
126        (),
127    >,
128    context: &KotlinContext<'_>,
129) -> Result<KotlinInput> {
130    if *sequence != JniSequence::Buffered {
131        return Err(
132            "callback sequence is only supported as the function's successful return".into(),
133        );
134    }
135    let child = input_kotlin(element, context)?;
136    if child.parameters.len() != 1 {
137        return Err("Kotlin collection elements must expose exactly one value; declare a data class for expanded fields".into());
138    }
139    let (parameter, ty) = child.parameters[0]
140        .split_once(": ")
141        .ok_or("invalid element parameter")?;
142    let name = carrier.name.to_string();
143    let state = format!("{name}_elements");
144    let JniWire::Rows { condition, .. } = &carrier.kind else {
145        return Err("sequence requires an object-array control".into());
146    };
147    let mut code = String::new();
148    for binding in &child.setup {
149        code.push_str(&format!("val {} = {}\n", binding.name, binding.expression));
150    }
151    let mut locks = if child.locks.is_empty() {
152        "kotlin.collections.emptyList<Nothing>()".into()
153    } else {
154        format!(
155            "kotlin.collections.listOfNotNull({})",
156            child.locks.join(", ")
157        )
158    };
159    for group in &child.lock_groups {
160        locks = format!("({locks} + ({group}))");
161    }
162    let expression=format!("if ({condition}) {name}?.toList()?.map {{ {parameter} ->\n{code}kotlin.Triple({{ kotlin.arrayOf<kotlin.Any?>({}) }}, {{ {locks} }}, {{ {}; kotlin.Unit }})\n}} else null",child.arguments.join(", "),child.consumed.join("; "));
163    Ok(KotlinInput {
164        parameters: vec![format!(
165            "{name}: kotlin.collections.List<{ty}>{}",
166            if condition == "true" { "" } else { "?" }
167        )],
168        setup: vec![KotlinBinding {
169            name: state.clone(),
170            expression,
171            absent: "null".into(),
172        }],
173        arguments: vec![format!(
174            "{state}?.map {{ it.first() }}?.toTypedArray<kotlin.Any?>()"
175        )],
176        locks: Vec::new(),
177        lock_groups: if child.locks.is_empty() && child.lock_groups.is_empty() {
178            vec![]
179        } else {
180            vec![format!("{state}.orEmpty().flatMap {{ it.second() }}")]
181        },
182        consumed: if child.consumed.is_empty() {
183            vec![]
184        } else {
185            vec![format!("{state}.orEmpty().forEach {{ it.third() }}")]
186        },
187    })
188}

190pub(crate) fn output_list(
191    element: &prebindgen_flat::WireNode<
192        JniWire,
193        crate::JniOptional,
194        crate::JniSequence,
195        JniGroup,
196        (),
197    >,
198    context: &KotlinContext<'_>,
199) -> Result<KotlinOutput> {
200    let row = format!("{}_row", context.output_prefix);
201    let prefix = format!("{}item_", context.output_prefix);
202    let failure = format!("{}failure", context.output_prefix);
203    let ctx = KotlinContext {
204        failure: &failure,
205        cleanup_failure: None,
206        result: &row,
207        output_prefix: &prefix,
208        ..*context
209    };
210    let child = output_kotlin(element, &ctx)?;
211    if child.delivered.len() != 1 {
212        return Err("Kotlin collection elements must deliver exactly one value; declare a data class for expanded outputs".into());
213    }
214    let ty = &child.result_types[0];
215    let value = &child.delivered[0];
216    let raw = format!("({}[0] as kotlin.Array<*>)", context.result);
217    let output = format!("{}list", context.output_prefix);
218    let actions = format!("{}cleanups", context.output_prefix);
219    let (failure_value, finish_cleanup) = match context.cleanup_failure {
220        Some(target) => (
221            format!("{target} ?: kotlin.RuntimeException(\"collection cleanup failed\")"),
222            format!("if ({target} == null && {failure}.suppressed.isNotEmpty()) {{ {target} = {failure}.suppressed.first(); {failure}.suppressed.drop(1).forEach {{ {target}!!.addSuppressed(it) }} }}"),
223        ),
224        None => (context.failure.to_owned(), String::new()),
225    };
226    Ok(KotlinOutput {
227        delivered:vec![format!("requireNotNull({output})")],result_types:vec![format!("kotlin.collections.List<{ty}>")],
228        output_locals:format!("var {output}: kotlin.collections.MutableList<{ty}>? = null\nvar {actions}: kotlin.collections.MutableList<(kotlin.Throwable) -> kotlin.Unit>? = null\n"),
229        materialize:format!("{actions} = kotlin.collections.mutableListOf()\n{output} = kotlin.collections.mutableListOf()\nfor (rawRow in {raw}) {{\nval {row} = rawRow as kotlin.Array<*>\n{}\n{actions}.add {{ {failure} -> {} }}\n{}\n{output}.add({value})\n}}\n",child.output_locals,child.cleanup,child.materialize),
230        cleanup:format!("val {failure} = {failure_value}\n{actions}?.forEach {{ it({failure}) }}\nfor (index in ({actions}?.size ?: 0) until {raw}.size) {{\nval {row} = {raw}[index] as kotlin.Array<*>\n{}\n{}\n}}\n{finish_cleanup}\n",child.output_locals,child.cleanup),
231    })
232}

234/// The adapter selects its native scope before looking for a sequence.
235#[allow(clippy::type_complexity)]
236pub(crate) fn return_sequence(
237    layout: &prebindgen_flat::WireNode<JniWire, JniOptional, JniSequence, JniGroup, ()>,
238) -> Option<(
239    &prebindgen_flat::Wire<JniWire>,
240    &prebindgen_flat::WireNode<JniWire, JniOptional, JniSequence, JniGroup, ()>,
241    &JniSequence,
242)> {
243    prebindgen_declarations::outer_sequences(layout)
244        .into_iter()
245        .next()
246        .and_then(|(element, controls, metadata)| match controls {
247            [carrier] => Some((carrier, element, metadata)),
248            _ => None,
249        })
250}
251pub(crate) fn callback_target(
252    layout: &prebindgen_flat::WireNode<
253        JniWire,
254        crate::JniOptional,
255        crate::JniSequence,
256        JniGroup,
257        (),
258    >,
259) -> Result<syn::Ident> {
260    if prebindgen_declarations::outer_sequences(layout).len() != 1
261        || crate::group::result_packet_fields(layout).len() > 2
262    {
263        return Err(
264            "callback_return permits only one sequence and its optional presence flag; cannot discard additional return values".into(),
265        );
266    }
267    return_sequence(layout)
268        .map(|(carrier, _, _)| carrier.name.clone())
269        .ok_or_else(|| "callback_return requires a sequence boundary".into())
270}

272/// Whether the successful return uses the function's sequence callback.
273pub(crate) fn callback_delivery(
274    node: &prebindgen_flat::WireNode<JniWire, JniOptional, JniSequence, JniGroup>,
275) -> Result<bool> {
276    let sequences = prebindgen_declarations::outer_sequences(node);
277    if !sequences
278        .iter()
279        .any(|(_, _, metadata)| **metadata == JniSequence::Callback)
280    {
281        return Ok(false);
282    }
283    callback_target(node)?;
284    if crate::group::result_packet_fields(node).len() != 1 {
285        return Err("callback_return cannot discard additional return values".into());
286    }
287    Ok(true)
288}

290/// Function sequence callbacks are not available in parameters, error channels or rows.
291pub(crate) fn validate_buffered(
292    node: &prebindgen_flat::WireNode<JniWire, JniOptional, JniSequence, JniGroup>,
293) -> Result<()> {
294    match node {
295        prebindgen_flat::WireNode::Wire(_) => {}
296        prebindgen_flat::WireNode::Sequence {
297            element, metadata, ..
298        } => {
299            if *metadata == JniSequence::Callback {
300                return Err(
301                    "callback sequence is only supported as the function's successful return"
302                        .into(),
303                );
304            }
305            validate_buffered(element)?;
306        }
307        prebindgen_flat::WireNode::Optional { element, .. } => validate_buffered(element)?,
308        prebindgen_flat::WireNode::Group { elements, .. }
309        | prebindgen_flat::WireNode::Alternatives { elements, .. } => {
310            for element in elements {
311                validate_buffered(element)?;
312            }
313        }
314    }
315    Ok(())
316}

318#[cfg(test)]
319mod delivery_tests {
320    use super::*;
321    use prebindgen_flat::WireNode;
322    fn leaf() -> prebindgen_flat::WireNode<JniWire, JniOptional, JniSequence, JniGroup> {
323        WireNode::Wire(vec![prebindgen_flat::Wire {
324            name: syn::parse_quote!(element),
325            kind: JniWire::scalar(Primitive::I32),
326        }])
327    }
328    fn sequence(
329        delivery: JniSequence,
330        name: &str,
331        element: prebindgen_flat::WireNode<JniWire, JniOptional, JniSequence, JniGroup>,
332    ) -> prebindgen_flat::WireNode<JniWire, JniOptional, JniSequence, JniGroup> {
333        let kind = match delivery {
334            JniSequence::Buffered | JniSequence::Callback => JniWire::Rows {
335                condition: "true".into(),
336                null_niche_available: true,
337            },
338        };
339        WireNode::Sequence {
340            controls: vec![prebindgen_flat::Wire {
341                name: syn::parse_str(name).unwrap(),
342                kind,
343            }],
344            element: Box::new(element),
345            metadata: delivery,
346        }
347    }
348    #[test]
349    fn delivery_is_explicit_and_callback_scope_is_checked() {
350        let visitor = sequence(JniSequence::Callback, "visitor", leaf());
351        assert!(callback_delivery(&visitor).unwrap());
352        assert!(validate_buffered(&visitor)
353            .unwrap_err()
354            .contains("successful return"));
355        let packed = sequence(JniSequence::Buffered, "rows", leaf());
356        assert!(!callback_delivery(&packed).unwrap());
357        validate_buffered(&packed).unwrap();
358        let nested = sequence(JniSequence::Buffered, "rows", visitor.clone());
359        assert!(crate::group::validate_layout(&nested)
360            .unwrap_err()
361            .contains("successful return"));
362        let extra = WireNode::Group {
363            elements: vec![visitor, leaf()],
364            metadata: JniGroup::Values,
365        };
366        assert!(callback_delivery(&extra)
367            .unwrap_err()
368            .contains("additional return"));
369    }
