6pub(crate) fn transport(
7    wire: syn::Ident,
8    element: &[prebindgen_flat::WireNode<JniWire, JniGroup>],
9    values: &[(syn::Ident, TokenStream)],
10) -> Result<SequenceOutput<JniGroup>> {
11    let env = prebindgen_conversions::template_parameter("environment");
12    let method = format_ident!("{wire}_callback_method");
13    let class = format_ident!("{wire}_callback_class");
14    let frame = format_ident!("{wire}_callback_frame");
15    let mut descriptor = String::from("(");
16    let mut encode = TokenStream::new();
17    let mut args = Vec::new();
18    let slots: usize = crate::group::callback_parameters(element)
19        .map(|w| w.kind.parameter_slots())
20        .sum();
21    if slots > 254 {
22        return Err("sequence callback exceeds JVM parameter limit".into());
23    }
24    for (i, (field, value)) in prebindgen_conversions::match_output_fields(
25        crate::group::callback_parameters(element),
26        values,
27    )?
28    .into_iter()
29    .enumerate()
30    {
31        descriptor.push_str(field.kind.jvm_descriptor());
32        let local = format_ident!("{wire}_callback_object_{i}");
33        let argument = field.kind.output_argument(value, &local);
34        encode.extend(argument.statements);
35        args.push(argument.value);
36    }
37    descriptor.push_str(")V");
38    let capacity = i32::try_from(crate::group::callback_parameters(element).len() + 8)
39        .map_err(|_| "too many callback fields")?;
40    Ok(SequenceOutput {
41        commit_each: false,
42        fallible: true,
43        metadata: JniGroup::Sequence(Wire {
44            name: wire.clone(),
45            kind: JniWire::Rows {
46                condition: "true".into(),
47                null_niche_available: true,
48            },
49        }),
50        begin: quote! {
51            let #class=#env.get_object_class(&__pb_callback).map_err(|e|e.to_string())?;
52            let #method=#env.get_method_id(&#class,"run",#descriptor).map_err(|e|e.to_string())?;
53            #env.delete_local_ref(#class).map_err(|e|e.to_string())?;
54        },
55        enter: quote!(let #frame=unsafe{::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;),
56        store: quote! {
57            #encode
58            // The method was resolved on this callback object with exactly these
59            // JVM wire types. The generated bridge only retains raw fields.
60            unsafe {#env.call_method_unchecked(&__pb_callback,#method,::prebindgen_jni_runtime::jni::signature::ReturnType::Primitive(::prebindgen_jni_runtime::jni::signature::Primitive::Void),&[#(#args),*])}.map_err(|e|e.to_string())?;
61            drop(#frame);
62        },
63        // Only a presence marker crosses back; row payloads crossed through run.
64        finish: quote!(let #wire=Some(::prebindgen_jni_runtime::array(&mut #env,0)?);),
65        values: vec![(wire.clone(), quote!(#wire))],
66        commit: quote!(),
67    })
68}

70pub(crate) struct Fold {
71    pub output: KotlinOutput,
72    pub parameter: String,
73    pub discard: String,
74}
75pub(crate) fn fold(
76    layout: &[prebindgen_flat::WireNode<JniWire, JniGroup>],
77    context: &KotlinContext<'_>,
78) -> Result<Fold> {
79    fn optional(nodes: &[prebindgen_flat::WireNode<JniWire, JniGroup>]) -> bool {
80        nodes.iter().any(|node| match node {
81            prebindgen_flat::WireNode::Optional(_) => true,
82            prebindgen_flat::WireNode::Group { children, .. } => optional(children.as_slice()),
83            _ => false,
84        })
85    }
86    let success = match layout {
87        [prebindgen_flat::WireNode::Outcomes { success, .. }] => success.as_slice(),
88        nodes => nodes,
89    };
90    let optional = optional(success);
91    let (carrier, element) =
92        crate::sequence::return_sequence(layout).ok_or("missing sequence callback layout")?;
93    let position = crate::group::result_packet_fields(layout)
94        .position(|w| w.name == carrier.name)
95        .unwrap();
96    let row = "__pb_fold_row";
97    let ctx = KotlinContext {
98        result: row,
99        output_prefix: "foldElement",
100        cleanup_failure: None,
101        ..*context
102    };
103    let child = kotlin::output_kotlin(element, &ctx)?;
104    let present = format!("{}[{position}] != null", context.result);
105    let raw = "__pb_rows";
106    let types = child.result_types.join(", ");
107    let args = child.delivered.join(", ");
108    let failure = context.failure;
109    Ok(Fold {
110        parameter:format!("fold: (A, {types}) -> A"),
111        discard:format!("fun __pb_discard({failure}: Throwable) {{ for (rawRow in __pb_rows) {{ val {row} = rawRow as kotlin.Array<*>\n{}\n{}\n}} }}\n",child.output_locals,child.cleanup),
112        output:KotlinOutput {
113            delivered:vec![if optional{"__pb_accumulator".into()}else{"__pb_accumulator as A".into()}],
114            result_types:vec![if optional{"A?".into()}else{"A".into()}],
115            output_locals:"var __pb_accumulator: A? = null\nvar __pb_processed = 0\nvar __pb_pending: ((Throwable) -> Unit)? = null\n".into(),
116            materialize:format!("if ({present}) {{\nvar current = initial\nfor (rawRow in {raw}) {{\nval {row} = rawRow as kotlin.Array<*>\n{}\n__pb_pending = {{ {failure} -> {} }}\n{}\ncurrent = fold(current, {args})\n__pb_pending = null\n__pb_processed++\n}}\n__pb_accumulator = current\n}}\n",child.output_locals,child.cleanup,child.materialize),
117            cleanup:format!("__pb_pending?.invoke({failure})\nif ({present}) for (index in (__pb_processed + if (__pb_pending != null) 1 else 0) until {raw}.size) {{\nval {row} = {raw}[index] as kotlin.Array<*>\n{}\n{}\n}}\n",child.output_locals,child.cleanup),
118        },
119    })
120}
