1//! Kotlin enum declarations and checked JNI integer conversions.
2use super::*;
3use prebindgen_declarations::{Condition, EnumMapping};

5/// A Kotlin enum class preserving a fieldless Rust enum's integer discriminants.
6#[derive(Clone)]
7pub struct EnumClassDecl {
8    /// Source enum type, resolved against the selected source.
9    pub ty: syn::Type,
10    /// Literal Kotlin class name, or the source type's last path segment.
11    pub name: Option<String>,
12    /// Explicit Kotlin entry-name overrides keyed by source variant name.
13    pub variant_names: Vec<(String, String)>,
14}
15impl EnumClassDecl {
16    /// Declare a fieldless enum class from its source type syntax.
17    pub fn new(ty: syn::Type) -> Self {
18        Self {
19            ty,
20            name: None,
21            variant_names: Vec::new(),
22        }
23    }
24    /// Override the Kotlin short class name.
25    pub fn name(mut self, name: impl Into<String>) -> Self {
26        self.name = Some(name.into());
27        self
28    }
29    /// Set a Kotlin entry name without changing the source discriminant.
30    /// Unknown source names, duplicate overrides and conflicting foreign names
31    /// are rejected during build.
32    pub fn variant_name(mut self, source: impl Into<String>, name: impl Into<String>) -> Self {
33        self.variant_names.push((source.into(), name.into()));
34        self
35    }
36}
37#[macro_export]
38/// Declare a fieldless Kotlin enum class without requiring the Rust type in scope.
39macro_rules! enum_class {($($ty:tt)+)=>{$crate::EnumClassDecl::new($crate::__type(stringify!($($ty)+)))};}
40
41/// A JNI enum's checked integer mapping and Kotlin class identity.
42#[derive(Clone, Debug)]
43pub struct JniEnum {
44    /// Source variants and their signed 32-bit representations.
45    pub mapping: EnumMapping,
46    /// Fully qualified generated Kotlin enum class name.
47    pub class_name: String,
48    /// Explicit Kotlin variant-name overrides; source names are otherwise retained.
49    pub variant_names: BTreeMap<String, String>,
50}
51impl JniEnum {
52    pub(crate) fn from_class<I, O>(flat: &Flat<I, O>, class: &Class) -> Result<Self> {
53        let source = flat
54            .enums
55            .get(&type_key(&class.ty))
56            .ok_or("enum_class requires a fieldless source enum")?;
57        let mut variant_names = BTreeMap::new();
58        for (source, name) in &class.variant_names {
59            if variant_names.insert(source.clone(), name.clone()).is_some() {
60                return Err("duplicate enum variant override".into());
61            }
62        }
63        let result = Self {
64            mapping: EnumMapping::new(source)?,
65            class_name: class.fqn(),
66            variant_names,
67        };
68        result.validate()?;
69        Ok(result)
70    }
71    pub(crate) fn variant_name(&self, source: &syn::Ident) -> String {
72        let name = source.to_string();
73        let source = name.trim_start_matches("r#");
74        self.variant_names
75            .get(source)
76            .cloned()
77            .unwrap_or_else(|| source.into())
78    }
79    pub(crate) fn validate(&self) -> Result<()> {
80        self.mapping.validate()?;
81        validate_package(&self.class_name)?;
82        for source in self.variant_names.keys() {
83            if !self
84                .mapping
85                .variants
86                .iter()
87                .any(|(n, _)| n.to_string().trim_start_matches("r#") == source)
88            {
89                return Err(format!("unknown enum variant override `{source}`"));
90            }
91        }
92        let mut names = BTreeSet::new();
93        for (source, _) in &self.mapping.variants {
94            let name = self.variant_name(source);
95            prebindgen_declarations::ident(&format!("v_{name}"))?;
96            if ["name", "ordinal", "entries", "kotlin"].contains(&name.as_str())
97                || name.chars().all(|c| c == '_')
98            {
99                return Err(format!(
100                    "Kotlin enum entry `{name}` is reserved; use variant_name to rename it"
101                ));
102            }
103            if !names.insert(name) {
104                return Err("duplicate Kotlin enum entry name".into());
105            }
106        }
107        Ok(())
108    }
109    fn member(&self, initial: &str) -> String {
110        let mut name = initial.to_string();
111        while self
112            .mapping
113            .variants
114            .iter()
115            .any(|(v, _)| self.variant_name(v) == name)
116        {
117            name.insert(0, '_');
118        }
119        name
120    }
121    pub(crate) fn value_member(&self) -> String {
122        self.member("wireValue")
123    }
124    pub(crate) fn decode_member(&self) -> String {
125        self.member("fromWire")
126    }
127    pub(crate) fn wire(&self) -> JniWire {
128        JniWire::Enum {
129            enum_type: Box::new(self.clone()),
130            next_integer_niche: self.mapping.initial_niche(),
131        }
132    }
133    pub(crate) fn input(
134        &self,
135        ty: &syn::Type,
136        w: syn::Ident,
137        v: syn::Ident,
138        active: &Condition,
139    ) -> Result<InputPlan<JniWire>> {
140        prebindgen_declarations::enum_input(&self.mapping, ty, w, v, active, self.wire())
141    }
142    pub(crate) fn output(&self, w: syn::Ident, v: TokenStream) -> OutputPlan<JniWire> {
143        prebindgen_declarations::enum_output(&self.mapping, w, v, self.wire())
144    }
145    pub(crate) fn render(&self, class: &Class) -> Result<String> {
146        let mut entries = Vec::new();
147        let mut cases = Vec::new();
148        for (variant, value) in &self.mapping.variants {
149            let name = self.variant_name(variant);
150            prebindgen_declarations::ident(&format!("v_{name}"))?;
151            entries.push(format!("    `{name}`({value})"));
152            cases.push(format!("            {value} -> `{name}`"));
153        }
154        let member = self.value_member();
155        let decode = self.decode_member();
156        let companion = self.member("Companion");
157        let parameter = self.member("value");
158        Ok(format!("package {}\n\npublic enum class {}(public val {member}: kotlin.Int) {{\n{};\n\n    public companion object {companion} {{\n        public fun {decode}({parameter}: kotlin.Int): {} = when ({parameter}) {{\n{}\n            else -> throw kotlin.IllegalArgumentException(\"invalid enum discriminant: ${parameter}\")\n        }}\n    }}\n}}\n",class.pkg,class.name,entries.join(",\n"),class.name,cases.join("\n")))
159    }
160}

162#[cfg(test)]
163mod tests {
164    use super::*;
165    fn flat(source: &str) -> Bindings {
166        Flat::parse(
167            syn::parse_file(source)
168                .unwrap()
169                .items
170                .into_iter()
171                .map(|i| (i, Default::default())),
172        )
173        .unwrap()
174    }
175    #[test]
176    fn enum_classes_are_order_independent_and_use_discriminants_in_data_fields() {
177        let generated = JniGen::builder().flat(flat("pub enum E { A=-2, B, C=7 } pub struct Record { pub value: E, pub maybe: Option<E> } pub fn echo(x:Record)->Record {}"))
178            .package(package!("records").class(data_class!(Record)).fun(fun!(echo)))
179            .package(package!("codes").class(enum_class!(E).name("Code")))
180            .set_package_prefix("test").build().unwrap();
181        let enumeration = &generated.kotlin["test/codes/Code.kt"];
182        assert!(enumeration.contains("`B`(-1)"));
183        assert!(enumeration.contains("`C`(7)"));
184        assert!(enumeration.contains("else -> throw kotlin.IllegalArgumentException"));
185        let record = &generated.kotlin["test/records/Record.kt"];
186        assert!(record.contains("test.codes.Code"));
187        assert!(record.contains("test.codes.Code?"));
188        syn::parse_file(&generated.rust).unwrap();
189    }
190    #[test]
191    fn enum_class_aliases_bad_discriminants_and_duplicate_bindings_fail() {
192        let fail = |source: &str| {
193            JniGen::builder()
194                .flat(flat(source))
195                .set_package_prefix("test")
196                .package(package!().class(enum_class!(E)))
197                .build()
198                .err()
199                .unwrap()
200        };
201        assert!(fail("pub type E = hidden::E;").contains("requires a fieldless source enum"));
202        assert!(fail("pub enum E { A = 2147483648 }").contains("does not fit"));
203        assert!(fail("pub enum E { A = CONST }").contains("integer literals"));
204        assert!(JniGen::builder()
205            .flat(flat("pub enum E { A, B }"))
206            .set_package_prefix("test")
207            .package(package!().class(enum_class!(E)).class(ptr_class!(E)))
208            .build()
209            .err()
210            .unwrap()
211            .contains("duplicate"));
212    }
213    #[test]
214    fn enum_wire_niches_validate_domain_skip_variants_and_keep_consumed_state() {
215        let source = flat("pub enum E { A = -2, B, C = 0, D = 7 }");
216        let e = JniEnum {
217            mapping: EnumMapping::new(source.enums.values().next().unwrap()).unwrap(),
218            class_name: "test.E".into(),
219            variant_names: BTreeMap::new(),
220        };
221        let mut wire = e.wire();
222        assert_eq!(wire.available_niche(), Some(JniNiche::Integer(-3)));
223        assert_eq!(
224            optional::remaining(&wire, &JniNiche::Integer(8)),
225            Some(JniNiche::Integer(6))
226        );
227        for bad in [
228            JniNiche::Null,
229            JniNiche::Integer(-1),
230            JniNiche::Integer(i64::MAX),
231        ] {
232            assert!(wire.set_niche(Some(bad)).is_err());
233            assert_eq!(wire.available_niche(), Some(JniNiche::Integer(-3)));
234        }
235        wire.clear_niche();
236        assert!(wire.available_niche().is_none());
237        wire.validate_niche(&JniNiche::Integer(-3)).unwrap();
238        wire.set_niche(Some(JniNiche::Integer(i32::MIN as i64)))
239            .unwrap();
240        assert!(optional::remaining(&wire, &JniNiche::Integer(i32::MIN as i64)).is_none());
241    }
242    #[test]
243    fn generated_infrastructure_avoids_source_variant_names() {
244        let generated = JniGen::builder()

246            .set_package_prefix("test")
247            .package(package!().class(enum_class!(E)))

251        assert!(kt.contains("public val _wireValue"));
252        assert!(kt.contains("companion object _Companion"));
253        assert!(kt.contains("fun _fromWire"));
254    }

260    fn build(decl: EnumClassDecl) -> Result<JniGen> {
261        let items = syn::parse_file("pub enum E { name=1, ordinal=2, entries=3 }")
262            .unwrap()
263            .items;

270    #[test]
271    fn reserved_entry_names_require_explicit_unique_overrides() {
272        assert!(build(enum_class!(E)).err().unwrap().contains("reserved"));
273        let valid = || {
274            enum_class!(E)
275                .variant_name("name", "Name")
276                .variant_name("ordinal", "Ordinal")
277                .variant_name("entries", "Entries")
278        };
279        assert!(build(valid()).is_ok());
280        assert!(build(valid().variant_name("unknown", "Other"))
281            .err()
282            .unwrap()
283            .contains("unknown"));
284        assert!(build(valid().variant_name("name", "Other"))
285            .err()
286            .unwrap()
287            .contains("duplicate"));
288        assert!(build(
289            enum_class!(E)
290                .variant_name("name", "Same")
291                .variant_name("ordinal", "Same")
292                .variant_name("entries", "Entries")
