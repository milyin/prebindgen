6/// A Kotlin data class whose fields are converted across the JNI boundary.
7///
8/// Supports owned public named-field records containing scalars, `String`,
9/// `Vec<u8>`, other declared data classes, declared opaque handles, and optional fields.
10/// Records containing handles (including nested ones) support returns only.
11/// Output ownership is declared with `expand_return!`: `field_self_ref()` borrows
12/// and `field_self_into()` moves, recursively. `project_self(Projection)` allows
13/// field overrides. Legacy `field_self()` and unexpanded outputs keep a fixed
14/// borrowed policy, including for handles (which then require `Clone`).
15/// Moving Rust fields requires that the record itself does not implement `Drop`.
16/// Callers close each returned handle explicitly; data-class `copy` shares those
17/// handle objects and does not duplicate native ownership. References, empty
18/// records, and recursive records are rejected.
19///
20/// ```
21/// use prebindgen_jni::{JniGen, data_class, package, fun};
22/// let source = syn::parse_file("pub struct Point { pub x: f64, pub y: f64 } pub fn echo(p: Point) -> Point { p }")?;
23/// let generated = JniGen::builder()
24///     .items(source.items.into_iter().map(|item| (item, Default::default())))
25///     .set_package_prefix("example")
26///     .package(package!().class(data_class!(Point)).fun(fun!(echo)))
27///     .build()?;
28/// assert!(generated.kotlin["example/Point.kt"].contains("data class Point"));
29/// assert!(generated.kotlin["example/Functions.kt"].contains(": example.Point"));
30/// # Ok::<(), Box<dyn std::error::Error>>(())
31/// ```
32#[derive(Clone)]
33pub struct DataClassDecl {
34    /// Source record type, resolved against the selected flat model.
35    pub ty: syn::Type,
36    /// Literal Kotlin short name, or the source type's last path segment.
37    pub name: Option<String>,
38}
39impl DataClassDecl {
40    /// Declare a data class for the source record `ty`.
41    pub fn new(ty: syn::Type) -> Self {
42        Self { ty, name: None }
43    }
44    /// Set a literal Kotlin short class name, validated during build.
45    pub fn name(mut self, name: impl Into<String>) -> Self {
46        self.name = Some(name.into());
47        self
48    }
49}
50/// A class declaration placed in a [`PackageDecl`].
51#[derive(Clone)]
52pub enum ClassDecl {
53    /// An owning native handle.
54    Ptr(PtrClassDecl),
55    /// A record mapped to a Kotlin data class.

57    /// A fieldless enum mapped to a Kotlin enum class.

59}

62        Self::Ptr(value)
63    }
64}

66    fn from(value: DataClassDecl) -> Self {
67        Self::Data(value)
68    }
69}
70impl From<EnumClassDecl> for ClassDecl {
71    fn from(value: EnumClassDecl) -> Self {
72        Self::Enum(value)
73    }
74}

104#[derive(Clone, Debug)]
105struct DataField {
106    name: syn::Ident,
107    ty: syn::Type,
108    shape: Shape,
109}
110#[derive(Clone, Debug)]
111enum Shape {
112    Optional(Box<Shape>),
113    Value(Value),
114    Record(Rc<RecordShape>),
115    Handle(String),
116    Enum(Box<JniEnum>),
117}
118impl Shape {
119    fn is_bytes(&self) -> bool {
120        match self {
121            Self::Value(Value::Bytes) => true,
122            Self::Optional(s) => s.is_bytes(),
123            _ => false,
124        }
125    }
126    fn kotlin_type(&self) -> String {
127        match self {
128            Self::Optional(inner) => format!("{}?", inner.kotlin_type()),
129            Self::Value(Value::Scalar(Primitive::I128 | Primitive::U128)) => {
130                "java.math.BigInteger".into()
131            }
132            Self::Value(Value::Scalar(p)) => format!("kotlin.{}", scalar::kotlin(*p)),
133            Self::Value(Value::Text) => "kotlin.String".into(),
134            Self::Value(Value::Bytes) => "kotlin.ByteArray".into(),
135            Self::Record(r) => r.fqn.clone(),
136            Self::Handle(fqn) => fqn.clone(),
137            Self::Enum(e) => e.class_name.clone(),
138        }
139    }
140    fn slots(&self) -> usize {
141        match self {
142            Self::Value(Value::Scalar(p)) => JniWire::scalar(*p).parameter_slots(),
143            _ => 1,
144        }
145    }
146}

168impl DataCodec {
169    pub(crate) fn new<I, O>(flat: &Flat<I, O>, classes: &BTreeMap<String, Class>) -> Result<Self> {
170        fn resolve<I, O>(
171            flat: &Flat<I, O>,
172            classes: &BTreeMap<String, Class>,
173            key: &str,
174            stack: &mut Vec<String>,
175            records: &mut BTreeMap<String, Rc<RecordShape>>,
176        ) -> Result<Rc<RecordShape>> {
177            if stack.iter().any(|k| k == key) {
178                return Err(format!("recursive data class `{key}`"));
179            }
180            if let Some(r) = records.get(key) {
181                return Ok(r.clone());
182            }
183            let class = &classes[key];
184            let record = flat
185                .record(&class.ty)
186                .map_err(|_| format!("data class `{key}` requires a named-field record"))?;
187            if record.fields.is_empty() {
188                return Err(format!("data class `{key}` must have at least one field"));
189            }
190            stack.push(key.into());
191            let mut fields = Vec::new();
192            let mut getters = BTreeSet::from(["getClass".to_string()]);
193            for f in &record.fields {
194                let name = f.name.to_string();
195                let name = name.trim_start_matches("r#");
196                validate_name(name)
197                    .map_err(|e| format!("data class `{key}` field `{name}`: {e}"))?;
198                // Distinct Rust names such as `x` and `X` can generate the
199                // same JVM accessor even though Kotlin properties are distinct.
200                let getter = if name.starts_with("is")
201                    && name.len() > 2
202                    && !name.as_bytes()[2].is_ascii_lowercase()
203                {
204                    name.to_string()
205                } else {
206                    format!("get{}{}", name[..1].to_ascii_uppercase(), &name[1..])
207                };
208                if !getters.insert(getter.clone()) {
209                    return Err(format!(
210                        "data class `{key}` field `{name}` has a conflicting JVM getter `{getter}`"
211                    ));
212                }
213                let optional = prebindgen_declarations::option_inner(&f.ty);
214                let field_ty = optional.unwrap_or(&f.ty);
215                if optional
216                    .and_then(prebindgen_declarations::option_inner)
217                    .is_some()
218                {
219                    return Err("nested Option data fields require an explicit expansion".into());
220                }
221                let field_key = type_key(field_ty);
222                let shape = if matches!(field_ty, syn::Type::Reference(_)) {
223                    return Err(format!(
224                        "data class `{key}` field `{name}` cannot be a reference"
225                    ));
226                } else if let Some(value) = Value::of(field_ty) {
227                    Shape::Value(value)
228                } else if classes.get(&field_key).is_some_and(|c| c.data) {
229                    Shape::Record(resolve(flat, classes, &field_key, stack, records)?)
230                } else if let Some(class) = classes.get(&field_key).filter(|c| c.enumeration) {
231                    Shape::Enum(Box::new(JniEnum::from_class(flat, class)?))
232                } else if let Some(handle) = classes.get(&field_key) {
233                    Shape::Handle(handle.fqn())
234                } else {
235                    return Err(format!("data class `{key}` field `{name}` has unsupported type `{field_key}`; expected a scalar, String, Vec<u8>, declared data class, or declared handle"));
236                };
237                fields.push(DataField {
238                    name: f.name.clone(),
239                    ty: f.ty.clone(),
240                    shape: if optional.is_some() {
241                        Shape::Optional(Box::new(shape))
242                    } else {
243                        shape
244                    },
245                });
246            }
247            // Kotlin's generated copy$default also needs a receiver, default masks,
248            // and a marker. Check that method, not just the primary constructor.
249            let slots = fields.iter().map(|f| f.shape.slots()).sum::<usize>()
250                + fields.len().div_ceil(32)
251                + 2;
252            if slots > 255 {
253                return Err(format!("data class `{key}` copy method needs {slots} JVM parameter slots; the limit is 255"));
254            }
255            stack.pop();
256            let shape = Rc::new(RecordShape {
257                ty: class.ty.clone(),
258                fqn: class.fqn(),
259                fields,
260            });
261            records.insert(key.into(), shape.clone());
262            Ok(shape)
263        }
264        for (key, _) in classes.iter().filter(|(_, c)| c.enumeration) {
265            if !flat.enums.contains_key(key) {
266                return Err("enum_class requires a fieldless source enum".into());
267            }
268        }
269        let mut records = BTreeMap::new();
270        for (key, class) in classes {
271            if class.data {
272                resolve(flat, classes, key, &mut Vec::new(), &mut records)?;
273            }
274        }
275        let mut enums = BTreeMap::new();
276        for (key, class) in classes.iter().filter(|(_, c)| c.enumeration) {
277            enums.insert(key.clone(), JniEnum::from_class(flat, class)?);
278        }
279        Ok(Self {
280            records,
281            enums,
282            callback_wire: None,
283            error_classes: BTreeMap::new(),
284            error_helpers: BTreeMap::new(),
285            error_layouts: BTreeMap::new(),
286        })
287    }
288    fn enum_type(&self, ty: &syn::Type) -> Option<&JniEnum> {
289        let ty = if let syn::Type::Reference(r) = ty {
290            &r.elem
291        } else {
292            ty
293        };
294        self.enums.get(&type_key(ty))
295    }
296    fn record(&self, ty: &syn::Type) -> Result<Option<&Rc<RecordShape>>> {
297        if let syn::Type::Reference(r) = ty {
298            if self.records.contains_key(&type_key(&r.elem)) {
299                return Err(format!(
300                    "data class `{}` supports owned occurrences only",
301                    type_key(&r.elem)
302                ));
303            }
304        }
305        Ok(self.records.get(&type_key(ty)))
306    }
307    pub(crate) fn render(&self, class: &Class) -> Result<String> {
308        let shape = &self.records[&type_key(&class.ty)];
309        let fields = shape
310            .fields
311            .iter()
312            .map(|f| {
313                format!(
314                    "    public val `{}`: {}",
315                    field_name(f),
316                    f.shape.kotlin_type()
317                )
318            })
319            .collect::<Vec<_>>()
320            .join(",\n");
321        let lifecycle = if shape.has_handles() {
322            "/** Close each handle member explicitly. Copies share the same handle objects. */\n"
323        } else {
324            ""
325        };
326        let mut out = format!(
327            "package {}\n\n{lifecycle}public data class {}(\n{fields}\n)",
328            class.pkg, class.name
329        );
330        if shape.fields.iter().any(|f| f.shape.is_bytes()) {
331            let equal = shape
332                .fields
333                .iter()
334                .map(|f| {
335                    let n = field_name(f);
336                    match f.shape {
337                        _ if f.shape.is_bytes() => {
338                            format!("this.`{n}`.contentEquals(other.`{n}`)")
339                        }
340                        Shape::Value(Value::Scalar(Primitive::F32 | Primitive::F64)) => {
341                            format!("this.`{n}`.compareTo(other.`{n}`) == 0")
342                        }
343                        _ => format!("this.`{n}` == other.`{n}`"),
344                    }
345                })
346                .collect::<Vec<_>>()
347                .join(" &&\n            ");
348            let hashes = shape
349                .fields
350                .iter()
351                .map(|f| {
352                    let n = field_name(f);
353                    if f.shape.is_bytes() {
354                        format!("this.`{n}`.contentHashCode()")
355                    } else {
356                        format!("this.`{n}`.hashCode()")
357                    }
358                })
359                .collect::<Vec<_>>();
360            let mut hash = hashes[0].clone();
361            for h in &hashes[1..] {
362                hash = format!("31 * ({hash}) + {h}");
363            }
364            let display = shape
365                .fields
366                .iter()
367                .map(|f| {
368                    let n = field_name(f);
369                    let suffix = if f.shape.is_bytes() {
370                        ".contentToString()"
371                    } else {
372                        ""
373                    };
374                    format!("{n}=${{this.`{n}`{suffix}}}")
375                })
376                .collect::<Vec<_>>()
377                .join(", ");
378            out.push_str(&format!(" {{\n    override fun equals(other: kotlin.Any?): kotlin.Boolean =\n        this === other || (other is {} &&\n            {equal})\n    override fun hashCode(): kotlin.Int = {hash}\n    override fun toString(): kotlin.String = \"{}({display})\"\n}}", class.name, class.name));
379        }
380        out.push('\n');
381        Ok(out)
382    }

388impl RecordShape {
389    fn has_handles(&self) -> bool {
390        self.fields.iter().any(|f| match &f.shape {
391            Shape::Optional(s) => s.has_handles(),
392            Shape::Handle(_) => true,
393            Shape::Record(r) => r.has_handles(),
394            Shape::Value(_) | Shape::Enum(_) => false,
395        })
396    }
397
398    fn input(
399        &self,
400        w: &syn::Ident,
401        v: &syn::Ident,
402        active: &Condition,
403    ) -> Result<InputPlan<JniWire>> {
404        if self.has_handles() {
405            return Err(format!(
406                "data class `{}` contains handles and supports returns only",
407                self.fqn
408            ));
409        }
410        let mut plan = prebindgen_declarations::record_input(
411            &JniCodec,
412            &self.ty,
413            v.clone(),
414            self.fields.iter().enumerate().map(|(i, f)| {
415                let fw = format_ident!("{w}_f{i}");
416                let fv = format_ident!("{v}_f{i}");
417                let child = f.shape.input(&f.ty, &fw, &fv, active)?;
418                Ok((f.name.clone(), child))
419            }),
420        )?;
421        debug_assert!(
422            plan.acquire.is_empty() && plan.resources.is_empty() && plan.acknowledgement.is_none(),
423            "data fields must remain value-only"
424        );
425        self.mark(&mut plan.node, w.to_string(), kotlin_condition(active))?;
426        Ok(plan)
427    }
428    fn output(
429        &self,
430        w: &syn::Ident,
431        value: TokenStream,
432        projection: &Projection,
433    ) -> Result<OutputPlan<JniWire>> {
434        for name in projection.fields.keys() {
435            if !self.fields.iter().any(|f| field_name(f) == *name) {
436                return Err(format!(
437                    "data class `{}`: unknown projection field `{name}`",
438                    self.fqn
439                ));
440            }
441        }
442        let owned = projection.mode == ProjectionMode::Owned;
443        let inherited = projection.inherited();
444        let convert = |i: usize, field: TokenStream| {
445            let f = &self.fields[i];
446            let policy = projection.fields.get(&field_name(f)).unwrap_or(&inherited);
447            if !owned && policy.mode == ProjectionMode::Owned {
448                return Err(format!("data class `{}` field `{}`: owned projection cannot consume a shared-reference occurrence", self.fqn, field_name(f)));
449            }
450            let value = if owned && policy.mode == ProjectionMode::Borrowed {
451                quote!(&(#field))
452            } else {
453                field
454            };
455            let fw = format_ident!("{w}_f{i}");
456            f.shape.output(&f.ty, &fw, value, policy)
457        };
458        let mut plan = if owned {
459            prebindgen_declarations::record_output_owned(
460                &JniCodec,
461                format_ident!("{w}_record"),
462                value,
463                self.fields.iter().map(|f| f.name.clone()),
464                convert,
465            )
466        } else {
467            prebindgen_declarations::record_output(
468                &JniCodec,
469                format_ident!("{w}_record"),
470                value,
471                self.fields.iter().map(|f| f.name.clone()),
472                convert,
473            )
474        }?;
475        self.mark(&mut plan.outcomes.success, w.to_string(), "true".into())?;
476        Ok(plan)
477    }
478    fn mark(
479        &self,
480        layout: &mut prebindgen_flat::WireNode<
481            JniWire,
482            crate::JniOptional,
483            crate::JniSequence,
484            JniGroup,
485            (),
486        >,
487        root: String,
488        condition: String,
489    ) -> Result<()> {
490        JniCodec.retire_niches(layout, None)?;
491        let mut nodes = std::mem::take(layout);
492        let prebindgen_flat::WireNode::Group { metadata, .. } = &mut nodes else {
493            return Err("record compiler must produce one field collection".into());
494        };
495        if !matches!(metadata, JniGroup::Fields(None)) {
496            return Err("record annotation requires a field collection".into());
497        }
498        *metadata = JniGroup::Fields(Some(DataGroup {
499            root,
500            class_name: self.fqn.clone(),
501            condition,
502        }));
503        *layout = nodes;
504        Ok(())
505    }
506}

651    }
652    fn consumes(&self, ty: &syn::Type, extra: &()) -> bool {
653        if let Some(inner) = prebindgen_declarations::option_inner(ty) {
654            return self.consumes(inner, extra);
655        }

657            && self.enum_type(ty).is_none()
658            && JniCodec.consumes(ty, extra)

663    fn finish_output(&self, plan: &mut OutputPlan<JniWire>, extra: &()) -> Result<()> {
664        JniCodec.finish_output(plan, extra)?;
665        if self.callback_wire.is_some() {
666            let env = prebindgen_conversions::template_parameter("environment");
667            plan.delivery.statements.extend(quote!(#env.set_boolean_array_region(&__pb_output_accepted,0,&[1]).map_err(|e|e.to_string())?;));
668        }
669        Ok(())
670    }
671    fn input_leaf(
672        &self,
673        ty: &syn::Type,
674        extra: &(),
675        wire: syn::Ident,
676        var: syn::Ident,
677        active: &Condition,
678    ) -> Result<InputPlan<JniWire>> {
679        if let Some(inner) = prebindgen_declarations::option_inner(ty) {
680            let tag = format_ident!("{wire}_present");
681            let child = self.input_leaf(
682                inner,
683                extra,
684                format_ident!("{wire}_some"),
685                format_ident!("{var}_some"),
686                &active.arm(tag.clone(), 1),
687            )?;
688            return prebindgen_declarations::optional_input(self, extra, tag, var, active, child);
689        }
690        if let Some(e) = self.enum_type(ty) {
691            return e.input(ty, wire, var, active);
692        }
693        let Some(record) = self.record(ty)? else {
694            return JniCodec.input_leaf(ty, extra, wire, var, active);
695        };
696        record.input(&wire, &var, active)
697    }
698    fn output_leaf(
699        &self,
700        ty: &syn::Type,
701        extra: &(),
702        wire: syn::Ident,
703        value: TokenStream,
704        borrowed: bool,
705    ) -> Result<OutputPlan<JniWire>> {
706        if let Some(inner) = prebindgen_declarations::option_inner(ty) {
707            let tag = format_ident!("{wire}_present");
708            let payload = format_ident!("{wire}_payload");
709            let child = self.output_leaf(
710                inner,
711                extra,
712                format_ident!("{wire}_some"),
713                quote!(#payload),
714                borrowed,
715            )?;
716            return prebindgen_declarations::optional_output(
717                self, extra, tag, payload, value, borrowed, child,
718            );
719        }
720        if let Some(e) = self.enum_type(ty) {
721            if matches!(ty,syn::Type::Reference(r) if r.mutability.is_some()) {
722                return Err("mutable-reference outputs are unsupported".into());
723            }
724            return Ok(e.output(wire, value));
725        }
726        let Some(record) = self.record(ty)? else {
727            return JniCodec.output_leaf(ty, extra, wire, value, borrowed);
728        };
729        // The legacy data-class leaf remains a borrowed projection, regardless
730        // of its fields. Ownership transfer requires an explicit projection.
731        record.output(&wire, value, &Projection::borrowed())
732    }
733    fn output_projected(
734        &self,
735        ty: &syn::Type,
736        extra: &(),
737        wire: syn::Ident,
738        value: TokenStream,
739        projection: &Projection,
740    ) -> Result<OutputPlan<JniWire>> {
741        if let Some(inner) = prebindgen_declarations::option_inner(ty) {
742            let tag = format_ident!("{wire}_present");
743            let payload = format_ident!("{wire}_payload");
744            let child = self.output_projected(
745                inner,
746                extra,
747                format_ident!("{wire}_some"),
748                quote!(#payload),
749                projection,
750            )?;
751            return prebindgen_declarations::optional_output(
752                self,
753                extra,
754                tag,
755                payload,
756                value,
757                projection.mode == ProjectionMode::Borrowed,
758                child,
759            );
760        }
761        if let Some(e) = self.enum_type(ty) {
762            if projection.mode == ProjectionMode::Owned && matches!(ty, syn::Type::Reference(_)) {
763                return Err("owned projection cannot consume a reference payload".into());
764            }
765            if !projection.fields.is_empty() {
766                return Err("enum values do not have projection fields".into());
767            }
768            return Ok(e.output(wire, value));
769        }
770        let Some(record) = self.record(ty)? else {
771            return JniCodec.output_projected(ty, extra, wire, value, projection);
772        };
773        record.output(&wire, value, projection)
774    }

835impl Shape {
836    fn has_handles(&self) -> bool {
837        match self {
838            Self::Handle(_) => true,
839            Self::Record(r) => r.has_handles(),
840            Self::Optional(s) => s.has_handles(),
841            _ => false,
842        }
843    }
844    fn input(
845        &self,
846        ty: &syn::Type,
847        w: &syn::Ident,
848        v: &syn::Ident,
849        active: &Condition,
850    ) -> Result<InputPlan<JniWire>> {
851        match self {
852            Self::Optional(s) => {
853                let tag = format_ident!("{w}_present");
854                let child = s.input(
855                    prebindgen_declarations::option_inner(ty).unwrap(),
856                    &format_ident!("{w}_some"),
857                    &format_ident!("{v}_some"),
858                    &active.arm(tag.clone(), 1),
859                )?;
860                prebindgen_declarations::optional_input(
861                    &JniCodec,
862                    &(),
863                    tag,
864                    v.clone(),
865                    active,
866                    child,
867                )
868            }
869            Self::Record(r) => r.input(w, v, active),
870            Self::Enum(e) => e.input(ty, w.clone(), v.clone(), active),
871            _ => JniCodec.input_leaf(ty, &(), w.clone(), v.clone(), active),
872        }
873    }
874    fn output(
875        &self,
876        ty: &syn::Type,
877        w: &syn::Ident,
878        value: TokenStream,
879        policy: &Projection,
880    ) -> Result<OutputPlan<JniWire>> {
881        match self {
882            Self::Optional(s) => {
883                let tag = format_ident!("{w}_present");
884                let payload = format_ident!("{w}_payload");
885                let child = s.output(
886                    prebindgen_declarations::option_inner(ty).unwrap(),
887                    &format_ident!("{w}_some"),
888                    quote!(#payload),
889                    policy,
890                )?;
891                prebindgen_declarations::optional_output(
892                    &JniCodec,
893                    &(),
894                    tag,
895                    payload,
896                    value,
897                    policy.mode == ProjectionMode::Borrowed,
898                    child,
899                )
900            }
901            Self::Record(r) => r.output(w, value, policy),
902            Self::Enum(e) => {
903                if !policy.fields.is_empty() {
904                    return Err("enum values do not have projection fields".into());
905                }
906                Ok(e.output(w.clone(), value))
907            }
908            _ => JniCodec.output_projected(ty, &(), w.clone(), value, policy),
909        }
910    }

913type FieldNode<'a> = (
914    &'a syn::Ident,
915    &'a prebindgen_flat::WireNode<JniWire, crate::JniOptional, crate::JniSequence, JniGroup, ()>,
916);
917fn fields(nodes: &[kotlin::JniNode]) -> Result<Vec<FieldNode<'_>>> {
918    nodes
919        .iter()
920        .map(|node| match node {
921            kotlin::JniNode::Group {
922                metadata: JniGroup::Field(name),
923                ..
924            } => Ok((name, node)),
925            _ => Err("Kotlin record requires named field children".into()),
926        })
927        .collect()
928}
929pub(crate) fn input_kotlin(
930    nodes: &[kotlin::JniNode],
931    d: &DataGroup,
932    context: &KotlinContext<'_>,
933) -> Result<KotlinInput> {
934    let inner = nodes
935        .iter()
936        .flat_map(kotlin::leaf_wires)
937        .collect::<Vec<_>>();
938    let mut out = KotlinInput::default();
939    let mut c = kotlin::input_components(nodes, context)?;
940    out.parameters.push(format!(
941        "{}: {}{}",
942        d.root,
943        d.class_name,
944        if d.condition == "true" { "" } else { "?" }
945    ));
946    let fields = fields(nodes)?;
947    if c.parameters.len() != fields.len() {
948        return Err("incomplete data-class input occurrence".into());
949    }
950    for (p, (field, children)) in c.parameters.iter().zip(&fields) {
951        let name = p.split_once(": ").unwrap().0;
952        let receiver = if d.condition == "true" {
953            d.root.clone()
954        } else {
955            format!("requireNotNull({})", d.root)
956        };
957        let expr = format!(
958            "{receiver}.`{}`",
959            field.to_string().trim_start_matches("r#")
960        );
961        for a in c
962            .arguments
963            .iter_mut()
964            .chain(c.locks.iter_mut())
965            .chain(c.consumed.iter_mut())
966            .chain(c.lock_groups.iter_mut())
967            .chain(c.setup.iter_mut().map(|b| &mut b.expression))
968        {
969            if matches!(kotlin::leaf_wires(children).as_slice(), [w] if matches!(w.kind, JniWire::Scalar { primitive: Primitive::I128 | Primitive::U128, .. }))
970            {
971                *a = a.replace(
972                    &format!("{name}?.toByteArray()"),
973                    &format!("{name}.toByteArray()"),
974                );
975            }
976            *a = optional::replace(a, name, &expr);
977        }
978    }
979    for (j, a) in c.arguments.into_iter().enumerate() {
980        out.arguments.push(if d.condition == "true" {
981            a
982        } else {
983            format!(
984                "if ({}) {a} else {}",
985                d.condition,
986                optional::default_kt(&inner[j].kind, None)
987            )
988        });
989    }
990    out.locks.extend(c.locks);
991    out.consumed.extend(c.consumed);
992    out.setup.extend(c.setup);
993    out.lock_groups.extend(c.lock_groups);
994    Ok(out)
995}
996pub(crate) fn output_kotlin(
997    nodes: &[kotlin::JniNode],
998    d: &DataGroup,
999    context: &KotlinContext<'_>,
1000) -> Result<KotlinOutput> {
1001    let mut c = kotlin::output_components(nodes, context)?;
1002    if c.delivered.len() != fields(nodes)?.len() {
1003        return Err("incomplete data-class output occurrence".into());
1004    }
1005    c.delivered = vec![format!("{}({})", d.class_name, c.delivered.join(", "))];
1006    c.result_types = vec![d.class_name.clone()];
1007    Ok(c)
1008}
