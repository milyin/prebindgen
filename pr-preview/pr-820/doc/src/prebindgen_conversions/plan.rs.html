6/// An input conversion split into call-wide validation, acquisition and construction.
7///
8/// Optional storage for the staged request protocol in this crate.
9/// Generators may use this protocol or implement their own request types.
10#[derive(Clone, Debug)]
11pub struct InputPlan<W: WireKind, C = (), G = ()> {
12    /// Whether validation or construction can return a binding error. This
13    /// excludes panics and allocation aborts. Composition preserves child flags.
14    pub fallible: bool,
15    /// Context type supplied by the language generator at emission.
16    pub context: std::marker::PhantomData<fn(&C)>,
17    /// Authoritative value tree. Adapter metadata records native encoding choices;
18    /// physical parameters are derived from this tree when the reader needs them.
19    pub layout: WireLayout<W, G>,
20    /// Shared call acknowledgement, if this input needs one.
21    pub acknowledgement: Option<TokenStream>,
22    /// Validate and prepare every active leaf before any ownership transfer.
23    pub validation: TokenStream,
24    /// Infallible acquisition of all owned inputs, before constructors can run.
25    pub acquire: TokenStream,
26    /// Active foreign resources whose identities must be checked across all arguments.
27    pub resources: Vec<Resource>,
28    /// Argument construction after all acquisitions, including source constructor calls.
29    pub statements: TokenStream,
30    /// Prepared value expression assigned to the caller-supplied target after `statements`.
31    pub argument: TokenStream,
32}
33/// Output conversion whose owned temporaries survive until successful delivery.
34///
35/// Optional concrete storage implementing [`prebindgen_flat::Conversion`].
36/// Delivery and ownership commit are separate requests.
37#[derive(Clone, Debug)]
38pub struct OutputPlan<W: WireKind, C = (), G = ()> {
39    /// Whether preparation or delivery can return a binding error, excluding
40    /// panics and allocation aborts. Adapters must mark custom fallible code.
41    pub fallible: bool,
42    /// Context type supplied by the language generator at emission.
43    pub context: std::marker::PhantomData<fn(&C)>,
44    /// Delivery and return code prepared with this result slot.
45    pub delivery: Delivery,
46    /// Authoritative success/error value tree. Adapter metadata on its nodes
47    /// records explicit native encodings without duplicating payload subtrees.
48    pub layout: WireLayout<W, G>,
49    /// Complete transfers for the selected protocol. Buffered outputs commit after
50    /// foreign delivery succeeds; explicit visitor protocols can commit each
51    /// element at callback entry during preparation.
52    pub commit: TokenStream,
53    /// Code producing owned output temporaries and evaluating observations before moves.
54    pub statements: TokenStream,
55    /// Expressions in native transport order, valid until commit. These describe
56    /// this reader's success delivery, not a flat list of logical output values.
57    pub values: Vec<TokenStream>,
58}
59/// Access to one foreign resource during an exported call.
60#[derive(Clone, Copy, Debug, PartialEq, Eq)]
61pub enum Access {
62    /// Read-only borrow; multiple active shared accesses may alias.
63    Shared,
64    /// Mutable borrow; conflicts with every other active use of the resource.
65    Exclusive,
66    /// Ownership transfer; conflicts with every other active use of the resource.
67    Move,
68}
69/// One allocation address, or a runtime-sized set of addresses with the same access mode.
70#[derive(Clone, Debug)]
71pub enum ResourceIdentity {
72    /// Expression evaluating to one allocation address.
73    Single(TokenStream),
74    /// Expression implementing IntoIterator<Item = usize>.
75    Many(TokenStream),
76}
77/// The adapter supplies identity; flat enforces call-wide access conflicts.
78#[derive(Clone, Debug)]
79pub struct Resource {
80    /// Side-effect-free Rust expression identifying a live resource, such as an address.
81    /// All identities compared within a call must share an equality-compatible domain.
82    pub identity: ResourceIdentity,
83    /// Borrow or ownership requirement imposed by this input conversion.
84    pub access: Access,
85    /// Side-effect-free Rust boolean expression; false resources do not conflict.
86    pub active: TokenStream,
87}
88/// Result-slot glue executed after output preparation and before ownership commit.
89#[derive(Clone, Debug)]
90pub struct Delivery {
91    /// Deliver every output; failure must leave owned temporaries uncommitted.
92    pub statements: TokenStream,
93    /// Infallible wrapper result expression evaluated after commit.
94    pub result: TokenStream,
95}
96impl Default for Delivery {
97    fn default() -> Self {
98        Self {
99            statements: TokenStream::new(),
100            result: quote!(()),
101        }
102    }
103}
