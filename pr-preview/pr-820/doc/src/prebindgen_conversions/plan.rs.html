6/// An input conversion split into call-wide validation, acquisition and construction.
7///
8/// Optional storage for the staged request protocol in this crate.
9/// Generators may use this protocol or implement their own request types.
10pub struct InputPlan<W: WireKind, C = (), O = (), S = (), G = (), A = ()> {
11    /// Whether validation or construction can return a binding error. This
12    /// excludes panics and allocation aborts. Composition preserves child flags.
13    pub fallible: bool,
14    /// Context type supplied by the language generator at emission.
15    pub context: std::marker::PhantomData<fn(&C)>,
16    /// Authoritative value tree. Adapter metadata records native encoding choices;
17    /// physical parameters are derived from this tree when the generator emits its signature.
18    pub node: prebindgen_flat::WireNode<W, O, S, G, A>,
19    /// Shared call acknowledgement, if this input needs one.
20    pub acknowledgement: Option<TokenStream>,
21    /// Validate and prepare every active leaf before any ownership transfer.
22    pub validation: TokenStream,
23    /// Infallible acquisition of all owned inputs, before constructors can run.
24    pub acquire: TokenStream,
25    /// Active foreign resources whose identities must be checked across all arguments.
26    pub resources: Vec<Resource>,
27    /// Argument construction after all acquisitions, including source constructor calls.
28    pub statements: TokenStream,
29    /// Prepared value expression assigned to the caller-supplied target after `statements`.
30    pub argument: TokenStream,
31}
32/// Output conversion whose owned temporaries survive until successful delivery.
33///
34/// Optional concrete storage implementing [`prebindgen_flat::Conversion`].
35/// Delivery and ownership commit are separate requests.
36pub struct OutputPlan<W: WireKind, C = (), O = (), S = (), G = (), A = ()> {
37    /// Whether preparation or delivery can return a binding error, excluding
38    /// panics and allocation aborts. Adapters must mark custom fallible code.
39    pub fallible: bool,
40    /// Context type supplied by the language generator at emission.
41    pub context: std::marker::PhantomData<fn(&C)>,
42    /// Delivery and return code prepared with this result slot.
43    pub delivery: Delivery,
44    /// Authoritative success/error value tree. Adapter metadata on its nodes
45    /// records explicit native encodings without duplicating payload subtrees.
46    pub outcomes: prebindgen_flat::Outcomes<prebindgen_flat::WireNode<W, O, S, G, A>>,
47    /// Complete transfers for the selected protocol. Buffered outputs commit after
48    /// foreign delivery succeeds; explicit visitor protocols can commit each
49    /// element at callback entry during preparation.
50    pub commit: TokenStream,
51    /// Code producing owned output temporaries and evaluating observations before moves.
52    pub statements: TokenStream,
53    /// Prepared success expressions keyed by their destination binding names.
54    /// Adapters choose packet/row order from the tree and match these by name.
55    pub values: Vec<(syn::Ident, TokenStream)>,
56}
57/// Access to one foreign resource during an exported call.
58#[derive(Clone, Copy, Debug, PartialEq, Eq)]
59pub enum Access {
60    /// Read-only borrow; multiple active shared accesses may alias.
61    Shared,
62    /// Mutable borrow; conflicts with every other active use of the resource.
63    Exclusive,
64    /// Ownership transfer; conflicts with every other active use of the resource.
65    Move,
66}
67/// One allocation address, or a runtime-sized set of addresses with the same access mode.
68#[derive(Clone, Debug)]
69pub enum ResourceIdentity {
70    /// Expression evaluating to one allocation address.
71    Single(TokenStream),
72    /// Expression implementing IntoIterator<Item = usize>.
73    Many(TokenStream),
74}
75/// The adapter supplies identity; flat enforces call-wide access conflicts.
76#[derive(Clone, Debug)]
77pub struct Resource {
78    /// Side-effect-free Rust expression identifying a live resource, such as an address.
79    /// All identities compared within a call must share an equality-compatible domain.
80    pub identity: ResourceIdentity,
81    /// Borrow or ownership requirement imposed by this input conversion.
82    pub access: Access,
83    /// Side-effect-free Rust boolean expression; false resources do not conflict.
84    pub active: TokenStream,
85}
86/// Result-slot glue executed after output preparation and before ownership commit.
87#[derive(Clone, Debug)]
88pub struct Delivery {
89    /// Deliver every output; failure must leave owned temporaries uncommitted.
90    pub statements: TokenStream,
91    /// Infallible wrapper result expression evaluated after commit.
92    pub result: TokenStream,
93}
94impl Default for Delivery {
95    fn default() -> Self {
96        Self {
97            statements: TokenStream::new(),
98            result: quote!(()),
99        }
100    }

110macro_rules! plan_traits {
111    ($plan:ident; $($field:ident),* $(,)?) => {
112        impl<W: WireKind + Clone, C, O: Clone, S: Clone, G: Clone, A: Clone> Clone for $plan<W, C, O, S, G, A> {
113            fn clone(&self) -> Self {
114                Self {
115                    context: self.context,
116                    $($field: self.$field.clone(),)*
117                }
118            }
119        }
120        impl<W: WireKind + std::fmt::Debug, C, O: std::fmt::Debug, S: std::fmt::Debug, G: std::fmt::Debug, A: std::fmt::Debug> std::fmt::Debug
121            for $plan<W, C, O, S, G, A>
122        {
123            fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
124                f.debug_struct(stringify!($plan))
125                    $(.field(stringify!($field), &self.$field))*
126                    .finish()
127            }
128        }
129    }
130}

134/// Match prepared expressions to the fields of a concrete result packet or row.
135/// The caller supplies destination fields in ABI order. Names, rather than the
136/// expression list's order, establish correspondence; missing/extra names fail.
137pub fn match_output_fields<'a, W: WireKind + 'a>(
138    fields: impl IntoIterator<Item = &'a prebindgen_flat::Wire<W>>,
139    bindings: &'a [(syn::Ident, TokenStream)],
140) -> prebindgen_flat::Result<Vec<(&'a prebindgen_flat::Wire<W>, &'a TokenStream)>> {
141    validate_value_bindings(bindings)?;
142    let mut remaining = bindings
143        .iter()
144        .map(|(name, value)| (name.to_string(), value))
145        .collect::<std::collections::BTreeMap<_, _>>();
146    let mut out = vec![];
147    for field in fields {
148        let value = remaining
149            .remove(&field.name.to_string())
150            .ok_or_else(|| format!("missing output binding `{}`", field.name))?;
151        out.push((field, value));
152    }
153    if let Some(name) = remaining.keys().next() {
154        return Err(format!("unexpected output binding `{name}`"));
155    }
156    Ok(out)
157}
158/// Check that each prepared output name is defined once.
159pub fn validate_value_bindings(
160    bindings: &[(syn::Ident, TokenStream)],
161) -> prebindgen_flat::Result<()> {
162    let mut names = std::collections::BTreeSet::new();
163    for (name, _) in bindings {
164        if !names.insert(name.to_string()) {
165            return Err(format!("duplicate output binding `{name}`"));
166        }
167    }
168    Ok(())
169}
