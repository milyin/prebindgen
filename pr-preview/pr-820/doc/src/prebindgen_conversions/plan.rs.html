6/// An input conversion split into call-wide validation, acquisition and construction.
7///
8/// Optional storage for the staged request protocol in this crate.
9/// Generators may use this protocol or implement their own request types.
10#[derive(Clone, Debug)]
11pub struct InputPlan<W: WireKind, C = (), G = ()> {
12    /// Logical value schema exposed through Conversion::layout.
13    pub logical: prebindgen_flat::WireLayout<W, G>,
14    /// Whether validation or construction can return a binding error. This
15    /// excludes panics and allocation aborts. Composition preserves child flags.
16    pub fallible: bool,
17    /// Context type supplied by the language generator at emission.
18    pub context: std::marker::PhantomData<fn(&C)>,
19    /// Native lowering selected by this staged-protocol reader.
20    pub transport: TransportLayout<W, G>,
21    /// Shared call acknowledgement, if this input needs one.
22    pub acknowledgement: Option<TokenStream>,
23    /// Validate and prepare every active leaf before any ownership transfer.
24    pub validation: TokenStream,
25    /// Infallible acquisition of all owned inputs, before constructors can run.
26    pub acquire: TokenStream,
27    /// Active foreign resources whose identities must be checked across all arguments.
28    pub resources: Vec<Resource>,
29    /// Argument construction after all acquisitions, including source constructor calls.
30    pub statements: TokenStream,
31    /// Prepared value expression assigned to the caller-supplied target after `statements`.
32    pub argument: TokenStream,
33}
34/// Output conversion whose owned temporaries survive until successful delivery.
35///
36/// Optional concrete storage implementing [`prebindgen_flat::Conversion`].
37/// Delivery and ownership commit are separate requests.
38#[derive(Clone, Debug)]
39pub struct OutputPlan<W: WireKind, C = (), G = ()> {
40    /// Logical success/error schemas exposed through Conversion::layout.
41    pub logical: prebindgen_flat::WireLayout<W, G>,
42    /// Whether preparation or delivery can return a binding error, excluding
43    /// panics and allocation aborts. Adapters must mark custom fallible code.
44    pub fallible: bool,
45    /// Context type supplied by the language generator at emission.
46    pub context: std::marker::PhantomData<fn(&C)>,
47    /// Delivery and return code prepared with this result slot.
48    pub delivery: Delivery,
49    /// Native success transport selected by this reader. Each physical entry
50    /// corresponds to one expression in `values`; logical outcomes and repeated
51    /// field definitions have no such arity requirement.
52    pub transport: TransportLayout<W, G>,
53    /// Complete transfers for the selected protocol. Buffered outputs commit after
54    /// foreign delivery succeeds; explicit visitor protocols can commit each
55    /// element at callback entry during preparation.
56    pub commit: TokenStream,
57    /// Code producing owned output temporaries and evaluating observations before moves.
58    pub statements: TokenStream,
59    /// Expressions in native transport order, valid until commit. These describe
60    /// this reader's success delivery, not a flat list of logical output values.
61    pub values: Vec<TokenStream>,
62}
63/// Access to one foreign resource during an exported call.
64#[derive(Clone, Copy, Debug, PartialEq, Eq)]
65pub enum Access {
66    /// Read-only borrow; multiple active shared accesses may alias.
67    Shared,
68    /// Mutable borrow; conflicts with every other active use of the resource.
69    Exclusive,
70    /// Ownership transfer; conflicts with every other active use of the resource.
71    Move,
72}
73/// One allocation address, or a runtime-sized set of addresses with the same access mode.
74#[derive(Clone, Debug)]
75pub enum ResourceIdentity {
76    /// Expression evaluating to one allocation address.
77    Single(TokenStream),
78    /// Expression implementing IntoIterator<Item = usize>.
79    Many(TokenStream),
80}
81/// The adapter supplies identity; flat enforces call-wide access conflicts.
82#[derive(Clone, Debug)]
83pub struct Resource {
84    /// Side-effect-free Rust expression identifying a live resource, such as an address.
85    /// All identities compared within a call must share an equality-compatible domain.
86    pub identity: ResourceIdentity,
87    /// Borrow or ownership requirement imposed by this input conversion.
88    pub access: Access,
89    /// Side-effect-free Rust boolean expression; false resources do not conflict.
90    pub active: TokenStream,
91}
92/// Result-slot glue executed after output preparation and before ownership commit.
93#[derive(Clone, Debug)]
94pub struct Delivery {
95    /// Deliver every output; failure must leave owned temporaries uncommitted.
96    pub statements: TokenStream,
97    /// Infallible wrapper result expression evaluated after commit.
98    pub result: TokenStream,
99}
100impl Default for Delivery {
101    fn default() -> Self {
102        Self {
103            statements: TokenStream::new(),
104            result: quote!(()),
105        }
106    }
107}

116impl<W: WireKind + Clone, C, G: Clone> InputPlan<W, C, G> {
117    /// Finalize this reader's logical payload specification from its chosen transport.
118    pub fn refresh_logical(&mut self) {
119        self.logical = self.transport.logical();
120    }
121    /// Finalize a freshly constructed native plan's logical specification.
122    pub fn with_logical(mut self) -> Self {
123        self.refresh_logical();
124        self
125    }
126}
127impl<W: WireKind + Clone, C, G: Clone> OutputPlan<W, C, G> {
128    /// Refresh the normal payload while retaining declared source/binding errors.
129    pub fn refresh_logical(&mut self) {
130        self.logical.replace_success(self.transport.logical());
131    }
132    /// Finalize a freshly constructed native plan's logical specification.
133    pub fn with_logical(mut self) -> Self {
134        self.refresh_logical();
135        self
136    }
