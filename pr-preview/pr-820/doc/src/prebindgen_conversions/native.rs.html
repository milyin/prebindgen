1//! Adapter-supplied views of native boundary scopes.
2use prebindgen_flat::{Conversion, Function, Result, Wire, WireKind, WireLayout, WireNode};
3use proc_macro2::TokenStream;
4
5/// Reads one native boundary scope from the authoritative value tree.
6///
7/// The adapter chooses which leaves and metadata values participate, their order,
8/// and which children belong to another scope. Shared conversion code consumes
9/// the resulting borrowed lists without prescribing packing or delivery rules.
10/// Readers may expose several values for one node, including pointer/length pairs.
11pub trait NativeReader<W: WireKind, G> {
12    /// Physical values in the requested scope, borrowed from these nodes.
13    /// Readers decide how to select output outcomes and interpret metadata.
14    fn wires(nodes: &[WireNode<W, G>]) -> std::vec::IntoIter<&Wire<W>>;
15    /// Edit those same values in the same order. Other scopes remain untouched.
16    fn wires_mut(nodes: &mut [WireNode<W, G>]) -> std::vec::IntoIter<&mut Wire<W>>;
17    /// Validate the schema and the adapter's boundary rules before installation.
18    fn validate(layout: &WireLayout<W, G>) -> Result<()>;
19}
20
21/// Check one adapter-selected namespace. The caller decides which values share
22/// a namespace; this helper knows nothing about tree shape or ABI placement.
23pub fn validate_wire_names<'a, W: WireKind + 'a>(
24    wires: impl IntoIterator<Item = &'a Wire<W>>,
25) -> Result<()> {
26    let mut names = std::collections::BTreeSet::new();
27    for wire in wires {
28        if !names.insert(wire.name.to_string()) {
29            return Err(format!("duplicate wire identifier `{}`", wire.name));
30        }
31    }
32    Ok(())
33}

35// The default plan reader is for direct Rust values without adapter metadata.
36// It has no collection transport. Adapters select their own reader explicitly.
37impl<W: WireKind> NativeReader<W, ()> for () {
38    fn wires(nodes: &[WireNode<W, ()>]) -> std::vec::IntoIter<&Wire<W>> {
39        fn visit<'a, W: WireKind>(nodes: &'a [WireNode<W, ()>], out: &mut Vec<&'a Wire<W>>) {
40            for node in nodes {
41                match node {
42                    WireNode::Wire(w) => out.push(w),
43                    WireNode::Group { children, .. } => visit(children.nodes(), out),
44                    WireNode::Outcomes { alternatives } => {
45                        if let Some(o) = alternatives
46                            .iter()
47                            .find(|o| o.kind == prebindgen_flat::OutcomeKind::Success)
48                        {
49                            visit(o.layout.nodes(), out);
50                        }
51                    }
52                    _ => {}
53                }
54            }
55        }
56        let mut out = vec![];
57        visit(nodes, &mut out);
58        out.into_iter()
59    }
60    fn wires_mut(nodes: &mut [WireNode<W, ()>]) -> std::vec::IntoIter<&mut Wire<W>> {
61        fn visit<'a, W: WireKind>(
62            nodes: &'a mut [WireNode<W, ()>],
63            out: &mut Vec<&'a mut Wire<W>>,
64        ) {
65            for node in nodes {
66                match node {
67                    WireNode::Wire(w) => out.push(w),
68                    WireNode::Group { children, .. } => visit(children.nodes_mut(), out),
69                    WireNode::Outcomes { alternatives } => {
70                        if let Some(o) = alternatives
71                            .iter_mut()
72                            .find(|o| o.kind == prebindgen_flat::OutcomeKind::Success)
73                        {
74                            visit(o.layout.nodes_mut(), out);
75                        }
76                    }
77                    _ => {}
78                }
79            }
80        }
81        let mut out = vec![];
82        visit(nodes, &mut out);
83        out.into_iter()
84    }
85    fn validate(layout: &WireLayout<W, ()>) -> Result<()> {
86        layout.validate()?;
87        if let [WireNode::Outcomes { alternatives }] = layout.nodes() {
88            for outcome in alternatives {
89                Self::validate(&outcome.layout)?;
90            }
91            return Ok(());
92        }
93        fn visit<W: WireKind>(nodes: &[WireNode<W, ()>]) -> Result<()> {
94            for node in nodes {
95                match node {
96                    WireNode::Wire(_) => {}
97                    WireNode::Group { children, .. } => visit(children.nodes())?,
98                    WireNode::Outcomes { alternatives } => {
99                        for o in alternatives {
100                            visit(o.layout.nodes())?;
101                        }
102                    }
103                    _ => return Err("the direct Rust reader requires a value or group".into()),
104                }
105            }
106            Ok(())
107        }
108        visit(layout.nodes())?;
109        validate_wire_names(Self::wires(layout.nodes()))
110    }
111}

113/// Physical-value expression requests for the adapter-selected native reader.
114pub trait NativeConversion: Conversion {
115    /// Reader used by this conversion; metadata has no required ABI interface.
116    type Reader: NativeReader<Self::Wire, Self::Group>;
117    /// Reader-specific requests concerning a physical value.
118    type WireFragment<'a>;
119    /// Emit an expression at this reader's boundary position.
120    fn code_fragment_wire(
121        &self,
122        pos: usize,
123        mode: Self::WireFragment<'_>,
124        expected: &syn::Type,
125    ) -> Result<TokenStream>;
126}
127/// Native input enumeration for a function's selected readers.
128pub trait NativeFunction {
129    /// Adapter's value description.
130    type Wire: WireKind;
131    /// Source-slot values; the adapter adds any invocation infrastructure.
132    fn native_inputs(&self) -> Result<std::vec::IntoIter<&Wire<Self::Wire>>>;
133}
134impl<I: NativeConversion, O: Conversion> NativeFunction for Function<I, O> {
135    type Wire = I::Wire;
136    fn native_inputs(&self) -> Result<std::vec::IntoIter<&Wire<I::Wire>>> {
137        Ok(self
138            .input_conversions()?
139            .into_iter()
140            .flat_map(|i| I::Reader::wires(i.layout().nodes()))
141            .collect::<Vec<_>>()
142            .into_iter())
143    }
144}

146#[cfg(test)]
147mod tests {
148    use super::*;
149    #[test]
150    fn direct_reader_validates_every_outcome_namespace() {
151        let wire = || Wire {
152            name: syn::parse_quote!(value),
153            kind: syn::parse_quote!(i32),
154        };
155        let mut layout: WireLayout<syn::Type> = vec![wire()].into();
156        layout.set_outcome(
157            prebindgen_flat::OutcomeKind::SourceError,
158            vec![wire()].into(),
159        );
160        layout.set_outcome(
161            prebindgen_flat::OutcomeKind::BindingFailure,
162            vec![wire(), wire()].into(),
163        );
164        assert!(<()>::validate(&layout)
165            .unwrap_err()
166            .contains("duplicate wire"));
167        layout.set_outcome(
168            prebindgen_flat::OutcomeKind::BindingFailure,
169            vec![wire()].into(),
170        );
171        <()>::validate(&layout).unwrap();
172    }
