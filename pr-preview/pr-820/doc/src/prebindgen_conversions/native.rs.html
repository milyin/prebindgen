7/// One node's explicit native encoding. This describes additional physical
8/// values only; payload structure remains exclusively in the WireNode tree.
9pub enum NativeEncoding<'a, W: WireKind> {
10    /// Additional controls precede the node's payload in the current scope.
11    Inline(&'a [Wire<W>]),
12    /// One physical value carries the payload, whose fields have another scope.
13    Carrier(&'a Wire<W>),
14}
15impl<'a, W: WireKind> NativeEncoding<'a, W> {
16    /// Carrier owned by this metadata, if this node uses one.
17    pub fn carrier(self) -> Option<&'a Wire<W>> {
18        match self {
19            NativeEncoding::Carrier(w) => Some(w),
20            _ => None,
21        }
22    }
23    /// Inline controls owned by this metadata, or None for a carrier.
24    pub fn controls(self) -> Option<&'a [Wire<W>]> {
25        match self {
26            NativeEncoding::Inline(ws) => Some(ws),
27            _ => None,
28        }
29    }
30}
31/// Mutable access to the additional values owned by one node's metadata.
32pub enum NativeEncodingMut<'a, W: WireKind> {
33    /// Controls in the current scope.
34    Inline(&'a mut [Wire<W>]),
35    /// Carrier in the current scope.
36    Carrier(&'a mut Wire<W>),
37}
38/// Adapter metadata interpreted by the optional staged native reader. Common
39/// Conversion does not require this trait or prescribe any native encoding.
40pub trait NativeMetadata<W: WireKind> {
41    /// Explicit encoding at this node; field/record metadata has no controls.
42    fn encoding(&self) -> NativeEncoding<'_, W>;
43    /// Edit those same physical values without copying their descriptions.
44    fn encoding_mut(&mut self) -> NativeEncodingMut<'_, W>;
45}
46impl<W: WireKind> NativeMetadata<W> for () {
47    fn encoding(&self) -> NativeEncoding<'_, W> {
48        NativeEncoding::Inline(&[])
49    }
50    fn encoding_mut(&mut self) -> NativeEncodingMut<'_, W> {
51        NativeEncodingMut::Inline(&mut [])
52    }
53}
54impl<W: WireKind> NativeMetadata<W> for Vec<Wire<W>> {
55    fn encoding(&self) -> NativeEncoding<'_, W> {
56        NativeEncoding::Inline(self)
57    }
58    fn encoding_mut(&mut self) -> NativeEncodingMut<'_, W> {
59        NativeEncodingMut::Inline(self)
60    }
61}

63/// Borrowed native parameter views. No second hierarchy or payload storage is
64/// constructed: each returned value belongs to a leaf or to adapter metadata.
65pub trait NativeLayoutExt {
66    /// Adapter's individual value description.
67    type Wire: WireKind;
68    /// Current scope's physical values in the reader's chosen order. At an
69    /// Outcomes root this selects Success; error payloads are read explicitly.
70    fn native_wires(&self) -> std::vec::IntoIter<&Wire<Self::Wire>>;
71    /// Mutable view of those same values, excluding carried element scopes.
72    fn native_wires_mut(&mut self) -> std::vec::IntoIter<&mut Wire<Self::Wire>>;
73    /// Number of physical values in this scope, derived from the tree.
74    fn native_len(&self) -> usize {
75        self.native_wires().len()
76    }
77    /// Whether the current scope has no physical values.
78    fn native_is_empty(&self) -> bool {
79        self.native_len() == 0
80    }
81    /// One physical value by position in this reader's derived view.
82    fn native_get(&self, pos: usize) -> Option<&Wire<Self::Wire>> {
83        self.native_wires().nth(pos)
84    }
85    /// Edit one physical value by its derived position.
86    fn native_get_mut(&mut self, pos: usize) -> Option<&mut Wire<Self::Wire>> {
87        self.native_wires_mut().nth(pos)
88    }
89    /// Check physical names, including carried elements, separately for each
90    /// outcome. Alternative outcomes may reuse names because they never coexist.
91    fn validate_native_names(&self) -> Result<()>;
92}
93impl<W: WireKind, G: NativeMetadata<W>> NativeLayoutExt for WireLayout<W, G> {
94    type Wire = W;
95    fn native_wires(&self) -> std::vec::IntoIter<&Wire<W>> {
96        native_node_wires(self.nodes())
97    }
98    fn native_wires_mut(&mut self) -> std::vec::IntoIter<&mut Wire<W>> {
99        fn walk<'a, W: WireKind, G: NativeMetadata<W>>(
100            layout: &'a mut WireLayout<W, G>,
101            out: &mut Vec<&'a mut Wire<W>>,
102        ) {
103            for node in layout.nodes_mut() {
104                match node {
105                    WireNode::Wire(w) => out.push(w),
106                    WireNode::Group { children, .. } => walk(children, out),
107                    WireNode::Optional(c) | WireNode::Sequence(c) => {
108                        match c.metadata.as_mut().map(NativeMetadata::encoding_mut) {
109                            Some(NativeEncodingMut::Carrier(w)) => out.push(w),
110                            encoding => {
111                                if let Some(NativeEncodingMut::Inline(controls)) = encoding {
112                                    out.extend(controls);
113                                }
114                                walk(&mut c.element, out);
115                            }
116                        }
117                    }
118                    WireNode::Alternatives {
119                        metadata,
120                        alternatives,
121                    } => {
122                        if let Some(metadata) = metadata {
123                            match metadata.encoding_mut() {
124                                NativeEncodingMut::Inline(controls) => out.extend(controls),
125                                NativeEncodingMut::Carrier(w) => {
126                                    out.push(w);
127                                    continue;
128                                }
129                            }
130                        }
131                        for arm in alternatives {
132                            walk(arm, out);
133                        }
134                    }
135                    WireNode::Outcomes { alternatives } => {
136                        if let Some(success) = alternatives
137                            .iter_mut()
138                            .find(|o| o.kind == OutcomeKind::Success)
139                        {
140                            walk(&mut success.layout, out);
141                        }
142                    }
143                }
144            }
145        }
146        let mut out = vec![];
147        walk(self, &mut out);
148        out.into_iter()
149    }
150    fn validate_native_names(&self) -> Result<()> {
151        fn insert<W: WireKind>(
152            w: &Wire<W>,
153            names: &mut std::collections::BTreeSet<String>,
154        ) -> Result<()> {
155            if !names.insert(w.name.to_string()) {
156                return Err(format!("duplicate wire identifier `{}`", w.name));
157            }
158            Ok(())
159        }
160        fn walk<W: WireKind, G: NativeMetadata<W>>(
161            layout: &WireLayout<W, G>,
162            names: &mut std::collections::BTreeSet<String>,
163        ) -> Result<()> {
164            for node in layout.nodes() {
165                match node {
166                    WireNode::Wire(w) => insert(w, names)?,
167                    WireNode::Group { metadata, children } => {
168                        if !matches!(metadata.encoding(), NativeEncoding::Inline([])) {
169                            return Err(
170                                "group metadata cannot carry native controls or a carrier".into()
171                            );
172                        }
173                        walk(children, names)?;
174                    }
175                    WireNode::Optional(c) | WireNode::Sequence(c) => {
176                        if let Some(m) = &c.metadata {
177                            match m.encoding() {
178                                NativeEncoding::Carrier(w) => insert(w, names)?,
179                                NativeEncoding::Inline(ws) => {
180                                    for w in ws {
181                                        insert(w, names)?;
182                                    }
183                                }
184                            }
185                        }
186                        walk(&c.element, names)?;
187                    }
188                    WireNode::Alternatives {
189                        metadata,
190                        alternatives,
191                    } => {
192                        if let Some(m) = metadata {
193                            match m.encoding() {
194                                NativeEncoding::Carrier(w) => insert(w, names)?,
195                                NativeEncoding::Inline(ws) => {
196                                    for w in ws {
197                                        insert(w, names)?;
198                                    }
199                                }
200                            }
201                        }
202                        for arm in alternatives {
203                            walk(arm, names)?;
204                        }
205                    }
206                    WireNode::Outcomes { alternatives } => {
207                        for arm in alternatives {
208                            walk(&arm.layout, &mut Default::default())?;
209                        }
210                    }
211                }
212            }
213            Ok(())
214        }
215        self.validate()?;
216        walk(self, &mut Default::default())
217    }
218}

220/// Physical values borrowed from logical child nodes and their encoding metadata.
221pub fn native_node_wires<W: WireKind, G: NativeMetadata<W>>(
222    nodes: &[WireNode<W, G>],
223) -> std::vec::IntoIter<&Wire<W>> {
224    fn walk<'a, W: WireKind, G: NativeMetadata<W>>(
225        nodes: &'a [WireNode<W, G>],
226        out: &mut Vec<&'a Wire<W>>,
227    ) {
228        for node in nodes {
229            match node {
230                WireNode::Wire(w) => out.push(w),
231                WireNode::Group { children, .. } => walk(children.nodes(), out),
232                WireNode::Optional(c) | WireNode::Sequence(c) => {
233                    match c.metadata.as_ref().map(NativeMetadata::encoding) {
234                        Some(NativeEncoding::Carrier(w)) => out.push(w),
235                        encoding => {
236                            if let Some(NativeEncoding::Inline(controls)) = encoding {
237                                out.extend(controls);
238                            }
239                            walk(c.element.nodes(), out);
240                        }
241                    }
242                }
243                WireNode::Alternatives {
244                    metadata,
245                    alternatives,
246                } => {
247                    if let Some(metadata) = metadata {
248                        match metadata.encoding() {
249                            NativeEncoding::Inline(controls) => out.extend(controls),
250                            NativeEncoding::Carrier(w) => {
251                                out.push(w);
252                                continue;
253                            }
254                        }
255                    }
256                    for arm in alternatives {
257                        walk(arm.nodes(), out);
258                    }
259                }
260                WireNode::Outcomes { alternatives } => {
261                    if let Some(success) =
262                        alternatives.iter().find(|o| o.kind == OutcomeKind::Success)
263                    {
264                        walk(success.layout.nodes(), out);
265                    }
266                }
267            }
268        }
269    }
270    let mut out = vec![];
271    walk(nodes, &mut out);
272    out.into_iter()
273}

275/// Opt-in requests for physical values derived from the conversion's value tree.
276pub trait NativeConversion: Conversion<Group: NativeMetadata<Self::Wire>> {
277    /// Reader-specific requests concerning a physical value.
278    type WireFragment<'a>;
279    /// Emit a physical-value expression in the derived native index domain.
280    fn code_fragment_wire(
281        &self,
282        pos: usize,
283        mode: Self::WireFragment<'_>,
284        expected: &syn::Type,
285    ) -> Result<TokenStream>;
286}
287/// Native input enumeration for the optional staged reader.
288pub trait NativeFunction {
289    /// Adapter's value description.
290    type Wire: WireKind;
291    /// Source slots lowered to native parameters. Reader infrastructure may add
292    /// callbacks and acknowledgements separately.
293    fn native_inputs(&self) -> Result<std::vec::IntoIter<&Wire<Self::Wire>>>;
294}
295impl<I: NativeConversion, O: Conversion> NativeFunction for Function<I, O> {
296    type Wire = I::Wire;
297    fn native_inputs(&self) -> Result<std::vec::IntoIter<&Wire<I::Wire>>> {
298        Ok(self
299            .input_conversions()?
300            .into_iter()
301            .flat_map(|i| i.layout().native_wires())
302            .collect::<Vec<_>>()
303            .into_iter())
304    }
305}

311    // Neither wire nor metadata needs Clone: native views borrow the only tree.
312    struct Kind(syn::Type);
313    impl WireKind for Kind {
314        fn rust_type(&self) -> syn::Type {
315            self.0.clone()
316        }
317    }
318    enum Encoding {
319        Inline(Vec<Wire<Kind>>),
320        Carrier(Box<Wire<Kind>>),
321        Record,
322    }
323    impl NativeMetadata<Kind> for Encoding {
324        fn encoding(&self) -> NativeEncoding<'_, Kind> {
325            match self {
326                Self::Inline(ws) => NativeEncoding::Inline(ws),
327                Self::Carrier(w) => NativeEncoding::Carrier(w),
328                Self::Record => NativeEncoding::Inline(&[]),
329            }
330        }
331        fn encoding_mut(&mut self) -> NativeEncodingMut<'_, Kind> {
332            match self {
333                Self::Inline(ws) => NativeEncodingMut::Inline(ws),
334                Self::Carrier(w) => NativeEncodingMut::Carrier(w),
335                Self::Record => NativeEncodingMut::Inline(&mut []),
336            }
337        }
338    }
339    type Layout = WireLayout<Kind, Encoding>;
340    fn leaf(name: &str) -> Wire<Kind> {
341        Wire {
342            name: syn::parse_str(name).unwrap(),
343            kind: Kind(syn::parse_quote!(i64)),
344        }
345    }
346    fn names(layout: &Layout) -> Vec<String> {
347        layout.native_wires().map(|w| w.name.to_string()).collect()
348    }

350    #[test]
351    fn native_views_borrow_payloads_and_metadata_from_one_tree() {
352        let mut element: Layout = vec![leaf("x"), leaf("y")].into();
353        element.enclose_optional(Some(Encoding::Inline(vec![leaf("item_present")])));
354        element.enclose_sequence(Some(Encoding::Carrier(leaf("rows").into())));
355        element.enclose_optional(Some(Encoding::Inline(vec![leaf("list_present")])));
356        element.enclose(Encoding::Record);
357        let mut layout: Layout = vec![leaf("before")].into();
358        layout.extend(element);
359        layout.push(leaf("after"));
360        layout.validate_native_names().unwrap();
361        assert_eq!(names(&layout), ["before", "list_present", "rows", "after"]);
362        assert_eq!(
363            layout
364                .iter()
365                .map(|w| w.name.to_string())
366                .collect::<Vec<_>>(),
367            ["before", "x", "y", "after"]
368        );
369        assert!(std::ptr::eq(
370            layout.get(0).unwrap(),
371            layout.native_get(0).unwrap()
372        ));
373        assert!(std::ptr::eq(
374            layout.get(3).unwrap(),
375            layout.native_get(3).unwrap()
376        ));
377        for (pos, wire) in layout.native_wires().enumerate() {
378            assert!(std::ptr::eq(wire, layout.native_get(pos).unwrap()));
379        }
380        assert!(layout.native_get(usize::MAX).is_none());
381        for wire in layout.native_wires_mut() {
382            wire.kind.0 = syn::parse_quote!(u64);
383        }
384        assert_eq!(prebindgen_flat::type_key(&layout[0].kind.0), "u64");
385        // Carried fields are another scope, so mutating outer values excludes them.
386        assert_eq!(prebindgen_flat::type_key(&layout[1].kind.0), "i64");
387        assert_eq!(
388            prebindgen_flat::type_key(&layout.native_get(2).unwrap().kind.0),
389            "u64"
390        );
391    }

393    #[test]
394    fn changing_node_metadata_changes_the_native_view_without_refreshing() {
395        let mut layout: Layout = vec![leaf("x"), leaf("y")].into();
396        layout.enclose_sequence(Some(Encoding::Carrier(leaf("rows").into())));
397        let addresses = layout
398            .entries()
399            .map(|entry| (entry.path, entry.wire as *const _ as usize))
400            .collect::<Vec<_>>();
401        assert_eq!(names(&layout), ["rows"]);
402        let WireNode::Sequence(c) = &mut layout.nodes_mut()[0] else {
403            panic!()
404        };
405        c.metadata = Some(Encoding::Inline(vec![leaf("count")]));
406        assert_eq!(names(&layout), ["count", "x", "y"]);
407        for (path, address) in addresses {
408            assert_eq!(
409                layout.get_path(&path).unwrap() as *const _ as usize,
410                address
411            );
412        }
413        layout.native_get_mut(0).unwrap().name = syn::parse_quote!(length);
414        assert_eq!(names(&layout), ["length", "x", "y"]);
415        assert_eq!(layout.len(), 2);
