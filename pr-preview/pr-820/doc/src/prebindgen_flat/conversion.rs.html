6/// Type information for one logical value. A generator may pass it directly or
7/// encode it as a packed field/callback argument.
8/// The returned type describes the boundary value, not the source value being converted.
9/// For example, a boolean conversion may use `u8` as its Rust carrier type.
10///
11/// ```
12/// use prebindgen_flat::WireKind;
13/// struct BoolCarrier;
14/// impl WireKind for BoolCarrier {
15///     fn rust_type(&self) -> syn::Type { syn::parse_quote!(u8) }
16/// }
17/// let ty = BoolCarrier.rust_type();
18/// assert_eq!(quote::quote!(#ty).to_string(), "u8");
19/// ```
20pub trait WireKind {
21    /// Rust type for this individual value; this does not prescribe a signature.
22    fn rust_type(&self) -> syn::Type;
23}
24impl WireKind for syn::Type {
25    fn rust_type(&self) -> syn::Type {
26        self.clone()
27    }
28}
29
30/// One typed value definition in the logical input/output contract.
31///
32/// `W` must implement [`WireKind`]. Call [`Wire::rust_type`]
33/// to obtain the Rust carrier type supplied by the kind.
34///
35/// ```
36/// use prebindgen_flat::Wire;
37/// let wire: Wire<syn::Type> = Wire {
38///     name: syn::parse_quote!(value),
39///     kind: syn::parse_quote!(i64),
40/// };
41/// let ty = wire.rust_type();
42/// assert_eq!(quote::quote!(#ty).to_string(), "i64");
43/// ```
44/// A wire kind without a Rust carrier type is rejected:
45///
46/// ```compile_fail
47/// use prebindgen_flat::Wire;
48/// struct Untyped;
49/// let wire = Wire { name: syn::parse_quote!(value), kind: Untyped };
50/// ```
51#[derive(Clone, Debug)]
52pub struct Wire<W: WireKind> {
53    /// Identifier used by conversion code within its element/branch context.
54    pub name: syn::Ident,
55    /// Individual value information implementing [`WireKind`].
56    /// Stores information about this individual value, such as conversion access
57    /// and available niches. Relationships between values belong on parent tree nodes.
58    pub kind: W,
59}
60
61impl<W: WireKind> Wire<W> {
62    /// Rust carrier type supplied by this wire's [`WireKind`].
63    /// A generator can use this type directly or select a packed transport.
64    pub fn rust_type(&self) -> syn::Type {
65        self.kind.rust_type()
66    }
67}

69/// Check that every bound function has an input converter for each parameter
70/// and an output converter, including unit results.
71///
72/// Functions with entirely empty slots remain source-only helpers. This check
73/// neither builds conversions nor changes the model. Adapters call it before
74/// writing wrappers. It does not validate code fragments supplied directly by callers.
75pub fn validate_bindings<I: Conversion, O: Conversion>(flat: &crate::Flat<I, O>) -> Result<()> {
76    for f in flat.functions.values() {
77        if f.ret.slot.is_some() {
78            for p in &f.params {
79                if p.ty.slot.is_none() {
80                    return Err(format!(
81                        "`{}`: parameter `{}` has no input converter",
82                        f.name, p.name
83                    ));
84                }
85            }
86        } else if f.params.iter().any(|p| p.ty.slot.is_some()) {
87            return Err(format!(
88                "`{}`: bound inputs have no output converter",
89                f.name
90            ));
91        }
92    }
93    Ok(())
94}
95/// Format generated tokens as a Rust source file.
96///
97/// `tokens` must be a sequence of Rust items. Returns a syntax error if they do not form a file. This does not
98/// resolve names or type-check generated code; the binding crate must compile it.
99pub fn rust_source(tokens: TokenStream) -> Result<String> {
100    let file = syn::parse2(tokens).map_err(|e| format!("generated Rust: {e}"))?;
101    Ok(prettyplease::unparse(&file))
102}

104/// Prepared slot conversion with a generator-defined request protocol.
105///
106/// Input and output slots share this interface. Their payload types may define
107/// different wire kinds and request enums; direction comes from the slot in
108/// [`crate::Flat`], not from this trait. An output request can, for example, carry
109/// the source-result identifier without imposing a shared context type.
110///
111/// Every successful request returns one Rust expression, including `()` for
112/// a no-op. Statements may appear inside a block expression. The caller decides
113/// how expressions are combined and binds any values needed by later requests.
114/// `expected_type` is the caller's expectation, not a type-checking guarantee:
115/// the implementation may use it to configure emission or reject a request.
116/// Rust compilation performs the final type check. No implicit casts are required.
117///
118/// Request enums belong to the generator. Each variant can carry only the names,
119/// settings or other data it needs; no common context or phase order is imposed.
120/// Their lifetime parameter allows requests to borrow that data.
121///
122/// The tree describes logical values. Readers choose transport, runtime
123/// repetition and branch selection. A leaf traversal index is not an ABI
124/// parameter position. Requests needing an element/branch context carry it in
125/// the generator-defined request enum; no per-wire ABI protocol is mandatory.
126///
127/// A reusable owned-box converter can define its own small request vocabulary.
128/// The pointee is preparation data, not a hardcoded application type. The slot
129/// request borrows an error message only for validation; acquisition needs no
130/// context. The caller supplies `expected_type` to every request and binds the
131/// resulting value itself. This example chooses syntactic type matching as its
132/// policy; other implementations may use the expectation to configure emission.
133///
134/// ```
135/// use prebindgen_flat::{bind_fragment, same_type, Conversion, ConversionExt, Result, Wire, WireKind};
136/// use proc_macro2::TokenStream;
137/// use quote::quote;
138/// use syn::{parse_quote, Type};
139///
140/// enum Carrier { OpaquePointer }
141/// impl WireKind for Carrier {
142///     fn rust_type(&self) -> Type { parse_quote!(*mut ::std::ffi::c_void) }
143/// }
144/// enum SlotRequest<'a> {
145///     Validate { null_message: &'a str },
146///     TakeOwnership,
147/// }
148/// enum WireRequest { Value, Active }
149/// struct OwnedBoxInput {
150///     pointee: Type,
151///     layout: prebindgen_flat::WireLayout<Carrier>,
152/// }
153/// fn expect(actual: &Type, expected: &Type) -> Result<()> {
154///     if same_type(actual, expected) { Ok(()) }
155///     else { Err("unsupported expected type".into()) }
156/// }
157/// impl Conversion for OwnedBoxInput {
158///     type Wire = Carrier;
159///     type Group = ();
160///     type SlotFragment<'a> = SlotRequest<'a>;
161///     fn layout(&self) -> &prebindgen_flat::WireLayout<Carrier> { &self.layout }
162///     fn code_fragment_slot(&self, mode: SlotRequest<'_>, expected: &Type)
163///         -> Result<TokenStream>
164///     {
165///         let name = &self.layout[0].name;
166///         let pointee = &self.pointee;
167///         match mode {
168///             SlotRequest::Validate { null_message } => {
169///                 expect(&parse_quote!(()), expected)?;
170///                 Ok(quote!({
171///                     if #name.is_null() { return Err(#null_message.into()); }
172///                 }))
173///             }
174///             SlotRequest::TakeOwnership => {
175///                 expect(&parse_quote!(::std::boxed::Box<#pointee>), expected)?;
176///                 // The caller guarantees a live, uniquely owned allocation
177///                 // created by compatible Box::into_raw. Validation alone cannot.
178///                 Ok(quote!(unsafe {
179///                     ::std::boxed::Box::<#pointee>::from_raw(#name.cast::<#pointee>())
180///                 }))
181///             }
182///         }
183///     }
184/// }
185/// // A generator may supply its own contextual wire-access API.
186/// impl OwnedBoxInput {
187///     fn code_fragment_wire(&self, pos: usize, mode: WireRequest, expected: &Type)
188///         -> Result<TokenStream>
189///     {
190///         let wire = self.layout.get(pos).ok_or("invalid wire position")?;
191///         match mode {
192///             WireRequest::Value => {
193///                 expect(&wire.rust_type(), expected)?;
194///                 let name = &wire.name;
195///                 Ok(quote!(#name))
196///             }
197///             WireRequest::Active => {
198///                 expect(&parse_quote!(Option<bool>), expected)?;
199///                 Ok(quote!(None)) // An expression meaning unconditional activity.
200///             }
201///         }
202///     }
203/// }
204/// let slot = OwnedBoxInput {
205///     pointee: parse_quote!(u64), // Supplied by preparation from the source type.
206///     layout: vec![Wire { name: parse_quote!(raw_value), kind: Carrier::OpaquePointer }].into(),
207/// };
208/// let validation = slot.code_fragment_slot(
209///     SlotRequest::Validate { null_message: "null owned handle" }, &parse_quote!(()))?;
210/// let expected = parse_quote!(::std::boxed::Box<u64>);
211/// let value = slot.code_fragment_slot(SlotRequest::TakeOwnership, &expected)?;
212/// let binding = bind_fragment(&parse_quote!(argument), &expected, value)?;
213/// let _: syn::Block = syn::parse2(quote!({ #validation; #binding }))?;
214/// let active = slot.code_fragment_wire(0, WireRequest::Active, &parse_quote!(Option<bool>))?;
215/// assert_eq!(active.to_string(), "None");
216/// let wire_value = slot.code_fragment_wire(0, WireRequest::Value, &slot.wire(0)?.rust_type())?;
217/// assert_eq!(wire_value.to_string(), "raw_value");
218/// # Ok::<(), Box<dyn std::error::Error>>(())
219/// ```
220pub trait Conversion {
221    /// Generator-defined type information for individual logical values.
222    type Wire: WireKind;
223    /// Metadata describing how child values form records, options or alternatives.
224    type Group;
225    /// Requests concerning the whole slot.
226    type SlotFragment<'a>;
227    /// Logical tree of values, repetition and outcomes. Its generator chooses ABI placement.
228    fn layout(&self) -> &crate::WireLayout<Self::Wire, Self::Group>;
229    /// Emit an expression for the whole slot.
230    fn code_fragment_slot(
231        &self,
232        mode: Self::SlotFragment<'_>,
233        expected_type: &syn::Type,
234    ) -> Result<TokenStream>;
235}

237/// Wire access derived exclusively from a conversion's tree.
238///
239/// Import this extension trait alongside [`Conversion`] to call `wires()` and
240/// `wire(position)`. Its blanket implementation prevents a slot implementation
241/// from supplying a different wire order than the tree used for grouping.
242pub trait ConversionExt: Conversion {
243    /// Logical leaf definitions in structural order, each schema visited once.
244    fn wires(&self) -> crate::Wires<'_, Self::Wire, Self::Group>;
245    /// A definition at its traversal index; this is not a native ABI position.
246    fn wire(&self, position: usize) -> Result<&Wire<Self::Wire>>;
247    /// Resolve a definition by element/outcome path, independently of transport.
248    fn wire_at(&self, path: &crate::WirePath) -> Result<&Wire<Self::Wire>> {
249        self.layout()
250            .get_path(path)
251            .ok_or_else(|| "invalid logical wire path".into())
252    }
253}
254impl<T: Conversion + ?Sized> ConversionExt for T {
255    fn wires(&self) -> crate::Wires<'_, Self::Wire, Self::Group> {
256        self.layout().iter()
257    }
258    fn wire(&self, position: usize) -> Result<&Wire<Self::Wire>> {
259        self.layout()
260            .get(position)
261            .ok_or_else(|| format!("invalid wire position {position}"))
262    }
263}

265/// Bind an emitted expression to a caller-selected name and expected Rust type.
266///
267/// Rejects empty or statement-only fragments. The generated type annotation
268/// asks the Rust compiler to check the expectation in the wrapper's environment.
269pub fn bind_fragment(
270    target: &syn::Ident,
271    expected_type: &syn::Type,
272    fragment: TokenStream,
273) -> Result<TokenStream> {
274    let expression: syn::Expr =
275        syn::parse2(fragment).map_err(|error| format!("expected a Rust expression: {error}"))?;
276    Ok(quote!(let #target: #expected_type = #expression;))
277}
278impl<I: Conversion, O: Conversion> crate::Function<I, O> {
279    /// Filled input slots, preserving source parameter order and grouping.
280    pub fn input_conversions(&self) -> Result<Vec<&I>> {
281        self.params
282            .iter()
283            .map(|p| {
284                p.ty.slot.as_ref().ok_or_else(|| {
285                    format!("`{}`: parameter `{}` has no conversion", self.name, p.name)
286                })
287            })
288            .collect()
289    }
290    /// Filled result slot, borrowed without requesting any generated code.
291    pub fn output_conversion(&self) -> Result<&O> {
292        let output = self
293            .ret
294            .slot
295            .as_ref()
296            .ok_or_else(|| format!("`{}`: result has no conversion", self.name))?;
297        Ok(output)
298    }
299    /// Logical input definitions, including each element/branch schema once.
300    pub fn inputs(&self) -> Result<impl Iterator<Item = &Wire<I::Wire>>> {
301        Ok(self
302            .input_conversions()?
303            .into_iter()
304            .flat_map(|i| i.wires()))
305    }
306}
