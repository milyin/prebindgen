1//! Source types, occurrence slots, and generator-defined conversion interfaces.
2//! Preparation fills slots; language generators interpret their request protocols.
3//! Named objects must have nonzero Rust size, including opaque aliases. Their
4//! layout is checked by the emitted compile-time guards, independently of the
5//! slot data types. Unit remains available for absence of a value.

18/// A model or generation result with a diagnostic suitable for a build script.
19pub type Result<T> = std::result::Result<T, String>;

21/// One function parameter or result type, paired with consumer-defined data.
22///
23/// Two parameters of the same type have independent slots.
24#[derive(Clone, Debug)]
25pub struct TypeUse<S> {
26    /// Qualified source type.
27    pub ty: Type,
28    /// Consumer-defined data for this occurrence.
29    pub slot: S,
30}
31/// A named function parameter, in source declaration order.
32#[derive(Clone, Debug)]
33pub struct Param<S> {
34    /// Source parameter name.
35    pub name: syn::Ident,
36    /// Its source type and occurrence slot.
37    pub ty: TypeUse<S>,
38}
39/// A validated free function and its independently configurable type occurrences.
40#[derive(Clone, Debug)]
41pub struct Function<I, O> {
42    /// Unqualified name used by binding declarations.
43    pub name: syn::Ident,
44    /// Qualified path used to call the source function from generated Rust.
45    pub path: syn::Path,
46    /// Parameters in call order.
47    pub params: Vec<Param<Option<I>>>,
48    /// Result occurrence; an omitted Rust return type becomes `()`.
49    pub ret: TypeUse<Option<O>>,
50    /// Origin recorded by source capture, for diagnostics and qualification.
51    pub location: SourceLocation,
52}
53/// A named-field record available for field construction and projection.
54#[derive(Clone, Debug)]
55pub struct Record {
56    /// Qualified record type.
57    pub ty: Type,
58    /// Public fields in declaration order.
59    pub fields: Vec<Field>,
60}
61/// A named record field, independent of function input and output data.
62#[derive(Clone, Debug)]
63pub struct Field {
64    /// Source field name.
65    pub name: syn::Ident,
66    /// Qualified source type.
67    pub ty: Type,
68}
69/// A fieldless Rust enum, retaining source discriminant expressions.
70#[derive(Clone, Debug)]
71pub struct FieldlessEnum {
72    /// Qualified source enum type.
73    pub ty: Type,
74    /// Variants in source order, without an imposed foreign representation.
75    pub variants: Vec<EnumVariant>,
76}
77/// A variant with no payload fields.
78#[derive(Clone, Debug)]
79pub struct EnumVariant {
80    /// Source variant name.
81    pub name: syn::Ident,
82    /// Explicit source discriminant, or an implicit successor when absent.
83    pub discriminant: Option<syn::Expr>,

86/// Source structure with independently configured input and output conversions.
87///
88/// Every parameter carries `Option<I>`; every result carries `Option<O>`.
89/// Record fields describe source structure. Slot payloads belong to the consumer;
90/// generators may use [`Conversion`] for expression
91/// emission. Source-only inspection uses `Flat<(), ()>` with empty slots.
92///
93/// Use [`Flat::parse`] or [`Flat::parse_selected`] to validate source syntax and
94/// qualify declared names. Aliases stay opaque. Named types must have nonzero
95/// Rust size; generated code must emit [`Self::guards`] to enforce that rule.
96///
97/// ```
98/// use prebindgen_flat::Flat;
99/// let items = syn::parse_file("pub fn add(x: i64, y: i64) -> i64 { x + y }")?.items;
100/// let flat = Flat::<(), ()>::parse(
101///     items.into_iter().map(|item| (item, Default::default())))?;
102/// assert!(flat.function("add")?.params[0].ty.slot.is_none());
103/// # Ok::<(), Box<dyn std::error::Error>>(())
104/// ```
105#[derive(Clone, Debug)]
106pub struct Flat<I = (), O = ()> {
107    /// Free functions indexed by unqualified source name.
108    pub functions: BTreeMap<String, Function<I, O>>,
109    /// Records indexed by `type_key` of their qualified type.
110    pub records: BTreeMap<String, Record>,
111    /// Fieldless enums indexed by qualified source type.
112    pub enums: BTreeMap<String, FieldlessEnum>,
113    /// Declared type names mapped to qualified Rust types, including opaque aliases.
114    pub types: BTreeMap<String, Type>,
115    /// Constants every adapter must emit: captured feature guards and generated
116    /// nonzero-size assertions. Layout assertions run when the bindings compile.
117    pub guards: Vec<syn::ItemConst>,
118}

120impl<I, O> Flat<I, O> {
121    /// Select the source declarations needed by this binding before validating
122    /// the supported flat subset. Items outside the selection are skipped without validation.
123    ///
124    /// Dependency discovery follows unqualified declared type names in function
125    /// signatures and record fields transitively. A type referenced only through
126    /// a qualified path (such as `crate::Foo`) must also be selected explicitly
127    /// through `types` using its unqualified name (`Foo`), or included by parsing
128    /// the complete item set with [`Self::parse`]. Opaque alias implementations
129    /// are not inspected. Unnamed constant guards are retained. Selected items
130    /// are validated by [`Self::parse`].
131    ///
132    /// # Arguments
133    ///
134    /// * `items` — captured source items with their locations, as returned by
135    ///   `prebindgen::Source::items_all`.
136    /// * `functions` — unqualified names of the functions to keep, including the
137    ///   constructors and getters that declarations refer to.
138    /// * `types` — further types to keep, with the types they refer to.
139    pub fn parse_selected(
140        items: impl IntoIterator<Item = (syn::Item, SourceLocation)>,
141        functions: &[String],
142        types: &[Type],
143    ) -> Result<Self> {
144        let items: Vec<_> = items.into_iter().collect();
145        fn name(item: &syn::Item) -> Option<String> {
146            match item {
147                syn::Item::Fn(i) => Some(i.sig.ident.to_string()),
148                syn::Item::Struct(i) => Some(i.ident.to_string()),
149                syn::Item::Enum(i) => Some(i.ident.to_string()),
150                syn::Item::Type(i) => Some(i.ident.to_string()),
151                _ => None,
152            }
153        }
154        struct References<'a>(&'a mut BTreeSet<String>);
155        impl VisitMut for References<'_> {
156            fn visit_type_path_mut(&mut self, path: &mut syn::TypePath) {
157                if path.qself.is_none() && path.path.segments.len() == 1 {
158                    self.0.insert(path.path.segments[0].ident.to_string());
159                }
160                syn::visit_mut::visit_type_path_mut(self, path);
161            }
162        }
163        let mut needed: BTreeSet<_> = functions.iter().cloned().collect();
164        for ty in types {
165            References(&mut needed).visit_type_mut(&mut ty.clone());
166        }
167        loop {
168            let old = needed.len();
169            for (item, _) in &items {
170                if !name(item).is_some_and(|n| needed.contains(&n)) {
171                    continue;
172                }
173                let mut visitor = References(&mut needed);
174                match item {
175                    syn::Item::Fn(f) => visitor.visit_signature_mut(&mut f.sig.clone()),
176                    syn::Item::Struct(s) => visitor.visit_fields_mut(&mut s.fields.clone()),
177                    // Aliases are opaque: their implementation is not part of the boundary.
178                    _ => {}
179                }
180            }
181            if needed.len() == old {
182                break;
183            }
184        }
185        Self::parse(items.into_iter().filter(|(item, _)| {
186            matches!(item, syn::Item::Const(c) if c.ident == "_")
187                || name(item).is_some_and(|n| needed.contains(&n))
188        }))
189    }
190
191    /// Read captured free functions, named records, fieldless enums and opaque aliases.
192    /// Only exhaustive enums with unconditional fieldless variants belong to this
193    /// subset. Rejects duplicate names, non-public declarations, unsupported
194    /// signatures and types, and inaccessible record fields.
195    ///
196    /// Input and output slots start at `None`; no default conversion is required.
197    /// Unqualified declared types and callable paths are qualified with the captured
198    /// crate name (or `crate` if absent).
199    /// Opaque alias implementations are not inspected. Rust layout and trait
200    /// requirements are checked later when the generated code compiles.
201    ///
202    /// `items` are captured source items with their locations, as returned by
203    /// `prebindgen::Source::items_all`.
204    pub fn parse(items: impl IntoIterator<Item = (syn::Item, SourceLocation)>) -> Result<Self> {
205        let items: Vec<_> = items.into_iter().collect();
206        let mut names = BTreeSet::new();
207        let mut types = BTreeMap::new();
208        for (item, loc) in &items {
209            let visibility = match item {
210                syn::Item::Fn(f) => Some(&f.vis),
211                syn::Item::Struct(s) => Some(&s.vis),
212                syn::Item::Enum(e) => Some(&e.vis),
213                syn::Item::Type(t) => Some(&t.vis),
214                _ => None,
215            };
216            if visibility.is_some_and(|v| !matches!(v, syn::Visibility::Public(_))) {
217                return Err("flat declarations must be public".into());
218            }
219            let id = match item {
220                syn::Item::Fn(f) => &f.sig.ident,
221                syn::Item::Struct(s) => &s.ident,
222                syn::Item::Enum(e) => &e.ident,
223                syn::Item::Type(t) => &t.ident,
224                syn::Item::Const(c) if c.ident == "_" => continue,
225                _ => return Err(format!("unsupported flat item: {}", item.to_token_stream())),
226            };
227            if !names.insert(id.to_string()) {
228                return Err(format!("duplicate flat name `{id}`"));
229            }
230            if !matches!(item, syn::Item::Fn(_)) {
231                let module: syn::Path = syn::parse_str(
232                    &loc.crate_name
233                        .as_deref()
234                        .unwrap_or("crate")
235                        .replace('-', "_"),
236                )
237                .map_err(|e| e.to_string())?;
238                types.insert(id.to_string(), syn::parse_quote!(#module::#id));
239            }
240        }
241        let mut flat = Self {
242            functions: BTreeMap::new(),
243            records: BTreeMap::new(),
244            enums: BTreeMap::new(),
245            types,
246            guards: Vec::new(),
247        };
248        for (item, location) in items {
249            match item {
250                syn::Item::Fn(f) => {
251                    if !f.sig.generics.params.is_empty()
252                        || f.sig.generics.where_clause.is_some()
253                        || f.sig.asyncness.is_some()
254                        || f.sig.unsafety.is_some()
255                        || f.sig.variadic.is_some()
256                        || f.sig.abi.is_some()
257                    {
258                        return Err(format!(
259                            "`{}`: only safe, synchronous, nongeneric free functions are supported",
260                            f.sig.ident
261                        ));
262                    }
263                    let module: syn::Path = syn::parse_str(
264                        &location
265                            .crate_name
266                            .as_deref()
267                            .unwrap_or("crate")
268                            .replace('-', "_"),
269                    )
270                    .map_err(|e| e.to_string())?;
271                    let name = f.sig.ident;
272                    let mut params = Vec::new();
273                    for arg in f.sig.inputs {
274                        let syn::FnArg::Typed(p) = arg else {
275                            return Err("receivers are not flat parameters".into());
276                        };
277                        let syn::Pat::Ident(id) = *p.pat else {
278                            return Err("a parameter must have a name".into());
279                        };
280                        if id.subpat.is_some() || id.by_ref.is_some() {
281                            return Err("parameter patterns are unsupported".into());
282                        }
283                        params.push(Param {
284                            name: id.ident,
285                            ty: TypeUse {
286                                ty: flat.resolve_type(&p.ty)?,
287                                slot: None,
288                            },
289                        });
290                    }
291                    let ret = match f.sig.output {
292                        syn::ReturnType::Default => syn::parse_quote!(()),
293                        syn::ReturnType::Type(_, t) => *t,
294                    };
295                    let ret = TypeUse {
296                        ty: flat.resolve_type(&ret)?,
297                        slot: None,
298                    };
299                    flat.functions.insert(
300                        name.to_string(),
301                        Function {
302                            path: syn::parse_quote!(#module::#name),
303                            name,
304                            params,
305                            ret,
306                            location,
307                        },
308                    );
309                }
310                syn::Item::Struct(s) => {
311                    if !s.generics.params.is_empty() || s.generics.where_clause.is_some() {
312                        return Err("generic records are unsupported".into());
313                    }
314                    let syn::Fields::Named(fields) = s.fields else {
315                        return Err(
316                            "only named records are supported; use an alias for an opaque type"
317                                .into(),
318                        );
319                    };
320                    let ty = flat.types[&s.ident.to_string()].clone();
321                    let mut out = Vec::new();
322                    for field in fields.named {
323                        if !matches!(field.vis, syn::Visibility::Public(_)) {
324                            return Err("record fields must be public".into());
325                        }
326                        out.push(Field {
327                            name: field.ident.unwrap(),
328                            ty: flat.resolve_type(&field.ty)?,
329                        });
330                    }
331                    flat.records
332                        .insert(type_key(&ty), Record { ty, fields: out });
333                }
334                syn::Item::Enum(e) => {
335                    if e.attrs.iter().any(|a| {
336                        a.path().is_ident("non_exhaustive")
337                            || a.path().is_ident("cfg")
338                            || a.path().is_ident("cfg_attr")
339                    }) {
340                        return Err("conditional or non-exhaustive enums are unsupported".into());
341                    }
342                    if !e.generics.params.is_empty() || e.generics.where_clause.is_some() {
343                        return Err("generic enums are unsupported".into());
344                    }
345                    if e.variants.is_empty() {
346                        return Err("fieldless enums require at least one variant".into());
347                    }
348                    let ty = flat.types[&e.ident.to_string()].clone();
349                    let mut variants = Vec::new();
350                    let mut names = BTreeSet::new();
351                    for variant in e.variants {
352                        if !matches!(variant.fields, syn::Fields::Unit) {
353                            return Err("only fieldless enum variants are supported".into());
354                        }
355                        if variant.attrs.iter().any(|a| {
356                            a.path().is_ident("cfg")
357                                || a.path().is_ident("cfg_attr")
358                                || a.path().is_ident("non_exhaustive")
359                        }) {
360                            return Err(
361                                "conditional or non-exhaustive enum variants are unsupported"
362                                    .into(),
363                            );
364                        }
365                        if !names.insert(
366                            variant
367                                .ident
368                                .to_string()
369                                .trim_start_matches("r#")
370                                .to_string(),
371                        ) {
372                            return Err("duplicate enum variant".into());
373                        }
374                        variants.push(EnumVariant {
375                            name: variant.ident,
376                            discriminant: variant.discriminant.map(|(_, e)| e),
377                        });
378                    }
379                    flat.enums
380                        .insert(type_key(&ty), FieldlessEnum { ty, variants });
381                }
382                syn::Item::Type(t) => {
383                    if !t.generics.params.is_empty() || t.generics.where_clause.is_some() {
384                        return Err("generic aliases are unsupported".into());
385                    }
386                    // The alias's contents remain opaque, as in the flat source contract.
387                }
388                syn::Item::Const(c) => flat.guards.push(c),
389                _ => unreachable!(),
390            }
391        }
392        // Rust, not source syntax, determines layout (especially for opaque
393        // aliases). Keep this flat-model invariant in the guards all adapters emit.
394        for (name, ty) in &flat.types {
395            let message = format!("flat object `{name}` must have nonzero size");
396            flat.guards.push(syn::parse_quote! {
397                const _: () = assert!(::core::mem::size_of::<#ty>() != 0, #message);
398            });
399        }
400        Ok(flat)
401    }
402}

404impl<I, O> Flat<I, O> {
405    /// Look up a free function by its unqualified source name.
406    ///
407    /// Returns an error if it was not included in this model.
408    pub fn function(&self, name: &str) -> Result<&Function<I, O>> {
409        self.functions
410            .get(name)
411            .ok_or_else(|| format!("unknown function `{name}`"))
412    }
413    /// Look up a function for occurrence-slot configuration.
414    ///
415    /// Returns an error for an unknown source name; changing slots does not change
416    /// the original Rust parameter or result types.
417    pub fn function_mut(&mut self, name: &str) -> Result<&mut Function<I, O>> {
418        self.functions
419            .get_mut(name)
420            .ok_or_else(|| format!("unknown function `{name}`"))
421    }
422    /// Look up a named record by its qualified type syntax, `ty`.
423    ///
424    /// Use [`Self::resolve_type`] first for an unqualified declaration name. Opaque
425    /// aliases have no record entry and return an error.
426    pub fn record(&self, ty: &Type) -> Result<&Record> {
427        self.records
428            .get(&type_key(ty))
429            .ok_or_else(|| format!("`{}` is not a record", type_key(ty)))
430    }
431    /// Qualify the declared names in `ty` and validate the supported source-type grammar.
432    ///
433    /// Accepts primitives, declared types, unit, strings, byte vectors/slices and
434    /// `Option<T>`, `Result<T, E>`, vectors and slices recursively, and references without explicit lifetimes. A successful result does not mean
435    /// a consumer can use the type in every position. Aliases remain opaque.
436    pub fn resolve_type(&self, ty: &Type) -> Result<Type> {
437        let mut ty = ty.clone();
438        struct Names<'a>(&'a BTreeMap<String, Type>);
439        impl VisitMut for Names<'_> {
440            fn visit_type_mut(&mut self, ty: &mut Type) {
441                if let Type::Path(p) = ty {
442                    if p.qself.is_none() && p.path.segments.len() == 1 {
443                        let seg = &p.path.segments[0];
444                        if matches!(seg.arguments, syn::PathArguments::None) {
445                            if let Some(named) = self.0.get(&seg.ident.to_string()) {
446                                *ty = named.clone();
447                                return;
448                            }
449                        }
450                    }
451                }
452                syn::visit_mut::visit_type_mut(self, ty);
453                if let Some(inner) = vec_inner(ty) {
454                    let inner = inner.clone();
455                    *ty = syn::parse_quote!(Vec<#inner>);
456                }
457                if let Some((ok, err)) = result_parts(ty) {
458                    let (ok, err) = (ok.clone(), err.clone());
459                    *ty = syn::parse_quote!(::core::result::Result<#ok, #err>);
460                }
461                if let Some(inner) = option_inner(ty) {
462                    let inner = inner.clone();
463                    *ty = syn::parse_quote!(Option<#inner>);
464                }
465            }
466        }
467        Names(&self.types).visit_type_mut(&mut ty);
468        self.validate_type(&ty)?;
469        Ok(ty)
470    }
471    fn validate_type(&self, ty: &Type) -> Result<()> {
472        if Primitive::from_type(ty).is_some() || self.types.values().any(|t| same_type(t, ty)) {
473            return Ok(());
474        }
475        if let Some(inner) = option_inner(ty) {
476            return self.validate_type(inner);
477        }
478        if let Some(inner) = vec_inner(ty) {
479            return self.validate_type(inner);
480        }
481        if let Some((ok, err)) = result_parts(ty) {
482            self.validate_type(ok)?;
483            return self.validate_type(err);
484        }
485        if matches!(ty, Type::Path(p) if p.path.segments.last().is_some_and(|s| s.ident == "Result"))
486        {
487            return Err("spell Result<T, E> with both arguments, optionally qualified by std::result or core::result".into());
488        }
489        match ty {
490            Type::Tuple(t) if t.elems.is_empty() => Ok(()),
491            Type::Reference(r) if r.lifetime.is_none() => self.validate_type(&r.elem),
492            Type::Slice(s) => self.validate_type(&s.elem),
493            Type::Path(_)
494                if matches!(
495                    type_key(ty).as_str(),
496                    "String" | "str" | "u8" | "Vec < u8 >"
497                ) =>
498            {
499                Ok(())
500            }
501            _ => Err(format!("unsupported source type `{}`", type_key(ty))),
502        }
503    }
504    /// Transform input and output data separately, preserving source structure and guards.
505    ///
506    /// Functions are visited by name; within each function, parameters are visited
507    /// in declaration order, followed by its result. Record declarations pass
508    /// through unchanged. Stops on the first error.
509    ///
510    /// # Arguments
511    ///
512    /// * `input` — called for each parameter with its source type and old input
513    ///   slot; returns the new slot.
514    /// * `output` — called for each result with its source type and old output
515    ///   slot; returns the new slot.
516    pub fn map_slots<I2, O2>(
517        self,
518        mut input: impl FnMut(&Type, Option<I>) -> Result<Option<I2>>,
519        mut output: impl FnMut(&Type, Option<O>) -> Result<Option<O2>>,
520    ) -> Result<Flat<I2, O2>> {
521        let mut functions = BTreeMap::new();
522        for (name, fun) in self.functions {
523            let params = fun
524                .params
525                .into_iter()
526                .map(|p| {
527                    let slot = input(&p.ty.ty, p.ty.slot)?;
528                    Ok(Param {
529                        name: p.name,
530                        ty: TypeUse { ty: p.ty.ty, slot },
531                    })
532                })
533                .collect::<Result<_>>()?;
534            let slot = output(&fun.ret.ty, fun.ret.slot)?;
535            functions.insert(
536                name,
537                Function {
538                    name: fun.name,
539                    path: fun.path,
540                    params,
541                    ret: TypeUse {
542                        ty: fun.ret.ty,
543                        slot,
544                    },
545                    location: fun.location,
546                },
547            );
548        }
549        Ok(Flat {
550            functions,
551            records: self.records,
552            enums: self.enums,
553            types: self.types,
554            guards: self.guards,
555        })
556    }
557}

559/// Return the token spelling of `ty`, used to index types.
560///
561/// This is syntactic identity, not Rust type equivalence or alias resolution.
562pub fn type_key(ty: &Type) -> String {
563    ty.to_token_stream().to_string()
564}
565/// Whether types `a` and `b` have the same token spelling; this does not resolve
566/// aliases or normalize types.
567pub fn same_type(a: &Type, b: &Type) -> bool {
568    type_key(a) == type_key(b)
569}

577/// Extract the payload type of a standard `Option<T>` spelling.
578pub fn option_inner(ty: &Type) -> Option<&Type> {
579    let Type::Path(p) = ty else { return None };
580    if p.qself.is_some() {
581        return None;
582    }
583    let names = p
584        .path
585        .segments
586        .iter()
587        .map(|s| s.ident.to_string())
588        .collect::<Vec<_>>()
589        .join("::");
590    if !matches!(
591        names.as_str(),
592        "Option" | "std::option::Option" | "core::option::Option"
593    ) {
594        return None;
595    }
596    let syn::PathArguments::AngleBracketed(a) = &p.path.segments.last()?.arguments else {
597        return None;
598    };
599    if a.args.len() != 1 {
600        return None;
601    }
602    match a.args.first()? {
603        syn::GenericArgument::Type(t) => Some(t),
604        _ => None,
605    }
606}

608/// Extract success and error types from a standard `Result<T, E>` spelling.
609/// Custom aliases and similarly named types in other modules remain opaque.
610pub fn result_parts(ty: &Type) -> Option<(&Type, &Type)> {
611    let Type::Path(p) = ty else { return None };
612    if p.qself.is_some() {
613        return None;
614    }
615    let names = p
616        .path
617        .segments
618        .iter()
619        .map(|s| s.ident.to_string())
620        .collect::<Vec<_>>()
621        .join("::");
622    if !matches!(
623        names.as_str(),
624        "Result" | "std::result::Result" | "core::result::Result"
625    ) {
626        return None;
627    }
628    // Generic arguments are permitted only on the final Result segment.
629    if p.path
630        .segments
631        .iter()
632        .take(p.path.segments.len() - 1)
633        .any(|s| !matches!(s.arguments, syn::PathArguments::None))
634    {
635        return None;
636    }
637    let syn::PathArguments::AngleBracketed(args) = &p.path.segments.last()?.arguments else {
638        return None;
639    };
640    if args.args.len() != 2 {
641        return None;
642    }
643    let mut args = args.args.iter();
644    match (args.next()?, args.next()?) {
645        (syn::GenericArgument::Type(ok), syn::GenericArgument::Type(err)) => Some((ok, err)),
646        _ => None,
647    }
648}
649
650/// Payload of a standard Vec spelling; byte vectors are included here.
651pub fn vec_inner(ty: &Type) -> Option<&Type> {
652    let Type::Path(p) = ty else { return None };
653    if p.qself.is_some() {
654        return None;
655    }
656    let name = p
657        .path
658        .segments
659        .iter()
660        .map(|s| s.ident.to_string())
661        .collect::<Vec<_>>()
662        .join("::");
663    if !matches!(name.as_str(), "Vec" | "std::vec::Vec" | "alloc::vec::Vec") {
664        return None;
665    }
666    let syn::PathArguments::AngleBracketed(args) = &p.path.segments.last()?.arguments else {
667        return None;
668    };
669    if args.args.len() != 1 {
670        return None;
671    }
672    match args.args.first()? {
673        syn::GenericArgument::Type(ty) => Some(ty),
674        _ => None,
675    }
676}
677/// Which Rust collection view an occurrence uses.
678#[derive(Clone, Copy, Debug, PartialEq, Eq)]
679pub enum SequenceView {
680    /// An owned Vec, whose elements may be transferred.
681    Owned,
682    /// A shared Vec reference.
683    Vector,
684    /// A shared slice reference.
685    Slice,
686    /// A mutable collection view, requiring unsupported write-back semantics.
687    Mutable,
688}
689/// Source collection shape, excluding byte vectors/slices which retain byte transport.
690pub fn sequence_parts(ty: &Type) -> Option<(&Type, SequenceView)> {
691    let (inner, reference) = match ty {
692        Type::Reference(r) => (r.elem.as_ref(), Some(r)),
693        _ => (ty, None),
694    };
695    let (element, slice) = if let Some(element) = vec_inner(inner) {
696        (element, false)
697    } else if let Type::Slice(s) = inner {
698        (s.elem.as_ref(), true)
699    } else {
700        return None;
701    };
702    if type_key(element) == "u8" {
703        return None;
704    }
705    let view = match reference {
706        Some(r) if r.mutability.is_some() => SequenceView::Mutable,
707        Some(_) if slice => SequenceView::Slice,
708        Some(_) => SequenceView::Vector,
709        None if slice => return None,
710        None => SequenceView::Owned,
711    };
712    Some((element, view))
713}
