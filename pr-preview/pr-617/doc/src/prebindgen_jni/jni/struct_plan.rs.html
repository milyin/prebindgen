1//! The shared recursive leaf plan of the data-class `fromParts` bridge.
2//!
3//! A whole-value struct crossing Rust→Kotlin is flattened into leaf wire
4//! slots: the Rust side encodes them and makes ONE
5//! `call_static_method("fromParts", …)`
6//! ([`flatten_struct_encode`](super::flatten_struct_encode)); the Kotlin side
7//! declares the matching `fromParts` factory that reassembles the object in
8//! bytecode ([`flatten_struct_factory`](super::flatten_struct_factory)). Both
9//! sides must enumerate the same leaves, in the same order, with matching
10//! wire slots and JVM descriptors.
11//!
12//! This module holds that agreement: [`build_struct_plan`] classifies every
13//! field ONCE, in one fixed priority order (projection → enum →
14//! `Option<enum>` → nested data-class → simple leaf), and both emitters walk
15//! the resulting [`StructPlan`] — so the two sides agree by construction
16//! instead of by hand-synchronized parallel walks.
17

36/// How a Rust-side simple leaf binds its encoded wire into the `JValue` slot.
37pub(crate) enum LeafForm {
38    /// Primitive wire: bind as the wire type, pass via `JValue::from`.
39    Prim,
40    /// `JString` / `JByteArray`: bind as `JObject` via `.into()`.
41    IntoObject,
42    /// Already-`JObject` wire (boxed `Option`, `List`, …): bind directly.
43    Object,
44}
45
46/// The COMPLETE Rust → wire conversion of one leaf: the rust-side stages a
47/// custom [`convert!`](prebindgen_registry::convert) declaration inserts (`Duration → u64`)
48/// followed by the wire-facing converter (`u64 → jlong`).
49///
50/// A leaf must carry the whole chain, not just
51/// [`converter_id`](prebindgen_registry::ConverterImpl::converter_id):
52/// calling only the wire-facing function would hand it the *semantic* value
53/// (a `Duration`) where it expects the *representation* (a `u64`), which does
54/// not compile. Structural wrappers (`Option<_>`, `Vec<_>`) already compose
55/// the chain; this is the same composition for the positions the flattened
56/// `fromParts` bridge encodes itself.
57pub(crate) struct ConvChain {
58    /// Rust-side stages in output execution order — each consumes the
59    /// previous one's result, the first consumes the Rust value.
60    pub stages: Vec<prebindgen_registry::OperationId>,
61    /// The wire-facing converter, applied last.
62    pub function: prebindgen_registry::OperationId,
63}
64
65impl ConvChain {
66    /// The artifacts this chain calls: its stages, then the wire-facing
67    /// converter.
68    pub(crate) fn calls(&self, out: &mut Vec<prebindgen_registry::write::ArtifactKey>) {
69        out.extend(
70            self.stages
71                .iter()
72                .chain([&self.function])
73                .cloned()
74                .map(prebindgen_registry::write::ArtifactKey::Operation),
75        );
76    }
77
78    /// Read the chain off a wire's resolved output conversion.
79    ///
80    /// `stages` comes from the wire rather than the converter: the stage order
81    /// is the fragment's conversion chain, and a `ConverterImpl` is the
82    /// wire-facing end of it alone.
83    fn of(
84        entry: &prebindgen_registry::ConverterImpl<KotlinMeta>,
85        stages: &[prebindgen_registry::generation::OperationId],
86    ) -> Self {
87        ConvChain {
88            stages: stages.to_vec(),
89            function: entry.converter_id().clone(),
90        }
91    }
92
93    /// The expression converting `value` (a Rust value expression) to this
94    /// leaf's wire form, propagating any stage error with `?`.
95    pub(crate) fn call(
96        &self,
97        env: &TokenStream,
98        value: &TokenStream,
99        base: &str,
100        emit: &prebindgen_registry::RustWriter,
101    ) -> TokenStream {
102        let function = emit.operation_ident("jni", &self.function);
103        if self.stages.is_empty() {
104            return quote! { #function(#env, #value.clone())? };
105        }
106        let mut body = TokenStream::new();
107        let mut previous = quote!(#value.clone());
108        for (order, stage) in self.stages.iter().enumerate() {
109            let stage = emit.operation_ident("jni", stage);
110            let next = format_ident!("__{}_s{}", base, order);
111            body.extend(quote! {
112                let #next = #stage(#env, #previous)
113                    .map_err(|__e| <__JniErr as ::core::convert::From<String>>::from(
114                        __e.to_string()))?;
115            });
116            previous = quote!(#next);
117        }
118        quote!({ #body #function(#env, #previous)? })
119    }
120}

122pub(crate) enum PlanFieldKind {
123    /// Projection leaf (opaque handle / `ULong`). Wire slot: `jlong` (`"J"`);
124    /// the factory rebuilds the typed value from `fqn`.
125    Projection {
126        conv: ConvChain,
127        proj: Projection,
128        fqn: String,
129    },
130    /// Bare enum → `jint` discriminant (`"I"`); factory calls `fromInt`.
131    Enum { conv: ConvChain, kotlin: KtType },
132    /// `Option<enum>` → primitive discriminant when the enum contributes a
133    /// niche, boxed `Integer` otherwise. The frozen Kotlin sentinel is the
134    /// same slot the registry-composed converter carved.
135    OptionEnum {
136        conv: ConvChain,
137        kotlin: KtType,
138        niche: Option<String>,
139    },
140    /// Nested plain data-class: its leaves inline here. `optional` prepends
141    /// a `present: Boolean` flag (`"Z"`) and defaults the child slots in the
142    /// `None` arm; the factory guards `Child.fromParts(…)` on the flag.
143    Nested {
144        optional: bool,
145        /// The child's registered Kotlin FQN (its `fromParts` owner). `None`
146        /// for an undeclared struct: the Rust encode can still inline it,
147        /// but the Kotlin factory (which must name the child class) aborts.
148        child_fqn: Option<String>,
149        plan: StructPlan,
150    },
151    /// Data-carrying enum (`sealed_class`): an `Int` **tag** slot naming the
152    /// live alternative, followed by one **leaf group per variant** laid side
153    /// by side. Exactly one group is live; the rest are wire-defaulted and
154    /// the tag tells both sides which to read.
155    ///
156    /// This is [`Self::Nested`]'s `optional` gating with `N` groups instead
157    /// of one and an `Int` tag instead of a `Boolean` flag — a unit-only
158    /// enum would degenerate to "just a tag", which is why `enum_class`
159    /// keeps its own simpler path.
160    Sum {
161        /// Path to the source enum, for the encoder's match arms.
162        source: syn::Path,
163        /// Kotlin FQN of the sealed interface, for the factory's `when`.
164        kotlin_fqn: String,
165        /// `Option<E>` keeps its own `present` flag ahead of the tag; the tag
166        /// domain is never overloaded with an "absent" value.
167        optional: bool,
168        /// Variants in declaration order; index == tag.
169        variants: Vec<SumPlanVariant>,
170    },
171    /// Simple leaf with its own output converter.
172    Leaf {
173        conv: ConvChain,
174        /// The converter's destination wire type (boxed: `syn::Type` is the
175        /// enum's size outlier).
176        wire: Box<syn::Type>,
177        form: LeafForm,
178        /// JVM descriptor of the slot (must match the factory param's type).
179        descriptor: String,
180        kotlin: KtType,
181        /// Kotlin-side `?` (an `Option` field whose wire is object-shaped).
182        nullable: bool,
183    },
184}

186impl StructPlan {
187    /// Every converter the encoder calls, field by field and through every
188    /// nested layout.
189    pub(crate) fn calls(&self, out: &mut Vec<prebindgen_registry::write::ArtifactKey>) {
190        for field in &self.fields {
191            field.kind.calls(out);
192        }
193    }
194}
195
196impl PlanFieldKind {
197    fn calls(&self, out: &mut Vec<prebindgen_registry::write::ArtifactKey>) {
198        match self {
199            Self::Projection { conv, .. }
200            | Self::Enum { conv, .. }
201            | Self::OptionEnum { conv, .. }
202            | Self::Leaf { conv, .. } => conv.calls(out),
203            Self::Nested { plan, .. } => plan.calls(out),
204            Self::Sum { variants, .. } => {
205                for field in variants.iter().flat_map(|variant| &variant.fields) {
206                    field.kind.calls(out);
207                }
208            }
209        }
210    }
211}

213/// One alternative of a [`PlanFieldKind::Sum`].
214pub(crate) struct SumPlanVariant {
215    /// Variant ident as declared in Rust — the encoder's match pattern.
216    pub rust_ident: syn::Ident,
217    /// Variant class name in Kotlin (after any `variant!(V).name(...)`).
218    pub kotlin_name: String,
219    /// This variant's payload, in declaration order. Empty for a unit
220    /// variant — the group that contributes nothing but its tag.
221    pub fields: Vec<SumPlanField>,
222}

224/// One payload field of a [`SumPlanVariant`]. Classified by exactly the same
225/// [`classify_field`] a struct field goes through, so a payload and a struct
226/// field of the same Rust type get the same slot, wire and Kotlin type.
227pub(crate) struct SumPlanField {
228    /// How the field is addressed in the encoder's match pattern.
229    pub member: syn::Member,
230    /// Slot-name fragment, `<variantCamel>_<prop>` (`exact_v0`). The Kotlin
231    /// property name it embeds is recomputed where needed from
232    /// [`sum_field_prop_name`], so there is one derivation rather than a
233    /// stored copy that could disagree with it.
234    pub slot: String,
235    pub kind: PlanFieldKind,
236}

238/// Classify `s`'s fields into the shared bridge plan. `None` aborts the
239/// whole-value bridge (an unresolved field converter or a missing Kotlin
240/// name) — consistently for BOTH sides, where the former parallel walks
241/// could silently diverge on such edge cases.
242pub(crate) fn build_struct_plan(
243    ext: &Declarations,
244    registry: &impl Conversions,
245    s: &prebindgen_registry::flat::Struct,
246    depth: usize,
247) -> Option<StructPlan> {
248    assert!(
249        depth <= 16,
250        "struct fromParts plan: recursion too deep at struct `{}` (cyclic data_class?)",
251        s.name
252    );
253    let mut fields: Vec<PlanField> = Vec::new();
254    for field in &s.fields {
255        // A tuple struct is an `Extern` in the model, never a `Struct`, so a
256        // nameless field cannot reach here.
257        let fname = field.name.as_ref()?.clone();
258        let owner = format!("{}.{}", s.name, fname);
259        let kind = classify_field(ext, registry, &field.ty, &owner, depth)?;
260        fields.push(PlanField { fname, kind });
261    }
262    Some(StructPlan { fields })
263}
264
265impl Declarations {
266    /// The one whole-value struct layout used by Rust encoding and Kotlin
267    /// declaration/factory emission. During resolution it is memoized; after
268    /// resolution the lookup is served exclusively by
269    /// [`crate::jni::generation::JniGenerationPlan`].
270    pub(crate) fn struct_plan(
271        &self,
272        registry: &impl Conversions,
273        s: &prebindgen_registry::flat::Struct,
274        depth: usize,
275    ) -> Option<std::rc::Rc<StructPlan>> {
276        let key = s.type_ref().key();
277        if let Some(generation) = &self.generation {
278            return generation.struct_plan(&key);
279        }
280        if let Some(hit) = self.struct_plans.borrow().get(&key) {
281            return hit.clone();
282        }
283        let plan = build_struct_plan(self, registry, s, depth).map(std::rc::Rc::new);
284        self.struct_plans.borrow_mut().insert(key, plan.clone());
285        plan
286    }
287}

289/// True when this plan's flattened `fromParts` takes at least one **raw
290/// native pointer** leaf — an opaque-handle projection, whose wire slot is a
291/// bare `jlong` the factory mints a handle from.
292///
293/// This is what decides whether that factory needs the raw-pointer guard.
294/// Marking every `fromParts` was over-broad: a pointer-free factory such as
295/// `Timestamp.fromParts(Long, ByteArray)` cannot forge anything, and guarding
296/// it only removed a safe factory from Java and forced unrelated consumers to
297/// opt into a contract it does not have.
298///
299/// Recursive for the same reason the factory is: a nested data class inlines
300/// its leaves into the parent's signature, so a handle two levels down still
301/// arrives as a raw `Long` parameter of the *parent's* `fromParts`.
302pub(crate) fn plan_mints_handle(plan: &StructPlan) -> bool {
303    plan.fields.iter().any(|f| kind_mints_handle(&f.kind))
304}
305
306fn kind_mints_handle(kind: &PlanFieldKind) -> bool {
307    match kind {
308        PlanFieldKind::Projection { proj, .. } => proj.kind == ProjectionKind::Handle,
309        PlanFieldKind::Nested { plan, .. } => plan_mints_handle(plan),
310        PlanFieldKind::Sum { variants, .. } => variants
311            .iter()
312            .flat_map(|v| &v.fields)
313            .any(|f| kind_mints_handle(&f.kind)),
314        // `ULong` is a value, not a pointer; the rest are enums and plain
315        // leaves that carry no address.
316        PlanFieldKind::Enum { .. }
317        | PlanFieldKind::OptionEnum { .. }
318        | PlanFieldKind::Leaf { .. } => false,
319    }
320}

322/// Classify ONE value position — a struct field or a sum's variant payload —
323/// into its bridge slot. Both callers go through here so a payload and a
324/// struct field of the same Rust type get the same slot, wire, descriptor and
325/// Kotlin type; the alternative (a second classification walk) is exactly the
326/// drift `StructPlan` exists to prevent.
327///
328/// `owner` is the dotted path used in diagnostics (`Config.mode`,
329/// `Reading::Exact.v0`).
330pub(crate) fn classify_field(
331    ext: &Declarations,
332    registry: &impl Conversions,
333    reading: &prebindgen_registry::flat::TypeRef,
334    owner: &str,
335    depth: usize,
336) -> Option<PlanFieldKind> {
337    // The **reading**, not a spelling. Every layer question below is answered
338    // from `kind` and cannot fail: holding a `TypeRef` is proof the model
339    // classified this type. Taking a `syn::Type` meant asking the registry per
340    // question, and a type it had never seen answered "no layer" rather than
341    // saying so — which is the missing `?` of #273 waiting to happen again.
342
343    // A sum is classified FIRST, because it is the one kind with no converter
344    // of its own: it crosses as a tag plus one leaf group per variant, never
345    // as a single wire, which is why `sealed_class` types are declared
346    // boundary-only. Demanding an output entry before this point would send
347    // every sum-typed field down the `None` path and fail the whole parent's
348    // plan with an unresolved-converter error naming the wrong thing.
349    // `Vec` is peeled alongside `Option` here purely to CLASSIFY: `type_kind`
350    // answers about a bare ident, so it reports `Vec<Reading>` as `Other` and
351    // a rejection guarded on the unpeeled type could never fire. Peeling first
352    // is what makes the `Vec<sum>` error reachable at all.
353    // Every layer question below is the MODEL's, asked once: a field spelled
354    // `Box<Option<T>>` is `Optional` and must classify, nest and render exactly
355    // as `Option<T>` does. Peeling by path segment answered "not optional" for
356    // it, and the seven peels in this function would then disagree with each
357    // other about the same field (#273).
358    let optional_inner = reading.optional_inner();
359    let bare_ref = optional_inner.unwrap_or(reading);
360    let seq_elem = bare_ref.sequence_elem();
361    let core = seq_elem.unwrap_or(bare_ref);
362    if matches!(ext.type_kind(registry, &core.key()), TypeKind::Sum) {
363        // A `Vec` of tag-gated groups has variable arity, exactly like a `Vec`
364        // of nested data classes — the flattened bridge is fixed-layout by
365        // construction.
366        if seq_elem.is_some() {
367            panic!(
368                "fromParts bridge: `Vec<{}>` sealed-class field (`{owner}`) is not supported \
369                 (variable arity)",
370                core,
371            );
372        }
373        return sum_plan_kind(
374            ext,
375            registry,
376            bare_ref,
377            owner,
378            optional_inner.is_some(),
379            depth,
380        );
381    }
382
383    let field_entry = ext.out_frag(reading)?;
384    field_entry.activate();
385    let stages = field_entry.stages();
386    let conv = ConvChain::of(&field_entry, &stages);
387
388    {
389        // Projection leaf (opaque handle / `ULong`).
390        if let Some(proj) = field_entry.metadata.projection.clone() {
391            if matches!(proj.strategy, FoldStrategy::Iterable(_)) {
392                panic!(
393                    "fromParts bridge: collection (`Vec<projection>`) field `{owner}` is not \
394                     supported — add array codegen to lift this guard"
395                );
396            }
397            let fqn = projection_leaf_kt(ext, &proj)?.to_string();
398            return Some(PlanFieldKind::Projection { conv, proj, fqn });
399        }
400        // Enum leaf, bare or under `Option` — asked ONCE, of the model, and of
401        // the already-peeled reading beside us.
402        //
403        // It used to ask `is_kotlin_enum` twice, of two spellings. That answers
404        // about the WRAPPER: `builder.rs` documents `Box<Priority>` as `false`
405        // for it and `true` for the reading form, so a wrapped enum field fell
406        // through to the plain-leaf arm and rendered as its wire instead of the
407        // Kotlin enum class — the #273 family, output-side. `flat_input.rs` had
408        // already moved to the reading; this is the other half of the same
409        // question finally giving the same answer.
410        //
411        // Optionality stays the CALLER's fact rather than the probe's:
412        // `enum_probe` peels `Option` as well as borrows, so asking it about
413        // the unpeeled reading would make `Priority` and `Option<Priority>`
414        // indistinguishable and collapse the two arms into one.
415        if ext.is_kotlin_enum_reading(bare_ref) {
416            return match optional_inner {
417                None => {
418                    let kotlin = field_entry.metadata.kotlin_name.clone()?;
419                    Some(PlanFieldKind::Enum { conv, kotlin })
420                }
421                Some(inner) => {
422                    let kotlin = ext.out_frag(inner)?.metadata.kotlin_name.clone()?;
423                    let niche = crate::jni::compile::option_enum_niche(
424                        ext,
425                        reading,
426                        prebindgen_registry::recipe::Direction::Deconstruct,
427                    );
428                    Some(PlanFieldKind::OptionEnum {
429                        conv,
430                        kotlin,
431                        niche,
432                    })
433                }
434            };
435        }
436        // Nested plain data-class (optionally under `Option`).
437        //
438        // A `Vec<data class>` does NOT arrive here and needs no guard of its
439        // own (#217). `type_kind` answers `DataStruct` only for a key that is a
440        // single identifier, which a `Vec<_>` key never is — so this branch
441        // cannot be entered with a sequence in hand, and the field falls
442        // through to the simple-leaf arm below. That is the right answer rather
443        // than a missed one: the field stays ONE slot whose own converter is
444        // the element's fixed folder, so the bridge keeps its fixed slot count
445        // and the elements still cross as raw leaves. #217 expected this to
446        // need array codegen — a count slot plus a per-element sub-plan in all
447        // three producers — and it does not, because the sequence never has to
448        // enter the fixed layout at all.
449        //
450        // The `Vec<sum>` refusal above is a different question and stays: a sum
451        // has no converter of its own, so there is no single slot to fall
452        // through to.
453        let inner_ty = bare_ref;
454        if let TypeKind::DataStruct { st, cfg } = ext.type_kind(registry, &inner_ty.key()) {
455            let child_fqn = cfg
456                .and_then(|c| c.name_spec.as_ref())
457                .map(|s| ext.fqn_of(s));
458            let plan = build_struct_plan(ext, registry, st, depth + 1)?;
459            return Some(PlanFieldKind::Nested {
460                optional: optional_inner.is_some(),
461                child_fqn,
462                plan,
463            });
464        }
465        // Simple leaf: derive the slot descriptor and the Rust binding form
466        // from the converter's wire — the one place this decision is made.
467        let wire = field_entry.destination.clone();
468        let kotlin = field_entry.metadata.kotlin_name.clone()?;
469        let (form, descriptor) = match jni_field_access(&wire) {
470            Some((sig, _, false)) => (LeafForm::Prim, sig.to_string()),
471            Some((sig, _, true)) => (LeafForm::IntoObject, sig.to_string()),
472            None => {
473                // Object-shaped wire with no fixed descriptor; the JVM slot
474                // must be the field's actual declared type (Option-stripped).
475                // Option-stripped off the MODEL: `optional_inner` is the
476                // layer's own reading, so there is nothing to re-look-up.
477                let slot = optional_inner.unwrap_or(reading);
478                let descriptor = ext
479                    .out_frag(slot)
480                    .and_then(|e| jni_field_access(&e.destination))
481                    .and_then(|(sig, _, is_obj)| {
482                        if is_obj {
483                            Some(sig.to_string())
484                        } else {
485                            // The inner type's own wire is a primitive, so
486                            // this field is an `Option<primitive-wire>` whose
487                            // converter delivers the `box_j*`-boxed OBJECT
488                            // (null for `None`) — the JVM slot is the box
489                            // class, not the primitive.
490                            box_descriptor_for_primitive(sig).map(str::to_string)
491                        }
492                    })
493                    .or_else(|| {
494                        // The NAME off the classification, not off the last
495                        // path segment: `Box<T>` IS `T` here, and taking the
496                        // spelling apart would answer about the wrapper.
497                        match slot.unwrapped().kind() {
498                            prebindgen_registry::flat::TypeKind::Named { id, .. } => id.ident(),
499                            _ => None,
500                        }
501                        .and_then(|name| {
502                            ext.kotlin_fqn(&TypeKey::from_ident(&name))
503                                .map(|v| format!("L{};", v.replace('.', "/")))
504                        })
505                    })
506                    .or_else(|| {
507                        // A run of values is what `kind` says it is.
508                        // `pat_match_top(.., "Vec")` compared the last path
509                        // segment, so a `Box<Vec<T>>` answered false.
510                        if slot.sequence_elem().is_some() {
511                            Some("Ljava/util/List;".to_string())
512                        } else {
513                            // The wire table already names every reference wire's
514                            // descriptor — String and the eight primitive arrays.
515                            jni_field_access(&wire).map(|(sig, _, _)| sig.to_string())
516                        }
517                    })
518                    .unwrap_or_else(|| "Ljava/lang/Object;".to_string());
519                (LeafForm::Object, descriptor)
520            }
521        };
522        let nullable = optional_inner.is_some() && !is_jni_primitive(&wire);
523        Some(PlanFieldKind::Leaf {
524            conv,
525            wire: Box::new(wire),
526            form,
527            descriptor,
528            kotlin,
529            nullable,
530        })
531    }
532}

534impl PlanFieldKind {
535    /// The Kotlin type of the `data class` **constructor property** this field
536    /// becomes.
537    ///
538    /// The class declaration, the `fromParts` factory and the Rust encoder are
539    /// three views of one classification, so all three read it from here — the
540    /// module docs' "agree by construction instead of by hand-synchronized
541    /// parallel walks" applied to the declaration too (#156). Deriving it
542    /// separately is what let a property's type disagree with its own
543    /// factory parameter.
544    ///
545    /// `owner` is the dotted path used in diagnostics.
546    pub(crate) fn property_type(&self, owner: &str) -> KtType {
547        match self {
548            // A projection's typed surface is its folded shape over the leaf
549            // class (`ZKeyExpr?`, `List<ZKeyExpr>`, `ULong`), which the plan
550            // already resolved into `fqn`.
551            PlanFieldKind::Projection { proj, fqn, .. } => {
552                handle_kt_type(&proj.strategy, &KtType::cls(fqn))
553            }
554            PlanFieldKind::Enum { kotlin, .. } => kotlin.clone(),
555            PlanFieldKind::OptionEnum { kotlin, .. } => kotlin.clone().nullable(),
556            PlanFieldKind::Nested {
557                optional,
558                child_fqn,
559                ..
560            } => {
561                let fqn = child_fqn.as_ref().unwrap_or_else(|| {
562                    panic!(
563                        "data class property `{owner}`: nested data-class field has no \
564                         registered Kotlin class — declare the child type in a package"
565                    )
566                });
567                let t = KtType::cls(fqn);
568                if *optional {
569                    t.nullable()
570                } else {
571                    t
572                }
573            }
574            PlanFieldKind::Sum {
575                kotlin_fqn,
576                optional,
577                ..
578            } => {
579                let t = KtType::cls(kotlin_fqn);
580                if *optional {
581                    t.nullable()
582                } else {
583                    t
584                }
585            }
586            // `nullable` is the plan's own rule — an `Option` field whose wire
587            // is object-shaped. An `Option` over a PRIMITIVE wire stays
588            // non-null, because the encoder passes the bare primitive with a
589            // sentinel and the JVM slot must match (`J`, not `Ljava/lang/Long;`).
590            PlanFieldKind::Leaf {
591                kotlin, nullable, ..
592            } => {
593                if *nullable {
594                    kotlin.clone().nullable()
595                } else {
596                    kotlin.clone()
597                }
598            }
599        }
600    }
601
602    /// The close strategy when this field **reaches** an owned native handle,
603    /// so the class implements `AutoCloseable` and `close()` walks it.
604    ///
605    /// Only an **owned** `Handle` projection is a handle: a `ULong` owns
606    /// nothing, and a borrowed handle is not this object's to release. But
607    /// reaching one is not the same as *being* one — a handle held inside a sum
608    /// payload or a nested data class is just as much this object's to release,
609    /// and used to fall through to `None` (#218). That made ownership depend on
610    /// how the field happened to be spelled: swapping a handle field for an
611    /// enum carrying that handle silently moved the free onto the consumer,
612    /// with nothing in the generated Kotlin saying so.
613    ///
614    /// The wrapped cases fold as `Base`/`Optional` over the field itself rather
615    /// than over the handle inside it, because the field's own generated type
616    /// is `AutoCloseable` too — a sum's `close()` is the `when` over its
617    /// alternatives (`Declarations::build_sealed_class`), a nested data
618    /// class's is this same cascade one level down. So every
619    /// container emits the same plain `field.close()`, and the walk into the
620    /// wrapper lives once, in the wrapped type, instead of at each use site.
621    pub(crate) fn destructible(&self) -> Option<FoldStrategy> {
622        match self {
623            PlanFieldKind::Projection { proj, .. }
624                if matches!(proj.kind, ProjectionKind::Handle) && proj.owned =>
625            {
626                Some(proj.strategy.clone())
627            }
628            // `sequence: false` is `classify_field`'s guarantee, not an
629            // assumption: it rejects `Vec<sum>` and `Vec<data class>` outright,
630            // so a classified field of either kind is never a sequence.
631            PlanFieldKind::Sum {
632                optional, variants, ..
633            } if variants.iter().any(SumPlanVariant::destructible) => {
634                Some(whole_value_close(*optional, false))
635            }
636            PlanFieldKind::Nested { optional, plan, .. } if plan.destructible() => {
637                Some(whole_value_close(*optional, false))
638            }
639            _ => None,
640        }
641    }
642}

661/// The fold of a wrapper that closes itself: the value is closed as a whole,
662/// `?.`-guarded when it is optional and `forEach`-ed when it is a sequence.
663/// Shared by the two forms of the reaches-a-handle question below so they
664/// cannot answer differently.
665///
666/// [`NullableKind::Boxed`] is not a guess here. The receiver is always a
667/// generated Kotlin *reference* — a sum or a data class — whose absent form is
668/// a JVM null; a niche encoding is a wire fact of a handle projection, and
669/// those come back carrying their own strategy without passing through this.
670fn whole_value_close(optional: bool, sequence: bool) -> FoldStrategy {
671    let mut fold = FoldStrategy::Base;
672    if sequence {
673        fold = FoldStrategy::Iterable(Box::new(fold));
674    }
675    if optional {
676        fold = FoldStrategy::Optional(NullableKind::Boxed, Box::new(fold));
677    }
678    fold
679}

681/// How to close a value of this type, or `None` when it **reaches** no owned
682/// native handle and so has nothing to release —
683/// [`PlanFieldKind::destructible`]'s question asked of a type rather than of an
684/// already-classified plan field.
685///
686/// Two callers hold a [`TypeRef`](prebindgen_registry::flat::TypeRef) and no
687/// plan: the sealed-interface emitter, deciding whether a sum is
688/// `AutoCloseable` and what each variant class's `close()` body does; and the
689/// callback interface builder, deciding whether a reassembled whole value is
690/// the proxy's to close after `run`.
691///
692/// The two forms must **agree wherever both answer**, and that is a tested
693/// invariant, not a structural one: `a_types_close_answer_matches_its_plans`
694/// asserts `type_close_strategy(ty).is_some() == plan(ty).destructible()` over
695/// every field of every declared shape in a set covering all the ways one can
696/// reach a handle. It is worth pinning because the walks are not identical.
697/// Two places they differ:
698///
699/// * **Order.** [`classify_field`] classifies a `Sum` *before* it consults
700///   `output_entry` (deliberately — see its comment); this asks `output_entry`
701///   first. Safe only while sums carry no converter of their own, which is a
702///   precondition rather than a construction.
703/// * **Totality.** A field [`classify_field`] refuses collapses the whole
704///   struct's plan to `None`, while this refuses nothing. So on a subtree the
705///   bridge rejects, the two answer differently *by design* — the plan
706///   builders are what diagnose those, with the path, and "is there anything
707///   to close" still has a defensible answer for every type.
708///
709/// A disagreement costs a Kotlin compile error in one direction and a silent
710/// leak — #218 again, at this seam — in the other.
711///
712/// `depth` bounds the same recursion `build_struct_plan` and `sum_plan_kind`
713/// bound, and asserts on the same bound rather than answering: a cycle deep
714/// enough to trip this is a declaration the plan builders already refuse
715/// loudly, and returning `None` for it would report "nothing to close" — the
716/// leak direction — for a shape nobody can compile anyway.
717pub(crate) fn type_close_strategy(
718    ext: &Declarations,
719    registry: &impl Conversions,
720    ty: &prebindgen_registry::flat::TypeRef,
721    depth: usize,
722) -> Option<FoldStrategy> {
723    assert!(
724        depth <= 16,
725        "close-strategy walk: recursion too deep at type `{}` (cyclic data_class?)",
726        ty
727    );
728    // An owned `Handle` projection is the one thing that actually owns
729    // something: a `ULong` owns nothing, and a borrowed handle is not ours to
730    // release. Asked of the whole reading, so the `Option`/`Vec` folds the
731    // projection carries come back in its own strategy.
732    if let Some(proj) = ext.out_frag(ty).and_then(|e| e.metadata.projection.clone()) {
733        return (matches!(proj.kind, ProjectionKind::Handle) && proj.owned)
734            .then(|| proj.strategy.clone());
735    }
736    // Peel the layers the model names, exactly as `classify_field` does: what
737    // a `Box<Option<T>>` reaches is what `T` reaches.
738    let bare = ty.optional_inner().unwrap_or(ty);
739    let core = bare.sequence_elem().unwrap_or(bare);
740    let reaches = match ext.type_kind(registry, &core.key()) {
741        TypeKind::Sum => core
742            .key()
743            .ident()
744            .and_then(|ident| registry.flat().declared_type(&ident))
745            .is_some_and(|ty| match ty {
746                prebindgen_registry::flat::Type::Variant(sum) => {
747                    sum.alternatives.iter().any(|alt| {
748                        alt.fields
749                            .iter()
750                            .any(|f| type_close_strategy(ext, registry, &f.ty, depth + 1).is_some())
751                    })
752                }
753                _ => false,
754            }),
755        TypeKind::DataStruct { st, .. } => st
756            .fields
757            .iter()
758            .any(|f| type_close_strategy(ext, registry, &f.ty, depth + 1).is_some()),
759        TypeKind::Handle | TypeKind::Enum | TypeKind::Other => false,
760    };
761    // Put back exactly the layers peeled above. `reaches` was answered about
762    // the ELEMENT, so a `Vec<sum-that-reaches-a-handle>` must close each
763    // element — `close()` on the `List` itself would not compile. The field
764    // bridge rejects that shape, but this predicate exists precisely for the
765    // callers that hold no plan and so meet no rejection.
766    reaches.then(|| {
767        whole_value_close(
768            ty.optional_inner().is_some(),
769            bare.sequence_elem().is_some(),
770        )
771    })
772}

774/// Build the [`PlanFieldKind::Sum`] for a `sealed_class`-declared enum: one
775/// leaf group per variant, each payload classified through
776/// [`classify_field`].
777///
778/// Recursion is bounded by this function's own depth guard (see below) — a
779/// sum reaching its own type has no `jobject_input`-style escape hatch, since
780/// the flatten plan is finite by construction.
781///
782/// `None` propagates the resolver's **deferral** protocol: a payload whose
783/// converter has not resolved *yet* means "retry on the next fixed-point
784/// iteration", exactly as [`classify_field`] signals it for a struct field.
785/// Panicking there instead would turn a transient state into a build failure
786/// whenever a payload's converter happened to resolve later than this plan
787/// was first attempted.
788fn sum_plan_kind(
789    ext: &Declarations,
790    registry: &impl Conversions,
791    ty: &prebindgen_registry::flat::TypeRef,
792    owner: &str,
793    optional: bool,
794    depth: usize,
795) -> Option<PlanFieldKind> {
796    // Sum expansion needs its OWN depth guard. A sum whose payload is a sum
797    // never passes through `build_struct_plan`, so that function's assert —
798    // the only one on this recursion before now — cannot see a chain made
799    // purely of sums. Rust's sizedness rules make an unindirected cycle
800    // impossible to declare, and every indirection either classifies as
801    // `Other` (`Box<E>` is not a bare ident) or is already rejected
802    // (`Vec<E>`), so this is defence in depth rather than a reachable path
803    // today. It costs one comparison and makes the bound true for every
804    // future shape instead of true-by-accident.
805    assert!(
806        depth <= 16,
807        "fromParts bridge: sealed-class expansion too deep at `{owner}` (recursive sum?)"
808    );
809    // The key's ident: a sum is a declared name, and the key is that name when
810    // the key is one identifier.
811    let ident = ty.key().ident().unwrap_or_else(|| {
812        panic!("fromParts bridge: sealed-class field `{owner}` is not a path type")
813    });
814    // The sum as the MODEL holds it: its alternatives' payloads are `TypeRef`s
815    // already, so classifying one asks nothing and cannot be asked about a type
816    // the model never saw. One lookup, not two — the `enum_item` that used to
817    // sit beside this only fed a `SumSpec` of what the element already says.
818    let Some(prebindgen_registry::flat::Type::Variant(sum)) = registry.flat().declared_type(&ident)
819    else {
820        panic!("fromParts bridge: sealed-class field `{owner}`: `{ident}` is not an indexed sum")
821    };
822    let key = TypeKey::from_ident(&ident);
823    let cfg = ext
824        .types
825        .get(&key)
826        .unwrap_or_else(|| panic!("fromParts bridge: `{ident}` is not declared"));
827    let sum_cfg = cfg
828        .sum()
829        .unwrap_or_else(|| panic!("fromParts bridge: `{ident}` is not a sealed class"));
830    let kotlin_fqn = cfg
831        .name_spec
832        .as_ref()
833        .map(|s| ext.fqn_of(s))
834        .unwrap_or_else(|| panic!("fromParts bridge: sealed class `{ident}` has no Kotlin name"));
835
836    let mut variants: Vec<SumPlanVariant> = Vec::new();
837    for alt in &sum.alternatives {
838        let kotlin_name = ext.sum_variant_class_name(sum_cfg, &alt.name);
839        let mut fields: Vec<SumPlanField> = Vec::new();
840        for field in &alt.fields {
841            let member = field.member();
842            let prop = sum_field_prop_name(&member);
843            let slot = sum_slot_fragment(&kotlin_name, &prop);
844            let owner = format!("{ident}::{}.{prop}", alt.name);
845            // `?` — a payload whose converter has not resolved yet defers the
846            // whole plan to the next iteration, it does not fail the build.
847            let kind = classify_field(ext, registry, &field.ty, &owner, depth + 1)?;
848            fields.push(SumPlanField { member, slot, kind });
849        }
850        variants.push(SumPlanVariant {
851            rust_ident: alt.name.clone(),
852            kotlin_name,
853            fields,
854        });
855    }
856
857    Some(PlanFieldKind::Sum {
858        source: {
859            let module = ext.fn_module(registry, &ident);
860            syn::parse_quote!(#module::#ident)
861        },
862        kotlin_fqn,
863        optional,
864        variants,
865    })
866}

868/// Kotlin property name of one sum payload field — a named field keeps its
869/// camelCased name, a tuple field becomes `v0`, `v1`. Must agree with the
870/// sealed-interface emitter, which is why both call this.
871///
872/// Takes the **member**, which is the whole of what the name depends on: every
873/// caller holds a `flat::Field` and asks `Field::member()`. It took a
874/// `types_util::SumField` when a second description of a sum still existed
875/// beside the model's (#289).
876pub(crate) fn sum_field_prop_name(member: &syn::Member) -> String {
877    match member {
878        syn::Member::Named(id) => mangle_kotlin_ident(&kt_snake_to_camel(&id.to_string())),
879        syn::Member::Unnamed(i) => format!("v{}", i.index),
880    }
881}

883/// The wire tag of one alternative: its declaration-order index, as the `jint`
884/// the selector leaf carries.
885///
886/// One place, because the tag has to agree in three: the leaf's `group`, the
887/// Kotlin `when` arm, and the Rust `match` arm. Three separate `as i32` casts
888/// agreed by coincidence rather than by construction.
889///
890/// Deliberately **not** a checked conversion. `usize` → `i32` can truncate in
891/// general, but not here: the index counts alternatives of one enum, and an
892/// enum with `i32::MAX` variants is not a thing rustc can be handed. A
893/// `try_from(..).expect(..)` would put a panic in the working path for a state
894/// the compiler cannot produce, which is the shape this crate avoids.
895pub(crate) fn sum_tag(alt: &prebindgen_registry::flat::Alternative) -> i32 {
896    alt.index as i32
897}

899/// Slot-name fragment for one variant field: `<variantCamel>_<prop>`. Keyed
900/// on the **Kotlin** variant name so a `variant!(V).name(...)` rename carries
901/// through to the slots.
902pub(crate) fn sum_slot_fragment(kotlin_variant: &str, prop: &str) -> String {
903    let mut chars = kotlin_variant.chars();
904    let head: String = match chars.next() {
905        Some(c) => c.to_lowercase().collect(),
906        None => String::new(),
907    };
908    format!("{head}{}_{prop}", chars.as_str())
