20/// The JNI adapter's answer for one site.
21///
22/// Both variants are boxed. A `PlanLeaf` embeds whole sub-plans and runs to
23/// ~856 bytes, a `ValueOutputPlan` to ~608, and every site pays the larger of
24/// the two — the same reason `FnOutputPlan::Value` boxes its own payload.
25pub(crate) enum JPlan {
26    /// One parameter, in the seven-way wire layout the emitters branch on.
27    Param(std::rc::Rc<crate::jni::fn_plan::PlanLeaf>),
28    /// A return that crosses as one value: what it converts through, and what
29    /// the Kotlin surface declares.
30    Return(std::rc::Rc<crate::jni::fn_plan::ValueOutputPlan>),
31    /// A return the binding takes apart: the values it hands out, in the order
32    /// the builder receives them.
33    ///
34    /// Consumed by ordinary decomposed-return delivery. Callback delivery
35    /// takes the same result in the following Invoke stage.
36    Decomposed(std::rc::Rc<Vec<OutWire>>),
37}
38
39impl JPlan {
40    /// This plan as a parameter's, or `None` if it is a return's.
41    pub(crate) fn param(self) -> Option<std::rc::Rc<crate::jni::fn_plan::PlanLeaf>> {
42        match self {
43            JPlan::Param(leaf) => Some(leaf),
44            _ => None,
45        }
46    }
47
48    /// This plan as a return's, or `None` if it is a parameter's.
49    pub(crate) fn returned(self) -> Option<std::rc::Rc<crate::jni::fn_plan::ValueOutputPlan>> {
50        match self {
51            JPlan::Return(plan) => Some(plan),
52            _ => None,
53        }
54    }
55
56    /// The values a decomposed return hands out, or `None` if it is not one.
57    pub(crate) fn decomposed(self) -> Option<std::rc::Rc<Vec<OutWire>>> {
58        match self {
59            JPlan::Decomposed(wires) => Some(wires),
60            _ => None,
61        }
62    }
63}

81/// Whether this input can select the registry's allocation-free Optional
82/// `pair` row.
83///
84/// This predicate intentionally reads the already-compiled payload fragment:
85/// niches, projections, and converter stages are destination facts and cannot
86/// be inferred correctly from Flat syntax.
87pub(crate) fn optional_pair_plan_candidate(ext: &Declarations, arg: &TypeRef) -> bool {
88    // A wrapper outside Optional is accepted only when the composed source
89    // policy can rebuild it. Ask before peeling, because Flat deliberately
90    // erases transparent wrappers from the structural kind.
91    if crate::jni::trait_impl::build_through_erased_wrappers(arg, quote!(__probe)).is_none() {
92        return false;
93    }
94    let Some(inner) = arg.optional_inner() else {
95        return false;
96    };
97    if inner.borrow_target().is_some() {
98        return false;
99    }
100    let Some(inner_entry) = ext.in_frag(inner) else {
101        return false;
102    };
103    JniPrim::from_wire(&inner_entry.wire).is_some()
104        && inner_entry.niches.clone().carve().is_none()
105        && inner_entry.metadata.projection.is_none()
106        && inner_entry.0.chain.steps().is_empty()
107}

109/// Kotlin spelling of the niche consumed by the outer Optional enum layer.
110/// The source is the inner fragment's next free slot: nested option fragments
111/// re-export their remaining slots, so each layer naturally takes a different
112/// discriminant without a second allocation policy in the renderer.
113pub(crate) fn option_enum_niche(
114    ext: &Declarations,
115    reading: &TypeRef,
116    direction: Direction,
117) -> Option<String> {
118    option_enum_niches(ext, reading, direction)
119        .into_iter()
120        .next()
121}

123/// Kotlin spellings of every niche consumed by nested Optional enum layers,
124/// outside-in. Input sites use the first one to encode Kotlin `null` as the
125/// outer `None`; output wrappers accept all of them because the deliberately
126/// collapsed Kotlin surface cannot distinguish `None` from `Some(None)`.
127pub(crate) fn option_enum_niches(
128    ext: &Declarations,
129    reading: &TypeRef,
130    direction: Direction,
131) -> Vec<String> {
132    let mut current = reading;
133    let mut sentinels = Vec::new();
134    while let Some(inner) = current.optional_inner() {
135        if !ext.is_kotlin_enum_reading(inner) {
136            break;
137        }
138        let fragment = match direction {
139            Direction::Construct => ext.in_frag(inner),
140            Direction::Deconstruct => ext.out_frag(inner),
141        };
142        let Some(sentinel) =
143            fragment.and_then(|fragment| fragment.metadata.niche_sentinels.first().cloned())
144        else {
145            break;
146        };
147        sentinels.push(sentinel);
148        current = inner;
149    }
150    sentinels
151}

153/// Describe the two-leaf ABI of a bare `Option<primitive>` / `Option<enum>`
154/// input, or leave the crossing on its one-value representation.
155pub(crate) fn optional_pair_plan(
156    ext: &Declarations,
157    param_name: &syn::Ident,
158    arg: &TypeRef,
159    root: &JFrag,
160) -> Option<OptionalPairPlan> {
161    optional_pair_plan_candidate(ext, arg).then_some(())?;
162    let inner = arg.optional_inner()?;
163    let inner_entry = ext.in_frag(inner)?;
164    let value_wire = inner_entry.wire.clone();
165    let prim = JniPrim::from_wire(&value_wire)?;
166    let is_enum = ext.is_kotlin_enum_reading(inner);
167    let chain = root.composed_chain()?;
168    if chain.layout.leaf_count() != 2 {
169        return None;
170    }
171    chain.activate();
172    Some(OptionalPairPlan {
173        present_ident: format_ident!("{}_present", param_name),
174        value_ident: format_ident!("{}_value", param_name),
175        value_wire,
176        chain,
177        present_kt: snake_to_camel(&format!("{}_present", param_name)),
178        value_kt: snake_to_camel(&format!("{}_value", param_name)),
179        value_kt_type: prim.kotlin_type().to_string(),
180        value_kt_zero: prim.kotlin_zero().to_string(),
181        is_enum,
182    })
183}

185/// JniGen's half of the canonical generation plan.
186///
187/// Declared here so a JNI fragment can state its conversion chain in the
188/// registry's vocabulary — [`prebindgen_registry::generation::ConversionChain`]
189/// — rather than in a row of its own. #613 step 4 is what puts JNI fragments
190/// and sites into `FragmentPlan` / `SitePlan`; this states the representation
191/// those will be over, and the chain is its first live use.
192/// One fragment's converter, and the two facts a reader of the canonical plan
193/// needs beside it.
194///
195/// `FragmentPlan` carries the shape, the chain, the niches and the identity;
196/// what it cannot carry is adapter syntax, and both of these are exactly that.
197/// The wire is the JNI type this fragment's converter produces or consumes —
198/// adapter syntax, not captured source syntax, which is what
199/// [`Representation`](prebindgen_registry::generation::Representation) permits
200/// an artifact to retain. The layout is this fragment's own nesting over its
201/// ABI leaves, which no other part of the plan states:
202/// [`AbiLayout`](prebindgen_registry::generation::AbiLayout) is a **site's**
203/// answer, and `ShapePlan` does not imply it — a transparent wrapper over an
204/// optional leaves the shape composed and the layout absent.
205#[derive(Clone)]
206pub(crate) struct JConverterArtifact {
207    /// The private Rust converter this fragment renders.
208    pub(crate) rust: crate::jni::chain::JFunction,
209    /// The JNI wire this converter produces (deconstruct) or consumes
210    /// (construct).
211    pub(crate) wire: syn::Type,
212    /// This fragment's own nesting over its ABI leaves, when it composes.
213    pub(crate) layout: Option<JLayout>,
214    /// The Kotlin-facing extras this conversion carries: the class name a
215    /// value surfaces as, the reading an error peel rides, the handle or
216    /// unsigned projection, and any niche sentinels. Adapter payload, which is
217    /// what an artifact is for — and the last thing `Conv` reaches for through
218    /// `ConverterImpl` rather than through the plan (#613 step 5c).
219    pub(crate) metadata: KotlinMeta,
220    /// Bit-patterns this fragment's wire can hold but its converter never
221    /// produces and rejects on input.
222    pub(crate) niches: prebindgen_registry::Niches,
223    /// The wire-facing conversion's identity.
224    pub(crate) converter: OperationId,
225    /// Semantic stages rendered beside the wire-facing converter. A fragment
226    /// can emit more than one converter function, and this is where it says so
227    /// — adapter payload, so the canonical plan carries it without the registry
228    /// needing a multi-artifact fragment (#613 step 8).
229    pub(crate) stages: Vec<crate::jni::chain::JFunction>,
230}

232impl JConverterArtifact {
233    /// Whether the wire is a raw pointer, which decides how a child value is
234    /// handed to its parent converter.
235    pub(crate) fn wire_is_pointer(&self) -> bool {
236        matches!(self.wire, syn::Type::Ptr(_))
237    }
238
239    /// Whether this fragment occupies several ABI leaves under one nesting.
240    pub(crate) fn layout_is_composed(&self) -> bool {
241        self.layout.as_ref().is_some_and(JLayout::is_composed)
242    }

268/// The JNI adapter's answer for one crossing.
269///
270/// What a `ConverterImpl` was, minus the bookkeeping the table now does.
271#[derive(Clone)]
272pub(crate) struct JFrag {
273    /// This fragment's canonical identity and the crossing it answers — what
274    /// [`FragmentPlan`](prebindgen_registry::generation::FragmentPlan) is keyed
275    /// and spelled by. Carried here so freezing reads them off the fragment
276    /// rather than re-deriving them from a site that has moved on.
277    pub(crate) id: prebindgen_registry::generation::FragmentId,
278    pub(crate) source: TypeRef,
279    /// The niche this fragment **consumed** to encode its own absence, when it
280    /// did. An `Option` over a value whose wire has a free bit-pattern spends
281    /// one of them on `None` and exposes only the rest; the spent slot lives
282    /// on in the bridge that tests it, so without recording it here the frozen
283    /// plan could describe what a parent may still use but not what this
284    /// fragment took (#621 review).
285    pub(crate) consumed_niche: Option<prebindgen_registry::niches::NicheSlot>,
286    /// The registry-composed shape of this fragment, in the canonical

291    pub(crate) conv: ConverterImpl<KotlinMeta>,
292    /// The conversions between this fragment's wire value and its source
293    /// value, in **execution order for this fragment's direction** — from the
294    /// wire when constructing, from the source when deconstructing.
295    ///
296    /// The registry's chain rather than a row of this adapter's own. It
297    /// replaced `ConverterImpl::pre_stages`, which stored one order and left
298    /// each reader to reverse it for input; storing execution order means the
299    /// reversal happens once, where the chain is built and the direction is
300    /// already in hand.
301    pub(crate) chain: prebindgen_registry::generation::ConversionChain<JRepresentation>,
302    pub(crate) rust: crate::jni::chain::JFunction,
303    /// Frozen semantic stages beside the wire-facing converter. The
304    /// compatibility ConverterImpl keeps marker functions for ordering and
305    /// call names; these are the artifacts that render those markers.
306    pub(crate) rust_stages: Vec<crate::jni::chain::JFunction>,
307    pub(crate) yields: Yield,
308    /// Shape of the single adapter-side intermediate over flattened ABI leaves.
309    pub(crate) layout: Option<JLayout>,
310    /// Payload composition retained until `choice` supplies the variant.
311    choice_arm: Option<JChoiceArmPlan>,
312    /// Composed element retained by a containing sequence for callback folding.
313    pub(crate) nested_chain: Option<ComposedChain>,
314    /// The wire values this crossing occupies, when it occupies more than the

316    ///

341    /// own conversion to emit, so "has wires" is the wrong test for what to
342    /// leave out of the file.
343    pub(crate) composed_only: bool,
344}

346/// A site's ordered ABI is the leaves that cross, and nothing else.
347///
348/// Not the fragment's `JLayout`: that is how the fragment's own intermediate
349/// nests, which is a different fact and a different count — a `data_class`
350/// parameter is one intermediate over two native parameters. Cross-checking
351/// them by count asserted they were the same thing, and they are not.
352/// What a site's ordered leaves are, which depends on which side of the
353/// boundary the site sits on.
354#[derive(Clone)]
355// `Whole` and `Invoked` carry the plan a site converts through, and until step
356// 5b's readers move onto the canonical plan nothing in production reads it back
357// — the site set is a check today. The other two variants are read by `len`.
358#[cfg_attr(not(test), allow(dead_code))]
359pub(crate) enum JAbiLeaves {
360    /// A parameter: its ordered native positions AND the operations that
361    /// decode them — the Rust and Kotlin param ops and the pipeline. The whole
362    /// leaf, shared, because those are what an emitter reads and a carrier
363    /// that held only the descriptors could not replace the plan.
364    Params(std::rc::Rc<crate::jni::fn_plan::PlanLeaf>),
365    /// The values a decomposed return hands out, in builder order.
366    Decomposed(std::rc::Rc<Vec<OutWire>>),
367    /// One returned value: the plan that converts it, which is the plan the
368    /// emitters read rather than the intermediate the site hook produces.
369    Whole(std::rc::Rc<crate::jni::fn_plan::ValueOutputPlan>),
370    /// One value a callback delivers whole: the argument's own output ABI,
371    /// shared with the `JInvokePart` the trampoline renders. A returned value
372    /// and a delivered one are converted by different plans, so they are
373    /// different layouts — what they have in common is occupying one leaf.
374    Invoked(std::rc::Rc<OutValueAbi>),
375}
376
377impl JAbiLeaves {
378    /// How many ordered ABI **leaves** these are — not how many JVM descriptor
379    /// slots they occupy. One native parameter is one leaf; `jvm_slots` says
380    /// how wide that leaf is in a descriptor, which is a different count and
381    /// summing it here made a two-slot `long` look like two leaves.
382    pub(crate) fn len(&self) -> usize {
383        match self {
384            Self::Params(leaf) => leaf.native.len(),
385            Self::Decomposed(wires) => wires.len(),
386            Self::Whole(_) | Self::Invoked(_) => 1,
387        }
388    }
389}

439/// Structural JNI intermediate layout. Leaves are ABI values; Products nest
440/// them into the one tuple type consumed or produced by the registry chain.
441#[derive(Clone)]
442pub(crate) enum JLayout {
443    Leaf,
444    Product(Vec<JLayout>),
445    Optional(Box<JLayout>),
446    Choice(Vec<JLayout>),
447}
448
449impl JLayout {
450    pub(crate) fn leaf_count(&self) -> usize {
451        match self {
452            Self::Leaf => 1,
453            Self::Product(parts) => parts.iter().map(Self::leaf_count).sum(),
454            Self::Optional(inner) => 1 + inner.leaf_count(),
455            Self::Choice(arms) => 1 + arms.iter().map(Self::leaf_count).sum::<usize>(),
456        }
457    }
458
459    pub(crate) fn expression(&self, leaves: &[syn::Ident]) -> TokenStream {
460        fn build(layout: &JLayout, leaves: &[syn::Ident], next: &mut usize) -> TokenStream {
461            match layout {
462                JLayout::Leaf => {
463                    let leaf = &leaves[*next];
464                    *next += 1;
465                    quote!(#leaf)
466                }
467                JLayout::Product(parts) => {
468                    let values = parts.iter().map(|part| build(part, leaves, next));
469                    quote!((#(#values,)*))
470                }
471                JLayout::Optional(inner) => {
472                    let present = build(&JLayout::Leaf, leaves, next);
473                    let value = build(inner, leaves, next);
474                    quote!((#present, #value))
475                }
476                JLayout::Choice(arms) => {
477                    let tag = build(&JLayout::Leaf, leaves, next);
478                    let arms = arms.iter().map(|arm| build(arm, leaves, next));
479                    quote!((#tag, #(#arms,)*))
480                }
481            }
482        }
483        assert_eq!(self.leaf_count(), leaves.len());
484        build(self, leaves, &mut 0)
485    }
486
487    pub(crate) fn is_composed(&self) -> bool {
488        matches!(self, Self::Product(_) | Self::Optional(_) | Self::Choice(_))
489    }
490
491    pub(crate) fn pattern(&self, leaves: &[syn::Ident]) -> TokenStream {
492        self.expression(leaves)
493    }
494}

504/// One step of the walk from the object a site names to a wire's value.
505#[derive(Clone, Debug, PartialEq, Eq)]
506pub(crate) struct Nav {
507    /// The Kotlin property read.
508    pub(crate) field: String,
509    /// Whether the value it is read **off** may be null, which makes the read a
510    /// safe call.
511    ///
512    /// Set by every gate above the step, not only the nearest: once a chain
513    /// passes through one `?.` every read after it is on a nullable value, so a
514    /// gate marks the whole chain below it rather than its first step.
515    pub(crate) gated: bool,
516}

518/// How Kotlin reaches one wire value, relative to the object the site names.
519///
520/// The walk is kept as steps rather than as one string, and that is what makes
521/// a gate composable: an optional is applied **after** the fields under it have
522/// composed, so it has to reach back and make every step below it a safe call.
523/// A string cannot say which of its dots are property reads — the `0.0` a
524/// non-nullable slot falls back to has one too.
525///
526/// Three forms, because two of them put the base in the **middle**: a sum's tag
527/// reads `when (<base>.f) { … }` and a sum's payload slot reads
528/// `(<base>.f as? I.V)?.v0`.
529#[derive(Clone, Debug, PartialEq, Eq)]
530pub(crate) enum Access {
531    /// `<base><walk><tail>` — the ordinary read. `tail` is whatever follows the
532    /// property chain: a presence comparison, an elvis default, a Kotlin enum's
533    /// `?.value`, or nothing.
534    Read {
535        /// The property chain from the base.
536        walk: Vec<Nav>,
537        /// What follows it.
538        tail: String,
539    },
540    /// `when (<base><walk>) { … }` — which alternative of a sum is live, as the
541    /// `jint` tag the arms below are numbered by.
542    Select {
543        /// The property chain reaching the sum value.
544        walk: Vec<Nav>,
545        /// One arm per alternative, in declaration order, without the `null`
546        /// one — a gate above adds that by setting `nullable`.
547        arms: Vec<String>,
548        /// Whether the value reached can be null, which is its own arm.
549        nullable: bool,
550    },
551    /// `(<base><walk> as? <class>)?.<read>[ ?: <zero>]` — one payload slot of
552    /// one alternative. Every alternative's slots cross on every call; the cast
553    /// yields null for the ones that are not live, and `zero` is what a
554    /// non-nullable wire carries instead.
555    Slot {
556        /// The property chain reaching the sum value.
557        walk: Vec<Nav>,
558        /// The Kotlin class of the alternative this slot belongs to. Empty
559        /// until [`Compile::choice`] names it — the arm's own composition
560        /// cannot, because a product hook is not told which alternative it is.
561        class: String,
562        /// What reads the payload off the cast alternative — `v0`, or
563        /// `v1?.value` for a Kotlin enum payload.
564        read: String,
565        /// What an inert slot carries, or `None` for a wire that rides a JVM
566        /// `null`.
567        zero: Option<String>,

571/// The Kotlin property chain, rooted at the object a site destructures.
572pub(crate) fn reached(base: &str, walk: &[Nav]) -> String {
573    let mut out = base.to_string();
574    for nav in walk {
575        out.push_str(if nav.gated { "?." } else { "." });
576        out.push_str(&nav.field);
577    }
578    out
579}

581impl Access {
582    /// An ordinary read of the base itself, with nothing walked.
583    fn read(tail: impl Into<String>) -> Self {
584        Access::Read {
585            walk: Vec::new(),
586            tail: tail.into(),
587        }
588    }
589
590    /// The Kotlin expression, rooted at the object this site destructures.
591    pub(crate) fn render(&self, base: &str) -> String {
592        match self {
593            Access::Read { walk, tail } => format!("{}{tail}", reached(base, walk)),
594            Access::Select {
595                walk,
596                arms,
597                nullable,
598            } => {
599                let arms = nullable
600                    .then(|| "null -> 0".to_string())
601                    .into_iter()
602                    .chain(arms.iter().cloned())
603                    .collect::<Vec<_>>()
604                    .join("; ");
605                format!("when ({}) {{ {arms} }}", reached(base, walk))
606            }
607            Access::Slot {
608                walk,
609                class,
610                read,
611                zero,
612            } => {
613                let zero = zero
614                    .as_ref()
615                    .map(|z| format!(" ?: {z}"))
616                    .unwrap_or_default();
617                format!("({} as? {class})?.{read}{zero}", reached(base, walk))
618            }
619        }
620    }
621
622    /// The property chain, whichever form the access takes.
623    fn walk_mut(&mut self) -> &mut Vec<Nav> {
624        match self {
625            Access::Read { walk, .. } | Access::Select { walk, .. } | Access::Slot { walk, .. } => {
626                walk
627            }
628        }
629    }
630
631    /// This access read from one field in, rather than from the object itself.
632    fn under(mut self, field: &str) -> Self {
633        self.walk_mut().insert(
634            0,
635            Nav {
636                field: field.to_string(),
637                gated: false,
638            },
639        );
640        self
641    }

965/// One stage in front of `inner`'s chain, in execution order for `direction`.
966///
967/// A stage converts between the fragment's own source value and the value the
968/// inner fragment carries, so the two endpoints are `ChainValue::Source` and
969/// the inner crossing's key. Which is `from` and which is `into` is the whole
970/// difference between the directions: constructing ends at the source value
971/// and deconstructing starts there, which is also why the inner steps run
972/// before the new one in one direction and after it in the other.
973fn staged_chain(
974    direction: Direction,
975    operation: OperationId,
976    inner_key: TypeKey,
977    inner: &prebindgen_registry::generation::ConversionChain<JRepresentation>,
978    failure: prebindgen_registry::generation::Failure,
979) -> prebindgen_registry::generation::ConversionChain<JRepresentation> {
980    use prebindgen_registry::generation::{ChainValue, Cleanup, ConversionChain, ConverterStep};
981    let (from, into) = match direction {
982        Direction::Construct => (
983            ChainValue::Intermediate(inner_key.clone()),
984            ChainValue::Source,
985        ),
986        Direction::Deconstruct => (
987            ChainValue::Source,
988            ChainValue::Intermediate(inner_key.clone()),
989        ),
990    };
991    let own = ConverterStep::new(from, into, operation, failure, Cleanup::None);
992    // `ChainValue::Source` names the source value of the fragment that OWNS
993    // the chain, so the inner fragment's steps change meaning when they are
994    // embedded in this one: what was the inner's source value is this
995    // fragment's `inner_key` carrier. Rebasing it is what keeps the steps
996    // adjacent — each step's `into` is the next step's `from` — with `Source`
997    // appearing only at the end of a construct chain and the start of a
998    // deconstruct one.
999    let rebase = |value: &ChainValue<TypeKey>| match value {
1000        ChainValue::Source => ChainValue::Intermediate(inner_key.clone()),
1001        ChainValue::Intermediate(key) => ChainValue::Intermediate(key.clone()),
1002    };
1003    let inner = inner.steps().iter().map(|step| {
1004        ConverterStep::new(
1005            rebase(step.from()),
1006            rebase(step.into()),
1007            step.operation().clone(),
1008            step.failure(),
1009            step.cleanup().clone(),
1010        )
1011    });
1012    let steps: Vec<ConverterStep<JRepresentation>> = match direction {
1013        Direction::Construct => inner.chain(std::iter::once(own)).collect(),
1014        Direction::Deconstruct => std::iter::once(own).chain(inner).collect(),
1015    };
1016    ConversionChain::Steps(steps)
1017}

1070    /// This fragment as a [`FragmentPlan`] — the canonical description, in the
1071    /// registry's own vocabulary rather than this adapter's.
1072    ///
1073    /// Nothing renders from it yet: #613 step 4 is putting JNI fragments and
1074    /// sites into the canonical plan, and this is the first half, built beside
1075    /// the fields it will replace so that the swap is a deletion rather than a
1076    /// rewrite — the shape #620 used for the same kind of move. Its only reader
1077    /// so far is the check that says the two describe the same fragment, which
1078    /// is why it carries a `dead_code` exemption outside tests; the exemption
1079    /// goes when the generation plan holds these instead of `JFrag`.
1080    ///
1081    /// A **composed-only** fragment freezes without an artifact. That is the
1082    /// canonical plan's existing answer for a fragment that emits nothing
1083    /// (`FragmentPlan::artifact` is already an `Option`), not a JNI concept:
1084    /// the `parts` recipe of a `data_class` occupies wires and states a
1085    /// composition, and has no converter of its own to render.
1086    pub(crate) fn freeze(&self) -> prebindgen_registry::generation::FragmentPlan<JRepresentation> {
1087        use prebindgen_registry::generation::{
1088            ConverterPlan, FragmentPlan, NichePlan, {self as generation},
1089        };
1090        let exposed: Vec<String> = self.conv.niches.slots.iter().map(niche_key).collect();
1091        // What this fragment SPENT, beside what it leaves for a parent. An
1092        // `Option` that encodes `None` in a free bit-pattern consumes one slot
1093        // and discriminates on it; the plan has to carry both domains, or it
1094        // describes a contract nothing can validate and a nested `Option` layer
1095        // cannot be reconstructed from it.
1096        let consumed: Vec<String> = self.consumed_niche.iter().map(niche_key).collect();
1097        let converter = ConverterPlan::with_chain(
1098            self.shape.clone(),
1099            // The authoritative conversion graph, not `Direct`: a `convert!`
1100            // declaration's stages, their order, and the artifacts they route
1101            // to live here since #615, and freezing without them would hand a
1102            // later reader a fragment with no stages at all.
1103            self.chain.clone(),
1104            NichePlan::new(consumed.len(), consumed, exposed),
1105            generation::Failure::Fallible,
1106            generation::Cleanup::None,
1107        );
1108        // The **shape-adjacent** intermediate, which is not always the wire.
1109        // A `convert!` declaration puts stages between the source value and the
1110        // value the shape converts — `Duration -> u64 -> jlong` — so for a
1111        // fragment with a chain the intermediate is the chain's own end: what
1112        // it produces when deconstructing, what it starts from when
1113        // constructing. Taking the destination unconditionally described a
1114        // chain that ends nowhere, which is what validating the collected
1115        // fragments as one plan reported (#621 review).
1116        let intermediate = match (&self.chain, self.id.direction()) {
1117            (generation::ConversionChain::Steps(steps), Direction::Deconstruct) => {
1118                steps.last().and_then(|step| match step.into() {
1119                    generation::ChainValue::Intermediate(key) => Some(key.clone()),
1120                    generation::ChainValue::Source => None,
1121                })
1122            }
1123            (generation::ConversionChain::Steps(steps), Direction::Construct) => {
1124                steps.first().and_then(|step| match step.from() {
1125                    generation::ChainValue::Intermediate(key) => Some(key.clone()),
1126                    generation::ChainValue::Source => None,
1127                })
1128            }
1129            (generation::ConversionChain::Direct, _) => None,
1130        }
1131        .unwrap_or_else(|| TypeKey::from_type(&self.conv.destination));
1132        let plan = FragmentPlan::new(
1133            self.id.clone(),
1134            self.source.clone(),
1135            intermediate,
1136            converter,
1137            self.yields.clone(),
1138        );
1139        match self.composed_only {
1140            true => plan,
1141            false => plan.with_artifact(self.artifact()),
1142        }
1143    }

1145    /// This fragment as the plan for one **site** that uses it.
1146    ///
1147    /// The multi-slot ABI is what JNI demonstrates and C does not need: a C
1148    /// site's value occupies the slots its own `CValue` states, while a JNI
1149    /// site may occupy several ordered leaves — a `data_class` parameter
1150    /// crossing as its fields, an optional as a presence flag ahead of its
1151    /// value — and `JLayout` is how their nesting is spelled. `AbiLayout`
1152    /// already carries exactly that pair: how many ordered leaves, and the
1153    /// adapter's own layout of them.
1154    ///
1155    /// Failure IS routed at every site, and `Some(())` is how JNI says so. The
1156    /// route carries no payload — a conversion failure goes to the error sink
1157    /// the wrapper already holds, so there is nothing to choose between, where
1158    /// a C site chooses an out-param or a panic. But presence is the semantic
1159    /// edge the canonical plan validates: a fallible fragment used by a site
1160    /// with no route is `MissingFailureRoute`, and freezing `None` made every
1161    /// JNI site that (#622 review).
1162    pub(crate) fn freeze_site(
1163        &self,
1164        bound: &prebindgen_registry::recipe::Bound,
1165        leaves: JAbiLeaves,
1166    ) -> prebindgen_registry::generation::SitePlan<JRepresentation> {
1167        use prebindgen_registry::generation::{AbiLayout, SiteId, SitePlan};
1168        // The leaves come from the plan that produced them — this does not
1169        // derive them a second time, which is the whole point: a carrier
1170        // derived beside the thing it carries is another parallel answer.
1171        let slots = leaves.len();
1172        SitePlan::new(
1173            SiteId::new(bound.site.clone()),
1174            bound.clone(),

1266/// How long what this conversion produces stays usable.
1267///
1268/// A property of the **conversion**, not of how the crossing was spelled. The
1269/// two disagree and the spelling is the wrong one to read: a `&T` output over a
1270/// declared opaque handle clones its referent into a fresh `Box`-handle, and a
1271/// `&str` output copies into a JVM string, so both are self-sufficient although
1272/// the crossing is a borrow.
1273fn validity_of(direction: Direction, mode: Mode) -> Validity {
1274    match (direction, mode) {
1275        // Rust to the JVM. Every JNI wire value is a `jlong` the Rust side
1276        // handed over or a JVM object the JVM now owns; nothing on this wire
1277        // points into the Rust value it came from.
1278        (Direction::Deconstruct, _) => Validity::SelfSufficient,
1279        // The JVM to Rust: Flat says whether the crossing lends the value. A
1280        // borrowed crossing is valid only for the call, which is exactly right
1281        // at a parameter and refused at a return.
1282        (Direction::Construct, Mode::Shared | Mode::Exclusive) => Validity::Borrowed,
1283        (Direction::Construct, Mode::Owned) => Validity::SelfSufficient,
1284    }
1285}

1601    /// Freeze this site into the canonical plan, taking its ordered ABI from
1602    /// the plan just built rather than deriving it again.
1603    fn freeze_site_of(&self, bound: &Bound, root: &JFrag, plan: &JPlan) {
1604        let leaves = match plan {
1605            // The whole leaf, not just its descriptors: `RustParamOp`,
1606            // `KotlinParamOp` and the pipeline are the site's operations, and
1607            // a carrier without them could not replace the plan (#622 review).
1608            JPlan::Param(leaf) => JAbiLeaves::Params(leaf.clone()),
1609            JPlan::Decomposed(wires) => JAbiLeaves::Decomposed(wires.clone()),
1610            // A whole return is frozen where its FINAL plan is built, with the
1611            // convert delivery attached — the hook's is an intermediate, and
1612            // freezing it made the carrier authoritative for something no
1613            // emitter reads (#622 review).
1614            JPlan::Return(_) => return,
1615        };
1616        let frozen = root.freeze_site(bound, leaves);
1617        // The carrier holds the SAME list the plan does, not a copy of it.
1618        // That is the property that makes it a replacement: a copy could only
1619        // be checked to match today, while one allocation has nothing to
1620        // match. Guarded here so a later `clone()` cannot quietly reintroduce
1621        // the second answer (#622 review).
1622        match (frozen.abi().payload(), plan) {
1623            (JAbiLeaves::Params(frozen), JPlan::Param(leaf)) => assert!(
1624                std::rc::Rc::ptr_eq(frozen, leaf),
1625                "the site's ABI is a copy of the plan's rather than the same leaf"
1626            ),
1627            (JAbiLeaves::Decomposed(frozen), JPlan::Decomposed(wires)) => assert!(
1628                std::rc::Rc::ptr_eq(frozen, wires),
1629                "the site's ABI is a copy of the plan's rather than the same list"
1630            ),
1631
1632            _ => panic!("a site's frozen ABI does not match the plan it was taken from"),
1633        }
1634        self.decls
1635            .site_plans
1636            .borrow_mut()
1637            .push(std::rc::Rc::new(frozen));
1638    }

1858    /// The borrow arms, which are neither a terminal nor an arity layer.
1859    fn borrow(&self, ty: &TypeRef, into_rust: bool) -> Option<ConverterImpl<KotlinMeta>> {
1860        let TypeKind::Ref {
1861            mutable,
1862            inner: borrowed,
1863            ..
1864        } = ty.unwrapped().kind()
1865        else {
1866            return None;
1867        };
1868        let mutable = *mutable;
1869        // The target through the accessor: an out-parameter's `MaybeUninit` is
1870        // the slot a `T` goes in, and it is the `T` that converts.
1871        let inner = ty.borrow_target().expect("a borrow");
1872        if into_rust {
1873            // An exclusive borrow crosses only when the borrowed value lives on
1874            // the Rust side — see `input_borrow`'s own note. `mutable` is read
1875            // off the `Ref` kind rather than through `is_exclusive_borrow`,
1876            // which answers `false` for a `&mut MaybeUninit<T>` out-parameter.
1877            let writes_reach_the_caller = !mutable
1878                || self
1879                    .decls
1880                    .types
1881                    .get(&borrowed.key())
1882                    .is_some_and(|cfg| cfg.is_opaque());
1883            if !writes_reach_the_caller {
1884                return None;
1885            }
1886            let mut c = self.decls.input_borrow(ty, inner)?;
1887            c.subs = vec![inner.key()];
1888            Some(c)
1889        } else {
1890            None
1891        }
1892    }

2680    /// Plan a Java List-backed Sequence as one intermediate collection.
2681    ///
2682    /// The registry owns source collection traversal, element conversion and
2683    /// collection construction. JNI supplies only its List iterator/appender
2684    /// operations. Primitive arrays, handle-vector builders and multi-slot
2685    /// element folds retain their specialized representations.
2686    fn planned_sequence(&self, at: At<'_>, elements: Mode, inner: &JFrag) -> Option<JFrag> {
2687        let source = at.crossing.value();
2688        let element = source.sequence_elem()?.clone();
2689        let direction = at.crossing.direction();
2690        match (direction, source.unwrapped().kind()) {
2691            (Direction::Construct, TypeKind::Vec(_)) => {}
2692            (Direction::Construct, TypeKind::Slice(_)) if at.crossing.mode() == Mode::Shared => {}
2693            (Direction::Deconstruct, TypeKind::Vec(_) | TypeKind::Slice(_)) => {}
2694            _ => return None,
2695        }
2696        // The same gate the product and choice planners apply. This one used to
2697        // omit the ownership clause, which a deconstructing `&Box<Vec<T>>`
2698        // needs: the emitted loop reads the source with `*v` and then consumes
2699        // it with `into_iter`, so a shared borrow would move the `Vec` out of
2700        // the `Box`.
2701        let wrappers = composable_wrappers(at.crossing)?;
2702
2703        let inner_kotlin = inner
2704            .conv
2705            .metadata
2706            .kotlin_name
2707            .clone()
2708            .or_else(|| self.decls.override_kotlin_name(&element.key(), None))?;
2709        let kotlin_name = self.decls.override_kotlin_name(
2710            &at.crossing.spelled().key(),
2711            Some(KtType::generic("List", [inner_kotlin])),
2712        );
2713        let projection = if direction == Direction::Deconstruct {
2714            inner
2715                .conv
2716                .metadata
2717                .projection
2718                .clone()
2719                .map(|projection| Projection {
2720                    strategy: FoldStrategy::Iterable(Box::new(projection.strategy)),
2721                    ..projection
2722                })
2723        } else {
2724            None
2725        };
2726
2727        let child_wire = inner.conv.destination.clone();
2728        let specialized_site = inner.wires.is_some()
2729            || inner.out_wires.is_some()
2730            || (!is_jobject_shaped_wire(&child_wire) && inner.conv.metadata.projection.is_some());
2731        if specialized_site {
2732            // Multi-slot elements cross at a site as a Kotlin-side fold or a
2733            // transient Vec handle. The Sequence row still owns the shape and
2734            // its surface metadata, but deliberately has no whole-list Rust
2735            // converter for those site-specific representations.
2736            let mut marker = JFrag::new(at, self.parts_marker(at, vec![element.key()]));
2737            marker.conv.destination = syn::parse_quote!(jni::objects::JObject);
2738            marker.conv.metadata = KotlinMeta {
2739                kotlin_name,
2740                value_reading: None,
2741                projection,
2742                niche_sentinels: Vec::new(),
2743            };
2744            marker.nested_chain = inner.composed_chain();
2745            marker.composed_only = true;
2746            return Some(marker);
2747        }
2748        if inner.composed_only || !is_jobject_shaped_wire(&child_wire) {
2749            return None;
2750        }
2751        if direction == Direction::Construct {
2752            reject_vec_of_handle(&inner.conv.metadata.projection, &element);
2753        }
2754        let child = Self::planned_child_mode(
2755            direction,
2756            elements,
2757            &inner.artifact(),
2758            &inner.chain,
2759            inner.conv.converter_id(),
2760        );
2761        let destination: syn::Type = syn::parse_quote!(jni::objects::JObject);
2762        // A shared slice input and an owned vector input both materialize the
2763        // same `Vec<T>` carrier. Give that produced model type to the plan,
2764        // so converter identity, name and emitted body are shared. The wrapper
2765        // still owns the crossing mode and adds the final borrow.
2766        let produced = if direction == Direction::Construct
2767            && matches!(source.kind(), TypeKind::Slice(_))
2768            && at.crossing.mode() == Mode::Shared
2769        {
2770            element.vector()
2771        } else {
2772            source.clone()
2773        };
2774        let operation = at.sequence_converter_for(&produced);
2775        let bridge = match direction {
2776            Direction::Construct => crate::jni::chain::JSequenceBridge::Input {
2777                child: Box::new(child_wire),
2778            },
2779            Direction::Deconstruct => crate::jni::chain::JSequenceBridge::Output,
2780        };
2781        let rust = crate::jni::chain::JFunction::sequence(crate::jni::chain::JSequencePlan {
2782            operation: operation.clone(),
2783            reachable: std::rc::Rc::new(std::cell::Cell::new(false)),
2784            dependencies: vec![inner.rust.clone()],
2785            mode: at.crossing.mode(),
2786            chain: prebindgen_registry::chain::Sequence {
2787                source: produced,
2788                element: element.clone(),
2789                direction,
2790                source_policy: crate::jni::chain::JSource { wrappers },
2791                bridge,
2792                child,
2793            },
2794        });
2795
2796        let niches = match direction {
2797            Direction::Construct => Niches::empty(),
2798            Direction::Deconstruct => default_niches_for_wire(&destination),
2799        };
2800        Some(JFrag {
2801            chain: prebindgen_registry::generation::ConversionChain::Direct,
2802            conv: ConverterImpl {
2803                destination,
2804                converter: operation,
2805                niches,
2806                metadata: KotlinMeta {
2807                    kotlin_name,
2808                    value_reading: None,
2809                    projection,
2810                    niche_sentinels: Vec::new(),
2811                },
2812                subs: vec![element.key()],
2813            },
2814            id: at.fragment_id(),
2815            source: at.crossing.spelled().clone(),
2816            consumed_niche: None,
2817            shape: prebindgen_registry::generation::ShapePlan::Atomic(rust.clone()),
2818            rust,
2819            rust_stages: Vec::new(),
2820            layout: Some(JLayout::Leaf),
2821            choice_arm: None,
2822            // Iterable callback delivery converts one element at a time. Keep
2823            // exposing the element chain there until Invoke composition owns
2824            // that site explicitly.
2825            nested_chain: inner.composed_chain(),
2826            wires: None,
2827            out_wires: None,
2828            composed_only: false,
2829            yields: Yield {
2830                ty: at.crossing.value().stripped_key(),

2920    /// Plan an Optional over one already-composed child intermediate without
2921    /// spelling its source type or generating its Rust body.
2922    ///
2923    /// Borrowed opaque handles use a dedicated frozen soundness-carrier plan;
2924    /// parts-only rows remain non-rendering markers.
2925    fn planned_optional(&self, at: At<'_>, inner: &JFrag, decoupled: bool) -> Option<JFrag> {
2926        let source = at.crossing.spelled();
2927        let element = source.optional_inner()?;
2928        let direction = at.crossing.direction();
2929        if let Some(plan) = self.planned_borrowed_optional_handle(at, inner) {
2930            return Some(plan);
2931        }
2932        let composed_child =
2933            !inner.composed_only && inner.layout.as_ref().is_some_and(JLayout::is_composed);
2934        match direction {
2935            Direction::Construct
2936                if inner.out_wires.is_some() || (inner.wires.is_some() && !composed_child) =>
2937            {
2938                return None;
2939            }
2940            Direction::Deconstruct
2941                if inner.wires.is_some() || (inner.out_wires.is_some() && !composed_child) =>
2942            {

3641        // A product whose own type is a sum is one **alternative's** payload:
3642        // the registry composes every arm through this hook and hands the lot
3643        // to `choice`. Which alternative that is stays `choice`'s to fill in,
3644        // being the only hook told — so both directions leave a hole here.

3646        if sum {
3647            let direction = at.crossing.direction();
3648            let mut frag = match direction {
3649                Direction::Construct => self.arm(at, parts),
3650                Direction::Deconstruct => self.out_arm(at, parts),
3651            };

3653                Direction::Construct => frag.wires.is_some(),
3654                Direction::Deconstruct => frag.out_wires.is_some(),

3660            // what `choice` reads back as that arm's parts.

3663        }
3664        if at.crossing.direction() == Direction::Deconstruct {

3666            product(&mut frag, parts);
3667            return Ok(frag);

3670        // contributes its own several — which is the recursion, stated once

3690                            .chain(t.iter().cloned())

3692                        }),
3693                        handle_nullable: w.handle_nullable,

3697                        // open, the part it belongs to closes.
3698                        field: w
3699                            .field
3700                            .clone()

3740    fn choice(
3741        &mut self,
3742        _cx: &mut Cx<'_>,
3743        at: At<'_>,
3744        arms: &[(&Alternative, &JFrag)],
3745    ) -> Frag<Self> {
3746        // Tagged selection among ordered arms, with each arm's exact arity part
3747        // of the frozen contract. Applied to whichever fragment answers below
3748        // except the whole-value decoder, which is a terminal.
3749        let chosen = |frag: &mut JFrag| {
3750            let parts: Vec<Vec<prebindgen_registry::generation::FragmentUse>> =
3751                arms.iter().map(|(_, arm)| arm_parts(arm)).collect();
3752            frag.shape = prebindgen_registry::generation::ShapePlan::Choice {
3753                bridge: prebindgen_registry::generation::ChoiceArity::new(
3754                    parts.iter().map(Vec::len).collect(),
3755                    frag.rust.clone(),
3756                ),
3757                arms: parts,
3758            };
3759        };
3760        match at.crossing.direction() {
3761            Direction::Construct => match self.selected(at, arms) {
3762                Some(wires) => {
3763                    if let Some(mut planned) =
3764                        self.planned_choice(at, arms, Some(wires.clone()), None)
3765                    {
3766                        chosen(&mut planned);
3767                        return Ok(planned);
3768                    }
3769                    let mut legacy = JFrag::new(at, self.parts_marker(at, parts_subs(arms)));
3770                    legacy.wires = Some(wires);
3771                    legacy.composed_only = true;
3772                    chosen(&mut legacy);
3773                    Ok(legacy)
3774                }
3775                // A payload this adapter has no slot for — a nested object or
3776                // handle — keeps the explicit whole-value sealed-class decoder.
3777                None => {
3778                    let conv = self
3779                        .decls
3780                        .in_frag(at.crossing.spelled())
3781                        .ok_or_else(|| refuse(at, "no JNI representation for this sum"))?;
3782                    Ok(JFrag::new(at, conv.converter_impl().clone()))
3783                }
3784            },
3785            Direction::Deconstruct => {
3786                let legacy = self.selected_out(at, arms)?;
3787                let out_wires = legacy.out_wires.clone();
3788                let mut frag = self
3789                    .planned_choice(at, arms, None, out_wires)
3790                    .unwrap_or(legacy);
3791                chosen(&mut frag);
3792                Ok(frag)
3793            }
3794        }
3795    }

4686        // `jint` discriminant, so the slot reads `.value` — without it the wire
4687        // would carry a `Priority?` where it wants an `Int`.
4688        let enum_value = self.decls.is_kotlin_enum_reading(&part.ty);
4689        let read = if enum_value {
4690            format!("{prop}?.value")
4691        } else {
4692            prop.clone()
4693        };
4694        let zero = if enum_value {
4695            option_enum_niche(self.decls, &part.ty, Direction::Construct)
4696                .or_else(|| prim.map(|p| p.kotlin_zero().to_string()))
4697        } else {
4698            prim.map(|p| p.kotlin_zero().to_string())
4699        };
4700        let mut kt_ty = crate::jni::emit::wire_kotlin_type(&frag.conv);
4701        if prim.is_none() && !kt_ty.ends_with('?') {
4702            kt_ty.push('?');
4703        }
4704        Some(Wire {
4705            ty: frag.conv.destination.clone(),
4706            kt_ty,
4707            path: prop,
4708            access: Access::Slot {
4709                walk: Vec::new(),
4710                // Named by `choice`, the only hook told which alternative this
4711                // payload belongs to.
4712                class: String::new(),
4713                read,
4714                zero,
4715            },
4716            entry: Some(frag.conv.clone()),
4717            stages: frag
4718                .chain
4719                .steps()
4720                .iter()
4721                .map(|step| step.operation().clone())
4722                .collect(),
4723            handle_target: None,
4724            handle_nullable: false,
4725            absent: None,
4726            field: None,
4727            whole_gate: false,
4728        })
4729    }

4732    /// declined, since a sum crosses whole or not at all.

4735            return None;
4736        };

4740

4742            .iter()
4743            .map(|(alt, _)| {
4744                format!(
4745                    "{iface}.{}",
4746                    self.decls.sum_variant_class_name(sum_cfg, &alt.name)

4748            })
4749            .collect();

4751            // A tag converts nothing: it is read on the Rust side.
4752            stages: Vec::new(),
4753            ty: syn::parse_quote!(jni::sys::jint),
4754            kt_ty: "Int".to_string(),
4755            path: "_tag".to_string(),
4756            access: Access::Select {
4757                walk: Vec::new(),
4758                arms: arms
4759                    .iter()

4771            field: None,
4772            whole_gate: false,
4773        }];

4775            let short = class.rsplit('.').next().unwrap_or(class);
4776            for w in frag.wires.as_ref()? {

4778                    return None;
4779                };

4795    /// The `(present, value)` pair a nullable primitive or enum crosses as, or
4796    /// `None` if this optional boxes instead.
4797    ///
4798    /// The conditions are the ones the walk applies: the inner is not a borrow,
4799    /// its wire is a JNI primitive, and it has nothing that would already carry
4800    /// the absence — no carved niche, no projection, no Rust-side stages.
4801    fn decoupled_optional(
4802        &self,
4803        at: At<'_>,
4804        inner: &JFrag,
4805        outer: Option<&ConverterImpl<KotlinMeta>>,
4806    ) -> Option<Vec<Wire>> {
4807        let inner_reading = at.crossing.value().optional_inner()?;
4808        if inner_reading.borrow_target().is_some() {
4809            return None;
4810        }
4811        let c = &inner.conv;
4812        let prim = crate::jni::JniPrim::from_wire(&c.destination)?;
4813        if c.niches.clone().carve().is_some() || !inner.chain.steps().is_empty() {
4814            return None;
4815        }
4816        // An unsigned representation is the one projection that still takes the
4817        // pair. Its Kotlin property is a `ULong?`, which boxes, and its wire is
4818        // the `Long` under it — so the gate is worth the same here as it is over
4819        // a signed primitive. Only where the optional has no primitive wire of
4820        // its own: a bounded representation whose range leaves a niche already
4821        // crosses as one value and keeps doing so.
4822        let unsigned = match c.metadata.projection.as_ref().map(|p| &p.kind) {
4823            None => false,
4824            Some(ProjectionKind::Unsigned64)
4825                if outer.is_some_and(|outer| {
4826                    crate::jni::JniPrim::from_wire(&outer.destination).is_none()
4827                }) =>
4828            {
4829                true
4830            }
4831            Some(_) => return None,
4832        };
4833        // A Kotlin enum's slot is its `value`, reached through the same gate —
4834        // what this pair was decoupled from is optional, so that read is a safe
4835        // call however the wire above it is composed.
4836        let step = if unsigned {
4837            // The `ULong` property's own representation, which is the wire.
4838            Some("toLong()")
4839        } else if self.decls.is_kotlin_enum_reading(inner_reading) {
4840            Some("value")
4841        } else {
4842            None
4843        };
4844        let value_access = Access::Read {
4845            walk: step
4846                .map(|field| {
4847                    vec![Nav {
4848                        field: field.to_string(),
4849                        gated: true,
4850                    }]
4851                })
4852                .unwrap_or_default(),
4853            tail: format!(" ?: {}", prim.kotlin_zero()),
4854        };
4855        Some(vec![
4856            Wire {
4857                ty: syn::parse_quote!(jni::sys::jboolean),
4858                kt_ty: "Boolean".to_string(),
4859                path: "present".to_string(),
4860                access: Access::read(" != null"),
4861                entry: None,
4862                stages: Vec::new(),
4863                handle_target: None,
4864                handle_nullable: false,
4865                absent: None,
4866                field: None,
4867                whole_gate: false,
4868            },
4869            Wire {
4870                ty: c.destination.clone(),
4871                kt_ty: crate::jni::emit::wire_kotlin_type(c),
4872                path: "value".to_string(),
4873                access: value_access,
4874                entry: Some(c.clone()),
4875                // Refused above unless the inner fragment's chain is empty, so
4876                // this value crosses through its converter alone.
4877                stages: Vec::new(),
4878                handle_target: None,
4879                handle_nullable: false,
4880                absent: None,
4881                field: None,
4882                whole_gate: false,
4883            },
4884        ])
4885    }

4889    /// The values a `data_class` hands out: its fields, and a field that is
4890    /// itself one contributes its own under the parent's name and chain.
4891    ///
4892    /// Model and declaration only, like [`Self::sum_out_wires`] — so the same
4893    /// answer serves the leaf synthesis that runs before `resolve` and the recipe
4894    /// that composes after it.
4895    ///
4896    /// `None` is this adapter declining, and it declines for the **whole**
4897    /// value rather than per field: one field it cannot state sends the whole
4898    /// object down the whole-value `fromParts` path, because a recipe that
4899    /// decomposed the rest of it would describe a shape nothing emits.
4900    ///
4901    /// What is left to decline is **structural**, not a missing transform. A
4902    /// handle, an `enum_class`, a sum, an optional nested class and a selector
4903    /// inside a gated group are all ordinary leaves here (#602): a fixed number
4904    /// of values, laid side by side, each carrying its own conversion. The
4905    /// declines are the shapes that have no such number:
4906    ///
4907    /// * a field the model does not hold as a named `data_class` field —
4908    ///   a positional field, or a type that is not a path;
4909    /// * a nested `data_class` **repeated** under a `Vec`, whose leaves would
4910    ///   be as many as the value happens to hold. That is a sequence rather
4911    ///   than a decomposition — and note that only a `Vec` DIRECTLY under a
4912    ///   declared `data_class` reaches that arm: `Vec<Child>` keys as a
4913    ///   sequence rather than a `DataStruct`, so it stays one leaf whose own
4914    ///   converter is the element folder (#217);

4975    fn struct_out_wires_at(
4976        &self,
4977        registry: &impl Conversions,
4978        ident: &syn::Ident,
4979        path: &[syn::Ident],
4980        name_prefix: &str,
4981        depth: usize,
4982    ) -> Option<Vec<OutWire>> {
4983        if depth > 16 {
4984            return None;
4985        }
4986        let prebindgen_registry::flat::Type::Struct(st) = registry.flat().declared_type(ident)?
4987        else {
4988            return None;
4989        };
4990        let mut wires = Vec::new();
4991        for field in &st.fields {
4992            // Named by construction — a tuple struct is an `Extern`, not a
4993            // `Struct` — so a positional field means the model and this walk
4994            // disagree, and declining is the safe answer.
4995            let fname = field.name.as_ref()?;
4996            let name =
4997                crate::jni::mangle_kotlin_ident(&crate::jni::kt_snake_to_camel(&fname.to_string()));
4998            let name = match name_prefix.is_empty() {
4999                true => name,
5000                false => format!("{name_prefix}__{name}"),
5001            };
5002            let mut field_path = path.to_vec();
5003            field_path.push(fname.clone());
5004
5005            // The layer questions off the field's own reading — `Optional` to
5006            // look through, `Vec` to decline — never a last path segment.
5007            let probe = field.ty.optional_inner().unwrap_or(&field.ty);
5008            // A `Vec` of tag-gated groups has variable arity, and a
5009            // decomposition is fixed-layout by construction: there is no
5010            // number of slots to lay a sum's alternatives in per element.
5011            // Loud rather than a silent decline, because the whole-value path
5012            // cannot carry it either — a sum has no converter of its own to
5013            // fall through to, so declining would surface as a missing
5014            // converter for `Vec<T>` far from the field that caused it.
5015            if let Some(elem) = probe.sequence_elem() {
5016                if matches!(
5017                    self.type_kind(registry, &elem.key()),
5018                    crate::jni::classify::TypeKind::Sum
5019                ) {
5020                    panic!(
5021                        "fromParts bridge: `Vec<{elem}>` sealed-class field \
5022                         (`{}.{fname}`) is not supported (variable arity)",
5023                        st.name,
5024                    );
5025                }
5026            }
5027            match self.type_kind(registry, &probe.key()) {
5028                // A handle or an enum_class field is an ordinary leaf: its own
5029                // output conversion carries it — a `jlong` for the handle the
5030                // receiver adopts, a `jint` discriminant for the enum — and
5031                // the whole-object encode has always emitted exactly that.
5032                crate::jni::classify::TypeKind::Handle | crate::jni::classify::TypeKind::Enum => {}
5033                // A `sealed_class` field contributes what a sealed_class
5034                // crossing does — a tag naming the live alternative, then one
5035                // group of leaves per alternative — reached through this
5036                // field. Only one group is live per value, which is what the
5037                // registry's segment gate emits as a single `match`.
5038                crate::jni::classify::TypeKind::Sum => {
5039                    let sum_ident = match probe.unwrapped().kind() {
5040                        TypeKind::Named { id, .. } => id.ident()?,
5041                        _ => return None,
5042                    };
5043                    let optional = field.ty.optional_inner().is_some();
5044                    let mut reach: Vec<prebindgen_registry::unfold::PathStep> =
5045                        field_path.iter().map(field_step).collect();
5046                    if optional {
5047                        // An OPTIONAL sum is two selectors, one owning the
5048                        // other: presence says whether there is an alternative
5049                        // at all, and the tag names which. The flag's reach
5050                        // ends AT the optional step it tests; the sum's leaves
5051                        // reach through it, and their arm paths gain the one
5052                        // group presence gates.
5053                        let mut flag_reach = reach.clone();
5054                        flag_reach[path.len()] =
5055                            prebindgen_registry::unfold::PathStep::field(fname.clone(), true);
5056                        wires.push(OutWire {
5057                            name: format!("{name}__{}", crate::jni::emit::PRESENT_LEAF),
5058                            out_ty: field.ty.clone(),
5059                            groups: Vec::new(),
5060                            through: Vec::new(),
5061                            from: OutFrom::Present,
5062                            reach: flag_reach,
5063                            nullable: false,
5064                            identity: false,
5065                            abi: None,
5066                        });
5067                        reach[path.len()] =
5068                            prebindgen_registry::unfold::PathStep::field(fname.clone(), true);
5069                    }
5070                    for wire in self.sum_out_wires(registry, &sum_ident, probe)? {
5071                        let groups = match optional {
5072                            true => std::iter::once(crate::jni::emit::PRESENT_ARM)
5073                                .chain(wire.groups.iter().copied())
5074                                .collect(),
5075                            false => wire.groups,
5076                        };
5077                        wires.push(OutWire {
5078                            name: format!("{name}__{}", wire.name),
5079                            reach: reach.clone(),
5080                            groups,
5081                            ..wire
5082                        });
5083                    }
5084                    continue;
5085                }
5086                // A nested `data_class` inlines: its own leaves are reached
5087                // through this field. An OPTIONAL one contributes a presence
5088                // flag ahead of them, and its group carries wire defaults when
5089                // the value is absent — the same shape a sum field's tag gives
5090                // its groups, with presence in place of an alternative.
5091                //
5092                // A `Vec` of them is still refused: a repeated value has no
5093                // fixed number of leaves to lay side by side, which is a
5094                // sequence rather than a decomposition.
5095                crate::jni::classify::TypeKind::DataStruct {
5096                    st: _,
5097                    cfg: Some(child_cfg),
5098                } => {
5099                    if matches!(field.ty.kind(), TypeKind::Vec(_)) {
5100                        return None;
5101                    }
5102                    let child = match probe.unwrapped().kind() {
5103                        TypeKind::Named { id, .. } => id.ident()?,
5104                        _ => return None,
5105                    };
5106                    let optional = field.ty.optional_inner().is_some();
5107                    if optional {
5108                        // The flag's own reach ends in the OPTIONAL step it
5109                        // tests: presence is the gate, and the walk emits a
5110                        // gate exactly where a segment's selector reaches
5111                        // through one. The group's leaves then reach on from
5112                        // the value that gate unwrapped.
5113                        let mut reach: Vec<prebindgen_registry::unfold::PathStep> =
5114                            field_path.iter().map(field_step).collect();
5115                        reach[path.len()] =
5116                            prebindgen_registry::unfold::PathStep::field(fname.clone(), true);
5117                        wires.push(OutWire {
5118                            name: format!("{name}__{}", crate::jni::emit::PRESENT_LEAF),
5119                            out_ty: field.ty.clone(),
5120                            groups: Vec::new(),
5121                            through: Vec::new(),
5122                            from: OutFrom::Present,
5123                            reach,
5124                            nullable: false,
5125                            identity: false,
5126                            abi: None,
5127                        });
5128                    }
5129                    let inlined =
5130                        self.struct_out_wires_at(registry, &child, &field_path, &name, depth + 1)?;
5131                    // Which class this boundary reassembles through, recorded
5132                    // where the flattening happens. `data_class!` named it, so
5133                    // this stays a declaration read like the rest of the walk.
5134                    let child_fqn = child_cfg.name_spec.as_ref().map(|spec| self.fqn_of(spec))?;
5135                    wires.extend(inlined.into_iter().map(|wire| {
5136                        let mut reach = wire.reach;
5137                        if optional {
5138                            // The step reaching THIS field is optional for the
5139                            // child's leaves: they are inside the value the
5140                            // presence flag tests, so the walk gates them on
5141                            // it. The flag's own reach stops at the `Option`,
5142                            // which is the value it tests rather than one
5143                            // inside it.
5144                            reach[path.len()] =
5145                                prebindgen_registry::unfold::PathStep::field(fname.clone(), true);
5146                        }
5147                        OutWire {
5148                            reach,
5149                            // Under a presence flag the child's leaves are the
5150                            // one group that flag gates — including a selector
5151                            // of the child's own, which keeps its own arms one
5152                            // level further in.
5153                            groups: match optional {
5154                                true => std::iter::once(crate::jni::emit::PRESENT_ARM)
5155                                    .chain(wire.groups.iter().copied())
5156                                    .collect(),
5157                                false => wire.groups.clone(),
5158                            },
5159                            through: std::iter::once(child_fqn.clone())
5160                                .chain(wire.through.iter().cloned())
5161                                .collect(),
5162                            ..wire
5163                        }
5164                    }));
5165                    continue;
5166                }
5167                _ => {}
5168            }
5169
5170            // A simple value: the field's own output conversion encodes it, and
5171            // the foreign `fromParts` forwards it verbatim. Nullability rides
5172            // that conversion — `Option<Box<String>>` is a `String?` — so the
5173            // value itself is not path-nullable.
5174            wires.push(OutWire {
5175                name,
5176                out_ty: field.ty.clone(),
5177                groups: Vec::new(),
5178                through: Vec::new(),
5179                from: OutFrom::Place,
5180                reach: field_path.iter().map(field_step).collect(),
5181                nullable: false,
5182                identity: false,
5183                abi: None,
5184            });
5185        }
5186        Some(wires)
5187    }

5262/// The transparent wrappers over a crossing's value, or `None` when a composed
5263/// chain cannot go through one of them.
5264///
5265/// A chain reaches the canonical value by peeling every wrapper and puts them
5266/// back afterwards, so it needs the operation for the direction it is going:
5267/// `Box::new` to build, `*b` to read. A wrapper missing that operation — `Cow`,
5268/// which has no read — is not composable, and neither is a wrapper this adapter
5269/// has no entry for at all.
5270///
5271/// Deconstructing additionally needs the crossing to **own** what it takes
5272/// apart. Reading through a shared borrow would move the value out of the
5273/// wrapper, which is what a `&Box<T>` would do.
5274///
5275/// Refusing here is what keeps the legacy path in control. `JSource::build` and
5276/// `JSource::read` expect the rejection to have happened while planning: past
5277/// this point an unsupported wrapper is a panic during `write_rust`, or
5278/// generated Rust that does not compile in the consumer's crate.
5279///
5280/// One function because it is one rule. Stated per planner, the choice copy
5281/// omitted it, while the sequence and Optional copies omitted the ownership
5282/// half.
5283fn composable_wrappers(
5284    crossing: &prebindgen_registry::recipe::Crossing,
5285) -> Option<Vec<&'static str>> {
5286    let wrappers = crossing.value().erased_wrappers();
5287    let usable = |wrapper: &&'static str| -> bool {
5288        let Some(ops) = crate::jni::trait_impl::wrapper_ops(wrapper) else {
5289            return false;
5290        };
5291        match crossing.direction() {
5292            Direction::Construct => ops.build.is_some(),
5293            Direction::Deconstruct => crossing.mode() == Mode::Owned && ops.read.is_some(),
5294        }
5295    };
5296    wrappers.iter().all(usable).then_some(wrappers)
5297}

5337/// One wire, read through a gate above it that may find nothing.
5338///
5339/// Every step below the gate becomes a safe call — not only the first: after
5340/// one `?.` the chain is on a nullable value and stays there. What each form
5341/// then needs is its own. An ordinary read carries the literal it stated for
5342/// exactly this case; a sum's `when` gains the `null` arm; a sum's `as?` slot
5343/// needs neither, a cast over a null subject being already null.
5344///
5345/// A wire whose value is a JVM object rides that `null` instead of a literal,
5346/// and its Kotlin type says so. Read **after** the literal rather than before,
5347/// because a `String` slot does both: it falls back to `""` when its own field
5348/// is present under a closed ancestor, and it is typed `String?`.
5349fn gated(w: &Wire) -> Wire {
5350    let mut gated = w.clone();
5351    for nav in gated
5352        .access
5353        .walk_mut()
5354        .iter_mut()
5355        .chain(gated.handle_target.iter_mut().flatten())
5356    {
5357        nav.gated = true;
5358    }
5359    match &mut gated.access {
5360        Access::Read { tail, .. } => {
5361            if let Some(absent) = gated.absent.take() {
5362                if !tail.contains(" ?: ") {
5363                    *tail = format!("{tail} ?: {absent}");
5364                }
5365            }
5366        }
5367        Access::Select { nullable, .. } => *nullable = true,
5368        Access::Slot { .. } => {}
5369    }
5370    if gated.entry.is_some()
5371        && matches!(gated.access, Access::Read { .. })
5372        && crate::jni::emit::is_jobject_shaped_wire(&gated.ty)
5373        && !gated.kt_ty.ends_with('?')
5374    {
5375        gated.kt_ty.push('?');
5376    }
5377    // Anything under the gate can be absent, however the field itself was
5378    // spelled.
5379    gated.handle_nullable = gated.handle_target.is_some() || gated.handle_nullable;
5380    gated
5381}
