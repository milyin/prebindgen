1//! Call-scoped resource conflicts for inputs constructed as they are decoded.
2use crate::Result;
3use std::hash::Hasher;
4
5/// No bookkeeping is needed for shared-only calls or a single fixed handle.
6#[derive(Default)]
7pub struct SharedInputs;
8impl SharedInputs {
9    /// Shared-only calls can construct rows directly without retaining a batch.
10    #[inline(always)]
11    pub fn needs_group(&self) -> bool {
12        false
13    }
14    /// Accept an access that cannot conflict within this generated call.
15    #[inline(always)]
16    pub fn access(&mut self, _: usize, _: bool) -> Result<()> {
17        Ok(())
18    }
19    /// Ignore a collection capacity hint.
20    #[inline(always)]
21    pub fn reserve(&mut self, _: usize) {}
22}
23/// Heap-free history for fixed accesses; larger sets use a stack hash table.
24pub struct FixedInputs<const N: usize> {
25    values: [(usize, bool); N],
26    len: usize,
27}
28impl<const N: usize> Default for FixedInputs<N> {
29    fn default() -> Self {
30        Self {
31            values: [(0, false); N],
32            len: 0,
33        }
34    }
35}
36impl<const N: usize> FixedInputs<N> {
37    /// Check this access before creating an owner or reference.
38    #[inline]
39    pub fn access(&mut self, address: usize, exclusive: bool) -> Result<()> {
40        if N <= 8 {
41            if self.values[..self.len]
42                .iter()
43                .any(|&(p, e)| p == address && (e || exclusive))
44            {
45                return Err("conflicting input aliases".into());
46            }
47            if self.len == N {
48                return Err("input access history capacity exceeded".into());
49            }
50            self.values[self.len] = (address, exclusive);
51            self.len += 1;
52        } else {
53            let mut hasher = AddressHasher::default();
54            hasher.write_usize(address);
55            let mut slot = hasher.finish() as usize % N;
56            for _ in 0..N {
57                let (previous, used_exclusively) = self.values[slot];
58                if previous == 0 {
59                    self.values[slot] = (address, exclusive);
60                    return Ok(());
61                }
62                if previous == address {
63                    if used_exclusively || exclusive {
64                        return Err("conflicting input aliases".into());
65                    }
66                    return Ok(());
67                }
68                slot = (slot + 1) % N;
69            }
70            return Err("input access history capacity exceeded".into());
71        }
72        Ok(())
73    }
74    /// Fixed input histories need no collection allocation.
75    pub fn reserve(&mut self, _: usize) {}
76}
77/// Hash native addresses without hashing arbitrary foreign data.
78#[derive(Default)]
79struct AddressHasher(u64);
80impl Hasher for AddressHasher {
81    fn finish(&self) -> u64 {
82        self.0
83    }
84    fn write(&mut self, bytes: &[u8]) {
85        let mut word = 0u64;
86        for (i, b) in bytes.iter().take(8).enumerate() {
87            word |= u64::from(*b) << (i * 8);
88        }
89        self.write_u64(word);
90    }
91    fn write_usize(&mut self, value: usize) {
92        self.write_u64(value as u64);
93    }
94    fn write_u64(&mut self, mut value: u64) {
95        value = (value ^ (value >> 30)).wrapping_mul(0xbf58476d1ce4e5b9);
96        value = (value ^ (value >> 27)).wrapping_mul(0x94d049bb133111eb);
97        self.0 = value ^ (value >> 31);
98    }
99}
100/// A shared-only group needs neither storage nor comparisons.
101pub struct SharedGroup;
102impl SharedGroup {
103    /// Ignore an access when the entire call uses shared borrows.
104    #[inline(always)]
105    pub fn push(&mut self, _: (usize, bool)) {}
106}
107impl SharedInputs {
108    /// Begin a collection whose accesses cannot conflict.
109    pub fn begin_group(&self, _: usize) -> SharedGroup {
110        SharedGroup
111    }
112    /// Complete a shared-only collection check.
113    pub fn accept_group(&mut self, _: SharedGroup) -> Result<()> {
114        Ok(())
115    }
116}
117/// Lazily allocated addresses from a collection that has not been acquired yet.
118pub struct InputGroup {
119    values: Vec<(usize, bool)>,
120    capacity: usize,
121}
122impl InputGroup {
123    /// Register an active resource without allocating for absent optional values.
124    pub fn push(&mut self, value: (usize, bool)) {
125        if self.values.capacity() == 0 {
126            self.values.reserve_exact(self.capacity);
127        }
128        self.values.push(value);
129    }
130}
131/// Sorted address history for dynamic collections. A collection checks all of
132/// its addresses together, before creating any owners or Rust references.
133#[derive(Default)]
134pub struct DynamicInputs {
135    values: Vec<(usize, bool)>,
136}
137impl DynamicInputs {
138    /// Dynamic exclusive accesses need a checked batch before construction.
139    #[inline(always)]
140    pub fn needs_group(&self) -> bool {
141        true
142    }
143    /// No allocation is needed until an active access or group is encountered.
144    pub fn reserve(&mut self, _: usize) {}
145    /// Collect a batch without allocating for absent optional resources.
146    pub fn begin_group(&self, capacity: usize) -> InputGroup {
147        InputGroup {
148            values: Vec::new(),
149            capacity,
150        }
151    }
152    /// Sort the new batch together with previous accesses, then reject every
153    /// shared/exclusive or exclusive/exclusive duplicate before using the batch.
154    pub fn accept_group(&mut self, group: InputGroup) -> Result<()> {
155        let mut group = group.values;
156        if self.values.is_empty() {
157            ::core::mem::swap(&mut self.values, &mut group);
158        } else {
159            self.values.append(&mut group);
160        }
161        self.values.sort_unstable_by_key(|entry| entry.0);
162        for pair in self.values.windows(2) {
163            if pair[0].0 == pair[1].0 && (pair[0].1 || pair[1].1) {
164                return Err("conflicting input aliases".into());
165            }
166        }
167        self.values.dedup_by_key(|entry| entry.0);
168        Ok(())
169    }
170    /// Check an individual argument against the history of previous arguments.
171    pub fn access(&mut self, address: usize, exclusive: bool) -> Result<()> {
172        match self.values.binary_search_by_key(&address, |entry| entry.0) {
173            Ok(i) if self.values[i].1 || exclusive => Err("conflicting input aliases".into()),
174            Ok(_) => Ok(()),
175            Err(i) => {
176                self.values.insert(i, (address, exclusive));
177                Ok(())
178            }
179        }
180    }
181}

183#[cfg(test)]
184mod tests {
185    use super::*;
186    #[test]
187    fn fixed_and_dynamic_histories_check_before_every_use() {
188        for exclusive in [false, true] {
189            let mut fixed = FixedInputs::<4>::default();
190            let mut dynamic = DynamicInputs::default();
191            dynamic.reserve(4);
192            for (address, access) in [(8, false), (16, exclusive), (8, false)] {
193                fixed.access(address, access).unwrap();
194                dynamic.access(address, access).unwrap();
195            }
196            assert!(fixed.access(8, true).is_err());
197            assert!(dynamic.access(8, true).is_err());
198            assert!(fixed.access(16, true).is_err());
199            assert!(dynamic.access(16, true).is_err());
200        }
201    }
202    #[test]
203    fn large_fixed_history_accepts_shared_aliases_and_detects_moves() {
204        let mut history = FixedInputs::<32>::default();
205        for i in 1..=24 {
206            history.access(i * 8, false).unwrap();
207        }
208        for _ in 0..100 {
209            history.access(8, false).unwrap();
210        }
211        assert!(history.access(8, true).is_err());
212        history.access(800, true).unwrap();
213        assert!(history.access(800, false).is_err());
214    }
215    #[test]
216    fn groups_check_internal_conflicts_and_previous_arguments() {
217        let mut history = DynamicInputs::default();
218        history.access(8, false).unwrap();
219        let mut group = history.begin_group(4);
220        group.push((16, true));
221        group.push((24, false));
222        group.push((24, false));
223        history.accept_group(group).unwrap();
224        assert!(history.access(16, false).is_err());
225        let mut next = history.begin_group(2);
226        next.push((32, false));
227        next.push((8, true));
228        assert!(history.accept_group(next).is_err());
229        let mut history = DynamicInputs::default();
230        let mut duplicates = history.begin_group(2);
231        duplicates.push((48, true));
232        duplicates.push((48, true));
233        assert!(history.accept_group(duplicates).is_err());
234    }
235    #[test]
236    fn inactive_dynamic_input_does_not_allocate() {
237        let mut history = DynamicInputs::default();
238        history.reserve(1000);
239        assert_eq!(history.values.capacity(), 0);
240        history.access(8, true).unwrap();
241        assert!(history.values.capacity() > 0);
242    }
