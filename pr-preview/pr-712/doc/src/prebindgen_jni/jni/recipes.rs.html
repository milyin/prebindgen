1//! JniGen's Kotlin class declarations, lowered into the registry's recipe
2//! vocabulary.
3//!
4//! A build script writes `ptr_class!`, `data_class!`, `enum_class!` or
5//! `sealed_class!`, each naming the Kotlin type one Rust type becomes, and
6//! `convert!` for a conversion the binding supplies itself. This turns those
7//! into recipes in a [`Recipes`] table: the shared statement of **which parts** a
8//! value gets across in, with nothing about the JNI wire in it.
9//!
10//! One recipe per declared type and direction, and every crossing nobody declared
11//! takes
12//! the recipe the registry derives from its kind. The chain of guesses this
13//! replaced — try a terminal, then a `Result`, then an optional, then a run,
14//! then a borrow, then a transparent bridge — is a lookup for the declared
15//! half, and the derived half is the registry's.
16

29/// Every type one captured item names, so a reading nested inside it can be
30/// found without the caller knowing which kind of item it came from.
31fn element_types(element: &prebindgen_registry::Element) -> Vec<&TypeRef> {
32    use prebindgen_registry::flat::Type;
33    match element {
34        prebindgen_registry::Element::Function(f) => f
35            .params
36            .iter()
37            .map(|p| &p.ty)
38            .chain(std::iter::once(&f.ret))
39            .collect(),
40        prebindgen_registry::Element::Type(Type::Struct(s)) => {
41            s.fields.iter().map(|f| &f.ty).collect()
42        }
43        prebindgen_registry::Element::Type(Type::Variant(v)) => v
44            .alternatives
45            .iter()
46            .flat_map(|a| a.fields.iter().map(|f| &f.ty))
47            .collect(),
48        prebindgen_registry::Element::Constant(c) => vec![&c.ty],
49        _ => Vec::new(),
50    }
51}

53/// One arm per alternative of a declared sum, each assembled from its own
54/// payload fields.
55///
56/// Empty for anything the model does not hold as a data-carrying enum, which
57/// includes a `sealed_class!` declared over a type the scan dropped — the scan
58/// reports that, and a recipe over no arms would report it a second time.
59fn alternatives(model: &Flat, ty: &TypeRef) -> Vec<Arm<Construct>> {
60    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
61        return Vec::new();
62    };
63    let Some(prebindgen_registry::flat::Type::Variant(v)) =
64        id.ident().and_then(|ident| model.declared_type(&ident))
65    else {
66        return Vec::new();
67    };
68    v.alternatives
69        .iter()
70        .map(|alt| Arm {
71            alternative: Some(alt.index),
72            op: Construct::Fields,
73        })
74        .collect()
75}

77/// One arm per alternative of a declared sum, each taken apart into its own
78/// payload fields.
79///
80/// The deconstructing twin of [`alternatives`]. Separate because the two directions
81/// name different operations — building an alternative is `Construct::Fields`,
82/// reading one is `Deconstruct::Fields` over the reaches that get there — and
83/// a shared helper would have to be generic over an operation neither side
84/// chooses.
85fn out_alternatives(model: &Flat, ty: &TypeRef) -> Vec<Arm<Deconstruct>> {
86    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
87        return Vec::new();
88    };
89    let Some(prebindgen_registry::flat::Type::Variant(v)) =
90        id.ident().and_then(|ident| model.declared_type(&ident))
91    else {
92        return Vec::new();
93    };
94    v.alternatives
95        .iter()
96        .map(|alt| Arm {
97            alternative: Some(alt.index),
98            op: Deconstruct::Fields((0..alt.fields.len()).map(Reach::Field).collect()),
99        })
100        .collect()
101}

104    /// How many values this type hands out, or 0 if it hands out none.
105    ///
106    /// Model and declaration only, like the compositions it asks — so a
107    /// declaration-time caller may ask it, which is what binding a return site
108    /// needs.
109    fn out_wire_count(
110        &self,
111        registry: &(impl prebindgen_registry::Conversions + ?Sized),
112        ty: &TypeRef,
113    ) -> usize {
114        let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
115            return 0;
116        };
117        let Some(ident) = id.ident() else { return 0 };
118        match registry.flat().declared_type(&ident) {
119            Some(prebindgen_registry::flat::Type::Variant(_)) => self
120                .sum_out_wires(registry.flat(), &ident, ty)
121                .map_or(0, |w| w.len()),
122            Some(prebindgen_registry::flat::Type::Struct(_)) => {
123                self.struct_out_wires(registry, ty).map_or(0, |w| w.len())
124            }
125            _ => 0,
126        }
127    }

129    /// The accessor a type's value form names, and one reach per field of the
130    /// struct it returns.
131    ///
132    /// `None` unless the declaration is exactly one `.fields(fields!(f))` whose
133    /// every field is a plain leaf. A record that states its own decomposition
134    /// — a per-field `expand_return!` override, an inlined nested class, a sum
135    /// laid out as a selector and its groups — is a shape this recipe does not
136    /// state yet, and stating half of one would describe a boundary nothing
137    /// emits.
138    fn value_form_of(
139        &self,
140        model: &Flat,
141        registry: &(impl prebindgen_registry::Conversions + ?Sized),
142        ty: &TypeRef,
143    ) -> Option<(syn::Ident, Vec<Reach>)> {
144        let decl = self
145            .return_expand_decls
146            .iter()
147            .find(|d| *d.key() == ty.stripped_key())?;
148        self.value_form_in(model, registry, ty, decl)
149    }

151    /// [`Self::value_form_of`] against ONE declaration — the type's own, or the
152    /// one a function wrote for its return. A per-function
153    /// `.expand_return(expand_return!(T).fields(..))` states its own accessor
154    /// and its own field list, which may differ from the type's or exist
155    /// without one, so the lowering reads the declaration it was handed rather
156    /// than looking the type up again.
157    fn value_form_in(
158        &self,
159        model: &Flat,
160        registry: &(impl prebindgen_registry::Conversions + ?Sized),
161        ty: &TypeRef,
162        decl: &crate::jni::ExpandReturnDecl,
163    ) -> Option<(syn::Ident, Vec<Reach>)> {
164        let (func, reaches, _) = self.value_form_parts(model, registry, ty, decl)?;
165        Some((func, reaches))
166    }

168    /// Each field of a value form whose declaration OVERRODE how it comes
169    /// apart: the field's name, the type it reaches, and the reaches that
170    /// override states.
171    ///
172    /// `FieldDecon::Records` is a list written at this site, replacing the
173    /// field type's own decomposition for this occurrence. It says exactly what
174    /// a `Deconstruct::Fields` row says, so it becomes one — under a name of
175    /// its own, since the type's own row is a different statement.
176    fn field_overrides(
177        &self,
178        model: &Flat,
179        registry: &(impl prebindgen_registry::Conversions + ?Sized),
180        ty: &TypeRef,
181        decl: &crate::jni::ExpandReturnDecl,
182    ) -> Vec<(String, TypeRef, Vec<Reach>)> {
183        use crate::unfold::{DeconRecord, FieldDecon};
184
185        let mut out = Vec::new();
186        let [crate::jni::LocalField::Fields(fields)] = decl.field_list() else {
187            return out;
188        };
189        let Some(st) = model.function(fields.func()).map(|f| f.ret.clone()) else {
190            return out;
191        };
192        let st = st.borrow_target().unwrap_or(&st);
193        let Some(ident) = st.stripped_key().ident() else {
194            return out;
195        };
196        let Some(prebindgen_registry::flat::Type::Struct(st)) = model.declared_type(&ident) else {
197            return out;
198        };
199        let _ = ty;
200        for record in self.lower_value_form(registry, decl.key(), fields) {
201            let FieldDecon::Records(records) = &record.decon else {
202                continue;
203            };
204            let Some(member) = record.members.first() else {
205                continue;
206            };
207            let Some(field) = st.fields.iter().find(|f| f.name.as_ref() == Some(member)) else {
208                continue;
209            };
210            let reaches: Option<Vec<Reach>> = records
211                .iter()
212                .map(|r| match r {
213                    DeconRecord::Acc { func, .. } => Some(Reach::Accessor(func.clone())),
214                    DeconRecord::LocalAcc { path, .. } => {
215                        Some(Reach::Accessor(path.segments.last()?.ident.clone()))
216                    }
217                    DeconRecord::Identity => Some(Reach::Identity),
218                    // A value form inside an override is a row of a shape this
219                    // one does not hold; it waits, rather than being stated
220                    // wrongly.
221                    _ => None,
222                })
223                .collect();
224            let Some(reaches) = reaches else { continue };
225            out.push((member.to_string(), field.ty.clone(), reaches));
226        }
227        out
228    }

230    /// [`Self::value_form_in`], with the answer it drops: whether each part is
231    /// SPLICED, taken apart by a binding on it, or crosses as one leaf.
232    ///
233    /// One walk answers both, because they have to agree. A spliced part is one
234    /// position of the row however many records the declaration flattened it
235    /// into, so a row grouped differently from the bindings would index them
236    /// apart (#701 decision 3).
237    fn value_form_parts(
238        &self,
239        model: &Flat,
240        registry: &(impl prebindgen_registry::Conversions + ?Sized),
241        ty: &TypeRef,
242        decl: &crate::jni::ExpandReturnDecl,
243    ) -> Option<(syn::Ident, Vec<Reach>, Vec<bool>)> {
244        use crate::unfold::{FieldDecon, FieldRecord};
245        let [crate::jni::LocalField::Fields(fields)] = decl.field_list() else {
246            return None;
247        };
248        let records: Vec<FieldRecord> = self.lower_value_form(registry, decl.key(), fields);
249        let ret = model.function(fields.func())?.ret.clone();
250        let ret = ret.borrow_target().unwrap_or(&ret);
251        let prebindgen_registry::flat::TypeKind::Named { id, .. } = ret.unwrapped().kind() else {
252            return None;
253        };
254        let prebindgen_registry::flat::Type::Struct(st) =
255            id.ident().and_then(|i| model.declared_type(&i))?
256        else {
257            return None;
258        };
259        let mut reaches: Vec<Reach> = Vec::new();
260        let mut spliced: Vec<bool> = Vec::new();
261        // The field each part already reached, so records an inlined nested
262        // class flattened into — one per leaf of the child, sharing a first hop
263        // — collapse to the single part that child is.
264        let mut reached: Vec<usize> = Vec::new();
265        for record in &records {
266            // A field the DECLARATION took apart — a decomposed sum, or an
267            // override written at this site — is a plain reach here, and the
268            // taking-apart is a part binding on it (design decision 3). The row
269            // says which field; the binding says that it comes apart, and to
270            // which row.
271            // A record with several members is an INLINED nested class: each
272            // member is one hop of a field-access chain. `Reach::Path` states
273            // that; before it existed this row declined and fell back to a
274            // placeholder (#613 step 10).
275            let mut here = st;
276            let mut indices = Vec::with_capacity(record.members.len());
277            for member in &record.members {
278                let hop = here
279                    .fields
280                    .iter()
281                    .position(|f| f.name.as_ref() == Some(member))?;
282                indices.push(hop);
283                if indices.len() < record.members.len() {
284                    let next = here.fields[hop].ty.stripped_key().ident()?;
285                    let prebindgen_registry::flat::Type::Struct(next) =
286                        model.declared_type(&next)?
287                    else {
288                        return None;
289                    };
290                    here = next;
291                }
292            }
293            // A CHAIN is an inlined nested class. `Reach::Path` spelled the
294            // whole chain in the parent's row; design decision 3 says the same
295            // thing as the first hop plus a part binding, and the binding is
296            // what #701 keeps.
297            //
298            // Collapsible only where the CHILD names these parts the way this
299            // declaration does, because a binding delegates naming to the
300            // child's own row. A parent that renamed a nested leaf —
301            // `.name("nodeId")` over the child's `node` — says something no
302            // binding can, and keeps the chain it had.
303            if indices.len() > 1 {
304                let head = indices[0];
305                let child = st.fields[head].ty.stripped_key();
306                let declared = self.return_expand_decls.iter().any(|d| *d.key() == child);
307                let prefix = st.fields[head]
308                    .name
309                    .as_ref()
310                    .map(|n| {
311                        crate::jni::mangle_kotlin_ident(&crate::jni::kt_snake_to_camel(
312                            &n.to_string(),
313                        ))
314                    })
315                    .unwrap_or_default();
316                // The name the spliced child will produce for its part, under
317                // the prefix the parent part contributes — which is exactly
318                // what `UnfoldPolicy::nest` composes.
319                let renamed = records
320                    .iter()
321                    .filter(|r| r.members.first() == record.members.first())
322                    .enumerate()
323                    .any(|(at, r)| {
324                        let leaf = if declared {
325                            self.leaf_name_at(&child, at)
326                        } else {
327                            r.members
328                                .get(1)
329                                .map(|m| {
330                                    crate::jni::mangle_kotlin_ident(&crate::jni::kt_snake_to_camel(
331                                        &m.to_string(),
332                                    ))
333                                })
334                                .unwrap_or_default()
335                        };
336                        r.name != format!("{prefix}__{leaf}")
337                    });
338                if !renamed {
339                    if reached.contains(&head) {
340                        continue;
341                    }
342                    reached.push(head);
343                    reaches.push(Reach::Field(head));
344                    spliced.push(true);
345                    continue;
346                }
347            }
348            let [index] = indices.as_slice() else {
349                reaches.push(Reach::Path(indices));
350                spliced.push(false);
351                continue;
352            };
353            let index = *index;
354            // A field that reaches the type being taken apart. The leaf
355            // synthesis treats `Box<T>` as a spelling of its own and stops
356            // there; a recipe keys a crossing by the value that crosses, so
357            // `Box<ZSample>` inside `ZSample`'s own value form is that recipe
358            // reaching itself. Refused here rather than by `Recipes::build`,
359            // which would refuse the whole binding over a shape the leaf
360            // synthesis handles.
361            if st.fields[index].ty.stripped_key() == ty.stripped_key() {
362                return None;
363            }
364            if reached.contains(&index) {
365                continue;
366            }
367            reached.push(index);
368            // Spliced two ways. The declaration may take the field apart HERE —
369            // a decomposed sum — or say `Default`, which means "by the field
370            // type's own deconstructor if it has one". A field whose type
371            // merely COULD come apart is not spliced.
372            let (_, core) = st.fields[index].ty.layer_stack();
373            let core = core.borrow_target().unwrap_or(core).stripped_key();
374            reaches.push(Reach::Field(index));
375            spliced.push(
376                !matches!(record.decon, FieldDecon::Default)
377                    || self.return_expand_decls.iter().any(|d| *d.key() == core),
378            );
379        }
380        Some((fields.func().clone(), reaches, spliced))
381    }
382}

478impl Declarations {
479    /// Every recipe this binding's declarations state.
480    ///
481    /// A type declared but absent from the model is skipped rather than
482    /// refused: the scan already reports it, and reporting it twice in
483    /// different words helps nobody.
484    /// This type's `parts` decomposition, read off its `expand_return!`
485    /// declaration in the recipe table's own vocabulary.
486    ///
487    /// `None` when the declaration states a shape a `Deconstruct` cannot hold —
488    /// a value-form record mixed with others, since `ValueForm` is the whole
489    /// row while `LocalField::Fields` is one record among many. No declaration
490    /// in this workspace has that shape (measured: every `Fields` record stands
491    /// alone), so the caller's `Atomic` fallback is a guard rather than a path
492    /// anything takes.
493    fn parts_deconstruct(
494        &self,
495        model: &Flat,
496        registry: &(impl prebindgen_registry::Conversions + ?Sized),
497        ty: &TypeRef,
498    ) -> Option<Deconstruct> {
499        let decl = self
500            .return_expand_decls
501            .iter()
502            .find(|d| *d.key() == ty.stripped_key())?;
503        self.decon_of(model, registry, ty, decl)
504    }
505
506    /// [`Self::parts_deconstruct`] against ONE declaration, whichever states
507    /// it: the type's own, or the one a function wrote for its return.
508    fn decon_of(
509        &self,
510        model: &Flat,
511        registry: &(impl prebindgen_registry::Conversions + ?Sized),
512        ty: &TypeRef,
513        decl: &crate::jni::ExpandReturnDecl,
514    ) -> Option<Deconstruct> {
515        let fields = decl.field_list();
516        if matches!(fields, [crate::jni::LocalField::Fields(_)]) {
517            // `ValueForm` means "call this accessor, then read these parts off
518            // its result" — the parts are how the compiler reaches the returned
519            // struct's fields, so an empty list is a decomposition with nothing
520            // in it rather than one of everything (#638 review).
521            //
522            // `value_form_of` already computes exactly that mapping, and it is
523            // not a count: `lower_value_form` returns FLATTENED records, one of
524            // which can splice a child's fields, so `Report` yields six records
525            // over a five-field struct. It maps each record back to the struct
526            // field it names and declines the shapes a row cannot hold.
527            let (func, parts) = self.value_form_in(model, registry, ty, decl)?;
528            return Some(Deconstruct::ValueForm { func, parts });
529        }
530        let mut reaches = Vec::with_capacity(fields.len());
531        for field in fields {
532            reaches.push(match field {
533                crate::jni::LocalField::Named(func, _) => Reach::Accessor(func.clone()),
534                // The handle itself — the leaf `Reach::Identity` was added for.
535                crate::jni::LocalField::SelfField => Reach::Identity,
536                // A binding-local accessor resolves by its path's last segment:
537                // `local_functions()` registers it under exactly that ident.
538                crate::jni::LocalField::Local { path, .. } => {
539                    Reach::Accessor(path.segments.last()?.ident.clone())
540                }
541                crate::jni::LocalField::Fields(_) => return None,
542            });
543        }
544        // A row names a part by the accessor that reaches it, so one accessor
545        // used twice — `.field(fun!(f).name("a")).field(fun!(f).name("b"))` —
546        // is two parts the row cannot tell apart. The declaration's name
547        // overrides say which is which and `Reach` has nowhere to put them, so
548        // the row declines rather than stating two leaves of one name.
549        let mut seen: Vec<&syn::Ident> = Vec::with_capacity(reaches.len());
550        for reach in &reaches {
551            if let Reach::Accessor(func) = reach {
552                if seen.contains(&func) {
553                    return None;
554                }
555                seen.push(func);
556            }
557        }
558        Some(Deconstruct::Fields(reaches))
559    }

561    pub(crate) fn recipes(
562        &self,
563        model: &Flat,
564        expansion_leaves: &[TypeRef],
565        registry: &(impl prebindgen_registry::Conversions + ?Sized),
566    ) -> Result<Recipes, Vec<RecipeError>> {
567        let mut recipes = Recipes::builder();
568        // How each `expand_param!`-declared type is built. Named rather than
569        // defaulted: the crossing's default stays the value's own conversion,
570        // which is what an identity arm is handed (#701 step 2).
571        // Which crossings already state a deconstructing `parts` row, so the
572        // decomposition block below adds one only where none was declared.
573        let mut parts_out: std::collections::HashSet<TypeKey> = std::collections::HashSet::new();
574        // Every Kotlin class declaration, and every `convert!`-declared
575        // conversion. The second matters as much as the first: a conversion may
576        // be declared on a type the registry would otherwise read as an arity
577        // layer, and `convert!(Option<Duration> => ..)` means the adapter emits
578        // that optional's conversion itself rather than wrapping `Duration`'s.
579        let mut declared: Vec<(TypeKey, Origin<syn::Type>)> = self
580            .declared_types()
581            .into_iter()
582            .chain(
583                self.convert_decls
584                    .iter()
585                    .map(|d| (d.key().clone(), d.rust_type().clone())),
586            )
587            .collect();
588        declared.sort_by(|a, b| a.0.as_str().cmp(b.0.as_str()));
589        declared.dedup_by(|a, b| a.0 == b.0);
590
591        for (_key, origin) in declared {
592            // The declarator's own tokens, re-parsed: a declared type may be an
593            // alias the model holds as an `Extern`, which carries no reading of
594            // its own to borrow.
595            let Ok(spelled) = syn::parse2::<syn::Type>(origin.declared_spelling()) else {
596                continue;
597            };
598            let Ok(ty) = model.classify(&spelled) else {
599                continue;
600            };
601            // Every declared Kotlin shape is one recipe with no parts today,
602            // including the two that plainly have some: a `data_class` arrives
603            // as separate JNI parameters and is rebuilt inside one generated
604            // function, and a `sealed_class` is a selector plus the live arm's
605            // slots inside another. `Atomic` is what a recipe says about that —
606            // the adapter emits the conversion itself, and how many wire values
607            // that costs is its own business — so it describes the adapter as it
608            // stands rather than as it will be.
609            // A `sealed_class` hands its value out as a tag plus every
610            // alternative's payloads, laid side by side — the deconstructing
611            // twin of the constructing recipe below, and the direction that
612            // actually has no alternative: a sum has no whole-value output
613            // conversion, because a tag plus groups is not one wire.
614            match self.types.get(&ty.key()).map(|c| &c.kind) {
615                Some(DeclaredKind::Sealed(_)) => {
616                    let arms = out_alternatives(model, &ty);
617                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
618                    if !arms.is_empty() {
619                        recipes.declare(ty.clone(), parts(), Deconstructing::Choice { arms });
620                        parts_out.insert(ty.stripped_key());
621                    }
622                }
623                // A `data_class` hands its value out as its fields too, so the
624                // foreign side reassembles the object and nothing builds one on
625                // the Rust side.
626                Some(DeclaredKind::Data) => {
627                    let reaches = out_fields(model, &ty);
628                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
629                    if !reaches.is_empty() {
630                        recipes.declare(
631                            ty.clone(),
632                            parts(),
633                            Deconstructing::Product(Deconstruct::Fields(reaches)),
634                        );
635                        parts_out.insert(ty.stripped_key());
636                    }
637                }
638                _ => {
639                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
640                }
641            }
642            // A value form: `expand_return!(T).fields(fields!(f))` calls `f`
643            // once and hands out the fields of the struct it returns. The one
644            // declaration shape that names its own accessor, and the reason
645            // `Deconstruct::ValueForm` exists.
646            if let Some((func, reaches)) = self.value_form_of(model, registry, &ty) {
647                recipes.declare(
648                    ty.clone(),
649                    parts(),
650                    Deconstructing::Product(Deconstruct::ValueForm {
651                        func,
652                        parts: reaches,
653                    }),
654                );
655                parts_out.insert(ty.stripped_key());
656            }
657            // A `data_class` also has a recipe that says what it is made of, so
658            // its constructing side names which of the two a site takes by
659            // default. See [`parts`] for why that is still `whole`.
660            match self.types.get(&ty.key()).map(|c| &c.kind) {
661                // A struct with no fields is nothing to be made of, so it
662                // states no `parts` recipe — the same condition the deconstructing
663                // side applies, and the reason the row is looked up by name before
664                // it is compiled.
665                Some(DeclaredKind::Data) if !fields_of(model, &ty).is_empty() => {
666                    recipes
667                        .declare_default(ty.clone(), whole(), Constructing::Atomic)
668                        .declare_unasked(ty, parts(), Constructing::Product(Construct::Fields));
669                }
670                // A `sealed_class` has one too, and it is a choice rather than
671                // a product: exactly one alternative is live, every one of them
672                // still crosses, and the tag says which. The arms are the
673                // model's — a recipe states which parts, never what they are.
674                Some(DeclaredKind::Sealed(_)) => {
675                    let arms = alternatives(model, &ty);
676                    recipes.declare_default(ty.clone(), whole(), Constructing::Atomic);
677                    if !arms.is_empty() {
678                        recipes.declare(ty, parts(), Constructing::Choice { arms });
679                    }
680                }
681                _ => {
682                    recipes.declare(ty, whole(), Constructing::Atomic);
683                }
684            }
685        }
686
687        // A type a callback delivers by taking it apart states that as a row,
688        // so the argument's site has a recipe to name. Without one the site
689        // could only be fabricated — a `Bound` no binding answered — which is
690        // what #622's first two attempts produced and why they were withdrawn.
691        //
692        // `Atomic` for the reason the declared-type loop above gives for the
693        // two shapes that plainly have parts: the adapter emits this
694        // conversion itself, and how many wire values that costs is its own
695        // business. The parts are stated by the deconstructor declaration the
696        // registry already resolved into an `UnfoldPlan`, which this row does
697        // not restate — it names the crossing that plan decomposes. Deleting
698        // that second statement is #613 step 5b's, not this row's.
699        //
700        // The source is the plan's own owned core, so a plan keyed under `&T`
701        // and one keyed under `T` state one row; a type that already declared
702        // `parts` — a `data_class`, a `sealed_class`, a value form — keeps it.
703        //
704        // A **whole-element fold** is not one of these and must not earn a row.
705        // `apply_leaf_vec_folds` files a plan for `impl Fn(&[T])` under `&[T]`
706        // whose `source` is the ELEMENT and whose `decon` is `None`: nothing is
707        // taken apart, each element crosses whole through its own converter.
708        // Declaring `parts` off it would state a decomposition of `T` that does
709        // not exist — and, worse, a scalar `T` argument elsewhere in the same
710        // model would then bind to that row and name a fragment compiled under
711        // a different recipe (#623 review). `decon` is the gate the model
712        // already carries for this: `None` only for the whole-element arm.
713        let mut decomposed: BTreeMap<TypeKey, TypeRef> = BTreeMap::new();
714        // EVERY decomposition, not only a callback argument's. A return and a
715        // domain error come apart by the same declaration — `Error` is one
716        // declaration reached from every fallible function in zenoh-flat-jni —
717        // so each states its row too. Without this the rows described a
718        // fraction of what a binding decomposes, which is what 68 entries of
719        // the parity inventory said (#701 step 3).
720        let unfolded = self.unfolded();
721        let every_plan = unfolded
722            .callback_arg_plans
723            .values()
724            .chain(unfolded.unfold_plans.values())
725            .chain(unfolded.error_plans.values());
726        for plan in every_plan {
727            if plan.decon.is_none() {
728                continue;
729            }
730            let key = plan.source.stripped_key();
731            if parts_out.contains(&key) {
732                continue;
733            }
734            decomposed.entry(key).or_insert_with(|| plan.source.clone());
735        }
736        for decl in &self.return_expand_decls {
737            if parts_out.contains(decl.key()) {
738                continue;
739            }
740            if let Some(ty) = decl
741                .key()
742                .ident()
743                .and_then(|ident| model.classify(&syn::parse_quote!(#ident)).ok())
744            {
745                decomposed.entry(decl.key().clone()).or_insert(ty);
746            }
747        }
748        // A per-FIELD override states how ONE occurrence of a field's type
749        // comes apart, which is a row under a name of its own on that type's
750        // crossing — beside whatever row the type itself states.
751        let type_forms = self.return_expand_decls.iter().map(|decl| (decl, parts()));
752        let fn_forms = self
753            .fn_return_expands
754            .iter()
755            .map(|(func, decl)| (decl, site_row(func)));
756        for (decl, parent) in type_forms.chain(fn_forms) {
757            let Some(owner) = registry.reading(decl.key()) else {
758                continue;
759            };
760            for (field, child, reaches) in self.field_overrides(model, registry, &owner, decl) {
761                recipes.declare(
762                    child,
763                    field_row(&parent, decl.key(), &field),
764                    Deconstructing::Product(Deconstruct::Fields(reaches)),
765                );
766            }
767        }
768
769        // A plan that states NO decomposition still states a crossing: a
770        // whole-element fold hands each element over through its own converter.
771        // The derived row is what that reads as, and declaring it here is what
772        // lets the differential compare such a plan instead of naming it.
773        let mut derived: std::collections::BTreeSet<TypeKey> = Default::default();
774        for plan in unfolded
775            .callback_arg_plans
776            .values()
777            .chain(unfolded.unfold_plans.values())
778            .chain(unfolded.error_plans.values())
779        {
780            // Once per crossing, and only where nothing else states one: a
781            // DECLARED class already has its `whole` default, and a second
782            // default is refused. Several functions fold runs of one element
783            // type, so the row they read is the element's, not each plan's.
784            let key = plan.source.stripped_key();
785            if plan.decon.is_none() && !self.types.contains_key(&key) && derived.insert(key) {
786                recipes.declare_derived_default(plan.source.clone(), Direction::Deconstruct);
787            }
788        }
789        for ty in decomposed.into_values() {
790            // A real row where the declaration can state one. #622 wrote
791            // `Atomic` here because `Reach` could not spell an identity leaf;
792            // it can now (#635), so the row says how the value comes apart
793            // instead of only existing to be selected (#613 step 10).
794            match self.parts_deconstruct(model, registry, &ty) {
795                Some(deconstruct) => {
796                    recipes.declare(ty, parts(), Deconstructing::Product(deconstruct));
797                }
798                None => {
799                    recipes.declare(ty, parts(), Deconstructing::Atomic);
800                }
801            }
802        }
803
804        // A per-function `.expand_return(...)` states a decomposition of its
805        // own, so it gets a row of its own, under a name of its own — the shape
806        // step 2 gave the parameter side (`expand-param@fn#param`), at the
807        // other end. Without it the type's `parts` row and the function's
808        // decomposition sat under one name describing two different things,
809        // which is why the differential could not compare either.
810        for (func, decl) in &self.fn_return_expands {
811            let Some(ty) = decl
812                .key()
813                .ident()
814                .and_then(|ident| model.classify(&syn::parse_quote!(#ident)).ok())
815            else {
816                continue;
817            };
818            match self.decon_of(model, registry, &ty, decl) {
819                Some(deconstruct) => {
820                    recipes.declare(ty, site_row(func), Deconstructing::Product(deconstruct));
821                }
822                None => {
823                    recipes.declare(ty, site_row(func), Deconstructing::Atomic);
824                }
825            }
826        }
827
828        // Every implicit Optional keeps its established whole/default input
829        // row and also offers the allocation-free primitive `pair` row. A
830        // parameter site selects `pair` only after the compiled payload proves
831        // that it is one JNI primitive with no niche, projection, or stage.
832        //
833        // Product-shaped payloads additionally offer `parts`; that row is
834        // selected by the existing flattening bindings below.
835        let mut implicit: BTreeMap<TypeKey, (TypeRef, TypeRef)> = self
836            .implicit_optionals(model)
837            .into_iter()
838            .map(|(key, (outer, inner))| (key, (outer.clone(), inner.clone())))
839            .collect();
840        // A combined constructor expansion represents an inactive arm by
841        // Option-wrapping each required leaf. Those readings are synthesized
842        // after the Flat model was built, so a model walk cannot discover
843        // them. File them in the same table as source-written implicit
844        // Optionals: otherwise a nullable primitive leaf falls back to a boxed
845        // JObject and Rust has to call `intValue()`/`longValue()` across JNI.
846        for leaf in expansion_leaves {
847            let Some(inner) = leaf.optional_inner() else {
848                continue;
849            };
850            if !self.explicitly_declares(leaf) {
851                implicit
852                    .entry(leaf.stripped_key())
853                    .or_insert_with(|| (leaf.clone(), inner.clone()));
854            }
855        }
856        for (outer, _) in implicit.values() {
857            recipes
858                .declare_default(outer.clone(), whole(), Constructing::Optional)
859                .declare(outer.clone(), pair(), Constructing::Optional)
860                .declare_unasked(outer.clone(), parts(), Constructing::Optional);
861        }
862        for (outer, _) in implicit.into_values() {
863            recipes
864                .declare_default(outer.clone(), whole(), Deconstructing::Optional)
865                .declare_unasked(outer.clone(), parts(), Deconstructing::Optional);
866        }
867
868        // A fixed-size array of JNI primitives is one Kotlin `ByteArray` or
869        // `LongArray`, bulk-copied with nothing boxed — one wire value, not a
870        // run this adapter walks. The registry reads `[T; N]` as a run unless
871        // something says otherwise, and this is JniGen saying otherwise for
872        // every array the model holds. Nothing is enumerated twice: a recipe for
873        // a crossing nobody uses is inert.
874        // Keyed by identity in a `BTreeMap`, so each array type is keyed once
875        // and the declaration order is the key order — no sort, no second pass
876        // rebuilding a key it already has, and nothing cloned before the entry
877        // turns out to be new.
878        let mut arrays: BTreeMap<TypeKey, &TypeRef> = BTreeMap::new();
879        for ty in model
880            .elements()
881            .flat_map(element_types)
882            .flat_map(|ty| ty.walk())
883            .filter(|ty| matches!(ty.kind(), prebindgen_registry::flat::TypeKind::Array { .. }))
884        {
885            arrays.entry(ty.key()).or_insert(ty);
886        }
887        for ty in arrays.into_values() {
888            recipes
889                .declare(ty.clone(), whole(), Deconstructing::Atomic)
890                .declare(ty.clone(), whole(), Constructing::Atomic);
891        }
892
893        // The `expand_param!` rows are NOT declared here. They live in the
894        // table `Declarations::expansion_plans` builds for itself, because the
895        // readings they deliver have to be known while the binding declares
896        // itself — before this table can exist. Declaring them here as well
897        // would put a second copy in front of a walk that never reaches them:
898        // no site names one, and both adapters' `choice` refuse the arm shape
899        // they use. They move here when something reads them, in #701's step 3.
900        recipes.build(model)
901    }
902}

904impl Declarations {
905    /// Which recipe each part of a `data_class` takes.
906    ///
907    /// A field that is itself a `data_class` crosses as **its** fields too, so
908    /// the part site asks for [`parts`] rather than letting the field take its
909    /// own default. That is the one thing a site can say that a recipe cannot: the
910    /// same crossing is read one way on its own and another inside a product.
911    ///
912    /// Without it a nested class contributes a single wire and the flattening
913    /// stops one layer down.
914    /// Whether a `data_class` field is itself flattened, or stays one value.
915    ///
916    /// A `data_class` declared `.jobject_input()` crosses Kotlin → Rust as a
917    /// single `JObject` — that is the whole point of the opt-in — so binding
918    /// its parts would flatten a boundary the declaration drew.
919    /// Whether this binding takes a type APART where it appears inside another
920    /// value, rather than crossing it whole.
921    ///
922    /// Three declarations say so: a `data_class`, a `sealed_class`, and a value
923    /// form — `expand_return!(T).fields(..)`, which names its own accessor. The
924    /// proxy for this used to be "declares a `parts` row", which held only
925    /// while an inlined class was the one thing that had one; every
926    /// decomposition states a row now (#701 step 3).
927    fn inlines_parts(
928        &self,
929        model: &Flat,
930        registry: &(impl prebindgen_registry::Conversions + ?Sized),
931        key: &TypeKey,
932    ) -> bool {
933        if self
934            .types
935            .get(key)
936            .is_some_and(|c| c.sum().is_some() || matches!(c.kind, DeclaredKind::Data))
937        {
938            return true;
939        }
940        key.ident()
941            .and_then(|ident| model.classify(&syn::parse_quote!(#ident)).ok())
942            .is_some_and(|ty| self.value_form_of(model, registry, &ty).is_some())
943    }
944
945    fn field_crosses_as_its_fields(&self, ty: &TypeRef) -> bool {
946        // The field's own key, not the stripped one: a `Box<Leaf>` field
947        // crosses as one value through `Box<Leaf>`'s conversion, and only a
948        // parameter spelled that way is peeled back to the class. Binding the
949        // stripped key here would flatten a field the rebuild has no node for.
950        self.types.get(&ty.key()).is_some_and(|c| match c.kind {
951            DeclaredKind::Data => !c.jobject_input,
952            // A `sealed_class` field crosses as a tag plus every alternative's
953            // slots. Whether it *can* is the adapter's answer at compile time,
954            // not a declaration: a payload with no slot form leaves the sum
955            // object-shaped, which is the fragment `Compile::choice` hands
956            // back. So the site asks, and the recipe answers.
957            DeclaredKind::Sealed(_) => true,
958            _ => false,
959        })
960    }
961
962    /// Optional crossings whose payload can use the shared `parts` recipe.
963    ///
964    /// Explicit declarations keep full control of the outer crossing: a
965    /// `convert!(Option<T> => ..)` is an atomic recipe, not an implicit request
966    /// to flatten `T`.
967    fn implicit_optionals<'a>(
968        &self,
969        model: &'a Flat,
970    ) -> BTreeMap<TypeKey, (&'a TypeRef, &'a TypeRef)> {
971        let mut optionals = BTreeMap::new();
972        for ty in model
973            .elements()
974            .flat_map(element_types)
975            .flat_map(|ty| ty.walk())
976        {
977            let Some(inner) = ty.optional_inner() else {
978                continue;
979            };
980            if !self.explicitly_declares(ty) {
981                optionals.entry(ty.stripped_key()).or_insert((ty, inner));
982            }
983        }
984        optionals
985    }
986
987    fn explicitly_declares(&self, ty: &TypeRef) -> bool {
988        self.types.contains_key(&ty.key())
989            || self
990                .convert_decls
991                .iter()
992                .any(|decl| decl.key() == &ty.key() || decl.key() == &ty.stripped_key())
993    }
994
995    pub(crate) fn bindings(
996        &self,
997        model: &Flat,
998        registry: &(impl prebindgen_registry::Conversions + ?Sized),
999        recipes: &prebindgen_registry::recipe::Recipes,
1000    ) -> Result<prebindgen_registry::recipe::Bindings, Vec<RecipeError>> {
1001        use prebindgen_registry::recipe::{
1002            Ask, Bindings, Crossing, Direction, Origin, RecipeName, Site,
1003        };
1004
1005        let mut bound = Bindings::builder();
1006
1007        // What crosses at a **convert** return is not the value the signature
1008        // names. The binding converts it first, and the converted value's type
1009        // is the function's own output expansion answer, which no model can
1010        // state. Bound here so the walk plans that crossing rather than the
1011        // return type, and so the site needs no adapter-private planning.
1012        // What an **expanded** return crosses. A return the binding converts or
1013        // takes apart does not cross the value its signature names, and only
1014        // the binding's own output expansion says what it does cross. Bound
1015        // here so the walk plans that crossing, which is what lets the site be
1016        // planned once rather than again at the emission site.
1017        //
1018        // One binding per function, in the same precedence `build_output`
1019        // applies: a function may have BOTH an output expansion and an error
1020        // plan, and the output expansion wins. Binding each in its own pass
1021        // would answer one site twice and be refused as a rebind.
1022        let mut owners: std::collections::BTreeSet<String> = Default::default();
1023        let unfolded = self.unfolded();
1024        owners.extend(unfolded.unfold_plans.keys().map(|f| f.to_string()));
1025        owners.extend(unfolded.error_plans.keys().map(|f| f.to_string()));
1026        for owner in owners {
1027            let func = quote::format_ident!("{owner}");
1028            let (crossed, ask) = match unfolded.unfold_plans.get(&func) {
1029                // A convert hands over one converted value and takes whatever
1030                // row that value has; a decomposition crosses the value it
1031                // takes apart and asks for the row that names its parts.
1032                Some(plan) => match plan.delivery {
1033                    crate::unfold::Delivery::Return => match plan.convert_out_ty.as_ref() {
1034                        Some(out) => (out.clone(), Ask::Default),
1035                        None => continue,
1036                    },
1037                    crate::unfold::Delivery::Callback => {
1038                        let crossing = Crossing::new(plan.source.clone(), Direction::Deconstruct);
1039                        // The row that states THIS function's decomposition: a
1040                        // function with its own `.expand_return(...)` takes
1041                        // apart differently from the way its type does, and
1042                        // this is the binding the return site resolves — the
1043                        // one with `Origin::Function`, which no weaker binding
1044                        // can displace.
1045                        let row = if self.fn_return_expands.iter().any(|(g, _)| *g == func) {
1046                            site_row(&func)
1047                        } else {
1048                            parts()
1049                        };
1050                        let ask = if recipes.key_of(&crossing.key(), &row).is_some() {
1051                            Ask::Recipe(row)
1052                        } else {
1053                            Ask::Default
1054                        };
1055                        (plan.source.clone(), ask)
1056                    }
1057                },
1058                // No output expansion, but the binding peels the error: only
1059                // the ok arm crosses at the return, because the error is thrown
1060                // rather than crossed there.
1061                None => {
1062                    let Some((ok, _)) = model
1063                        .function(&func)
1064                        .and_then(|function| function.ret.fallible_parts())
1065                    else {
1066                        continue;
1067                    };
1068                    (ok.clone(), Ask::Default)
1069                }
1070            };
1071            bound.bind(
1072                Site {
1073                    owner: func,
1074                    role: prebindgen_registry::recipe::Role::Return,
1075                },
1076                Crossing::new(crossed, Direction::Deconstruct),
1077                ask,
1078                Origin::Function,
1079            );
1080        }
1081
1082        let mut declared: Vec<TypeKey> = self
1083            .types
1084            .iter()
1085            .filter(|(_, c)| matches!(c.kind, DeclaredKind::Data))
1086            .map(|(k, _)| k.clone())
1087            .collect();
1088        declared.sort_by(|a, b| a.as_str().cmp(b.as_str()));
1089
1090        // The named Optional row is the representation suitable inside another
1091        // composed shape. Where its child has a Product row it delegates to that
1092        // row; a scalar child stays on its default row and lets the JNI compiler
1093        // choose a niche or a `(present, value)` intermediate from the compiled
1094        // child facts. The whole/default Optional row remains untouched.
1095        //
1096        // Bind both directions: function inputs construct the Rust optional,
1097        // while returns and callback arguments deconstruct it.
1098        for (outer, inner) in self.implicit_optionals(model).into_values() {
1099            for direction in [Direction::Construct, Direction::Deconstruct] {
1100                let outer = Crossing::new(outer.clone(), direction);
1101                let row = outer.row(parts());
1102                let inner = Crossing::new(inner.clone(), direction);
1103                // A DECLARED CLASS's parts are inlined here; anything else
1104                // crosses whole. "Has a `parts` row" was the proxy for that
1105                // while only a declared class had one — every decomposition
1106                // states a row now, so the proxy would inline a field the
1107                // decomposition crosses whole. The condition says what it
1108                // meant (#701 step 3).
1109                let inlines = self.inlines_parts(model, registry, &inner.key().ty)
1110                    && recipes.key_of(&inner.key(), &parts()).is_some();
1111                bound.bind(
1112                    Site::part(&row, 0),
1113                    inner.clone(),
1114                    if inlines {
1115                        Ask::Recipe(parts())
1116                    } else {
1117                        Ask::Default
1118                    },
1119                    Origin::Part,
1120                );
1121            }
1122        }
1123
1124        for key in declared {
1125            let Some(ident) = key.ident() else { continue };
1126            let Some(prebindgen_registry::flat::Type::Struct(s)) = model.declared_type(&ident)
1127            else {
1128                continue;
1129            };
1130            let Ok(ty) = model.classify(&syn::parse_quote!(#ident)) else {
1131                continue;
1132            };
1133            let building = Crossing::new(ty.clone(), Direction::Construct);
1134            let handing_out = Crossing::new(ty, Direction::Deconstruct);
1135            for (index, field) in s.fields.iter().enumerate() {
1136                // A field selects `parts` exactly when its crossing declares
1137                // that row AND the declaration says this occurrence decomposes.
1138                // Direct classes retain the existing `.jobject_input()` and
1139                // erased-wrapper boundaries; every implicit Optional selects its
1140                // part representation, which decides whether the child is
1141                // another composition or one scalar intermediate.
1142                if field.ty.optional_inner().is_none()
1143                    && !self.field_crosses_as_its_fields(&field.ty)
1144                {
1145                    continue;
1146                }
1147                for (of, direction) in [
1148                    (&building, Direction::Construct),
1149                    (&handing_out, Direction::Deconstruct),
1150                ] {
1151                    // Handing a value OUT, an optional field asks the question
1152                    // of what is inside the layer: a decomposition looks
1153                    // through `Option<T>` only where it inlines T's fields, and
1154                    // crosses `Option<Summary>` — an optional handle — whole.
1155                    // "Declares a `parts` row" was the proxy for that while only
1156                    // an inlined class had one; every decomposition states one
1157                    // now, so the proxy would look through a layer the
1158                    // decomposition crosses whole (#701 step 3).
1159                    //
1160                    // Building one is unchanged: a `(present, value)` pair is
1161                    // how every optional parameter arrives, whatever is inside.
1162                    if direction == Direction::Deconstruct
1163                        && field
1164                            .ty
1165                            .optional_inner()
1166                            .is_some_and(|inner| !self.field_crosses_as_its_fields(inner))
1167                    {
1168                        continue;
1169                    }
1170                    let row = of.row(parts());
1171                    let field_crossing = Crossing::new(field.ty.clone(), direction);
1172                    if recipes.key_of(&field_crossing.key(), &parts()).is_none() {
1173                        continue;
1174                    }
1175                    bound.bind(
1176                        Site::part(&row, index),
1177                        field_crossing,
1178                        Ask::Recipe(parts()),
1179                        Origin::Part,
1180                    );
1181                }
1182            }
1183        }
1184        // Every function whose return the binding takes apart, bound to the
1185        // `parts` recipe of its return type.
1186        //
1187        // This is what a site naming several values is: `Ask::Recipe` already
1188        // lets a site pick a recipe, and a `parts` fragment already occupies
1189        // several wires — the two facts stages 3 and 4 established, meeting.
1190        // The registry needs nothing new to express a decomposed return.
1191        for f in model.functions() {
1192            // The value that crosses, through the layers a decomposition looks
1193            // through: a `&T` return decomposes T, and so does an `Option<T>`
1194            // or a `Vec<T>` — the shape rides the delivery, not the recipe.
1195            let ret = f.ret.borrow_target().unwrap_or(&f.ret);
1196            let ret = ret.optional_inner().unwrap_or(ret);
1197            let ret = ret.sequence_elem().unwrap_or(ret);
1198            let ret = ret.borrow_target().unwrap_or(ret);
1199            let crossing = Crossing::new(ret.clone(), Direction::Deconstruct);
1200            // Only where the type states one. A `sealed_class` always does; a
1201            // `data_class` states one unless a field of it declines, and a
1202            // return whose type states none crosses whole.
1203            if recipes.key_of(&crossing.key(), &parts()).is_none() {
1204                continue;
1205            }
1206            // And only where it is genuinely several. A decomposition that
1207            // yields ONE value takes `Delivery::Return`: the wrapper hands that
1208            // value back through its own conversion rather than through a
1209            // builder, so the return site's crossing is the value's, not this
1210            // type's. Asked of the composition, which is model and declaration
1211            // only and so answerable here — the same property that let one
1212            // composition serve both sides of `resolve`.
1213            if self.out_wire_count(registry, ret) < 2 {
1214                continue;
1215            }
1216            // A function that wrote its own `.expand_return(...)` names the row
1217            // that states IT, not the type's.
1218            bound.bind(
1219                Site {
1220                    owner: f.name.clone(),
1221                    role: prebindgen_registry::recipe::Role::Return,
1222                },
1223                crossing,
1224                Ask::Recipe(parts()),
1225                Origin::Adapter,
1226            );
1227        }
1228
1229        // A callback's `Invoke` recipe owns its argument conversions. Bind a
1230        // model-derived data-class argument to `parts` so the Product fragment
1231        // reaches `Compile::callback` before the trampoline is rendered.
1232        for f in model.functions() {
1233            for param in &f.params {
1234                let prebindgen_registry::flat::TypeKind::Callback { args } =
1235                    param.ty.unwrapped().kind()
1236                else {
1237                    continue;
1238                };
1239                let callback = Crossing::new(param.ty.clone(), Direction::Construct);
1240                let row = callback.row(RecipeName::derived());
1241                for (index, arg) in args.iter().enumerate() {
1242                    let crossed = arg.borrow_target().unwrap_or(arg);
1243                    let core = crossed.optional_inner().unwrap_or(crossed);
1244                    if let Some(element) = core.sequence_elem() {
1245                        let element = element.borrow_target().unwrap_or(element);
1246                        if !self.field_crosses_as_its_fields(element) {
1247                            continue;
1248                        }
1249                        let sequence = Crossing::new(arg.clone(), Direction::Deconstruct);
1250                        let element_crossing =
1251                            Crossing::new(element.borrowed(), Direction::Deconstruct);
1252                        if recipes.key_of(&element_crossing.key(), &parts()).is_none() {
1253                            continue;
1254                        }
1255                        bound.bind(
1256                            Site::part(&sequence.row(RecipeName::derived()), 0),
1257                            element_crossing,
1258                            Ask::Recipe(parts()),
1259                            Origin::Adapter,
1260                        );
1261                        continue;
1262                    }
1263                    // A model-derived data class, or a type whose deconstructor
1264                    // declaration earned the `parts` row above. Both take that
1265                    // row here, so the fragment the trampoline delivers and the
1266                    // one the argument's own `Role::CallbackArg` site names are
1267                    // one fragment rather than two rows over one crossing.
1268                    if !self.field_crosses_as_its_fields(core)
1269                        && !self.unfolded().callback_arg_plans.contains_key(&arg.key())
1270                    {
1271                        continue;
1272                    }
1273                    let crossing = Crossing::new(arg.clone(), Direction::Deconstruct);
1274                    if recipes.key_of(&crossing.key(), &parts()).is_none() {
1275                        continue;
1276                    }
1277                    bound.bind(
1278                        Site::part(&row, index),
1279                        crossing,
1280                        Ask::Recipe(parts()),
1281                        Origin::Adapter,
1282                    );
1283                }
1284            }
1285        }
1286
1287        // Each value a callback delivers is a root site of its own — the
1288        // function-unique `Role::CallbackArg` the registry names, not the
1289        // `Role::Part` above, which is keyed by the callback recipe every
1290        // function with that signature shares.
1291        //
1292        // Bound to `parts` exactly where the argument's crossing states that
1293        // row, which is now every argument the trampoline takes apart: a
1294        // `data_class`, a `sealed_class`, an implicit Optional over one of
1295        // those, or a type whose deconstructor declaration earned the row
1296        // declared above. An argument that crosses whole names nothing here and
1297        // takes its crossing's default, attributed to the adapter.
1298        for f in model.functions() {
1299            for (param, p) in f.params.iter().enumerate() {
1300                let prebindgen_registry::flat::TypeKind::Callback { args } =
1301                    p.ty.unwrapped().kind()
1302                else {
1303                    continue;
1304                };
1305                for (arg, ty) in args.iter().enumerate() {
1306                    let crossing = Crossing::new(ty.clone(), Direction::Deconstruct);
1307                    if recipes.key_of(&crossing.key(), &parts()).is_none() {
1308                        continue;
1309                    }
1310                    bound.bind(
1311                        Site {
1312                            owner: f.name.clone(),
1313                            role: prebindgen_registry::recipe::Role::CallbackArg { param, arg },
1314                        },
1315                        crossing,
1316                        Ask::Recipe(parts()),
1317                        Origin::Adapter,
1318                    );
1319                }
1320            }
1321        }
1322
1323        // Which accessor parts are taken apart further, which is a binding
1324        // rather than a property of the row: a `data_class` states a `parts`
1325        // row and still crosses whole as an accessor's return. The rule is the
1326        // decomposition's own — splice where the part's type has a
1327        // deconstructor DECLARATION — stated here so the leaf view and the
1328        // compiler read one answer (#701 step 3, design decision 3).
1329        for decl in &self.return_expand_decls {
1330            let Some(owner) = registry.reading(decl.key()) else {
1331                continue;
1332            };
1333            let row = Crossing::new(owner, Direction::Deconstruct).row(parts());
1334            for (index, field) in decl.field_list().iter().enumerate() {
1335                let crate::jni::LocalField::Named(func, _) = field else {
1336                    continue;
1337                };
1338                let Some(ret) = model.function(func).map(|f| f.ret.clone()) else {
1339                    continue;
1340                };
1341                // An OPTIONAL part comes apart like any other: it crosses as
1342                // the accessor's return states it, LAYERS INCLUDED, which is
1343                // the crossing both the compiler and the leaf view build for
1344                // it. The binding names `Option<&Report>` and the payload's
1345                // own row is reached through the Optional's value slot,
1346                // exactly as it is for an optional field.
1347                // The bounded delivery peel, not `stripped_key`: that reaches
1348                // the leaf type through ANY nesting, so a part typed
1349                // `Vec<Option<T>>` would be read as the declaration's own `T`
1350                // and spliced — making the row reach its own crossing. The
1351                // inner optional is part of the element, and the part crosses
1352                // as the distinct `Vec<Option<T>>`.
1353                let (_, core) = ret.layer_stack();
1354                let core = core.borrow_target().unwrap_or(core);
1355                // Two conditions, and both are needed. A DECLARATION is what
1356                // says this part comes apart — a type may state a `parts` row
1357                // and still cross whole here, which is the difference this
1358                // binding carries. And the row has to exist to be named: a
1359                // declared type whose row this table does not state yet is
1360                // step 3's remaining work, not a binding to write now.
1361                // Two questions of two types. WHAT COMES APART is the value
1362                // inside the layers, and a DECLARATION is what says it does.
1363                // WHAT CROSSES is the accessor's return as written, and that is
1364                // what the binding names.
1365                if !self
1366                    .return_expand_decls
1367                    .iter()
1368                    .any(|d| *d.key() == core.stripped_key())
1369                {
1370                    continue;
1371                }
1372                // A part that reaches the type being taken apart crosses as
1373                // ONE leaf — the conditional-handle idiom, `fn(&T) -> Option<&T>`,
1374                // which re-hands the receiver rather than reading something out
1375                // of it. Splicing it would make this row reach its own
1376                // crossing; refused here, at the binding, so only this part is
1377                // affected and the rest of the decomposition still comes apart.
1378                if core.stripped_key() == *decl.key() {
1379                    continue;
1380                }
1381                let part = Crossing::new(ret.clone(), Direction::Deconstruct);
1382                if recipes.key_of(&part.key(), &parts()).is_none() {
1383                    continue;
1384                }
1385                bound.bind(
1386                    Site::arm_part(&row, None, index),
1387                    part,
1388                    Ask::Recipe(parts()),
1389                    Origin::Adapter,
1390                );
1391            }
1392        }
1393
1394        // A VALUE FORM's parts, bound the same way an accessor's are. The row
1395        // states which field of the returned struct each part reaches; a
1396        // binding states that the part comes apart further and to what. That
1397        // split is design decision 3, and it is what lets a sum-typed field —
1398        // a selector plus one group per alternative, which no `Reach` spells —
1399        // be a `Choice` part rather than a shape smuggled into the row.
1400        // Every value form, under the row that states it: a type's own under
1401        // `parts`, and a function's own under `expand-return@fn`. A per-function
1402        // `.expand_return(..)` declares a real `ValueForm` row of its own, so
1403        // its sum-typed field needs the same binding — without one the row says
1404        // `b : ZValOutcome` where the decomposition says a tag and two groups.
1405        let type_forms = self.return_expand_decls.iter().map(|decl| (decl, parts()));
1406        let fn_forms = self
1407            .fn_return_expands
1408            .iter()
1409            .map(|(func, decl)| (decl, site_row(func)));
1410        for (decl, name) in type_forms.chain(fn_forms) {
1411            let Some(owner) = registry.reading(decl.key()) else {
1412                continue;
1413            };
1414            let Some((func, reaches, spliced)) =
1415                self.value_form_parts(model, registry, &owner, decl)
1416            else {
1417                continue;
1418            };
1419            let Some(st) = model.function(&func).map(|f| f.ret.clone()) else {
1420                continue;
1421            };
1422            let st = st.borrow_target().unwrap_or(&st);
1423            let Some(ident) = st.stripped_key().ident() else {
1424                continue;
1425            };
1426            let Some(prebindgen_registry::flat::Type::Struct(st)) = model.declared_type(&ident)
1427            else {
1428                continue;
1429            };
1430            let row = Crossing::new(owner, Direction::Deconstruct).row(name.clone());
1431            for (index, reach) in reaches.iter().enumerate() {
1432                let prebindgen_registry::recipe::Reach::Field(field) = reach else {
1433                    continue;
1434                };
1435                if !spliced.get(index).copied().unwrap_or(false) {
1436                    continue;
1437                }
1438                let Some(field) = st.fields.get(*field) else {
1439                    continue;
1440                };
1441                // The row this occurrence comes apart by: the one the
1442                // declaration wrote for THIS field where it wrote one, and the
1443                // field type's own `parts` otherwise.
1444                let overridden = field
1445                    .name
1446                    .as_ref()
1447                    .map(|n| field_row(&name, decl.key(), &n.to_string()))
1448                    .filter(|r| {
1449                        recipes
1450                            .key_of(
1451                                &Crossing::new(field.ty.clone(), Direction::Deconstruct).key(),
1452                                r,
1453                            )
1454                            .is_some()
1455                    });
1456                let name = overridden.unwrap_or_else(parts);
1457                let crossing = Crossing::new(field.ty.clone(), Direction::Deconstruct);
1458                if recipes.key_of(&crossing.key(), &name).is_none() {
1459                    continue;
1460                }
1461                bound.bind(
1462                    Site::part(&row, index),
1463                    crossing,
1464                    Ask::Recipe(name),
1465                    Origin::Adapter,
1466                );
1467            }
1468        }
1469
1470        bound.build(recipes, model)
1471    }
