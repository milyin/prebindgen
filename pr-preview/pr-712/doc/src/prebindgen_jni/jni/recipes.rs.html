1//! JniGen's Kotlin class declarations, lowered into the registry's recipe
2//! vocabulary.
3//!
4//! A build script writes `ptr_class!`, `data_class!`, `enum_class!` or
5//! `sealed_class!`, each naming the Kotlin type one Rust type becomes, and
6//! `convert!` for a conversion the binding supplies itself. This turns those
7//! into recipes in a [`Recipes`] table: the shared statement of **which parts** a
8//! value gets across in, with nothing about the JNI wire in it.
9//!
10//! One recipe per declared type and direction, and every crossing nobody declared
11//! takes
12//! the recipe the registry derives from its kind. The chain of guesses this
13//! replaced — try a terminal, then a `Result`, then an optional, then a run,
14//! then a borrow, then a transparent bridge — is a lookup for the declared
15//! half, and the derived half is the registry's.
16

29/// Every type one captured item names, so a reading nested inside it can be
30/// found without the caller knowing which kind of item it came from.
31fn element_types(element: &prebindgen_registry::Element) -> Vec<&TypeRef> {
32    use prebindgen_registry::flat::Type;
33    match element {
34        prebindgen_registry::Element::Function(f) => f
35            .params
36            .iter()
37            .map(|p| &p.ty)
38            .chain(std::iter::once(&f.ret))
39            .collect(),
40        prebindgen_registry::Element::Type(Type::Struct(s)) => {
41            s.fields.iter().map(|f| &f.ty).collect()
42        }
43        prebindgen_registry::Element::Type(Type::Variant(v)) => v
44            .alternatives
45            .iter()
46            .flat_map(|a| a.fields.iter().map(|f| &f.ty))
47            .collect(),
48        prebindgen_registry::Element::Constant(c) => vec![&c.ty],
49        _ => Vec::new(),
50    }
51}

53/// One arm per alternative of a declared sum, each assembled from its own
54/// payload fields.
55///
56/// Empty for anything the model does not hold as a data-carrying enum, which
57/// includes a `sealed_class!` declared over a type the scan dropped — the scan
58/// reports that, and a recipe over no arms would report it a second time.
59fn alternatives(model: &Flat, ty: &TypeRef) -> Vec<Arm<Construct>> {
60    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
61        return Vec::new();
62    };
63    let Some(prebindgen_registry::flat::Type::Variant(v)) =
64        id.ident().and_then(|ident| model.declared_type(&ident))
65    else {
66        return Vec::new();
67    };
68    v.alternatives
69        .iter()
70        .map(|alt| Arm {
71            alternative: Some(alt.index),
72            op: Construct::Fields,
73        })
74        .collect()
75}

77/// One arm per alternative of a declared sum, each taken apart into its own
78/// payload fields.
79///
80/// The deconstructing twin of [`alternatives`]. Separate because the two directions
81/// name different operations — building an alternative is `Construct::Fields`,
82/// reading one is `Deconstruct::Fields` over the reaches that get there — and
83/// a shared helper would have to be generic over an operation neither side
84/// chooses.
85fn out_alternatives(model: &Flat, ty: &TypeRef) -> Vec<Arm<Deconstruct>> {
86    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
87        return Vec::new();
88    };
89    let Some(prebindgen_registry::flat::Type::Variant(v)) =
90        id.ident().and_then(|ident| model.declared_type(&ident))
91    else {
92        return Vec::new();
93    };
94    v.alternatives
95        .iter()
96        .map(|alt| Arm {
97            alternative: Some(alt.index),
98            op: Deconstruct::Fields((0..alt.fields.len()).map(Reach::Field).collect()),
99        })
100        .collect()
101}

104    /// How many values this type hands out, or 0 if it hands out none.
105    ///
106    /// Model and declaration only, like the compositions it asks — so a
107    /// declaration-time caller may ask it, which is what binding a return site
108    /// needs.
109    fn out_wire_count(
110        &self,
111        registry: &(impl prebindgen_registry::Conversions + ?Sized),
112        ty: &TypeRef,
113    ) -> usize {
114        let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
115            return 0;
116        };
117        let Some(ident) = id.ident() else { return 0 };
118        match registry.flat().declared_type(&ident) {
119            Some(prebindgen_registry::flat::Type::Variant(_)) => self
120                .sum_out_wires(registry.flat(), &ident, ty)
121                .map_or(0, |w| w.len()),
122            Some(prebindgen_registry::flat::Type::Struct(_)) => {
123                self.struct_out_wires(registry, ty).map_or(0, |w| w.len())
124            }
125            _ => 0,
126        }
127    }

129    /// The accessor a type's value form names, and one reach per field of the
130    /// struct it returns.
131    ///
132    /// `None` unless the declaration is exactly one `.fields(fields!(f))` whose
133    /// every field is a plain leaf. A record that states its own decomposition
134    /// — a per-field `expand_return!` override, an inlined nested class, a sum
135    /// laid out as a selector and its groups — is a shape this recipe does not
136    /// state yet, and stating half of one would describe a boundary nothing
137    /// emits.
138    fn value_form_of(
139        &self,
140        model: &Flat,
141        registry: &(impl prebindgen_registry::Conversions + ?Sized),
142        ty: &TypeRef,
143    ) -> Option<(syn::Ident, Vec<Reach>)> {
144        let decl = self
145            .return_expand_decls
146            .iter()
147            .find(|d| *d.key() == ty.stripped_key())?;
148        self.value_form_in(model, registry, ty, decl)
149    }
150
151    /// [`Self::value_form_of`] against ONE declaration — the type's own, or the
152    /// one a function wrote for its return. A per-function
153    /// `.expand_return(expand_return!(T).fields(..))` states its own accessor
154    /// and its own field list, which may differ from the type's or exist
155    /// without one, so the lowering reads the declaration it was handed rather
156    /// than looking the type up again.
157    fn value_form_in(
158        &self,
159        model: &Flat,
160        registry: &(impl prebindgen_registry::Conversions + ?Sized),
161        ty: &TypeRef,
162        decl: &crate::jni::ExpandReturnDecl,
163    ) -> Option<(syn::Ident, Vec<Reach>)> {
164        use crate::unfold::{FieldDecon, FieldRecord};
165        let [crate::jni::LocalField::Fields(fields)] = decl.field_list() else {
166            return None;
167        };
168        let records: Vec<FieldRecord> = self.lower_value_form(registry, decl.key(), fields);
169        let ret = model.function(fields.func())?.ret.clone();
170        let ret = ret.borrow_target().unwrap_or(&ret);
171        let prebindgen_registry::flat::TypeKind::Named { id, .. } = ret.unwrapped().kind() else {
172            return None;
173        };
174        let prebindgen_registry::flat::Type::Struct(st) =
175            id.ident().and_then(|i| model.declared_type(&i))?
176        else {
177            return None;
178        };
179        let mut reaches = Vec::new();
180        for record in &records {
181            // A field the ADAPTER took apart — a decomposed sum, a selector
182            // plus one group per alternative — is a plain reach here, and the
183            // taking-apart is a part binding on it (design decision 3). The row
184            // says which field; the binding says that it comes apart, and to
185            // what. Only `Records`, which replaces the field's type default
186            // wholesale with a list written at this site, is a statement no row
187            // makes yet.
188            if matches!(record.decon, FieldDecon::Records(_)) {
189                return None;
190            }
191            // A record with several members is an INLINED nested class: each
192            // member is one hop of a field-access chain. `Reach::Path` states
193            // that; before it existed this row declined and fell back to a
194            // placeholder (#613 step 10).
195            let mut here = st;
196            let mut indices = Vec::with_capacity(record.members.len());
197            for member in &record.members {
198                let hop = here
199                    .fields
200                    .iter()
201                    .position(|f| f.name.as_ref() == Some(member))?;
202                indices.push(hop);
203                if indices.len() < record.members.len() {
204                    let next = here.fields[hop].ty.stripped_key().ident()?;
205                    let prebindgen_registry::flat::Type::Struct(next) =
206                        model.declared_type(&next)?
207                    else {
208                        return None;
209                    };
210                    here = next;
211                }
212            }
213            let [index] = indices.as_slice() else {
214                reaches.push(Reach::Path(indices));
215                continue;
216            };
217            let index = *index;
218            // A field that reaches the type being taken apart. The leaf
219            // synthesis treats `Box<T>` as a spelling of its own and stops
220            // there; a recipe keys a crossing by the value that crosses, so
221            // `Box<ZSample>` inside `ZSample`'s own value form is that recipe
222            // reaching itself. Refused here rather than by `Recipes::build`,
223            // which would refuse the whole binding over a shape the leaf
224            // synthesis handles.
225            if st.fields[index].ty.stripped_key() == ty.stripped_key() {
226                return None;
227            }
228            reaches.push(Reach::Field(index));
229        }
230        Some((fields.func().clone(), reaches))
231    }
232}

235    /// What a value form calls each of its fields, by the Rust field ident.
236    ///
237    /// The declaration's answer, so a `.name(..)` rename carries through to the
238    /// builder parameter. `None` for a type with no value form, or one whose
239    /// records this recipe does not state — the same test [`Self::value_form_of`]
240    /// makes when it declares the recipe.
241    pub(crate) fn value_form_names(
242        &self,
243        registry: &(impl prebindgen_registry::Conversions + ?Sized),
244        ty: &TypeRef,
245    ) -> Option<std::collections::HashMap<String, String>> {
246        use crate::unfold::FieldDecon;
247        let decl = self
248            .return_expand_decls
249            .iter()
250            .find(|d| *d.key() == ty.stripped_key())?;
251        let [crate::jni::LocalField::Fields(fields)] = decl.field_list() else {
252            return None;
253        };
254        self.lower_value_form(registry, decl.key(), fields)
255            .into_iter()
256            .map(|r| match (r.members.as_slice(), &r.decon) {
257                ([member], FieldDecon::Default) => Some((member.to_string(), r.name)),
258                _ => None,
259            })
260            .collect()
261    }

264/// One reach per field of a declared struct, in the model's order.
265///
266/// Empty for anything the model does not hold as a struct, which includes a
267/// `data_class!` over a type the scan dropped — the scan reports that, and a
268/// recipe over no fields would report it a second time.
269fn out_fields(model: &Flat, ty: &TypeRef) -> Vec<Reach> {
270    (0..fields_of(model, ty).len()).map(Reach::Field).collect()
271}

273/// The fields of a declared struct, or none for anything the model does not
274/// hold as one.
275fn fields_of<'a>(model: &'a Flat, ty: &TypeRef) -> &'a [prebindgen_registry::flat::Field] {
276    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
277        return &[];
278    };
279    match id.ident().and_then(|ident| model.declared_type(&ident)) {
280        Some(prebindgen_registry::flat::Type::Struct(s)) => &s.fields,
281        _ => &[],
282    }
283}

317/// Whether a reach's type mentions `owner` anywhere inside it.
318///
319/// Deliberately NOT a delivery peel. [`TypeRef::layer_stack`] answers "what
320/// does this crossing deliver", and is bounded because `Vec<Option<T>>` has
321/// the optional inside the ELEMENT rather than in the boundary. The question
322/// here is a different one — "would splicing this part make the row reach its
323/// own crossing" — and a self-reference nested out of position still closes the
324/// loop: `T`'s row reaches `Vec<Option<T>>`, whose element row reaches `T`.
325/// Answering it with a peel of either depth conflates the two.
326fn mentions(ty: &TypeRef, owner: &TypeKey) -> bool {
327    ty.walk().iter().any(|node| node.stripped_key() == *owner)
328}

331    /// Every recipe this binding's declarations state.
332    ///
333    /// A type declared but absent from the model is skipped rather than
334    /// refused: the scan already reports it, and reporting it twice in
335    /// different words helps nobody.
336    /// This type's `parts` decomposition, read off its `expand_return!`
337    /// declaration in the recipe table's own vocabulary.
338    ///
339    /// `None` when the declaration states a shape a `Deconstruct` cannot hold —
340    /// a value-form record mixed with others, since `ValueForm` is the whole
341    /// row while `LocalField::Fields` is one record among many. No declaration
342    /// in this workspace has that shape (measured: every `Fields` record stands
343    /// alone), so the caller's `Atomic` fallback is a guard rather than a path
344    /// anything takes.
345    fn parts_deconstruct(
346        &self,
347        model: &Flat,
348        registry: &(impl prebindgen_registry::Conversions + ?Sized),
349        ty: &TypeRef,
350    ) -> Option<Deconstruct> {
351        let decl = self
352            .return_expand_decls
353            .iter()
354            .find(|d| *d.key() == ty.stripped_key())?;
355        self.decon_of(model, registry, ty, decl)
356    }

358    /// [`Self::parts_deconstruct`] against ONE declaration, whichever states
359    /// it: the type's own, or the one a function wrote for its return.
360    fn decon_of(
361        &self,
362        model: &Flat,
363        registry: &(impl prebindgen_registry::Conversions + ?Sized),
364        ty: &TypeRef,
365        decl: &crate::jni::ExpandReturnDecl,
366    ) -> Option<Deconstruct> {
367        let fields = decl.field_list();
368        if matches!(fields, [crate::jni::LocalField::Fields(_)]) {
369            // `ValueForm` means "call this accessor, then read these parts off
370            // its result" — the parts are how the compiler reaches the returned
371            // struct's fields, so an empty list is a decomposition with nothing
372            // in it rather than one of everything (#638 review).
373            //
374            // `value_form_of` already computes exactly that mapping, and it is
375            // not a count: `lower_value_form` returns FLATTENED records, one of
376            // which can splice a child's fields, so `Report` yields six records
377            // over a five-field struct. It maps each record back to the struct
378            // field it names and declines the shapes a row cannot hold.
379            let (func, parts) = self.value_form_in(model, registry, ty, decl)?;
380            return Some(Deconstruct::ValueForm { func, parts });
381        }
382        let mut reaches = Vec::with_capacity(fields.len());
383        for field in fields {
384            // A field that reaches the type being taken apart — the
385            // conditional-handle idiom, `fn(&T) -> Option<&T>`. Splicing it
386            // would make this row reach its own crossing; the field crosses as
387            // one nullable handle leaf instead. Refused here for the same
388            // reason `value_form_of` refuses its own case: `Recipes::build`
389            // would otherwise fail the whole binding over one field.
390            let reaches_self = match field {
391                crate::jni::LocalField::Local { sig, .. } => match &sig.output {
392                    syn::ReturnType::Type(_, ret) => model
393                        .classify(ret)
394                        .is_ok_and(|r| mentions(&r, &ty.stripped_key())),
395                    syn::ReturnType::Default => false,
396                },
397                crate::jni::LocalField::Named(func, _) => model
398                    .function(func)
399                    .is_some_and(|f| mentions(&f.ret, &ty.stripped_key())),
400                _ => false,
401            };
402            if reaches_self {
403                return None;
404            }
405            reaches.push(match field {
406                crate::jni::LocalField::Named(func, _) => Reach::Accessor(func.clone()),
407                // The handle itself — the leaf `Reach::Identity` was added for.
408                crate::jni::LocalField::SelfField => Reach::Identity,
409                // A binding-local accessor resolves by its path's last segment:
410                // `local_functions()` registers it under exactly that ident.
411                crate::jni::LocalField::Local { path, .. } => {
412                    Reach::Accessor(path.segments.last()?.ident.clone())
413                }
414                crate::jni::LocalField::Fields(_) => return None,
415            });
416        }
417        // A row names a part by the accessor that reaches it, so one accessor
418        // used twice — `.field(fun!(f).name("a")).field(fun!(f).name("b"))` —
419        // is two parts the row cannot tell apart. The declaration's name
420        // overrides say which is which and `Reach` has nowhere to put them, so
421        // the row declines rather than stating two leaves of one name.
422        let mut seen: Vec<&syn::Ident> = Vec::with_capacity(reaches.len());
423        for reach in &reaches {
424            if let Reach::Accessor(func) = reach {
425                if seen.contains(&func) {
426                    return None;
427                }
428                seen.push(func);
429            }
430        }
431        Some(Deconstruct::Fields(reaches))
432    }

434    pub(crate) fn recipes(
435        &self,
436        model: &Flat,
437        expansion_leaves: &[TypeRef],
438        registry: &(impl prebindgen_registry::Conversions + ?Sized),
439    ) -> Result<Recipes, Vec<RecipeError>> {
440        let mut recipes = Recipes::builder();
441        // How each `expand_param!`-declared type is built. Named rather than
442        // defaulted: the crossing's default stays the value's own conversion,
443        // which is what an identity arm is handed (#701 step 2).
444        // Which crossings already state a deconstructing `parts` row, so the
445        // decomposition block below adds one only where none was declared.
446        let mut parts_out: std::collections::HashSet<TypeKey> = std::collections::HashSet::new();
447        // Every Kotlin class declaration, and every `convert!`-declared
448        // conversion. The second matters as much as the first: a conversion may
449        // be declared on a type the registry would otherwise read as an arity
450        // layer, and `convert!(Option<Duration> => ..)` means the adapter emits
451        // that optional's conversion itself rather than wrapping `Duration`'s.
452        let mut declared: Vec<(TypeKey, Origin<syn::Type>)> = self
453            .declared_types()
454            .into_iter()
455            .chain(
456                self.convert_decls
457                    .iter()
458                    .map(|d| (d.key().clone(), d.rust_type().clone())),
459            )
460            .collect();
461        declared.sort_by(|a, b| a.0.as_str().cmp(b.0.as_str()));
462        declared.dedup_by(|a, b| a.0 == b.0);
463
464        for (_key, origin) in declared {
465            // The declarator's own tokens, re-parsed: a declared type may be an
466            // alias the model holds as an `Extern`, which carries no reading of
467            // its own to borrow.
468            let Ok(spelled) = syn::parse2::<syn::Type>(origin.declared_spelling()) else {
469                continue;
470            };
471            let Ok(ty) = model.classify(&spelled) else {
472                continue;
473            };
474            // Every declared Kotlin shape is one recipe with no parts today,
475            // including the two that plainly have some: a `data_class` arrives
476            // as separate JNI parameters and is rebuilt inside one generated
477            // function, and a `sealed_class` is a selector plus the live arm's
478            // slots inside another. `Atomic` is what a recipe says about that —
479            // the adapter emits the conversion itself, and how many wire values
480            // that costs is its own business — so it describes the adapter as it
481            // stands rather than as it will be.
482            // A `sealed_class` hands its value out as a tag plus every
483            // alternative's payloads, laid side by side — the deconstructing
484            // twin of the constructing recipe below, and the direction that
485            // actually has no alternative: a sum has no whole-value output
486            // conversion, because a tag plus groups is not one wire.
487            match self.types.get(&ty.key()).map(|c| &c.kind) {
488                Some(DeclaredKind::Sealed(_)) => {
489                    let arms = out_alternatives(model, &ty);
490                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
491                    if !arms.is_empty() {
492                        recipes.declare(ty.clone(), parts(), Deconstructing::Choice { arms });
493                        parts_out.insert(ty.stripped_key());
494                    }
495                }
496                // A `data_class` hands its value out as its fields too, so the
497                // foreign side reassembles the object and nothing builds one on
498                // the Rust side.
499                Some(DeclaredKind::Data) => {
500                    let reaches = out_fields(model, &ty);
501                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
502                    if !reaches.is_empty() {
503                        recipes.declare(
504                            ty.clone(),
505                            parts(),
506                            Deconstructing::Product(Deconstruct::Fields(reaches)),
507                        );
508                        parts_out.insert(ty.stripped_key());
509                    }
510                }
511                _ => {
512                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
513                }
514            }
515            // A value form: `expand_return!(T).fields(fields!(f))` calls `f`
516            // once and hands out the fields of the struct it returns. The one
517            // declaration shape that names its own accessor, and the reason
518            // `Deconstruct::ValueForm` exists.
519            if let Some((func, reaches)) = self.value_form_of(model, registry, &ty) {
520                recipes.declare(
521                    ty.clone(),
522                    parts(),
523                    Deconstructing::Product(Deconstruct::ValueForm {
524                        func,
525                        parts: reaches,
526                    }),
527                );
528                parts_out.insert(ty.stripped_key());
529            }
530            // A `data_class` also has a recipe that says what it is made of, so
531            // its constructing side names which of the two a site takes by
532            // default. See [`parts`] for why that is still `whole`.
533            match self.types.get(&ty.key()).map(|c| &c.kind) {
534                // A struct with no fields is nothing to be made of, so it
535                // states no `parts` recipe — the same condition the deconstructing
536                // side applies, and the reason the row is looked up by name before
537                // it is compiled.
538                Some(DeclaredKind::Data) if !fields_of(model, &ty).is_empty() => {
539                    recipes
540                        .declare_default(ty.clone(), whole(), Constructing::Atomic)
541                        .declare_unasked(ty, parts(), Constructing::Product(Construct::Fields));
542                }
543                // A `sealed_class` has one too, and it is a choice rather than
544                // a product: exactly one alternative is live, every one of them
545                // still crosses, and the tag says which. The arms are the
546                // model's — a recipe states which parts, never what they are.
547                Some(DeclaredKind::Sealed(_)) => {
548                    let arms = alternatives(model, &ty);
549                    recipes.declare_default(ty.clone(), whole(), Constructing::Atomic);
550                    if !arms.is_empty() {
551                        recipes.declare(ty, parts(), Constructing::Choice { arms });
552                    }
553                }
554                _ => {
555                    recipes.declare(ty, whole(), Constructing::Atomic);
556                }
557            }
558        }
559
560        // A type a callback delivers by taking it apart states that as a row,
561        // so the argument's site has a recipe to name. Without one the site
562        // could only be fabricated — a `Bound` no binding answered — which is
563        // what #622's first two attempts produced and why they were withdrawn.
564        //
565        // `Atomic` for the reason the declared-type loop above gives for the
566        // two shapes that plainly have parts: the adapter emits this
567        // conversion itself, and how many wire values that costs is its own
568        // business. The parts are stated by the deconstructor declaration the
569        // registry already resolved into an `UnfoldPlan`, which this row does
570        // not restate — it names the crossing that plan decomposes. Deleting
571        // that second statement is #613 step 5b's, not this row's.
572        //
573        // The source is the plan's own owned core, so a plan keyed under `&T`
574        // and one keyed under `T` state one row; a type that already declared
575        // `parts` — a `data_class`, a `sealed_class`, a value form — keeps it.
576        //
577        // A **whole-element fold** is not one of these and must not earn a row.
578        // `apply_leaf_vec_folds` files a plan for `impl Fn(&[T])` under `&[T]`
579        // whose `source` is the ELEMENT and whose `decon` is `None`: nothing is
580        // taken apart, each element crosses whole through its own converter.
581        // Declaring `parts` off it would state a decomposition of `T` that does
582        // not exist — and, worse, a scalar `T` argument elsewhere in the same
583        // model would then bind to that row and name a fragment compiled under
584        // a different recipe (#623 review). `decon` is the gate the model
585        // already carries for this: `None` only for the whole-element arm.
586        let mut decomposed: BTreeMap<TypeKey, TypeRef> = BTreeMap::new();
587        // EVERY decomposition, not only a callback argument's. A return and a
588        // domain error come apart by the same declaration — `Error` is one
589        // declaration reached from every fallible function in zenoh-flat-jni —
590        // so each states its row too. Without this the rows described a
591        // fraction of what a binding decomposes, which is what 68 entries of
592        // the parity inventory said (#701 step 3).
593        let unfolded = self.unfolded();
594        let every_plan = unfolded
595            .callback_arg_plans
596            .values()
597            .chain(unfolded.unfold_plans.values())
598            .chain(unfolded.error_plans.values());
599        for plan in every_plan {
600            if plan.decon.is_none() {
601                continue;
602            }
603            let key = plan.source.stripped_key();
604            if parts_out.contains(&key) {
605                continue;
606            }
607            decomposed.entry(key).or_insert_with(|| plan.source.clone());
608        }
609        for decl in &self.return_expand_decls {
610            if parts_out.contains(decl.key()) {
611                continue;
612            }
613            if let Some(ty) = decl
614                .key()
615                .ident()
616                .and_then(|ident| model.classify(&syn::parse_quote!(#ident)).ok())
617            {
618                decomposed.entry(decl.key().clone()).or_insert(ty);
619            }
620        }
621        for ty in decomposed.into_values() {
622            // A real row where the declaration can state one. #622 wrote
623            // `Atomic` here because `Reach` could not spell an identity leaf;
624            // it can now (#635), so the row says how the value comes apart
625            // instead of only existing to be selected (#613 step 10).
626            match self.parts_deconstruct(model, registry, &ty) {
627                Some(deconstruct) => {
628                    recipes.declare(ty, parts(), Deconstructing::Product(deconstruct));
629                }
630                None => {
631                    recipes.declare(ty, parts(), Deconstructing::Atomic);
632                }
633            }
634        }
635
636        // A per-function `.expand_return(...)` states a decomposition of its
637        // own, so it gets a row of its own, under a name of its own — the shape
638        // step 2 gave the parameter side (`expand-param@fn#param`), at the
639        // other end. Without it the type's `parts` row and the function's
640        // decomposition sat under one name describing two different things,
641        // which is why the differential could not compare either.
642        for (func, decl) in &self.fn_return_expands {
643            let Some(ty) = decl
644                .key()
645                .ident()
646                .and_then(|ident| model.classify(&syn::parse_quote!(#ident)).ok())
647            else {
648                continue;
649            };
650            match self.decon_of(model, registry, &ty, decl) {
651                Some(deconstruct) => {
652                    recipes.declare(ty, site_row(func), Deconstructing::Product(deconstruct));
653                }
654                None => {
655                    recipes.declare(ty, site_row(func), Deconstructing::Atomic);
656                }
657            }
658        }
659
660        // Every implicit Optional keeps its established whole/default input
661        // row and also offers the allocation-free primitive `pair` row. A
662        // parameter site selects `pair` only after the compiled payload proves
663        // that it is one JNI primitive with no niche, projection, or stage.
664        //
665        // Product-shaped payloads additionally offer `parts`; that row is
666        // selected by the existing flattening bindings below.
667        let mut implicit: BTreeMap<TypeKey, (TypeRef, TypeRef)> = self
668            .implicit_optionals(model)
669            .into_iter()
670            .map(|(key, (outer, inner))| (key, (outer.clone(), inner.clone())))
671            .collect();
672        // A combined constructor expansion represents an inactive arm by
673        // Option-wrapping each required leaf. Those readings are synthesized
674        // after the Flat model was built, so a model walk cannot discover
675        // them. File them in the same table as source-written implicit
676        // Optionals: otherwise a nullable primitive leaf falls back to a boxed
677        // JObject and Rust has to call `intValue()`/`longValue()` across JNI.
678        for leaf in expansion_leaves {
679            let Some(inner) = leaf.optional_inner() else {
680                continue;
681            };
682            if !self.explicitly_declares(leaf) {
683                implicit
684                    .entry(leaf.stripped_key())
685                    .or_insert_with(|| (leaf.clone(), inner.clone()));
686            }
687        }
688        for (outer, _) in implicit.values() {
689            recipes
690                .declare_default(outer.clone(), whole(), Constructing::Optional)
691                .declare(outer.clone(), pair(), Constructing::Optional)
692                .declare_unasked(outer.clone(), parts(), Constructing::Optional);
693        }
694        for (outer, _) in implicit.into_values() {
695            recipes
696                .declare_default(outer.clone(), whole(), Deconstructing::Optional)
697                .declare_unasked(outer.clone(), parts(), Deconstructing::Optional);
698        }
699
700        // A fixed-size array of JNI primitives is one Kotlin `ByteArray` or
701        // `LongArray`, bulk-copied with nothing boxed — one wire value, not a
702        // run this adapter walks. The registry reads `[T; N]` as a run unless
703        // something says otherwise, and this is JniGen saying otherwise for
704        // every array the model holds. Nothing is enumerated twice: a recipe for
705        // a crossing nobody uses is inert.
706        // Keyed by identity in a `BTreeMap`, so each array type is keyed once
707        // and the declaration order is the key order — no sort, no second pass
708        // rebuilding a key it already has, and nothing cloned before the entry
709        // turns out to be new.
710        let mut arrays: BTreeMap<TypeKey, &TypeRef> = BTreeMap::new();
711        for ty in model
712            .elements()
713            .flat_map(element_types)
714            .flat_map(|ty| ty.walk())
715            .filter(|ty| matches!(ty.kind(), prebindgen_registry::flat::TypeKind::Array { .. }))
716        {
717            arrays.entry(ty.key()).or_insert(ty);
718        }
719        for ty in arrays.into_values() {
720            recipes
721                .declare(ty.clone(), whole(), Deconstructing::Atomic)
722                .declare(ty.clone(), whole(), Constructing::Atomic);
723        }
724
725        // The `expand_param!` rows are NOT declared here. They live in the
726        // table `Declarations::expansion_plans` builds for itself, because the
727        // readings they deliver have to be known while the binding declares
728        // itself — before this table can exist. Declaring them here as well
729        // would put a second copy in front of a walk that never reaches them:
730        // no site names one, and both adapters' `choice` refuse the arm shape
731        // they use. They move here when something reads them, in #701's step 3.
732        recipes.build(model)
733    }
734}

736impl Declarations {
737    /// Which recipe each part of a `data_class` takes.
738    ///
739    /// A field that is itself a `data_class` crosses as **its** fields too, so
740    /// the part site asks for [`parts`] rather than letting the field take its
741    /// own default. That is the one thing a site can say that a recipe cannot: the
742    /// same crossing is read one way on its own and another inside a product.
743    ///
744    /// Without it a nested class contributes a single wire and the flattening
745    /// stops one layer down.
746    /// Whether a `data_class` field is itself flattened, or stays one value.
747    ///
748    /// A `data_class` declared `.jobject_input()` crosses Kotlin → Rust as a
749    /// single `JObject` — that is the whole point of the opt-in — so binding
750    /// its parts would flatten a boundary the declaration drew.
751    /// Whether this binding takes a type APART where it appears inside another
752    /// value, rather than crossing it whole.
753    ///
754    /// Three declarations say so: a `data_class`, a `sealed_class`, and a value
755    /// form — `expand_return!(T).fields(..)`, which names its own accessor. The
756    /// proxy for this used to be "declares a `parts` row", which held only
757    /// while an inlined class was the one thing that had one; every
758    /// decomposition states a row now (#701 step 3).
759    fn inlines_parts(
760        &self,
761        model: &Flat,
762        registry: &(impl prebindgen_registry::Conversions + ?Sized),
763        key: &TypeKey,
764    ) -> bool {
765        if self
766            .types
767            .get(key)
768            .is_some_and(|c| c.sum().is_some() || matches!(c.kind, DeclaredKind::Data))
769        {
770            return true;
771        }
772        key.ident()
773            .and_then(|ident| model.classify(&syn::parse_quote!(#ident)).ok())
774            .is_some_and(|ty| self.value_form_of(model, registry, &ty).is_some())
775    }
776
777    fn field_crosses_as_its_fields(&self, ty: &TypeRef) -> bool {
778        // The field's own key, not the stripped one: a `Box<Leaf>` field
779        // crosses as one value through `Box<Leaf>`'s conversion, and only a
780        // parameter spelled that way is peeled back to the class. Binding the
781        // stripped key here would flatten a field the rebuild has no node for.
782        self.types.get(&ty.key()).is_some_and(|c| match c.kind {
783            DeclaredKind::Data => !c.jobject_input,
784            // A `sealed_class` field crosses as a tag plus every alternative's
785            // slots. Whether it *can* is the adapter's answer at compile time,
786            // not a declaration: a payload with no slot form leaves the sum
787            // object-shaped, which is the fragment `Compile::choice` hands
788            // back. So the site asks, and the recipe answers.
789            DeclaredKind::Sealed(_) => true,
790            _ => false,
791        })
792    }
793
794    /// Optional crossings whose payload can use the shared `parts` recipe.
795    ///
796    /// Explicit declarations keep full control of the outer crossing: a
797    /// `convert!(Option<T> => ..)` is an atomic recipe, not an implicit request
798    /// to flatten `T`.
799    fn implicit_optionals<'a>(
800        &self,
801        model: &'a Flat,
802    ) -> BTreeMap<TypeKey, (&'a TypeRef, &'a TypeRef)> {
803        let mut optionals = BTreeMap::new();
804        for ty in model
805            .elements()
806            .flat_map(element_types)
807            .flat_map(|ty| ty.walk())
808        {
809            let Some(inner) = ty.optional_inner() else {
810                continue;
811            };
812            if !self.explicitly_declares(ty) {
813                optionals.entry(ty.stripped_key()).or_insert((ty, inner));
814            }
815        }
816        optionals
817    }
818
819    fn explicitly_declares(&self, ty: &TypeRef) -> bool {
820        self.types.contains_key(&ty.key())
821            || self
822                .convert_decls
823                .iter()
824                .any(|decl| decl.key() == &ty.key() || decl.key() == &ty.stripped_key())
825    }
826
827    pub(crate) fn bindings(
828        &self,
829        model: &Flat,
830        registry: &(impl prebindgen_registry::Conversions + ?Sized),
831        recipes: &prebindgen_registry::recipe::Recipes,
832    ) -> Result<prebindgen_registry::recipe::Bindings, Vec<RecipeError>> {
833        use prebindgen_registry::recipe::{
834            Ask, Bindings, Crossing, Direction, Origin, RecipeName, Site,
835        };
836
837        let mut bound = Bindings::builder();
838
839        // What crosses at a **convert** return is not the value the signature
840        // names. The binding converts it first, and the converted value's type
841        // is the function's own output expansion answer, which no model can
842        // state. Bound here so the walk plans that crossing rather than the
843        // return type, and so the site needs no adapter-private planning.
844        // What an **expanded** return crosses. A return the binding converts or
845        // takes apart does not cross the value its signature names, and only
846        // the binding's own output expansion says what it does cross. Bound
847        // here so the walk plans that crossing, which is what lets the site be
848        // planned once rather than again at the emission site.
849        //
850        // One binding per function, in the same precedence `build_output`
851        // applies: a function may have BOTH an output expansion and an error
852        // plan, and the output expansion wins. Binding each in its own pass
853        // would answer one site twice and be refused as a rebind.
854        let mut owners: std::collections::BTreeSet<String> = Default::default();
855        let unfolded = self.unfolded();
856        owners.extend(unfolded.unfold_plans.keys().map(|f| f.to_string()));
857        owners.extend(unfolded.error_plans.keys().map(|f| f.to_string()));
858        for owner in owners {
859            let func = quote::format_ident!("{owner}");
860            let (crossed, ask) = match unfolded.unfold_plans.get(&func) {
861                // A convert hands over one converted value and takes whatever
862                // row that value has; a decomposition crosses the value it
863                // takes apart and asks for the row that names its parts.
864                Some(plan) => match plan.delivery {
865                    crate::unfold::Delivery::Return => match plan.convert_out_ty.as_ref() {
866                        Some(out) => (out.clone(), Ask::Default),
867                        None => continue,
868                    },
869                    crate::unfold::Delivery::Callback => {
870                        let crossing = Crossing::new(plan.source.clone(), Direction::Deconstruct);
871                        // The row that states THIS function's decomposition: a
872                        // function with its own `.expand_return(...)` takes
873                        // apart differently from the way its type does, and
874                        // this is the binding the return site resolves — the
875                        // one with `Origin::Function`, which no weaker binding
876                        // can displace.
877                        let row = if self.fn_return_expands.iter().any(|(g, _)| *g == func) {
878                            site_row(&func)
879                        } else {
880                            parts()
881                        };
882                        let ask = if recipes.key_of(&crossing.key(), &row).is_some() {
883                            Ask::Recipe(row)
884                        } else {
885                            Ask::Default
886                        };
887                        (plan.source.clone(), ask)
888                    }
889                },
890                // No output expansion, but the binding peels the error: only
891                // the ok arm crosses at the return, because the error is thrown
892                // rather than crossed there.
893                None => {
894                    let Some((ok, _)) = model
895                        .function(&func)
896                        .and_then(|function| function.ret.fallible_parts())
897                    else {
898                        continue;
899                    };
900                    (ok.clone(), Ask::Default)
901                }
902            };
903            bound.bind(
904                Site {
905                    owner: func,
906                    role: prebindgen_registry::recipe::Role::Return,
907                },
908                Crossing::new(crossed, Direction::Deconstruct),
909                ask,
910                Origin::Function,
911            );
912        }
913
914        let mut declared: Vec<TypeKey> = self
915            .types
916            .iter()
917            .filter(|(_, c)| matches!(c.kind, DeclaredKind::Data))
918            .map(|(k, _)| k.clone())
919            .collect();
920        declared.sort_by(|a, b| a.as_str().cmp(b.as_str()));
921
922        // The named Optional row is the representation suitable inside another
923        // composed shape. Where its child has a Product row it delegates to that
924        // row; a scalar child stays on its default row and lets the JNI compiler
925        // choose a niche or a `(present, value)` intermediate from the compiled
926        // child facts. The whole/default Optional row remains untouched.
927        //
928        // Bind both directions: function inputs construct the Rust optional,
929        // while returns and callback arguments deconstruct it.
930        for (outer, inner) in self.implicit_optionals(model).into_values() {
931            for direction in [Direction::Construct, Direction::Deconstruct] {
932                let outer = Crossing::new(outer.clone(), direction);
933                let row = outer.row(parts());
934                let inner = Crossing::new(inner.clone(), direction);
935                // A DECLARED CLASS's parts are inlined here; anything else
936                // crosses whole. "Has a `parts` row" was the proxy for that
937                // while only a declared class had one — every decomposition
938                // states a row now, so the proxy would inline a field the
939                // decomposition crosses whole. The condition says what it
940                // meant (#701 step 3).
941                let inlines = self.inlines_parts(model, registry, &inner.key().ty)
942                    && recipes.key_of(&inner.key(), &parts()).is_some();
943                bound.bind(
944                    Site::part(&row, 0),
945                    inner.clone(),
946                    if inlines {
947                        Ask::Recipe(parts())
948                    } else {
949                        Ask::Default
950                    },
951                    Origin::Part,
952                );
953            }
954        }
955
956        for key in declared {
957            let Some(ident) = key.ident() else { continue };
958            let Some(prebindgen_registry::flat::Type::Struct(s)) = model.declared_type(&ident)
959            else {
960                continue;
961            };
962            let Ok(ty) = model.classify(&syn::parse_quote!(#ident)) else {
963                continue;
964            };
965            let building = Crossing::new(ty.clone(), Direction::Construct);
966            let handing_out = Crossing::new(ty, Direction::Deconstruct);
967            for (index, field) in s.fields.iter().enumerate() {
968                // A field selects `parts` exactly when its crossing declares
969                // that row AND the declaration says this occurrence decomposes.
970                // Direct classes retain the existing `.jobject_input()` and
971                // erased-wrapper boundaries; every implicit Optional selects its
972                // part representation, which decides whether the child is
973                // another composition or one scalar intermediate.
974                if field.ty.optional_inner().is_none()
975                    && !self.field_crosses_as_its_fields(&field.ty)
976                {
977                    continue;
978                }
979                for (of, direction) in [
980                    (&building, Direction::Construct),
981                    (&handing_out, Direction::Deconstruct),
982                ] {
983                    // Handing a value OUT, an optional field asks the question
984                    // of what is inside the layer: a decomposition looks
985                    // through `Option<T>` only where it inlines T's fields, and
986                    // crosses `Option<Summary>` — an optional handle — whole.
987                    // "Declares a `parts` row" was the proxy for that while only
988                    // an inlined class had one; every decomposition states one
989                    // now, so the proxy would look through a layer the
990                    // decomposition crosses whole (#701 step 3).
991                    //
992                    // Building one is unchanged: a `(present, value)` pair is
993                    // how every optional parameter arrives, whatever is inside.
994                    if direction == Direction::Deconstruct
995                        && field
996                            .ty
997                            .optional_inner()
998                            .is_some_and(|inner| !self.field_crosses_as_its_fields(inner))
999                    {
1000                        continue;
1001                    }
1002                    let row = of.row(parts());
1003                    let field_crossing = Crossing::new(field.ty.clone(), direction);
1004                    if recipes.key_of(&field_crossing.key(), &parts()).is_none() {
1005                        continue;
1006                    }
1007                    bound.bind(
1008                        Site::part(&row, index),
1009                        field_crossing,
1010                        Ask::Recipe(parts()),
1011                        Origin::Part,
1012                    );
1013                }
1014            }
1015        }
1016        // Every function whose return the binding takes apart, bound to the
1017        // `parts` recipe of its return type.
1018        //
1019        // This is what a site naming several values is: `Ask::Recipe` already
1020        // lets a site pick a recipe, and a `parts` fragment already occupies
1021        // several wires — the two facts stages 3 and 4 established, meeting.
1022        // The registry needs nothing new to express a decomposed return.
1023        for f in model.functions() {
1024            // The value that crosses, through the layers a decomposition looks
1025            // through: a `&T` return decomposes T, and so does an `Option<T>`
1026            // or a `Vec<T>` — the shape rides the delivery, not the recipe.
1027            let ret = f.ret.borrow_target().unwrap_or(&f.ret);
1028            let ret = ret.optional_inner().unwrap_or(ret);
1029            let ret = ret.sequence_elem().unwrap_or(ret);
1030            let ret = ret.borrow_target().unwrap_or(ret);
1031            let crossing = Crossing::new(ret.clone(), Direction::Deconstruct);
1032            // Only where the type states one. A `sealed_class` always does; a
1033            // `data_class` states one unless a field of it declines, and a
1034            // return whose type states none crosses whole.
1035            if recipes.key_of(&crossing.key(), &parts()).is_none() {
1036                continue;
1037            }
1038            // And only where it is genuinely several. A decomposition that
1039            // yields ONE value takes `Delivery::Return`: the wrapper hands that
1040            // value back through its own conversion rather than through a
1041            // builder, so the return site's crossing is the value's, not this
1042            // type's. Asked of the composition, which is model and declaration
1043            // only and so answerable here — the same property that let one
1044            // composition serve both sides of `resolve`.
1045            if self.out_wire_count(registry, ret) < 2 {
1046                continue;
1047            }
1048            // A function that wrote its own `.expand_return(...)` names the row
1049            // that states IT, not the type's.
1050            bound.bind(
1051                Site {
1052                    owner: f.name.clone(),
1053                    role: prebindgen_registry::recipe::Role::Return,
1054                },
1055                crossing,
1056                Ask::Recipe(parts()),
1057                Origin::Adapter,
1058            );
1059        }
1060
1061        // A callback's `Invoke` recipe owns its argument conversions. Bind a
1062        // model-derived data-class argument to `parts` so the Product fragment
1063        // reaches `Compile::callback` before the trampoline is rendered.
1064        for f in model.functions() {
1065            for param in &f.params {
1066                let prebindgen_registry::flat::TypeKind::Callback { args } =
1067                    param.ty.unwrapped().kind()
1068                else {
1069                    continue;
1070                };
1071                let callback = Crossing::new(param.ty.clone(), Direction::Construct);
1072                let row = callback.row(RecipeName::derived());
1073                for (index, arg) in args.iter().enumerate() {
1074                    let crossed = arg.borrow_target().unwrap_or(arg);
1075                    let core = crossed.optional_inner().unwrap_or(crossed);
1076                    if let Some(element) = core.sequence_elem() {
1077                        let element = element.borrow_target().unwrap_or(element);
1078                        if !self.field_crosses_as_its_fields(element) {
1079                            continue;
1080                        }
1081                        let sequence = Crossing::new(arg.clone(), Direction::Deconstruct);
1082                        let element_crossing =
1083                            Crossing::new(element.borrowed(), Direction::Deconstruct);
1084                        if recipes.key_of(&element_crossing.key(), &parts()).is_none() {
1085                            continue;
1086                        }
1087                        bound.bind(
1088                            Site::part(&sequence.row(RecipeName::derived()), 0),
1089                            element_crossing,
1090                            Ask::Recipe(parts()),
1091                            Origin::Adapter,
1092                        );
1093                        continue;
1094                    }
1095                    // A model-derived data class, or a type whose deconstructor
1096                    // declaration earned the `parts` row above. Both take that
1097                    // row here, so the fragment the trampoline delivers and the
1098                    // one the argument's own `Role::CallbackArg` site names are
1099                    // one fragment rather than two rows over one crossing.
1100                    if !self.field_crosses_as_its_fields(core)
1101                        && !self.unfolded().callback_arg_plans.contains_key(&arg.key())
1102                    {
1103                        continue;
1104                    }
1105                    let crossing = Crossing::new(arg.clone(), Direction::Deconstruct);
1106                    if recipes.key_of(&crossing.key(), &parts()).is_none() {
1107                        continue;
1108                    }
1109                    bound.bind(
1110                        Site::part(&row, index),
1111                        crossing,
1112                        Ask::Recipe(parts()),
1113                        Origin::Adapter,
1114                    );
1115                }
1116            }
1117        }
1118
1119        // Each value a callback delivers is a root site of its own — the
1120        // function-unique `Role::CallbackArg` the registry names, not the
1121        // `Role::Part` above, which is keyed by the callback recipe every
1122        // function with that signature shares.
1123        //
1124        // Bound to `parts` exactly where the argument's crossing states that
1125        // row, which is now every argument the trampoline takes apart: a
1126        // `data_class`, a `sealed_class`, an implicit Optional over one of
1127        // those, or a type whose deconstructor declaration earned the row
1128        // declared above. An argument that crosses whole names nothing here and
1129        // takes its crossing's default, attributed to the adapter.
1130        for f in model.functions() {
1131            for (param, p) in f.params.iter().enumerate() {
1132                let prebindgen_registry::flat::TypeKind::Callback { args } =
1133                    p.ty.unwrapped().kind()
1134                else {
1135                    continue;
1136                };
1137                for (arg, ty) in args.iter().enumerate() {
1138                    let crossing = Crossing::new(ty.clone(), Direction::Deconstruct);
1139                    if recipes.key_of(&crossing.key(), &parts()).is_none() {
1140                        continue;
1141                    }
1142                    bound.bind(
1143                        Site {
1144                            owner: f.name.clone(),
1145                            role: prebindgen_registry::recipe::Role::CallbackArg { param, arg },
1146                        },
1147                        crossing,
1148                        Ask::Recipe(parts()),
1149                        Origin::Adapter,
1150                    );
1151                }
1152            }
1153        }
1154
1155        // Which accessor parts are taken apart further, which is a binding
1156        // rather than a property of the row: a `data_class` states a `parts`
1157        // row and still crosses whole as an accessor's return. The rule is the
1158        // decomposition's own — splice where the part's type has a
1159        // deconstructor DECLARATION — stated here so the leaf view and the
1160        // compiler read one answer (#701 step 3, design decision 3).
1161        for decl in &self.return_expand_decls {
1162            let Some(owner) = registry.reading(decl.key()) else {
1163                continue;
1164            };
1165            let row = Crossing::new(owner, Direction::Deconstruct).row(parts());
1166            for (index, field) in decl.field_list().iter().enumerate() {
1167                let crate::jni::LocalField::Named(func, _) = field else {
1168                    continue;
1169                };
1170                let Some(ret) = model.function(func).map(|f| f.ret.clone()) else {
1171                    continue;
1172                };
1173                // An OPTIONAL part comes apart like any other: it crosses as
1174                // the accessor's return states it, LAYERS INCLUDED, which is
1175                // the crossing both the compiler and the leaf view build for
1176                // it. The binding names `Option<&Report>` and the payload's
1177                // own row is reached through the Optional's value slot,
1178                // exactly as it is for an optional field.
1179                // The bounded delivery peel, not `stripped_key`: that reaches
1180                // the leaf type through ANY nesting, so a part typed
1181                // `Vec<Option<T>>` would be read as the declaration's own `T`
1182                // and spliced — making the row reach its own crossing. The
1183                // inner optional is part of the element, and the part crosses
1184                // as the distinct `Vec<Option<T>>`.
1185                let (_, core) = ret.layer_stack();
1186                let core = core.borrow_target().unwrap_or(core);
1187                // Two conditions, and both are needed. A DECLARATION is what
1188                // says this part comes apart — a type may state a `parts` row
1189                // and still cross whole here, which is the difference this
1190                // binding carries. And the row has to exist to be named: a
1191                // declared type whose row this table does not state yet is
1192                // step 3's remaining work, not a binding to write now.
1193                // Two questions of two types. WHAT COMES APART is the value
1194                // inside the layers, and a DECLARATION is what says it does.
1195                // WHAT CROSSES is the accessor's return as written, and that is
1196                // what the binding names.
1197                if !self
1198                    .return_expand_decls
1199                    .iter()
1200                    .any(|d| *d.key() == core.stripped_key())
1201                {
1202                    continue;
1203                }
1204                let part = Crossing::new(ret.clone(), Direction::Deconstruct);
1205                if recipes.key_of(&part.key(), &parts()).is_none() {
1206                    continue;
1207                }
1208                bound.bind(
1209                    Site::arm_part(&row, None, index),
1210                    part,
1211                    Ask::Recipe(parts()),
1212                    Origin::Adapter,
1213                );
1214            }
1215        }
1216
1217        // A VALUE FORM's parts, bound the same way an accessor's are. The row
1218        // states which field of the returned struct each part reaches; a
1219        // binding states that the part comes apart further and to what. That
1220        // split is design decision 3, and it is what lets a sum-typed field —
1221        // a selector plus one group per alternative, which no `Reach` spells —
1222        // be a `Choice` part rather than a shape smuggled into the row.
1223        // Every value form, under the row that states it: a type's own under
1224        // `parts`, and a function's own under `expand-return@fn`. A per-function
1225        // `.expand_return(..)` declares a real `ValueForm` row of its own, so
1226        // its sum-typed field needs the same binding — without one the row says
1227        // `b : ZValOutcome` where the decomposition says a tag and two groups.
1228        let type_forms = self.return_expand_decls.iter().map(|decl| (decl, parts()));
1229        let fn_forms = self
1230            .fn_return_expands
1231            .iter()
1232            .map(|(func, decl)| (decl, site_row(func)));
1233        for (decl, name) in type_forms.chain(fn_forms) {
1234            let Some(owner) = registry.reading(decl.key()) else {
1235                continue;
1236            };
1237            let Some((func, reaches)) = self.value_form_in(model, registry, &owner, decl) else {
1238                continue;
1239            };
1240            let Some(st) = model.function(&func).map(|f| f.ret.clone()) else {
1241                continue;
1242            };
1243            let st = st.borrow_target().unwrap_or(&st);
1244            let Some(ident) = st.stripped_key().ident() else {
1245                continue;
1246            };
1247            let Some(prebindgen_registry::flat::Type::Struct(st)) = model.declared_type(&ident)
1248            else {
1249                continue;
1250            };
1251            // The DECLARATION's answer for each field, in the same order as
1252            // the reaches: whether this occurrence comes apart. Having a row of
1253            // its own is not enough — `a_whole_value_crossing_uses_only_its_
1254            // flat_structure` is a field whose type is a `data_class` and which
1255            // this form crosses whole — and the record's `decon` is where the
1256            // declaration says which (design decision 3).
1257            let [crate::jni::LocalField::Fields(form)] = decl.field_list() else {
1258                continue;
1259            };
1260            let records = self.lower_value_form(registry, decl.key(), form);
1261            let row = Crossing::new(owner, Direction::Deconstruct).row(name);
1262            for (index, reach) in reaches.iter().enumerate() {
1263                let prebindgen_registry::recipe::Reach::Field(field) = reach else {
1264                    continue;
1265                };
1266                if records
1267                    .get(index)
1268                    .is_none_or(|r| matches!(r.decon, crate::unfold::FieldDecon::Default))
1269                {
1270                    continue;
1271                }
1272                let Some(field) = st.fields.get(*field) else {
1273                    continue;
1274                };
1275                let crossing = Crossing::new(field.ty.clone(), Direction::Deconstruct);
1276                if recipes.key_of(&crossing.key(), &parts()).is_none() {
1277                    continue;
1278                }
1279                bound.bind(
1280                    Site::part(&row, index),
1281                    crossing,
1282                    Ask::Recipe(parts()),
1283                    Origin::Adapter,
1284                );
1285            }
1286        }
1287
1288        bound.build(recipes)
1289    }
