1//! The return side, read off the rows rather than off the decomposition.
2//!
3//! `unfold` still builds every plan the emitters use. This reads the same
4//! decompositions off the `parts` rows the recipe table already carries, and
5//! holds the two to each other — a build fails on any difference (#701 step 3).
6//!
7//! The comparison goes with the older path, once every decomposition this
8//! binding declares has been through it. Until then it is the oracle a fixture
9//! cannot be: it runs over `covertest-kotlin`, `perftest-kotlin`, `emitcheck`
10//! and zenoh-flat-jni, which is the whole declared surface.
11

15    leaf::{Hoist, LeafSource, UnfoldLeaf},
16    recipe::{Reach, Recipes},
17};

21/// What the JVM calls the values a decomposition delivers.
22///
23/// Every answer is one the decomposition writes into a leaf today, and the
24/// comparison below is what holds them to it.
25pub(crate) struct JniUnfold<'a> {
26    decls: &'a Declarations,
27    /// The registry, for the one name lookup that needs a signature.
28    registry: &'a dyn prebindgen_registry::Conversions,
29}

31impl UnfoldPolicy for JniUnfold<'_> {
32    fn selector(&self, source: &TypeRef) -> UnfoldLeaf {
33        UnfoldLeaf {
34            name: "tag".to_string(),
35            path: Vec::new(),
36            // The sum, not the `i32` it crosses as: this is how the emitter
37            // finds the enum to match on. Registered and not required — a sum
38            // has no whole-value output converter (#282).
39            out_ty: source.clone(),
40            identity: false,
41            nullable: false,
42            source: LeafSource::SumTag,
43            groups: Vec::new(),
44        }
45    }
46
47    fn presence(&self, name: &str, source: &TypeRef) -> UnfoldLeaf {
48        UnfoldLeaf {
49            name: format!("{name}__present"),
50            path: Vec::new(),
51            // The OPTIONAL as written: the emitter tests that value to set the
52            // flag, so the reading it names is the one it tests. It is not a
53            // `bool` on this side — what crosses is the presence of a value the
54            // JVM sees as nullable.
55            out_ty: source.clone(),
56            identity: false,
57            nullable: false,
58            source: LeafSource::Presence,
59            groups: Vec::new(),
60        }
61    }
62
63    fn part_name(
64        &self,
65        row: &prebindgen_registry::recipe::RecipeKey,
66        owner: &TypeRef,
67        reach: &Reach,
68        index: usize,
69        field: Option<&syn::Ident>,
70    ) -> String {
71        // Named under the declaration that OWNS the part. A spliced child's
72        // `.name(..)` overrides live on the child's declaration, so asking
73        // under the walk's root would answer with the accessor-derived name
74        // and rename every part a nested decomposition declared.
75        // The VALUE's key: a part reached through a lending accessor arrives
76        // as `&T`, and a row is keyed by what crosses, not by how it was
77        // reached. Keyed by the borrow, the declaration lookup misses and every
78        // nested part falls back to its accessor's ident.
79        let key = owner.borrow_target().unwrap_or(owner).stripped_key();
80        match reach {
81            Reach::Accessor(func) => self.decls.leaf_name_of(row.name(), &key, func, index),
82            // A synthesized `data_class` decomposition names its slots after
83            // the struct's own fields — as the KOTLIN property is named, which
84            // camel-cases and escapes a name that is a keyword there. A
85            // declared decomposition names its parts itself.
86            _ => match field {
87                Some(name) => crate::jni::mangle_kotlin_ident(&crate::jni::kt_snake_to_camel(
88                    &name.to_string(),
89                )),
90                None => self.decls.leaf_name_at(&key, index),
91            },
92        }
93    }
94
95    fn arm_part_name(
96        &self,
97        sum: &TypeRef,
98        variant: &syn::Ident,
99        member: &syn::Member,
100        _index: usize,
101    ) -> String {
102        // The Kotlin variant class's name and the Kotlin property's, which is
103        // what the slot is called on the far side — neither is the Rust name.
104        // Read off the SUM's own declaration: a sum reached as a part of
105        // something else renames its variants for itself.
106        let kotlin = match self
107            .decls
108            .types
109            .get(&sum.stripped_key())
110            .and_then(|c| c.sum())
111        {
112            Some(cfg) => self.decls.sum_variant_class_name(cfg, variant),
113            None => variant.to_string(),
114        };
115        crate::jni::struct_plan::sum_slot_fragment(
116            &kotlin,
117            &crate::jni::struct_plan::sum_field_prop_name(member),
118        )
119    }
120
121    fn value_form_part(
122        &self,
123        owner: &prebindgen_registry::TypeKey,
124        index: usize,
125    ) -> Option<String> {
126        // A value form's parts are named by the declaration that lists them —
127        // the OWNER's, since a form spliced inside another names its own.
128        self.decls.value_form_part_name(self.registry, owner, index)
129    }
130
131    fn identity_name(&self) -> String {
132        // A root identity leaf is `handle`; a nested one takes the name of the
133        // part it was reached through, which the view supplies instead.
134        "handle".to_string()
135    }
136
137    fn nest(&self, outer: &str, inner: &str) -> String {
138        format!("{outer}__{inner}")
139    }
140}

142impl Declarations {
143    /// Read every decomposition back off its row and hold the older path to it.
144    ///
145    /// Temporary, and deleted with `unfold::apply`. A decomposition whose row
146    /// the view does not read yet is skipped rather than failing: the forms it
147    /// refuses are the ones step 3 is still replacing, and the point of this is
148    /// what the two paths say about the forms it DOES read.
149    pub(crate) fn check_unfold_parity(
150        &self,
151        model: &Flat,
152        registry: &dyn prebindgen_registry::Conversions,
153        recipes: &Recipes,
154        bindings: &prebindgen_registry::recipe::Bindings,
155    ) -> Result<(), String> {
156        let folding = Folding::new(recipes, model);
157        // What was NOT compared, and why. Reported rather than discarded: a
158        // skip is work #701's step 3 still owes, and a list of them is how a
159        // reader sees the differential's reach shrink or grow.
160        let mut skipped: Vec<String> = Vec::new();
161        let mut compared = 0usize;
162        let unfolded = self.unfolded();
163        let mut plans: Vec<(String, &crate::unfold::UnfoldPlan)> = unfolded
164            .unfold_plans
165            .iter()
166            .map(|(f, p)| (format!("`{f}`'s return"), p))
167            .chain(
168                unfolded
169                    .error_plans
170                    .iter()
171                    .map(|(f, p)| (format!("`{f}`'s error"), p)),
172            )
173            .chain(
174                unfolded
175                    .callback_arg_plans
176                    .iter()
177                    .map(|(k, p)| (format!("the callback argument `{k}`"), p)),
178            )
179            .collect();
180        // Sorted, so a binding with two differences reports the same one every
181        // time rather than whichever the hash order reached first.
182        plans.sort_by(|(a, _), (b, _)| a.cmp(b));
183
184        for (what, plan) in plans {
185            // EVERY decomposition lands in exactly one bucket: compared, or
186            // named in the skip list. A `continue` before the list is a
187            // decomposition leaving the comparison with nothing to say so, and
188            // the pinned set then covers a part of the surface rather than all
189            // of it — which is this check disabling itself one level up from
190            // where it looks.
191            //
192            // A whole-element fold takes nothing apart: the emitter folds over
193            // the run and each element crosses through its own converter. That
194            // is a decomposition with nothing IN it rather than one the rows do
195            // not reach, so it is compared like any other — against the
196            // crossing's own derived row, which states the run as a `Sequence`
197            // layer and yields the empty leaf list the plan carries.
198            //
199            // A per-function `.expand_return(...)` states a decomposition of
200            // its own, under a row of its own; the type's `parts` answers the
201            // rest.
202            let row = match &plan.decon {
203                Some(prebindgen_registry::leaf::DeconId::PerFn(_, func)) => {
204                    crate::jni::recipes::site_row(&quote::format_ident!("{func}"))
205                }
206                Some(prebindgen_registry::leaf::DeconId::Default(_)) => {
207                    crate::jni::recipes::parts()
208                }
209                // Whatever this crossing's DEFAULT row is: the element of a
210                // whole-element fold reads as the row it already has, which is
211                // the declared class's `whole` where it is one and the derived
212                // row where nothing declared it.
213                None => prebindgen_registry::recipe::RecipeName::derived(),
214            };
215            // A `sealed_class` that ALSO carries an `expand_return!` states two
216            // decompositions under one row name: the sum's arms, declared by
217            // the class, and the fields, declared by the author. The row table
218            // keeps the first and the decomposition used the second, so the two
219            // describe different things and comparing them says nothing. One
220            // row name for two meanings is step 3's to resolve.
221            if self.is_sum(&plan.source) && self.declares_return_expand(&plan.source) {
222                skipped.push(format!("{what}: sum-and-expand-return"));
223                continue;
224            }
225            let policy = JniUnfold {
226                decls: self,
227                registry,
228            };
229
230            // The reading as the decomposition holds it: a plan reached by
231            // reference is lent the value, and its root hands out a borrow
232            // rather than the value itself. `plan.source` is the owned core, so
233            // the borrow has to be put back for the walk to see it.
234            let reading = match plan.by_ref {
235                true => plan.source.borrowed(),
236                false => plan.source.clone(),
237            };
238            let row = match plan.decon {
239                Some(_) => row,
240                None => recipes
241                    .recipe(&prebindgen_registry::recipe::Crossing::new(
242                        reading.clone(),
243                        prebindgen_registry::recipe::Direction::Deconstruct,
244                    ))
245                    .0
246                    .name()
247                    .clone(),
248            };
249            let (leaves, hoists, coverage) = match folding.unfold(&policy, bindings, &reading, &row)
250            {
251                Ok(read) => read,
252                // A shape the view does not represent yet — no row states
253                // this decomposition, or it states one in a form still
254                // being replaced. Step 3's remaining work rather than a
255                // disagreement.
256                Err(e) if e.is_not_yet_readable() => {
257                    skipped.push(format!("{what}: {}", e.code()));
258                    continue;
259                }
260                // Anything else is a row that is WRONG: two leaves of one
261                // name, two identities, a cycle, an accessor or an
262                // alternative the model does not have. Skipping those would
263                // let a bad row disable the check that found it.
264                Err(e) => return Err(format!("{what} has a row that cannot be read: {e}")),
265            };
266            // The walk says where it stopped; nothing is inferred from the
267            // SIZE of what came back. A measure that skips whenever one side is
268            // smaller cannot tell a shape the rows do not state yet from a
269            // defect that made the read smaller — and a defect of exactly that
270            // kind is what the previous commit fixed, invisible because it made
271            // one side smaller.
272            //
273            // So: an incomplete read is skipped and named, and a complete one
274            // is compared in full, with no exemptions.
275            //
276            // The one place a stop is the ANSWER rather than a shortfall is a
277            // plan that states no decomposition. `Atomic` there says "the
278            // adapter emits this conversion itself", which is exactly what a
279            // whole-element fold does, and the declaration states no parts for
280            // it to have compared less than. Both leaf lists are still compared
281            // in full below.
282            if !coverage.is_complete() && plan.decon.is_some() {
283                skipped.push(format!("{what}: {}", coverage.unread().join(", ")));
284                continue;
285            }
286            // What THIS adapter has not lowered to a binding yet. The walk read
287            // what the rows say; whether the older statement says more is a
288            // question about the declarations, which only this side can answer
289            // — and answering it here names the exact declaration rather than
290            // inferring from what came back.
291            if let Some(unlowered) = self.unlowered_parts(model, registry, recipes, &plan.source) {
292                skipped.push(format!("{what}: {unlowered}"));
293                continue;
294            }
295            compared += 1;
296            let from_row = describe(&leaves, &hoists);
297            let from_decl = describe(&plan.leaves, &plan.hoists);
298            if from_row != from_decl {
299                return Err(format!(
300                    "{what} reads back differently from its row.\nfrom the row:\n{from_row}\n\
301                     from the declarations:\n{from_decl}"
302                ));
303            }
304        }
305        // A differential that quietly compares less than it did is the failure
306        // this whole check exists to avoid, so what it did NOT compare is
307        // pinned rather than printed. The binding states the set it expects,
308        // and any growth fails the build naming what appeared.
309        //
310        // Shrinking fails too, and deliberately: each entry is a binding #701's
311        // step 3 still owes, and removing one from the expectation is how that
312        // work is recorded as done.
313        // How many decompositions the differential actually compared. The skip
314        // set names what left the comparison; this names what stayed, and one
315        // leaving the POPULATION shows up in neither.
316        if let Some(expected) = self.parity_compared {
317            if compared != expected {
318                return Err(format!(
319                    "the differential compared {compared} decomposition(s), and this binding \
320                     states {expected}. State the current count with \
321                     `.expect_parity_compared({compared})`."
322                ));
323            }
324        }
325        skipped.sort();
326        // Only where the binding stated one: a fixture exercising a single
327        // shape has no reach worth holding, and the four bindings this
328        // workspace builds all state theirs.
329        let Some(expected) = self.parity_skips.clone() else {
330            return Ok(());
331        };
332        if skipped != expected {
333            let only = |a: &[String], b: &[String]| {
334                a.iter()
335                    .filter(|x| !b.contains(x))
336                    .cloned()
337                    .collect::<Vec<_>>()
338                    .join("\n  ")
339            };
340            return Err(format!(
341                "the decompositions NOT compared against their rows have changed.\n                 newly skipped:\n  {}\nno longer skipped:\n  {}\n\n                 state the current set with `.expect_parity_skips([..])`:\n  {}",
342                only(&skipped, &expected),
343                only(&expected, &skipped),
344                skipped.join("\n  ")
345            ));
346        }
347        Ok(())
348    }
349}

351impl Declarations {
352    /// Which part of this type's decomposition declaration has no binding yet,
353    /// if any.
354    ///
355    /// `Declarations::bindings` writes a part binding for an accessor part
356    /// whose type carries a declaration of its own. One shape is left: a
357    /// declared type whose `parts` row the table does not state, which is the
358    /// same unfinished work seen from the other side.
359    ///
360    /// Named here rather than guessed at from the leaves, because this is where
361    /// the declaration is.
362    fn unlowered_parts(
363        &self,
364        model: &Flat,
365        _registry: &dyn prebindgen_registry::Conversions,
366        recipes: &Recipes,
367        source: &prebindgen_registry::flat::TypeRef,
368    ) -> Option<&'static str> {
369        let key = source.stripped_key();
370        let decl = self.return_expand_decls.iter().find(|d| *d.key() == key)?;
371        // A VALUE FORM is represented completely by `Deconstruct::ValueForm`
372        // now — the call, the hoist, each field's reach and leaf, a field the
373        // declaration overrode, and whether the form was handed the value and
374        // moves its parts out. Nothing about one is unlowered.
375        if matches!(decl.field_list(), [crate::jni::LocalField::Fields(_)]) {
376            return None;
377        }
378        for field in decl.field_list() {
379            let crate::jni::LocalField::Named(func, _) = field else {
380                continue;
381            };
382            let Some(ret) = model.function(func).map(|f| f.ret.clone()) else {
383                continue;
384            };
385            let core = ret.optional_inner().unwrap_or(&ret);
386            let core = core.borrow_target().unwrap_or(core);
387            let Some(_child) = self
388                .return_expand_decls
389                .iter()
390                .find(|d| *d.key() == core.stripped_key())
391            else {
392                continue;
393            };
394            // `bindings` writes a binding only where the row it would name
395            // exists. A declared type whose `parts` row this table does not
396            // state yet is the same unfinished work, seen from the other side.
397            let part = prebindgen_registry::recipe::Crossing::new(
398                core.clone(),
399                prebindgen_registry::recipe::Direction::Deconstruct,
400            );
401            if recipes
402                .key_of(&part.key(), &crate::jni::recipes::parts())
403                .is_none()
404            {
405                return Some("part-without-parts-row");
406            }
407        }
408        None
409    }
410}

412/// One decomposition, flattened far enough that two can be compared.
413fn describe(leaves: &[UnfoldLeaf], hoists: &[Hoist]) -> String {
414    use std::fmt::Write;
415    let path_of = |steps: &[prebindgen_registry::leaf::PathStep]| {
416        steps
417            .iter()
418            .map(|step| {
419                format!(
420                    "{}{}",
421                    step.ident(),
422                    if step.is_optional() { "?" } else { "" }
423                )
424            })
425            .collect::<Vec<_>>()
426            .join(".")
427    };
428    let mut out = String::new();
429    for leaf in leaves {
430        let _ = writeln!(
431            out,
432            "  {} : {} path=[{}] identity={} nullable={} source={} groups={:?}",
433            leaf.name,
434            leaf.out_ty.key(),
435            path_of(&leaf.path),
436            leaf.identity,
437            leaf.nullable,
438            source_of(&leaf.source),
439            leaf.groups
440        );
441    }
442    for hoist in hoists {
443        let _ = writeln!(
444            out,
445            "  hoist [{}] consuming={}",
446            path_of(&hoist.prefix),
447            hoist.consuming
448        );
449    }
450    out
451}

453/// How a leaf is reached, for the comparison.
454///
455/// A nested sum's payload is deliberately not distinguished from a plain reach.
456/// The two paths record it differently — `synth_value_struct_leaves` maps a
457/// rebased payload wire to `Reach`, dropping the alternative, where the view
458/// keeps `VariantField` — and which of the two is right is not this check's
459/// question. What the emitter reads off such a leaf is its `groups`, which both
460/// paths agree on and this compares.
461fn source_of(source: &LeafSource) -> String {
462    match source {
463        LeafSource::VariantField { .. } | LeafSource::Reach => "reach".to_string(),
464        LeafSource::SumTag => "tag".to_string(),
465        LeafSource::Presence => "present".to_string(),
466    }
