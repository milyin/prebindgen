1//! How a value crosses the JNI boundary, in both languages and both
2//! directions.
3//!
4//! Every value crosses as a flat list of [`Leaf`]s — JNI primitives,
5//! strings, primitive arrays, object arrays — and generated Rust never reads
6//! a Kotlin object's fields. One layer of a type is read with
7//! [`Plan::shape`]; everything here recurses on it, so the leaf list and the
8//! four conversions cannot drift apart:
9//!
10//! | | into Rust | out of Rust |
11//! |---|---|---|
12//! | leaves | [`Plan::leaves`] (`Dir::In`) | [`Plan::leaves`] (`Dir::Out`) |
13//! | Kotlin | [`Plan::kt_encode`]: value → leaf expressions | [`Plan::kt_decode`]: leaf expressions → value |
14//! | Rust | [`Plan::rs_decode`]: wires → value | [`Plan::rs_encode`]: value → wires |
15//!
16//! A data class is the concatenation of its fields; an optional value is a
17//! presence flag beside its inner leaves (a single object leaf is simply
18//! nullable, and a single primitive leaving Rust is boxed); a sum is a tag
19//! plus every alternative's fields; a sequence is a count plus one array per
20//! element leaf.
21//!
22//! [`deliver`] hands a value to Kotlin as a list of parameters (a builder's,
23//! a callback's, an error handler's); [`select`] builds a parameter from a
24//! choice of constructors.
25

38fn bare_declared_name(ty: &TypeRef) -> Option<&str> {
39    match ty.kind() {
40        TypeKind::Named { id, .. } => Some(&id.name),
41        TypeKind::String => Some("String"),
42        _ => None,
43    }
44}
45
46fn declared_name(ty: &TypeRef) -> Option<&str> {
47    let core = match ty.kind() {
48        TypeKind::Ref { inner, .. } => inner.as_ref(),
49        _ => ty,
50    };
51    bare_declared_name(core)
52}

68pub(crate) fn scalar_prim(k: ScalarKind) -> Prim {
69    match k {
70        ScalarKind::Bool => Prim::Z,
71        ScalarKind::I8 => Prim::B,
72        ScalarKind::I16 => Prim::S,
73        ScalarKind::I32 | ScalarKind::U8 | ScalarKind::U16 => Prim::I,
74        ScalarKind::I64
75        | ScalarKind::Isize
76        | ScalarKind::U32
77        | ScalarKind::U64
78        | ScalarKind::Usize => Prim::J,
79        ScalarKind::F32 => Prim::F,
80        ScalarKind::F64 => Prim::D,
81    }
82}
83
84fn scalar_kt(k: ScalarKind) -> &'static str {
85    match k {
86        ScalarKind::Bool => "Boolean",
87        ScalarKind::I8 => "Byte",
88        ScalarKind::I16 => "Short",
89        ScalarKind::I32 | ScalarKind::U8 | ScalarKind::U16 => "Int",
90        ScalarKind::I64 | ScalarKind::Isize | ScalarKind::U32 | ScalarKind::Usize => "Long",
91        ScalarKind::U64 => "ULong",
92        ScalarKind::F32 => "Float",
93        ScalarKind::F64 => "Double",
94    }
95}

97/// The primitive an array element crosses as: unsigned elements keep their
98/// bits in the same-width signed primitive.
99pub(crate) fn array_prim(k: ScalarKind) -> Prim {
100    match k {
101        ScalarKind::U8 => Prim::B,
102        ScalarKind::U16 => Prim::S,
103        ScalarKind::U32 => Prim::I,
104        ScalarKind::U64 | ScalarKind::Usize => Prim::J,
105        other => scalar_prim(other),
106    }
107}

141/// A Kotlin identifier, escaped when it is a keyword.
142pub(crate) fn kt_ident(s: &str) -> String {
143    if KT_KEYWORDS.contains(&s) {
144        format!("`{s}`")
145    } else {
146        s.to_string()
147    }
148}

150/// The Kotlin property name of a field: camelCase, or `v<i>` positionally.
151pub(crate) fn kt_prop(f: &Field) -> String {
152    match &f.name {
153        Some(n) => kt_ident(&names::camel(&names::bare(n))),
154        None => format!("v{}", f.index),
155    }
156}

158/// The leaf path segment of a field.
159pub(crate) fn field_seg(f: &Field) -> String {
160    match &f.name {
161        Some(n) => names::bare(n),
162        None => format!("v{}", f.index),
163    }
164}

166/// The leaf path segment of a sum alternative.
167pub(crate) fn alt_seg(alt: &Alternative) -> String {
168    names::snake(&names::bare(&alt.name))
169}

171/// The Kotlin class name of a sum alternative.
172pub(crate) fn alt_kt_name(class: &Class, alt: &Alternative) -> String {
173    let rust = names::bare(&alt.name);
174    if let ClassKind::Sealed { renames } = &class.kind {
175        if let Some(n) = renames.get(&rust) {
176            return n.clone();
177        }
178    }
179    match rust.as_str() {
180        "None" => "None_".to_string(),
181        _ => rust,
182    }
183}

185/// A Rust ident for a leaf at `path` under `root`.
186pub(crate) fn leaf_ident(root: &str, path: &str) -> syn::Ident {
187    names::ident(&join(root, path))
188}

191    /// One layer of `ty`. A borrow of a value the JNI side rebuilds (a data
192    /// class, a converted type) is a [`Shape::Ref`] over it: the value is
193    /// decoded and lent. A handle keeps its [`Access`].
194    pub(crate) fn shape<'t>(&self, ty: &'t TypeRef) -> Res<Shape<'t, &Setting>> {
195        let s = shape(ty, |candidate| {
196            bare_declared_name(candidate).and_then(|name| self.types.get(name))
197        })
198        .map_err(|e| crate::Error(format!("`{ty}`: {e}")))?;
199        Ok(match s {
200            Shape::Declared {
201                ty, declaration, ..
202            } if Access::of(ty) != Access::Owned
203                && !matches!(declaration, Setting::Class(c) if matches!(c.kind, ClassKind::Ptr { .. })) =>
204            {
205                Shape::Ref {
206                    inner: ty.borrow_target().expect("a borrow"),
207                    access: Access::of(ty),
208                }
209            }
210            Shape::Str {
211                kind: TextKind::Str,
212                access: Access::Owned,
213            }
214            | Shape::Seq {
215                kind: SequenceKind::Slice,
216                access: Access::Owned,
217                ..
218            } => {
219                return err(format!(
220                    "`{ty}`: an unsized value cannot cross by value; borrow it or use Cow"
221                ))
222            }
223            Shape::Str {
224                kind: TextKind::Str,
225                access: Access::Exclusive,
226            }
227            | Shape::Seq {
228                kind: SequenceKind::Slice,
229                access: Access::Exclusive,
230                ..
231            } => {
232                return err(format!(
233                    "`{ty}`: mutable unsized values have no JVM representation"
234                ))
235            }
236            Shape::Undeclared(name) => {
237                return err(format!(
238                    "`{name}` is not declared as a class or a conversion"
239                ))
240            }
241            Shape::Result { .. } => {
242                return err(format!("`{ty}`: `Result` crosses only as a result"))
243            }
244            Shape::Out(_) => return err(format!("`{ty}` has no JVM representation")),
245            s => s,
246        })
247    }

270    /// `(name, discriminant literal)` for each value of an enum class.
271    pub(crate) fn enum_arms(&self, class: &Class) -> Res<Vec<(syn::Ident, proc_macro2::Literal)>> {
272        let Some(prebindgen_flat::flat::Type::Enum(e)) = self.flat.declared_type(&class.rust)
273        else {
274            return err(format!("`{}` is not a fieldless enum", class.rust));
275        };
276        let values = e.discriminant_values().map_err(|v| {
277            crate::Error(format!(
278                "`{}::{v}`: discriminant is not a literal",
279                class.rust
280            ))
281        })?;
282        Ok(values
283            .into_iter()
284            .map(|(n, d)| (n.clone(), proc_macro2::Literal::i32_unsuffixed(d as i32)))
285            .collect())
286    }

290    /// The leaves `ty` crosses on.
291    pub(crate) fn leaves(&self, ty: &TypeRef, dir: Dir) -> Res<Vec<Leaf>> {
292        Ok(match self.shape(ty)? {
293            Shape::Unit => Vec::new(),
294            Shape::Scalar(k) => vec![Leaf::new(LeafTy::Prim(scalar_prim(k)))],
295            Shape::Str { .. } => vec![Leaf::new(LeafTy::String)],
296            Shape::Seq { elem, .. } if is_u8(elem) => vec![Leaf::new(LeafTy::PrimArray(Prim::B))],
297            Shape::Array { elem, .. } => match elem.kind() {
298                prebindgen_flat::flat::TypeKind::Scalar(k) => {
299                    vec![Leaf::new(LeafTy::PrimArray(array_prim(*k)))]
300                }
301                _ => return err(format!("`{ty}`: only arrays of primitives cross")),
302            },
303            Shape::Declared { declaration, .. } => match declaration {
304                Setting::Converted(c) => self.leaves(self.conv_repr(c), dir)?,
305                Setting::Class(c) => match &c.kind {
306                    ClassKind::Ptr { .. } => vec![Leaf::new(LeafTy::Prim(Prim::J))],
307                    ClassKind::Enum => vec![Leaf::new(LeafTy::Prim(Prim::I))],
308                    ClassKind::Data => {
309                        let mut out = Vec::new();
310                        for f in &self.struct_of(c)?.fields {
311                            let seg = field_seg(f);
312                            out.extend(self.leaves(&f.ty, dir)?.into_iter().map(|l| l.under(&seg)));
313                        }
314                        out
315                    }
316                    ClassKind::Sealed { .. } => {
317                        let mut out = vec![Leaf::new(LeafTy::Prim(Prim::I)).under("_tag")];
318                        for alt in &self.variant_of(c)?.alternatives {
319                            let aseg = alt_seg(alt);
320                            for f in &alt.fields {
321                                let seg = join(&aseg, &field_seg(f));
322                                out.extend(self.leaves(&f.ty, dir)?.into_iter().map(|mut l| {
323                                    l.nullable |= l.is_obj();
324                                    l.under(&seg)
325                                }));
326                            }
327                        }
328                        out
329                    }
330                },
331            },
332            Shape::Option(inner) => {
333                let inner_leaves = self.leaves(inner, dir)?;
334                if inner_leaves.len() == 1 && inner_leaves[0].is_obj() {
335                    let mut l = inner_leaves.into_iter().next().unwrap();
336                    l.nullable = true;
337                    vec![l]
338                } else if dir == Dir::Out && inner_leaves.len() == 1 {
339                    let p = inner_leaves[0].prim().expect("a primitive leaf");
340                    vec![Leaf::new(LeafTy::Boxed(p))]
341                } else {
342                    let mut out = vec![Leaf::new(LeafTy::Prim(Prim::Z)).under("_present")];
343                    out.extend(inner_leaves.into_iter().map(|mut l| {
344                        l.nullable |= l.is_obj();
345                        l
346                    }));
347                    out
348                }
349            }
350            Shape::Seq { elem, .. } => {
351                let mut out = vec![Leaf::new(LeafTy::Prim(Prim::I)).under("_n")];
352                out.extend(self.leaves(elem, dir)?.iter().map(Leaf::column));
353                out
354            }
355            Shape::Ref { inner, .. } | Shape::Boxed(inner) => self.leaves(inner, dir)?,
356            Shape::Cow(inner) => self.leaves(&cow_view(inner), dir)?,
357            Shape::Callback(_) => vec![Leaf::new(LeafTy::Callback(self.callback_raw_fqn(ty)?))],
358            Shape::Undeclared(_) | Shape::Result { .. } | Shape::Out(_) => {
359                unreachable!("refused by shape")
360            }
361        })
362    }

366    /// The Kotlin type of a value of `ty`.
367    pub(crate) fn kt_type(&self, ty: &TypeRef) -> Res<String> {
368        Ok(match self.shape(ty)? {
369            Shape::Unit => "Unit".to_string(),
370            Shape::Scalar(k) => scalar_kt(k).to_string(),
371            Shape::Str { .. } => "String".to_string(),
372            Shape::Seq { elem, .. } if is_u8(elem) => "ByteArray".to_string(),
373            Shape::Array { elem, .. } => match elem.kind() {
374                prebindgen_flat::flat::TypeKind::Scalar(k) => array_prim(*k).kt_array(),
375                _ => return err(format!("`{ty}`: only arrays of primitives cross")),
376            },
377            Shape::Declared { declaration, .. } => match declaration {
378                Setting::Class(c) => c.fqn(),
379                Setting::Converted(c) => self.kt_type(self.conv_repr(c))?,
380            },
381            Shape::Option(inner) => format!("{}?", self.kt_type(inner)?),
382            Shape::Seq { elem, .. } => format!("List<{}>", self.kt_type(elem)?),
383            Shape::Ref { inner, .. } | Shape::Boxed(inner) => self.kt_type(inner)?,
384            Shape::Cow(inner) => self.kt_type(&cow_view(inner))?,
385            Shape::Callback(_) => self.callback_fqn(ty)?,
386            Shape::Undeclared(_) | Shape::Result { .. } | Shape::Out(_) => {
387                unreachable!("refused by shape")
388            }
389        })
390    }

392    /// Whether a value of `ty`, held in Kotlin, owns a native handle (and so
393    /// must be closed).
394    pub(crate) fn owns_handle(&self, ty: &TypeRef) -> Res<bool> {
395        Ok(match self.shape(ty)? {
396            Shape::Declared {
397                declaration: Setting::Class(c),
398                ..
399            } => match &c.kind {
400                ClassKind::Ptr { .. } => Access::of(ty) != Access::Exclusive,
401                ClassKind::Data => {
402                    let mut any = false;
403                    for f in &self.struct_of(c)?.fields {
404                        any |= self.owns_handle(&f.ty)?;
405                    }
406                    any
407                }
408                ClassKind::Sealed { .. } => {
409                    let mut any = false;
410                    for a in &self.variant_of(c)?.alternatives {
411                        for f in &a.fields {
412                            any |= self.owns_handle(&f.ty)?;
413                        }
414                    }
415                    any
416                }
417                ClassKind::Enum => false,
418            },
419            Shape::Option(t)
420            | Shape::Seq { elem: t, .. }
421            | Shape::Ref { inner: t, .. }
422            | Shape::Boxed(t) => self.owns_handle(t)?,
423            Shape::Cow(t) => self.owns_handle(&cow_view(t))?,
424            _ => false,
425        })
426    }

429/// Whether `ty` is `u8`: a sequence of it is a byte array.
430pub(crate) fn is_u8(ty: &TypeRef) -> bool {
431    matches!(
432        ty.kind(),
433        prebindgen_flat::flat::TypeKind::Scalar(ScalarKind::U8)
434    )
435}

437/// A Cow's unsized target is viewed through a shared borrow when choosing its
438/// Kotlin type and leaves. This does not admit bare unsized boundary values.
439pub(crate) fn cow_view(inner: &TypeRef) -> std::borrow::Cow<'_, TypeRef> {
440    use prebindgen_flat::flat::TypeKind;
441    match inner.kind() {
442        TypeKind::Str | TypeKind::Slice(_) => std::borrow::Cow::Owned(inner.borrowed()),
443        _ => std::borrow::Cow::Borrowed(inner),
444    }
