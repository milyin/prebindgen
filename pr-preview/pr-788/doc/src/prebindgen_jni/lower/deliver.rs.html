1//! Deliveries: a value handed to Kotlin as a *list of parameters* — a
2//! builder's, a callback's, an error handler's.
3//!
4//! A value delivers as one typed parameter unless its type has an output
5//! expansion (`expand_return!`): then it delivers as its fields, each field
6//! itself a delivery — spliced when the field's type has an expansion of its
7//! own, inlined when it comes from a value form (`fields!`) and is a data
8//! class, typed otherwise. An optional value with an expansion delivers its
9//! fields gated by one presence flag: every one of them `null` when absent.
10

26/// One Kotlin-visible parameter of a delivery.
27#[derive(Clone, Debug)]
28pub(crate) struct DParam {
29    pub name: String,
30    pub kt: String,
31    /// Builds the parameter's value from the raw leaves (named by
32    /// [`raw_name`]).
33    pub decode: String,
34    /// Closes the value after a callback returns, when it owns handles.
35    pub close: Option<String>,
36}

38/// A delivered value: its parameters, its raw leaves and the Rust encoding.
39pub(crate) struct Delivery {
40    pub params: Vec<DParam>,
41    pub leaves: Vec<Leaf>,
42    pub output: Output<Leaf>,
43}

45/// The Kotlin name of a raw leaf parameter.
46pub(crate) fn raw_name(l: &Leaf) -> String {
47    let n = names::camel(&l.name);
48    kt_ident(if n.is_empty() { "value" } else { &n })
49}

51fn gate(gates: &[String], decode: String) -> String {
52    if gates.is_empty() {
53        decode
54    } else {
55        format!("(if ({}) {decode} else null)", gates.join(" && "))
56    }
57}

59fn nullable(kt: String, gated: bool) -> String {
60    if gated && !kt.ends_with('?') {
61        format!("{kt}?")
62    } else {
63        kt
64    }
65}

68    /// The expansion that applies to a value of `ty`: the explicit one, else
69    /// the type's own; `None` when it would only hand over the value itself.
70    pub(crate) fn expansion<'e>(
71        &'e self,
72        ty: &TypeRef,
73        explicit: Option<&'e ExpandReturnDecl>,
74    ) -> Option<&'e ExpandReturnDecl> {
75        let e = match explicit {
76            Some(e) => e,
77            None => match ty.kind() {
78                TypeKind::Named { id, .. } => self.ret_exp.get(&id.name)?,
79                _ => return None,
80            },
81        };
82        let identity = e.fields.len() == 1 && matches!(e.fields[0], ReturnField::Handle);
83        (!identity).then_some(e)
84    }

86    /// Deliver `value` (of `ty`) as parameters named under `name`, with raw
87    /// leaves under `root`. `gates` are the presence flags every parameter
88    /// hangs on; `inline` spreads a data class's fields.
89    #[allow(clippy::too_many_arguments)]
90    pub(crate) fn deliver(
91        &self,
92        ty: &TypeRef,
93        value: TokenStream,
94        root: &str,
95        name: &str,
96        explicit: Option<&ExpandReturnDecl>,
97        gates: &[String],
98        inline: bool,
99        depth: usize,
100    ) -> Res<Delivery> {
101        // An expansion with no fields drops the value: nothing crosses.
102        if explicit.is_some_and(|e| e.fields.is_empty()) {
103            return Ok(Delivery {
104                params: Vec::new(),
105                leaves: Vec::new(),
106                output: Output::unit(ty, value),
107            });
108        }
109        let (core, v, borrowed) = peel(ty, value.clone());
110        // An optional value whose inner type expands: one presence gate.
111        if let TypeKind::Optional(inner) = core.kind() {
112            let (inner_core, _, _) = peel(inner, quote!(__unused));
113            if self.expansion(inner_core, explicit).is_some() {
114                let present = Leaf::new(LeafTy::Prim(Prim::Z)).under(&join(root, "_present"));
115                let x = format_ident!("__x{}", depth);
116                let x_value = if borrowed { quote!(&#x) } else { quote!(#x) };
117                let mut g = gates.to_vec();
118                g.push(raw_name(&present));
119                let inner_d =
120                    self.deliver(inner, x_value, root, name, explicit, &g, false, depth + 1)?;
121                let defaults = inner_d.leaves.iter().map(Leaf::rs_default);
122                let bind = inner_d.output.bind();
123                let names: Vec<syn::Ident> = inner_d
124                    .output
125                    .wires()
126                    .iter()
127                    .map(|w| w.name.clone())
128                    .collect();
129                let presence = Wire::new(leaf_ident(root, "_present"), present.clone());
130                let scrutinee = if borrowed { quote!(#v.as_ref()) } else { v };
131                let output = Output {
132                    expr: quote!(match #scrutinee {
133                        ::core::option::Option::Some(#x) => { #bind (1u8, #(#names),*) }
134                        ::core::option::Option::None => (0u8, #(#defaults),*),
135                    }),
136                    fallible: inner_d.output.fallible,
137                    form: Form {
138                        ty: core.clone(),
139                        kind: FormKind::Optional {
140                            presence: Some(presence),
141                            inner: Box::new(inner_d.output.form),
142                        },
143                    },
144                };
145                let mut leaves = vec![present];
146                leaves.extend(inner_d.leaves);
147                return Ok(Delivery {
148                    params: inner_d.params,
149                    leaves,
150                    output,
151                });
152            }
153        }
154        if let Some(e) = self.expansion(core, explicit) {
155            return self.expanded(core, e, v, borrowed, root, name, gates, depth);
156        }
157        if inline {
158            if let Shape::Declared {
159                declaration: Setting::Class(c),
160                ..
161            } = self.shape(core)?
162            {
163                if let ClassKind::Data = c.kind {
164                    let v = if borrowed {
165                        quote!(::core::clone::Clone::clone(#v))
166                    } else {
167                        v
168                    };
169                    return self.fields_of(self.struct_of(c)?, None, v, root, name, gates, depth);
170                }
171            }
172        }
173        self.plain(ty, value, root, name, gates, depth)
174    }

176    /// Whether `ty` is an enum (`Some(false)`) or an optional enum
177    /// (`Some(true)`), under borrows and boxes.
178    fn enum_value(&self, ty: &TypeRef) -> Res<Option<bool>> {
179        let is_enum = |t: &TypeRef| {
180            let (core, _, _) = peel(t, TokenStream::new());
181            matches!(core.kind(), TypeKind::Named { id, .. }
182                if matches!(self.class(&id.name).map(|c| &c.kind), Some(ClassKind::Enum)))
183        };
184        let (core, _, _) = peel(ty, TokenStream::new());
185        if is_enum(core) {
186            return Ok(Some(false));
187        }
188        if let TypeKind::Optional(inner) = core.kind() {
189            if is_enum(inner) {
190                return Ok(Some(true));
191            }
192        }
193        Ok(None)
194    }

196    /// One typed parameter.
197    fn plain(
198        &self,
199        ty: &TypeRef,
200        value: TokenStream,
201        root: &str,
202        name: &str,
203        gates: &[String],
204        depth: usize,
205    ) -> Res<Delivery> {
206        let gated = !gates.is_empty();
207        let leaves: Vec<Leaf> = self
208            .leaves(ty, Dir::Out)?
209            .into_iter()
210            .map(|mut l| {
211                l.nullable |= gated && l.is_obj();
212                l.under(root)
213            })
214            .collect();
215        let raws: Vec<String> = leaves.iter().map(raw_name).collect();
216        // An enum reaches a delivered parameter as its value, an `Int`.
217        let as_int = self.enum_value(ty)?;
218        let decode = match as_int {
219            Some(_) => gate(gates, raws[0].clone()),
220            None => gate(gates, self.kt_decode(ty, &raws, gated, depth)?),
221        };
222        let pname = kt_ident(&names::camel(if name.is_empty() { "value" } else { name }));
223        let close = self.kt_close(ty, &pname)?.map(|c| {
224            if gated && !c.contains("?.") {
225                c.replacen(&format!("{pname}."), &format!("{pname}?."), 1)
226            } else {
227                c
228            }
229        });
230        Ok(Delivery {
231            params: vec![DParam {
232                name: pname,
233                kt: match as_int {
234                    Some(true) => "Int?".to_string(),
235                    Some(false) => nullable("Int".to_string(), gated),
236                    None => nullable(self.kt_type(ty)?, gated),
237                },
238                decode,
239                close,
240            }],
241            leaves,
242            output: self.rs_encode(ty, value, root, depth)?,
243        })
244    }

246    /// The fields of a struct value, each delivered (inlining data classes),
247    /// with the per-field expansions and names of a value form.
248    #[allow(clippy::too_many_arguments)]
249    fn fields_of(
250        &self,
251        s: &Struct,
252        form: Option<&FieldsDecl>,
253        value: TokenStream,
254        root: &str,
255        name: &str,
256        gates: &[String],
257        depth: usize,
258    ) -> Res<Delivery> {
259        let head = self.q.path(&s.name);
260        let binds: Vec<syn::Ident> = (0..s.fields.len())
261            .map(|i| format_ident!("__f{}_{}", depth, i))
262            .collect();
263        let pat = Record::Struct(s).pattern(&head, &binds);
264        let mut parts = Vec::new();
265        for (f, b) in s.fields.iter().zip(&binds) {
266            let seg = field_seg(f);
267            let explicit = form.and_then(|d| field_entry(&d.overrides, &seg));
268            let pname = match form.and_then(|d| field_entry(&d.names, &seg)) {
269                Some(n) => n.clone(),
270                None => join(name, &seg),
271            };
272            let d = self.deliver(
273                &f.ty,
274                b.to_token_stream(),
275                &join(root, &seg),
276                &pname,
277                explicit,
278                gates,
279                true,
280                depth + 1,
281            )?;
282            parts.push((Seg::field(f), d));
283        }
284        Ok(concat(
285            s.type_ref(),
286            quote!(let #pat = #value;),
287            parts,
288            Vec::new(),
289        ))
290    }

292    /// A value delivered through its output expansion.
293    #[allow(clippy::too_many_arguments)]
294    fn expanded(
295        &self,
296        ty: &TypeRef,
297        e: &ExpandReturnDecl,
298        value: TokenStream,
299        borrowed: bool,
300        root: &str,
301        name: &str,
302        gates: &[String],
303        depth: usize,
304    ) -> Res<Delivery> {
305        let v = format_ident!("__v{}", depth);
306        let type_name = core_name(ty).unwrap_or_default();
307        let type_snake = names::snake(&type_name);
308        let mut parts = Vec::new();
309        let mut deferred = Vec::new();
310        let mut prelude = quote!(let #v = #value;);
311        for field in &e.fields {
312            match field {
313                ReturnField::Getter(g) => {
314                    let func = g.fun.resolve(self.flat).map_err(crate::Error)?;
315                    let callee = g.fun.callee(&self.q);
316                    let fname = g.name.clone().unwrap_or_else(|| {
317                        let n = names::bare(&func.name);
318                        n.strip_prefix(&format!("{type_snake}_"))
319                            .unwrap_or(&n)
320                            .to_string()
321                    });
322                    let arg = match func.params.first().map(|p| p.ty.kind()) {
323                        Some(TypeKind::Ref { .. }) if borrowed => quote!(#v),
324                        Some(TypeKind::Ref { .. }) => quote!(&#v),
325                        Some(_) if borrowed => quote!(::core::clone::Clone::clone(#v)),
326                        Some(_) => quote!(::core::clone::Clone::clone(&#v)),
327                        None => return err(format!("getter `{}` takes no argument", func.name)),
328                    };
329                    let seg = names::snake(&fname);
330                    let (r, n) = (join(root, &seg), join(name, &seg));
331                    // A field of the expanded type itself is that value, not
332                    // another expansion of it.
333                    let field = Seg::Field(seg.clone());
334                    parts.push((
335                        field,
336                        if core_name(&func.ret).as_deref() == Some(type_name.as_str()) {
337                            self.plain(&func.ret, quote!(#callee(#arg)), &r, &n, gates, depth + 1)?
338                        } else {
339                            let call = quote!(#callee(#arg));
340                            self.deliver(&func.ret, call, &r, &n, None, gates, false, depth + 1)?
341                        },
342                    ));
343                }
344                ReturnField::Handle => {
345                    let handle = if borrowed {
346                        quote!(::core::clone::Clone::clone(#v))
347                    } else {
348                        quote!(#v)
349                    };
350                    let (r, n) = (join(root, "handle"), join(name, "handle"));
351                    deferred.push(parts.len());
352                    let d = self.plain(ty, handle, &r, &n, gates, depth + 1)?;
353                    parts.push((Seg::Field("handle".into()), d));
354                }
355                ReturnField::Form { form, consume } => {
356                    let fun = &form.fun;
357                    let f = self.flat.function(fun).ok_or_else(|| {
358                        crate::Error(format!("`{fun}` is not a #[prebindgen] function"))
359                    })?;
360                    let callee = self.q.path(fun);
361                    let arg = match (consume, borrowed) {
362                        (true, true) => quote!(::core::clone::Clone::clone(#v)),
363                        (true, false) => quote!(#v),
364                        (false, true) => quote!(#v),
365                        (false, false) => quote!(&#v),
366                    };
367                    let s = match f.ret.kind() {
368                        TypeKind::Named { id, .. } => match self.flat.declared_type(&id.name) {
369                            Some(FlatType::Struct(s)) => s,
370                            _ => return err(format!("value form `{fun}` must return a struct")),
371                        },
372                        _ => return err(format!("value form `{fun}` must return a struct")),
373                    };
374                    let sv = format_ident!("__s{}", depth);
375                    prelude.extend(quote!(let #sv = #callee(#arg);));
376                    let d = self.fields_of(
377                        s,
378                        Some(form),
379                        sv.to_token_stream(),
380                        root,
381                        name,
382                        gates,
383                        depth + 1,
384                    )?;
385                    // The value form stands in for the value: its fields.
386                    parts.push((Seg::Repr, d));
387                }
388            }
389        }
390        Ok(concat(ty, prelude, parts, deferred))
391    }

395    /// The deliveries of a callback's arguments, bound to `__a0`, `__a1`, …
396    pub(crate) fn callback_args(&self, args: &[TypeRef]) -> Res<Vec<Delivery>> {
397        args.iter()
398            .enumerate()
399            .map(|(i, a)| self.callback_arg(i, a, &format_ident!("__a{}", i).to_token_stream()))
400            .collect()
401    }

403    fn callback_arg(&self, i: usize, ty: &TypeRef, value: &TokenStream) -> Res<Delivery> {
404        let name = match self.shape(ty) {
405            Ok(Shape::Declared { ty, .. }) => super::declared_name(ty)
406                .map(names::snake)
407                .unwrap_or_else(|| format!("arg{i}")),
408            Ok(Shape::Scalar(k)) => k.as_str().to_string(),
409            _ => format!("arg{i}"),
410        };
411        self.deliver(
412            ty,
413            value.clone(),
414            &format!("a{i}"),
415            &name,
416            None,
417            &[],
418            false,
419            1,
420        )
421    }

423    fn callback_base(&self, ty: &TypeRef) -> Res<String> {
424        let TypeKind::Callback { args } = ty.kind() else {
425            return err(format!("`{ty}` is not a callback"));
426        };
427        if args.is_empty() {
428            return Ok("Void".to_string());
429        }
430        Ok(args.iter().map(type_base).collect())
431    }

433    /// The package of a callback type's interfaces: the class of its one
434    /// argument, else the base package.
435    fn callback_pkg(&self, ty: &TypeRef) -> String {
436        if let TypeKind::Callback { args } = ty.kind() {
437            if let [arg] = args.as_slice() {
438                if let Some(c) = core_name(arg).and_then(|n| self.class(&n)) {
439                    return c.pkg.clone();
440                }
441            }
442        }
443        self.base_pkg.clone()
444    }

458    /// The Rust closure a callback object `w` becomes: every call delivers
459    /// the arguments through one upcall.
460    pub(crate) fn callback_closure(
461        &self,
462        ty: &TypeRef,
463        args: &[TypeRef],
464        w: &syn::Ident,
465    ) -> Res<TokenStream> {
466        let leaves: Vec<Leaf> = self
467            .callback_args(args)?
468            .into_iter()
469            .flat_map(|d| d.leaves)
470            .collect();
471        let desc = method_desc(&leaves, "V");
472        let frame = 32 + leaves.len() as i32 * 2;
473        let what = format!("callback {ty}");
474        let names: Vec<_> = (0..args.len()).map(|i| format_ident!("__a{}", i)).collect();
475        let tys = args.iter().map(|t| self.q.ty_elided(t));
476        let outs = args
477            .iter()
478            .zip(&names)
479            .enumerate()
480            .map(|(i, (ty, n))| {
481                self.callback_arg(i, ty, &n.to_token_stream())
482                    .map(|d| d.output)
483            })
484            .collect::<Res<Vec<_>>>()?;
485        let binds: TokenStream = outs.iter().map(Output::bind).collect();
486        let values = outs
487            .iter()
488            .flat_map(|o| o.wires())
489            .zip(&leaves)
490            .map(|(wr, l)| l.jvalue(&wr.name.to_token_stream()));
491        Ok(quote! {{
492            let __up = ::prebindgen_jni_runtime::Upcall::new(env, &#w, "run", #desc, #frame)?;
493            move |#(#names: #tys),*| {
494                let __res = (|| -> ::core::result::Result<(), ::std::string::String> {
495                    __up.call_void(|env| {
496                        #binds
497                        ::core::result::Result::Ok(::std::vec![#(#values),*])
498                    })?;
499                    ::core::result::Result::Ok(())
500                })();
501                if let ::core::result::Result::Err(__err) = __res {
502                    ::prebindgen_jni_runtime::report_callback_error(#what, &__err);
503                }
504            }
505        }})
506    }
507}

514/// `ty` under its transparent layers and borrows: the core type, the value
515/// expression reaching it, and whether that value is borrowed.
516fn peel(ty: &TypeRef, value: TokenStream) -> (&TypeRef, TokenStream, bool) {
517    match ty.kind() {
518        TypeKind::Boxed(inner) => peel(inner, quote!((*#value))),
519        TypeKind::Ref { inner, .. } => {
520            let (t, v, _) = peel(inner, value);
521            (t, v, true)
522        }
523        _ => (ty, value, false),
524    }
525}

527/// The base a callback interface is named from.
528fn type_base(ty: &TypeRef) -> String {
529    match ty.kind() {
530        TypeKind::Named { id, .. } => id.name.clone(),
531        TypeKind::Scalar(k) => k.as_str().to_string(),
532        TypeKind::String | TypeKind::Str => "String".to_string(),
533        TypeKind::Ref { inner, .. } | TypeKind::Boxed(inner) | TypeKind::Cow { inner, .. } => {
534            type_base(inner)
535        }
536        TypeKind::Vec(e) | TypeKind::Slice(e) => format!("{}List", type_base(e)),
537        TypeKind::Optional(t) => format!("Optional{}", type_base(t)),
538        TypeKind::Array { elem, .. } => format!("{}Array", type_base(elem)),
539        _ => "Value".to_string(),
540    }
541}
542
543/// The named type under `Option`, `Box` and borrows.
544fn core_name(ty: &TypeRef) -> Option<String> {
545    match ty.kind() {
546        TypeKind::Named { id, .. } => Some(id.name.clone()),
547        TypeKind::Optional(t) | TypeKind::Boxed(t) | TypeKind::Ref { inner: t, .. } => core_name(t),
548        _ => None,
549    }
550}

552/// Deliveries side by side, bound in order except `deferred` (bound last —
553/// they consume the value the others read), wires in declaration order.
554fn concat(
555    ty: &TypeRef,
556    prelude: TokenStream,
557    parts: Vec<(Seg, Delivery)>,
558    deferred: Vec<usize>,
559) -> Delivery {
560    let fallible = parts.iter().any(|(_, p)| p.output.fallible);
561    let mut binds = Vec::new();
562    let mut late = Vec::new();
563    for (i, (_, p)) in parts.iter().enumerate() {
564        if deferred.contains(&i) {
565            late.push(p.output.bind());
566        } else {
567            binds.push(p.output.bind());
568        }
569    }
570    let mut params = Vec::new();
571    let mut leaves = Vec::new();
572    let mut forms = Vec::new();
573    for (seg, p) in parts {
574        // A handle delivered as a field of a larger value is the receiver's
575        // to keep; only a whole delivered value is closed after a callback.
576        params.extend(p.params.into_iter().map(|p| DParam { close: None, ..p }));
577        leaves.extend(p.leaves);
578        forms.push((seg, p.output.form));
579    }
580    let form = Form {
581        ty: ty.clone(),
582        kind: FormKind::Parts(forms),
583    };
584    let names: Vec<syn::Ident> = form.wires().iter().map(|w| w.name.clone()).collect();
585    let tuple = if names.len() == 1 {
586        let n = &names[0];
587        quote!(#n)
588    } else {
589        quote!((#(#names),*))
590    };
591    let expr = quote!({ #prelude #(#binds)* #(#late)* #tuple });
592    Delivery {
593        params,
594        leaves,
595        output: Output {
596            form,
597            expr,
598            fallible,
599        },
600    }
