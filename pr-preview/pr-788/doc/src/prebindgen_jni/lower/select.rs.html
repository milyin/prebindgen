1//! Input selectors: a parameter supplied in one of several ways, chosen at
2//! run time (`expand_param!(T).variant(fun!(ctor)).variant_self()`).
3//!
4//! Kotlin passes a selector `Int` plus every variant's arguments side by
5//! side, the inactive ones `null`: a constructor's arguments, each nullable,
6//! or the handle itself. Rust builds the value with the constructor, or
7//! takes (or borrows) the handle. `split_on_param` adds one typed overload
8//! per variant.
9//!
10//! An expansion with a single constructor and no handle variant is
11//! *direct*: there is nothing to choose, so Kotlin passes the constructor's
12//! arguments as the parameter and no selector crosses. An optional
13//! parameter is direct only when its absence shows in the arguments — the
14//! constructor takes at least one, and none is itself optional; otherwise it
15//! keeps the selector, whose `-1` means `None`.
16

35/// A resolved selector parameter.
36#[derive(Clone)]
37pub(crate) struct Selector {
38    pub param: FlatParam,
39    pub class: Rc<Class>,
40    pub variants: Vec<SelVariant>,
41    /// `Option<..>`: selector `-1` is `None`.
42    pub optional: bool,
43    /// How the callee takes the value.
44    pub access: Access,
45}

47#[derive(Clone)]
48#[allow(clippy::large_enum_variant)] // one per selector variant
49pub(crate) enum SelVariant {
50    Build { func: Function, callee: TokenStream },
51    Handle,
52}

54impl Selector {
55    /// A single constructor and nothing else, and — for an optional
56    /// parameter — arguments that can say the value is absent: no selector
57    /// crosses.
58    pub(crate) fn is_direct(&self) -> bool {
59        let [SelVariant::Build { func, .. }] = self.variants.as_slice() else {
60            return false;
61        };
62        !self.optional
63            || (!func.params.is_empty()
64                && func
65                    .params
66                    .iter()
67                    .all(|p| !matches!(p.ty.kind(), TypeKind::Optional(_))))
68    }
69}

71/// The Kotlin name of argument `j` of variant `i` of selector parameter
72/// `p`: `p1` for a one-argument constructor, `p01`, `p02` otherwise.
73fn arg_name(p: &str, i: usize, j: usize, arity: usize) -> String {
74    if arity == 1 {
75        format!("{p}{i}")
76    } else {
77        format!("{p}{i}{j}")
78    }
79}

82    /// The selector for `param`, if an input expansion applies to it.
83    pub(crate) fn selector(
84        &self,
85        param: &FlatParam,
86        explicit: Option<&ExpandParamDecl>,
87    ) -> Res<Option<Selector>> {
88        let (core, optional) = match param.ty.kind() {
89            TypeKind::Optional(inner) => (&**inner, true),
90            _ => (&param.ty, false),
91        };
92        let (name, access) = match shape(core, |candidate| {
93            super::bare_declared_name(candidate).and_then(|name| self.types.get(name))
94        }) {
95            Ok(Shape::Declared { ty, .. }) => (
96                super::declared_name(ty).expect("a declared class name"),
97                Access::of(ty),
98            ),
99            Ok(Shape::Undeclared(name)) => (name, Access::Owned),
100            _ => return Ok(None),
101        };
102        if access == Access::Exclusive {
103            return Ok(None);
104        }
105        let decl = match explicit {
106            Some(d) => d,
107            None => match self.param_exp.get(name) {
108                Some(d) => d,
109                None => return Ok(None),
110            },
111        };
112        if decl.variants.len() == 1 && matches!(decl.variants[0], ParamVariant::Handle) {
113            return Ok(None);
114        }
115        let Some(class) = self.class(name).cloned() else {
116            return err(format!(
117                "`{name}`: an input expansion needs a declared class"
118            ));
119        };
120        let mut variants = Vec::new();
121        for v in &decl.variants {
122            variants.push(match v {
123                ParamVariant::Build(f) => SelVariant::Build {
124                    func: f.fun.resolve(self.flat).map_err(crate::Error)?,
125                    callee: f.fun.callee(&self.q),
126                },
127                ParamVariant::Handle => SelVariant::Handle,
128            });
129        }
130        Ok(Some(Selector {
131            param: param.clone(),
132            class,
133            variants,
134            optional,
135            access,
136        }))
137    }

139    /// The Kotlin parameters of a selector: `pSel`, then each variant's.
140    pub(crate) fn selector_params(&self, s: &Selector) -> Res<Vec<(String, String)>> {
141        let p = kt_ident(&names::camel(&names::bare(&s.param.name)));
142        if let (true, [SelVariant::Build { func, .. }]) = (s.is_direct(), s.variants.as_slice()) {
143            let n = func.params.len();
144            return func
145                .params
146                .iter()
147                .enumerate()
148                .map(|(j, fp)| {
149                    let name = if n == 1 { p.clone() } else { format!("{p}{j}") };
150                    let t = self.kt_type(&fp.ty)?;
151                    Ok((name, if s.optional { nullable(t) } else { t }))
152                })
153                .collect();
154        }
155        let mut out = vec![(format!("{p}Sel"), "Int".to_string())];
156        for (i, v) in s.variants.iter().enumerate() {
157            match v {
158                SelVariant::Build { func, .. } => {
159                    let n = func.params.len();
160                    for (j, fp) in func.params.iter().enumerate() {
161                        let t = self.kt_type(&fp.ty)?;
162                        out.push((arg_name(&p, i, j, n), nullable(t)));
163                    }
164                }
165                SelVariant::Handle => out.push((format!("{p}{i}"), format!("{}?", s.class.fqn()))),
166            }
167        }
168        Ok(out)
169    }

171    /// The raw leaves of a selector.
172    pub(crate) fn selector_leaves(&self, s: &Selector) -> Res<Vec<Leaf>> {
173        let root = names::bare(&s.param.name);
174        let mut out = Vec::new();
175        if !s.is_direct() {
176            out.push(Leaf::new(LeafTy::Prim(Prim::I)).under(&format!("{root}_sel")));
177        }
178        for (i, v) in s.variants.iter().enumerate() {
179            match v {
180                SelVariant::Build { func, .. } => {
181                    for (j, fp) in func.params.iter().enumerate() {
182                        let opt = self.variant_arg_ty(s, &fp.ty);
183                        out.extend(
184                            self.leaves(&opt, Dir::In)?
185                                .into_iter()
186                                .map(|l| l.under(&format!("{root}_{i}{j}"))),
187                        );
188                    }
189                }
190                SelVariant::Handle => {
191                    out.push(Leaf::new(LeafTy::Prim(Prim::J)).under(&format!("{root}_{i}")))
192                }
193            }
194        }
195        Ok(out)
196    }

198    /// Kotlin leaf expressions for a selector, from its Kotlin parameters.
199    pub(crate) fn selector_encode(&self, s: &Selector, cx: &mut KtEnc) -> Res<Vec<String>> {
200        let p = kt_ident(&names::camel(&names::bare(&s.param.name)));
201        if s.is_direct() {
202            let names: Vec<String> = self
203                .selector_params(s)?
204                .into_iter()
205                .map(|(n, _)| n)
206                .collect();
207            let [SelVariant::Build { func, .. }] = s.variants.as_slice() else {
208                unreachable!("a direct selector");
209            };
210            let mut out = Vec::new();
211            for (fp, n) in func.params.iter().zip(&names) {
212                let ty = self.variant_arg_ty(s, &fp.ty);
213                out.extend(self.kt_encode(&ty, n, false, cx, true)?);
214            }
215            return Ok(out);
216        }
217        let mut out = vec![format!("{p}Sel")];
218        for (i, v) in s.variants.iter().enumerate() {
219            match v {
220                SelVariant::Build { func, .. } => {
221                    let n = func.params.len();
222                    for (j, fp) in func.params.iter().enumerate() {
223                        let opt = self.variant_arg_ty(s, &fp.ty);
224                        out.extend(self.kt_encode(
225                            &opt,
226                            &arg_name(&p, i, j, n),
227                            false,
228                            cx,
229                            true,
230                        )?);
231                    }
232                }
233                SelVariant::Handle => {
234                    let e = format!("{p}{i}");
235                    out.extend(self.kt_encode_handle(&s.class, &e, s.access == Access::Owned, cx));
236                }
237            }
238        }
239        Ok(out)
240    }

242    fn kt_encode_handle(
243        &self,
244        class: &Rc<Class>,
245        expr: &str,
246        consumed: bool,
247        cx: &mut KtEnc,
248    ) -> Vec<String> {
249        cx.handles.push(HandleSite {
250            expr: expr.to_string(),
251            nullable: true,
252            consumed,
253            class: class.clone(),
254        });
255        vec![format!("({expr}?.ptr ?: 0L)")]
256    }

258    /// How argument type `ty` of a variant crosses: optional, unless the
259    /// selector is direct and not optional itself.
260    fn variant_arg_ty(
261        &self,
262        s: &Selector,
263        ty: &prebindgen_flat::flat::TypeRef,
264    ) -> prebindgen_flat::flat::TypeRef {
265        let already = matches!(ty.kind(), TypeKind::Optional(_));
266        if already || (s.is_direct() && !s.optional) {
267            ty.clone()
268        } else {
269            ty.optional()
270        }
271    }

273    /// The Rust input of a direct selector: the constructor called on its
274    /// arguments (`None` when an optional parameter's are absent).
275    fn direct_input(&self, s: &Selector, func: &Function, callee: &TokenStream) -> Res<Param> {
276        let root = names::bare(&s.param.name);
277        let mut parts = Vec::new();
278        let mut args = Vec::new();
279        let mut binds = Vec::new();
280        for (j, fp) in func.params.iter().enumerate() {
281            let ty = self.variant_arg_ty(s, &fp.ty);
282            let input = self.rs_decode(&ty, &format!("{root}_0{j}"), 0)?;
283            let e = input.result();
284            parts.push((Seg::Param(names::bare(&fp.name)), input.form));
285            let a = quote::format_ident!("__a{}", j);
286            binds.push(quote!(let #a = #e?;));
287            args.push(a);
288        }
289        let call = |args: &[syn::Ident]| match func.ret.kind() {
290            TypeKind::Fallible { .. } => {
291                quote!(#callee(#(#args),*).map_err(|__e| ::std::string::ToString::to_string(&__e))?)
292            }
293            _ => quote!(#callee(#(#args),*)),
294        };
295        let built = call(&args);
296        let wrap = |b: TokenStream| match s.access {
297            Access::Owned => b,
298            _ => quote!(::prebindgen_jni_runtime::MaybeOwned::Owned(#b)),
299        };
300        let expr = if s.optional {
301            let some = wrap(built);
302            quote!({
303                #(#binds)*
304                match (#(#args,)*) {
305                    (#(::core::option::Option::Some(#args),)*) => ::core::option::Option::Some(#some),
306                    _ => ::core::option::Option::None,
307                }
308            })
309        } else {
310            let v = wrap(built);
311            quote!({ #(#binds)* #v })
312        };
313        // The parameter is built from the constructor's arguments.
314        let form = Form {
315            ty: s.param.ty.clone(),
316            kind: FormKind::Parts(parts),
317        };
318        Ok(selector_param(s, form, expr))
319    }

321    /// The Rust input of a selector parameter.
322    pub(crate) fn selector_input(&self, s: &Selector) -> Res<Param> {
323        if let (true, [SelVariant::Build { func, callee }]) = (s.is_direct(), s.variants.as_slice())
324        {
325            return self.direct_input(s, func, callee);
326        }
327        let root = names::bare(&s.param.name);
328        let sel = leaf_ident(&root, "sel");
329        let t = self.q.path(&names::ident(&s.class.rust));
330        let tag = Wire::new(sel.clone(), Leaf::new(LeafTy::Prim(Prim::I)));
331        let mut alts = Vec::new();
332        let mut arms = Vec::new();
333        for (i, v) in s.variants.iter().enumerate() {
334            let body = match v {
335                SelVariant::Build { func, callee } => {
336                    let mut args = Vec::new();
337                    let mut parts = Vec::new();
338                    for (j, fp) in func.params.iter().enumerate() {
339                        let r = format!("{root}_{i}{j}");
340                        let input = self.rs_decode(&self.variant_arg_ty(s, &fp.ty), &r, 0)?;
341                        let e = input.result();
342                        parts.push((Seg::Param(names::bare(&fp.name)), input.form));
343                        // An optional argument's `None` is its value, not a
344                        // missing argument.
345                        if let TypeKind::Optional(_) = fp.ty.kind() {
346                            args.push(quote!(#e?));
347                            continue;
348                        }
349                        let pname = names::bare(&fp.name);
350                        let msg = format!("missing argument `{pname}` for `{root}` variant {i}");
351                        args.push(quote!(#e?.ok_or_else(|| ::std::string::String::from(#msg))?));
352                    }
353                    let built = match func.ret.kind() {
354                        TypeKind::Fallible { .. } => {
355                            quote!(#callee(#(#args),*).map_err(|__e| ::std::string::ToString::to_string(&__e))?)
356                        }
357                        _ => quote!(#callee(#(#args),*)),
358                    };
359                    alts.push(parts);
360                    match s.access {
361                        Access::Owned => built,
362                        _ => quote!(::prebindgen_jni_runtime::MaybeOwned::Owned(#built)),
363                    }
364                }
365                SelVariant::Handle => {
366                    let w = leaf_ident(&root, &i.to_string());
367                    // The value itself, as an existing handle.
368                    let handle = Wire::new(w.clone(), Leaf::new(LeafTy::Prim(Prim::J)));
369                    let form = Form {
370                        ty: s.param.ty.clone(),
371                        kind: FormKind::Wire(handle),
372                    };
373                    alts.push(vec![(Seg::Param("self".into()), form)]);
374                    match s.access {
375                        Access::Owned => quote!(::prebindgen_jni_runtime::take_handle::<#t>(#w)?),
376                        _ => quote!(::prebindgen_jni_runtime::MaybeOwned::Borrowed(
377                            ::prebindgen_jni_runtime::borrow_handle::<#t>(#w)?
378                        )),
379                    }
380                }
381            };
382            let i = i as i32;
383            arms.push(if s.optional {
384                quote!(#i => ::core::option::Option::Some(#body))
385            } else {
386                quote!(#i => #body)
387            });
388        }
389        if s.optional {
390            arms.push(quote!(-1 => ::core::option::Option::None));
391        }
392        let msg = format!("invalid selector {{}} for `{root}`");
393        let expr = quote!(match #sel {
394            #(#arms,)*
395            __s => return ::core::result::Result::Err(::std::format!(#msg, __s)),
396        });
397        // A selector is a tag choosing among the variants.
398        let form = Form {
399            ty: s.param.ty.clone(),
400            kind: FormKind::Sum { tag, alts },
401        };
402        Ok(selector_param(s, form, expr))
403    }

405    /// The typed overloads of a selector-form function, one per combination
406    /// of the split parameters' variants.
407    pub(crate) fn split_overloads(
408        &self,
409        head: &str,
410        name: &str,
411        params: &[SigParam],
412        tail: &[(String, String)],
413        ret: &str,
414        prefix_names: bool,
415    ) -> Res<Vec<String>> {
416        let mut combos: Vec<Vec<(usize, usize)>> = vec![Vec::new()];
417        for (pi, p) in params.iter().enumerate() {
418            if let SigParam::Split(s) = p {
419                let mut next = Vec::new();
420                for c in &combos {
421                    for vi in 0..s.variants.len() {
422                        let mut c2 = c.clone();
423                        c2.push((pi, vi));
424                        next.push(c2);
425                    }
426                }
427                combos = next;
428            }
429        }
430        let mut out = Vec::new();
431        for combo in combos {
432            let mut sig = Vec::new();
433            let mut args = Vec::new();
434            for (pi, p) in params.iter().enumerate() {
435                match p {
436                    SigParam::Plain(n, t) => {
437                        sig.push(format!("{n}: {t}"));
438                        args.push(n.clone());
439                    }
440                    SigParam::Split(s) => {
441                        let vi = combo.iter().find(|(i, _)| *i == pi).unwrap().1;
442                        let base = kt_ident(&names::camel(&names::bare(&s.param.name)));
443                        args.push(vi.to_string());
444                        for (i, v) in s.variants.iter().enumerate() {
445                            match v {
446                                SelVariant::Build { func, .. } => {
447                                    for fp in &func.params {
448                                        if i == vi {
449                                            let pn = names::bare(&fp.name);
450                                            let n = if prefix_names {
451                                                kt_ident(&names::camel(&format!(
452                                                    "{}_{pn}",
453                                                    names::bare(&s.param.name)
454                                                )))
455                                            } else {
456                                                kt_ident(&names::camel(&pn))
457                                            };
458                                            sig.push(format!("{n}: {}", self.kt_type(&fp.ty)?));
459                                            args.push(n);
460                                        } else {
461                                            args.push("null".to_string());
462                                        }
463                                    }
464                                }
465                                SelVariant::Handle => {
466                                    if i == vi {
467                                        sig.push(format!("{base}: {}", s.class.fqn()));
468                                        args.push(base.clone());
469                                    } else {
470                                        args.push("null".to_string());
471                                    }
472                                }
473                            }
474                        }
475                    }
476                    SigParam::Selector(s) => {
477                        for (n, t) in self.selector_params(s)? {
478                            sig.push(format!("{n}: {t}"));
479                            args.push(n);
480                        }
481                    }
482                }
483            }
484            for (n, t) in tail {
485                sig.push(format!("{n}: {t}"));
486                args.push(n.clone());
487            }
488            out.push(format!(
489                "{head}{name}({}): {ret} =\n    {name}({})\n",
490                sig.join(", "),
491                args.join(", ")
492            ));
493        }
494        Ok(out)
495    }
496}

498/// A Kotlin wrapper parameter: plain, a selector group, or a selector group
499/// that is split into overloads.
500pub(crate) enum SigParam {
501    Plain(String, String),
502    Selector(Selector),
503    Split(Selector),
504}
505
506fn nullable(t: String) -> String {
507    if t.ends_with('?') {
508        t
509    } else {
510        format!("{t}?")
511    }
512}

514/// A selector parameter's input over `form`, and how its value is lent.
515fn selector_param(s: &Selector, form: Form<Leaf>, expr: TokenStream) -> Param {
516    let name = &s.param.name;
517    let pass = match (s.access, s.optional) {
518        (Access::Owned, _) => None,
519        (_, false) => Some(quote!(&*#name)),
520        (_, true) => Some(quote!(#name.as_deref())),
521    };
522    Param {
523        input: Input {
524            form,
525            expr,
526            fallible: true,
527        },
528        pass,
529    }
