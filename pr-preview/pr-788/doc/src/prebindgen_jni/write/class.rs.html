20fn indent(text: &str, by: usize) -> String {
21    let pad = " ".repeat(by);
22    text.lines()
23        .map(|l| {
24            if l.is_empty() {
25                String::new()
26            } else {
27                format!("{pad}{l}")
28            }
29        })
30        .collect::<Vec<_>>()
31        .join("\n")
32}

34/// One class with its members: the Kotlin text into its package, the
35/// members' externs (and a handle's `freePtr`) into the Rust file.
36pub(crate) fn class(
37    plan: &Plan,
38    out: &mut Out,
39    c: &Class,
40    methods: &[Binding],
41    ctors: &[Binding],
42) -> Res<()> {
43    let methods: Vec<Wrapper> = methods
44        .iter()
45        .map(|m| function(plan, out, m))
46        .collect::<Res<_>>()?;
47    let ctors: Vec<Wrapper> = ctors
48        .iter()
49        .map(|m| function(plan, out, m))
50        .collect::<Res<_>>()?;
51    let text = match &c.kind {
52        ClassKind::Ptr { gc } => {
53            out.rust.push(free_ptr(plan, c));
54            ptr_class(plan, c, *gc, &methods, &ctors)
55        }
56        ClassKind::Data => data_class(plan, c, &methods, &ctors)?,
57        ClassKind::Enum => enum_class(plan, c)?,
58        ClassKind::Sealed { .. } => sealed_class(plan, c)?,
59    };
60    out.kt(&c.pkg, text);
61    Ok(())
62}

64/// A handle class's `freePtr`, and the alignment its tag bit relies on.
65fn free_ptr(plan: &Plan, c: &Class) -> proc_macro2::TokenStream {
66    let sym = format_ident!("{}", jni_symbol(&c.fqn(), "freePtr"));
67    let t = plan.q.path(&names::ident(&c.rust));
68    let msg = format!(
69        "`{}`: a handle type must have alignment >= 2 (bit 0 is the closed tag)",
70        c.rust
71    );
72    quote::quote! {
73        #[no_mangle]
74        #[allow(non_snake_case)]
75        pub unsafe extern "system" fn #sym(
76            _env: ::prebindgen_jni_runtime::jni::JNIEnv,
77            _class: ::prebindgen_jni_runtime::jni::objects::JClass,
78            ptr: ::prebindgen_jni_runtime::jni::sys::jlong,
79        ) {
80            ::prebindgen_jni_runtime::free_handle::<#t>(ptr)
81        }
82        const _: () = assert!(::core::mem::align_of::<#t>() >= 2, #msg);
83    }
84}

86fn supertypes(c: &Class, base: &[String]) -> String {
87    let mut all: Vec<String> = base.to_vec();
88    all.extend(c.iface.clone());
89    all.extend(c.implements.iter().cloned());
90    if all.is_empty() {
91        String::new()
92    } else {
93        format!(" : {}", all.join(", "))
94    }
95}

97fn iface_decl(c: &Class, extends: &str, members: Vec<String>) -> String {
98    let Some(name) = &c.iface else {
99        return String::new();
100    };
101    let body = members
102        .iter()
103        .map(|m| format!("    {m}"))
104        .collect::<Vec<_>>()
105        .join("\n\n");
106    format!("public interface {name}{extends} {{\n{body}\n}}\n\n")
107}

109fn ptr_class(plan: &Plan, c: &Class, gc: bool, methods: &[Wrapper], ctors: &[Wrapper]) -> String {
110    let name = &c.name;
111    let base = &plan.base_pkg;
112    let parent = if gc {
113        format!("{base}.GcNativeHandle(initialPtr)")
114    } else {
115        format!("{base}.NativeHandle(initialPtr)")
116    };
117    let over = if c.iface.is_some() { "override " } else { "" };
118    let lifecycle = if gc {
119        format!(
120            "    private val __cleanable = {base}.registerGcHandle(this) {{ freePtr(it) }}\n\n\
121             \x20   @Synchronized\n    override fun close() {{\n        val p = {base}.releaseCell(cell)\n        if (p != 0L) freePtr(p)\n        __cleanable?.clean()\n    }}\n\n\
122             \x20   @Synchronized\n    public {over}fun take(): {name} {{\n        val p = {base}.releaseCell(cell)\n        __cleanable?.clean()\n        return {name}(if (p != 0L) p else cell.get())\n    }}\n"
123        )
124    } else {
125        format!(
126            "    @Synchronized\n    override fun close() {{\n        val p = ptr\n        if (p != 0L && (p and 1L) == 0L) {{\n            ptr = p or 1L\n            freePtr(p)\n        }}\n    }}\n\n\
127             \x20   @Synchronized\n    public {over}fun take(): {name} {{\n        val p = ptr\n        ptr = p or 1L\n        return {name}(p)\n    }}\n"
128        )
129    };
130    let mut members = vec![
131        "fun peek(): Long".to_string(),
132        "fun isClosed(): Boolean".to_string(),
133        format!("fun take(): {name}"),
134    ];
135    members.extend(methods.iter().map(|m| m.header.clone()));
136    let iface = iface_decl(c, " : AutoCloseable", members);
137    let methods = methods
138        .iter()
139        .map(|m| indent(&m.text, 4))
140        .collect::<Vec<_>>()
141        .join("\n\n");
142    let ctors = ctors
143        .iter()
144        .map(|m| indent(&m.text, 8))
145        .collect::<Vec<_>>()
146        .join("\n\n");
147    format!(
148        "{iface}public class {name}(initialPtr: Long){} {{\n{lifecycle}\n{methods}\n\n    public companion object {{\n        @JvmStatic\n        external fun freePtr(ptr: Long)\n\n{ctors}\n    }}\n}}\n",
149        supertypes(c, &[parent])
150    )
151}

153fn data_class(plan: &Plan, c: &Class, methods: &[Wrapper], ctors: &[Wrapper]) -> Res<String> {
154    let s = plan.struct_of(c)?;
155    let over = if c.iface.is_some() { "override " } else { "" };
156    let mut props = Vec::new();
157    let mut iface_members = Vec::new();
158    let mut args = Vec::new();
159    let mut closes = Vec::new();
160    let mut arrays = Vec::new();
161    let mut types = Vec::new();
162    for f in &s.fields {
163        let p = kt_prop(f);
164        let t = plan.kt_type(&f.ty)?;
165        types.push(t.clone());
166        props.push(format!("{over}val {p}: {t}"));
167        iface_members.push(format!("val {p}: {t}"));
168        args.push(format!("{p}: {t}"));
169        closes.extend(plan.kt_close(&f.ty, &p)?);
170        arrays.push((p, array_eq(plan, &f.ty)?));
171    }
172    let name = &c.name;
173    // `fromParts` takes the same parameters as the constructor.
174    super::check_slots(&format!("the constructor of `{name}`"), &types)?;
175    let owns = !closes.is_empty();
176    let supers = if owns {
177        vec!["AutoCloseable".to_string()]
178    } else {
179        Vec::new()
180    };
181    iface_members.extend(methods.iter().map(|m| m.header.clone()));
182    let iface = iface_decl(c, "", iface_members);
183    let mut body = Vec::new();
184    if owns {
185        body.push(format!(
186            "    override fun close() {{\n        {}\n    }}",
187            closes.join("\n        ")
188        ));
189    }
190    if arrays.iter().any(|(_, a)| a.is_some()) {
191        body.push(array_members(name, &arrays));
192    }
193    body.extend(methods.iter().map(|m| indent(&m.text, 4)));
194    let names_list = s.fields.iter().map(kt_prop).collect::<Vec<_>>().join(", ");
195    let from_parts = format!(
196        "        @JvmStatic\n        public fun fromParts({}): {name} = {name}({names_list})",
197        args.join(", ")
198    );
199    let companion = std::iter::once(from_parts)
200        .chain(ctors.iter().map(|m| indent(&m.text, 8)))
201        .collect::<Vec<_>>()
202        .join("\n\n");
203    Ok(format!(
204        "{iface}public data class {name}({}){} {{\n{}\n\n    public companion object {{\n{companion}\n    }}\n}}\n",
205        props.join(", "),
206        supertypes(c, &supers),
207        body.join("\n\n")
208    ))
209}

211/// How a field compares when it is array-backed: `Some((eq, hash, show))`
212/// templates over `$a` / `$b`, or `None` for ordinary equality.
213fn array_eq(plan: &Plan, ty: &TypeRef) -> Res<Option<ArrayEq>> {
214    Ok(match plan.shape(ty)? {
215        Shape::Array { .. } => Some(content_eq()),
216        Shape::Seq { elem, .. } if is_u8(elem) => Some(content_eq()),
217        Shape::Seq { elem, .. } => array_eq(plan, elem)?.map(|(eq, hash, show)| {
218            (
219                format!(
220                    "($a.size == $b.size && $a.indices.all {{ __i -> val __x = $a[__i]; val __y = $b[__i]; {} }})",
221                    eq.replace("$a", "__x").replace("$b", "__y")
222                ),
223                format!("($a.fold(1) {{ __acc, __e -> 31 * __acc + {} }})", hash.replace("$a", "__e")),
224                format!(
225                    "${{$a.joinToString(\", \", \"[\", \"]\") {{ __e -> \"{}\" }}}}",
226                    show.replace("$a", "__e")
227                ),
228            )
229        }),
230        Shape::Cow(inner) => match inner.kind() {
231            prebindgen_flat::flat::TypeKind::Str | prebindgen_flat::flat::TypeKind::Slice(_) =>
232                array_eq(plan, &crate::lower::cow_view(inner))?,
233            _ => None,
234        },
235        Shape::Option(inner) => array_eq(plan, inner)?.map(|(eq, hash, show)| {
236            (
237                format!(
238                    "(if ($a == null || $b == null) $a === $b else {})",
239                    eq.replace("$a", "$a!!").replace("$b", "$b!!")
240                ),
241                format!("($a?.let {{ {} }} ?: 0)", hash.replace("$a", "it")),
242                show,
243            )
244        }),
245        _ => None,
246    })
247}

249fn content_eq() -> ArrayEq {
250    (
251        "$a.contentEquals($b)".to_string(),
252        "$a.contentHashCode()".to_string(),
253        "${$a.contentToString()}".to_string(),
254    )
255}

257fn array_members(name: &str, fields: &[(String, Option<ArrayEq>)]) -> String {
258    let eqs = fields
259        .iter()
260        .map(|(p, a)| match a {
261            Some((eq, _, _)) => eq.replace("$a", p).replace("$b", &format!("other.{p}")),
262            None => format!("{p} == other.{p}"),
263        })
264        .collect::<Vec<_>>()
265        .join(" && ");
266    let hashes: Vec<String> = fields
267        .iter()
268        .map(|(p, a)| match a {
269            Some((_, h, _)) => h.replace("$a", p),
270            None => format!("{p}.hashCode()"),
271        })
272        .collect();
273    let mut hash = format!(
274        "        var result = {}\n",
275        hashes.first().cloned().unwrap_or("0".into())
276    );
277    for h in hashes.iter().skip(1) {
278        hash.push_str(&format!("        result = 31 * result + {h}\n"));
279    }
280    let shows = fields
281        .iter()
282        .map(|(p, a)| match a {
283            Some((_, _, s)) => format!("{p}={}", s.replace("$a", p)),
284            None => format!("{p}=${{{p}}}"),
285        })
286        .collect::<Vec<_>>()
287        .join(", ");
288    format!(
289        "    override fun equals(other: Any?): Boolean {{\n        if (this === other) return true\n        if (other !is {name}) return false\n        return {eqs}\n    }}\n\n\
290         \x20   override fun hashCode(): Int {{\n{hash}        return result\n    }}\n\n\
291         \x20   override fun toString(): String = \"{name}({shows})\""
292    )
293}

295fn enum_class(plan: &Plan, c: &Class) -> Res<String> {
296    let values = plan.enum_arms(c)?;
297    let over = if c.iface.is_some() { "override " } else { "" };
298    let entries = values
299        .iter()
300        .map(|(n, d)| format!("    {}({d})", names::snake(&names::bare(n)).to_uppercase()))
301        .collect::<Vec<_>>()
302        .join(",\n");
303    let iface = iface_decl(c, "", vec!["val value: Int".to_string()]);
304    let name = &c.name;
305    Ok(format!(
306        "{iface}public enum class {name}(public {over}val value: Int){} {{\n{entries};\n\n    public companion object {{\n        @JvmStatic\n        public fun fromInt(value: Int): {name} = entries.first {{ it.value == value }}\n    }}\n}}\n",
307        supertypes(c, &[])
308    ))
309}

311fn sealed_class(plan: &Plan, c: &Class) -> Res<String> {
312    let v = plan.variant_of(c)?;
313    let name = &c.name;
314    let mut owns = false;
315    for a in &v.alternatives {
316        for f in &a.fields {
317            owns |= plan.owns_handle(&f.ty)?;
318        }
319    }
320    let close_body = |closes: Vec<String>| {
321        if closes.is_empty() {
322            "        override fun close() {\n        }".to_string()
323        } else {
324            format!(
325                "        override fun close() {{\n            {}\n        }}",
326                closes.join("\n            ")
327            )
328        }
329    };
330    let mut variants = Vec::new();
331    let mut from_params = vec!["tag: Int".to_string()];
332    let mut from_types = vec!["Int".to_string()];
333    let mut arms = Vec::new();
334    let mut companion_clash = false;
335    for (i, alt) in v.alternatives.iter().enumerate() {
336        let vname = alt_kt_name(c, alt);
337        companion_clash |= vname == "Companion";
338        if alt.fields.is_empty() {
339            variants.push(if owns {
340                format!(
341                    "    public data object {vname} : {name} {{\n{}\n    }}",
342                    close_body(Vec::new())
343                )
344            } else {
345                format!("    public data object {vname} : {name}")
346            });
347            arms.push(format!("{i} -> {vname}"));
348            continue;
349        }
350        let mut props = Vec::new();
351        let mut closes = Vec::new();
352        let mut args = Vec::new();
353        let mut arrays = Vec::new();
354        for f in &alt.fields {
355            let p = kt_prop(f);
356            let t = plan.kt_type(&f.ty)?;
357            props.push(format!("public val {p}: {t}"));
358            closes.extend(plan.kt_close(&f.ty, &p)?);
359            arrays.push((p.clone(), array_eq(plan, &f.ty)?));
360            let fp = format!("{}_{}", names::snake(&names::bare(&alt.name)), field_seg(f));
361            from_params.push(format!("{fp}: {t}"));
362            from_types.push(t);
363            args.push(fp);
364        }
365        let mut members = Vec::new();
366        if owns {
367            members.push(close_body(closes));
368        }
369        if arrays.iter().any(|(_, a)| a.is_some()) {
370            members.push(indent(&array_members(&vname, &arrays), 4));
371        }
372        let body = if members.is_empty() {
373            String::new()
374        } else {
375            format!(" {{\n{}\n    }}", members.join("\n\n"))
376        };
377        variants.push(format!(
378            "    public data class {vname}({}) : {name}{body}",
379            props.join(", ")
380        ));
381        arms.push(format!("{i} -> {vname}({})", args.join(", ")));
382    }
383    arms.push(format!(
384        "else -> throw IllegalArgumentException(\"{name}: invalid tag $tag\")"
385    ));
386    // Each alternative's constructor takes a subset of these.
387    super::check_slots(&format!("`{name}.fromParts`"), &from_types)?;
388    let companion = if companion_clash {
389        "companion object Companion_"
390    } else {
391        "companion object"
392    };
393    let supers = if owns {
394        vec!["AutoCloseable".to_string()]
395    } else {
396        Vec::new()
397    };
398    Ok(format!(
399        "public sealed interface {name}{} {{\n{}\n\n    public {companion} {{\n        @JvmStatic\n        public fun fromParts({}): {name} =\n            when (tag) {{\n                {}\n            }}\n    }}\n}}\n",
400        supertypes(c, &supers),
401        variants.join("\n\n"),
402        from_params.join(", "),
403        arms.join("\n                ")
404    ))
405}

407/// A constant: a private getter and a lazily initialized `val`.
408pub(crate) fn constant(plan: &Plan, out: &mut Out, pkg: &str, name: &str, b: &Binding) -> Res<()> {
409    let w = function(plan, out, b)?;
410    let ty = plan.kt_type(&b.func.ret)?;
411    let private = w.text.replacen("public fun", "private fun", 1);
412    out.kt(
413        pkg,
414        format!(
415            "{private}\npublic val {name}: {ty} by lazy {{ {}({}.JniErrorHandler {{ je -> error(je ?: \"const {name}: JNI getter failed\") }}) }}\n",
416            b.kt_name, plan.base_pkg
417        ),
418    );
419    Ok(())
