10/// A function a declaration refers to: a flat item by name, or a
11/// binding-local path with its signature stated.
12#[derive(Clone, Debug)]
13pub struct FnRef {
14    /// Path to call in generated Rust; a bare name without [`Self::sig`]
15    /// refers to a captured flat function.
16    pub path: syn::Path,
17    /// Signature supplied for a function outside the flat model.
18    pub sig: Option<syn::Signature>,
19}
20
21impl FnRef {
22    /// Create a reference to a flat function or a binding-local function.
23    /// Supply [`Self::sig`] for the latter before [`Self::resolve`].
24    pub fn new(path: syn::Path) -> Self {
25        Self { path, sig: None }
26    }
27
28    /// State the signature of a function the flat model does not know.
29    pub fn sig(mut self, sig: syn::Signature) -> Self {
30        self.sig = Some(sig);
31        self
32    }
33
34    /// The last path segment: the flat name, or the local function's name.
35    pub fn name(&self) -> &syn::Ident {
36        &self.path.segments.last().expect("a non-empty path").ident
37    }
38
39    /// Whether this names a flat item (a bare name, no stated signature).
40    pub fn is_flat(&self) -> bool {
41        self.sig.is_none() && self.path.segments.len() == 1 && self.path.leading_colon.is_none()
42    }
43
44    /// The function as the model reads it: the flat item, or the stated
45    /// signature read against the flat model's type names.
46    pub fn resolve(&self, flat: &Flat) -> Result<Function, String> {
47        match &self.sig {
48            Some(sig) => {
49                let mut sig = sig.clone();
50                sig.ident = self.name().clone();
51                let item: syn::ItemFn = syn::parse_quote!(pub #sig { unimplemented!() });
52                flat.lower_signature(&item)
53                    .map_err(|e| format!("`{}`: {e}", path_string(&self.path)))
54            }
55            None if !self.is_flat() => Err(format!(
56                "`{}` is not a #[prebindgen] function: state its signature with `.sig(sig!(..))`",
57                path_string(&self.path)
58            )),
59            None => flat
60                .function(self.name())
61                .cloned()
62                .ok_or_else(|| format!("`{}` is not a #[prebindgen] function", self.name())),
63        }
64    }
65
66    /// The path generated code calls.
67    pub fn callee(&self, q: &Qualifier<'_>) -> TokenStream {
68        if self.is_flat() {
69            q.path(self.name())
70        } else {
71            self.path.to_token_stream()
72        }
73    }
74}
75
76fn path_string(p: &syn::Path) -> String {
77    p.to_token_stream().to_string().replace(' ', "")
78}

80/// How one direction of a conversion is carried out.
81#[derive(Clone, Debug)]
82pub enum Via {
83    /// Call a source or binding-local function; see [`FnRef`].
84    Fn(FnRef),
85    /// Use `Source::from(Repr)` for input conversion.
86    From(syn::Type),
87    /// Use `Source::into(Repr)` for output conversion.
88    Into(syn::Type),
89    /// Use `Source::try_from(Repr)` for fallible input conversion.
90    TryFrom(syn::Type),
91    /// Use `Source::try_into(Repr)` for fallible output conversion.
92    TryInto(syn::Type),
93}
94
95impl From<FnRef> for Via {
96    fn from(f: FnRef) -> Self {
97        Via::Fn(f)
98    }
99}

101/// A declared conversion for one source type.
102#[derive(Clone, Debug)]
103pub struct Conversion {
104    /// The source type that will cross as a representation type.
105    pub ty: syn::Type,
106    /// Representation to source; `None` leaves this direction undeclared.
107    pub input: Option<Via>,
108    /// Source to representation; `None` leaves this direction undeclared.
109    pub output: Option<Via>,
110}
111
112impl Conversion {
113    /// Start a declaration for `ty`; the [`convert!`](crate::convert!) macro is shorthand.
114    pub fn new(ty: syn::Type) -> Self {
115        Self {
116            ty,
117            input: None,
118            output: None,
119        }
120    }
121
122    /// How a representation value becomes the source type.
123    pub fn input(mut self, via: impl Into<Via>) -> Self {
124        self.input = Some(via.into());
125        self
126    }
127
128    /// How the source type becomes its representation.
129    pub fn output(mut self, via: impl Into<Via>) -> Self {
130        self.output = Some(via.into());
131        self
132    }
133
134    /// Resolve both directions against the model.
135    ///
136    /// Each declared direction is read first, for how it converts and which
137    /// representation type it names. The directions must name the same one,
138    /// and at least one must be declared; that type is then the conversion's
139    /// representation.
140    pub fn resolve(&self, flat: &Flat) -> Result<ResolvedConversion, String> {
141        let target = flat
142            .classify(&self.ty)
143            .map_err(|e| format!("convert!({}): {e}", self.ty.to_token_stream()))?;
144        let read = |via: &Option<Via>, dir| {
145            via.as_ref()
146                .map(|v| Stage::resolve(flat, v, &target, dir))
147                .transpose()
148        };
149        let (input, output) = (
150            read(&self.input, Direction::In)?,
151            read(&self.output, Direction::Out)?,
152        );
153        let repr = match (&input, &output) {
154            (Some((_, i)), Some((_, o))) if i.key() != o.key() => {
155                return Err(format!(
156                    "convert!({target}): the input converts from `{i}` but the output \
157                     converts to `{o}`; both directions must use one representation type"
158                ))
159            }
160            (Some((_, r)), _) | (None, Some((_, r))) => r.clone(),
161            (None, None) => {
162                return Err(format!(
163                    "convert!({target}) declares neither `.input(..)` nor `.output(..)`"
164                ))
165            }
166        };
167        Ok(ResolvedConversion {
168            target,
169            repr,
170            input: input.map(|(s, _)| s),
171            output: output.map(|(s, _)| s),
172        })
173    }
174}

176/// A conversion resolved against the model: the source type crosses as
177/// its representation type, converted on the way in, on the way out, or
178/// both. Both directions share the representation, so the foreign side sees
179/// one type for the source type whichever way a value goes.
180///
181/// An adapter keeps it as its declaration for the source type, and passes
182/// it to [`Input::via`](crate::Input::via) or
183/// [`Output::via`](crate::Output::via) when it builds a value of that type.
184#[derive(Clone, Debug)]
185pub struct ResolvedConversion {
186    target: TypeRef,
187    repr: TypeRef,
188    /// Representation → source.
189    input: Option<Stage>,
190    /// Source → representation.
191    output: Option<Stage>,
192}
193
194impl ResolvedConversion {
195    /// The representation type: what the source type crosses as.
196    pub fn repr(&self) -> &TypeRef {
197        &self.repr
198    }
199
200    /// The functions the conversion calls, so an adapter can count them as
201    /// used.
202    pub fn functions(&self) -> impl Iterator<Item = &FnRef> {
203        [&self.input, &self.output]
204            .into_iter()
205            .flatten()
206            .filter_map(|s| match &s.how {
207                How::Call { fun, .. } => Some(&**fun),
208                _ => None,
209            })
210    }
211
212    /// The source type being converted.
213    pub(crate) fn target(&self) -> &TypeRef {
214        &self.target
215    }
216
217    /// The conversion call in direction `dir` applied to `value`, and
218    /// whether it can fail. Fails when that direction is not declared.
219    pub(crate) fn apply(
220        &self,
221        dir: Direction,
222        q: &Qualifier<'_>,
223        value: &TokenStream,
224    ) -> Result<(TokenStream, bool), String> {
225        let (stage, what) = match dir {
226            Direction::In => (&self.input, "input"),
227            Direction::Out => (&self.output, "output"),
228        };
229        let stage = stage
230            .as_ref()
231            .ok_or_else(|| format!("convert!({}) declares no {what}", self.target))?;
232        Ok((
233            stage.apply(q, &self.target, &self.repr, value),
234            stage.fallible,
235        ))
236    }
237}

239/// Which way a conversion runs: into the source type, or out of it.
240#[derive(Clone, Copy, PartialEq, Eq, Debug)]
241pub(crate) enum Direction {
242    In,
243    Out,
244}

246/// One direction of a conversion: how it converts. The types it converts
247/// between are the conversion's.
248#[derive(Clone, Debug)]
249struct Stage {
250    fallible: bool,
251    how: How,
252}
253
254#[derive(Clone, Debug)]
255enum How {
256    /// Call a function; `by_ref` when it takes its argument by reference.
257    Call {
258        fun: Box<FnRef>,
259        by_ref: bool,
260    },
261    From,
262    Into,
263    TryFrom,
264    TryInto,
265}
266
267impl Stage {
268    /// Read one declared direction: the stage, and the representation type
269    /// it converts from (input) or to (output).
270    fn resolve(
271        flat: &Flat,
272        via: &Via,
273        target: &TypeRef,
274        dir: Direction,
275    ) -> Result<(Self, TypeRef), String> {
276        let classify = |t: &syn::Type| {
277            flat.classify(t)
278                .map_err(|e| format!("conversion type `{}`: {e}", t.to_token_stream()))
279        };
280        let stage = |fallible, how| Stage { fallible, how };
281        Ok(match via {
282            Via::Fn(fun) => {
283                let f = fun.resolve(flat)?;
284                let name = path_string(&fun.path);
285                let [param] = f.params.as_slice() else {
286                    return Err(format!(
287                        "conversion fn `{name}` must take exactly one argument, takes {}",
288                        f.params.len()
289                    ));
290                };
291                let (ret, fallible) = match f.ret.kind() {
292                    TypeKind::Fallible { ok, .. } => ((**ok).clone(), true),
293                    _ => (f.ret.clone(), false),
294                };
295                // The side facing the converted type must be that type: an
296                // input stage returns it, an output stage takes it (or a
297                // borrow of it). The other side is the representation.
298                let (facing, repr) = match dir {
299                    Direction::In => (&ret, param.ty.clone()),
300                    Direction::Out => match param.ty.kind() {
301                        TypeKind::Ref { inner, .. } => (&**inner, ret.clone()),
302                        _ => (&param.ty, ret.clone()),
303                    },
304                };
305                if facing.key() != target.key() {
306                    let (what, side) = match dir {
307                        Direction::In => ("input", "return"),
308                        Direction::Out => ("output", "take"),
309                    };
310                    return Err(format!(
311                        "{what} conversion fn `{name}` must {side} `{target}`, not `{facing}`"
312                    ));
313                }
314                let by_ref =
315                    dir == Direction::Out && matches!(param.ty.kind(), TypeKind::Ref { .. });
316                let fun = Box::new(fun.clone());
317                (stage(fallible, How::Call { fun, by_ref }), repr)
318            }
319            Via::From(t) => (stage(false, How::From), classify(t)?),
320            Via::Into(t) => (stage(false, How::Into), classify(t)?),
321            Via::TryFrom(t) => (stage(true, How::TryFrom), classify(t)?),
322            Via::TryInto(t) => (stage(true, How::TryInto), classify(t)?),
323        })
324    }
325
326    /// Apply the stage to `value`, between `target` and its representation
327    /// `repr`. For an input stage `value` is the representation and the
328    /// result the source type; for an output stage the other way round. A
329    /// fallible stage produces an expression using `?` on a
330    /// `Result<_, String>`.
331    fn apply(
332        &self,
333        q: &Qualifier<'_>,
334        target: &TypeRef,
335        repr: &TypeRef,
336        value: &TokenStream,
337    ) -> TokenStream {
338        let (t, r) = (q.ty(target), q.ty(repr));
339        let call = match &self.how {
340            How::Call { fun, by_ref } => {
341                let callee = fun.callee(q);
342                if *by_ref {
343                    quote!(#callee(&#value))
344                } else {
345                    quote!(#callee(#value))
346                }
347            }
348            How::From => quote!(<#t as ::core::convert::From<#r>>::from(#value)),
349            How::Into => quote!(<#t as ::core::convert::Into<#r>>::into(#value)),
350            How::TryFrom => quote!(<#t as ::core::convert::TryFrom<#r>>::try_from(#value)),
351            How::TryInto => quote!(<#t as ::core::convert::TryInto<#r>>::try_into(#value)),
352        };
353        if self.fallible {
354            quote!(#call.map_err(|__e| ::std::string::ToString::to_string(&__e))?)
355        } else {
356            call
357        }
358    }
