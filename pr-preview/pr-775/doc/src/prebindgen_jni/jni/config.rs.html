1//! Global settings of a [`JniGenBuilder`] instance — the target
2//! package, name-mangling rules, native-init hook and handle-lock toggle.
3//!
4//! Every setter carries the `set_` prefix: unlike the declaration methods
5//! ([`JniGenBuilder::package`], [`JniGenBuilder::convert`], …) which each add
6//! one item to the binding surface, a `set_` method changes how *all* other
7//! declarations are interpreted. Setters are **order-independent by
8//! construction**: the builder stores only raw inputs — settings here,
9//! declaration name specs ([`NameSpec`]) in the type table — and every
10//! setting-derived value (class FQN, `FindClass` path, JNI extern symbol
11//! path) is computed at the point of use, so there is no stored derived
12//! state that could go stale.
13//!
14//! # Name-mangle hooks
15//!
16//! Placement-aware hooks receive the fully-qualified Kotlin **package where
17//! the named object is emitted**, followed by the derived default name for its
18//! tier. The method hook additionally receives the final containing class
19//! short name. The harness is the exception: it has one fixed placement and
20//! receives only its default short name. Hooks return the final short name;
21//! unset hooks are identity.
22//!
23//! | hook | names | input (the derived default) | default |
24//! |---|---|---|---|
25//! | [`set_harness_name_mangle`](JniGenBuilder::set_harness_name_mangle) | the centralized externs object | `"JNINative"` | identity |
26//! | [`set_fun_name_mangle`](JniGenBuilder::set_fun_name_mangle) | top-level package functions | package, camelCased Rust fn name (`put_publisher` → `"putPublisher"`) | identity |
27//! | [`set_ptr_class_name_mangle`](JniGenBuilder::set_ptr_class_name_mangle) | `ptr_class` Kotlin classes | package, Rust type short name (`"KeyExpr"`) | identity |
28//! | [`set_data_class_name_mangle`](JniGenBuilder::set_data_class_name_mangle) | `data_class` Kotlin classes | package, Rust type short name | identity |
29//! | [`set_enum_name_mangle`](JniGenBuilder::set_enum_name_mangle) | `enum_class` Kotlin classes | package, Rust type short name | identity |
30//! | [`set_method_name_mangle`](JniGenBuilder::set_method_name_mangle) | class methods/factories and JNI extern methods | package, final class name, full camelCase Rust fn name | identity |
31//!
32//! One further hook does NOT follow the identity rule, because its input and
33//! output are two names that must differ:
34//!
35//! | hook | names | input | default |
36//! |---|---|---|---|
37//! | [`set_interface_name_mangle`](JniGenBuilder::set_interface_name_mangle) | the generated `.interface()` interface | package, final **class** name (`"Storage"`) | append `"Api"` (`"StorageApi"`); identity is a hard error |
38//!
39//! A per-decl `.name()` / `.interface_name()` override is always verbatim and
40//! **bypasses** the hooks entirely.
41

44/// Which per-kind mangle hook applies when deriving a declared class's
45/// Kotlin name.
46#[derive(Clone, Copy)]
47pub(crate) enum NameKind {
48    Ptr,
49    Enum,
50    Data,
51}
52
53/// Raw naming spec of one declared class type, stored in [`TypeConfig`] as
54/// declared and turned into a concrete Kotlin FQN only when read
55/// ([`JniGenBuilder::fqn_of`]), against whatever the settings are at that moment.
56#[derive(Clone)]
57pub(crate) struct NameSpec {
58    pub(crate) subpackage: String,
59    /// `rust_short_name(&key)` — the name part of the mangle hook's input;
60    /// `subpackage` supplies its package part.
61    pub(crate) short: String,
62    /// Per-decl `.name()` — resolved against package + subpackage,
63    /// bypasses the mangle hook.
64    pub(crate) name_override: Option<String>,
65    pub(crate) kind: NameKind,
66}

68impl JniGenBuilder {
69    /// Set the JVM/Kotlin **base** package (dot-separated, e.g.
70    /// `"io.zenoh.jni"`). All derived forms (slash-separated `FindClass`
71    /// paths, `_`-mangled JNI extern idents, Kotlin `package` declarations)
72    /// are computed from this. Empty = no prefix.
73    pub fn set_package_prefix(mut self, p: impl Into<String>) -> Self {
74        let trimmed = p.into().trim_matches('.').trim_matches('/').to_string();
75        // Sanitize each segment to a valid Kotlin identifier (issue #89); a
76        // no-op for already-legal package names. The base package is read
77        // directly by many emitters, so mangling at this single storage
78        // point keeps every reader consistent; warn here (the raw input is
79        // only available now) when a segment was changed.
80        self.decls.package = mangle_kotlin_package(&trimmed);
81        if self.decls.package != trimmed {
82            println!(
83                "cargo:warning=prebindgen: package prefix `{trimmed}` sanitized to `{}` \
84                 (invalid Kotlin package identifier)",
85                self.decls.package
86            );
87        }
88        self
89    }

91    /// Emit `code` inside an `init { … }` block of the generated centralized
92    /// externs object (`JNINative`). Because every generated native call
93    /// routes through that object, its static initializer is the single
94    /// point at which the consumer can trigger native-library loading
95    /// transparently — e.g.
96    /// `.set_jni_native_init("io.zenoh.jni.NativeLibrary.ensureLoaded()")` so
97    /// any call into the generated bindings loads the library first. The
98    /// referenced loader is the consumer's own (hand-written) code; this
99    /// keeps the generator free of any concrete loading logic. Unset = no
100    /// init block.
101    pub fn set_jni_native_init(mut self, code: impl Into<String>) -> Self {
102        self.decls.jni_native_init = Some(code.into());
103        self
104    }

106    /// Set the closure that renames the framework "harness" class (the
107    /// centralized extern holder). Receives the derived default `"JNINative"`
108    /// and replaces it wholesale (`|_| "MyNative".to_string()`); default = identity (see the
109    /// module-level table). Affects the generated Kotlin class name and
110    /// the derived JNI extern symbol path on the Rust side.
111    pub fn set_harness_name_mangle<F>(mut self, f: F) -> Self
112    where
113        F: Fn(&str) -> String + Send + Sync + 'static,
114    {
115        self.decls.harness_name_mangle = Some(Arc::new(f));
116        self
117    }
118
119    /// Set the closure that mangles top-level package function names. It
120    /// receives the fully-qualified target package and camelCased Rust name.
121    /// JNI externs are methods of the generated harness and therefore use
122    /// [`set_method_name_mangle`](Self::set_method_name_mangle) instead.
123    pub fn set_fun_name_mangle<F>(mut self, f: F) -> Self
124    where
125        F: Fn(&str, &str) -> String + Send + Sync + 'static,
126    {
127        self.decls.fun_name_mangle = Some(Arc::new(f));
128        self
129    }

131    /// Set the closure that turns a class name into the name of its generated
132    /// `.interface()` interface. Receives its package and the **final class
133    /// name**, and must return a DIFFERENT name — identity is a hard error (a class and its
134    /// interface cannot share a name in one package), the one deviation from
135    /// the uniform hook contract in the module-level table. Default when
136    /// unset = append `"Api"` (`Storage` → `StorageApi`). A per-decl
137    /// [`interface_name`](crate::PtrClassDecl::interface_name) wins over
138    /// the hook.
139    pub fn set_interface_name_mangle<F>(mut self, f: F) -> Self
140    where
141        F: Fn(&str, &str) -> String + Send + Sync + 'static,
142    {
143        self.decls.interface_name_mangle = Some(Arc::new(f));
144        self
145    }

181    /// Set the closure that mangles methods and companion factories. Receives
182    /// the fully-qualified target package, final containing class short name,
183    /// and the full camelCase Rust function name. It is also used for methods
184    /// of generated infrastructure classes (`JNINative`, handle `freePtr`,
185    /// vector helpers). Per-method `.name()` remains verbatim.
186    pub fn set_method_name_mangle<F>(mut self, f: F) -> Self
187    where
188        F: Fn(&str, &str, &str) -> String + Send + Sync + 'static,
189    {
190        self.decls.method_name_mangle = Some(Arc::new(f));
191        self
192    }
193
194    /// Toggle the per-call locking the generator wraps around every handle a
195    /// wrapper touches (which guards against a handle being `close()`d on
196    /// another thread mid-call). Defaults to `true`.
197    ///
198    /// The locks are **not** a performance trade-off: benchmarks (perftest,
199    /// N = 5M per op, JDK 21 / macOS arm64, 2026-07-15) show locks-on vs
200    /// locks-off deltas within run-to-run noise on every op that does real
201    /// work (±3%, mixed sign); the cheapest op (`put` with no string field,
202    /// ~33 ns/call) bounds the whole uncontended lock pair at about one
203    /// nanosecond. This setting exists so that claim can be independently
204    /// re-verified on your own workload — generate once with `false`,
205    /// benchmark both, and keep the default — not as an optimization knob.
206    pub fn set_emit_handle_locks(mut self, emit: bool) -> Self {
207        self.decls.emit_handle_locks = emit;
208        self
209    }

212impl Declarations {
213    /// Materialize a [`NameSpec`] into a concrete Kotlin FQN under the
214    /// current settings. Precedence for a declared class: per-decl
215    /// `name_override` (package-resolved, mangle-bypassed), then the mangle
216    /// hook for the class kind + package resolution.
217    pub(crate) fn fqn_of(&self, spec: &NameSpec) -> String {
218        let NameSpec {
219            subpackage,
220            short,
221            name_override,
222            kind,
223        } = spec;
224        if let Some(n) = name_override {
225            return self.resolve_class_fqn(subpackage, n);
226        }
227        let package = self.package_name(subpackage);
228        let mangled = match kind {
229            NameKind::Ptr => self.mangle_ptr_class(&package, short),
230            NameKind::Enum => self.mangle_enum(&package, short),
231            NameKind::Data => self.mangle_data_class(&package, short),
232        };
233        self.resolve_class_fqn(subpackage, &mangled)
234    }
235
236    /// The registered Kotlin FQN for a canonical Rust type key, derived on
237    /// demand from the type's stored [`NameSpec`]. A typed direct lookup —
238    /// declaration keys and probe keys share the [`TypeKey`] constructor, so
239    /// the former linear string scan is gone (issue #95).
240    pub(crate) fn kotlin_fqn(&self, key: &TypeKey) -> Option<String> {
241        self.types
242            .get(key)
243            .and_then(|cfg| cfg.name_spec.as_ref())
244            .map(|spec| self.fqn_of(spec))
245    }
246
247    /// Derived on demand: `package.replace('.', "/")` — the slash-separated
248    /// prefix `FindClass` strings are built from.
249    pub(crate) fn java_class_prefix(&self) -> String {
250        self.package.replace('.', "/")
251    }
252
253    /// Derived on demand: the spec-escaped JNI export symbol
254    /// `Java_<package>_<jni_native_class_name()>_<method>` for a method on
255    /// the centralized Native object every emitted wrapper hangs off
256    /// (see [`super::symbol`], #86).
257    pub(crate) fn native_method_symbol(&self, method: &str) -> String {
258        super::symbol::native_symbol(&self.package, &self.jni_native_class_name(), method)
259    }
