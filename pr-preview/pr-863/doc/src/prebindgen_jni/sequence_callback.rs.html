5pub(crate) fn transport(
6    wire: syn::Ident,
7    element: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
8    values: &[(syn::Ident, TokenStream)],
9) -> Result<SequenceOutput<JniWire, crate::JniChain>> {
10    let env = prebindgen_conversions::template_parameter("environment");
11    let method = format_ident!("{wire}_callback_method");
12    let class = format_ident!("{wire}_callback_class");
13    let frame = format_ident!("{wire}_callback_frame");
14    let mut descriptor = String::from("(");
15    let mut encode = TokenStream::new();
16    let mut args = Vec::new();
17    let mut needs_frame = false;
18    let slots: usize = crate::group::callback_parameters(element)
19        .map(|w| w.kind.parameter_slots())
20        .sum();
21    if slots > 254 {
22        return Err("sequence callback exceeds JVM parameter limit".into());
23    }
24    for (i, (field, value)) in prebindgen_conversions::match_output_fields(
25        crate::group::callback_parameters(element),
26        values,
27    )?
28    .into_iter()
29    .enumerate()
30    {
31        descriptor.push_str(field.kind.jvm_descriptor());
32        let local = format_ident!("{wire}_callback_object_{i}");
33        let argument = field.kind.output_argument(value, &local);
34        // This is the same classification used by output_argument to allocate
35        // a JVM object. Primitive wire values, including handles, create none.
36        needs_frame |= field.kind.is_reference();
37        encode.extend(argument.statements);
38        args.push(argument.value);
39    }
40    descriptor.push_str(")V");
41    let (enter, leave) = if needs_frame {
42        let capacity = i32::try_from(crate::group::callback_parameters(element).len() + 8)
43            .map_err(|_| "too many callback fields")?;
44        (
45            quote!(let #frame=unsafe{::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;),
46            quote!(drop(#frame);),
47        )
48    } else {
49        (TokenStream::new(), TokenStream::new())
50    };
51    let commit_element = prebindgen_conversions::template_parameter("commit_element");
52    Ok(SequenceOutput {
53        controls: vec![prebindgen_flat::Wire {
54            name: wire.clone(),
55            kind: JniWire::Rows {
56                condition: ("true").into(),
57                null_niche_available: true,
58            },
59        }],
60        commit_each: true,
61        fallible: true,
62        metadata: JniChain::Sequence(JniSequence::Callback),
63        begin: quote! {
64            // Every accepted row belongs to the Kotlin collector, including
65            // when a later row fails. Publish this before any row can arrive.
66            #env.set_boolean_array_region(&__pb_output_accepted,0,&[1]).map_err(|e|e.to_string())?;
67            let #class=#env.get_object_class(&__pb_callback).map_err(|e|e.to_string())?;
68            let #method=#env.get_method_id(&#class,"run",#descriptor).map_err(|e|e.to_string())?;
69            #env.delete_local_ref(#class).map_err(|e|e.to_string())?;
70        },
71        enter,
72        store: quote! {
73            #encode
74            // The method was resolved on this callback object with exactly these
75            // JVM wire types. The generated bridge only retains raw fields.
76            unsafe {#env.call_method_unchecked(&__pb_callback,#method,::prebindgen_jni_runtime::jni::signature::ReturnType::Primitive(::prebindgen_jni_runtime::jni::signature::Primitive::Void),&[#(#args),*])}.map_err(|e|e.to_string())?;
77            #commit_element
78            #leave
79        },
80        // Only a presence marker crosses back; row payloads crossed through run.
81        finish: quote!(let #wire=Some(::prebindgen_jni_runtime::array(&mut #env,0)?);),
82        values: vec![(wire.clone(), quote!(#wire))],
83        commit: quote!(),
84    })
85}

87pub(crate) struct Fold {
88    pub output: KotlinOutput,
89    pub parameter: String,
90    pub discard: String,
91}
92pub(crate) fn fold(
93    layout: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
94    context: &KotlinContext<'_>,
95) -> Result<Fold> {
96    fn optional(
97        node: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
98    ) -> bool {
99        match node {
100            prebindgen_flat::WireNode::Chain {
101                meta: JniChain::Optional(_),
102                ..
103            } => true,
104            prebindgen_flat::WireNode::Branch {
105                meta: JniBranch::Group(_),
106                children: elements,
107                ..
108            } => elements.iter().any(optional),
109            _ => false,
110        }
111    }
112    let optional = optional(layout);
113    let (carrier, element, _) =
114        crate::sequence::return_sequence(layout).ok_or("missing sequence callback layout")?;
115    let position = crate::group::result_packet_fields(layout)
116        .position(|w| w.name == carrier.name)
117        .unwrap();
118    let row = "__pb_fold_row";
119    let ctx = KotlinContext {
120        result: row,
121        output_prefix: "foldElement",
122        cleanup_failure: None,
123        ..*context
124    };
125    // Lexical row cleanup keeps partially constructed wrappers in scope. Only
126    // dynamic nested collections need the shared owner journal; fixed fields
127    // can use their existing locals without allocating a journal or a lambda.
128    fn nested_owners(node: &kotlin::JniNode) -> bool {
129        match node {
130            prebindgen_flat::WireNode::Chain {
131                meta: JniChain::Sequence(_),
132                child,
133                ..
134            } => kotlin::input_handles(child, false),
135            prebindgen_flat::WireNode::Chain { child, .. } => nested_owners(child),
136            prebindgen_flat::WireNode::Branch { children, .. } => {
137                children.iter().any(nested_owners)
138            }
139            _ => false,
140        }
141    }
142    let unread = kotlin::output_kotlin(element, &ctx)?;
143    let shared = nested_owners(element);
144    let owners = "__pb_foldOwners";
145    let child = if shared {
146        kotlin::output_kotlin_scoped(element, &ctx, Some(owners))?
147    } else {
148        unread.clone()
149    };
150    let present = format!("{}[{position}] != null", context.result);
151    let raw = "__pb_rows";
152    let types = child.callback_type_parameters()?;
153    let args = child.delivered.join(", ");
154    let failure = context.failure;
155    let resources = !unread.cleanup.is_empty();
156    let mut locals = String::from("var __pb_accumulator: A? = null\n");
157    let mut initialize = String::new();
158    let mut row = format!("{}\ncurrent = fold(current, {args})\n", child.materialize);
159    let mut discard = String::new();
160    let mut cleanup = String::new();
161    if resources {
162        locals.push_str("var __pb_processed = 0\n");
163        let release = if shared {
164            locals.push_str(&format!(
165                "var {owners}: {}.CollectionOwners? = null\n",
166                context.package
167            ));
168            initialize = format!(
169                "if ({raw}.isNotEmpty()) {owners} = {}.CollectionOwners()\n",
170                context.package
171            );
172            row = format!(
173                "{}\n{owners}!!.complete = true\ncurrent = fold(current, {args})\n",
174                child.materialize
175            );
176            format!(
177                "{owners}?.close({failure})\nif ({owners}?.complete != true) {{\n{}\n}}",
178                child.cleanup
179            )
180        } else {
181            child.cleanup.clone()
182        };
183        let delivered = if shared {
184            format!("{owners}!!.release()\n")
185        } else {
186            String::new()
187        };
188        // Account for the current row before cleanup, so a secondary cleanup
189        // failure cannot make the outer failure scan destroy it a second time.
190        row = format!("try {{\n{row}}} catch ({failure}: Throwable) {{\n__pb_processed++\ntry {{\n{release}\n}} catch (cleanup: Throwable) {{ if ({failure} !== cleanup) {failure}.addSuppressed(cleanup) }}\nthrow {failure}\n}}\n__pb_processed++\n{delivered}");
191        discard = format!("fun __pb_discard({failure}: Throwable, start: Int = 0) {{\nfor (index in start until {raw}.size) {{\ntry {{\nval __pb_fold_row = {raw}[index] as kotlin.Array<*>\n{}\n{}\n}} catch (cleanup: Throwable) {{ if ({failure} !== cleanup) {failure}.addSuppressed(cleanup) }}\n}}\n}}\n", unread.output_locals, unread.cleanup);
192        cleanup = format!("if ({present}) __pb_discard({failure}, __pb_processed)\n");
193    }
194    Ok(Fold {
195        parameter: format!("fold: (A, {types}) -> A"),
196        discard,
197        output: KotlinOutput {
198            result_names: vec![None],
199            delivered: vec![if optional { "__pb_accumulator".into() } else { "__pb_accumulator as A".into() }],
200            result_types: vec![if optional { "A?".into() } else { "A".into() }],
201            output_locals: locals,
202            materialize: format!("if ({present}) {{\n{initialize}var current = initial\nfor (rawRow in {raw}) {{\nval __pb_fold_row = rawRow as kotlin.Array<*>\n{}\n{row}}}\n__pb_accumulator = current\n}}\n", child.output_locals),
203            cleanup,
204        },
205    })
