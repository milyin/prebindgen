3/// Metadata-only input choices. Explicit conversions use structural node children.
4pub enum JniInputRequest {
5    /// Partial named enum fields; children are variant branches.
6    EnumFields,
7    /// One partial variant; the children correspond to these source field names.
8    VariantFields {
9        /// Source variant name.
10        variant: String,
11        /// Source field names corresponding to the structural children.
12        fields: Vec<String>,
13    },
14    /// Resolve the canonical value representation.
15    Value,
16    /// Select a local Rust value conversion.
17    Conversion(Box<DeclConvert>),
18    /// Resolve this occurrence's configured default.
19    Auto,
20    /// Partial callback metadata; domain/codomain own all conversion children.
21    Callback(Box<prebindgen_declarations::CallbackRequest>),
22    /// Apply the unary child at this source type, including nested occurrences.
23    ForType {
24        /// Source type at which the unary child applies.
25        ty: Box<Type>,
26    },
27    /// Infer constructor arguments from their defaults.
28    Call(String),
29}

31/// Metadata-only output choices. Explicit conversions use structural node children.
32pub enum JniOutputRequest {
33    /// Select per-element delivery of the unary child's resolved sequence.
34    SequenceCallback {
35        /// Permit decomposition of an ordinary collection identity.
36        structural: bool,
37    },
38    /// Partial named enum fields; children are variant branches.
39    EnumFields,
40    /// One partial variant; children correspond to these source field names.
41    VariantFields {
42        /// Source variant name.
43        variant: String,
44        /// Source field names corresponding to the structural children.
45        fields: Vec<String>,
46    },
47    /// Resolve the canonical value representation with optional projection policy.
48    Value {
49        /// Explicit ownership and field projection policy.
50        projection: Option<Projection>,
51    },
52    /// Select a local Rust value conversion.
53    Conversion(Box<DeclConvert>),
54    /// Resolve this occurrence's configured default.
55    Auto,
56    /// Apply the unary child at this source type, including nested occurrences.
57    ForType {
58        /// Source type at which the unary child applies.
59        ty: Box<Type>,
60    },
61    /// Partial record fields, corresponding to the branch children.
62    NamedFields {
63        /// Source field names corresponding to the structural children.
64        fields: Vec<String>,
65        /// Observe or move the selected fields.
66        mode: ProjectionMode,
67    },
68}
69/// Fluent conversion construction for this language.
70pub trait JniInputExt<InRequest, OutRequest>: Sized {
71    /// Group this explicit expansion into a bulk carrier.
72    fn bulk(self) -> Self;
73    /// Describe source record fields in declaration order.
74    fn fields(fields: impl IntoIterator<Item = JniInput<InRequest, OutRequest>>) -> Self;
75    /// Construct a source value with explicit argument conversions.
76    fn constructor(
77        name: impl Into<String>,
78        args: impl IntoIterator<Item = JniInput<InRequest, OutRequest>>,
79    ) -> Self;
80    /// Choose between owned representations of the same input type.
81    ///
82    /// `arms` holds one recipe per alternative; each must produce the input's
83    /// type. The caller passes the zero-based index of the chosen alternative in an
84    /// extra wire ahead of all arms' wires.
85    ///
86    /// Requires at least one arm. Each arm retains its own wires; this does not
87    /// merge equivalent wire parameters or generate destination-language overloads.
88    fn selector(arms: impl IntoIterator<Item = JniInput<InRequest, OutRequest>>) -> Self;
89    /// Describe source variants and their field conversions in source order.
90    fn enum_variants(variants: Vec<Vec<Self>>, value: bool) -> Self;
91    /// Read the field conversions of one enum variant.
92    fn variant_fields(&self) -> crate::Result<&[Self]>;
93    /// Convert each element of a source collection.
94    fn sequence(element: Self) -> Self;
95}
96impl<InRequest, OutRequest> JniInputExt<InRequest, OutRequest> for JniInput<InRequest, OutRequest> {
97    /// Group this explicit expansion into a bulk carrier.
98    fn bulk(self) -> Self {
99        JniInput::Unary {
100            operation: (JniInputUnary::Bulk).into(),
101            child: Box::new(self),
102        }
103    }
104    /// Construct a record from field recipes, or unit from an empty list.
105    ///
106    /// `fields` holds one recipe per field of the source record, in declaration
107    /// order. Binding verifies the count against the record.
108    fn fields(fields: impl IntoIterator<Item = JniInput<InRequest, OutRequest>>) -> Self {
109        JniInput::Branch {
110            operation: (JniInputBranch::Fields).into(),
111            children: fields.into_iter().collect(),
112        }
113    }
114    /// Call a named source constructor with the supplied argument recipes.
115    ///
116    /// Binding checks its result type and argument count. Argument validation and
117    /// ownership acquisition precede the source call.
118    ///
119    /// # Arguments
120    ///
121    /// * `name` — unqualified name of a source function whose result type is the
122    ///   input's type.
123    /// * `args` — one recipe per parameter of that function, in order.
124    fn constructor(
125        name: impl Into<String>,
126        args: impl IntoIterator<Item = JniInput<InRequest, OutRequest>>,
127    ) -> Self {
128        JniInput::Branch {
129            operation: (JniInputBranch::Constructor {
130                function: name.into(),
131            })
132            .into(),
133            children: args.into_iter().collect(),
134        }
135    }
136    /// Choose between owned representations of the same input type.
137    ///
138    /// `arms` holds one recipe per alternative; each must produce the input's
139    /// type. The caller passes the zero-based index of the chosen alternative in an
140    /// extra wire ahead of all arms' wires.
141    ///
142    /// Requires at least one arm. Each arm retains its own wires; this does not
143    /// merge equivalent wire parameters or generate destination-language overloads.
144    fn selector(arms: impl IntoIterator<Item = JniInput<InRequest, OutRequest>>) -> Self {
145        JniInput::Branch {
146            operation: (JniInputBranch::Selector).into(),
147            children: arms.into_iter().collect(),
148        }
149    }
150
151    /// Describe source variants, each with an ordinary branch of field conversions.
152    fn enum_variants(variants: Vec<Vec<Self>>, value: bool) -> Self {
153        Self::Branch {
154            operation: (JniInputBranch::Enum { value }).into(),
155            children: variants
156                .into_iter()
157                .map(|children| Self::Branch {
158                    operation: (JniInputBranch::Variant).into(),
159                    children,
160                })
161                .collect(),
162        }
163    }
164    /// Read an enum variant's field conversions.
165    fn variant_fields(&self) -> crate::Result<&[Self]> {
166        match self {
167            Self::Branch {
168                operation: prebindgen_declarations::Operation::Resolved(JniInputBranch::Variant),
169                children,
170            } => Ok(children),
171            _ => Err("enum children must be variant branches".into()),
172        }
173    }
174
175    /// Convert each incoming collection element using the supplied subtree.
176    fn sequence(element: Self) -> Self {
177        Self::Unary {
178            operation: (JniInputUnary::Sequence).into(),
179            child: Box::new(element),
180        }
181    }
182}

184/// Fluent conversion construction for this language.
185pub trait JniInputRequests: Sized {
186    /// Expand the tag and selected fields. Keys are source variant and field
187    /// names (or tuple indices). Unspecified fields inherit boundary defaults.
188    fn enum_fields<'a>(
189        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
190    ) -> Self;
191    /// Request the canonical value representation independently of expansion defaults.
192    fn value() -> Self;
193    /// Resolve the configured default for this source occurrence.
194    fn auto() -> Self;
195    /// Apply a conversion to the matching source type, including nested occurrences.
196    fn for_type(ty: Type, input: Self) -> Self;
197    /// Infer a source constructor's argument conversions from registered defaults.
198    fn call(name: impl Into<String>) -> Self;
199    /// Explicitly construct owned backing storage for a borrowed input.
200    fn reconstruct(self) -> Self;
201    /// Append a constructor alternative.
202    fn variant(self, name: impl Into<String>) -> Self;
203    /// Append an explicit input alternative.
204    fn add_alternative(self, arm: Self) -> Self;
205    /// Append the direct value representation as an input alternative.
206    fn variant_self(self) -> Self;
207}
208impl JniInputRequests for JniInput<JniInputRequest, JniOutputRequest> {
209    /// Expand the tag and selected fields. Keys are source variant and field
210    /// names (or tuple indices). Unspecified fields inherit boundary defaults.
211    fn enum_fields<'a>(
212        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
213    ) -> Self {
214        let overrides = prebindgen_declarations::enum_overrides(overrides);
215        JniInput::Branch {
216            operation: prebindgen_declarations::Operation::Request(JniInputRequest::EnumFields),
217            children: overrides
218                .into_iter()
219                .map(|(variant, fields)| {
220                    let (fields, children) = fields.into_iter().unzip();
221                    JniInput::Branch {
222                        operation: prebindgen_declarations::Operation::Request(
223                            JniInputRequest::VariantFields { variant, fields },
224                        ),
225                        children,
226                    }
227                })
228                .collect(),
229        }
230    }
231    /// Request the canonical value representation, bypassing boundary defaults.
232    fn value() -> Self {
233        JniInput::Request(JniInputRequest::Value)
234    }
235    /// Request this source type's configured default during preparation.
236    fn auto() -> Self {
237        JniInput::Request(JniInputRequest::Auto)
238    }
239    /// Associate a conversion with a source type for a default or nested override.
240    fn for_type(ty: Type, input: Self) -> Self {
241        JniInput::Unary {
242            operation: prebindgen_declarations::Operation::Request(JniInputRequest::ForType {
243                ty: Box::new(ty),
244            }),
245            child: Box::new(input),
246        }
247    }
248    /// Infer a source constructor's argument conversions from registered defaults.
249    fn call(name: impl Into<String>) -> Self {
250        JniInput::Request(JniInputRequest::Call(name.into()))
251    }
252    /// Explicitly reconstruct an owned temporary for a shared-reference input.
253    /// Slices construct a Vec, str constructs a String, and sized references
254    /// construct their referent. Child recipes keep their ordinary ownership
255    /// semantics; this operation authorizes the temporary's allocation/copying.
256    fn reconstruct(self) -> Self {
257        match self {
258            JniInput::Unary {
259                operation:
260                    prebindgen_declarations::Operation::Request(JniInputRequest::ForType { ty }),
261                child: input,
262            } => JniInput::for_type(
263                *ty,
264                JniInput::Unary {
265                    operation: (JniInputUnary::Temporary).into(),
266                    child: input,
267                },
268            ),
269            other => JniInput::Unary {
270                operation: (JniInputUnary::Temporary).into(),
271                child: Box::new(other),
272            },
273        }
274    }
275    /// Add a constructor alternative to this conversion tree.
276    fn variant(self, name: impl Into<String>) -> Self {
277        self.add_alternative(JniInput::call(name))
278    }
279    /// Add a direct codec alternative, bypassing this type's expansion default.
280    fn variant_self(self) -> Self {
281        self.add_alternative(JniInput::Leaf(JniInputLeafOp::Value))
282    }
283    fn add_alternative(self, arm: Self) -> Self {
284        match self {
285            JniInput::Unary {
286                operation:
287                    prebindgen_declarations::Operation::Request(JniInputRequest::ForType { ty }),
288                child: input,
289            } => JniInput::for_type(*ty, input.add_alternative(arm)),
290            JniInput::Branch {
291                operation: prebindgen_declarations::Operation::Resolved(JniInputBranch::Selector),
292                children: mut arms,
293            } => {
294                arms.push(arm);
295                JniInput::Branch {
296                    operation: (JniInputBranch::Selector).into(),
297                    children: arms,
298                }
299            }
300            other => JniInput::Branch {
301                operation: (JniInputBranch::Selector).into(),
302                children: vec![other, arm],
303            },
304        }
305    }
306}

308/// Fluent conversion construction for this language.
309pub trait JniOutputExt<InRequest, OutRequest>: Sized {
310    /// Group this explicit expansion into a bulk carrier.
311    fn bulk(self) -> Self;
312    /// Describe source record fields in declaration order.
313    fn fields(fields: impl IntoIterator<Item = JniOutput<InRequest, OutRequest>>) -> Self;
314    /// Observe a source getter and convert its result.
315    fn getter(name: impl Into<String>, output: JniOutput<InRequest, OutRequest>) -> Self;
316    /// Deliver multiple views of one source value.
317    fn combine(parts: impl IntoIterator<Item = JniOutput<InRequest, OutRequest>>) -> Self;
318    /// Group callable argument conversions in invocation order.
319    fn arguments(children: Vec<Self>) -> Self;
320    /// Read the source-ordered conversions of a callable domain.
321    fn argument_nodes(&self) -> crate::Result<&[Self]>;
322    /// Describe source variants, each with an ordinary branch of field conversions.
323    fn enum_variants(variants: Vec<Vec<Self>>, value: bool, mode: Option<ProjectionMode>) -> Self;
324    /// Read the field conversions of one enum variant.
325    fn variant_fields(&self) -> crate::Result<&[Self]>;
326    /// Set the ownership contract of one callable argument.
327    fn callback_argument(self, ownership: CallbackArgumentOwnership) -> Self;
328    /// Read an argument's conversion and ownership contract.
329    fn callback_argument_parts(&self) -> crate::Result<(CallbackArgumentOwnership, &Self)>;
330}
331impl<InRequest, OutRequest> JniOutputExt<InRequest, OutRequest>
332    for JniOutput<InRequest, OutRequest>
333{
334    /// Group this explicit expansion into a bulk carrier.
335    fn bulk(self) -> Self {
336        JniOutput::Unary {
337            operation: (JniOutputUnary::Bulk).into(),
338            child: Box::new(self),
339        }
340    }
341    /// Observe record fields in declaration order; use an empty list for unit.
342    ///
343    /// `fields` holds one recipe per field of the source record, in declaration order.
344    fn fields(fields: impl IntoIterator<Item = JniOutput<InRequest, OutRequest>>) -> Self {
345        JniOutput::Branch {
346            operation: (JniOutputBranch::Fields).into(),
347            children: fields.into_iter().collect(),
348        }
349    }
350    /// Observe a value through a source getter and convert the getter's result.
351    ///
352    /// The getter takes `&T` for the current type `T`; adapters retaining a final
353    /// output owner may also support `&mut T`. Its result uses the selected
354    /// output conversion, including supported reference results.
355    ///
356    /// # Arguments
357    ///
358    /// * `name` — unqualified name of the source getter.
359    /// * `output` — recipe for the getter's result.
360    fn getter(name: impl Into<String>, output: JniOutput<InRequest, OutRequest>) -> Self {
361        JniOutput::Unary {
362            operation: (JniOutputUnary::Getter {
363                function: name.into(),
364            })
365            .into(),
366            child: Box::new(output),
367        }
368    }
369    /// Produce multiple outputs from one value.
370    ///
371    /// `parts` holds one recipe per view of the value; their wires appear in this
372    /// order. Binding rejects an empty list or more than one consuming part. The
373    /// framework uses [`Codec::output_consumes`] to order observations before a move,
374    /// unless [`Codec::output_owner`] first retains the final output allocation.
375    fn combine(parts: impl IntoIterator<Item = JniOutput<InRequest, OutRequest>>) -> Self {
376        JniOutput::Branch {
377            operation: (JniOutputBranch::Combine).into(),
378            children: parts.into_iter().collect(),
379        }
380    }
381
382    /// Group a callable's domain in source parameter order.
383    fn arguments(children: Vec<Self>) -> Self {
384        if children.is_empty() {
385            Self::None
386        } else {
387            Self::Branch {
388                operation: (JniOutputBranch::Arguments).into(),
389                children,
390            }
391        }
392    }
393    /// Read the source-ordered components of a callable domain.
394    fn argument_nodes(&self) -> crate::Result<&[Self]> {
395        match self {
396            Self::None => Ok(&[]),
397            Self::Branch {
398                operation: prebindgen_declarations::Operation::Resolved(JniOutputBranch::Arguments),
399                children,
400            } => Ok(children),
401            _ => Err("callback domain requires an argument branch or None".into()),
402        }
403    }
404
405    /// Describe source variants, each with an ordinary branch of field conversions.
406    fn enum_variants(variants: Vec<Vec<Self>>, value: bool, mode: Option<ProjectionMode>) -> Self {
407        Self::Branch {
408            operation: (JniOutputBranch::Enum { value, mode }).into(),
409            children: variants
410                .into_iter()
411                .map(|children| Self::Branch {
412                    operation: (JniOutputBranch::Variant).into(),
413                    children,
414                })
415                .collect(),
416        }
417    }
418    /// Read an enum variant's field conversions.
419    fn variant_fields(&self) -> crate::Result<&[Self]> {
420        match self {
421            Self::Branch {
422                operation: prebindgen_declarations::Operation::Resolved(JniOutputBranch::Variant),
423                children,
424            } => Ok(children),
425            _ => Err("enum children must be variant branches".into()),
426        }
427    }
428
429    /// Attach the lifetime of produced foreign owners to this callable argument.
430    fn callback_argument(self, ownership: CallbackArgumentOwnership) -> Self {
431        Self::Unary {
432            operation: (JniOutputUnary::CallbackArgument { ownership }).into(),
433            child: Box::new(self),
434        }
435    }
436    /// Read an argument's conversion and ownership contract from its own subtree.
437    fn callback_argument_parts(&self) -> crate::Result<(CallbackArgumentOwnership, &Self)> {
438        match self {
439            Self::Unary {
440                operation:
441                    prebindgen_declarations::Operation::Resolved(JniOutputUnary::CallbackArgument {
442                        ownership,
443                    }),
444                child,
445            } => Ok((*ownership, child)),
446            _ => Err("callback argument requires an explicit ownership contract".into()),
447        }
448    }
449}

451/// Fluent conversion construction for this language.
452pub trait JniOutputRequests: Sized {
453    /// Expand the tag and selected fields. Keys are source variant and field
454    /// names (or tuple indices). Unspecified fields inherit boundary defaults.
455    fn enum_fields<'a>(
456        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
457    ) -> Self;
458    /// Request the canonical value representation independently of expansion defaults.
459    fn value() -> Self;
460    /// Resolve the configured default for this source occurrence.
461    fn auto() -> Self;
462    /// Apply a conversion to the matching source type, including nested occurrences.
463    fn for_type(ty: Type, output: Self) -> Self;
464    /// Append another simultaneous output view.
465    fn view(self, view: impl Into<Self>) -> Self;
466    /// Append a getter view using its default result conversion.
467    fn field(self, name: impl Into<String>) -> Self;
468    /// Append the direct value representation.
469    fn field_self(self) -> Self;
470    /// Append an explicitly borrowed projection.
471    fn field_self_ref(self) -> Self;
472    /// Append an explicitly owned projection.
473    fn field_self_into(self) -> Self;
474    /// Append a projection with explicit ownership and field policies.
475    fn project_self(self, policy: Projection) -> Self;
476    /// Append a consuming accessor followed by owned field decomposition.
477    fn fields_into(self, view: Self) -> Self;
478    /// Deprecated spelling of fields_into.
479    fn fields_self_into(self, view: Self) -> Self;
480    /// Override one named source field's conversion.
481    fn field_override(self, name: impl Into<String>, output: Self) -> Self;
482}
483impl JniOutputRequests for JniOutput<JniInputRequest, JniOutputRequest> {
484    /// Expand an enum's tag and active fields using source variant/member keys.
485    fn enum_fields<'a>(
486        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
487    ) -> Self {
488        let overrides = prebindgen_declarations::enum_overrides(overrides);
489        JniOutput::Branch {
490            operation: prebindgen_declarations::Operation::Request(JniOutputRequest::EnumFields),
491            children: overrides
492                .into_iter()
493                .map(|(variant, fields)| {
494                    let (fields, children) = fields.into_iter().unzip();
495                    JniOutput::Branch {
496                        operation: prebindgen_declarations::Operation::Request(
497                            JniOutputRequest::VariantFields { variant, fields },
498                        ),
499                        children,
500                    }
501                })
502                .collect(),
503        }
504    }
505    /// Request the canonical value representation, bypassing boundary defaults.
506    fn value() -> Self {
507        JniOutput::Request(JniOutputRequest::Value { projection: None })
508    }
509    /// Request this source type's configured output default during preparation.
510    fn auto() -> Self {
511        JniOutput::Request(JniOutputRequest::Auto)
512    }
513    /// Associate a conversion with a source type for a default or nested override.
514    fn for_type(ty: Type, output: Self) -> Self {
515        JniOutput::Unary {
516            operation: prebindgen_declarations::Operation::Request(JniOutputRequest::ForType {
517                ty: Box::new(ty),
518            }),
519            child: Box::new(output),
520        }
521    }
522    /// Append another simultaneous output view.
523    fn view(self, view: impl Into<Self>) -> Self {
524        let view = view.into();
525        match self {
526            JniOutput::Unary {
527                operation:
528                    prebindgen_declarations::Operation::Request(JniOutputRequest::ForType { ty }),
529                child: output,
530            } => JniOutput::for_type(*ty, output.view(view)),
531            JniOutput::Branch {
532                operation: prebindgen_declarations::Operation::Resolved(JniOutputBranch::Combine),
533                children: mut parts,
534            } => {
535                parts.push(view);
536                JniOutput::Branch {
537                    operation: (JniOutputBranch::Combine).into(),
538                    children: parts,
539                }
540            }
541            other => JniOutput::Branch {
542                operation: (JniOutputBranch::Combine).into(),
543                children: vec![other, view],
544            },
545        }
546    }
547    /// Append a source getter whose result uses its registered conversion default.
548    fn field(self, name: impl Into<String>) -> Self {
549        self.view(JniOutput::getter(name, JniOutput::auto()))
550    }
551    /// Append the direct codec representation, bypassing this type's expansion default.
552    fn field_self(self) -> Self {
553        self.view(JniOutput::Leaf(JniOutputLeafOp::Value))
554    }
555    /// Append an explicit borrowed projection.
556    fn field_self_ref(self) -> Self {
557        self.project_self(Projection::borrowed())
558    }
559    /// Append an explicit owned projection.
560    fn field_self_into(self) -> Self {
561        self.project_self(Projection::owned())
562    }
563    /// Append a projection with explicit ownership and nested field policies.
564    fn project_self(self, policy: Projection) -> Self {
565        self.view(JniOutput::Leaf(JniOutputLeafOp::Projected(policy)))
566    }
567    /// Append a consuming accessor followed by owned field decomposition.
568    fn fields_into(self, view: Self) -> Self {
569        self.view(owned_field_view(view))
570    }
571    /// Deprecated spelling of [`JniOutput::fields_into`].
572    fn fields_self_into(self, view: Self) -> Self {
573        self.fields_into(view)
574    }
575    /// Override one named source field, preserving defaults for its siblings.
576    fn field_override(self, name: impl Into<String>, output: Self) -> Self {
577        prebindgen_declarations::override_output_field(
578            self,
579            name.into(),
580            output,
581            |tree| match tree {
582                JniOutput::Branch {
583                    operation:
584                        prebindgen_declarations::Operation::Request(JniOutputRequest::NamedFields {
585                            fields,
586                            ..
587                        }),
588                    children,
589                } => Some((fields, children)),
590                _ => None,
591            },
592            |tree| match tree {
593                JniOutput::Unary {
594                    operation:
595                        prebindgen_declarations::Operation::Resolved(
596                            JniOutputUnary::Getter { .. } | JniOutputUnary::Into { .. },
597                        )
598                        | prebindgen_declarations::Operation::Request(JniOutputRequest::ForType {
599                            ..
600                        }),
601                    child,
602                } => Some(child),
603                _ => None,
604            },
605        )
606    }
607}

630pub(super) fn owned_field_view(
631    view: JniOutput<JniInputRequest, JniOutputRequest>,
632) -> JniOutput<JniInputRequest, JniOutputRequest> {
633    let JniOutput::Unary {
634        operation: prebindgen_declarations::Operation::Resolved(JniOutputUnary::Getter { function }),
635        child,
636    } = view
637    else {
638        panic!("fields_into requires a fields! accessor")
639    };
640    let JniOutput::Branch {
641        operation:
642            prebindgen_declarations::Operation::Request(JniOutputRequest::NamedFields {
643                fields, ..
644            }),
645        children,
646    } = *child
647    else {
648        panic!("fields_into requires named fields")
649    };
650    JniOutput::Unary {
651        operation: JniOutputUnary::Into { function }.into(),
652        child: Box::new(JniOutput::Branch {
653            operation: prebindgen_declarations::Operation::Request(JniOutputRequest::NamedFields {
654                fields,
655                mode: ProjectionMode::Owned,
656            }),
657            children,
658        }),
659    }
