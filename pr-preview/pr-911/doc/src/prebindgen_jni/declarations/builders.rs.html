2/// Input alternatives for a source type, built directly as a [`JniInput`] tree.
3#[derive(Clone, Debug)]
4pub struct DeclJniExpandParam {
5    /// Source type targeted by this default or occurrence override.
6    pub ty: syn::Type,
7    input: JniInput<JniInputRequest, JniOutputRequest>,
8    bulk: bool,
9    reconstruct: bool,
10}
11impl DeclJniExpandParam {
12    /// Start an input declaration; add at least one alternative before use.
13    pub fn new(ty: syn::Type) -> Self {
14        Self {
15            ty,
16            input: JniInput::Branch {
17                operation: JniInputBranch::Selector,
18                children: vec![],
19            },
20            bulk: false,
21            reconstruct: false,
22        }
23    }
24    /// Group this expansion into a C-representation bulk carrier.
25    /// The adapter chooses storage; constructor and local conversion rules are retained.
26    pub fn bulk(mut self) -> Self {
27        self.bulk = true;
28        self
29    }
30    /// Explicitly construct an owned temporary and borrow it for the source call.
31    /// With no alternatives, use the owned value's default input conversion.
32    /// Alternatives describe the owned referent, not the reference itself.
33    pub fn reconstruct(mut self) -> Self {
34        self.reconstruct = true;
35        self
36    }
37    /// Append a constructor reference; its arguments use their type defaults.
38    pub fn variant(mut self, function: impl Into<DeclJniFunction>) -> Self {
39        self.input = self.input.variant(function.into().reference_name(false));
40        self
41    }
42    /// Append the type's direct language representation.
43    pub fn variant_self(mut self) -> Self {
44        self.input = self.input.variant_self();
45        self
46    }
47}
48impl From<DeclJniExpandParam> for JniInput<JniInputRequest, JniOutputRequest> {
49    fn from(decl: DeclJniExpandParam) -> Self {
50        let mut input = decl.input;
51        if decl.reconstruct {
52            if matches!(&input, JniInput::Branch { operation: JniInputBranch::Selector, children: arms } if arms.is_empty())
53            {
54                input = JniInput::auto();
55            }
56            input = JniInput::Unary {
57                operation: JniInputUnary::Temporary,
58                child: Box::new(input),
59            };
60        }
61        if decl.bulk {
62            input = JniInput::Unary {
63                operation: JniInputUnary::Bulk,
64                child: Box::new(input),
65            };
66        }
67        JniInput::for_type(decl.ty, input)
68    }
69}

71/// Simultaneous views of a source return value, built directly as a [`JniOutput`] tree.
72#[derive(Clone, Debug)]
73pub struct DeclJniExpandReturn {
74    /// Source type targeted by this default or occurrence override.
75    pub ty: syn::Type,
76    output: JniOutput<JniInputRequest, JniOutputRequest>,
77    bulk: bool,
78}
79impl DeclJniExpandReturn {
80    /// Start an output declaration; add views before registering a type default.
81    pub fn new(ty: syn::Type) -> Self {
82        Self {
83            ty,
84            output: JniOutput::Branch {
85                operation: JniOutputBranch::Combine,
86                children: vec![],
87            },
88            bulk: false,
89        }
90    }
91    /// Group this expansion into a C-representation bulk carrier.
92    /// Getter, ownership and local conversion rules are retained. Ordinary
93    /// collections combine these expansions into one frame. Callback delivery
94    /// keeps the expansion within each callback element instead: bulk `Row`
95    /// produces one frame per row of `Vec<Row>`, while bulk `Vec<Row>` produces
96    /// one frame per inner collection of `Vec<Vec<Row>>`.
97    pub fn bulk(mut self) -> Self {
98        self.bulk = true;
99        self
100    }
101    /// Append a getter view. An explicit `.name()` labels the delivered view.
102    /// Shared results are observed before another view moves the owner. An
103    /// owning opaque copy requires an explicit projection or cloning accessor.
104    pub fn field(mut self, function: impl Into<DeclJniFunction>) -> Self {
105        let function = function.into();
106        let name = function.foreign.clone();
107        let getter = JniOutput::getter(function.reference_name(true), JniOutput::auto());
108        self.output = self.output.view(match name {
109            Some(name) => JniOutput::Unary {
110                operation: JniOutputUnary::Named { name },
111                child: Box::new(getter),
112            },
113            None => getter,
114        });
115        self
116    }
117    /// Append a getter's named record fields as separate output values.
118    pub fn fields(self, fields: DeclJniFields) -> Self {
119        self.view(fields)
120    }
121    /// Append another explicit output view.
122    pub fn view(mut self, output: impl Into<JniOutput<JniInputRequest, JniOutputRequest>>) -> Self {
123        self.output = self.output.view(output.into());
124        self
125    }
126    /// Append a consuming accessor followed by owned field decomposition.
127    pub fn fields_into(mut self, fields: DeclJniFields) -> Self {
128        self.output = self.output.fields_into(fields.into());
129        self
130    }
131    /// Deprecated spelling of [`Self::fields_into`].
132    #[deprecated(note = "use fields_into instead")]
133    pub fn fields_self_into(self, fields: DeclJniFields) -> Self {
134        self.fields_into(fields)
135    }
136    /// Append the language's direct representation, bypassing this type's default.
137    pub fn field_self(mut self) -> Self {
138        self.output = self.output.field_self();
139        self
140    }
141    /// Append an explicit shared projection. Owning opaque representations clone
142    /// the value; record/collection projections permit copying borrowed leaves.
143    pub fn field_self_ref(self) -> Self {
144        self.project_self(Projection::borrowed())
145    }
146    /// Append an explicit owned projection.
147    pub fn field_self_into(self) -> Self {
148        self.project_self(Projection::owned())
149    }
150    /// Append a projection with explicit ownership and field policies.
151    pub fn project_self(mut self, policy: Projection) -> Self {
152        self.output = self.output.project_self(policy);
153        self
154    }
155}
156impl From<DeclJniExpandReturn> for JniOutput<JniInputRequest, JniOutputRequest> {
157    fn from(decl: DeclJniExpandReturn) -> Self {
158        JniOutput::for_type(
159            decl.ty,
160            if decl.bulk {
161                JniOutput::Unary {
162                    operation: JniOutputUnary::Bulk,
163                    child: Box::new(decl.output),
164                }
165            } else {
166                decl.output
167            },
168        )
169    }
170}

172/// A source accessor followed by named field decomposition.
173/// Field adjustments edit the accessor's output tree directly.
174#[derive(Clone, Debug)]
175pub struct DeclJniFields {
176    function: String,
177    overrides: BTreeMap<String, JniOutput<JniInputRequest, JniOutputRequest>>,
178}
179impl DeclJniFields {
180    /// Select a getter whose returned record supplies the fields.
181    pub fn new(function: syn::Ident) -> Self {
182        Self {
183            function: function.to_string(),
184            overrides: BTreeMap::new(),
185        }
186    }
187    /// Replace one field's conversion. An empty declaration omits the whole field,
188    /// including any enclosing Option or sequence representation.
189    pub fn field(self, name: impl AsRef<str>, output: DeclJniExpandReturn) -> Self {
190        self.field_override(name.as_ref().to_owned(), output)
191    }
192    /// Set an explicit conversion tree for one field, with the same omission
193    /// semantics as [`Self::field`] for an empty typed declaration.
194    pub fn field_override(
195        mut self,
196        name: impl Into<String>,
197        output: impl Into<JniOutput<JniInputRequest, JniOutputRequest>>,
198    ) -> Self {
199        let output = match output.into() {
200            JniOutput::Request(JniOutputRequest::ForType { ty, output }) if matches!(output.as_ref(),JniOutput::Branch { operation: JniOutputBranch::Combine, children: parts } if parts.is_empty()) => {
201                JniOutput::for_type(*ty, JniOutput::Leaf(JniOutputLeafOp::Discard))
202            }
203            output => output,
204        };
205        assert!(
206            self.overrides.insert(name.into(), output).is_none(),
207            "duplicate field override"
208        );
209        self
210    }
211}
212impl From<DeclJniFields> for JniOutput<JniInputRequest, JniOutputRequest> {
213    fn from(fields: DeclJniFields) -> Self {
214        JniOutput::getter(
215            fields.function,
216            JniOutput::Request(JniOutputRequest::NamedFields {
217                overrides: fields.overrides,
218                mode: prebindgen_declarations::ProjectionMode::Borrowed,
219            }),
220        )
221    }
222}

224/// A type-wide source-error message mapping.
225#[derive(Clone, Debug)]
226pub struct DeclJniError {
227    /// Source error type.
228    pub ty: syn::Type,
229    /// Source helper taking E or &E and returning String.
230    pub mapper: DeclJniFunction,
231}
232impl DeclJniError {
233    /// Select the source helper for this error type.
234    pub fn new(ty: syn::Type, mapper: DeclJniFunction) -> Self {
235        Self { ty, mapper }
236    }
237}
238/// A type-wide input, output, or error-mapping declaration.
239/// Generators immediately consume these builders into their conversion trees.
240#[derive(Clone, Debug)]
241pub enum DeclJniExpand {
242    /// Input alternatives.
243    Param(DeclJniExpandParam),
244    /// Simultaneous output views.
245    Return(DeclJniExpandReturn),
246    /// A source-error message mapping.
247    Error(Box<DeclJniError>),
248}
249impl From<DeclJniExpandParam> for DeclJniExpand {
250    fn from(value: DeclJniExpandParam) -> Self {
251        Self::Param(value)
252    }
253}
254impl From<DeclJniExpandReturn> for DeclJniExpand {
255    fn from(value: DeclJniExpandReturn) -> Self {
256        Self::Return(value)
257    }
258}
259impl From<DeclJniError> for DeclJniExpand {
260    fn from(value: DeclJniError) -> Self {
261        Self::Error(Box::new(value))
262    }
263}

265/// Callback declaration containing this language's argument and result trees.
266/// ```
267/// use prebindgen_jni::{callback,expand_param,expand_return,fun};
268/// let handler = callback!(impl Fn(Packet) -> Decision)
269///     .name("PacketHandler")
270///     .expand_param(0, expand_return!(Packet).field(fun!(packet_payload)))
271///     .expand_return(expand_param!(Decision).variant(fun!(decision_new)));
272/// let _ = fun!(register).expand_param("callback", handler);
273/// ```
274pub type DeclJniCallback = prebindgen_declarations::DeclCallback<
275    JniInput<JniInputRequest, JniOutputRequest>,
276    JniOutput<JniInputRequest, JniOutputRequest>,
277>;
278impl From<DeclJniCallback> for JniInputRequest {
279    fn from(decl: DeclJniCallback) -> Self {
280        Self::ForType {
281            ty: Box::new(decl.ty.clone()),
282            input: Box::new(JniInput::Request(Self::Callback(Box::new(decl)))),
283        }
284    }
