2/// Input alternatives for a source type, built directly as a [`JniInput`] tree.
3#[derive(Clone, Debug)]
4pub struct DeclJniExpandParam {
5    /// Source type targeted by this default or occurrence override.
6    pub ty: syn::Type,
7    input:
8        prebindgen_declarations::DeclInputAlternatives<JniInput<JniInputRequest, JniOutputRequest>>,
9    bulk: bool,
10}
11impl DeclJniExpandParam {
12    /// Start an input declaration; add at least one alternative before use.
13    pub fn new(ty: syn::Type) -> Self {
14        Self {
15            ty,
16            input: Default::default(),
17            bulk: false,
18        }
19    }
20    /// Group this expansion into a C-representation bulk carrier.
21    /// The adapter chooses storage; constructor and local conversion rules are retained.
22    pub fn bulk(mut self) -> Self {
23        self.bulk = true;
24        self
25    }
26    /// Explicitly construct an owned temporary and borrow it for the source call.
27    /// With no alternatives, use the owned value's default input conversion.
28    /// Alternatives describe the owned referent, not the reference itself.
29    pub fn reconstruct(mut self) -> Self {
30        self.input.reconstruct();
31        self
32    }
33    /// Append a constructor reference; its arguments use their type defaults.
34    pub fn variant(mut self, function: impl Into<DeclJniFunction>) -> Self {
35        self.input
36            .push(JniInput::call(function.into().reference_name(false)));
37        self
38    }
39    /// Append the type's direct language representation.
40    pub fn variant_self(mut self) -> Self {
41        self.input.push(JniInput::Leaf(JniInputLeafOp::Value));
42        self
43    }
44}
45impl From<DeclJniExpandParam> for JniInput<JniInputRequest, JniOutputRequest> {
46    fn from(decl: DeclJniExpandParam) -> Self {
47        let mut input = decl
48            .input
49            .build(JniInput::selector, JniInput::auto, |input| {
50                JniInput::Unary {
51                    operation: JniInputUnary::Temporary,
52                    child: Box::new(input),
53                }
54            });
55        if decl.bulk {
56            input = JniInput::Unary {
57                operation: JniInputUnary::Bulk,
58                child: Box::new(input),
59            };
60        }
61        JniInput::for_type(decl.ty, input)
62    }
63}

65/// Simultaneous views of a source return value, built directly as a [`JniOutput`] tree.
66#[derive(Clone, Debug)]
67pub struct DeclJniExpandReturn {
68    /// Source type targeted by this default or occurrence override.
69    pub ty: syn::Type,
70    output: Vec<JniOutput<JniInputRequest, JniOutputRequest>>,
71    bulk: bool,
72}
73impl DeclJniExpandReturn {
74    /// Start an output declaration; add views before registering a type default.
75    pub fn new(ty: syn::Type) -> Self {
76        Self {
77            ty,
78            output: Default::default(),
79            bulk: false,
80        }
81    }
82    /// Group this expansion into a C-representation bulk carrier.
83    /// Getter, ownership and local conversion rules are retained. Ordinary
84    /// collections combine these expansions into one frame. Callback delivery
85    /// keeps the expansion within each callback element instead: bulk `Row`
86    /// produces one frame per row of `Vec<Row>`, while bulk `Vec<Row>` produces
87    /// one frame per inner collection of `Vec<Vec<Row>>`.
88    pub fn bulk(mut self) -> Self {
89        self.bulk = true;
90        self
91    }
92    /// Append a getter view. An explicit `.name()` labels the delivered view.
93    /// Shared results are observed before another view moves the owner. An
94    /// owning opaque copy requires an explicit projection or cloning accessor.
95    pub fn field(mut self, function: impl Into<DeclJniFunction>) -> Self {
96        self.output.push(prebindgen_declarations::getter_view(
97            function.into(),
98            |name| JniOutput::getter(name, JniOutput::auto()),
99            |name, output| JniOutput::Unary {
100                operation: JniOutputUnary::Named { name },
101                child: Box::new(output),
102            },
103        ));
104        self
105    }
106    /// Append a getter's named record fields as separate output values.
107    pub fn fields(self, fields: DeclJniFields) -> Self {
108        self.view(fields)
109    }
110    /// Append another explicit output view.
111    pub fn view(mut self, output: impl Into<JniOutput<JniInputRequest, JniOutputRequest>>) -> Self {
112        self.output.push(output.into());
113        self
114    }
115    /// Append a consuming accessor followed by owned field decomposition.
116    pub fn fields_into(mut self, fields: DeclJniFields) -> Self {
117        self.output
118            .push(super::requests::owned_field_view(fields.into()));
119        self
120    }
121    /// Deprecated spelling of [`Self::fields_into`].
122    #[deprecated(note = "use fields_into instead")]
123    pub fn fields_self_into(self, fields: DeclJniFields) -> Self {
124        self.fields_into(fields)
125    }
126    /// Append the language's direct representation, bypassing this type's default.
127    pub fn field_self(mut self) -> Self {
128        self.output.push(JniOutput::Leaf(JniOutputLeafOp::Value));
129        self
130    }
131    /// Append an explicit shared projection. Owning opaque representations clone
132    /// the value; record/collection projections permit copying borrowed leaves.
133    pub fn field_self_ref(self) -> Self {
134        self.project_self(Projection::borrowed())
135    }
136    /// Append an explicit owned projection.
137    pub fn field_self_into(self) -> Self {
138        self.project_self(Projection::owned())
139    }
140    /// Append a projection with explicit ownership and field policies.
141    pub fn project_self(mut self, policy: Projection) -> Self {
142        self.output
143            .push(JniOutput::Leaf(JniOutputLeafOp::Projected(policy)));
144        self
145    }
146}
147impl From<DeclJniExpandReturn> for JniOutput<JniInputRequest, JniOutputRequest> {
148    fn from(decl: DeclJniExpandReturn) -> Self {
149        JniOutput::for_type(
150            decl.ty,
151            if decl.bulk {
152                JniOutput::Unary {
153                    operation: JniOutputUnary::Bulk,
154                    child: Box::new(JniOutput::combine(decl.output)),
155                }
156            } else {
157                JniOutput::combine(decl.output)
158            },
159        )
160    }
161}

163/// A source accessor followed by named field decomposition.
164/// Field adjustments edit the accessor's output tree directly.
165#[derive(Clone, Debug)]
166pub struct DeclJniFields {
167    function: String,
168    overrides: BTreeMap<String, JniOutput<JniInputRequest, JniOutputRequest>>,
169}
170impl DeclJniFields {
171    /// Select a getter whose returned record supplies the fields.
172    pub fn new(function: syn::Ident) -> Self {
173        Self {
174            function: function.to_string(),
175            overrides: BTreeMap::new(),
176        }
177    }
178    /// Replace one field's conversion. An empty declaration omits the whole field,
179    /// including any enclosing Option or sequence representation.
180    pub fn field(self, name: impl AsRef<str>, output: DeclJniExpandReturn) -> Self {
181        self.field_override(name.as_ref().to_owned(), output)
182    }
183    /// Set an explicit conversion tree for one field, with the same omission
184    /// semantics as [`Self::field`] for an empty typed declaration.
185    pub fn field_override(
186        mut self,
187        name: impl Into<String>,
188        output: impl Into<JniOutput<JniInputRequest, JniOutputRequest>>,
189    ) -> Self {
190        let output = match output.into() {
191            JniOutput::Request(JniOutputRequest::ForType { ty, output }) if matches!(output.as_ref(), JniOutput::Branch { operation: JniOutputBranch::Combine, children } if children.is_empty()) => {
192                JniOutput::for_type(*ty, JniOutput::Leaf(JniOutputLeafOp::Discard))
193            }
194            output => output,
195        };
196        assert!(
197            self.overrides.insert(name.into(), output).is_none(),
198            "duplicate field override"
199        );
200        self
201    }
202}
203impl From<DeclJniFields> for JniOutput<JniInputRequest, JniOutputRequest> {
204    fn from(fields: DeclJniFields) -> Self {
205        JniOutput::getter(
206            fields.function,
207            JniOutput::Request(JniOutputRequest::NamedFields {
208                overrides: fields.overrides,
209                mode: prebindgen_declarations::ProjectionMode::Borrowed,
210            }),
211        )
212    }
213}

215/// A type-wide source-error mapping using this adapter's function declaration.
216pub type DeclJniError = prebindgen_declarations::DeclError<
217    JniInput<JniInputRequest, JniOutputRequest>,
218    JniOutput<JniInputRequest, JniOutputRequest>,

220/// A type-wide input, output, or error-mapping declaration.
221/// Generators immediately consume these builders into their conversion trees.
222#[derive(Clone, Debug)]
223pub enum DeclJniExpand {
224    /// Input alternatives.
225    Param(DeclJniExpandParam),
226    /// Simultaneous output views.
227    Return(DeclJniExpandReturn),
228    /// A source-error message mapping.
229    Error(Box<DeclJniError>),
230}
231impl From<DeclJniExpandParam> for DeclJniExpand {
232    fn from(value: DeclJniExpandParam) -> Self {
233        Self::Param(value)
234    }
235}
236impl From<DeclJniExpandReturn> for DeclJniExpand {
237    fn from(value: DeclJniExpandReturn) -> Self {
238        Self::Return(value)
239    }
240}
241impl From<DeclJniError> for DeclJniExpand {
242    fn from(value: DeclJniError) -> Self {
243        Self::Error(Box::new(value))
244    }
245}

247/// Callback declaration containing this language's argument and result trees.
248/// ```
249/// use prebindgen_jni::{callback,expand_param,expand_return,fun};
250/// let handler = callback!(impl Fn(Packet) -> Decision)
251///     .name("PacketHandler")
252///     .expand_param(0, expand_return!(Packet).field(fun!(packet_payload)))
253///     .expand_return(expand_param!(Decision).variant(fun!(decision_new)));
254/// let _ = fun!(register).expand_param("callback", handler);
255/// ```
256pub type DeclJniCallback = prebindgen_declarations::DeclCallback<
257    JniInput<JniInputRequest, JniOutputRequest>,
258    JniOutput<JniInputRequest, JniOutputRequest>,
259>;
260impl From<DeclJniCallback> for JniInputRequest {
261    fn from(decl: DeclJniCallback) -> Self {
262        Self::ForType {
263            ty: Box::new(decl.ty.clone()),
264            input: Box::new(JniInput::Request(Self::Callback(Box::new(decl)))),
265        }
266    }
