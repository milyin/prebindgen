3/// Source-dependent input choices resolved before compilation.
4pub enum CInputRequest {
5    /// Expand enum fields, inheriting defaults for fields not explicitly overridden.
6    EnumFields(
7        std::collections::BTreeMap<
8            String,
9            std::collections::BTreeMap<String, CInput<CInputRequest, COutputRequest>>,
10        >,
11    ),
12    /// Resolve only the canonical value form, independently of boundary defaults.
13    Value,
14    /// Select a local single-value conversion.
15    Conversion(Box<DeclConvert>),
16    /// Select the configured default for this source occurrence.
17    Auto,
18    /// Select or override a callback signature's argument and result conversions.
19    Callback(Box<DeclCCallback>),
20    /// Apply a type-specific override, including inside optional/sequence sources.
21    ForType {
22        /// Source type targeted by the override.
23        ty: Box<Type>,
24        /// The requested conversion for that type.
25        input: Box<CInput<CInputRequest, COutputRequest>>,
26    },
27    /// Infer a constructor's argument conversions from its source signature.
28    Call(String),
29}

31/// Source-dependent output choices resolved before compilation.
32pub enum COutputRequest {
33    /// Select per-element delivery after resolving defaults and local expansion.
34    SequenceCallback {
35        /// Expansion whose outer sequence supplies callback elements.
36        output: Box<COutput<CInputRequest, COutputRequest>>,
37        /// Allow structural decomposition of a default collection identity.
38        structural: bool,
39    },
40    /// Expand enum fields, inheriting boundary defaults for unspecified fields.
41    EnumFields(
42        std::collections::BTreeMap<
43            String,
44            std::collections::BTreeMap<String, COutput<CInputRequest, COutputRequest>>,
45        >,
46    ),
47    /// Resolve only the canonical value form, independently of boundary defaults.
48    Value {
49        /// Explicit field ownership/projection policy, if any.
50        projection: Option<Projection>,
51    },
52    /// Select a local single-value conversion.
53    Conversion(Box<DeclConvert>),
54    /// Select the configured output default for this source occurrence.
55    Auto,
56    /// Apply a type-specific override, including inside optional/sequence sources.
57    ForType {
58        /// Source type targeted by the override.
59        ty: Box<Type>,
60        /// Requested conversion for the matching type.
61        output: Box<COutput<CInputRequest, COutputRequest>>,
62    },
63    /// Resolve source record fields by name and fill unspecified conversions.
64    NamedFields {
65        /// Explicit conversions keyed by source field name.
66        overrides: std::collections::BTreeMap<String, COutput<CInputRequest, COutputRequest>>,
67        /// Ownership of field decomposition.
68        mode: ProjectionMode,
69    },
70}
71/// Fluent conversion construction for this language.
72pub trait CInputExt<InRequest, OutRequest>: Sized {
73    /// Describe source record fields in declaration order.
74    fn fields(fields: impl IntoIterator<Item = CInput<InRequest, OutRequest>>) -> Self;
75    /// Construct a source value with explicit argument conversions.
76    fn constructor(
77        name: impl Into<String>,
78        args: impl IntoIterator<Item = CInput<InRequest, OutRequest>>,
79    ) -> Self;
80    /// Choose between alternative input conversions.
81    fn selector(arms: impl IntoIterator<Item = CInput<InRequest, OutRequest>>) -> Self;
82    /// Describe source variants and their field conversions in source order.
83    fn enum_variants(variants: Vec<Vec<Self>>, value: bool) -> Self;
84    /// Read the field conversions of one enum variant.
85    fn variant_fields(&self) -> crate::Result<&[Self]>;
86    /// Convert each element of a source collection.
87    fn sequence(element: Self) -> Self;
88}
89impl<InRequest, OutRequest> CInputExt<InRequest, OutRequest> for CInput<InRequest, OutRequest> {
90    /// Construct a record from field recipes, or unit from an empty list.
91    ///
92    /// `fields` holds one recipe per field of the source record, in declaration
93    /// order. Binding verifies the count against the record.
94    fn fields(fields: impl IntoIterator<Item = CInput<InRequest, OutRequest>>) -> Self {
95        CInput::Branch {
96            operation: CInputBranch::Fields,
97            children: fields.into_iter().collect(),
98        }
99    }
100    /// Call a named source constructor with the supplied argument recipes.
101    ///
102    /// Binding checks its result type and argument count. Argument validation and
103    /// ownership acquisition precede the source call.
104    ///
105    /// # Arguments
106    ///
107    /// * `name` — unqualified name of a source function whose result type is the
108    ///   input's type.
109    /// * `args` — one recipe per parameter of that function, in order.
110    fn constructor(
111        name: impl Into<String>,
112        args: impl IntoIterator<Item = CInput<InRequest, OutRequest>>,
113    ) -> Self {
114        CInput::Branch {
115            operation: CInputBranch::Constructor {
116                function: name.into(),
117            },
118            children: args.into_iter().collect(),
119        }
120    }
121    /// Choose between owned representations of the same input type.
122    ///
123    /// `arms` holds one recipe per alternative; each must produce the input's
124    /// type. The caller passes the zero-based index of the chosen alternative in an
125    /// extra wire ahead of all arms' wires.
126    ///
127    /// Requires at least one arm. Each arm retains its own wires; this does not
128    /// merge equivalent wire parameters or generate destination-language overloads.
129    fn selector(arms: impl IntoIterator<Item = CInput<InRequest, OutRequest>>) -> Self {
130        CInput::Branch {
131            operation: CInputBranch::Selector,
132            children: arms.into_iter().collect(),
133        }
134    }
135    /// Describe source variants, each with an ordinary branch of field conversions.
136    fn enum_variants(variants: Vec<Vec<Self>>, value: bool) -> Self {
137        Self::Branch {
138            operation: CInputBranch::Enum { value },
139            children: variants
140                .into_iter()
141                .map(|children| Self::Branch {
142                    operation: CInputBranch::Variant,
143                    children,
144                })
145                .collect(),
146        }
147    }
148    /// Read an enum variant's field conversions.
149    fn variant_fields(&self) -> crate::Result<&[Self]> {
150        match self {
151            Self::Branch {
152                operation: CInputBranch::Variant,
153                children,
154            } => Ok(children),
155            _ => Err("enum children must be variant branches".into()),
156        }
157    }
158    /// Convert each incoming collection element using the supplied subtree.
159    fn sequence(element: Self) -> Self {
160        Self::Unary {
161            operation: CInputUnary::Sequence,
162            child: Box::new(element),
163        }
164    }
165}
166/// Fluent conversion construction for this language.
167pub trait CInputRequests: Sized {
168    /// Expand source enum fields with local variant/member overrides.
169    fn enum_fields<'a>(
170        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
171    ) -> Self;
172    /// Request the canonical value representation independently of expansion defaults.
173    fn value() -> Self;
174    /// Resolve the configured default for this source occurrence.
175    fn auto() -> Self;
176    /// Apply a conversion to the matching source type, including nested occurrences.
177    fn for_type(ty: Type, input: Self) -> Self;
178    /// Infer a source constructor's arguments from their defaults.
179    fn call(name: impl Into<String>) -> Self;
180    /// Explicitly construct owned backing storage for a borrowed input.
181    fn reconstruct(self) -> Self;
182    /// Append a constructor alternative.
183    fn variant(self, name: impl Into<String>) -> Self;
184    /// Append an explicit input alternative.
185    fn add_alternative(self, arm: Self) -> Self;
186    /// Append the direct value representation as an input alternative.
187    fn variant_self(self) -> Self;
188}
189impl CInputRequests for CInput<CInputRequest, COutputRequest> {
190    /// Expand the tag and selected fields. Keys are source variant and field
191    /// names (or tuple indices). Unspecified fields inherit boundary defaults.
192    fn enum_fields<'a>(
193        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
194    ) -> Self {
195        CInput::Request(CInputRequest::EnumFields(
196            prebindgen_declarations::enum_overrides(overrides),
197        ))
198    }
199    /// Request the canonical value representation, bypassing boundary defaults.
200    fn value() -> Self {
201        CInput::Request(CInputRequest::Value)
202    }
203    /// Request this source type's configured default during preparation.
204    fn auto() -> Self {
205        CInput::Request(CInputRequest::Auto)
206    }
207    /// Associate a conversion with a source type for a default or nested override.
208    fn for_type(ty: Type, input: Self) -> Self {
209        CInput::Request(CInputRequest::ForType {
210            ty: Box::new(ty),
211            input: Box::new(input),
212        })
213    }
214    /// Infer a source constructor's argument conversions from registered defaults.
215    fn call(name: impl Into<String>) -> Self {
216        CInput::Request(CInputRequest::Call(name.into()))
217    }
218    /// Explicitly reconstruct an owned temporary for a shared-reference input.
219    /// Slices construct a Vec, str constructs a String, and sized references
220    /// construct their referent. Child recipes keep their ordinary ownership
221    /// semantics; this operation authorizes the temporary's allocation/copying.
222    fn reconstruct(self) -> Self {
223        match self {
224            CInput::Request(CInputRequest::ForType { ty, input }) => CInput::for_type(
225                *ty,
226                CInput::Unary {
227                    operation: CInputUnary::Temporary,
228                    child: input,
229                },
230            ),
231            other => CInput::Unary {
232                operation: CInputUnary::Temporary,
233                child: Box::new(other),
234            },
235        }
236    }
237    /// Add a constructor alternative to this conversion tree.
238    fn variant(self, name: impl Into<String>) -> Self {
239        self.add_alternative(CInput::call(name))
240    }
241    /// Add a direct codec alternative, bypassing this type's expansion default.
242    fn variant_self(self) -> Self {
243        self.add_alternative(CInput::Leaf(CInputLeafOp::Value))
244    }
245    fn add_alternative(self, arm: Self) -> Self {
246        match self {
247            CInput::Request(CInputRequest::ForType { ty, input }) => {
248                CInput::for_type(*ty, input.add_alternative(arm))
249            }
250            CInput::Branch {
251                operation: CInputBranch::Selector,
252                children: mut arms,
253            } => {
254                arms.push(arm);
255                CInput::Branch {
256                    operation: CInputBranch::Selector,
257                    children: arms,
258                }
259            }
260            other => CInput::Branch {
261                operation: CInputBranch::Selector,
262                children: vec![other, arm],
263            },
264        }
265    }
266}
267/// Fluent conversion construction for this language.
268pub trait COutputExt<InRequest, OutRequest>: Sized {
269    /// Describe source record fields in declaration order.
270    fn fields(fields: impl IntoIterator<Item = COutput<InRequest, OutRequest>>) -> Self;
271    /// Observe a source getter and convert its result.
272    fn getter(name: impl Into<String>, output: COutput<InRequest, OutRequest>) -> Self;
273    /// Deliver multiple views of one source value.
274    fn combine(parts: impl IntoIterator<Item = COutput<InRequest, OutRequest>>) -> Self;
275    /// Group callable argument conversions in invocation order.
276    fn arguments(children: Vec<Self>) -> Self;
277    /// Read the source-ordered conversions of a callable domain.
278    fn argument_nodes(&self) -> crate::Result<&[Self]>;
279    /// Describe source variants and their field conversions in source order.
280    fn enum_variants(variants: Vec<Vec<Self>>, value: bool, mode: Option<ProjectionMode>) -> Self;
281    /// Read the field conversions of one enum variant.
282    fn variant_fields(&self) -> crate::Result<&[Self]>;
283    /// Set the ownership contract of one callable argument.
284    fn callback_argument(self, ownership: CallbackArgumentOwnership) -> Self;
285    /// Read an argument's conversion and ownership contract.
286    fn callback_argument_parts(&self) -> crate::Result<(CallbackArgumentOwnership, &Self)>;
287}
288impl<InRequest, OutRequest> COutputExt<InRequest, OutRequest> for COutput<InRequest, OutRequest> {
289    /// Observe record fields in declaration order; use an empty list for unit.
290    ///
291    /// `fields` holds one recipe per field of the source record, in declaration order.
292    fn fields(fields: impl IntoIterator<Item = COutput<InRequest, OutRequest>>) -> Self {
293        COutput::Branch {
294            operation: COutputBranch::Fields,
295            children: fields.into_iter().collect(),
296        }
297    }
298    /// Observe a value through a source getter and convert the getter's result.
299    ///
300    /// The getter takes `&T` for the current type `T`; adapters retaining a final
301    /// output owner may also support `&mut T`. Its result uses the selected
302    /// output conversion, including supported reference results.
303    ///
304    /// # Arguments
305    ///
306    /// * `name` — unqualified name of the source getter.
307    /// * `output` — recipe for the getter's result.
308    fn getter(name: impl Into<String>, output: COutput<InRequest, OutRequest>) -> Self {
309        COutput::Unary {
310            operation: COutputUnary::Getter {
311                function: name.into(),
312            },
313            child: Box::new(output),
314        }
315    }
316    /// Produce multiple outputs from one value.
317    ///
318    /// `parts` holds one recipe per view of the value; their wires appear in this
319    /// order. Binding rejects an empty list or more than one consuming part. The
320    /// framework uses [`Codec::output_consumes`] to order observations before a move,
321    /// unless [`Codec::output_owner`] first retains the final output allocation.
322    fn combine(parts: impl IntoIterator<Item = COutput<InRequest, OutRequest>>) -> Self {
323        COutput::Branch {
324            operation: COutputBranch::Combine,
325            children: parts.into_iter().collect(),
326        }
327    }
328    /// Group a callable's domain in source parameter order.
329    fn arguments(children: Vec<Self>) -> Self {
330        if children.is_empty() {
331            Self::None
332        } else {
333            Self::Branch {
334                operation: COutputBranch::Arguments,
335                children,
336            }
337        }
338    }
339    /// Read the source-ordered components of a callable domain.
340    fn argument_nodes(&self) -> crate::Result<&[Self]> {
341        match self {
342            Self::None => Ok(&[]),
343            Self::Branch {
344                operation: COutputBranch::Arguments,
345                children,
346            } => Ok(children),
347            _ => Err("callback domain requires an argument branch or None".into()),
348        }
349    }
350    /// Describe source variants, each with an ordinary branch of field conversions.
351    fn enum_variants(variants: Vec<Vec<Self>>, value: bool, mode: Option<ProjectionMode>) -> Self {
352        Self::Branch {
353            operation: COutputBranch::Enum { value, mode },
354            children: variants
355                .into_iter()
356                .map(|children| Self::Branch {
357                    operation: COutputBranch::Variant,
358                    children,
359                })
360                .collect(),
361        }
362    }
363    /// Read an enum variant's field conversions.
364    fn variant_fields(&self) -> crate::Result<&[Self]> {
365        match self {
366            Self::Branch {
367                operation: COutputBranch::Variant,
368                children,
369            } => Ok(children),
370            _ => Err("enum children must be variant branches".into()),
371        }
372    }
373    /// Attach the lifetime of produced foreign owners to this callable argument.
374    fn callback_argument(self, ownership: CallbackArgumentOwnership) -> Self {
375        Self::Unary {
376            operation: COutputUnary::CallbackArgument { ownership },
377            child: Box::new(self),
378        }
379    }
380    /// Read an argument's conversion and ownership contract from its own subtree.
381    fn callback_argument_parts(&self) -> crate::Result<(CallbackArgumentOwnership, &Self)> {
382        match self {
383            Self::Unary {
384                operation: COutputUnary::CallbackArgument { ownership },
385                child,
386            } => Ok((*ownership, child)),
387            _ => Err("callback argument requires an explicit ownership contract".into()),
388        }
389    }
390}
391/// Fluent conversion construction for this language.
392pub trait COutputRequests: Sized {
393    /// Expand source enum fields with local variant/member overrides.
394    fn enum_fields<'a>(
395        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
396    ) -> Self;
397    /// Request the canonical value representation independently of expansion defaults.
398    fn value() -> Self;
399    /// Resolve the configured default for this source occurrence.
400    fn auto() -> Self;
401    /// Apply a conversion to the matching source type, including nested occurrences.
402    fn for_type(ty: Type, output: Self) -> Self;
403    /// Append an explicit output view.
404    fn view(self, view: impl Into<Self>) -> Self;
405    /// Append a getter view using its default result conversion.
406    fn field(self, name: impl Into<String>) -> Self;
407    /// Append the direct value representation.
408    fn field_self(self) -> Self;
409    /// Append an explicitly borrowed projection.
410    fn field_self_ref(self) -> Self;
411    /// Append an explicitly owned projection.
412    fn field_self_into(self) -> Self;
413    /// Append a projection with explicit ownership and field policies.
414    fn project_self(self, policy: Projection) -> Self;
415    /// Append consuming decomposition of an accessor's record result.
416    fn fields_into(self, view: Self) -> Self;
417    /// Deprecated spelling of fields_into.
418    fn fields_self_into(self, view: Self) -> Self;
419    /// Override one named source field's conversion.
420    fn field_override(self, name: impl Into<String>, output: Self) -> Self;
421}
422impl COutputRequests for COutput<CInputRequest, COutputRequest> {
423    /// Expand an enum's tag and active fields using source variant/member keys.
424    fn enum_fields<'a>(
425        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
426    ) -> Self {
427        COutput::Request(COutputRequest::EnumFields(
428            prebindgen_declarations::enum_overrides(overrides),
429        ))
430    }
431    /// Request the canonical value representation, bypassing boundary defaults.
432    fn value() -> Self {
433        COutput::Request(COutputRequest::Value { projection: None })
434    }
435    /// Request this source type's configured output default during preparation.
436    fn auto() -> Self {
437        COutput::Request(COutputRequest::Auto)
438    }
439    /// Associate a conversion with a source type for a default or nested override.
440    fn for_type(ty: Type, output: Self) -> Self {
441        COutput::Request(COutputRequest::ForType {
442            ty: Box::new(ty),
443            output: Box::new(output),
444        })
445    }
446    /// Append another simultaneous output view.
447    fn view(self, view: impl Into<Self>) -> Self {
448        let view = view.into();
449        match self {
450            COutput::Request(COutputRequest::ForType { ty, output }) => {
451                COutput::for_type(*ty, output.view(view))
452            }
453            COutput::Branch {
454                operation: COutputBranch::Combine,
455                children: mut parts,
456            } => {
457                parts.push(view);
458                COutput::Branch {
459                    operation: COutputBranch::Combine,
460                    children: parts,
461                }
462            }
463            other => COutput::Branch {
464                operation: COutputBranch::Combine,
465                children: vec![other, view],
466            },
467        }
468    }
469    /// Append a source getter whose result uses its registered conversion default.
470    fn field(self, name: impl Into<String>) -> Self {
471        self.view(COutput::getter(name, COutput::auto()))
472    }
473    /// Append the direct codec representation, bypassing this type's expansion default.
474    fn field_self(self) -> Self {
475        self.view(COutput::Leaf(COutputLeafOp::Value))
476    }
477    /// Append an explicit borrowed projection.
478    fn field_self_ref(self) -> Self {
479        self.project_self(Projection::borrowed())
480    }
481    /// Append an explicit owned projection.
482    fn field_self_into(self) -> Self {
483        self.project_self(Projection::owned())
484    }
485    /// Append a projection with explicit ownership and nested field policies.
486    fn project_self(self, policy: Projection) -> Self {
487        self.view(COutput::Leaf(COutputLeafOp::Projected(policy)))
488    }
489    /// Append a consuming accessor followed by owned field decomposition.
490    fn fields_into(self, view: Self) -> Self {
491        self.view(owned_field_view(view))
492    }
493    /// Deprecated spelling of [`COutput::fields_into`].
494    fn fields_self_into(self, view: Self) -> Self {
495        self.fields_into(view)
496    }
497    /// Override one named source field, preserving defaults for its siblings.
498    fn field_override(self, name: impl Into<String>, output: Self) -> Self {
499        prebindgen_declarations::override_output_field(
500            self,
501            name.into(),
502            output,
503            |tree| match tree {
504                COutput::Request(COutputRequest::NamedFields { overrides, .. }) => Some(overrides),
505                _ => None,
506            },
507            |tree| match tree {
508                COutput::Unary {
509                    operation: COutputUnary::Getter { .. } | COutputUnary::Into { .. },
510                    child,
511                }
512                | COutput::Request(COutputRequest::ForType { output: child, .. }) => Some(child),
513                _ => None,
514            },
515        )
516    }
517}

535pub(super) fn owned_field_view(
536    view: COutput<CInputRequest, COutputRequest>,
537) -> COutput<CInputRequest, COutputRequest> {
538    let COutput::Unary {
539        operation: COutputUnary::Getter { function },
540        child: output,
541    } = view
542    else {
543        panic!("fields_into requires a fields! accessor")
544    };
545    let COutput::Request(COutputRequest::NamedFields { overrides, .. }) = *output else {
546        panic!("fields_into requires named fields")
547    };
548    COutput::Unary {
549        operation: COutputUnary::Into { function },
550        child: Box::new(COutput::Request(COutputRequest::NamedFields {
551            overrides,
552            mode: ProjectionMode::Owned,
553        })),
554    }
