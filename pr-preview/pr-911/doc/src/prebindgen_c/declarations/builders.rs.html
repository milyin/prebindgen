1use super::*;
2/// Input alternatives for a source type, built directly as a [`CInput`] tree.
3#[derive(Clone, Debug)]
4pub struct DeclCExpandParam {
5    /// Source type targeted by this default or occurrence override.
6    pub ty: syn::Type,
7    input: prebindgen_declarations::DeclInputAlternatives<CInput<CInputRequest, COutputRequest>>,
8}
9impl DeclCExpandParam {
10    /// Start an input declaration; add at least one alternative before use.
11    pub fn new(ty: syn::Type) -> Self {
12        Self {
13            ty,
14            input: Default::default(),
15        }
16    }
17    /// Explicitly construct an owned temporary and borrow it for the source call.
18    /// With no alternatives, use the owned value's default input conversion.
19    /// Alternatives describe the owned referent, not the reference itself.
20    pub fn reconstruct(mut self) -> Self {
21        self.input.reconstruct();
22        self
23    }
24    /// Append a constructor reference; its arguments use their type defaults.
25    pub fn variant(mut self, function: impl Into<DeclCFunction>) -> Self {
26        self.input
27            .push(CInput::call(function.into().reference_name(false)));
28        self
29    }
30    /// Append the type's direct language representation.
31    pub fn variant_self(mut self) -> Self {
32        self.input.push(CInput::Leaf(CInputLeafOp::Value));
33        self
34    }
35}
36impl From<DeclCExpandParam> for CInput<CInputRequest, COutputRequest> {
37    fn from(decl: DeclCExpandParam) -> Self {
38        let input = decl
39            .input
40            .build(CInput::selector, CInput::auto, |input| CInput::Unary {
41                operation: CInputUnary::Temporary,
42                child: Box::new(input),
43            });
44        CInput::for_type(decl.ty, input)
45    }
46}
47/// Simultaneous views of a source return value, built directly as a [`COutput`] tree.
48#[derive(Clone, Debug)]
49pub struct DeclCExpandReturn {
50    /// Source type targeted by this default or occurrence override.
51    pub ty: syn::Type,
52    output: Vec<COutput<CInputRequest, COutputRequest>>,
53}
54impl DeclCExpandReturn {
55    /// Start an output declaration; add views before registering a type default.
56    pub fn new(ty: syn::Type) -> Self {
57        Self {
58            ty,
59            output: Default::default(),
60        }
61    }
62    /// Append a getter view. An explicit `.name()` labels the delivered view.
63    /// Shared results are observed before another view moves the owner. An
64    /// owning opaque copy requires an explicit projection or cloning accessor.
65    pub fn field(mut self, function: impl Into<DeclCFunction>) -> Self {
66        self.output.push(prebindgen_declarations::getter_view(
67            function.into(),
68            |name| COutput::getter(name, COutput::auto()),
69            |name, output| COutput::Unary {
70                operation: COutputUnary::Named { name },
71                child: Box::new(output),
72            },
73        ));
74        self
75    }
76    /// Append a getter's named record fields as separate output values.
77    pub fn fields(self, fields: DeclCFields) -> Self {
78        self.view(fields)
79    }
80    /// Append another explicit output view.
81    pub fn view(mut self, output: impl Into<COutput<CInputRequest, COutputRequest>>) -> Self {
82        self.output.push(output.into());
83        self
84    }
85    /// Append a consuming accessor followed by owned field decomposition.
86    pub fn fields_into(mut self, fields: DeclCFields) -> Self {
87        self.output
88            .push(super::requests::owned_field_view(fields.into()));
89        self
90    }
91    /// Deprecated spelling of [`Self::fields_into`].
92    #[deprecated(note = "use fields_into instead")]
93    pub fn fields_self_into(self, fields: DeclCFields) -> Self {
94        self.fields_into(fields)
95    }
96    /// Append the language's direct representation, bypassing this type's default.
97    pub fn field_self(mut self) -> Self {
98        self.output.push(COutput::Leaf(COutputLeafOp::Value));
99        self
100    }
101    /// Append an explicit shared projection. Owning opaque representations clone
102    /// the value; record/collection projections permit copying borrowed leaves.
103    pub fn field_self_ref(self) -> Self {
104        self.project_self(Projection::borrowed())
105    }
106    /// Append an explicit owned projection.
107    pub fn field_self_into(self) -> Self {
108        self.project_self(Projection::owned())
109    }
110    /// Append a projection with explicit ownership and field policies.
111    pub fn project_self(mut self, policy: Projection) -> Self {
112        self.output
113            .push(COutput::Leaf(COutputLeafOp::Projected(policy)));
114        self
115    }
116}
117impl From<DeclCExpandReturn> for COutput<CInputRequest, COutputRequest> {
118    fn from(decl: DeclCExpandReturn) -> Self {
119        COutput::for_type(decl.ty, COutput::combine(decl.output))
120    }
121}
122/// A source accessor followed by named field decomposition.
123/// Field adjustments edit the accessor's output tree directly.
124#[derive(Clone, Debug)]
125pub struct DeclCFields {
126    function: String,
127    overrides: BTreeMap<String, COutput<CInputRequest, COutputRequest>>,
128}
129impl DeclCFields {
130    /// Select a getter whose returned record supplies the fields.
131    pub fn new(function: syn::Ident) -> Self {
132        Self {
133            function: function.to_string(),
134            overrides: BTreeMap::new(),
135        }
136    }
137    /// Replace one field's conversion. An empty declaration omits the whole field,
138    /// including any enclosing Option or sequence representation.
139    pub fn field(self, name: impl AsRef<str>, output: DeclCExpandReturn) -> Self {
140        self.field_override(name.as_ref().to_owned(), output)
141    }
142    /// Set an explicit conversion tree for one field, with the same omission
143    /// semantics as [`Self::field`] for an empty typed declaration.
144    pub fn field_override(
145        mut self,
146        name: impl Into<String>,
147        output: impl Into<COutput<CInputRequest, COutputRequest>>,
148    ) -> Self {
149        let output = match output.into() {
150            COutput::Request(COutputRequest::ForType { ty, output }) if matches!(output.as_ref(), COutput::Branch { operation: COutputBranch::Combine, children } if children.is_empty()) => {
151                COutput::for_type(*ty, COutput::Leaf(COutputLeafOp::Discard))
152            }
153            output => output,
154        };
155        assert!(
156            self.overrides.insert(name.into(), output).is_none(),
157            "duplicate field override"
158        );
159        self
160    }
161}
162impl From<DeclCFields> for COutput<CInputRequest, COutputRequest> {
163    fn from(fields: DeclCFields) -> Self {
164        COutput::getter(
165            fields.function,
166            COutput::Request(COutputRequest::NamedFields {
167                overrides: fields.overrides,
168                mode: prebindgen_declarations::ProjectionMode::Borrowed,
169            }),
170        )
171    }
172}
173/// A type-wide source-error mapping using this adapter's function declaration.
174pub type DeclCError = prebindgen_declarations::DeclError<
175    CInput<CInputRequest, COutputRequest>,
176    COutput<CInputRequest, COutputRequest>,
177>;
178/// A type-wide input, output, or error-mapping declaration.
179/// Generators immediately consume these builders into their conversion trees.
180#[derive(Clone, Debug)]
181pub enum DeclCExpand {
182    /// Input alternatives.
183    Param(DeclCExpandParam),
184    /// Simultaneous output views.
185    Return(DeclCExpandReturn),
186    /// A source-error message mapping.
187    Error(Box<DeclCError>),
188}
189impl From<DeclCExpandParam> for DeclCExpand {
190    fn from(value: DeclCExpandParam) -> Self {
191        Self::Param(value)
192    }
193}
194impl From<DeclCExpandReturn> for DeclCExpand {
195    fn from(value: DeclCExpandReturn) -> Self {
196        Self::Return(value)
197    }
198}
199impl From<DeclCError> for DeclCExpand {
200    fn from(value: DeclCError) -> Self {
201        Self::Error(Box::new(value))
202    }
203}
204/// Callback declaration containing this language's argument and result trees.
205pub type DeclCCallback = prebindgen_declarations::DeclCallback<
206    CInput<CInputRequest, COutputRequest>,
207    COutput<CInputRequest, COutputRequest>,
208>;
209impl From<DeclCCallback> for CInputRequest {
210    fn from(decl: DeclCCallback) -> Self {
211        Self::ForType {
212            ty: Box::new(decl.ty.clone()),
213            input: Box::new(CInput::Request(Self::Callback(Box::new(decl)))),
214        }
215    }
