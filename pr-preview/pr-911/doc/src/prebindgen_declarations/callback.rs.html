2use prebindgen_flat::{callback_signature, Flat, Result, Schema};
3use syn::Type;

5/// Source identity and foreign declaration policy for one callback occurrence.
6///
7/// The Arrow's domain and codomain own the conversions. Both directions are
8/// fully resolved before the adapter compiles the callback.
9#[derive(Clone, Debug)]
10pub struct CallbackOperation<Extension = std::convert::Infallible> {
11    /// Adapter-specific callback settings retained in the resolved tree.
12    pub extension: Option<Extension>,
13    /// Source callback signature; canonicalized when the request resolves.
14    pub ty: Type,
15    /// Literal foreign interface name, if explicitly selected.
16    pub foreign: Option<String>,
17    /// Foreign namespace selected by the adapter.
18    pub namespace: Option<String>,
19    /// Whether the callback result has an explicit callback-level conversion.
20    pub explicit_return: bool,
21}
22
23impl<Extension> CallbackOperation<Extension> {
24    /// Verify the configured signature against this source callback occurrence.
25    pub fn validate_for<M: Schema>(&self, flat: &Flat<M>, ty: &Type) -> Result<()> {
26        if input_key(flat, ty)? != prebindgen_flat::type_key(&self.ty) {
27            return Err("prepared callback type does not match occurrence".into());
28        }
29        callback_signature(&self.ty)?.ok_or("prepared callback requires impl Fn(...) -> R")?;
30        Ok(())
31    }
32}
33
34/// Resolve a callback signature for configuration identity, excluding marker bounds.
35pub fn input_type<M: Schema>(flat: &Flat<M>, ty: &Type) -> Result<Type> {
36    let Some(signature) = callback_signature(ty)? else {
37        return flat.resolve_type(ty);
38    };
39    let arguments = signature
40        .arguments
41        .iter()
42        .map(|t| flat.resolve_type(t))
43        .collect::<Result<Vec<_>>>()?;
44    let ret = flat.resolve_type(&signature.return_type)?;
45    Ok(syn::parse_quote!(impl Fn(#(#arguments),*) -> #ret))
46}
47
48// Source occurrences have already been resolved by Flat. Only callback marker
49// bounds need normalization when looking up defaults.
50/// Canonical callback identity without lifetimes or marker bounds.
51pub fn input_key<M: Schema>(flat: &Flat<M>, ty: &Type) -> Result<String> {
52    if callback_signature(ty)?.is_some() {
53        Ok(prebindgen_flat::type_key(&input_type(flat, ty)?))
54    } else {
55        Ok(prebindgen_flat::type_key(ty))
56    }
57}

68use std::collections::BTreeMap;
69/// A retained callback's foreign declaration and independently selected conversions.
70///
71/// Arguments leave Rust and accept output declarations; the returned value enters
72/// Rust and accepts an input declaration. Unspecified settings inherit the
73/// callback signature's defaults, followed by ordinary value-type defaults.
74/// Each overridden argument/result replaces its entire conversion tree.
75///
76#[derive(Clone, Debug)]
77pub struct DeclCallback<InTree, OutTree, Extension = std::convert::Infallible> {
78    /// Adapter-defined settings, preserved by common fluent methods.
79    pub extension: Option<Extension>,
80    /// Source `impl Fn(...) -> R` signature; marker bounds do not identify defaults.
81    pub ty: Type,
82    /// Literal destination interface name.
83    pub foreign: Option<String>,
84    /// Destination namespace, supplied by the language adapter's placement declaration.
85    pub namespace: Option<String>,
86    /// Argument output overrides, indexed in source invocation order.
87    pub params: BTreeMap<usize, OutTree>,
88    /// Returned-value input override.
89    pub ret: Option<InTree>,
90}
91impl<InTree, OutTree, Extension> DeclCallback<InTree, OutTree, Extension> {
92    /// Declare defaults or an occurrence override for a source callback signature.
93    pub fn new(ty: Type) -> Self {
94        Self {
95            extension: None,
96            ty,
97            foreign: None,
98            namespace: None,
99            params: BTreeMap::new(),
100            ret: None,
101        }
102    }
103    /// Configure adapter-specific settings without replacing common declaration data.
104    pub fn with_extension<Next>(self, extension: Next) -> DeclCallback<InTree, OutTree, Next> {
105        DeclCallback {
106            extension: Some(extension),
107            ty: self.ty,
108            foreign: self.foreign,
109            namespace: self.namespace,
110            params: self.params,
111            ret: self.ret,
112        }
113    }
114    /// Set a literal destination interface name.
115    pub fn name(mut self, name: impl Into<String>) -> Self {
116        self.foreign = Some(name.into());
117        self
118    }
119    /// Override one argument's Rust-to-foreign conversion.
120    ///
121    /// Like a function parameter override, duplicate indices panic. Build checks
122    /// that the index exists and the conversion matches the source argument type.
123    pub fn expand_param(mut self, index: usize, output: impl Into<OutTree>) -> Self {
124        assert!(
125            self.params.insert(index, output.into()).is_none(),
126            "duplicate callback parameter override"
127        );
128        self
129    }
130    /// Override the callback result's foreign-to-Rust conversion.
131    pub fn expand_return(mut self, input: impl Into<InTree>) -> Self {
132        self.ret = Some(input.into());
133        self
134    }
135}

137/// Metadata for a callback whose explicit conversions live on structural tree edges.
138#[derive(Clone, Debug)]
139pub struct CallbackRequest<Extension = std::convert::Infallible> {
140    /// Source identity, foreign placement and adapter settings.
141    pub operation: CallbackOperation<Extension>,
142    /// Source argument indices corresponding to the domain's explicit children.
143    pub arguments: Vec<usize>,
144}
145/// Adapter construction of a retained callback's complete input tree.
146/// This bridge permits the shared declaration's `Into<InputNode>` implementation
147/// while the adapter selects its own operation and grouping metadata.
148pub trait CallbackInput<InTree, OutTree, Extension = std::convert::Infallible> {
149    /// Put explicit argument/result recipes on structural edges, preserving type scope.
150    fn callback_input(declaration: DeclCallback<InTree, OutTree, Extension>) -> InTree;
151}
152/// Assemble a callback declaration using the adapter's structural node constructors.
153pub fn callback_input<I, O, E>(
154    declaration: DeclCallback<I, O, E>,
155    auto: impl FnOnce() -> I,
156    arguments: impl FnOnce(Vec<O>) -> O,
157    arrow: impl FnOnce(CallbackRequest<E>, O, I) -> I,
158    typed: impl FnOnce(Type, I) -> I,
159) -> I {
160    let ty = declaration.ty;
161    let explicit_return = declaration.ret.is_some();
162    let (indices, children) = declaration.params.into_iter().unzip();
163    let metadata = CallbackRequest {
164        operation: CallbackOperation {
165            ty: ty.clone(),
166            foreign: declaration.foreign,
167            namespace: declaration.namespace,
168            extension: declaration.extension,
169            explicit_return,
170        },
171        arguments: indices,
172    };
173    let domain = arguments(children);
174    let codomain = declaration.ret.unwrap_or_else(auto);
175    typed(ty, arrow(metadata, domain, codomain))
176}
177impl<I: crate::InputOps, O: crate::OutputOps, IR, OR, E>
178    From<DeclCallback<crate::InputNode<I, O, IR, OR>, crate::OutputNode<I, O, IR, OR>, E>>
179    for crate::InputNode<I, O, IR, OR>
180where
181    IR: CallbackInput<crate::InputNode<I, O, IR, OR>, crate::OutputNode<I, O, IR, OR>, E>,
182{
183    fn from(declaration: DeclCallback<Self, crate::OutputNode<I, O, IR, OR>, E>) -> Self {
184        IR::callback_input(declaration)
185    }
