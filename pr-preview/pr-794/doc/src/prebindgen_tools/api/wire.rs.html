6/// The adapter's closed set of boundary types — one enum per adapter.
7///
8/// A variant may say more than its Rust type does: JNI's `jlong` is a Kotlin
9/// `Long`, a `ULong` or a handle of some class, and each is its own variant.
10/// The foreign-side writer reads that meaning off the wire, so it never has to
11/// repeat a decision that an override or a default made for it.
12pub trait WireType: Clone + fmt::Debug {
13    /// The Rust type of a slot of this kind in the generated file.
14    fn rust(&self) -> TokenStream;
15    /// The value a slot holds when it carries nothing: an absent optional,
16    /// an alternative of a sum that was not taken.
17    fn placeholder(&self) -> TokenStream;
18}
19
20/// One named slot of the generated boundary.
21#[derive(Clone, Debug)]
22pub struct Wire<W> {
23    pub name: syn::Ident,
24    pub ty: W,
25}
26
27impl<W: WireType> Wire<W> {
28    pub fn new(name: syn::Ident, ty: W) -> Self {
29        Self { name, ty }
30    }
31
32    /// `name: RustType`, for a parameter list or a struct body.
33    pub fn decl(&self) -> TokenStream {
34        let (n, t) = (&self.name, self.ty.rust());
35        quote!(#n: #t)
36    }
37}
