10/// A way that serves both directions.

13/// One way a type may cross, as registered in [`Ways`].
14///
15/// Each variant holds the model item that fixes the structure below it, so
16/// what a way can contain is decided by the model, never by the adapter:
17/// [`Way::Fields`] has one part per field of its struct, [`Way::Constructed`]
18/// one per parameter of its function.
19pub enum Way<'f, L> {
20    /// Whole, on one wire; `L` is the adapter's declaration (a handle, an
21    /// enum, a mirror) that its lowering turns into the wire.
22    Whole(L),
23    /// As the fields of this struct.
24    Fields(&'f Struct),
25    /// As a tag and the fields of every alternative of this sum.
26    Alternatives(&'f Variant),
27    /// As the representation this conversion declares.
28    Converted(Box<ResolvedConversion>),
29    /// Into Rust only: as the arguments of this function, which builds the
30    /// value.
31    Constructed(&'f Function),
32}

34/// A registered way, typed by the directions it can serve: [`Both`], or only
35/// [`In`] or [`Out`]. A `WayId<Both>` converts into either.
36pub struct WayId<D> {
37    index: usize,
38    _direction: PhantomData<fn() -> D>,
39}

76struct Registered<'f, L> {
77    /// The type the way is for.
78    key: String,
79    way: Way<'f, L>,
80    /// Whether it builds a value and so serves [`In`] only.
81    in_only: bool,
82}
83
84/// Every way each type may cross, registered once and checked against the
85/// model when registered. Which one an occurrence takes is a [`Choices`].
86pub struct Ways<'f, L> {
87    ways: Vec<Registered<'f, L>>,
88}
89
90impl<L> Default for Ways<'_, L> {
91    fn default() -> Self {
92        Self { ways: Vec::new() }
93    }
94}

96/// The identity a type is registered under: a named type by its name, any
97/// other type by its canonical form.
98pub(crate) fn key_of(ty: &TypeRef) -> String {
99    match ty.kind() {
100        TypeKind::Named { id, .. } => id.name.clone(),
101        TypeKind::String => "String".into(),
102        _ => ty.key().as_str().to_string(),
103    }
104}
105
106impl<'f, L> Ways<'f, L> {
107    pub fn new() -> Self {
108        Self::default()
109    }
110
111    fn push<D>(&mut self, ty: &TypeRef, way: Way<'f, L>, in_only: bool) -> WayId<D> {
112        self.ways.push(Registered {
113            key: key_of(ty),
114            way,
115            in_only,
116        });
117        WayId::new(self.ways.len() - 1)
118    }
119
120    /// `ty` crosses whole, as `leaf` says.
121    pub fn whole(&mut self, ty: &TypeRef, leaf: L) -> WayId<Both> {
122        self.push(ty, Way::Whole(leaf), false)
123    }
124
125    /// The struct crosses as its fields.
126    pub fn fields(&mut self, s: &'f Struct) -> WayId<Both> {
127        self.push(s.type_ref(), Way::Fields(s), false)
128    }
129
130    /// The sum crosses as a tag and its alternatives' fields.
131    pub fn alternatives(&mut self, v: &'f Variant) -> WayId<Both> {
132        self.push(v.type_ref(), Way::Alternatives(v), false)
133    }
134
135    /// The conversion's target crosses as its representation.
136    pub fn converted(&mut self, conversion: ResolvedConversion) -> WayId<Both> {
137        let ty = conversion.target().clone();
138        self.push(&ty, Way::Converted(Box::new(conversion)), false)
139    }
140
141    /// The type `f` returns (directly or as the `Ok` of a `Result`) is built
142    /// by calling `f` on its arguments.
143    pub fn constructed(&mut self, f: &'f Function) -> Result<WayId<In>, String> {
144        let ret = match f.ret.kind() {
145            TypeKind::Fallible { ok, .. } => &**ok,
146            _ => &f.ret,
147        };
148        if !matches!(ret.kind(), TypeKind::Named { .. }) {
149            return Err(format!(
150                "constructor `{}` must return a named type, returns `{ret}`",
151                f.name
152            ));
153        }
154        Ok(self.push(&ret.clone(), Way::Constructed(f), true))
155    }
156
157    /// The way `id` names.
158    pub fn get<D>(&self, id: WayId<D>) -> &Way<'f, L> {
159        &self.ways[id.index].way
160    }
161
162    /// The ways registered for `ty`, or for the type it borrows.
163    fn of<D: Direction>(&self, ty: &TypeRef) -> Vec<usize> {
164        let keys = [Some(key_of(ty)), ty.borrow_target().map(key_of)];
165        for key in keys.iter().flatten() {
166            let ids: Vec<usize> = (0..self.ways.len())
167                .filter(|&i| &self.ways[i].key == key && (D::IN || !self.ways[i].in_only))
168                .collect();
169            if !ids.is_empty() {
170                return ids;
171            }
172        }
173        Vec::new()
174    }
175
176    /// Whether way `index` is for `ty` or the type it borrows.
177    fn is_for(&self, index: usize, ty: &TypeRef) -> bool {
178        let key = &self.ways[index].key;
179        *key == key_of(ty) || ty.borrow_target().is_some_and(|t| *key == key_of(t))
180    }
181}

183/// Which registered way each occurrence takes, in direction `D`: a default
184/// per type, replaced at chosen places.
185///
186/// A type with a single way registered for `D` needs no choice.
187pub struct Choices<D> {
188    defaults: HashMap<String, usize>,
189    places: HashMap<Place, usize>,
190    _direction: PhantomData<fn() -> D>,
191}
192
193impl<D> Default for Choices<D> {
194    fn default() -> Self {
195        Self {
196            defaults: HashMap::new(),
197            places: HashMap::new(),
198            _direction: PhantomData,
199        }
200    }
201}
202
203impl<D: Direction> Choices<D> {
204    pub fn new() -> Self {
205        Self::default()
206    }
207
208    /// Every occurrence of the way's type takes it, unless a place chooses
209    /// otherwise.
210    pub fn choose<L>(&mut self, ways: &Ways<'_, L>, way: impl Into<WayId<D>>) {
211        let index = way.into().index;
212        self.defaults.insert(ways.ways[index].key.clone(), index);
213    }
214
215    /// The occurrence at `place` takes `way`. Resolving fails if the type
216    /// at `place` is not the way's.
217    pub fn choose_at(&mut self, place: Place, way: impl Into<WayId<D>>) {
218        self.places.insert(place, way.into().index);
219    }
220
221    /// The way the occurrence of `ty` at `place` takes: the place's choice,
222    /// else the type's default, else its only way. `None` when the type has
223    /// no way registered; an error when the choice is ambiguous or names
224    /// another type's way.
225    pub(crate) fn pick<'w, 'f, L>(
226        &self,
227        ways: &'w Ways<'f, L>,
228        ty: &TypeRef,
229        place: &Place,
230    ) -> Result<Option<&'w Way<'f, L>>, String> {
231        if let Some(&i) = self.places.get(place) {
232            if !ways.is_for(i, ty) {
233                return Err(format!(
234                    "{place}: the way chosen here is for `{}`, the type here is `{ty}`",
235                    ways.ways[i].key
236                ));
237            }
238            return Ok(Some(&ways.ways[i].way));
239        }
240        let ids = ways.of::<D>(ty);
241        if let Some(&i) = ids
242            .iter()
243            .find(|&&i| self.defaults.get(&ways.ways[i].key) == Some(&i))
244        {
245            return Ok(Some(&ways.ways[i].way));
246        }
247        match ids.as_slice() {
248            [] => Ok(None),
249            [i] => Ok(Some(&ways.ways[*i].way)),
250            _ => Err(format!(
251                "{place}: `{ty}` has {} ways to cross {}; choose one",
252                ids.len(),
253                D::NAME
254            )),
255        }
256    }
257}
