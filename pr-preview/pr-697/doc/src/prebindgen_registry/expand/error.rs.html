1//! Errors produced while resolving constructor-expansion declarations.
2
3/// Errors surfaced while resolving [`Expansions`](super::Expansions).
4#[derive(Debug)]
5pub enum ExpandError {
6    /// An expansion whose leaves hold more than one callback.
7    ///
8    /// A site names a callback's delivered value by the SOURCE parameter it
9    /// arrived on — `Role::CallbackArg { param, arg }` — so two callbacks under
10    /// one parameter give two distinct positions the same identity. Refused
11    /// where the expansion is declared, rather than left to collide when the
12    /// sites are planned.
13    TwoCallbacksInOneParam {
14        func: syn::Ident,
15        param: syn::Ident,
16    },
17    UnknownFunction(syn::Ident),
18    UnknownParam(syn::Ident, syn::Ident),
19    UnknownConstructor(syn::Ident),
20    NoConstructor {
21        func: syn::Ident,
22        param: syn::Ident,
23        target: String,
24    },
25    TargetMismatch {
26        ctor: String,
27        produces: String,
28        expected: String,
29    },
30    UnsupportedOptional {
31        func: syn::Ident,
32        param: syn::Ident,
33        reason: &'static str,
34    },
35    /// An explicit per-fn input flatten targeted a read accessor — accessors
36    /// are never parameter-composed.
37    ConstructOnAccessor {
38        func: syn::Ident,
39    },
40    /// A per-fn `.expand_param(name, expand_param!(T))` decl whose `T` does
41    /// not match the named parameter's peeled type.
42    ParamTypeMismatch {
43        func: syn::Ident,
44        param: syn::Ident,
45        declared: String,
46        actual: String,
47    },
48    /// Recursive input reached a type already on the build chain (`A → … → A`).
49    InputCycle {
50        ty: String,
51    },
52    /// A recursive-input shape that is declared-but-not-yet-supported (recursion
53    /// under a selector-dispatched variant, or on an `Option<…>` parameter).
54    UnsupportedRecursive {
55        func: syn::Ident,
56        reason: &'static str,
57    },
58    /// Structurally invalid declaration records — empty variant lists or
59    /// duplicate targets. All offenders are collected before failing
60    /// (mirrors `ScanError::DeclaredNotFound`).
61    InvalidDeclarations {
62        entries: Vec<ExpandDeclError>,
63    },
64}
65
66/// One structurally invalid expansion declaration (see
67/// [`ExpandError::InvalidDeclarations`]).
68#[derive(Debug)]
69pub enum ExpandDeclError {
70    /// A constructor declaration with no variants.
71    EmptyConstructor { target: String },
72    /// A per-fn expand with an empty variant subset.
73    EmptySubset { func: syn::Ident, param: syn::Ident },
74    /// Two constructor declarations for the same target type.
75    DuplicateConstructor { target: String },
76    /// Two per-fn expands for the same `(fn, param)`.
77    DuplicateExpand { func: syn::Ident, param: syn::Ident },
78}
79
80impl std::fmt::Display for ExpandDeclError {
81    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
82        match self {
83            ExpandDeclError::EmptyConstructor { target } => {
84                write!(f, "constructor for `{target}` declares no variants")
85            }
86            ExpandDeclError::EmptySubset { func, param } => write!(
87                f,
88                "expand for parameter `{param}` of `{func}` declares no variants"
89            ),
90            ExpandDeclError::DuplicateConstructor { target } => {
91                write!(f, "duplicate constructor declaration for `{target}`")
92            }
93            ExpandDeclError::DuplicateExpand { func, param } => write!(
94                f,
95                "duplicate expand declaration for parameter `{param}` of `{func}`"
96            ),
97        }
98    }
99}

101impl std::fmt::Display for ExpandError {
102    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
103        match self {
104            ExpandError::TwoCallbacksInOneParam { func, param } => write!(
105                f,
106                "expand: `{func}`'s parameter `{param}` expands into more than one callback. \
107                 A delivered value is named by the parameter it arrived on, so two callbacks \
108                 under one parameter would give two positions the same site",
109            ),
110            ExpandError::UnknownFunction(name) => {
111                write!(f, "expand: function `{}` is not a #[prebindgen] item", name)
112            }
113            ExpandError::ConstructOnAccessor { func } => write!(
114                f,
115                "expand: param-variant override on accessor fn `{}` — an accessor is never \
116                 parameter-composed (remove the override, or declare it as `.fun`)",
117                func
118            ),
119            ExpandError::InputCycle { ty } => write!(
120                f,
121                "expand: recursive input forms a cycle through `{}` — a constructor \
122                 parameter's type transitively constructs itself",
123                ty
124            ),
125            ExpandError::UnsupportedRecursive { func, reason } => write!(
126                f,
127                "expand: `{}`: recursive input not supported here: {}",
128                func, reason
129            ),
130            ExpandError::UnknownParam(func, param) => write!(
131                f,
132                "expand: function `{}` has no parameter named `{}`",
133                func, param
134            ),
135            ExpandError::ParamTypeMismatch {
136                func,
137                param,
138                declared,
139                actual,
140            } => write!(
141                f,
142                "expand: `{}`.expand_param(\"{}\", expand_param!({declared})): the parameter's \
143                 type is `{actual}`, not `{declared}` — declare the decl for the parameter's \
144                 actual type",
145                func, param
146            ),
147            ExpandError::UnknownConstructor(name) => write!(
148                f,
149                "expand: constructor `{}` is not a #[prebindgen] item",
150                name
151            ),
152            ExpandError::NoConstructor {
153                func,
154                param,
155                target,
156            } => write!(
157                f,
158                "expand: no constructor registered for `{}` (parameter `{}` of `{}`)",
159                target, param, func
160            ),
161            ExpandError::TargetMismatch {
162                ctor,
163                produces,
164                expected,
165            } => write!(
166                f,
167                "expand: constructor `{}` produces `{}` but the parameter expects `{}`",
168                ctor, produces, expected
169            ),
170            ExpandError::UnsupportedOptional {
171                func,
172                param,
173                reason,
174            } => write!(
175                f,
176                "expand: optional parameter `{}` of `{}` is not supported: {}",
177                param, func, reason
178            ),
179            ExpandError::InvalidDeclarations { entries } => {
180                writeln!(f, "expand: invalid declarations:")?;
181                for e in entries {
182                    writeln!(f, "  - {e}")?;
183                }
184                Ok(())
185            }
186        }
187    }
188}
