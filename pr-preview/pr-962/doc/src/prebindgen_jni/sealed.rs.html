8/// Declare a Kotlin sealed interface for a Rust payload enum or concrete Result.
9#[derive(Clone)]
10pub struct DeclJniSealedClass {
11    /// Source enum type.
12    pub ty: syn::Type,
13    /// Literal Kotlin short name; required for concrete generic types.
14    pub name: Option<String>,
15    /// Source variant name and foreign class-name overrides.
16    pub variant_names: Vec<(String, String)>,
17}
18impl DeclJniSealedClass {
19    /// Declare the ordinary value representation of a source enum.
20    pub fn new(ty: syn::Type) -> Self {
21        Self {
22            ty,
23            name: None,
24            variant_names: vec![],
25        }
26    }
27    /// Override the Kotlin short name.
28    pub fn name(mut self, name: impl Into<String>) -> Self {
29        self.name = Some(name.into());
30        self
31    }
32    /// Rename a source variant without changing its tag or tag constant.
33    pub fn variant_name(mut self, source: impl Into<String>, name: impl Into<String>) -> Self {
34        self.variant_names.push((source.into(), name.into()));
35        self
36    }
37}
38/// Declare an ordinary sealed Kotlin representation without importing the Rust type.
39#[macro_export]
40macro_rules! sealed_class {($($ty:tt)+)=>{$crate::DeclJniSealedClass::new($crate::__type(stringify!($($ty)+)))};}

42pub(crate) fn class<'a>(
43    types: &'a crate::metadata::JniTypes,
44    r_type: &TypeRef,
45) -> Result<&'a Class> {
46    crate::metadata::source_enum(types, r_type)?;
47    crate::metadata::source_type(types, r_type)?
48        .metadata
49        .class
50        .as_deref()
51        .filter(|class| matches!(class.representation, ClassRepresentation::Sealed { .. }))
52        .ok_or_else(|| "ordinary payload enum requires a declared sealed class".into())
53}
54pub(crate) fn variant_name(class: &Class, variant: &syn::Ident) -> String {
55    let source = variant.to_string();
56    let source = source.trim_start_matches("r#");
57    let ClassRepresentation::Sealed { variant_names, .. } = &class.representation else {
58        unreachable!("variant naming requires a sealed class")
59    };
60    variant_names
61        .iter()
62        .find(|(name, _)| name.trim_start_matches("r#") == source)
63        .map_or_else(|| source.to_owned(), |(_, name)| name.clone())
64}
65fn validate_variants(class: &Class, source: &PayloadEnum<(), JniField>) -> Result<()> {
66    let ClassRepresentation::Sealed { variant_names, .. } = &class.representation else {
67        return Err("variant naming requires a sealed class".into());
68    };
69    let mut overrides = BTreeSet::new();
70    for (source_name, _) in variant_names {
71        let name = source_name.trim_start_matches("r#");
72        if !source
73            .variants
74            .iter()
75            .any(|v| v.name.to_string().trim_start_matches("r#") == name)
76        {
77            return Err(format!("unknown enum variant `{name}`"));
78        }
79        if !overrides.insert(name) {
80            return Err("duplicate enum variant override".into());
81        }
82    }
83    let mut seen = BTreeSet::new();
84    for variant in &source.variants {
85        let name = variant_name(class, &variant.name);
86        validate_class_name(&name)?;
87        if !seen.insert(name) {
88            return Err("duplicate Kotlin enum variant name".into());
89        }
90    }
91    Ok(())
92}

94pub(crate) fn encoding<
95    Chain: group::ChainMetadata,
96    Branch,
97    V: prebindgen_conversions::Conversion<Wire = JniWire, Chain = Chain, Branch = Branch>,
98>(
99    request: prebindgen_declarations::EnumVariants<'_, JniCodec, V>,
100    direction: prebindgen_declarations::Direction,
101    wrap: impl FnOnce(TypeRef, EnumForm) -> Branch,
102) -> Result<EnumEncoding<Branch>> {
103    let source = request.flat.payload_enum(request.ty)?;
104    let class = request
105        .flat
106        .type_info(request.ty)
107        .ok()
108        .and_then(|t| t.metadata.class.as_ref())
109        .filter(|c| matches!(c.representation, ClassRepresentation::Sealed { .. }));
110    let grouped = request.form == EnumForm::Value;
111    if grouped && class.is_none() {
112        return Err(format!(
113            "ordinary payload enum `{}` requires sealed_class!",
114            type_key(request.ty)
115        ));
116    }
117    if let Some(class) = class {
118        validate_variants(class, source)?;
119    }
120    for arm in request.arms.iter_mut() {
121        prebindgen_declarations::retire_optional_niches(
122            crate::group::values_mut(arm),
123            None,
124            |_, _| None,
125            |w, niche| w.set_niche(niche),
126        )?;
127    }
128    request
129        .selector
130        .kind
131        .set_niche(Some(JniNiche::Integer(-2)))?;
132    let inactive = if direction == prebindgen_declarations::Direction::Output {
133        request
134            .arms
135            .iter()
136            .map(|arm| {
137                crate::group::values(arm)
138                    .map(|wire| {
139                        Ok((
140                            wire.name.clone(),
141                            crate::optional::absent(&wire.kind, None, false)?,
142                        ))
143                    })
144                    .collect()
145            })
146            .collect::<Result<Vec<_>>>()?
147    } else {
148        vec![]
149    };
150    Ok(EnumEncoding {
151        metadata: wrap(request.flat.type_ref(request.ty)?, request.form),
152        inactive,
153    })
154}

156// Source field identities and order come from the canonical variant. Occurrence
157// nodes only describe conversions of those fields.
158#[allow(clippy::type_complexity)] // Borrow each canonical field with its directional conversion subtree.
159pub(crate) fn variant_fields<
160    'a,
161    Chain: group::ChainMetadata,
162    Branch,
163    Annotation: group::AnnotationMetadata,
164    Container: group::ContainerMetadata,
165>(
166    node: &'a JniWireNode<Chain, Branch, Annotation, Container>,
167    source: &'a PayloadVariant<(), JniField>,
168) -> Result<
169    impl Iterator<
170        Item = Result<(
171            &'a PayloadField<JniField>,
172            &'a JniWireNode<Chain, Branch, Annotation, Container>,
173        )>,
174    >,
175> {
176    let WireNode::Branch { children, .. } = node else {
177        return Err("enum arm must group fields".into());
178    };
179    if children.len() != source.fields.len() {
180        return Err(format!(
181            "enum occurrence differs from source variant `{}` fields",
182            source.name
183        ));
184    }
185    Ok(source
186        .fields
187        .iter()
188        .zip(children)
189        .map(|(source_field, node)| {
190            if data::field(node)?.to_string().trim_start_matches("r#") != source_field.property() {
191                return Err(format!(
192                    "enum occurrence differs from source variant `{}` fields",
193                    source.name
194                ));
195            }
196            Ok((source_field, node))
197        }))
198}

200pub(crate) fn input(
201    nodes: &[JniInputWireNode],
202    controls: &[Wire<JniWire>],
203    r_type: &TypeRef,
204    k_name: Option<&str>,
205    context: &KotlinContext<'_>,
206    active: &str,
207) -> Result<KotlinInput> {
208    let source = crate::metadata::source_enum(context.source_types, r_type)?;
209    let declaration = k_name
210        .map(|_| class(context.source_types, r_type))
211        .transpose()?;
212    if source.variants.len() != nodes.len() {
213        return Err("enum occurrence differs from its source variants".into());
214    }
215    let class_name = declaration.map(Class::fqn);
216    let [tag] = controls else {
217        return Err("enum needs one tag".into());
218    };
219    if class_name.is_none() {
220        let tree = WireNode::Branch {
221            controls: controls.to_vec(),
222            children: nodes.to_vec(),
223            meta: JniInputWireBranch::RChoices,
224        };
225        let mut out = kotlin::input_kotlin_when(&tree, context, active)?;
226        for path in &mut out.parameter_paths {
227            for step in &mut path.0 {
228                if let prebindgen_declarations::ComponentSegment::Alternative(index) = step {
229                    *step = prebindgen_declarations::ComponentSegment::Named(
230                        source.variants[*index]
231                            .name
232                            .to_string()
233                            .trim_start_matches("r#")
234                            .into(),
235                    );
236                }
237            }
238        }
239        return Ok(out);
240    }
241    let class = class_name.as_ref().unwrap();
242    let root = k_name.expect("KSealed input binding");
243    let tag_name = tag.name.to_string();
244    let cases = source
245        .variants
246        .iter()
247        .enumerate()
248        .map(|(i, variant)| {
249            let name = variant_name(declaration.unwrap(), &variant.name);
250            format!("is {class}.{name} -> {i}")
251        })
252        .collect::<Vec<_>>()
253        .join("\n");
254    let select = format!("when (val value = requireNotNull({root})) {{ {cases} }}");
255    let select = if active == "true" {
256        select
257    } else {
258        format!("if ({active}) {select} else 0")
259    };
260    let mut out = KotlinInput::default();
261    out.parameters.push(format!(
262        "{root}: {class}{}",
263        if active == "true" { "" } else { "?" }
264    ));
265    out.setup.push(KotlinBinding {
266        name: tag_name.clone(),
267        expression: select,
268        absent: "0".into(),
269    });
270    out.arguments.push(tag_name.clone());
271    out.prepared_arguments.push(tag_name.clone());
272    out.handle_inputs = Some(vec![]);
273    for (index, (node, variant)) in nodes.iter().zip(&source.variants).enumerate() {
274        let variant_class = variant_name(declaration.unwrap(), &variant.name);
275        let selected = if active == "true" {
276            format!("{tag_name} == {index}")
277        } else {
278            format!("({active}) && {tag_name} == {index}")
279        };
280        for pair in variant_fields(node, variant)? {
281            let (field, node) = pair?;
282            let mut child = kotlin::input_kotlin_when(node, context, &selected)?;
283            child.prepare_arguments(kotlin::has_handles(node, false))?;
284            data::check_property_input(&field.metadata.kotlin_type, &child.parameters).map_err(
285                |error| format!("{class}.{}.{}: {error}", variant_class, field.property()),
286            )?;
287            if let Some(parameter) = child.parameters.first() {
288                let (parameter_name, ty) = parameter
289                    .split_once(": ")
290                    .ok_or("invalid enum field parameter")?;
291                let name = parameter_name.to_owned();
292                let absent = match ty.trim_start_matches("kotlin.") {
293                    "ULong" => "0uL",
294                    "UInt" => "0u",
295                    "UShort" => "0.toUShort()",
296                    "UByte" => "0.toUByte()",
297                    ty => crate::wire::jvm::primitive_local_default(ty).unwrap_or("null"),
298                };
299                let property = kotlin_identifier(&field.property());
300                let expression=format!("(if ({selected}) (requireNotNull({root}) as {class}.{}).{property} else {absent})",variant_class);
301                replace_input(&mut child, &name, &expression);
302            }
303            child.parameters.clear();
304            child.parameter_paths.clear();
305            kotlin::merge_input(&mut out, child)?;
306        }
307    }
308    Ok(out)
309}
310fn replace_input(input: &mut KotlinInput, old: &str, new: &str) {
311    for text in input
312        .arguments
313        .iter_mut()
314        .chain(&mut input.prepared_arguments)
315        .chain(&mut input.locks)
316        .chain(&mut input.lock_groups)
317        .chain(&mut input.consumed)
318        .chain(input.setup.iter_mut().map(|b| &mut b.expression))
319        .chain(
320            input
321                .handle_inputs
322                .iter_mut()
323                .flatten()
324                .flat_map(|h| std::iter::once(&mut h.value).chain(h.consumed.iter_mut())),
325        )
326    {
327        *text = optional::replace(text, old, new);
328    }
329}

331pub(crate) fn output(
332    nodes: &[JniOutputWireNode],
333    controls: &[Wire<JniWire>],
334    r_type: &TypeRef,
335    declaration: Option<&Class>,
336    context: &KotlinContext<'_>,
337    owners: Option<&str>,
338) -> Result<KotlinOutput> {
339    let source = crate::metadata::source_enum(context.source_types, r_type)?;
340    if source.variants.len() != nodes.len() {
341        return Err("enum occurrence differs from its source variants".into());
342    }
343    let class_name = declaration.map(Class::fqn);
344    let tag = kotlin::output_leaves(&controls.iter().collect::<Vec<_>>(), context, owners)?;
345    let tag_expr = tag.delivered.first().ok_or("missing enum tag")?;
346    let mut out = KotlinOutput::default();
347    let local = format!("{}enumValue", context.output_prefix);
348    let mut body = String::new();
349    let mut cleanup = String::new();
350    let mut offset = controls.len();
351    let mut expanded = Vec::new();
352    if class_name.is_none() {
353        out.delivered.extend(tag.delivered.clone());
354        out.result_types.extend(tag.result_types);
355        out.result_names.push(Some("tag".into()));
356    }
357    for (index, (node, variant)) in nodes.iter().zip(&source.variants).enumerate() {
358        let prefix = format!("{}a{index}_", context.output_prefix);
359        let ctx = KotlinContext {
360            output_prefix: &prefix,
361            ..*context
362        };
363        let mut child = if declaration.is_some() {
364            let mut fields = variant_fields(node, variant)?;
365            let WireNode::Branch { children, .. } = node else {
366                unreachable!()
367            };
368            kotlin::output_components_checked(children, &ctx, owners, |child| {
369                let (field, _) = fields.next().ok_or("missing enum field")??;
370                data::check_property_output(&field.metadata.kotlin_type, &child.result_types)?;
371                match child.delivered.len() {
372                    0 => { child.delivered.push("kotlin.Unit".into()); child.result_types.push("kotlin.Unit".into()); }
373                    1 => {}
374                    _ => return Err("ordinary enum fields require one Kotlin value; use enum_fields for expanded delivery".into()),
375                }
376                Ok(())
377            })?
378        } else {
379            kotlin::output_kotlin_scoped(node, &ctx, owners)?
380        };
381        let count = crate::group::kotlin_result_fields(node).len();
382        kotlin::remap_output(&mut child, context.result, context.result, offset, count);
383        offset += count;
384        out.output_locals.push_str(&child.output_locals);
385        if !child.cleanup.is_empty() {
386            cleanup.push_str(&format!(
387                "if ({tag_expr} == {index}) {{ {} }}\n",
388                child.cleanup
389            ));
390        }
391        if let Some(class) = &class_name {
392            let variant_class = variant_name(declaration.unwrap(), &variant.name);
393            let args = child.delivered.join(", ");
394            let constructor = if variant.fields.is_empty() {
395                format!("{class}.{variant_class}")
396            } else {
397                format!("{class}.{variant_class}({args})")
398            };
399            body.push_str(&format!(
400                "{index} -> {{ {} {constructor} }}\n",
401                child.materialize
402            ));
403        } else {
404            let source_names = expanded_names(node, &ctx, owners)?;
405            let count = child.result_types.len();
406            for (i, (ty, expression)) in child
407                .result_types
408                .into_iter()
409                .zip(child.delivered)
410                .enumerate()
411            {
412                let name = format!("{}a{index}_value{i}", context.output_prefix);
413                let primitive = matches!(
414                    ty.trim_start_matches("kotlin."),
415                    "UByte"
416                        | "UShort"
417                        | "UInt"
418                        | "Boolean"
419                        | "Byte"
420                        | "Short"
421                        | "Int"
422                        | "Long"
423                        | "ULong"
424                        | "Float"
425                        | "Double"
426                        | "Char"
427                );
428                let absent = match ty.trim_start_matches("kotlin.") {
429                    "Boolean" => "false",
430                    "Byte" => "0.toByte()",
431                    "Short" => "0.toShort()",
432                    "Int" => "0",
433                    "Long" => "0L",
434                    "ULong" => "0uL",
435                    "UInt" => "0u",
436                    "UShort" => "0.toUShort()",
437                    "UByte" => "0.toUByte()",
438
439                    "Float" => "0.0f",
440                    "Double" => "0.0",
441                    "Char" => "'\\u0000'",
442                    _ => "null",
443                };
444                let result_type = if primitive || ty.ends_with('?') {
445                    ty
446                } else {
447                    format!("{ty}?")
448                };
449                out.output_locals
450                    .push_str(&format!("var {name}: {result_type} = {absent}\n"));
451                out.result_types.push(result_type);
452                let variant = lower_camel(variant.name.to_string().trim_start_matches("r#"));
453                let field_name = child
454                    .result_names
455                    .get(i)
456                    .cloned()
457                    .flatten()
458                    .or_else(|| source_names.get(i).cloned().flatten());
459                out.result_names.push(Some(if let Some(name) = field_name {
460                    format!("{variant}{}", upper_first(&name))
461                } else if count == 1 && variant != "tag" {
462                    variant
463                } else {
464                    format!("{variant}Value{i}")
465                }));
466                out.delivered.push(name.clone());
467                expanded.push((index, format!("{name} = {expression}\n")));
468            }
469            body.push_str(&format!(
470                "if ({tag_expr} == {index}) {{ {} {} }}\n",
471                child.materialize,
472                expanded
473                    .iter()
474                    .filter(|(arm, _)| *arm == index)
475                    .map(|(_, s)| s.as_str())
476                    .collect::<String>()
477            ));
478        }
479    }
480    if let Some(class) = &class_name {
481        out.output_locals
482            .push_str(&format!("var {local}: {class}? = null\n"));
483        out.materialize = format!(
484            "{local} = when ({tag_expr}) {{ {body} else -> error(\"invalid native enum tag\") }}\n"
485        );
486        out.delivered = vec![format!("requireNotNull({local})")];
487        out.result_types = vec![class.clone()];
488        out.result_names = vec![None];
489    } else {
490        out.materialize = body;
491    }
492    out.cleanup = cleanup;
493    Ok(out)
494}

496fn lower_camel(name: &str) -> String {
497    let value = camel(name.trim_start_matches("r#"));
498    let mut chars = value.chars();
499    chars
500        .next()
501        .map(|c| c.to_lowercase().collect::<String>() + chars.as_str())
502        .unwrap_or_default()
503}
504fn upper_first(name: &str) -> String {
505    let mut chars = name.chars();
506    chars
507        .next()
508        .map(|c| c.to_uppercase().collect::<String>() + chars.as_str())
509        .unwrap_or_default()
510}
511// Read source field paths from the same resolved tree. Publicly grouped values
512// remain one property; expanded child fields retain their meaningful names.
513fn expanded_names(
514    node: &JniOutputWireNode,
515    context: &KotlinContext<'_>,
516    owners: Option<&str>,
517) -> Result<Vec<Option<String>>> {
518    if let WireNode::Branch {
519        meta:
520            JniOutputWireBranch::KDataClass { .. }
521            | JniOutputWireBranch::KSealed { .. }
522            | JniOutputWireBranch::KEnumParams { .. },
523        ..
524    } = node
525    {
526        return Ok(kotlin::output_kotlin_scoped(node, context, owners)?.result_names);
527    }
528    if let WireNode::Annotation {
529        meta: JniOutputWireAnnotation::RField { name: field },
530        child,
531        ..
532    } = node
533    {
534        let mut names = expanded_names(child, context, owners)?;
535        let field = field.to_string();
536        let tuple = field
537            .strip_prefix('v')
538            .is_some_and(|n| !n.is_empty() && n.chars().all(|c| c.is_ascii_digit()));
539        if !tuple {
540            let field = lower_camel(&field);
541            for name in &mut names {
542                *name = Some(
543                    name.as_ref()
544                        .map_or_else(|| field.clone(), |n| format!("{field}{}", upper_first(n))),
545                );
546            }
547        }
548        return Ok(names);
549    }
550    if let WireNode::Branch { children, .. } = node {
551        return Ok(children
552            .iter()
553            .map(|n| expanded_names(n, context, owners))
554            .collect::<Result<Vec<_>>>()?
555            .into_iter()
556            .flatten()
557            .collect());
558    }
559    Ok(kotlin::output_kotlin_scoped(node, context, owners)?.result_names)
560}

562pub(crate) fn render<I: KotlinInputConversion, O: KotlinOutputConversion>(
563    flat: &Flat<JniSchema<I, O>>,
564    class: &Class,
565    ty: &syn::Type,
566) -> Result<String> {
567    let source = flat.payload_enum(ty)?;
568    validate_variants(class, source)?;
569    let mut out = format!(
570        "package {}\n\npublic sealed interface {}{} {{\n",
571        class.pkg,
572        class.name,
573        if class.has_handles() {
574            " : AutoCloseable"
575        } else {
576            ""
577        }
578    );
579    for arm in &source.variants {
580        let name = variant_name(class, &arm.name);
581        if arm.fields.is_empty() {
582            out.push_str(&format!(
583                "public data object {name} : {}{}\n",
584                class.fqn(),
585                if class.has_handles() {
586                    " { override fun close() {} }"
587                } else {
588                    ""
589                }
590            ));
591            continue;
592        }
593        let nested = Class {
594            name: name.clone(),
595            pkg: class.pkg.clone(),
596            representation: ClassRepresentation::Data { has_handles: false },
597        };
598        let rendered = data::render_record_fields(
599            &nested,
600            arm.fields.iter().map(|f| (f.property(), &f.metadata)),
601        )?;
602        let text = rendered
603            .split_once("public data class")
604            .ok_or("invalid variant class")?
605            .1;
606        let mut text = format!("public data class{text}");
607        let at = text.find("\n)").ok_or("invalid variant constructor")? + 2;
608        text.insert_str(at, &format!(" : {}", class.fqn()));
609        if class.has_handles() {
610            let mut close = String::new();
611            for f in &arm.fields {
612                close.push_str(&close_member(
613                    flat,
614                    &f.ty,
615                    &format!("this.{}", kotlin_identifier(&f.property())),
616                    0,
617                )?);
618            }
619            if text.trim_end().ends_with('}') {
620                let at = text.rfind('}').unwrap();
621                text.insert_str(at, &format!("\noverride fun close() {{ {close} }}\n"));
622            } else {
623                text.push_str(&format!(" {{ override fun close() {{ {close} }} }}\n"));
624            }
625        }
626        out.push_str(&text);
627    }
628    out.push_str("}\n");
629    Ok(out)
630}
631fn close_member<S: prebindgen_flat::Schema<Field = data::JniField, Type = JniType>>(
632    flat: &Flat<S>,
633    ty: &syn::Type,
634    value: &str,
635    depth: usize,
636) -> Result<String> {
637    if let Some(inner) = prebindgen_flat::option_inner(ty) {
638        let name = format!("member{depth}");
639        let body = close_member(flat, inner, &name, depth + 1)?;
640        return Ok(if body.is_empty() {
641            String::new()
642        } else {
643            format!("{value}?.let {{ {name} -> {body} }}\n")
644        });
645    }
646    if let Some(inner) = prebindgen_flat::vec_inner(ty) {
647        let name = format!("member{depth}");
648        let body = close_member(flat, inner, &name, depth + 1)?;
649        return Ok(if body.is_empty() {
650            String::new()
651        } else {
652            format!("{value}.forEach {{ {name} -> {body} }}\n")
653        });
654    }
655    let Some(info) = flat.type_info(ty).ok() else {
656        return Ok(String::new());
657    };
658    if let Some(c) = info.metadata.conversion.input.as_ref().or(info
659        .metadata
660        .conversion
661        .output
662        .as_ref())
663    {
664        return close_member(flat, &c.representation, value, depth + 1);
665    }
666    let Some(class) = &info.metadata.class else {
667        return Ok(String::new());
668    };
669    Ok(match &class.representation {
670        ClassRepresentation::Handle { .. } => format!("{value}.close()\n"),
671        ClassRepresentation::Sealed {
672            has_handles: true, ..
673        } => format!("{value}.close()\n"),
674        ClassRepresentation::Data { .. } => flat
675            .record(ty)?
676            .fields
677            .iter()
678            .map(|f| {
679                close_member(
680                    flat,
681                    &f.ty,
682                    &format!(
683                        "{value}.{}",
684                        kotlin_identifier(f.name.to_string().trim_start_matches("r#"))
685                    ),
686                    depth + 1,
687                )
688            })
689            .collect::<Result<Vec<_>>>()?
690            .join(""),
691        ClassRepresentation::ConfiguredData
692        | ClassRepresentation::ConfiguredEnum { .. }
693        | ClassRepresentation::ConfiguredSealed { .. } => {
694            return Err(format!("JNI class `{}` is unresolved", class.fqn()))
695        }
696        ClassRepresentation::Enum(_) | ClassRepresentation::Sealed { .. } => String::new(),
697    })
698}

700fn tag_word(label: &str) -> String {
701    let mut stem = String::new();
702    let mut previous_lower = false;
703    for c in label.trim_start_matches("r#").chars() {
704        if c.is_uppercase() && previous_lower {
705            stem.push('_');
706        }
707        stem.push(c.to_ascii_uppercase());
708        previous_lower = c.is_lowercase() || c.is_ascii_digit();
709    }
710    stem
711}
712fn tag_stem(ty: &syn::Type) -> String {
713    let mut stem = tag_word(&names::type_label(ty));
714    if matches!(ty,syn::Type::Path(p) if p.path.segments.iter().any(|s|!matches!(s.arguments,syn::PathArguments::None)))
715    {
716        stem.push_str(&format!(
717            "_{:016X}",
718            prebindgen_reprc::schema_id(&names::type_identity(ty))
719        ));
720    }
721    stem
722}
723/// Publish source tags for every occurrence package and verify the complete
724/// source-tag catalogue against the loaded native library before any native call.
725pub(crate) fn emit_tags<I: KotlinInputConversion, O: KotlinOutputConversion>(
726    flat: &Flat<JniSchema<I, O>>,
727    rust: &mut TokenStream,
728    native: &mut String,
729    files: &mut BTreeMap<String, String>,
730) -> Result<()> {
731    fn visit<
732        Chain: group::ChainMetadata,
733        Branch: group::BranchMetadata,
734        Annotation: group::AnnotationMetadata,
735        Container: group::ContainerMetadata,
736    >(
737        node: &JniWireNode<Chain, Branch, Annotation, Container>,
738        package: &str,
739        by_package: &mut BTreeMap<String, BTreeSet<TypeRef>>,
740    ) -> Result<()> {
741        match node {
742            WireNode::Leaf(wires) => {
743                for wire in wires {
744                    if let JniWire::KCallback { callback } = &wire.kind {
745                        for root in callback.bulk_nodes() {
746                            match root {
747                                metadata::BoundaryNode::Input(node) => {
748                                    visit(node, &callback.package, by_package)?;
749                                }
750                                metadata::BoundaryNode::Output(node) => {
751                                    visit(node, &callback.package, by_package)?;
752                                }
753                            }
754                        }
755                    }
756                }
757            }
758            WireNode::Annotation { child, .. }
759            | WireNode::Chain { child, .. }
760            | WireNode::Container { child, .. } => visit(child, package, by_package)?,
761            WireNode::Branch { meta, children, .. } => {
762                if let Some(e) = meta.r_enum() {
763                    by_package
764                        .entry(package.into())
765                        .or_default()
766                        .insert(e.clone());
767                }
768                for child in children {
769                    visit(child, package, by_package)?;
770                }
771            }
772        }
773        Ok(())
774    }
775    let mut packages = BTreeMap::new();
776    for f in crate::bindings::views(flat)? {
777        {
778            let metadata = f.metadata;
779            for input in f.input_conversions()? {
780                visit(input.input_node(), &metadata.package, &mut packages)?;
781            }
782            let outcomes = f.output_conversion()?.output_outcomes();
783            for node in std::iter::once(&outcomes.success)
784                .chain(outcomes.source_error.iter())
785                .chain(outcomes.binding_failure.iter())
786            {
787                visit(node, &metadata.package, &mut packages)?;
788            }
789            if let Some(error) = f.output_conversion()?.domain_error() {
790                visit(
791                    &error.output().output_outcomes().success,
792                    &metadata.package,
793                    &mut packages,
794                )?;
795                if let Some((package, _)) = error.handler.rsplit_once('.') {
796                    visit(
797                        &error.output().output_outcomes().success,
798                        package,
799                        &mut packages,
800                    )?;
801                }
802            }
803        }
804    }
805    for callback in crate::callback::declarations(flat)? {
806        visit(callback.input_node(), &flat.metadata.package, &mut packages)?;
807    }
808    if packages.is_empty() {
809        return Ok(());
810    }
811    let mut catalogue = BTreeMap::new();
812    for (package, enums) in packages {
813        let mut constants = BTreeMap::new();
814        for reference in enums {
815            let source = crate::metadata::source_enum(&flat.types, &reference)?;
816            let identity = names::type_identity(&source.ty);
817            let stem = tag_stem(&source.ty);
818            for (index, variant) in source.variants.iter().enumerate() {
819                let variant = tag_word(&variant.name.to_string());
820                let name = format!("{stem}_{variant}_TAG");
821                validate_name(&name)?;
822                if constants
823                    .insert(
824                        name.clone(),
825                        format!("public const val {name}: Int = {index}\n"),
826                    )
827                    .is_some()
828                {
829                    return Err("payload enum tag constant collision".into());
830                }
831            }
832            catalogue.insert(identity, source);
833        }
834        files.insert(
835            format!("{}/PayloadTags.kt", package.replace('.', "/")),
836            format!(
837                "package {package}\n\n{}",
838                constants.values().cloned().collect::<String>()
839            ),
840        );
841    }
842    let schema = catalogue
843        .iter()
844        .map(|(ty, source)| {
845            format!(
846                "{ty}:{}",
847                source
848                    .variants
849                    .iter()
850                    .map(|v| v.name.to_string().trim_start_matches("r#").to_owned())
851                    .collect::<Vec<_>>()
852                    .join(";")
853            )
854        })
855        .collect::<Vec<_>>()
856        .join("|");
857    let hash = prebindgen_reprc::schema_id(&schema) as i64;
858    let symbol = super::symbol(&flat.metadata.package, "payloadEnumSchema");
859    rust.extend(quote!(#[no_mangle] pub extern "system" fn #symbol(_env: ::prebindgen_jni_runtime::jni::JNIEnv,_class: ::prebindgen_jni_runtime::jni::objects::JClass)->i64 {#hash}));
860    native.push_str(&format!("    @JvmStatic external fun payloadEnumSchema(): Long\n    init {{ require(payloadEnumSchema() == {hash}L) {{ \"incompatible payload enum tag mapping\" }} }}\n"));
861    Ok(())
862}
