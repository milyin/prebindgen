6/// Prepared boundary roots, including typed error converters and declarations
7/// whose callback interfaces are emitted without a direct exported occurrence.
8pub(crate) fn nodes<I: KotlinInputConversion, O: KotlinOutputConversion>(
9    flat: &Flat<JniSchema<I, O>>,
10) -> Vec<(bool, &kotlin::JniNode)> {
11    let mut nodes = Vec::new();
12    for function in flat.functions.values() {
13        for param in &function.params {
14            if let Some(plan) = &param.ty.slot {
15                nodes.push((true, plan.input_node()));
16            }
17        }
18        if let Some(plan) = &function.ret.slot {
19            let outcomes = plan.output_outcomes();
20            nodes.extend(
21                std::iter::once(&outcomes.success)
22                    .chain(outcomes.source_error.iter())
23                    .chain(outcomes.binding_failure.iter())
24                    .map(|node| (false, node)),
25            );
26        }
27        if let Some(error) = function.metadata.as_ref().and_then(|m| m.error.as_deref()) {
28            nodes.push((false, &error.output.output_outcomes().success));
29        }
30    }
31    for input in &flat.metadata.callbacks {
32        nodes.push((true, input.input_node()));
33    }
34    nodes
35}

37/// Metadata schema for a Kotlin/JNI binding model.
38/// `I` and `O` select occurrence converter types while declaration metadata keeps
39/// the JNI representation. Defaults also accept custom converters through their
40/// owned containers. No schema value is needed to construct Flat.
41pub struct JniSchema<I = JniInput, O = JniOutput>(std::marker::PhantomData<fn() -> (I, O)>);
42impl<I, O> prebindgen_flat::Schema for JniSchema<I, O> {
43    type Input = I;
44    type Output = O;
45    type Module = JniModule;
46    type Function = Option<JniFunction>;
47    type Type = JniType;
48    type Field = data::JniField;
49    type Variant = ();
50}
51/// Settings shared by the generated Kotlin/JNI module.
52#[derive(Clone, Debug, Default)]
53pub struct JniModule {
54    /// Root Kotlin package containing JNINative and shared infrastructure.
55    pub package: String,
56    /// Kotlin initialization statements in JNINative.
57    pub native_init: String,
58    /// Prepared callback declarations, including interfaces unused by exported functions.
59    pub callbacks: Vec<JniInput>,
60    /// Use bulk for materialized collection outputs with automatic transport.
61    pub bulk_collection_outputs: bool,
62}
63/// Export and placement decisions belonging to one function.
64#[derive(Clone, Debug, Default)]
65pub struct JniFunction {
66    /// Fully qualified destination package.
67    pub package: String,
68    /// Destination function name.
69    pub name: String,
70    /// Declaration order used when rendering functions.
71    pub order: usize,
72    pub(crate) error: Option<Rc<errors::ErrorBinding>>,
73}
74/// Kotlin/JNI declaration information shared by occurrences of a Rust type.
75#[derive(Clone, Debug, Default)]
76pub struct JniType {
77    pub(crate) class: Option<Rc<Class>>,
78    pub(crate) enumeration: Option<Rc<JniEnum>>,
79    pub(crate) errors: BTreeMap<bool, Rc<errors::ErrorBinding>>,
80    pub(crate) error_class: Option<String>,
81}
