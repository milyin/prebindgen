1//! Shared immutable row classes and their native constructor registration.
2use super::*;
3use kotlin::JniNode;
4use prebindgen_flat::WireNode;

6pub(crate) fn supported(element: &JniNode) -> bool {
7    group::row_fields(element)
8        .map(|w| w.kind.parameter_slots())
9        .sum::<usize>()
10        <= 254
11}
12pub(crate) fn cache(element: &JniNode) -> Result<syn::Ident> {
13    let (name, _) = sequence_callback::capture(element, &mut names::Names::default())?;
14    Ok(format_ident!("__pb_row_type_{name}"))
15}
16fn visit<'a>(node: &'a JniNode, input: bool, rows: &mut Vec<&'a JniNode>) {
17    match node {
18        WireNode::Chain {
19            meta: JniChain::Sequence(JniSequence::TypedBuffered),
20            child,
21            ..
22        } if !input => {
23            rows.push(child);
24            visit(child, input, rows);
25        }
26        WireNode::Chain { child, .. } => visit(child, input, rows),
27        WireNode::Branch { children, .. } => {
28            for child in children {
29                visit(child, input, rows)
30            }
31        }
32        WireNode::Leaf(wires) => {
33            for wire in wires {
34                if let JniWire::Callback { callback } = &wire.kind {
35                    for (input, node) in callback.bulk_nodes() {
36                        visit(node, input, rows);
37                    }
38                }
39            }
40        }
41    }
42}
43pub(crate) fn emit<I: KotlinInputConversion, O: KotlinOutputConversion>(
44    flat: &Flat<JniSchema<I, O>>,
45    rust: &mut TokenStream,
46    native: &mut String,
47    names: &mut names::Names,
48    emitted: &mut BTreeSet<String>,
49) -> Result<()> {
50    let mut rows = Vec::new();
51    for (input, node) in metadata::nodes(flat) {
52        visit(node, input, &mut rows);
53    }
54    let mut schemas = BTreeMap::new();
55    for row in rows {
56        let (name, declaration) = sequence_callback::capture(row, names)?;
57        if emitted.insert(name.clone()) {
58            native.push_str(&declaration);
59        }
60        schemas.insert(name, row);
61    }
62    if schemas.is_empty() {
63        return Ok(());
64    }
65    let method = names.name("init", "OutputRows", &["typed output row constructors"])?;
66    let symbol = symbol(&flat.metadata.package, &method);
67    let mut initialize = TokenStream::new();
68    let mut classes = Vec::new();
69    for (index, (name, row)) in schemas.into_iter().enumerate() {
70        let cache = cache(row)?;
71        let descriptors = group::row_fields(row)
72            .map(|w| w.kind.jvm_descriptor())
73            .collect::<Vec<_>>();
74        rust.extend(quote! {
75            #[allow(non_upper_case_globals)]
76            static #cache: ::prebindgen_jni_runtime::OutputRowType = ::prebindgen_jni_runtime::OutputRowType::new();
77        });
78        let index = index as i32;
79        initialize.extend(quote! {
80            let class=env.get_object_array_element(&classes,#index).map_err(|e|e.to_string())?;
81            let class=env.auto_local(::prebindgen_jni_runtime::jni::objects::JClass::from(class));
82            #cache.initialize(&mut env,&class,&[#(#descriptors),*])?;
83        });
84        classes.push(format!("{name}::class.java"));
85    }
86    let count = classes.len() as i32;
87    rust.extend(quote! {
88        #[no_mangle]
89        pub extern "system" fn #symbol(mut env: ::prebindgen_jni_runtime::jni::JNIEnv<'_>,_class: ::prebindgen_jni_runtime::jni::objects::JClass<'_>,classes: ::prebindgen_jni_runtime::jni::objects::JObjectArray<'_>) {
90            if let Err(error)=::prebindgen_runtime::boundary(|| {
91                if env.get_array_length(&classes).map_err(|e|e.to_string())? != #count {return Err("output row type count differs".into());}
92                #initialize
93                Ok(())
94            }) {::prebindgen_jni_runtime::fail(&mut env,&error);}
95        }
96    });
97    // Actual Class objects come from the binding's loader. Registration runs
98    // after the configured library loader and before any native wrapper call.
99    native.push_str(&format!("    @JvmStatic private external fun {method}(classes: Array<Class<*>>)\n    init {{ {method}(arrayOf({})) }}\n",classes.join(", ")));
100    Ok(())
101}
