27/// Resolved incoming conversion. Configuration cannot enter callback domains:
28/// ```compile_fail
29/// use prebindgen_jni::{JniInputArrow,JniInputConfigArrow};
30/// fn final_arrow(config:JniInputConfigArrow)->JniInputArrow { config }
31/// ```
32/// Final leaves and enum branches cannot contain configuration payloads:
33/// ```compile_fail
34/// use prebindgen_jni::{JniInput,JniInputLeafOp};
35/// let _:JniInput=JniInput::Leaf(JniInputLeafOp::ConfigDefault(()));
36/// ```
37/// ```compile_fail
38/// use prebindgen_jni::{JniInput,JniInputBranch};
39/// let _:JniInput=JniInput::Branch {
40///     operation:JniInputBranch::ConfigEnumFields(false),children:vec![],
41/// };
42/// ```
43pub type JniInput = InputNode<JniInputOps, JniOutputOps>;
44/// Resolved outgoing conversion. Partial named fields have no final payload:
45/// ```compile_fail
46/// use prebindgen_jni::{JniOutput,JniOutputBranch,JniNamedFields,ProjectionMode};
47/// let _:JniOutput=JniOutput::Branch {
48///     operation:JniOutputBranch::ConfigNamedFields(JniNamedFields {fields:vec![],mode:ProjectionMode::Owned}),
49///     children:vec![],
50/// };
51/// ```
52/// None is a configuration payload, not an inhabitant of the final payload type:
53/// ```compile_fail
54/// use prebindgen_jni::{JniOutput,JniOutputLeafOp};
55/// let _:JniOutput=JniOutput::Leaf(JniOutputLeafOp::ConfigValue(None));
56/// ```
57pub type JniOutput = OutputNode<JniInputOps, JniOutputOps>;
58/// Incoming conversion configuration.
59/// ```
60/// use prebindgen_jni::{JniInputConfig,JniOutputConfig,prelude::*};
61/// let _:JniInputConfig=JniInputConfig::auto();
62/// let _:JniOutputConfig=JniOutputConfig::auto();
63/// ```
64pub type JniInputConfig = InputNode<JniInputConfigOps, JniOutputConfigOps>;
65/// Outgoing conversion configuration.
66pub type JniOutputConfig = OutputNode<JniInputConfigOps, JniOutputConfigOps>;
67/// Incoming operation set with concrete configuration payloads.

331pub(crate) fn lift_input_bulk(tree: JniInput) -> JniInput {
332    match tree {
333        JniInput::Branch {
334            operation: JniInputBranch::Enum { value },
335            children: variants,
336        } => JniInput::Branch {
337            operation: JniInputBranch::Enum { value },
338            children: variants.into_iter().map(lift_input_bulk).collect(),
339        },
340        JniInput::Unary {
341            operation: JniInputUnary::Convert { conversion },
342            child: input,
343        } => JniInput::Unary {
344            operation: JniInputUnary::Convert { conversion },
345            child: Box::new(lift_input_bulk(*input)),
346        },
347        JniInput::Unary {
348            operation: JniInputUnary::Bulk,
349            child: c,
350        } => JniInput::Unary {
351            operation: JniInputUnary::Bulk,
352            child: Box::new(lift_input_bulk(*c)),
353        },
354        JniInput::Unary {
355            operation: JniInputUnary::Temporary,
356            child: c,
357        } => JniInput::Unary {
358            operation: JniInputUnary::Temporary,
359            child: Box::new(lift_input_bulk(*c)),
360        },
361        JniInput::Unary {
362            operation: JniInputUnary::Sequence,
363            child: c,
364        } => {
365            let child = lift_input_bulk(*c);
366            if input_has_bulk(&child) {
367                JniInput::Unary {
368                    operation: JniInputUnary::Bulk,
369                    child: Box::new(JniInput::Unary {
370                        operation: JniInputUnary::Sequence,
371                        child: Box::new(child),
372                    }),
373                }
374            } else {
375                JniInput::Unary {
376                    operation: JniInputUnary::Sequence,
377                    child: Box::new(child),
378                }
379            }
380        }
381        JniInput::Unary {
382            operation: JniInputUnary::Optional,
383            child: c,
384        } => match lift_input_bulk(*c) {
385            JniInput::Unary {
386                operation: JniInputUnary::Bulk,
387                child: c,
388            } => JniInput::Unary {
389                operation: JniInputUnary::Bulk,
390                child: Box::new(JniInput::Unary {
391                    operation: JniInputUnary::Optional,
392                    child: c,
393                }),
394            },
395            c => JniInput::Unary {
396                operation: JniInputUnary::Optional,
397                child: Box::new(c),
398            },
399        },
400        JniInput::Branch {
401            operation: JniInputBranch::Fields,
402            children: c,
403        } => JniInput::Branch {
404            operation: JniInputBranch::Fields,
405            children: c.into_iter().map(lift_input_bulk).collect(),
406        },
407        JniInput::Branch {
408            operation: JniInputBranch::Record,
409            children: c,
410        } => JniInput::Branch {
411            operation: JniInputBranch::Record,
412            children: c.into_iter().map(lift_input_bulk).collect(),
413        },
414        JniInput::Branch {
415            operation: JniInputBranch::Selector,
416            children: c,
417        } => JniInput::Branch {
418            operation: JniInputBranch::Selector,
419            children: c.into_iter().map(lift_input_bulk).collect(),
420        },
421        JniInput::Branch {
422            operation: JniInputBranch::Constructor { function },
423            children: args,
424        } => JniInput::Branch {
425            operation: JniInputBranch::Constructor { function },
426            children: args.into_iter().map(lift_input_bulk).collect(),
427        },
428        JniInput::Branch {
429            operation,
430            children,
431        } => JniInput::Branch {
432            operation,
433            children: children.into_iter().map(lift_input_bulk).collect(),
434        },
435        c => c,
436    }
437}
438pub(crate) fn lift_output_bulk(tree: JniOutput) -> JniOutput {
439    match tree {
440        JniOutput::Branch {
441            operation: JniOutputBranch::Enum { value, mode },
442            children: variants,
443        } => JniOutput::Branch {
444            operation: JniOutputBranch::Enum { value, mode },
445            children: variants.into_iter().map(lift_output_bulk).collect(),
446        },
447        JniOutput::Unary {
448            operation: JniOutputUnary::Convert { conversion },
449            child: output,
450        } => JniOutput::Unary {
451            operation: JniOutputUnary::Convert { conversion },
452            child: Box::new(lift_output_bulk(*output)),
453        },
454        JniOutput::Unary {
455            operation: JniOutputUnary::Bulk,
456            child: c,
457        } => JniOutput::Unary {
458            operation: JniOutputUnary::Bulk,
459            child: Box::new(lift_output_bulk(*c)),
460        },
461        JniOutput::Unary {
462            operation: JniOutputUnary::Sequence { delivery, mode },
463            child: element,
464        } => {
465            let child = lift_output_bulk(*element);
466            if delivery != SequenceDelivery::Callback && output_has_bulk(&child) {
467                JniOutput::Unary {
468                    operation: JniOutputUnary::Bulk,
469                    child: Box::new(JniOutput::Unary {
470                        operation: JniOutputUnary::Sequence { delivery, mode },
471                        child: Box::new(child),
472                    }),
473                }
474            } else {
475                JniOutput::Unary {
476                    operation: JniOutputUnary::Sequence { delivery, mode },
477                    child: Box::new(child),
478                }
479            }
480        }
481        JniOutput::Unary {
482            operation: JniOutputUnary::Optional,
483            child: c,
484        } => match lift_output_bulk(*c) {
485            JniOutput::Unary {
486                operation: JniOutputUnary::Bulk,
487                child: c,
488            } => JniOutput::Unary {
489                operation: JniOutputUnary::Bulk,
490                child: Box::new(JniOutput::Unary {
491                    operation: JniOutputUnary::Optional,
492                    child: c,
493                }),
494            },
495            c => JniOutput::Unary {
496                operation: JniOutputUnary::Optional,
497                child: Box::new(c),
498            },
499        },
500        JniOutput::Branch {
501            operation: JniOutputBranch::Fields,
502            children: c,
503        } => JniOutput::Branch {
504            operation: JniOutputBranch::Fields,
505            children: c.into_iter().map(lift_output_bulk).collect(),
506        },
507        JniOutput::Branch {
508            operation: JniOutputBranch::OwnedFields,
509            children: c,
510        } => JniOutput::Branch {
511            operation: JniOutputBranch::OwnedFields,
512            children: c.into_iter().map(lift_output_bulk).collect(),
513        },
514        JniOutput::Branch {
515            operation: JniOutputBranch::Record { mode },
516            children: fields,
517        } => JniOutput::Branch {
518            operation: JniOutputBranch::Record { mode },
519            children: fields.into_iter().map(lift_output_bulk).collect(),
520        },
521        JniOutput::Branch {
522            operation: JniOutputBranch::Combine,
523            children: c,
524        } => JniOutput::Branch {
525            operation: JniOutputBranch::Combine,
526            children: c.into_iter().map(lift_output_bulk).collect(),
527        },
528        JniOutput::Unary {
529            operation: JniOutputUnary::Result { error },
530            child: ok,
531        } => JniOutput::Unary {
532            operation: JniOutputUnary::Result { error },
533            child: Box::new(lift_output_bulk(*ok)),
534        },
535        JniOutput::Unary {
536            operation: JniOutputUnary::Getter { function },
537            child: output,
538        } => JniOutput::Unary {
539            operation: JniOutputUnary::Getter { function },
540            child: Box::new(lift_output_bulk(*output)),
541        },
542        JniOutput::Unary {
543            operation: JniOutputUnary::Into { function },
544            child: output,
545        } => JniOutput::Unary {
546            operation: JniOutputUnary::Into { function },
547            child: Box::new(lift_output_bulk(*output)),
548        },
549        JniOutput::Unary {
550            operation: JniOutputUnary::Named { name },
551            child: output,
552        } => JniOutput::Unary {
553            operation: JniOutputUnary::Named { name },
554            child: Box::new(lift_output_bulk(*output)),
555        },
556        JniOutput::Branch {
557            operation,
558            children,
559        } => JniOutput::Branch {
560            operation,
561            children: children.into_iter().map(lift_output_bulk).collect(),
562        },
563        c => c,
564    }
565}

567fn input_has_bulk(tree: &JniInput) -> bool {
568    match tree {
569        JniInput::Branch {
570            operation: JniInputBranch::Enum { .. },
571            children: variants,
572            ..
573        } => variants.iter().any(input_has_bulk),
574        JniInput::Unary {
575            operation: JniInputUnary::Convert { .. },
576            child: input,
577            ..
578        } => input_has_bulk(input),
579        JniInput::Unary {
580            operation: JniInputUnary::Bulk,
581            child: _,
582        } => true,
583        JniInput::Unary {
584            operation: JniInputUnary::Optional,
585            child: c,
586        }
587        | JniInput::Unary {
588            operation: JniInputUnary::Sequence,
589            child: c,
590        } => input_has_bulk(c),
591        JniInput::Branch {
592            operation: JniInputBranch::Fields,
593            children: c,
594        }
595        | JniInput::Branch {
596            operation: JniInputBranch::Record,
597            children: c,
598        }
599        | JniInput::Branch {
600            operation: JniInputBranch::Selector,
601            children: c,
602        } => c.iter().any(input_has_bulk),
603        JniInput::Branch {
604            operation: JniInputBranch::Constructor { .. },
605            children: args,
606            ..
607        } => args.iter().any(input_has_bulk),
608        JniInput::Branch { children, .. } => children.iter().any(input_has_bulk),
609        _ => false,
610    }
611}
612fn output_has_bulk(tree: &JniOutput) -> bool {
613    match tree {
614        JniOutput::Branch {
615            operation: JniOutputBranch::Enum { .. },
616            children: variants,
617            ..
618        } => variants.iter().any(output_has_bulk),
619        // Frames inside callback elements belong to those elements, including
620        // when the callback sequence is nested in another structural node.
621        JniOutput::Unary {
622            operation:
623                JniOutputUnary::Sequence {
624                    delivery: SequenceDelivery::Callback,
625                    ..
626                },
627            ..
628        } => false,
629        JniOutput::Unary {
630            operation: JniOutputUnary::Convert { .. },
631            child: output,
632            ..
633        } => output_has_bulk(output),
634        JniOutput::Unary {
635            operation: JniOutputUnary::Bulk,
636            child: _,
637        } => true,
638        JniOutput::Unary {
639            operation: JniOutputUnary::Optional,
640            child: c,
641        }
642        | JniOutput::Unary {
643            operation: JniOutputUnary::Named { .. },
644            child: c,
645            ..
646        }
647        | JniOutput::Unary {
648            operation: JniOutputUnary::Getter { .. },
649            child: c,
650            ..
651        }
652        | JniOutput::Unary {
653            operation: JniOutputUnary::Into { .. },
654            child: c,
655            ..
656        }
657        | JniOutput::Unary {
658            operation: JniOutputUnary::Sequence { .. },
659            child: c,
660            ..
661        } => output_has_bulk(c),
662        JniOutput::Branch {
663            operation: JniOutputBranch::Fields,
664            children: c,
665        }
666        | JniOutput::Branch {
667            operation: JniOutputBranch::OwnedFields,
668            children: c,
669        }
670        | JniOutput::Branch {
671            operation: JniOutputBranch::Combine,
672            children: c,
673        }
674        | JniOutput::Branch {
675            operation: JniOutputBranch::Record { .. },
676            children: c,
677            ..
678        } => c.iter().any(output_has_bulk),
679        JniOutput::Branch { children, .. } => children.iter().any(output_has_bulk),
680        _ => false,
681    }
682}

684pub(crate) fn sequence_callback(
685    cx: &mut crate::JniResolve<'_>,
686    ty: &syn::Type,
687    mut output: JniOutput,
688    structural_default: bool,
689) -> Result<JniOutput> {
690    fn select(output: &mut JniOutput) -> usize {
691        match output {
692            JniOutput::Unary {
693                operation: JniOutputUnary::Sequence { delivery, .. },
694                ..
695            } => {
696                *delivery = SequenceDelivery::Callback;
697                1
698            }
699            JniOutput::Unary {
700                operation: JniOutputUnary::Optional,
701                child,
702            }
703            | JniOutput::Unary {
704                operation: JniOutputUnary::Bulk,
705                child,
706            }
707            | JniOutput::Unary {
708                operation: JniOutputUnary::Named { .. },
709                child,
710                ..
711            }
712            | JniOutput::Unary {
713                operation: JniOutputUnary::Getter { .. },
714                child,
715                ..
716            }
717            | JniOutput::Unary {
718                operation: JniOutputUnary::Into { .. },
719                child,
720                ..
721            }
722            | JniOutput::Unary {
723                operation: JniOutputUnary::Convert { .. },
724                child,
725                ..
726            } => select(child),
727            JniOutput::Branch {
728                operation: JniOutputBranch::Combine,
729                children,
730            }
731            | JniOutput::Branch {
732                operation: JniOutputBranch::Fields,
733                children,
734            }
735            | JniOutput::Branch {
736                operation: JniOutputBranch::OwnedFields,
737                children,
738            }
739            | JniOutput::Branch {
740                operation: JniOutputBranch::Record { .. },
741                children,
742                ..
743            } => children.iter_mut().map(select).sum(),
744            _ => 0,
745        }
746    }
747    if select(&mut output) != 1 {
748        fn structural(ty: &syn::Type, output: &JniOutput) -> Option<crate::JniOutputConfig> {
749            match output {
750                JniOutput::Leaf(JniOutputLeafOp::Value)
751                    if prebindgen_flat::sequence_parts(ty).is_some() =>
752                {
753                    Some(crate::JniOutputConfig::Unary {
754                        operation: crate::JniOutputUnary::Sequence {
755                            delivery: SequenceDelivery::Callback,
756                            mode: None,
757                        },
758                        child: Box::new(crate::JniOutputConfig::auto()),
759                    })
760                }
761                JniOutput::Unary {
762                    operation: JniOutputUnary::Optional,
763                    child,
764                } => Some(crate::JniOutputConfig::Unary {
765                    operation: crate::JniOutputUnary::Optional,
766                    child: Box::new(structural(prebindgen_flat::option_inner(ty)?, child)?),
767                }),
768                _ => None,
769            }
770        }
771        if structural_default {
772            if let Some(tree) = structural(ty, &output) {
773                output = cx.output(ty, &tree)?;
774            } else {
775                return Err("callback_return requires exactly one outer sequence".into());
776            }
777        } else {
778            return Err("callback_return requires exactly one outer sequence".into());
779        }
780    }
781    Ok(output)
782}
