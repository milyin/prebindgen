1//! Generic function binding declarations. Trees belong to the adapter.
2use std::collections::BTreeMap;
3
4#[macro_export]
5/// Reference a source function and configure its exported binding.
6macro_rules! fun {
7    ($name:ident) => {
8        $crate::DeclFunction::<_, _>::new(
9            $crate::ident(stringify!($name)).expect("function identifier"),
10        )
11    };
12}
13/// Which sequence return variants an adapter should expose.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub enum CallbackReturn {
16    /// Keep the ordinary return and add a callback variant.
17    Additional,
18    /// Expose only the callback variant, including multi-value Kotlin elements
19    /// that have no ordinary List representation.
20    Only,
21}
22
23/// A source callable to export, with optional destination name and occurrence overrides.
24#[derive(Clone, Debug)]
25pub struct DeclFunction<InTree, OutTree, Extension = std::convert::Infallible> {
26    /// Adapter-defined settings, preserved by common fluent methods.
27    pub extension: Option<Extension>,
28    /// Treat an outer standard Result as ordinary variant data.
29    pub result_value: bool,
30    /// Also emit an explicit callback-delivery variant for a sequence return.
31    pub callback_return: Option<CallbackReturn>,
32    /// Unqualified source function name in the flat namespace.
33    pub name: String,
34    /// Explicit destination name; supported by JNI, currently rejected by C.
35    pub foreign: Option<String>,
36    /// Input expansion overrides keyed by source parameter name.
37    pub params: BTreeMap<String, InTree>,
38    /// Result expansion override, replacing the type default.
39    pub ret: Option<OutTree>,
40    /// Source mapper for a top-level Result error; it takes E or &E and returns String.
41    pub error: Option<String>,
42}
43impl<InTree, OutTree, Extension> DeclFunction<InTree, OutTree, Extension> {
44    /// Reference a source function without overriding names or conversions.
45    ///
46    /// `name` is the function's unqualified source name.
47    pub fn new(name: syn::Ident) -> Self {
48        Self {
49            extension: None,
50            result_value: false,
51            callback_return: None,
52            name: name.to_string(),
53            foreign: None,
54            params: BTreeMap::new(),
55            ret: None,
56            error: None,
57        }
58    }
59    /// Configure adapter-specific settings without replacing common declaration data.
60    pub fn with_extension<Next>(self, extension: Next) -> DeclFunction<InTree, OutTree, Next> {
61        DeclFunction {
62            extension: Some(extension),
63            result_value: self.result_value,
64            callback_return: self.callback_return,
65            name: self.name,
66            foreign: self.foreign,
67            params: self.params,
68            ret: self.ret,
69            error: self.error,
70        }
71    }
72    /// Extract an undecorated helper's source name, rejecting binding-only settings.
73    pub fn source_name(self) -> String {
74        self.reference_name(false)
75    }
76    /// Source reference used by constructor, getter and error-mapper declarations.
77    /// Extract a helper source name, optionally accepting a getter view name.
78    pub fn reference_name(self, allow_name: bool) -> String {
79        assert!(
80            self.extension.is_none()
81                && self.params.is_empty()
82                && self.ret.is_none()
83                && self.error.is_none()
84                && !self.result_value
85                && self.callback_return.is_none(),
86            "unsupported settings on this source helper"
87        );
88        assert!(
89            allow_name || self.foreign.is_none(),
90            "this helper reference does not support renaming"
91        );
92        self.name
93    }
94    /// Replace one parameter's default conversion using the production API.
95    pub fn expand_param(self, name: impl AsRef<str>, decl: impl Into<InTree>) -> Self {
96        self.input(name, decl)
97    }
98    /// Replace this return's default conversion using the production API.
99    pub fn expand_return(self, decl: impl Into<OutTree>) -> Self {
100        self.output(decl)
101    }
102    /// Deliver the whole source Result as data instead of routing its Err arm.
103    /// Return overrides then address the complete Result. Incompatible with on_error.
104    pub fn result_as_value(mut self) -> Self {
105        self.result_value = true;
106        self
107    }
108    /// Add callback delivery alongside the ordinary sequence return.
109    ///
110    /// C emits `<function>_each`, whose visitor receives each element's converted
111    /// fields. Kotlin emits `<function>Fold(initial, ..., fold)`, whose folder
112    /// receives an accumulator and each element's converted Kotlin values.
113    /// Supports one projected sequence, optionally inside Option and an outer Result.
114    /// Existing buffer/list returns are preserved. Adapter builders validate the
115    /// return shape and generated-name collisions.
116    ///
117    /// On adapters supporting bulk, expansions stay within each callback element.
118    /// The ordinary variant
119    /// and callback variant are prepared separately from the same declaration.
120    /// For a source returning `Vec<Row>`, this selects a frame per row for the
121    /// callback variant; its ordinary return keeps a frame for the collection:
122    /// For a source returning `Vec<Vec<Row>>`, target the inner collection to
123    /// select one bulk frame per batch:
124    /// An explicit bulk wrapper around the entire callback sequence is invalid;
125    /// apply bulk to its element expansion instead. This also applies to a bulk
126    /// wrapper supplied by a whole-collection type default. Nested automatic
127    /// sequences stay ordinary inside a callback element; explicit bulk opts in
128    /// to packing them. Chain order does not affect these policies, and
129    /// per-function expansions override type-wide defaults.
130    pub fn callback_return(mut self) -> Self {
131        self.callback_return = Some(CallbackReturn::Additional);
132        self
133    }
134    /// Expose only callback delivery, without generating an ordinary return.
135    ///
136    /// This explicitly permits decomposed multi-value Kotlin elements without
137    /// requiring a Kotlin data class. Generated names are the same as for
138    /// `callback_return`: C `_each` and Kotlin `Fold`.
139    pub fn callback_return_only(mut self) -> Self {
140        self.callback_return = Some(CallbackReturn::Only);
141        self
142    }
143    /// Map the error of a top-level `Result<T, E>` return with a source function.
144    ///
145    /// The helper must take exactly `E` or `&E` and return `String`. Only its
146    /// source name is accepted; it is selected for capture but is not exported.
147    /// The success value uses normal return defaults and `expand_return` overrides.
148    /// Replaces any previously selected mapper. Non-Result returns reject this option.
149    pub fn on_error(mut self, mapper: impl Into<DeclFunction<InTree, OutTree, Extension>>) -> Self {
150        self.error = Some(mapper.into().reference_name(false));
151        self
152    }
153    /// Set a literal destination name, subject to adapter validation.
154    ///
155    /// The JNI adapter uses it instead of its default camel-case name.
156    pub fn name(mut self, name: impl Into<String>) -> Self {
157        self.foreign = Some(name.into());
158        self
159    }
160    /// Replace the expansion for one source parameter.
161    ///
162    /// # Arguments
163    ///
164    /// * `name` — the parameter's name, as written in the source.
165    /// * `decl` — how to build that argument instead of its type's default.
166    ///
167    /// # Panics
168    /// Panics if that parameter already has an override. Resolution checks that the
169    /// parameter exists and the expansion type matches.
170    pub fn input(mut self, name: impl AsRef<str>, decl: impl Into<InTree>) -> Self {
171        assert!(
172            self.params
173                .insert(name.as_ref().into(), decl.into())
174                .is_none(),
175            "duplicate parameter override"
176        );
177        self
178    }
179    /// Replace the result expansion, including any previously assigned override.
180    ///
181    /// Configuration syntax uses the adapter's configuration operation sets.
182    /// Resolved trees can be embedded with [`crate::OutputNode::map_into`] before
183    /// registration, or passed directly to [`crate::compile_output`].
184    pub fn output(mut self, decl: impl Into<OutTree>) -> Self {
185        self.ret = Some(decl.into());
186        self
187    }
188}

190impl<InTree, OutTree, Extension> From<&str> for DeclFunction<InTree, OutTree, Extension> {
191    fn from(name: &str) -> Self {
192        Self::new(crate::ident(name).expect("source function identifier"))
193    }
194}
195impl<InTree, OutTree, Extension> From<String> for DeclFunction<InTree, OutTree, Extension> {
196    fn from(name: String) -> Self {
197        Self::from(name.as_str())
198    }
199}
