1//! The deconstructing half of the view: leaves read off a row that takes a
2//! value apart.
3//!
4//! The constructing half is in the parent module, and the two answer the same
5//! question in opposite directions — what a row costs on the foreign signature,
6//! in order, with names. What a row does not say, the target does, through an
7//! [`UnfoldPolicy`]: what a selector and a presence flag are, what one part is
8//! called, and how a nested part's name joins its parent's.
9//!
10//! Which parts are taken apart further is a **binding**, not a property of the
11//! row: a part whose type states a row still crosses whole unless something
12//! says to read it in pieces. That is #701's design decision 3, and it is what
13//! `Bindings` carries here.
14//!
15//! Only what a row can already state is here. `Reach::Path` and `Reach::Nested`
16//! are the two forms step 3 replaces with part bindings; until they are, a row
17//! using one is refused rather than read as something else.
18

27/// What a target decides when a row is taken apart into leaves.
28///
29/// A row states which parts there are and how each is reached. It does not
30/// state what any of them is *called* — an accessor's leaf name is the
31/// declaration's, not the function's — nor what a synthesized leaf carries.
32pub trait UnfoldPolicy {
33    /// The leaf saying which alternative of a sum is live.
34    ///
35    /// Its type is the sum itself rather than the integer it crosses as: what
36    /// the emitter needs is which sum it is choosing between.
37    fn selector(&self, source: &TypeRef) -> UnfoldLeaf;
38
39    /// The leaf saying whether an optional value the decomposition looks
40    /// through is present.
41    fn presence(&self, name: &str) -> UnfoldLeaf;
42
43    /// Whether the parts being named are a value form's.
44    ///
45    /// A value form's parts are named by the declaration that lists them, and a
46    /// product's own by whatever the target calls the field. The two are
47    /// different lists, and only the caller knows which is being walked.
48    fn value_form_part(&self, _index: usize) -> Option<String> {
49        None
50    }
51
52    /// What the part reached by `reach` is called.
53    ///
54    /// The declaration's answer, which is why the row cannot give it: two
55    /// bindings may read one accessor under two names. `field` is the model's
56    /// name for the field a [`Reach::Field`] reads, which a target that names
57    /// its slots after the struct's own fields needs and one that names them
58    /// after the declaration ignores.
59    fn part_name(&self, reach: &Reach, index: usize, field: Option<&syn::Ident>) -> String;
60
61    /// What one payload of one alternative is called.
62    ///
63    /// Not [`Self::part_name`] with the alternative prefixed: a target names an
64    /// arm's payload after what the alternative is called on ITS side, which is
65    /// not the Rust variant's name, and after how it addresses the field. Both
66    /// are the target's, and neither is in the row.
67    fn arm_part_name(&self, variant: &syn::Ident, member: &syn::Member, index: usize) -> String;
68
69    /// What the value itself is called when it is one of the parts.
70    fn identity_name(&self) -> String;
71
72    /// How a nested part's name joins the name of the part it was reached
73    /// through.
74    fn nest(&self, outer: &str, inner: &str) -> String;
75}

77/// Why a row could not be taken apart into leaves.
78#[derive(Debug)]
79pub enum UnfoldViewError {
80    /// A row named a function the model does not have.
81    UnknownAccessor(syn::Ident),
82    /// A row states a form the view does not read yet.
83    NotYetReadable {
84        /// The crossing whose row was walked.
85        crossing: String,
86        /// The form it used.
87        form: &'static str,
88    },
89    /// More than one part of one level is the value itself.
90    ManyIdentities {
91        /// The value being taken apart.
92        ty: String,
93    },
94    /// Two leaves of one decomposition carry the same name.
95    DuplicateName {
96        /// The value being taken apart.
97        ty: String,
98        /// The name written twice.
99        name: String,
100    },
101    /// A part's row reaches the value being taken apart.
102    Cycle {
103        /// The type the walk came back to.
104        ty: String,
105    },
106}
107
108impl std::fmt::Display for UnfoldViewError {
109    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
110        match self {
111            UnfoldViewError::UnknownAccessor(name) => {
112                write!(f, "no `#[prebindgen]` function named `{name}`")
113            }
114            UnfoldViewError::NotYetReadable { crossing, form } => write!(
115                f,
116                "`{crossing}`'s row uses {form}, which the leaf view does not read yet"
117            ),
118            UnfoldViewError::ManyIdentities { ty } => write!(
119                f,
120                "`{ty}` hands itself over more than once, and it has only one to give"
121            ),
122            UnfoldViewError::DuplicateName { ty, name } => {
123                write!(f, "`{ty}` gives two of its leaves the name `{name}`")
124            }
125            UnfoldViewError::Cycle { ty } => {
126                write!(
127                    f,
128                    "`{ty}` is taken apart into a part that is taken apart into it"
129                )
130            }
131        }
132    }
133}

135/// What one walk carries: the target's answers, the row name a nested part is
136/// looked up under, what has been read so far, and the chain of types being
137/// taken apart, which is what catches a value reached from itself.
138struct Walk<'w> {
139    policy: &'w dyn UnfoldPolicy,
140    bindings: &'w crate::recipe::Bindings,
141    row: &'w RecipeName,
142    leaves: Vec<UnfoldLeaf>,
143    hoists: Vec<Hoist>,
144    reading: Vec<String>,
145}
146
147/// Where one level of the walk stands: the path reached so far, the name
148/// prefix, and whether anything above was optional.
149#[derive(Clone)]
150struct At {
151    path: Vec<PathStep>,
152    name: Option<String>,
153    nullable: bool,
154    /// Whether these parts are a value form's, which names them itself.
155    value_form: bool,
156}
157
158impl Folding<'_> {
159    /// The leaves one value is taken apart into, and the value forms to bind
160    /// once on the way.
161    ///
162    /// `reading` is the value's own type. The row walked is the one named,
163    /// because taking a value apart is what a site opts into — the crossing's
164    /// default row is the value crossing whole.
165    pub fn unfold(
166        &self,
167        policy: &dyn UnfoldPolicy,
168        bindings: &crate::recipe::Bindings,
169        reading: &TypeRef,
170        row: &RecipeName,
171    ) -> Result<(Vec<UnfoldLeaf>, Vec<Hoist>), UnfoldViewError> {
172        let shape = self
173            .deconstructing(reading, row)
174            .ok_or_else(|| UnfoldViewError::NotYetReadable {
175                crossing: reading.to_string(),
176                form: "no row that takes the value apart",
177            })?
178            .clone();
179        let mut walk = Walk {
180            policy,
181            bindings,
182            row,
183            leaves: Vec::new(),
184            hoists: Vec::new(),
185            reading: vec![reading.key().to_string()],
186        };
187        let at = At {
188            path: Vec::new(),
189            name: None,
190            nullable: false,
191            value_form: false,
192        };
193        self.level(&mut walk, reading, &shape, &at)?;
194        unique_names(&walk.leaves, reading)?;
195        Ok((walk.leaves, walk.hoists))
196    }
197
198    /// One row, at one place in the value.
199    fn level(
200        &self,
201        walk: &mut Walk<'_>,
202        source: &TypeRef,
203        shape: &Shape<Deconstruct>,
204        at: &At,
205    ) -> Result<(), UnfoldViewError> {
206        match shape {
207            Shape::Product(Deconstruct::Fields(reaches)) => self.parts(walk, source, reaches, at),
208            Shape::Product(Deconstruct::ValueForm { func, parts }) => {
209                // The form is called once and every part hangs off that one
210                // call, which is what a hoist records.
211                let mut root = at.path.clone();
212                root.push(PathStep::call(func.clone(), false, true));
213                walk.hoists.push(Hoist {
214                    prefix: root.clone(),
215                    consuming: self.consumes(func)?,
216                });
217                let inner = At {
218                    path: root,
219                    name: at.name.clone(),
220                    nullable: at.nullable,
221                    value_form: true,
222                };
223                // The parts are read off what the call RETURNS, not off the
224                // value it was called on: `Reach::Field(0)` here is the first
225                // field of the bound struct.
226                let bound = self.returns(func)?;
227                self.parts(walk, &bound, parts, &inner)
228            }
229            Shape::Choice { arms } => {
230                // The selector first, then one group per arm. Every arm's
231                // leaves are written, and the selector says which of them is
232                // live — an arm the value did not take fills its slots with the
233                // wire's default rather than being absent.
234                let mut tag = walk.policy.selector(source);
235                // A sum reached AS A PART carries that part's path and name:
236                // its leaves are read out of the field the walk arrived
237                // through, not off the value the row belongs to.
238                tag.path = at.path.clone();
239                tag.nullable = at.nullable;
240                if let Some(outer) = &at.name {
241                    tag.name = walk.policy.nest(outer, &tag.name);
242                }
243                walk.leaves.push(tag);
244                for arm in arms {
245                    let Some(alternative) = arm.alternative else {
246                        return Err(UnfoldViewError::NotYetReadable {
247                            crossing: source.to_string(),
248                            form: "an arm naming no alternative, which only a parameter builds",
249                        });
250                    };
251                    let Deconstruct::Fields(reaches) = &arm.op else {
252                        return Err(UnfoldViewError::NotYetReadable {
253                            crossing: source.to_string(),
254                            form: "an arm read through a value form",
255                        });
256                    };
257                    let Some((variant, fields)) = self.alternative(source, alternative) else {
258                        return Err(UnfoldViewError::NotYetReadable {
259                            crossing: source.to_string(),
260                            form: "an alternative the model does not have",
261                        });
262                    };
263                    for (index, reach) in reaches.iter().enumerate() {
264                        let Reach::Field(field) = reach else {
265                            return Err(UnfoldViewError::NotYetReadable {
266                                crossing: source.to_string(),
267                                form: reach_form(reach),
268                            });
269                        };
270                        let Some(payload) = fields.get(*field) else {
271                            return Err(UnfoldViewError::NotYetReadable {
272                                crossing: source.to_string(),
273                                form: "a payload field the alternative does not have",
274                            });
275                        };
276                        // Bound by a pattern in the arm rather than reached off
277                        // the value, so the leaf carries no path.
278                        // Named under the ALTERNATIVE: every arm writes its
279                        // own slots, so two arms reading their first payload
280                        // field would otherwise be two leaves of one name. What
281                        // that name is, is the target's — see
282                        // [`UnfoldPolicy::arm_part_name`].
283                        let member = member_of(payload, *field);
284                        let name = walk.policy.arm_part_name(variant, &member, index);
285                        walk.leaves.push(UnfoldLeaf {
286                            name: match &at.name {
287                                Some(outer) => walk.policy.nest(outer, &name),
288                                None => name,
289                            },
290                            path: at.path.clone(),
291                            out_ty: payload.ty.clone(),
292                            identity: false,
293                            nullable: at.nullable,
294                            source: LeafSource::VariantField {
295                                variant: variant.clone(),
296                                member,
297                            },
298                            groups: vec![alternative as i32],
299                        });
300                    }
301                }
302                Ok(())
303            }
304            Shape::Atomic => Ok(()),
305            other => Err(UnfoldViewError::NotYetReadable {
306                crossing: source.to_string(),
307                form: name_of(other),
308            }),
309        }
310    }

312    /// Every part of one product.
313    fn parts(
314        &self,
315        walk: &mut Walk<'_>,
316        source: &TypeRef,
317        reaches: &[Reach],
318        at: &At,
319    ) -> Result<(), UnfoldViewError> {
320        // One value to give, so one part may be it. It keeps the position the
321        // row puts it in: what goes last is its EMISSION, after every borrow
322        // taken off the value has ended, and that is the emitter's ordering
323        // rather than the leaf list's.
324        let mut seen_identity = false;
325        for (index, reach) in reaches.iter().enumerate() {
326            if !matches!(reach, Reach::Identity) {
327                self.part(walk, source, reach, index, at)?;
328                continue;
329            }
330            if seen_identity {
331                return Err(UnfoldViewError::ManyIdentities {
332                    ty: source.key().to_string(),
333                });
334            }
335            seen_identity = true;
336            let name = match &at.name {
337                Some(outer) => outer.clone(),
338                None => walk.policy.identity_name(),
339            };
340            walk.leaves.push(UnfoldLeaf {
341                name,
342                path: at.path.clone(),
343                // The value as it stands: owned where it is ours to give,
344                // borrowed where it is lent and the leaf clones through its own
345                // converter.
346                out_ty: if at.path.is_empty() {
347                    source.clone()
348                } else {
349                    source.borrowed()
350                },
351                identity: true,
352                nullable: at.nullable,
353                source: LeafSource::Reach,
354                groups: Vec::new(),
355            });
356        }
357        Ok(())
358    }

360    /// One part, spliced through its own row when it has one.
361    fn part(
362        &self,
363        walk: &mut Walk<'_>,
364        source: &TypeRef,
365        reach: &Reach,
366        index: usize,
367        at: &At,
368    ) -> Result<(), UnfoldViewError> {
369        let name = match at
370            .value_form
371            .then(|| walk.policy.value_form_part(index))
372            .flatten()
373        {
374            Some(name) => name,
375            // A value form that names no part at this position falls back to
376            // the ordinary answer, the same as a product's own part.
377            None => walk
378                .policy
379                .part_name(reach, index, self.field_name(source, reach)),
380        };
381        let full = match &at.name {
382            Some(outer) => walk.policy.nest(outer, &name),
383            None => name,
384        };
385        let (step, ty) = match reach {
386            Reach::Omit => return Ok(()),
387            Reach::Accessor(func) => {
388                let ret = self.returns(func)?;
389                let optional = ret.optional_inner().is_some();
390                let core = ret.optional_inner().unwrap_or(&ret);
391                let owned = core.borrow_target().is_none();
392                (
393                    PathStep::call(func.clone(), optional, owned),
394                    core.borrow_target().unwrap_or(core).clone(),
395                )
396            }
397            Reach::Field(field) => {
398                let ty = self.field_ty(source, *field).ok_or_else(|| {
399                    UnfoldViewError::NotYetReadable {
400                        crossing: source.to_string(),
401                        form: "a field the model does not have",
402                    }
403                })?;
404                let ident = self.field_ident(source, *field);
405                (PathStep::field(ident, false), ty)
406            }
407            Reach::Identity => unreachable!("handled by the caller"),
408            other => {
409                return Err(UnfoldViewError::NotYetReadable {
410                    crossing: source.to_string(),
411                    form: reach_form(other),
412                })
413            }
414        };
415        let optional = step.is_optional();
416        let mut path = at.path.clone();
417        path.push(step);
418
419        // A part is taken apart further only where a binding says so. Having a
420        // row of its own is not enough — a `data_class` states one and still
421        // crosses whole as an accessor's return, which is the difference a
422        // binding carries and a row cannot (#701 decision 3).
423        if let Some(child) = self.bound_row(walk, source, index, &ty).cloned() {
424            let key = ty.key().to_string();
425            if walk.reading.contains(&key) {
426                return Err(UnfoldViewError::Cycle { ty: key });
427            }
428            walk.reading.push(key);
429            let inner = At {
430                path,
431                name: Some(full),
432                nullable: at.nullable || optional,
433                value_form: false,
434            };
435            self.level(walk, &ty, &child, &inner)?;
436            walk.reading.pop();
437            return Ok(());
438        }
439
440        walk.leaves.push(UnfoldLeaf {
441            name: full,
442            path,
443            out_ty: self.leaf_ty(reach, source, index),
444            identity: false,
445            nullable: at.nullable,
446            source: LeafSource::Reach,
447            groups: Vec::new(),
448        });
449        Ok(())
450    }

452    /// The model's name for the field a reach reads, when it reads one.
453    fn field_name(&self, source: &TypeRef, reach: &Reach) -> Option<&syn::Ident> {
454        let Reach::Field(index) = reach else {
455            return None;
456        };
457        self.fields_of(source).get(*index)?.name.as_ref()
458    }

460    /// The row a part is taken apart by, if a binding names one.
461    ///
462    /// The site is the part's own position in the row being walked, which is
463    /// the key `Compiler::part_of` builds for the same part — so an adapter
464    /// writes one binding and both the compiler and this view find it.
465    fn bound_row(
466        &self,
467        walk: &Walk<'_>,
468        source: &TypeRef,
469        index: usize,
470        part: &TypeRef,
471    ) -> Option<&Shape<Deconstruct>> {
472        use crate::recipe::{Crossing, Direction, Site};
473
474        let owner = Crossing::new(source.clone(), Direction::Deconstruct);
475        let site = Site::arm_part(&owner.row(walk.row.clone()), None, index);
476        let crossing = Crossing::new(part.clone(), Direction::Deconstruct);
477        let bound = walk.bindings.resolve(&site, &crossing, self.recipes())?;
478        match self.recipes().get(&bound.recipe)? {
479            Recipe::Deconstructing(shape) => Some(shape),
480            Recipe::Constructing(_) => None,
481        }
482    }

484    /// One alternative of a declared sum: its name, and its payload fields.
485    fn alternative(
486        &self,
487        source: &TypeRef,
488        index: usize,
489    ) -> Option<(&syn::Ident, &[prebindgen_flat::flat::Field])> {
490        let value = source.borrow_target().unwrap_or(source).unwrapped();
491        let prebindgen_flat::flat::TypeKind::Named { id, .. } = value.kind() else {
492            return None;
493        };
494        let prebindgen_flat::flat::Type::Variant(v) = self.model().resolve(id)? else {
495            return None;
496        };
497        let alternative = v.alternatives.get(index)?;
498        Some((&alternative.name, alternative.fields.as_slice()))
499    }

501    /// This crossing's deconstructing row under `row`, if it states one.
502    fn deconstructing(&self, ty: &TypeRef, row: &RecipeName) -> Option<&Shape<Deconstruct>> {
503        let crossing =
504            crate::recipe::Crossing::new(ty.clone(), crate::recipe::Direction::Deconstruct).key();
505        match self.recipes().get(self.recipes().key_of(&crossing, row)?)? {
506            Recipe::Deconstructing(shape) => Some(shape),
507            Recipe::Constructing(_) => None,
508        }
509    }

511    /// What a part's own leaf carries, when it is not spliced.
512    fn leaf_ty(&self, reach: &Reach, source: &TypeRef, index: usize) -> TypeRef {
513        match reach {
514            Reach::Accessor(func) => self.returns(func).unwrap_or_else(|_| source.clone()),
515            Reach::Field(field) => self
516                .field_ty(source, *field)
517                .unwrap_or_else(|| source.clone()),
518            _ => {
519                let _ = index;
520                source.clone()
521            }
522        }
523    }
524
525    fn returns(&self, func: &syn::Ident) -> Result<TypeRef, UnfoldViewError> {
526        self.model()
527            .function(func)
528            .map(|f| f.ret.clone())
529            .ok_or_else(|| UnfoldViewError::UnknownAccessor(func.clone()))
530    }
531
532    fn consumes(&self, func: &syn::Ident) -> Result<bool, UnfoldViewError> {
533        let f = self
534            .model()
535            .function(func)
536            .ok_or_else(|| UnfoldViewError::UnknownAccessor(func.clone()))?;
537        Ok(f.params
538            .first()
539            .is_some_and(|p| p.ty.borrow_target().is_none()))
540    }
541
542    fn field_ty(&self, source: &TypeRef, index: usize) -> Option<TypeRef> {
543        self.fields_of(source).get(index).map(|f| f.ty.clone())
544    }
545
546    /// How a field is addressed: its name, or its position for a tuple struct.
547    fn field_ident(&self, source: &TypeRef, index: usize) -> syn::Ident {
548        self.fields_of(source)
549            .get(index)
550            .and_then(|f| f.name.clone())
551            .unwrap_or_else(|| {
552                syn::Ident::new(&format!("_{index}"), proc_macro2::Span::call_site())
553            })
554    }
555}

557/// Two leaves of one decomposition may not share a name: the foreign signature
558/// names its arguments, and two of one name is one the emitter cannot write.
559fn unique_names(leaves: &[UnfoldLeaf], source: &TypeRef) -> Result<(), UnfoldViewError> {
560    let mut seen = std::collections::HashSet::new();
561    for leaf in leaves {
562        if !seen.insert(leaf.name.clone()) {
563            return Err(UnfoldViewError::DuplicateName {
564                ty: source.key().to_string(),
565                name: leaf.name.clone(),
566            });
567        }
568    }
569    Ok(())
570}

572fn name_of(shape: &Shape<Deconstruct>) -> &'static str {
573    match shape {
574        Shape::Atomic => "atomic",
575        Shape::Optional => "an optional layer",
576        Shape::Sequence => "a sequence",
577        Shape::Invoke => "a callable",
578        Shape::Product(_) => "a product",
579        Shape::Choice { .. } => "a choice",
580    }
581}

583/// How a payload field is addressed in its arm's pattern: by name, or by
584/// position for a tuple variant.
585fn member_of(field: &prebindgen_flat::flat::Field, index: usize) -> syn::Member {
586    match &field.name {
587        Some(name) => syn::Member::Named(name.clone()),
588        None => syn::Member::Unnamed(syn::Index::from(index)),
589    }
590}
591
592fn reach_form(reach: &Reach) -> &'static str {
593    match reach {
594        Reach::Path(_) => "a field-of-a-field chain",
595        Reach::Nested { .. } => "a field taken apart in place",
596        _ => "a reach the view does not read",
597    }
598}
