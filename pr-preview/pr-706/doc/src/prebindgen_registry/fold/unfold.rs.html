1//! The deconstructing half of the view: leaves read off a row that takes a
2//! value apart.
3//!
4//! The constructing half is in the parent module, and the two answer the same
5//! question in opposite directions — what a row costs on the foreign signature,
6//! in order, with names. What a row does not say, the target does, through an
7//! [`UnfoldPolicy`]: what a selector and a presence flag are, what one part is
8//! called, and how a nested part's name joins its parent's.
9//!
10//! Which parts are taken apart further is a **binding**, not a property of the
11//! row: a part whose type states a row still crosses whole unless something
12//! says to read it in pieces. That is #701's design decision 3, and it is what
13//! `Bindings` carries here.
14//!
15//! Only what a row can already state is here. `Reach::Path` and `Reach::Nested`
16//! are the two forms step 3 replaces with part bindings; until they are, a row
17//! using one is refused rather than read as something else.
18

27/// What a target decides when a row is taken apart into leaves.
28///
29/// A row states which parts there are and how each is reached. It does not
30/// state what any of them is *called* — an accessor's leaf name is the
31/// declaration's, not the function's — nor what a synthesized leaf carries.
32pub trait UnfoldPolicy {
33    /// The leaf saying which alternative of a sum is live.
34    ///
35    /// Its type is the sum itself rather than the integer it crosses as: what
36    /// the emitter needs is which sum it is choosing between.
37    fn selector(&self, source: &TypeRef) -> UnfoldLeaf;
38
39    /// The leaf saying whether an optional value the decomposition looks
40    /// through is present.
41    fn presence(&self, name: &str) -> UnfoldLeaf;
42
43    /// Whether the parts being named are a value form's.
44    ///
45    /// A value form's parts are named by the declaration that lists them, and a
46    /// product's own by whatever the target calls the field. The two are
47    /// different lists, and only the caller knows which is being walked.
48    fn value_form_part(&self, _index: usize) -> Option<String> {
49        None
50    }
51
52    /// What the part reached by `reach` is called.
53    ///
54    /// The declaration's answer, which is why the row cannot give it: two
55    /// bindings may read one accessor under two names. `field` is the model's
56    /// name for the field a [`Reach::Field`] reads, which a target that names
57    /// its slots after the struct's own fields needs and one that names them
58    /// after the declaration ignores.
59    fn part_name(&self, reach: &Reach, index: usize, field: Option<&syn::Ident>) -> String;
60
61    /// What one payload of one alternative is called.
62    ///
63    /// Not [`Self::part_name`] with the alternative prefixed: a target names an
64    /// arm's payload after what the alternative is called on ITS side, which is
65    /// not the Rust variant's name, and after how it addresses the field. Both
66    /// are the target's, and neither is in the row.
67    fn arm_part_name(&self, variant: &syn::Ident, member: &syn::Member, index: usize) -> String;
68
69    /// What the value itself is called when it is one of the parts.
70    fn identity_name(&self) -> String;
71
72    /// How a nested part's name joins the name of the part it was reached
73    /// through.
74    fn nest(&self, outer: &str, inner: &str) -> String;
75}

77/// Why a row could not be taken apart into leaves.
78#[derive(Debug)]
79pub enum UnfoldViewError {
80    /// A row named a function the model does not have.
81    UnknownAccessor(syn::Ident),
82    /// A row states a form the view does not read yet.
83    ///
84    /// Not a defect in the row: these are the shapes #701's step 3 is still
85    /// replacing, and a caller running the view beside the older path skips
86    /// them. Every other variant here IS a defect, which is why they are
87    /// separate — [`Self::is_not_yet_readable`] is what a caller may skip on,
88    /// and skipping on any error would let a bad row disable the very check
89    /// that found it.
90    NotYetReadable {
91        /// The crossing whose row was walked.
92        crossing: String,
93        /// The form it used.
94        form: &'static str,
95    },
96    /// More than one part of one level is the value itself.
97    ManyIdentities {
98        /// The value being taken apart.
99        ty: String,
100    },
101    /// Two leaves of one decomposition carry the same name.
102    DuplicateName {
103        /// The value being taken apart.
104        ty: String,
105        /// The name written twice.
106        name: String,
107    },
108    /// A part's row reaches the value being taken apart.
109    Cycle {
110        /// The type the walk came back to.
111        ty: String,
112    },
113}
114
115impl UnfoldViewError {
116    /// Whether this is a shape the view does not represent yet, rather than a
117    /// row that is wrong.
118    pub fn is_not_yet_readable(&self) -> bool {
119        matches!(self, UnfoldViewError::NotYetReadable { .. })
120    }
121}
122
123impl std::fmt::Display for UnfoldViewError {
124    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
125        match self {
126            UnfoldViewError::UnknownAccessor(name) => {
127                write!(f, "no `#[prebindgen]` function named `{name}`")
128            }
129            UnfoldViewError::NotYetReadable { crossing, form } => write!(
130                f,
131                "`{crossing}`'s row uses {form}, which the leaf view does not read yet"
132            ),
133            UnfoldViewError::ManyIdentities { ty } => write!(
134                f,
135                "`{ty}` hands itself over more than once, and it has only one to give"
136            ),
137            UnfoldViewError::DuplicateName { ty, name } => {
138                write!(f, "`{ty}` gives two of its leaves the name `{name}`")
139            }
140            UnfoldViewError::Cycle { ty } => {
141                write!(
142                    f,
143                    "`{ty}` is taken apart into a part that is taken apart into it"
144                )
145            }
146        }
147    }
148}

150/// What one walk of a row did NOT read.
151///
152/// A caller running this beside an older decomposition needs to know where the
153/// two are not comparable, and needs to know it from the walk rather than by
154/// measuring the answers against each other. Measuring cannot tell a shape the
155/// rows do not state yet from a defect that made the read smaller — which is
156/// the same mistake in the small that a silent fallback is in the large.
157///
158/// Empty means the walk read everything the row states. Anything else names a
159/// place it stopped, and the leaves are that much short of the whole value.
160#[derive(Clone, Debug, Default)]
161pub struct Coverage {
162    unread: Vec<&'static str>,
163}
164
165impl Coverage {
166    /// Whether the walk read the whole of what the row states.
167    pub fn is_complete(&self) -> bool {
168        self.unread.is_empty()
169    }
170
171    /// The places it stopped, for a caller that reports them.
172    pub fn unread(&self) -> &[&'static str] {
173        &self.unread
174    }
175
176    fn note(&mut self, what: &'static str) {
177        if !self.unread.contains(&what) {
178            self.unread.push(what);
179        }
180    }
181}

183/// What one walk carries: the target's answers, the row name a nested part is
184/// looked up under, what has been read so far, and the chain of types being
185/// taken apart, which is what catches a value reached from itself.
186struct Walk<'w> {
187    policy: &'w dyn UnfoldPolicy,
188    bindings: &'w crate::recipe::Bindings,
189    leaves: Vec<UnfoldLeaf>,
190    hoists: Vec<Hoist>,
191    reading: Vec<String>,
192    coverage: Coverage,
193}
194
195/// Where one level of the walk stands: the path reached so far, the name
196/// prefix, and whether anything above was optional.
197#[derive(Clone)]
198struct At {
199    path: Vec<PathStep>,
200    name: Option<String>,
201    nullable: bool,
202    /// Whether these parts are a value form's, which names them itself.
203    value_form: bool,
204    /// The row being read at this level, as a key.
205    ///
206    /// The KEY and not a name: a part's binding is written against the row its
207    /// product belongs to, and after a binding has taken a part to some other
208    /// row, that row is what its own parts are keyed by. Carrying one root name
209    /// down the walk would look for a grandchild's binding under the root's
210    /// name and miss it — while `Compiler::part_of` keys the same part by the
211    /// row it is actually at.
212    row: crate::recipe::RecipeKey,
213}
214
215impl Folding<'_> {
216    /// The leaves one value is taken apart into, and the value forms to bind
217    /// once on the way.
218    ///
219    /// `reading` is the value's own type. The row walked is the one named,
220    /// because taking a value apart is what a site opts into — the crossing's
221    /// default row is the value crossing whole.
222    pub fn unfold(
223        &self,
224        policy: &dyn UnfoldPolicy,
225        bindings: &crate::recipe::Bindings,
226        reading: &TypeRef,
227        row: &RecipeName,
228    ) -> Result<(Vec<UnfoldLeaf>, Vec<Hoist>, Coverage), UnfoldViewError> {
229        let shape = self
230            .deconstructing(reading, row)
231            .ok_or_else(|| UnfoldViewError::NotYetReadable {
232                crossing: reading.to_string(),
233                form: "no row that takes the value apart",
234            })?
235            .clone();
236        let mut walk = Walk {
237            policy,
238            bindings,
239            leaves: Vec::new(),
240            hoists: Vec::new(),
241            reading: vec![reading.key().to_string()],
242            coverage: Coverage::default(),
243        };
244        let at = At {
245            path: Vec::new(),
246            name: None,
247            nullable: false,
248            value_form: false,
249            row: crate::recipe::Crossing::new(
250                reading.clone(),
251                crate::recipe::Direction::Deconstruct,
252            )
253            .row(row.clone()),
254        };
255        self.level(&mut walk, reading, &shape, &at)?;
256        unique_names(&walk.leaves, reading)?;
257        Ok((walk.leaves, walk.hoists, walk.coverage))
258    }

260    /// One row, at one place in the value.
261    fn level(
262        &self,
263        walk: &mut Walk<'_>,
264        source: &TypeRef,
265        shape: &Shape<Deconstruct>,
266        at: &At,
267    ) -> Result<(), UnfoldViewError> {
268        match shape {
269            Shape::Product(Deconstruct::Fields(reaches)) => self.parts(walk, source, reaches, at),
270            Shape::Product(Deconstruct::ValueForm { func, parts }) => {
271                // The form is called once and every part hangs off that one
272                // call, which is what a hoist records.
273                let mut root = at.path.clone();
274                root.push(PathStep::call(func.clone(), false, true));
275                let consuming = self.consumes(func)?;
276                if consuming {
277                    // A part of a form that was handed the value may be moved
278                    // out rather than read, and which parts those are is the
279                    // target's answer about their types. A row states the same
280                    // reach either way, so the walk cannot say.
281                    walk.coverage
282                        .note("a consuming value form, whose parts may be handed over");
283                }
284                walk.hoists.push(Hoist {
285                    prefix: root.clone(),
286                    consuming,
287                });
288                let inner = At {
289                    path: root,
290                    name: at.name.clone(),
291                    nullable: at.nullable,
292                    value_form: true,
293                    // A value form's parts belong to the row that states the
294                    // form, not to the struct the call returns.
295                    row: at.row.clone(),
296                };
297                // The parts are read off what the call RETURNS, not off the
298                // value it was called on: `Reach::Field(0)` here is the first
299                // field of the bound struct.
300                let bound = self.returns(func)?;
301                self.parts(walk, &bound, parts, &inner)
302            }
303            Shape::Choice { arms } => {
304                // The selector first, then one group per arm. Every arm's
305                // leaves are written, and the selector says which of them is
306                // live — an arm the value did not take fills its slots with the
307                // wire's default rather than being absent.
308                let mut tag = walk.policy.selector(source);
309                // A sum reached AS A PART carries that part's path and name:
310                // its leaves are read out of the field the walk arrived
311                // through, not off the value the row belongs to.
312                tag.path = at.path.clone();
313                tag.nullable = at.nullable;
314                if let Some(outer) = &at.name {
315                    tag.name = walk.policy.nest(outer, &tag.name);
316                }
317                walk.leaves.push(tag);
318                for arm in arms {
319                    let Some(alternative) = arm.alternative else {
320                        return Err(UnfoldViewError::NotYetReadable {
321                            crossing: source.to_string(),
322                            form: "an arm naming no alternative, which only a parameter builds",
323                        });
324                    };
325                    let Deconstruct::Fields(reaches) = &arm.op else {
326                        return Err(UnfoldViewError::NotYetReadable {
327                            crossing: source.to_string(),
328                            form: "an arm read through a value form",
329                        });
330                    };
331                    let Some((variant, fields)) = self.alternative(source, alternative) else {
332                        return Err(UnfoldViewError::NotYetReadable {
333                            crossing: source.to_string(),
334                            form: "an alternative the model does not have",
335                        });
336                    };
337                    for (index, reach) in reaches.iter().enumerate() {
338                        let Reach::Field(field) = reach else {
339                            return Err(UnfoldViewError::NotYetReadable {
340                                crossing: source.to_string(),
341                                form: reach_form(reach),
342                            });
343                        };
344                        let Some(payload) = fields.get(*field) else {
345                            return Err(UnfoldViewError::NotYetReadable {
346                                crossing: source.to_string(),
347                                form: "a payload field the alternative does not have",
348                            });
349                        };
350                        // Bound by a pattern in the arm rather than reached off
351                        // the value, so the leaf carries no path.
352                        // Named under the ALTERNATIVE: every arm writes its
353                        // own slots, so two arms reading their first payload
354                        // field would otherwise be two leaves of one name. What
355                        // that name is, is the target's — see
356                        // [`UnfoldPolicy::arm_part_name`].
357                        let member = member_of(payload, *field);
358                        let name = walk.policy.arm_part_name(variant, &member, index);
359                        let full = match &at.name {
360                            Some(outer) => walk.policy.nest(outer, &name),
361                            None => name,
362                        };
363                        // A payload may itself be taken apart, and a binding on
364                        // one part of one ARM is what says so — the case
365                        // `ArmKey` exists to tell from a part of the product
366                        // itself. `Compiler::part_of` follows the same key.
367                        let arm_key = ArmKey::Alternative(alternative);
368                        if let Some((row, child)) =
369                            self.bound_row(walk, at, Some(arm_key), index, &payload.ty)
370                        {
371                            let child = child.clone();
372                            let seen = payload.ty.key().to_string();
373                            if walk.reading.contains(&seen) {
374                                return Err(UnfoldViewError::Cycle { ty: seen });
375                            }
376                            walk.reading.push(seen);
377                            let inner = At {
378                                // Bound by a pattern in the arm, so the payload
379                                // is reached from the binding rather than from
380                                // the value: its own parts hang off no path.
381                                path: Vec::new(),
382                                name: Some(full),
383                                nullable: at.nullable,
384                                value_form: false,
385                                row,
386                            };
387                            let before = walk.leaves.len();
388                            self.level(walk, &payload.ty, &child, &inner)?;
389                            walk.reading.pop();
390                            // Everything the payload contributed is live only
391                            // in this arm, exactly as the payload itself would
392                            // have been.
393                            for leaf in &mut walk.leaves[before..] {
394                                leaf.groups.insert(0, alternative as i32);
395                            }
396                            continue;
397                        }
398                        walk.leaves.push(UnfoldLeaf {
399                            name: full,
400                            path: at.path.clone(),
401                            out_ty: payload.ty.clone(),
402                            identity: false,
403                            nullable: at.nullable,
404                            source: LeafSource::VariantField {
405                                variant: variant.clone(),
406                                member,
407                            },
408                            groups: vec![alternative as i32],
409                        });
410                    }
411                }
412                Ok(())
413            }
414            Shape::Atomic => {
415                // A placeholder a crossing carries so a site can select it: it
416                // says the adapter emits this conversion itself, and says
417                // nothing about parts.
418                walk.coverage.note("a row that states no parts");
419                Ok(())
420            }
421            other => Err(UnfoldViewError::NotYetReadable {
422                crossing: source.to_string(),
423                form: name_of(other),
424            }),
425        }
426    }

428    /// Every part of one product.
429    fn parts(
430        &self,
431        walk: &mut Walk<'_>,
432        source: &TypeRef,
433        reaches: &[Reach],
434        at: &At,
435    ) -> Result<(), UnfoldViewError> {
436        // One value to give, so one part may be it. It keeps the position the
437        // row puts it in: what goes last is its EMISSION, after every borrow
438        // taken off the value has ended, and that is the emitter's ordering
439        // rather than the leaf list's.
440        let mut seen_identity = false;
441        for (index, reach) in reaches.iter().enumerate() {
442            if !matches!(reach, Reach::Identity) {
443                self.part(walk, source, reach, index, at)?;
444                continue;
445            }
446            if seen_identity {
447                return Err(UnfoldViewError::ManyIdentities {
448                    ty: source.key().to_string(),
449                });
450            }
451            seen_identity = true;
452            let name = match &at.name {
453                Some(outer) => outer.clone(),
454                None => walk.policy.identity_name(),
455            };
456            walk.leaves.push(UnfoldLeaf {
457                name,
458                path: at.path.clone(),
459                // The value as it stands: owned where it is ours to give,
460                // borrowed where it is lent and the leaf clones through its own
461                // converter.
462                out_ty: if at.path.is_empty() {
463                    source.clone()
464                } else {
465                    source.borrowed()
466                },
467                identity: true,
468                nullable: at.nullable,
469                source: LeafSource::Reach,
470                groups: Vec::new(),
471            });
472        }
473        Ok(())
474    }

476    /// One part, spliced through its own row when it has one.
477    fn part(
478        &self,
479        walk: &mut Walk<'_>,
480        source: &TypeRef,
481        reach: &Reach,
482        index: usize,
483        at: &At,
484    ) -> Result<(), UnfoldViewError> {
485        let name = match at
486            .value_form
487            .then(|| walk.policy.value_form_part(index))
488            .flatten()
489        {
490            Some(name) => name,
491            // A value form that names no part at this position falls back to
492            // the ordinary answer, the same as a product's own part.
493            None => walk
494                .policy
495                .part_name(reach, index, self.field_name(source, reach)),
496        };
497        let full = match &at.name {
498            Some(outer) => walk.policy.nest(outer, &name),
499            None => name,
500        };
501        let (step, ty) = match reach {
502            Reach::Omit => return Ok(()),
503            Reach::Accessor(func) => {
504                let ret = self.returns(func)?;
505                let optional = ret.optional_inner().is_some();
506                let core = ret.optional_inner().unwrap_or(&ret);
507                let owned = core.borrow_target().is_none();
508                (
509                    PathStep::call(func.clone(), optional, owned),
510                    core.borrow_target().unwrap_or(core).clone(),
511                )
512            }
513            Reach::Field(field) => {
514                let ty = self.field_ty(source, *field).ok_or_else(|| {
515                    UnfoldViewError::NotYetReadable {
516                        crossing: source.to_string(),
517                        form: "a field the model does not have",
518                    }
519                })?;
520                let ident = self.field_ident(source, *field);
521                // A field's own `Option` rides the leaf's own converter: a part
522                // that crosses whole takes the `Option` with it. Looking
523                // THROUGH one — the nesting step a decomposition makes of an
524                // optional nested class — is the `Optional` part #701's
525                // decision 3 describes, and waits for it.
526                (PathStep::field(ident, false), ty)
527            }
528            Reach::Identity => unreachable!("handled by the caller"),
529            other => {
530                return Err(UnfoldViewError::NotYetReadable {
531                    crossing: source.to_string(),
532                    form: reach_form(other),
533                })
534            }
535        };
536        let optional = step.is_optional();
537        let mut path = at.path.clone();
538        path.push(step);
539
540        // A part is taken apart further only where a binding says so. Having a
541        // row of its own is not enough — a `data_class` states one and still
542        // crosses whole as an accessor's return, which is the difference a
543        // binding carries and a row cannot (#701 decision 3).
544        if let Some((row, child)) = self.bound_row(walk, at, None, index, &ty) {
545            let child = child.clone();
546            let seen = ty.key().to_string();
547            if walk.reading.contains(&seen) {
548                return Err(UnfoldViewError::Cycle { ty: seen });
549            }
550            walk.reading.push(seen);
551            let inner = At {
552                path,
553                name: Some(full),
554                nullable: at.nullable || optional,
555                value_form: false,
556                // The row the binding took this part to, so ITS parts are keyed
557                // by it rather than by the row above.
558                row,
559            };
560            self.level(walk, &ty, &child, &inner)?;
561            walk.reading.pop();
562            return Ok(());
563        }
564
565        // Nothing bound this part, so it crosses whole. If its value states
566        // parts of its own, there was something here the walk did not look
567        // through — the binding for it is step 3's remaining work, and saying
568        // so is what keeps a caller from having to guess by measuring.
569        let value = ty.optional_inner().unwrap_or(&ty);
570        let value = value.borrow_target().unwrap_or(value);
571        if self.deconstructing_any(value) {
572            walk.coverage
573                .note("a part whose value states parts of its own, and no binding takes it apart");
574        }
575        walk.leaves.push(UnfoldLeaf {
576            name: full,
577            path,
578            out_ty: self.leaf_ty(reach, source, index),
579            identity: false,
580            nullable: at.nullable,
581            source: LeafSource::Reach,
582            groups: Vec::new(),
583        });
584        Ok(())
585    }

587    /// The model's name for the field a reach reads, when it reads one.
588    fn field_name(&self, source: &TypeRef, reach: &Reach) -> Option<&syn::Ident> {
589        let Reach::Field(index) = reach else {
590            return None;
591        };
592        self.fields_of(source).get(*index)?.name.as_ref()
593    }

595    /// The row a part is taken apart by, if a binding names one.
596    ///
597    /// The site is the part's own position in the row being walked, which is
598    /// the key `Compiler::part_of` builds for the same part — so an adapter
599    /// writes one binding and both the compiler and this view find it.
600    fn bound_row(
601        &self,
602        walk: &Walk<'_>,
603        at: &At,
604        arm: Option<crate::recipe::ArmKey>,
605        index: usize,
606        part: &TypeRef,
607    ) -> Option<(crate::recipe::RecipeKey, &Shape<Deconstruct>)> {
608        use crate::recipe::{Crossing, Direction, Site};
609
610        // Keyed by the row this level is AT, which is what `Compiler::part_of`
611        // keys the same part by — so an adapter writes one binding and both
612        // find it.
613        let site = Site::arm_part(&at.row, arm, index);
614        // DECLARED, not merely resolved. `Bindings::resolve` answers for every
615        // site, falling back to the crossing's default row — which is the whole
616        // value crossing, and following it here would take a part apart because
617        // its type has a row rather than because something said to. That is the
618        // opposite of the rule: only a binding takes a part apart.
619        if !walk.bindings.is_declared(&site) {
620            return None;
621        }
622        let crossing = Crossing::new(part.clone(), Direction::Deconstruct);
623        let bound = walk.bindings.resolve(&site, &crossing, self.recipes())?;
624        match self.recipes().get(&bound.recipe)? {
625            Recipe::Deconstructing(shape) => Some((bound.recipe, shape)),
626            Recipe::Constructing(_) => None,
627        }
628    }

630    /// One alternative of a declared sum: its name, and its payload fields.
631    fn alternative(
632        &self,
633        source: &TypeRef,
634        index: usize,
635    ) -> Option<(&syn::Ident, &[prebindgen_flat::flat::Field])> {
636        let value = source.borrow_target().unwrap_or(source).unwrapped();
637        let prebindgen_flat::flat::TypeKind::Named { id, .. } = value.kind() else {
638            return None;
639        };
640        let prebindgen_flat::flat::Type::Variant(v) = self.model().resolve(id)? else {
641            return None;
642        };
643        let alternative = v.alternatives.get(index)?;
644        Some((&alternative.name, alternative.fields.as_slice()))
645    }
646
647    /// Whether this type states any deconstructing row at all.
648    ///
649    /// Asked of a part the walk did not look through, to say whether there was
650    /// something there to look through — a plain field is read in full, and
651    /// only one whose value has parts of its own leaves the read short.
652    fn deconstructing_any(&self, ty: &TypeRef) -> bool {
653        let crossing =
654            crate::recipe::Crossing::new(ty.clone(), crate::recipe::Direction::Deconstruct).key();
655        self.recipes()
656            .names_of(&crossing)
657            .iter()
658            .any(|name| **name != crate::recipe::RecipeName::derived())
659    }
660
661    /// This crossing's deconstructing row under `row`, if it states one.
662    fn deconstructing(&self, ty: &TypeRef, row: &RecipeName) -> Option<&Shape<Deconstruct>> {
663        let crossing =
664            crate::recipe::Crossing::new(ty.clone(), crate::recipe::Direction::Deconstruct).key();
665        match self.recipes().get(self.recipes().key_of(&crossing, row)?)? {
666            Recipe::Deconstructing(shape) => Some(shape),
667            Recipe::Constructing(_) => None,
668        }
669    }
670
671    /// What a part's own leaf carries, when it is not spliced.
672    fn leaf_ty(&self, reach: &Reach, source: &TypeRef, index: usize) -> TypeRef {
673        match reach {
674            Reach::Accessor(func) => self.returns(func).unwrap_or_else(|_| source.clone()),
675            Reach::Field(field) => self
676                .field_ty(source, *field)
677                .unwrap_or_else(|| source.clone()),
678            _ => {
679                let _ = index;
680                source.clone()
681            }
682        }
683    }
684
685    fn returns(&self, func: &syn::Ident) -> Result<TypeRef, UnfoldViewError> {
686        self.model()
687            .function(func)
688            .map(|f| f.ret.clone())
689            .ok_or_else(|| UnfoldViewError::UnknownAccessor(func.clone()))
690    }
691
692    fn consumes(&self, func: &syn::Ident) -> Result<bool, UnfoldViewError> {
693        let f = self
694            .model()
695            .function(func)
696            .ok_or_else(|| UnfoldViewError::UnknownAccessor(func.clone()))?;
697        Ok(f.params
698            .first()
699            .is_some_and(|p| p.ty.borrow_target().is_none()))
700    }
701
702    fn field_ty(&self, source: &TypeRef, index: usize) -> Option<TypeRef> {
703        self.fields_of(source).get(index).map(|f| f.ty.clone())
704    }
705
706    /// How a field is addressed: its name, or its position for a tuple struct.
707    fn field_ident(&self, source: &TypeRef, index: usize) -> syn::Ident {
708        self.fields_of(source)
709            .get(index)
710            .and_then(|f| f.name.clone())
711            .unwrap_or_else(|| {
712                syn::Ident::new(&format!("_{index}"), proc_macro2::Span::call_site())
713            })
714    }
715}
716
717/// Two leaves of one decomposition may not share a name: the foreign signature
718/// names its arguments, and two of one name is one the emitter cannot write.
719fn unique_names(leaves: &[UnfoldLeaf], source: &TypeRef) -> Result<(), UnfoldViewError> {
720    let mut seen = std::collections::HashSet::new();
721    for leaf in leaves {
722        if !seen.insert(leaf.name.clone()) {
723            return Err(UnfoldViewError::DuplicateName {
724                ty: source.key().to_string(),
725                name: leaf.name.clone(),
726            });
727        }
728    }
729    Ok(())
730}

732fn name_of(shape: &Shape<Deconstruct>) -> &'static str {
733    match shape {
734        Shape::Atomic => "atomic",
735        Shape::Optional => "an optional layer",
736        Shape::Sequence => "a sequence",
737        Shape::Invoke => "a callable",
738        Shape::Product(_) => "a product",
739        Shape::Choice { .. } => "a choice",
740    }
741}

743/// How a payload field is addressed in its arm's pattern: by name, or by
744/// position for a tuple variant.
745fn member_of(field: &prebindgen_flat::flat::Field, index: usize) -> syn::Member {
746    match &field.name {
747        Some(name) => syn::Member::Named(name.clone()),
748        None => syn::Member::Unnamed(syn::Index::from(index)),
749    }
750}

752fn reach_form(reach: &Reach) -> &'static str {
753    match reach {
754        Reach::Path(_) => "a field-of-a-field chain",
755        Reach::Nested { .. } => "a field taken apart in place",
756        _ => "a reach the view does not read",
757    }
