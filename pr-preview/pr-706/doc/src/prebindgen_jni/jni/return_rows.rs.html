1//! The return side, read off the rows rather than off the decomposition.
2//!
3//! `unfold` still builds every plan the emitters use. This reads the same
4//! decompositions off the `parts` rows the recipe table already carries, and
5//! holds the two to each other — a build fails on any difference (#701 step 3).
6//!
7//! The comparison goes with the older path, once every decomposition this
8//! binding declares has been through it. Until then it is the oracle a fixture
9//! cannot be: it runs over `covertest-kotlin`, `perftest-kotlin`, `emitcheck`
10//! and zenoh-flat-jni, which is the whole declared surface.
11

15    leaf::{Hoist, LeafSource, UnfoldLeaf},

17};

21/// What the JVM calls the values a decomposition delivers.
22///
23/// Every answer is one the decomposition writes into a leaf today, and the
24/// comparison below is what holds them to it.
25pub(crate) struct JniUnfold<'a> {
26    decls: &'a Declarations,
27    /// The registry, for the one name lookup that needs a signature.
28    registry: &'a dyn prebindgen_registry::Conversions,
29    /// The type whose declaration names these parts. A part name is the
30    /// declaration's answer, and a declaration is per type.
31    key: prebindgen_registry::TypeKey,
32}
33
34impl UnfoldPolicy for JniUnfold<'_> {
35    fn selector(&self, source: &TypeRef) -> UnfoldLeaf {
36        UnfoldLeaf {
37            name: "tag".to_string(),
38            path: Vec::new(),
39            // The sum, not the `i32` it crosses as: this is how the emitter
40            // finds the enum to match on. Registered and not required — a sum
41            // has no whole-value output converter (#282).
42            out_ty: source.clone(),
43            identity: false,
44            nullable: false,
45            source: LeafSource::SumTag,
46            groups: Vec::new(),
47        }
48    }
49
50    fn presence(&self, name: &str) -> UnfoldLeaf {
51        UnfoldLeaf {
52            name: format!("{name}__present"),
53            path: Vec::new(),
54            out_ty: TypeRef::scalar(ScalarKind::Bool),
55            identity: false,
56            nullable: false,
57            source: LeafSource::Presence,
58            groups: Vec::new(),
59        }
60    }
61
62    fn part_name(&self, reach: &Reach, index: usize, field: Option<&syn::Ident>) -> String {
63        match reach {
64            Reach::Accessor(func) => self.decls.leaf_name_of(&self.key, func, index),
65            // A synthesized `data_class` decomposition names its slots after
66            // the struct's own fields — as the KOTLIN property is named, which
67            // camel-cases and escapes a name that is a keyword there. A
68            // declared decomposition names its parts itself.
69            _ => match field {
70                Some(name) => crate::jni::mangle_kotlin_ident(&crate::jni::kt_snake_to_camel(
71                    &name.to_string(),
72                )),
73                None => self.decls.leaf_name_at(&self.key, index),
74            },
75        }
76    }
77
78    fn arm_part_name(&self, variant: &syn::Ident, member: &syn::Member, _index: usize) -> String {
79        // The Kotlin variant class's name and the Kotlin property's, which is
80        // what the slot is called on the far side — neither is the Rust name.
81        let kotlin = match self.decls.types.get(&self.key).and_then(|c| c.sum()) {
82            Some(cfg) => self.decls.sum_variant_class_name(cfg, variant),
83            None => variant.to_string(),
84        };
85        crate::jni::struct_plan::sum_slot_fragment(
86            &kotlin,
87            &crate::jni::struct_plan::sum_field_prop_name(member),
88        )
89    }
90
91    fn value_form_part(&self, index: usize) -> Option<String> {
92        // A value form's parts are named by the declaration that lists them,
93        // which is the one record `field_list` holds for such a declaration.
94        self.decls
95            .value_form_part_name(self.registry, &self.key, index)
96    }
97
98    fn identity_name(&self) -> String {
99        // A root identity leaf is `handle`; a nested one takes the name of the
100        // part it was reached through, which the view supplies instead.
101        "handle".to_string()
102    }
103
104    fn nest(&self, outer: &str, inner: &str) -> String {
105        format!("{outer}__{inner}")
106    }
107}

109impl Declarations {
110    /// Read every decomposition back off its row and hold the older path to it.
111    ///
112    /// Temporary, and deleted with `unfold::apply`. A decomposition whose row
113    /// the view does not read yet is skipped rather than failing: the forms it
114    /// refuses are the ones step 3 is still replacing, and the point of this is
115    /// what the two paths say about the forms it DOES read.
116    pub(crate) fn check_unfold_parity(
117        &self,
118        model: &Flat,
119        registry: &dyn prebindgen_registry::Conversions,
120        recipes: &Recipes,
121        bindings: &prebindgen_registry::recipe::Bindings,
122    ) -> Result<(), String> {
123        let folding = Folding::new(recipes, model);
124        let unfolded = self.unfolded();
125        let mut plans: Vec<(String, &crate::unfold::UnfoldPlan)> = unfolded
126            .unfold_plans
127            .iter()
128            .map(|(f, p)| (format!("`{f}`'s return"), p))
129            .chain(
130                unfolded
131                    .error_plans
132                    .iter()
133                    .map(|(f, p)| (format!("`{f}`'s error"), p)),
134            )
135            .chain(
136                unfolded
137                    .callback_arg_plans
138                    .iter()
139                    .map(|(k, p)| (format!("the callback argument `{k}`"), p)),
140            )
141            .collect();
142        // Sorted, so a binding with two differences reports the same one every
143        // time rather than whichever the hash order reached first.
144        plans.sort_by(|(a, _), (b, _)| a.cmp(b));
145
146        for (what, plan) in plans {
147            // A whole-element fold takes nothing apart: each element crosses
148            // through its own converter, so there is no row and nothing to
149            // compare.
150            let Some(decon) = &plan.decon else {
151                continue;
152            };
153            // A per-function `.expand_return(...)` states a decomposition of
154            // its own, which is a row of its own under a name of its own —
155            // the shape step 2 gave the parameter side, and step 3's remaining
156            // work here. The type's row does not describe it, so comparing the
157            // two would compare two different decompositions.
158            if !matches!(decon, prebindgen_registry::leaf::DeconId::Default(_)) {
159                continue;
160            }
161            // A `sealed_class` that ALSO carries an `expand_return!` states two
162            // decompositions under one row name: the sum's arms, declared by
163            // the class, and the fields, declared by the author. The row table
164            // keeps the first and the decomposition used the second, so the two
165            // describe different things and comparing them says nothing. One
166            // row name for two meanings is step 3's to resolve.
167            if self.is_sum(&plan.source) && self.declares_return_expand(&plan.source) {
168                continue;
169            }
170            let policy = JniUnfold {
171                decls: self,
172                registry,
173                key: plan.source.stripped_key(),
174            };
175            let row = RecipeName::new("parts");
176            let (leaves, hoists) = match folding.unfold(&policy, bindings, &plan.source, &row) {
177                Ok(read) => read,
178                // A shape the view does not represent yet — no row states this
179                // decomposition, or it states one in a form still being
180                // replaced. Step 3's remaining work rather than a disagreement.
181                Err(e) if e.is_not_yet_readable() => continue,
182                // Anything else is a row that is WRONG: two leaves of one name,
183                // two identities, a cycle, an accessor or an alternative the
184                // model does not have. Skipping those would let a bad row
185                // disable the check that found it.
186                Err(e) => return Err(format!("{what} has a row that cannot be read: {e}")),
187            };
188            // A row that reads LESS than the decomposition has not disagreed
189            // with it — it has not stated the whole decomposition yet. Three
190            // measures say so, and each names work step 3 has still to do.
191            //
192            // Fewer leaves: an `Atomic` placeholder, which a crossing carries
193            // so a site can select it and which says nothing about parts, or a
194            // row whose reaches the view still refuses.
195            //
196            // Shallower leaves: a part the decomposition takes apart further
197            // and no binding does, so the row stops where the decomposition
198            // goes through. Every part binding step 3 has yet to write is here,
199            // including looking THROUGH an optional field — the `Optional`
200            // part of #701's decision 3.
201            //
202            // Fewer parts handed over: a handle field of a CONSUMING value form
203            // moves out rather than being read, which the decomposition records
204            // as an identity leaf. Whether a part moves is the target's answer
205            // about its type, and the row states `Reach::Field` either way, so
206            // the view has no way to say it yet.
207            //
208            // A row that reads as far, as much and as strongly is compared, and
209            // any difference in it is a real one.
210            if leaves.len() < plan.leaves.len()
211                || depth(&leaves) < depth(&plan.leaves)
212                || identities(&leaves) < identities(&plan.leaves)
213            {
214                continue;
215            }
216            let from_row = describe(&leaves, &hoists);
217            let from_decl = describe(&plan.leaves, &plan.hoists);
218            if from_row != from_decl {
219                return Err(format!(
220                    "{what} reads back differently from its row.\nfrom the row:\n{from_row}\n\
221                     from the declarations:\n{from_decl}"
222                ));
223            }
224        }
225        Ok(())
226    }
227}

229/// How far the deepest leaf is reached from the value.
230fn depth(leaves: &[UnfoldLeaf]) -> usize {
231    leaves.iter().map(|leaf| leaf.path.len()).max().unwrap_or(0)
232}
233
234/// How many parts are handed over rather than read.
235fn identities(leaves: &[UnfoldLeaf]) -> usize {
236    leaves.iter().filter(|leaf| leaf.identity).count()
237}
238
239/// One decomposition, flattened far enough that two can be compared.
240fn describe(leaves: &[UnfoldLeaf], hoists: &[Hoist]) -> String {
241    use std::fmt::Write;
242    let path_of = |steps: &[prebindgen_registry::leaf::PathStep]| {
243        steps
244            .iter()
245            .map(|step| {
246                format!(
247                    "{}{}",
248                    step.ident(),
249                    if step.is_optional() { "?" } else { "" }
250                )
251            })
252            .collect::<Vec<_>>()
253            .join(".")
254    };
255    let mut out = String::new();
256    for leaf in leaves {
257        let _ = writeln!(
258            out,
259            "  {} : {} path=[{}] identity={} nullable={} source={} groups={:?}",
260            leaf.name,
261            leaf.out_ty.key(),
262            path_of(&leaf.path),
263            leaf.identity,
264            leaf.nullable,
265            source_of(&leaf.source),
266            leaf.groups
267        );
268    }
269    for hoist in hoists {
270        let _ = writeln!(
271            out,
272            "  hoist [{}] consuming={}",
273            path_of(&hoist.prefix),
274            hoist.consuming
275        );
276    }
277    out
278}

280/// How a leaf is reached, for the comparison.
281///
282/// A nested sum's payload is deliberately not distinguished from a plain reach.
283/// The two paths record it differently — `synth_value_struct_leaves` maps a
284/// rebased payload wire to `Reach`, dropping the alternative, where the view
285/// keeps `VariantField` — and which of the two is right is not this check's
286/// question. What the emitter reads off such a leaf is its `groups`, which both
287/// paths agree on and this compares.
288fn source_of(source: &LeafSource) -> String {
289    match source {
290        LeafSource::VariantField { .. } | LeafSource::Reach => "reach".to_string(),
291        LeafSource::SumTag => "tag".to_string(),
292        LeafSource::Presence => "present".to_string(),
293    }
