58/// Short name of the generated `@RequiresOptIn` marker annotation class.
59///
60/// The Kotlin half of the raw-pointer guard: it marks every generated entry
61/// point that hands a raw native pointer to, or takes one from, safe Kotlin
62/// and cannot be hidden outright — `NativeHandle.peek()` and the `fromParts`
63/// factories, both reached from Rust by JNI reflection, so renaming them is
64/// not an option. [`JVM_SYNTHETIC`] is the other half, for Java.
65///
66/// See [`Declarations::unsafe_marker_fqn`] for the qualified form.
67pub(crate) const UNSAFE_MARKER: &str = "UnsafeNativeApi";
68
69/// `kotlin.jvm.JvmSynthetic` — the JVM half of the raw-pointer guard.
70///
71/// `internal` and `@RequiresOptIn` are Kotlin-source constructs and stop at
72/// the Kotlin compiler: `internal` members are emitted as **public** JVM
73/// methods under a mangled name, and javac neither knows nor enforces an
74/// opt-in marker. `@JvmSynthetic` sets `ACC_SYNTHETIC`, which javac skips when
75/// resolving a call, so a Java consumer cannot name the member at all.
76///
77/// It leaves the name and the JVM signature alone, so JNI's own lookup
78/// (`GetMethodID` / `GetStaticMethodID`, which ignore the flag) still finds
79/// `peek`, `fromParts`, and the `external fun` externs. That is exactly what
80/// `internal` could not offer. It is **not** applicable to a constructor
81/// (Kotlin rejects the target), which is why a handle's constructor is instead
82/// `private` behind a synthetic factory — see [`HANDLE_FACTORY`].
83pub(crate) const JVM_SYNTHETIC: &str = "JvmSynthetic";
84
85/// Name of the generated per-handle factory that replaces the raw-pointer
86/// constructor: `Storage.fromRawPtr(p)` rather than `Storage(p)`.
87///
88/// A handle's constructor is `private` — `internal` would still be a public
89/// JVM constructor and `@JvmSynthetic` cannot be applied to one, so `new
90/// Storage(0xdeadbeefL)` compiled fine from Java. The factory is `internal` +
91/// [`JVM_SYNTHETIC`], reachable from generated Kotlin and from nowhere else.
92pub(crate) const HANDLE_FACTORY: &str = "fromRawPtr";

102/// An `internal` member of the generated surface, hidden from Java too.
103///
104/// Always pair the two: `internal` alone leaves a public JVM method behind a
105/// mangled name, and `handle.setPtr$mymodule(0xdeadbeefL)` from Java would
106/// then repoint a live handle at an address of the caller's choosing, which

133/// Options of a type declared with `ptr_class!` — the payload of
134/// [`DeclaredKind::Ptr`], which is what marks the type as an opaque handle.
135/// The unified Kotlin emitter writes a typed-handle `.kt` file (and the Rust
136/// side its `freePtr` destructor) for every such type.
137///
138/// The typed-handle Kotlin FQN (e.g. `"io.zenoh.jni.JNISession"`) lives
139/// in the surrounding [`TypeConfig::name_spec`] slot — FQN-consumers
140/// (typed-handle class emission, `instanceof` dispatch,
141/// return-value constructor wrap) read it from there. The
142/// value-context Kotlin name for the same type (`"Long"`) is produced
143/// independently by the rank-0 opaque handler in [`KotlinMeta`], so
144/// the two roles don't collide despite sharing the `TypeConfig`.
145#[derive(Clone, Default)]
146pub(crate) struct OpaqueConfig {
147    /// `ptr_class!(X).gc_managed()`: the typed handle stores its pointer in
148    /// a separate atomic cell and registers a `Cleaner` action that frees the
149    /// native box if no other release path (close/take/consumption) won the
150    /// untagged→tagged CAS ticket first.
151    pub gc_managed: bool,
152}

154/// Options of a type declared with `enum_class!` — the payload of
155/// [`DeclaredKind::Enum`], which is what marks a `#[prebindgen]`-scanned
156/// `enum` as a Kotlin enum class. There are none yet: the declaration itself
157/// carries all the information, so this is the empty slot future enum options
158/// go in.
159///
160/// The rank-0 converter arms emit `jint ↔ Rust enum` bodies (via
161/// `TryFrom<i32>` for input and `as jni::sys::jint` for output), and the
162/// Kotlin emitter writes an `enum class` file with SCREAMING_SNAKE_CASE
163/// variants and a discriminant-keyed `fromInt(...)` companion. The Kotlin FQN
164/// lives in the surrounding [`TypeConfig::name_spec`] slot, same as
165/// [`OpaqueConfig`].
166#[derive(Clone, Default)]
167pub(crate) struct EnumConfig {}

169/// Per-variant naming of a type declared with `sealed_class!` — the payload
170/// of [`DeclaredKind::Sealed`], which is what marks a `#[prebindgen]`

173/// from the model's [`Variant`](prebindgen_registry::flat::Variant) — its

208/// Which of the five class declarators a type is registered under, **with
209/// that declarator's own options**. A type gets exactly one — this is a sum,
210/// not a set of independent markers: two declarators would emit two Kotlin
211/// declarations for the same FQN, so the second is a hard error (see
212/// [`DeclaredKind::merge`]). Reopening the *same* declarator stays legal —
213/// `merge` folds the incoming payload into the stored one.
214///
215/// Adding a sixth class kind is one variant here plus its emitter: there is
216/// no flag to add and no precedence chain to extend, because every consumer
217/// reads this one field (via [`TypeConfig`]'s accessors, [`JniGenBuilder::type_kind`],
218/// or a direct match).
219#[derive(Clone)]
220pub(crate) enum DeclaredKind {
221    /// `ptr_class!` — an opaque-handle type: jlong wire,
222    /// `Box::into_raw`/`Box::from_raw` conventions, instanceof dispatch, and
223    /// Kotlin typed-handle class emission.
224    Ptr(OpaqueConfig),
225    /// `enum_class!` — a `#[prebindgen]` fieldless enum mirrored as a Kotlin
226    /// `enum class`: jint wire (input + output via `TryFrom<i32>` / `as
227    /// jint`) and a generated `.kt` file.
228    Enum(EnumConfig),
229    /// `sealed_class!` — a `#[prebindgen]` data-carrying enum mirrored as a
230    /// Kotlin `sealed interface`: a tag plus one leaf group per variant.
231    Sealed(SumConfig),
232    /// `data_class!` — a `#[prebindgen]` struct mirrored as a Kotlin `data
233    /// class`, flattened field-by-field at the boundary. The kind with no
234    /// options of its own.
235    Data,
236}

245/// All configuration the structured builder accumulates for one
246/// canonical Rust type key. The declared kind is mandatory (an entry exists
247/// only because some declarator created it); the remaining, cross-kind
248/// options default to unset and are populated by the decl that carries them.
249#[derive(Clone)]
250pub(crate) struct TypeConfig {
251    /// The class declarator this type is registered under, carrying that
252    /// kind's own options. Every entry in [`JniGenBuilder::types`] has one: entries
253    /// are created only by a class declarator (see
254    /// `JniGenBuilder::register_class`), which is why presence in the table *is*
255    /// "declared as a class" — declared classes are required in **both**
256    /// directions at scan (their converters always resolve both ways),
257    /// unlike a wrapper registration, which is required per **usage**
258    /// direction.
259    pub kind: DeclaredKind,
260    /// The type this declaration was **written with**, e.g. the `Foo` in
261    /// `ptr_class!(Foo)`.
262    ///
263    /// A class declarator receives a real `syn::Type` and used to keep only the
264    /// key derived from it, so every later site that needed the tokens back had
265    /// to ask the key for them. That is the wrong direction: the declaration is
266    /// where the type came from, and this is where it stays (#291).
267    pub rust_type: Origin<syn::Type>,
268    /// Raw naming spec of the type as declared — verbatim Kotlin type or
269    /// settings-derived class name. Required for any type emitted in
270    /// Kotlin; the concrete FQN (`Sample` → `"io.zenoh.jni.Sample"`,
271    /// `Vec<u8>` → `"ByteArray"`) is materialized only at read time via
272    /// [`JniGenBuilder::fqn_of`], which is what makes the `set_*` settings
273    /// order-independent w.r.t. declarations.
274    pub name_spec: Option<NameSpec>,
275    /// Explicit opt-in for a `data_class` to cross Kotlin → Rust as one
276    /// `JObject`. Unmarked data classes are required to flatten completely;
277    /// this flag is sticky across reopened declarations.
278    pub jobject_input: bool,
279    /// Emit the generated interface mirroring the class's public instance
280    /// surface, and make the class implement it with `override` on every
281    /// class-body member (`.interface()` / implied by `.interface_name()`).
282    pub interface_enabled: bool,
283    /// Per-decl literal interface name (`.interface_name(...)`, relative, no
284    /// dots) — bypasses the `set_interface_name_mangle` hook.
285    pub interface_name_override: Option<String>,
286    /// Kotlin interfaces added to the generated class's supertype list
287    /// (`.implements(...)`, any class kind) — the class implements them
288    /// nominally; the class body and lifecycle members are unaffected.
289    /// Orthogonal to [`Self::interface_enabled`].
290    pub interfaces: Vec<String>,
291}

421/// JNI back-end. Global settings are applied with the order-insensitive
422/// `set_*` methods; declarations are accepted as pre-built objects
423/// (`PackageDecl`, `ExpandParamDecl`, `ExpandReturnDecl`,
424/// `ConvertDecl` — see `decl.rs`) built

427/// ```

430/// let jni = JniGenBuilder::new()
431///     .set_package_prefix("io.test.jni")
432///     .package(
433///         prebindgen_jni::package!("keyexpr")
434///             .class(prebindgen_jni::ptr_class!(KeyExpr)
435///                 .method(prebindgen_registry::fun!(keyexpr_get_str).name("getStr"))

442///             .variant(prebindgen_registry::fun!(keyexpr_new_try_from))

445///     .expand(prebindgen_registry::expand_return!(KeyExpr).field(prebindgen_registry::fun!(keyexpr_get_str)));

448/// can be written.

481/// How a reach renders: its field steps, joined — the accessor call every leaf
482/// hangs off is the site's, not the value's, so it is dropped.
483#[cfg(test)]
484fn reach_of(path: &[unfold::PathStep]) -> String {
485    use crate::unfold::PathStep;
486    path.iter()
487        .filter_map(|step| match step {
488            PathStep::Field { ident, .. } => Some(ident.to_string()),
489            PathStep::Call { .. } => None,
490        })
491        .collect::<Vec<_>>()
492        .join(".")
493}

495/// How the Rust side reaches one plan leaf, in the form a wire renders it —
496/// the accessor call every leaf hangs off is the site's, not the value's, so it
497/// is dropped to line the two up.
498#[cfg(test)]
499fn leaf_reach(leaf: &unfold::UnfoldLeaf) -> String {
500    use crate::unfold::LeafSource;
501    match &leaf.source {
502        LeafSource::SumTag => "tag".to_string(),
503        LeafSource::Presence => format!("present({})", reach_of(&leaf.path)),
504        LeafSource::VariantField { variant, member } => format!(
505            "{variant}.{}",
506            crate::jni::struct_plan::sum_field_prop_name(member)
507        ),
508        _ => reach_of(&leaf.path),
509    }
510}

537    fn render_plan_for_test(
538        &self,
539        ty: syn::Type,
540        direction: prebindgen_registry::recipe::Direction,
541        parts: bool,
542    ) -> Result<(bool, String, String), String> {
543        use prebindgen_registry::recipe::{Compiler, Crossing};
544
545        let reading = self
546            .registry
547            .flat()
548            .classify(&ty)
549            .map_err(|e| format!("the model does not classify {}: {e}", quote::quote!(#ty)))?;
550        let crossing = Crossing::new(reading, direction);
551        let mut compiler = Compiler::resume(
552            &self.registry,
553            self.decls.recipe_table(),
554            self.decls.site_bindings(),
555            self.decls.compiled.borrow().clone(),
556        );
557        let mut adapter = crate::jni::compile::JCompile {
558            decls: &self.decls,
559            declared_return: None,
560        };
561        let fragment = if parts {
562            // By its row rather than by name: asking for a row the crossing does
563            // not have is the caller's own claim, and answering it with the
564            // default would file the default under a recipe that does not exist.
565            let row = self
566                .decls
567                .recipe_table()
568                .key_of(&crossing.key(), &crate::jni::recipes::parts())
569                .cloned()
570                .ok_or_else(|| format!("{} states no `parts` recipe", crossing.key()))?;
571            compiler.row(&mut adapter, &crossing, &row)
572        } else {
573            compiler.crossing(&mut adapter, &crossing).map(|(f, _)| f)
574        }
575        .map_err(|e| e.to_string())?;
576        fragment.rust.mark_reachable();
577        let rendered =
578            fragment
579                .rust
580                .render_fn(&prebindgen_registry::RustWriter::for_registry_test(
581                    &self.registry,
582                ));
583        Ok((
584            fragment.composed_only,
585            rendered.sig.ident.to_string(),
586            quote::quote!(#rendered).to_string(),
587        ))
588    }

926        // onto it — so this is what says it is ready to.

931        // dependency edges name every call its artifacts render — the

952    /// Every compiled fragment states the same thing twice while step 5 is in
953    /// progress — as a [`JFrag`](crate::jni::compile::JFrag) and as the
954    /// `FragmentPlan` the canonical plan holds — and this is what says the two
955    /// agree.
956    ///
957    /// Identity, source and artifact presence are compared because they are
958    /// what a plan is keyed and spelled by, and what an emitter would read
959    /// instead of the fragment. A composed-only fragment renders nothing, and
960    /// the canonical plan says so by carrying no artifact at all.
961    ///
962    /// The plan itself is **not** rebuilt here. Since 5a it is built and
963    /// validated in `JniGenerationPlan::freeze`, so a set-level defect — a
964    /// chain whose endpoints do not join up, an edge naming a fragment nothing
965    /// compiled, a duplicate site identity — fails the binding rather than one
966    /// test. What is left for this check is the per-fragment agreement that
967    /// only survives while `JFrag` still exists; 5c deletes both.
968    #[cfg(test)]
969    fn assert_fragments_freeze(&self) {
970        let generation = self
971            .decls
972            .generation
973            .as_ref()
974            .expect("resolved JniGen has no frozen generation plan");
975        // Driven from the PLAN, not from the compiled store. `build` keeps the
976        // fragments reachable from a site and prunes the rest, so the store is
977        // a superset — a converter the assembly still reaches on its own is in
978        // it and not in the plan. Closing that is step 8's, where artifact
979        // ordering moves onto the plan; until then the plan's own fragments are
980        // what there is to agree about.
981        let compiled = self.decls.compiled.borrow();
982        for plan in generation.plan().fragments() {
983            let fragment = compiled
984                .recipe_fragment(plan.id().spelling(), plan.id().recipe())
985                .unwrap_or_else(|| {
986                    panic!(
987                        "`{}`: the canonical plan holds a fragment nothing compiled",
988                        plan.id().spelling()
989                    )
990                });
991            assert_eq!(
992                plan.id(),
993                &fragment.id,
994                "a frozen fragment plan changed the fragment's identity"
995            );
996            assert_eq!(
997                plan.source().key(),
998                fragment.source.key(),
999                "`{}`: a frozen fragment plan changed the crossing it answers",
1000                fragment.source.key()
1001            );
1002            assert_eq!(
1003                plan.artifact().is_some(),
1004                !fragment.composed_only,
1005                "`{}`: a composed-only fragment renders nothing, and only such a \
1006                 fragment freezes without an artifact",
1007                fragment.source.key()
1008            );
1009            // The conversion graph, step for step. Comparing only that a chain
1010            // exists would not have caught freezing every staged custom
1011            // conversion as `Direct`, which is what the first version of this
1012            // did (#621 review): the operations and their order ARE the
1013            // contract, and a reader of the plan routes to stage artifacts
1014            // through them.
1015            let steps = |chain: &prebindgen_registry::generation::ConversionChain<
1016                crate::jni::compile::JRepresentation,
1017            >| match chain {
1018                prebindgen_registry::generation::ConversionChain::Direct => Vec::new(),
1019                prebindgen_registry::generation::ConversionChain::Steps(steps) => steps
1020                    .iter()
1021                    .map(|step| {
1022                        format!(
1023                            "{:?}:{:?}->{:?}:{:?}",
1024                            step.operation(),
1025                            step.from(),
1026                            step.into(),
1027                            step.failure()
1028                        )
1029                    })
1030                    .collect(),
1031            };
1032            assert_eq!(
1033                steps(plan.converter().chain()),
1034                steps(&fragment.chain),
1035                "`{}`: a frozen fragment plan changed the conversion graph",
1036                fragment.source.key()
1037            );
1038            // Both niche domains: what the fragment spends on its own absence,
1039            // and what it leaves for a parent. Exposing the remainder while
1040            // losing the consumed slot was the other thing this check could not
1041            // see.
1042            // Both niche domains, by identity rather than by count. Comparing
1043            // how many there are would pass a substituted sentinel, and
1044            // `GenerationPlanBuilder` cannot see one either — a niche is a
1045            // bit-pattern contract, so what it IS is the whole of it.
1046            let niches = plan.converter().niches();
1047            let key = crate::jni::compile::niche_key;
1048            assert_eq!(
1049                niches.consumed(),
1050                fragment
1051                    .consumed_niche
1052                    .iter()
1053                    .map(key)
1054                    .collect::<Vec<_>>()
1055                    .as_slice(),
1056                "`{}`: a frozen fragment plan changed the niche it consumed",
1057                fragment.source.key()
1058            );
1059            assert_eq!(
1060                niches.discriminants(),
1061                niches.consumed().len(),
1062                "`{}`: a consumed niche is what the fragment discriminates on",
1063                fragment.source.key()
1064            );
1065            assert_eq!(
1066                niches.exposed(),
1067                fragment
1068                    .conv
1069                    .niches
1070                    .slots
1071                    .iter()
1072                    .map(key)
1073                    .collect::<Vec<_>>()
1074                    .as_slice(),
1075                "`{}`: a frozen fragment plan changed the niches it leaves for a parent",
1076                fragment.source.key()
1077            );
1078            // A composition's shape has to be the composition, not the
1079            // placeholder every fragment starts with: `JLayout` is what JNI
1080            // says today about occupying several wires, so a fragment whose
1081            // layout composes must not freeze as `Atomic`.
1082            if matches!(
1083                fragment.layout,
1084                Some(crate::jni::compile::JLayout::Product(_))
1085                    | Some(crate::jni::compile::JLayout::Optional(_))
1086                    | Some(crate::jni::compile::JLayout::Choice(_))
1087            ) {
1088                assert!(
1089                    !matches!(
1090                        plan.converter().shape(),
1091                        prebindgen_registry::generation::ShapePlan::Atomic(_)
1092                    ),
1093                    "`{}`: a fragment whose layout composes froze as atomic — the \
1094                     hook that composed it did not state its shape",
1095                    fragment.source.key()
1096                );
1097            }
1098        }
1099        // Every site the compilation stated reached the stored plan. The
1100        // set-level properties — a duplicate identity, a failure route that
1101        // does not match its fragment's fallibility, an edge naming a fragment
1102        // nothing compiled — are `build`'s, and `build` now runs in `freeze`,
1103        // so reaching here at all means it passed. What this adds is that
1104        // nothing was dropped between stating a site and storing it.
1105        let stated = self.decls.site_plans.borrow();
1106        assert_eq!(
1107            stated.len(),
1108            generation.plan().sites().len(),
1109            "the canonical plan holds a different number of sites than were stated"
1110        );
1111        for site in stated.iter() {
1112            // The slot count is what the ABI itself says it is, so the two
1113            // cannot drift: `slots` is taken from the leaves rather than
1114            // counted beside them.
1115            assert_eq!(
1116                site.abi().slots(),
1117                site.abi().payload().len(),
1118                "site {}: the slot count and the ABI disagree",
1119                site.id().site()

1425/// Holds the [`Declarations`] being filled and the sources to parse, and it is the
