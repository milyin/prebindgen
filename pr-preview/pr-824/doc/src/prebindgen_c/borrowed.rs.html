1//! Non-owning numeric collection views, with no per-element output conversion.
2use super::*;
3use prebindgen_declarations::{Output, SequenceDelivery};
4use std::collections::{BTreeMap, BTreeSet};

6pub(crate) fn compatible(element: Primitive) -> bool {
7    !matches!(
8        element,
9        Primitive::Bool | Primitive::Char | Primitive::I128 | Primitive::U128
10    )
11}
12/// A primitive whose configured C representation is still the native scalar.
13pub(crate) fn primitive(
14    ty: &syn::Type,
15    types: &BTreeMap<String, prebindgen_flat::TypeInfo<CType>>,
16) -> Option<Primitive> {
17    let primitive = Primitive::from_type(ty).filter(|p| compatible(*p))?;
18    if types.values().any(|info| {
19        Primitive::from_type(&info.ty) == Some(primitive)
20            && (info.metadata.pointer.is_some()
21                || info.metadata.owned.is_some()
22                || info.metadata.enumeration.is_some())
23    }) {
24        return None;
25    }
26    Some(primitive)
27}
28pub(crate) fn native_element_type(element: Primitive) -> syn::Type {
29    let name = format_ident!("{}", prebindgen_flat::type_key(&element.rust_type()));
30    syn::parse_quote!(::core::primitive::#name)
31}
32pub(crate) fn ident(element: Primitive) -> syn::Ident {
33    format_ident!(
34        "__PbSliceView_{}",
35        prebindgen_flat::type_key(&element.rust_type())
36    )
37}
38pub(crate) fn c_name(element: Primitive, prefix: &str) -> String {
39    format!(
40        "{prefix}_{}_slice_view",
41        prebindgen_flat::type_key(&element.rust_type())
42    )
43}
44#[allow(clippy::too_many_arguments)]
45pub(crate) fn sequence(
46    flat: &Bindings,
47    ty: &syn::Type,
48    wire: syn::Ident,
49    value: TokenStream,
50    recipe: &Output,
51    delivery: SequenceDelivery,
52    temporary: bool,
53) -> Result<Option<OutputPlan<CWire>>> {
54    // An explicitly requested visitor does not allocate row storage. Its leaves
55    // still enforce their own ownership/copy policies.
56    if delivery == SequenceDelivery::Callback {
57        return Ok(None);
58    }
59    if temporary
60        && !matches!(ty, syn::Type::Reference(r) if r.lifetime.as_ref().is_some_and(|l|l.ident=="static"))
61    {
62        return Err("borrowed C collection view cannot escape a temporary owner; return its owner alongside a generated heap view getter or explicitly permit a collection copy".into());
63    }
64    let (element_type, view) =
65        prebindgen_flat::sequence_parts(ty).ok_or("borrowed sequence requires a collection")?;
66    if view == prebindgen_flat::SequenceView::Mutable {
67        return Err("mutable C collection views are unsupported".into());
68    }
69    let primitive = primitive(element_type, &flat.types);
70    let Some(element) = primitive.filter(|_| matches!(recipe, Output::Leaf)) else {
71        return Err(format!("borrowed C collection `{}` cannot allocate converted rows implicitly; use an address-preserving numeric element representation or explicitly permit a collection-level copy with expand_return!(&[T]).field_self_ref() (not an element-only expansion)", prebindgen_flat::type_key(ty)));
72    };
73    let element_type = native_element_type(element);
74    let carrier = ident(element);
75    let source = format_ident!("{wire}_source");
76    Ok(Some(OutputPlan {
77        fallible: false,
78        context: Default::default(),
79        delivery: Default::default(),
80        commit: TokenStream::new(),
81        outcomes: prebindgen_flat::Outcomes {
82            success: prebindgen_flat::WireNode::Leaf(vec![Wire {
83                name: wire.clone(),
84                kind: CWire::SliceView { element },
85            }]),
86            source_error: None,
87            binding_failure: None,
88        },
89        statements: quote!(
90            let #source = &(#value);
91            let #wire: &[#element_type] = ::core::convert::AsRef::<[#element_type]>::as_ref(#source);
92            let #wire = #carrier { data: <[#element_type]>::as_ptr(#wire), len: <[#element_type]>::len(#wire) };
93        ),
94        values: vec![(wire.clone(), quote!(#wire))],
95    }))
96}
97/// A loan anywhere in a packet, including fields inside converted rows.
98pub(crate) fn has_loans(node: &prebindgen_flat::WireNode<CWire, CChain, CBranch>) -> bool {
99    prebindgen_flat::wire_definitions(std::iter::once(node)).any(|wire| {
100        matches!(
101            wire.kind,
102            CWire::View
103                | CWire::MutableView
104                | CWire::SliceView { .. }
105                | CWire::Owned {
106                    access: Access::Shared | Access::Exclusive,
107                    ..
108                }
109        ) || matches!(&wire.kind, CWire::Pointer {pointer,..} if pointer.access != Access::Move)
110    })
111}
112pub(crate) fn collect(
113    node: &prebindgen_flat::WireNode<CWire, CChain, CBranch>,
114    out: &mut BTreeMap<String, Primitive>,
115) {
116    for wire in prebindgen_flat::wire_definitions(std::iter::once(node)) {
117        match &wire.kind {
118            CWire::SliceView { element } => {
119                out.insert(prebindgen_flat::type_key(&element.rust_type()), *element);
120            }
121            CWire::Callback { callback } => callback.slice_views(out),
122            _ => {}
123        }
124    }
125}
126pub(crate) fn emit(
127    element: Primitive,
128    prefix: &str,
129    symbols: &mut BTreeSet<String>,
130    tokens: &mut TokenStream,
131    header: &mut String,
132) -> Result<()> {
133    let name = c_name(element, prefix);
134    if !symbols.insert(name.clone()) {
135        return Err("generated C slice-view type collides with another C symbol".into());
136    }
137    let rust = ident(element);
138    let ty = native_element_type(element);
139    tokens.extend(quote!(#[allow(non_camel_case_types)] #[repr(C)] #[derive(Clone,Copy)] pub struct #rust {pub data: *const #ty,pub len: usize}));
140    let c = c_scalar(element, prefix);
141    header.push_str(&format!("/* Shared borrowed slice; never free or consume. Keep source storage alive; do not mutate it or obtain overlapping exclusive/mutable access until the view's last use. Empty views may have a non-null dangling data pointer; never dereference when len is zero. Callback argument views last only for that callback invocation. */\ntypedef struct {{ const {c} *data; size_t len; }} {name};\n"));
142    Ok(())
143}
