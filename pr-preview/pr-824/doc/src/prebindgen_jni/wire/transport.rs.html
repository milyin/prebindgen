1//! JNI representation and code generation for individual wire values.
2use super::jvm::JvmType;
3use super::*;

5/// Code preparing one JNI method argument from a prepared output value.
6///
7/// Emit `statements` before using `value`, in the same local-reference frame.
8/// The caller owns that frame and must keep its references alive until the
9/// method invocation finishes. Neither fragment commits an ownership transfer.
10#[derive(Clone, Debug)]
11#[non_exhaustive]
12pub struct JniArgument {
13    /// Fallible preparation, including any Java object allocation.
14    pub statements: TokenStream,
15    /// A `jni::sys::jvalue` expression referring to the prepared value.
16    pub value: TokenStream,
17}
18
19impl JniWire {
20    fn jvm_type(&self) -> JvmType {
21        match self {
22            Self::Scalar { primitive, .. } => match primitive {
23                Primitive::Bool => JvmType::Boolean,
24                Primitive::I8 => JvmType::Byte,
25                Primitive::I16 => JvmType::Short,
26                Primitive::U8 | Primitive::U16 | Primitive::I32 | Primitive::Char => JvmType::Int,
27                Primitive::U32
28                | Primitive::U64
29                | Primitive::I64
30                | Primitive::Usize
31                | Primitive::Isize => JvmType::Long,
32                Primitive::F32 => JvmType::Float,
33                Primitive::F64 => JvmType::Double,
34                Primitive::I128 | Primitive::U128 => JvmType::Bytes,
35            },
36            Self::Handle { .. } => JvmType::Long,
37            Self::Selector { .. } | Self::Enum { .. } => JvmType::Int,
38            Self::Text { .. } => JvmType::Text,
39            Self::Bytes { .. } => JvmType::Bytes,
40            Self::Rows { .. } => JvmType::Rows,
41            Self::Callback { .. } => JvmType::Object,
42        }
43    }
44    /// JVM field descriptor used in method signatures, independent of Kotlin spelling.
45    pub fn jvm_descriptor(&self) -> &'static str {
46        self.jvm_type().descriptor()
47    }
48    /// Kotlin type used by the native declaration, before typed projections.
49    pub fn kotlin_type(&self) -> &'static str {
50        self.jvm_type().kotlin_type()
51    }
52    /// JVM argument slots occupied by this value (`long` and `double` occupy two).
53    pub fn parameter_slots(&self) -> usize {
54        self.jvm_type().parameter_slots()
55    }
56    /// Whether the native input is a JNI reference rather than a copyable scalar.
57    ///
58    /// Handles are integers, so this returns false for them. Rust i128/u128
59    /// arrive as byte-array references, so this returns true for those scalars.
60    pub fn is_reference(&self) -> bool {
61        self.jvm_type().is_reference()
62    }
63    /// Kotlin placeholder for an inactive wire, without selecting an option niche.
64    pub fn kotlin_default(&self) -> &'static str {
65        self.jvm_type().kotlin_default()
66    }
67    pub(crate) fn kotlin_integer(&self, value: i64) -> String {
68        if matches!(self.jvm_type(), JvmType::Long) {
69            format!("{value}L")
70        } else {
71            value.to_string()
72        }
73    }
74    /// Generate an expression reading a checked, boxed field of a Java `Object[]`.
75    ///
76    /// The expression returns `Result<rust_type()>`, without applying Rust-domain
77    /// validation or acquiring handles. Boolean inputs are normalized to `u8`.
78    /// Reference fields preserve Java null for optional/inactive values. The
79    /// caller supplies the `environment` template parameter and owns local refs.
80    pub fn read_array_element(&self, array: &TokenStream, index: usize) -> TokenStream {
81        self.jvm_type().read_array_element(array, index)
82    }
83    pub(crate) fn callback_return_type(&self) -> TokenStream {
84        self.jvm_type().return_type()
85    }
86    pub(crate) fn read_callback_return(&self, value: &syn::Ident) -> TokenStream {
87        self.jvm_type().read_return(value)
88    }
89
90    /// Generate a Java object expression from a prepared output, boxing primitives.
91    ///
92    /// This accepts the representation in `OutputPlan::values`, not a native
93    /// input: text/bytes are prepared nullable JVM objects, i128/u128 are encoded bytes,
94    /// handles are allocation pointers, and rows are optional JNI arrays.
95    /// The expression can fail through `?`, uses the `environment` template
96    /// parameter, and returns a local reference owned by the caller's frame.
97    /// Text/byte payloads are consumed; each prepared object may be delivered
98    /// once. Other wire kinds retain their declared representation.
99    /// It does not commit any ownership transfer.
100    pub fn output_object(&self, value: &TokenStream) -> TokenStream {
101        let env = prebindgen_conversions::template_parameter("environment");
102        if let Some(boxed) = self.jvm_type().boxed_output(&self.output_scalar(value)) {
103            return boxed;
104        }
105        match self.jvm_type() {
106            JvmType::Text => quote!(#value),
107            JvmType::Bytes if matches!(self, Self::Scalar { .. }) => {
108                quote!(::prebindgen_jni_runtime::bytes_object(&mut #env, &(#value))?)
109            }
110            JvmType::Bytes => quote!(#value),
111            JvmType::Object => quote!(#env.new_local_ref(&(#value)).map_err(|e|e.to_string())?),
112            JvmType::Rows => {
113                quote!(match (#value).as_ref(){Some(value)=>#env.new_local_ref(value).map_err(|e|e.to_string())?,None=>::prebindgen_jni_runtime::jni::objects::JObject::null()})
114            }
115            _ => unreachable!("JVM primitives use their declared boxing conversion"),
116        }
117    }
118    fn output_scalar(&self, value: &TokenStream) -> TokenStream {
119        if matches!(self, Self::Handle { .. }) {
120            quote!((#value) as i64)
121        } else {
122            value.clone()
123        }
124    }
125    /// Generate one JNI method argument from the same prepared output as
126    /// [`Self::output_object`], leaving JVM primitives unboxed.
127    ///
128    /// `local` must be a fresh identifier for this argument's Java object.
129    /// Emit the returned statements before its value and keep references alive
130    /// through the invocation. Text/byte prepared payloads are consumed once.
131    /// The caller supplies `environment`, validates the
132    /// method signature, and commits ownership only after successful delivery.
133    /// A custom generator can render both fragments with its [`JniContext`]:
134    ///
135    /// ```
136    /// use prebindgen_conversions::TemplateContext;
137    /// use prebindgen_jni::{JniContext, JniWire};
138    /// use quote::quote;
139    /// let context = JniContext {
140    ///     environment: syn::parse_quote!(env),
141    ///     acknowledgement: syn::parse_quote!(accepted),
142    ///     domain_output: syn::parse_quote!(domain_output),
143    ///     is_domain_error: false,
144    ///     output_committed: quote!(),
145    /// };
146    /// let argument = JniWire::text().output_argument(
147    ///     &quote!(prepared_text), &syn::parse_quote!(java_text));
148    /// let statements = context.render(&argument.statements)?;
149    /// let value = context.render(&argument.value)?;
150    /// // Emit statements, invoke with value, then release the local frame.
151    /// # assert!(!statements.to_string().contains("__prebindgen_emit"));
152    /// # assert!(!value.to_string().contains("__prebindgen_emit"));
153    /// # Ok::<(), String>(())
154    /// ```
155    pub fn output_argument(&self, value: &TokenStream, local: &syn::Ident) -> JniArgument {
156        let member = format_ident!("{}", self.jvm_type().value_member());
157        let (statements, value) = if self.is_reference() {
158            let object = self.output_object(value);
159            (quote!(let #local=#object;), quote!(#local.as_raw()))
160        } else {
161            (TokenStream::new(), self.output_scalar(value))
162        };
163        JniArgument {
164            statements,
165            value: quote!(::prebindgen_jni_runtime::jni::sys::jvalue { #member:#value }),
166        }
167    }
168}

172    }
