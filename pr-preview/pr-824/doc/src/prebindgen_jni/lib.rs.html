157/// Generated Rust/JNI bindings and their Kotlin source files.
158///
159/// Start with [`Self::builder`]. Source capture, expansion declarations and package
160/// placement are resolved before wrapper emission. Data classes map supported
161/// record fields to Kotlin properties, including owning handle members in returns.
162/// Handle classes own native allocations.
163/// Selectors keep all alternatives
164/// in one native signature; multiple outputs use a Kotlin builder callback.
165///
166/// ```
167/// use prebindgen_jni::{JniGen, package, ptr_class, fun};
168/// let source = syn::parse_file(r#"
169///     pub struct Counter { pub value: i64 }
170///     pub fn new_counter(value: i64) -> Counter { Counter { value } }
171/// "#)?;
172/// let generated = JniGen::builder()
173///     .items(source.items.into_iter().map(|item| (item, Default::default())))
174///     .set_package_prefix("org.example")
175///     .package(package!()
176///         .class(ptr_class!(Counter))
177///         .fun(fun!(new_counter)))
178///     .build()?;
179/// assert!(generated.kotlin.contains_key("org/example/Counter.kt"));
180/// assert!(generated.kotlin["org/example/Functions.kt"].contains("newCounter"));
181/// syn::parse_file(&generated.rust)?;
182/// # Ok::<(), Box<dyn std::error::Error>>(())
183/// ```
184///
185/// In `build.rs`, replace `.items(...)` with `.source(flat_crate::PREBINDGEN_OUT_DIR)`,
186/// then write both outputs. Add the source crate as both a build dependency and
187/// a normal dependency. Include the Rust file in a binding crate depending on
188/// `prebindgen-runtime`, `prebindgen-jni-runtime` and `konst` (used by captured
189/// source feature guards). Compile all generated Kotlin
190/// files together. Calls must go through those wrappers: their locks, consumed
191/// state and ownership acknowledgement are part of the native safety contract.
192pub struct JniGen {
193    /// Formatted native Rust source, including conversion code and flat-model guards.
194    pub rust: String,
195    /// Kotlin sources keyed by package-relative paths such as `org/example/Counter.kt`.
196    pub kotlin: BTreeMap<String, String>,
197}

206///

229    /// Adjust the selected flat model before declaration defaults are applied.
230    ///
231    /// `f` receives the model and may attach converters to it.
232    ///
233    /// Use this to attach converters with [`prebindgen_declarations::bind_input`] and
234    /// [`prebindgen_declarations::bind_output`], using [`JniCodec`]. Callbacks run in
235    /// registration order; errors stop generation. Source selection and registration
236    /// of module settings, class declarations, function placement and `error_class`
237    /// policies have already happened. Callbacks run before class completion, error
238    /// preparation and converter filling; their placement edits are revalidated.
239    /// References introduced here must already exist in the selected source model.
240    /// Per-function declarations can explicitly override converters assigned here.
241    ///
242    /// Class and function packages are fully qualified at registration. Changing
243    /// `flat.metadata.package` changes native infrastructure and the fallback error
244    /// handler package; it does not relocate already declared classes or functions.
245    ///
246    /// ```
247    /// use prebindgen_jni::{Bindings, JniGen, JniCodec, fun, package};
248    /// use prebindgen_declarations::{bind_input, bind_output, Input, Output};
249    /// let source = syn::parse_file("pub fn echo(x: i64) -> i64 { x }")?;
250    /// let flat = Bindings::parse(
251    ///     source.items.into_iter().map(|item| (item, Default::default())),
252    /// )?;
253    /// let generated = JniGen::builder().flat(flat)
254    ///     .set_package_prefix("example")
255    ///     .configure(|flat| {
256    ///         bind_input(flat, &JniCodec, "echo", "x", &Input::Leaf, &())?;
257    ///         bind_output(flat, &JniCodec, "echo", &Output::Leaf, &())
258    ///     })
259    ///     .package(package!().fun(fun!(echo)))
260    ///     .build()?;
261    /// syn::parse_file(&generated.rust)?;
262    /// # Ok::<(), Box<dyn std::error::Error>>(())
263    /// ```
264    pub fn configure(mut self, f: impl FnOnce(&mut Bindings) -> Result<()> + 'static) -> Self {
265        self.configure.push(Box::new(f));
266        self
267    }

376    /// Resolve declarations and return a complete Flat binding model for emission.
377    ///
378    /// Rejects ambiguous source selection, invalid or duplicate names, unsupported
379    /// recipes, undeclared handle types and signatures exceeding JVM parameter limits.
380    /// Layout, Rust trait bounds and injected Kotlin syntax are checked by the
381    /// respective compilers when consuming the generated output.
382    pub fn prepare(mut self) -> Result<Bindings> {
383        for package in &self.packages {
384            let namespace = if package.name.is_empty() {
385                self.prefix.clone()
386            } else {
387                format!("{}.{}", self.prefix, package.name)
388            };
389            for class in &package.classes {
390                if let ClassDecl::Callback(declaration) = class {
391                    let mut declaration = declaration.clone();
392                    declaration.namespace = Some(namespace.clone());
393                    self.inputs.push(declaration.into());
394                }
395            }
396        }
397        let references = prebindgen_declarations::declarations::source_functions(
398            &self.inputs,
399            &self.outputs,
400            &self.error_mappers,
401            self.packages.iter().flat_map(|p| &p.functions),
402        );
403        let types: Vec<_> = self
404            .packages
405            .iter()
406            .flat_map(|p| p.classes.iter().map(|c| c.ty().clone()))
407            .chain(self.error_classes.iter().map(|(ty, _)| ty.clone()))
408            .chain(self.error_mappers.iter().map(|(ty, _)| ty.clone()))
409            .collect();
410        let (mut flat, mut callback_sources) = match (self.flat, self.source, self.items) {
411            (Some(flat), None, None) => (flat, Vec::new()),
412            (None, source, items) => {
413                let items = match (source, items) {
414                    (Some(path), None) => prebindgen::Source::new(path).items_all().collect(),
415                    (None, Some(items)) => items,
416                    _ => return Err("supply exactly one source, item stream or flat model".into()),
417                };
418                let callbacks = callback::source_parameters(&items);
419                (Flat::parse_selected(items, &references, &types)?, callbacks)
420            }
421            _ => return Err("supply exactly one source, item stream or flat model".into()),
422        };
423        validate_package(&self.prefix)?;
424        flat.metadata = JniModule {
425            package: self.prefix,
426            native_init: self.init,
427            callbacks: Vec::new(),
428        };
429        let mut declarations = Vec::new();
430        let mut class_types = BTreeSet::new();
431        let mut bound = BTreeSet::new();
432        let mut callback_types = vec![];
433        for p in self.packages {
434            let pkg = if p.name.is_empty() {
435                flat.metadata.package.clone()
436            } else {
437                format!("{}.{}", flat.metadata.package, p.name)
438            };
439            validate_package(&pkg)?;
440            for c in p.classes {
441                if let ClassDecl::Callback(declaration) = c {
442                    callback_types.push(declaration.ty);
443                    continue;
444                }
445                let ty = flat.resolve_type(c.ty())?;
446                let name = match c.name() {
447                    Some(n) => n.clone(),
448                    None => match c.ty() {
449                        syn::Type::Path(p) => p.path.segments.last().unwrap().ident.to_string(),
450                        _ => return Err("class needs a name".into()),
451                    },
452                };
453                validate_class_name(&name)?;
454                if !class_types.insert(type_key(&ty)) {
455                    return Err("duplicate class for source type".into());
456                }
457                let class = Class {
458                    pkg: pkg.clone(),
459                    name,
460                    data: matches!(c, ClassDecl::Data(_)),
461                    enumeration: matches!(c, ClassDecl::Enum(_)),
462                    has_handles: false,
463                    variant_names: match c {
464                        ClassDecl::Enum(c) => c.variant_names,
465                        _ => Vec::new(),
466                    },
467                };
468                let metadata = flat.type_metadata_mut(&ty)?;
469                metadata.class = Some(std::rc::Rc::new(class));
470                metadata.enumeration = None;
471            }
472            for f in p.functions {
473                let name = f.foreign.clone().unwrap_or_else(|| camel(&f.name));
474                validate_name(&name)?;
475                if !bound.insert(f.name.clone()) {
476                    return Err("binding one source function twice is outside this subset".into());
477                }
478                flat.function_mut(&f.name)?.metadata = Some(JniFunction {
479                    package: pkg.clone(),
480                    name,
481                    order: declarations.len(),
482                    error: None,
483                });
484                declarations.push(f);
485            }
486        }
487        for (ty, class) in self.error_classes {
488            let mut parts = class.split('$');
489            validate_package(parts.next().unwrap())?;
490            for nested in parts {
491                validate_name(nested)?;
492            }
493            let ty = flat.resolve_type(&ty)?;
494            if flat
495                .type_metadata_mut(&ty)?
496                .error_class
497                .replace(class)
498                .is_some()
499            {
500                return Err("duplicate exception class for source error type".into());
501            }
502        }
503        let codec = data::DataCodec::default();
504        for configure in self.configure {
505            configure(&mut flat)?;
506        }
507        callback_sources.extend(
508            flat.functions
509                .values()
510                .flat_map(|function| &function.params)
511                .filter(|parameter| matches!(&parameter.ty.ty, syn::Type::ImplTrait(_)))
512                .map(|parameter| parameter.ty.ty.clone()),
513        );
514        validate_package(&flat.metadata.package)?;
515        for declaration in &declarations {
516            if flat.function(&declaration.name)?.metadata.is_none() {
517                return Err(format!(
518                    "declared JNI function `{}` has no export metadata",
519                    declaration.name
520                ));
521            }
522        }
523        let output_defaults = callback::declared_outputs(&flat, &self.outputs)?;
524        let defaults = prebindgen_declarations::Declarations::new(
525            &flat,
526            self.inputs,
527            self.outputs,
528            self.error_mappers,
529        )?;
530        validate_declaration_names(&flat)?;
531        data::prepare(&mut flat)?;
532        errors::prepare(&mut flat, &codec, &defaults, &declarations)?;
533        let codec = data::DataCodec {
534            callback_outputs: Some(&output_defaults),
535        };
536        for ty in callback_types {
537            let recipe = defaults.resolve_input(&flat, &codec, &ty, &Input::auto(), &())?;
538            let Input::Callback(configuration) = recipe else {
539                return Err("callback declaration requires a callback conversion".into());
540            };
541            callback::validate_source_occurrence(&flat, &configuration, &callback_sources)?;
542            let plan = callback::input(
543                &flat,
544                &configuration,
545                &output_defaults,
546                &configuration.ty,
547                format_ident!("declared_callback"),
548                format_ident!("declared_callback_value"),
549                &Default::default(),
550            )?;
551            flat.metadata.callbacks.push(plan.into());
552        }
553        let mut next_order = declarations.len();
554        for declaration in declarations {
555            defaults.fill(&mut flat, &codec, &declaration, &())?;
556            if declaration.callback_return.is_some() {
557                let variant_name = format!("{}_fold", declaration.name);
558                sequence::callback_target(
559                    &(flat.function(&declaration.name)?.output_conversion()?)
560                        .output_outcomes()
561                        .success,
562                )?;
563                let variant = defaults.callback_variant(
564                    &mut flat,
565                    &codec,
566                    &declaration,
567                    &variant_name,
568                    &(),
569                )?;
570                let mut placement = flat
571                    .function(&declaration.name)?
572                    .metadata
573                    .as_ref()
574                    .ok_or("callback source has no JNI export metadata")?
575                    .clone();
576                placement.name = format!("{}Fold", placement.name);
577                let foreign = format!("{}.{}", placement.package, placement.name);
578                if foreign_name_exists(&flat, &foreign) {
579                    return Err("callback return foreign name collision".into());
580                }
581                placement.order = next_order;
582                next_order += 1;
583                flat.function_mut(&variant_name)?.metadata = Some(placement);
584                defaults.fill(&mut flat, &codec, &variant, &())?;
585                if declaration.callback_return
586                    == Some(prebindgen_declarations::CallbackReturn::Only)
587                {
588                    let original = flat.functions.get_mut(&declaration.name).unwrap();
589                    original.metadata = None;
590                    original.ret.slot = None;
591                    for param in &mut original.params {
592                        param.ty.slot = None;
593                    }
594                }
595            }
596        }
597        prebindgen_flat::validate_bindings(&flat)?;
598        for (name, f) in &flat.functions {
599            if f.ret.slot.is_some() && f.metadata.is_none() {
600                return Err(format!(
601                    "configured function `{name}` has no Kotlin declaration"

626/// Validate the declarations currently held by Flat, including configuration edits.
627fn validate_declaration_names(flat: &Bindings) -> Result<()> {
628    let mut names = BTreeSet::new();
629    for ty in flat.types.values() {
630        if let Some(class) = &ty.metadata.class {
631            validate_package(&class.pkg)?;
632            validate_name(&class.name)?;
633            let name = class.fqn();
634            if !names.insert(name.clone()) {
635                return Err(format!("duplicate foreign name `{name}`"));
636            }
637        }
638    }
639    for function in flat.functions.values() {
640        if let Some(m) = &function.metadata {
641            validate_package(&m.package)?;
642            validate_name(&m.name)?;
643            if !names.insert(format!("{}.{}", m.package, m.name)) {
644                return Err("duplicate foreign function name".into());
645            }
646        }
647    }
648    Ok(())
649}
650/// Check a prospective generated declaration against canonical class, function and handler names.
651fn foreign_name_exists(flat: &Bindings, name: &str) -> bool {
652    flat.types.values().any(|t| {
653        t.metadata.class.as_ref().is_some_and(|c| c.fqn() == name)
654            || t.metadata.errors.values().any(|e| e.handler == name)
655    }) || flat.functions.values().any(|f| {
656        f.metadata
657            .as_ref()
658            .is_some_and(|m| format!("{}.{}", m.package, m.name) == name)
659    })

814    fn input_leaf(
815        &self,
816        _flat: &prebindgen_flat::Flat<Self::Schema>,
817        ty: &syn::Type,
818        _: &(),
819        w: syn::Ident,
820        v: syn::Ident,
821        active: &prebindgen_declarations::Condition,
822    ) -> Result<InputPlan<JniWire>> {
823        if prebindgen_flat::sequence_parts(ty).is_some() {
824            return Err("collections require a Sequence recipe, not an opaque leaf; use an alias for opaque storage".into());
825        }
826        if prebindgen_declarations::option_inner(ty).is_some() {
827            return Err("resolve Optional operations before leaf compilation".into());
828        }
829        let env = prebindgen_conversions::template_parameter("environment");
830        let (decode, kind) = match Value::of(ty) {
831            Some(Value::Scalar(p)) => (scalar::decode(p, &w), JniWire::scalar(p)),
832            Some(Value::Text) => (
833                quote!(::prebindgen_jni_runtime::text(&mut #env,&#w)?),
834                JniWire::text(),
835            ),
836            Some(Value::Bytes) => (
837                quote!(::prebindgen_jni_runtime::bytes(&mut #env,&#w)?),
838                JniWire::bytes(),
839            ),
840            None => {
841                return prebindgen_conversions::boxed::input(
842                    &storage(),
843                    ty,
844                    w,
845                    v,
846                    &active.rust(),
847                    |b| JniWire::handle(b, kotlin_condition(active)),
848                )
849            }
850        };
851        prebindgen_conversions::value::input(ty, w, v, &active.rust(), decode, kind)
852    }
853    fn output_leaf(
854        &self,
855        _flat: &prebindgen_flat::Flat<Self::Schema>,
856        ty: &syn::Type,
857        _: &(),
858        w: syn::Ident,
859        v: TokenStream,
860        borrowed: bool,
861    ) -> Result<OutputPlan<JniWire>> {
862        if prebindgen_flat::sequence_parts(ty).is_some() {
863            return Err("collections require a Sequence recipe, not an opaque leaf; use an alias for opaque storage".into());
864        }
865        // Materialize JVM buffers while their source is still in scope. Optional
866        // payloads, getter temporaries and collection rows may be dropped before
867        // final delivery, so carrying a borrowed Rust view that far is unsound.
868        if let Some(value_kind @ (Value::Text | Value::Bytes)) = Value::of(ty) {
869            let env = prebindgen_conversions::template_parameter("environment");
870            let (encode, kind) = match value_kind {
871                Value::Text => (
872                    quote!(::prebindgen_jni_runtime::text_object(
873                        &mut #env, ::core::convert::AsRef::<str>::as_ref(&#w))?),
874                    JniWire::text(),
875                ),
876                Value::Bytes => (
877                    quote!(::prebindgen_jni_runtime::bytes_object(
878                        &mut #env, ::core::convert::AsRef::<[u8]>::as_ref(&#w))?),
879                    JniWire::bytes(),
880                ),
881                Value::Scalar(_) => unreachable!(),
882            };
883            return Ok(OutputPlan {
884                fallible: true,
885                context: Default::default(),
886                delivery: Default::default(),
887                outcomes: prebindgen_flat::Outcomes {
888                    success: prebindgen_flat::WireNode::Leaf(vec![prebindgen_flat::Wire {
889                        name: w.clone(),
890                        kind,
891                    }]),
892                    source_error: None,
893                    binding_failure: None,
894                },
895                statements: quote!(let #w: ::prebindgen_jni_runtime::jni::objects::JObject<'_> = {
896                    let #w = #v; #encode
897                };),
898                values: vec![(w.clone(), quote!(#w))],
899                commit: quote!(),
900            });
901        }
902        if let syn::Type::Reference(reference) = ty {
903            if reference.mutability.is_some() {
904                return Err("mutable-reference outputs are unsupported".into());
905            }
906            let value = if borrowed { quote!(*#v) } else { v };
907            return self.output_leaf(_flat, &reference.elem, &(), w, value, true);
908        }
909        if prebindgen_declarations::option_inner(ty).is_some() {
910            return Err("resolve Optional operations before leaf compilation".into());
911        }
912        match Value::of(ty) {
913            Some(Value::Scalar(p)) => prebindgen_conversions::value::output(
914                ty,
915                w,
916                v,
917                borrowed,
918                |v| scalar::encode(p, v),
919                JniWire::scalar(p),
920            ),
921            Some(Value::Text | Value::Bytes) => unreachable!("buffers prepared above"),
922            None => prebindgen_conversions::boxed::output(&storage(), ty, w, v, borrowed, |b| {
923                JniWire::handle(b, "true")
924            }),
925        }
926    }

977    }

979pub(crate) fn encode_array(
980    fields: Vec<(&prebindgen_flat::Wire<JniWire>, &TokenStream)>,
981    array: &syn::Ident,
982    offset: usize,
983) -> TokenStream {
984    let env = prebindgen_conversions::template_parameter("environment");
985    let mut encode = TokenStream::new();
986    for (i, (wire, value)) in fields.into_iter().enumerate() {
987        let expr = wire.kind.output_object(value);
988        let i = (i + offset) as i32;
989        encode.extend(quote! {
990            let object = #expr;
991            #env.set_object_array_element(&#array, #i, &object).map_err(|e|e.to_string())?;
992            #env.delete_local_ref(object).map_err(|e|e.to_string())?;
993        });
994    }
995
996    encode
997}

1012        let path = path.as_ref();

1014            path.to_owned()

1111        } else if upper {

1120/// JNI escaping is defined over UTF-16, including underscores in identifiers.
1121fn escape(s: &str) -> String {
1122    let mut out = String::new();
1123    for c in s.encode_utf16() {
1124        match c {
1125            46 | 47 => out.push('_'),
1126            95 => out.push_str("_1"),
1127            59 => out.push_str("_2"),
1128            91 => out.push_str("_3"),
1129            c if (c as u8).is_ascii_alphanumeric() && c < 128 => out.push(char::from(c as u8)),
1130            c => out.push_str(&format!("_0{c:04x}")),
1131        }
1132    }
1133    out
1134}

1143/// Emit Rust/JNI and Kotlin using only the prepared Flat model.
1144/// All project-dependent settings and declarations are read from its metadata.
1145pub fn generate(flat: &Bindings) -> Result<JniGen> {
1146    let prefix = &flat.metadata.package;
1147    validate_package(prefix)?;
1148    for f in flat.functions.values().filter(|f| f.ret.slot.is_some()) {
1149        let metadata = f
1150            .metadata
1151            .as_ref()
1152            .ok_or_else(|| format!("bound JNI function `{}` has no export metadata", f.name))?;
1153        validate_package(&metadata.package)?;
1154        validate_name(&metadata.name)?;
1155    }
1156    let init = &flat.metadata.native_init;
1157    let classes = flat
1158        .types
1159        .values()
1160        .filter_map(|t| {
1161            t.metadata
1162                .class
1163                .as_ref()
1164                .filter(|c| !c.data && !c.enumeration)
1165                .map(|c| (type_key(&t.ty), (&t.ty, c)))
1166        })
1167        .collect::<BTreeMap<_, _>>();
1168    let mut functions = flat
1169        .functions
1170        .values()
1171        .filter_map(|f| f.metadata.as_ref().map(|m| (&f.name, m)))
1172        .collect::<Vec<_>>();
1173    functions.sort_by_key(|(_, m)| m.order);
1174    let errors = flat
1175        .functions
1176        .iter()
1177        .filter_map(|(name, f)| {
1178            f.metadata
1179                .as_ref()
1180                .and_then(|m| m.error.as_deref())
1181                .map(|e| (name.clone(), e))
1182        })
1183        .collect::<BTreeMap<_, _>>();
1184    let mut files = BTreeMap::new();
1185    for t in flat.types.values() {
1186        if let Some(class) = &t.metadata.class {
1187            if class.data {
1188                files.insert(
1189                    format!("{}/{}.kt", class.pkg.replace('.', "/"), class.name),
1190                    data::render_record(flat, class, &t.ty)?,
1191                );
1192            }
1193            if class.enumeration {
1194                files.insert(
1195                    format!("{}/{}.kt", class.pkg.replace('.', "/"), class.name),
1196                    t.metadata
1197                        .enumeration
1198                        .as_deref()
1199                        .ok_or("missing prepared enum declaration")?
1200                        .render(class)?,
1201                );
1202            }
1203        }
1204    }
1205    let mut rust = errors::emit(&errors)?;
1206    let mut packages: BTreeMap<String, String> = BTreeMap::new();
1207    errors::interfaces(prefix, &mut files)?;
1208    let mut native = format!("package {prefix}\n\ninternal object JNINative {{\n");

1748        let domain = if let Some(error) = error {
1749            let mut delivery = errors::delivery(error, &kotlin_context)?;
1750            if native_return.is_some() {
1751                let fields =
1752                    crate::group::result_packet_fields(&error.output.output_outcomes().success)
1753                        .collect::<Vec<_>>();
1754                let values = (0..fields.len())
1755                    .map(|i| format!("returnCapture.d{i}"))
1756                    .collect::<Vec<_>>();
1757                delivery =
1758                    returns::remap_text(&delivery, kotlin_context.result, &fields, 1, &values);
1759                format!("if (returnCapture.status == 3) return@run run {{ {delivery} }}")
1760            } else {
1761                format!("true -> {{ {delivery} }}")
1762            }
1763        } else {
1764            String::new()
1765        };
1766        let callback_failure = if callback {
1767            format!("if (__pb_outputAccepted[0]) __pb_discard(IllegalArgumentException({result}[0] as String)); ")
1768        } else {
1769            String::new()
1770        };
1771        let native_call = if native_return.is_some() {
1772            let callback_discard = if callback {
1773                "if (__pb_outputAccepted[0]) __pb_discard(IllegalArgumentException(returnCapture.message));"
1774            } else {
1775                ""
1776            };
1777            format!("val nativeResult = {call}\n    if (returnCapture.status == 2) {{ {callback_discard}{discard_success}\n        return@run {binding_name}.run(returnCapture.message) }}\n    {domain}\n    {end}")
1778        } else {
1779            format!("val {result} = {call}\n    when ({result}[0]) {{\n        null -> {{ {end} }}\n        {domain}\n        is String -> {{ {callback_failure}{binding_name}.run({result}[0] as String) }}\n        else -> error(\"invalid native result channel\")\n    }}")
1780        };
1781        let native_call = if let Some(name) = &capture_name {
1782            format!("val returnCapture = {prefix}.{name}.acquire()\n    try {{\n{native_call}\n    }} finally {{ returnCapture.release() }}")
1783        } else {
1784            native_call
1785        };
1786        let native_call = native_call
1787            .lines()
1788            .map(str::trim_end)
1789            .collect::<Vec<_>>()
1790            .join("\n");
1791        let text = packages.entry(pkg.clone()).or_default();
1792        text.push_str(&format!(
1793            "{throws}public fun {generic}`{name}`({}): {ret} = run {{\n    {setup}{native_call}\n}}\n\n",
1794            kt_params.join(", ")
1795        ));
1796    }
1797    native.push_str("}\n");
1798    files.insert(format!("{}/JNINative.kt", prefix.replace('.', "/")), native);
1799    for (pkg, body) in packages {
1800        let body = body.trim_end();
1801        let path = format!("{}/Functions.kt", pkg.replace('.', "/"));
1802        if files
1803            .insert(
1804                path,
1805                format!("@file:Suppress(\"UNUSED_VARIABLE\", \"UNCHECKED_CAST\", \"UNNECESSARY_NOT_NULL_ASSERTION\")\npackage {pkg}\n\n{body}\n"),
1806            )
1807            .is_some()
1808        {
1809            return Err("Kotlin file collision".into());
1810        }
1811    }
1812    let guards = &flat.guards;
1813    rust.extend(quote!(#(#guards)*));
1814    Ok(JniGen {
1815        rust: prebindgen_flat::rust_source(rust)?,
1816        kotlin: files,
1817    })
1818}
