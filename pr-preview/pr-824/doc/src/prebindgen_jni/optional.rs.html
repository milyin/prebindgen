3pub(crate) fn absent(w: &JniWire, niche: Option<&JniNiche>, input: bool) -> Result<TokenStream> {
4    // Prepared text/bytes already are JVM references. Null means absent;
5    // an empty buffer is a non-null JVM object, with no Rust Option wrapper.
6    if matches!(w, JniWire::Text { .. } | JniWire::Bytes { .. }) {
7        return Ok(if input {
8            quote!(Default::default())
9        } else {
10            quote!(::prebindgen_jni_runtime::jni::objects::JObject::null())
11        });
12    }
13    if matches!(w, JniWire::Rows { .. }) {
14        return Ok(if input {
15            quote!(Default::default())
16        } else {
17            quote!(None)
18        });
19    }
20    if !input && matches!(w, JniWire::Handle { .. }) {
21        return Ok(quote!(std::ptr::null_mut()));
22    }
23    if let Some(n) = niche {
24        return Ok(n.integer_value());
25    }
26    Ok(match w {
27        JniWire::Scalar {
28            primitive: Primitive::F32 | Primitive::F64,
29            ..
30        } => quote!(0.0),
31        JniWire::Scalar {
32            primitive: Primitive::I128 | Primitive::U128,
33            ..
34        } if !input => quote!([0u8; 16]),
35        JniWire::Scalar {
36            primitive: Primitive::I128 | Primitive::U128,
37            ..
38        } => {
39            quote!(Default::default())
40        }
41        _ => quote!(0),
42    })
43}
44pub(crate) fn default_kt(w: &JniWire, niche: Option<&JniNiche>) -> String {
45    if matches!(niche, Some(JniNiche::Null))
46        && matches!(
47            w,
48            JniWire::Rows { .. } | JniWire::Text { .. } | JniWire::Bytes { .. }
49        )
50    {
51        return "null".into();
52    }
53    if let Some(n) = niche {
54        let v = match n {
55            JniNiche::Null => 0,
56            JniNiche::Integer(n) => *n,
57        };
58        return w.kotlin_integer(v);
59    }
60    w.kotlin_default().into()
61}
62// Replace generated identifiers only, not substrings of other generated names.
63pub(crate) fn replace(s: &str, from: &str, to: &str) -> String {
64    let mut out = String::new();
65    let mut word = String::new();
66    for c in s.chars().chain(std::iter::once(' ')) {
67        if c.is_ascii_alphanumeric() || c == '_' {
68            word.push(c)
69        } else {
70            if word == from {
71                out.push_str(to)
72            } else {
73                out.push_str(&word)
74            }
75            word.clear();
76            out.push(c);
77        }
78    }
79    out.pop();
80    out
81}
82pub(crate) fn input_kotlin(
83    nodes: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
84    o: &JniOptional,
85    context: &KotlinContext<'_>,
86) -> Result<KotlinInput> {
87    let root = &o.presence().to_string();
88    let inner = kotlin::leaf_wires(nodes);
89    let mut out = KotlinInput::default();
90    let mut c = kotlin::input_kotlin(nodes, context)?;
91    // A single converted value is idiomatic T?. Multiple views retain an
92    // explicit presence argument so Some with nullable components is distinct.
93    let presence = if c.parameters.len() == 1 && !kotlin::has_optional(nodes) {
94        let (name, ty) = c.parameters[0].split_once(": ").unwrap();
95        let name = name.to_string();
96        let ty = ty.trim_end_matches('?');
97        out.parameters.push(format!("{name}: {ty}?"));
98        for a in c
99            .arguments
100            .iter_mut()
101            .chain(c.locks.iter_mut())
102            .chain(c.consumed.iter_mut())
103            .chain(c.lock_groups.iter_mut())
104            .chain(c.setup.iter_mut().map(|b| &mut b.expression))
105        {
106            *a = a
107                .replace(&format!("{name}?.raw ?: 0L"), &format!("{name}.raw"))
108                .replace(&format!("{name}?."), &format!("{name}."))
109                .replace(&format!("{name}!!"), &name);
110            *a = replace(a, &name, &format!("requireNotNull({name})"));
111        }
112        format!("{name} != null")
113    } else {
114        out.parameters.push(format!("{}: Boolean", root));
115        out.parameters.extend(c.parameters.clone());
116        root.to_string()
117    };
118    let tag = format!("(if ({presence}) 1 else 0)");
119    for a in c
120        .arguments
121        .iter_mut()
122        .chain(c.locks.iter_mut())
123        .chain(c.consumed.iter_mut())
124        .chain(c.lock_groups.iter_mut())
125        .chain(c.setup.iter_mut().map(|b| &mut b.expression))
126    {
127        // Every payload expression below runs only in the present branch.
128        *a = a.replace(&format!("{} == 1", root), "true");
129        *a = replace(a, root, "1");
130    }
131    if matches!(o, JniOptional::Flag(_)) {
132        out.arguments.push(tag);
133    }
134    for (j, a) in c.arguments.into_iter().enumerate() {
135        let niche = o.niche().filter(|_| inner[j].name == *o.discriminator());
136        let default = default_kt(&inner[j].kind, niche);
137        let a = match niche {
138            Some(JniNiche::Integer(_)) => format!("run {{ val value = {a}; require(value != {default}) {{ \"invalid option payload\" }}; value }}"),
139            Some(JniNiche::Null) if matches!(inner[j].kind, JniWire::Rows { .. } | JniWire::Text { .. } | JniWire::Bytes { .. }) => {
140                // An explicit presence parameter must not silently become None
141                // when a nullable raw reference is passed for its discriminator.
142                format!("requireNotNull({a}) {{ \"invalid option payload\" }}")
143            }
144            _ => a,
145        };
146        out.arguments
147            .push(format!("if ({presence}) {a} else {default}"));
148    }
149    out.setup.extend(c.setup.into_iter().map(|mut b| {
150        b.expression = format!("if ({presence}) ({}) else ({})", b.expression, b.absent);
151        b
152    }));
153    out.lock_groups.extend(
154        c.lock_groups
155            .into_iter()
156            .map(|s| format!("if ({presence}) ({s}) else kotlin.collections.emptyList()")),
157    );
158    out.locks.extend(
159        c.locks
160            .into_iter()
161            .map(|s| format!("if ({presence}) {s} else null")),
162    );
163    out.consumed.extend(
164        c.consumed
165            .into_iter()
166            .map(|s| format!("if ({presence}) {{ {s} }}")),
167    );
168    Ok(out)
169}
170pub(crate) fn output_kotlin(
171    nodes: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
172    o: &JniOptional,
173    context: &KotlinContext<'_>,
174) -> Result<KotlinOutput> {
175    let wires = kotlin::leaf_wires(nodes);
176    let flag = usize::from(matches!(o, JniOptional::Flag(_)));
177    let p = if let Some(n) = o.niche() {
178        let i = wires
179            .iter()
180            .position(|w| w.name == *o.discriminator())
181            .ok_or("missing optional discriminator")?;
182        let absent = default_kt(&wires[i].kind, Some(n));
183        format!("{}[{i}] != {absent}", context.result)
184    } else {
185        format!("{}[0] == 1", context.result)
186    };
187    let local = format!("{}_payload", context.result);
188    let ctx = KotlinContext {
189        result: &local,
190        ..*context
191    };
192    let mut c = kotlin::output_kotlin(nodes, &ctx)?;
193    kotlin::remap_output(&mut c, &local, context.result, flag, wires.len());
194    let mut out = KotlinOutput {
195        output_locals: c.output_locals,
196        ..Default::default()
197    };
198    if c.delivered.is_empty() || kotlin::has_optional(nodes) {
199        out.delivered.push(p.clone());
200        out.result_types.push("Boolean".into());
201        out.result_names.push(None);
202    }
203    out.materialize = format!("            if ({p}) {{\n{}            }}\n", c.materialize);
204    out.cleanup = format!("            if ({p}) {{\n{}            }}\n", c.cleanup);
205    c.result_names.resize(c.result_types.len(), None);
206    out.result_names.extend(c.result_names);
207    for (v, t) in c.delivered.into_iter().zip(c.result_types) {
208        out.delivered.push(format!("if ({p}) {v} else null"));
209        out.result_types
210            .push(format!("{}?", t.trim_end_matches('?')));
211    }
212    Ok(out)
213}

215/// A wire value excluded from the representation of a present value.
216#[derive(Clone, Debug, PartialEq, Eq)]
217pub enum JniNiche {
218    /// Null reference or address (including integer-encoded handles).
219    Null,
220    /// An invalid integer representation, within the wire's integer range.
221    Integer(i64),
222}
223impl JniNiche {
224    /// Expression recognizing the present case without decoding the payload.
225    pub(crate) fn present(&self, wire: &JniWire, value: TokenStream) -> TokenStream {
226        match self {
227            Self::Null
228                if matches!(
229                    wire,
230                    JniWire::Rows { .. } | JniWire::Text { .. } | JniWire::Bytes { .. }
231                ) =>
232            {
233                quote!(!#value.is_null())
234            }
235            Self::Null => quote!((#value as usize) != 0),
236            Self::Integer(n) => {
237                let n = syn::parse_str::<syn::Expr>(&n.to_string()).unwrap();
238                quote!(#value != #n)
239            }
240        }
241    }
242    /// Absence expression for a numeric wire. Pointer codecs supply a null pointer.
243    pub(crate) fn integer_value(&self) -> TokenStream {
244        match self {
245            Self::Null => quote!(0),
246            Self::Integer(n) => syn::parse_str::<TokenStream>(&n.to_string()).unwrap(),
247        }
248    }
249}

251pub(crate) fn remaining(wire: &JniWire, niche: &JniNiche) -> Option<JniNiche> {
252    wire.validate_niche(niche).ok()?;
253    let JniNiche::Integer(n) = niche else {
254        return None;
255    };
256    if let JniWire::Enum { enum_type, .. } = wire {
257        return i32::try_from(*n)
258            .ok()?
259            .checked_sub(1)
260            .and_then(|n| enum_type.mapping.unused_at_or_below(n))
261            .map(|n| JniNiche::Integer(n.into()));
262    }
263    let next = JniNiche::Integer(n.checked_sub(1)?);
264    // Stop at the carrier boundary or the next valid payload value. Do not
265    // advertise a reserved range's neighbour merely because it fits an integer.
266    wire.validate_niche(&next).ok().map(|()| next)
267}
