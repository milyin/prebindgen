1//! Bindings supplied by this language generator during slot emission.
2use prebindgen_conversions::{render_template, TemplateContext};
3use prebindgen_flat::Result;
4use proc_macro2::TokenStream;
5use quote::ToTokens;
6
7/// Concrete wrapper bindings used by this language generator.
8#[derive(Clone, Debug)]
9pub struct JniContext {
10    /// Incoming ownership reporting selected by the wrapper boundary.
11    pub input_transfers: prebindgen_conversions::InputTransfers,
12    /// Explicit function-return transport. Source callback argument scopes use Packet.
13    pub return_delivery: crate::JniReturnDelivery,
14    /// Delivery prepared once for the active result converter, if adapter-selected.
15    pub prepared_delivery: Option<prebindgen_conversions::Delivery>,
16    /// Mutable JNI environment parameter.
17    pub environment: syn::Ident,
18    /// Local retaining a committed domain-error packet or delivery marker across Rust unwinding.
19    pub domain_output: syn::Ident,
20    /// Borrowed synchronous function-return capture. None preserves custom
21    /// packet delivery and source-callback argument conversion.
22    pub return_sink: Option<syn::Ident>,
23    /// Whether this delivery belongs to the typed domain-error channel.
24    pub is_domain_error: bool,
25    /// Infallible statements recording completed output ownership transfer.
26    pub output_committed: TokenStream,
27}
28impl TemplateContext for JniContext {
29    fn render(&self, code: &TokenStream) -> Result<TokenStream> {
30        // Generated Kotlin validates aliases while acquiring handle locks.
31        render_template(
32            code,
33            &|name| match name {
34                "environment" => Ok(self.environment.to_token_stream()),
35                "domain_output" => Ok(self.domain_output.to_token_stream()),
36                "return_sink" => Ok(self
37                    .return_sink
38                    .as_ref()
39                    .map_or_else(|| quote::quote!(None), |sink| quote::quote!(Some(&#sink)))),
40                "mark_output" => {
41                    let env = &self.environment;
42                    Ok(if self.is_domain_error {
43                        quote::quote!(::prebindgen_jni_runtime::mark_domain_packet(&mut #env, &result)?;)
44                    } else {
45                        TokenStream::new()
46                    })
47                }
48                "output_committed" => Ok(self.output_committed.clone()),
49                name if name.starts_with("bulk_read_") => {
50                    Ok(quote::quote!(::prebindgen_jni_runtime::bulk::input))
51                }
52                _ => Err(format!("unknown emission parameter `{name}`")),
53            },
54            &|count| self.input_transfers.render(count),
55        )
56    }
57}
58
59impl JniContext {
60    /// Use the shared callback boundary's prepared bulk carriers in native input
61    /// validation. The code uses the standard bulk-read template markers and
62    /// the same wire tree as the callback's input converter.
63    pub fn cached_callback_validation(
64        &self,
65        validation: &TokenStream,
66        node: &prebindgen_flat::WireNode<crate::JniWire, crate::JniChain, crate::JniBranch>,
67    ) -> Result<TokenStream> {
68        crate::bulk::cached_reads(validation, node)
69    }
70    /// Prepare native result delivery from the successful wire expressions.
71    /// This is the same operation used by built-in output plans. Values may
72    /// refer to source_result_expr; the output fragment resolves that binding.
73    pub fn output_delivery(
74        &self,
75        outcomes: &prebindgen_flat::Outcomes<
76            prebindgen_flat::WireNode<crate::JniWire, crate::JniChain, crate::JniBranch>,
77        >,
78        values: &[(syn::Ident, TokenStream)],
79    ) -> Result<prebindgen_conversions::Delivery> {
80        let node = &outcomes.success;
81        match self.return_delivery {
82            crate::JniReturnDelivery::Packet => crate::returns::packet_delivery(node, values),
83            crate::JniReturnDelivery::Capture => crate::returns::callback_delivery(
84                node,
85                values,
86                self.return_sink
87                    .as_ref()
88                    .ok_or("capture delivery requires a return sink")?,
89                self.is_domain_error,
90            ),
91            crate::JniReturnDelivery::Direct => {
92                if self.is_domain_error {
93                    return Err("domain errors require capture or packet delivery".into());
94                }
95                if crate::group::result_packet_fields(node).len() > 1 {
96                    return Err("direct delivery requires at most one wire".into());
97                }
98                crate::returns::delivery(
99                    node,
100                    values,
101                    self.return_sink
102                        .as_ref()
103                        .ok_or("direct delivery requires an error capture")?,
104                    false,
105                )
106            }
107        }
108    }
109}
