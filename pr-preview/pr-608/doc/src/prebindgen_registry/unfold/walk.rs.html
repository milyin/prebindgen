1//! The walk over a decomposed value.
2//!
3//! A decomposition — an [`UnfoldPlan`](crate::unfold::UnfoldPlan) — says which
4//! **leaves** a value is delivered as and which **hoists** (intermediate value
5//! forms) they are reached through. Performing that walk is the registry's:
6//! binding each hoist once, reaching each leaf through its own path, and
7//! deciding at every step whether what it holds is owned or borrowed.
8//!
9//! What an adapter supplies is the target-language half — how a leaf is
10//! encoded, and how the delivered values are handed over — plus the two facts
11//! this module cannot know: where a source item is qualified from
12//! (`qualify`), and what each leaf is ([`DecomposedLeaf`]).
13
14use prebindgen_flat::flat::TypeRef;
15use proc_macro2::TokenStream;
16use quote::{format_ident, quote};
17
18use crate::unfold::{steps_are_movable, PathStep};

20/// What the walk needs to know about one delivered leaf.
21///
22/// The adapter's own leaf carries far more — its wire type, its JNI or C
23/// encoding, its projection — and none of that reaches here.
24pub trait DecomposedLeaf {
25    /// Steps from the decomposed value to this leaf.
26    fn reach(&self) -> &[PathStep];
27
28    /// The leaf's own name, used in diagnostics and in local names.
29    fn name(&self) -> &str;
30
31    /// Whether this leaf *is* the decomposed value rather than a part of it.
32    fn identity(&self) -> bool;
33
34    /// The reading the leaf delivers.
35    fn source(&self) -> &TypeRef;
36
37    /// Whether the last step reaching it is a field read rather than a call.
38    ///
39    /// Derived from [`Self::reach`], so no implementation can disagree with
40    /// its own path — and the answer decides whether the reached place is
41    /// cloned, which is an ownership decision rather than a spelling.
42    fn is_field_read(&self) -> bool {
43        matches!(self.reach().last(), Some(PathStep::Field { .. }))
44    }
45}

47/// A leaf's reach, rendered **around** an encoding body: the walk hands the
48/// body the reached Rust expression and gets back the encoded value, so an
49/// absent value short-circuits to the adapter's own absence rather than to one
50/// the walk chose. `dyn` on both halves because each side is written where it
51/// is known and neither can name the other's closure.
52pub type Reach<'a> = dyn Fn(&dyn Fn(TokenStream) -> TokenStream) -> TokenStream + 'a;
53
54/// The adapter's half of a delivery.
55///
56/// The walk in this module owns how a decomposed value is reached — which
57/// hoist a leaf sits under, what is owned, what is borrowed. What a leaf
58/// becomes in the target language, and how the delivered values are handed to
59/// the call, is the adapter's, and this is where it says so. Stated over
60/// [`DecomposedLeaf`] rather than over an adapter's own leaf type, so the walk
61/// can call it.
62pub trait DeliveryBridge {
63    /// The adapter's leaf.
64    type Leaf: DecomposedLeaf;
65
66    /// Where the source item declaring `ident` is qualified from — the one
67    /// fact about a captured path this crate cannot answer for itself.
68    fn qualify(&self, ident: &syn::Ident) -> syn::Path;
69
70    /// The sum a selector leaf names: the qualified source path of the type,
71    /// and its Flat shape.
72    fn sum(&self, leaf: &Self::Leaf) -> (syn::Path, &prebindgen_flat::flat::Variant);
73
74    /// Encode one leaf into the value the delivery call receives, and bind
75    /// that value to `slot`.
76    ///
77    /// The adapter emits the binding rather than the walk, because the slot's
78    /// type is the adapter's: one leaf may cross as a scalar and the next as a
79    /// reference. `index` is the leaf's position in the delivery, which is
80    /// what any temporary the encode needs is named from. An encoding that can
81    /// fail routes its failure through `fail`.
82    fn encode(
83        &self,
84        leaf: &Self::Leaf,
85        index: usize,
86        slot: &syn::Ident,
87        reach: &Reach<'_>,
88        fail: &dyn Fn(TokenStream) -> TokenStream,
89        emit: &crate::RustWriter,
90    ) -> TokenStream;
91
92    /// The expression that passes `slot` as one argument of the delivery call.
93    fn argument(&self, leaf: &Self::Leaf, slot: &syn::Ident) -> TokenStream;
94}

96/// Compose one [`PathStep`] onto the reference expression reached so far.
97/// A `Call` applies its accessor (origin-qualified); a `Field` reads the field
98/// and re-borrows, so the result is a reference either way and steps chain
99/// uniformly.
100pub fn compose_step(
101    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
102    step: &PathStep,
103    e: TokenStream,
104) -> TokenStream {
105    match step {
106        PathStep::Call { ident, .. } => {
107            let m = qualify(ident);
108            quote!(#m::#ident(#e))
109        }
110        PathStep::Field { ident, .. } => quote!(&(#e).#ident),
111    }
112}
113/// Fold a run of steps onto `e`, borrowing wherever ownership demands it.
114///
115/// [`compose_step`] hands a `Call` its receiver as written, and an accessor
116/// takes that receiver **by reference** — so an owned value in hand has to be
117/// borrowed before the next call composes onto it. A value is in hand whenever
118/// the previous step returned one (`f(..) -> T` rather than `-> &T`), which is
119/// what [`PathStep::yields_owned`] records; `owned` says whether `e` itself
120/// started that way.
121///
122/// A `Field` step needs no borrow either way — it composes as `&(e).f`, which
123/// reads through a value and a reference alike.
124///
125/// This is the ONE place the rule lives, so every fold — a leaf's reach, a
126/// conditional hoist's prefix, a sum's matched value — answers it the same way.
127/// Splitting it produced exactly the bug it exists to prevent: ownership was
128/// handled at the optional binding and at the value form, and lost at every
129/// ordinary call in between.
130pub fn fold_steps(
131    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
132    steps: &[PathStep],
133    mut e: TokenStream,
134    mut owned: bool,
135) -> TokenStream {
136    for step in steps {
137        if owned && matches!(step, PathStep::Call { .. }) {
138            e = quote!(&#e);
139        }
140        e = compose_step(qualify, step, e);
141        owned = step.yields_owned();
142    }
143    e
144}
145/// Compose a value form's OWN call — the one step in the whole system whose
146/// receiver may be by value.
147///
148/// Four cases, from what the fold ended up holding crossed with what the
149/// accessor takes. A CONSUMING form takes its receiver by value: hand it the
150/// value when that is ours, clone when it is not — the same cost the borrowing
151/// form of the accessor would have paid, which keeps one declaration usable by
152/// both owned and `&T` roots. A borrowing form takes a reference, so an owned
153/// value is borrowed for it.
154///
155/// The decision is made from the fold's RESULT, never from where the fold
156/// began: that is what lets a consuming form sit behind ordinary accessors,
157/// where the chain in front borrows and the form itself still moves.
158///
159/// One function because both hoist paths — the conditional binding and the
160/// ordinary one — need exactly this rule, and stating it twice is what turned
161/// each new shape into a new defect.
162fn compose_value_form_call(
163    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
164    call: &PathStep,
165    e: TokenStream,
166    e_owned: bool,
167    consuming: bool,
168) -> TokenStream {
169    match (consuming, e_owned) {
170        (true, owned) => {
171            let (m, f) = (qualify(call.ident()), call.ident());
172            // Parenthesized: the clone applies to whatever the fold holds, and
173            // `&x.clone()` would parse as `&(x.clone())`.
174            let arg = if owned { e } else { quote!((#e).clone()) };
175            quote!(#m::#f(#arg))
176        }
177        (false, true) => compose_step(qualify, call, quote!(&#e)),
178        (false, false) => compose_step(qualify, call, e),
179    }
180}
181/// Start a reach from `base`, projecting the **leading run of plain field
182/// steps** directly (`&base.a.b`) instead of through a borrow of the base
183/// (`&(&base).a.b`). Returns the expression and how many steps it consumed.
184///
185/// The two forms name the same value, but the second borrows the base **as a
186/// whole**, which the borrow checker rejects once a sibling leaf has moved a
187/// different field out of it. Projecting directly makes each leaf's borrow
188/// disjoint, so a consuming value form's field moves are order-independent
189/// rather than compiling only while the borrowing leaves happen to be declared
190/// first.
191pub fn project_leading_fields(
192    base: &TokenStream,
193    base_is_ref: bool,
194    path: &[PathStep],
195) -> (TokenStream, usize) {
196    if base_is_ref {
197        return (base.clone(), 0);
198    }
199    let n = path.iter().take_while(|s| s.is_plain_field()).count();
200    if n == 0 {
201        return (quote!(&#base), 0);
202    }
203    let segs: Vec<&syn::Ident> = path[..n].iter().map(PathStep::ident).collect();
204    (quote!(&#base #(.#segs)*), n)
205}

207/// Fold a leaf's whole `path` over `base` with no optional-step handling, then
208/// apply the terminal treatment the leaf calls for: a leaf reached by a field
209/// read is **cloned** out of the place it reached, because its converter takes
210/// the field type as written (owned); every other leaf keeps the borrow its
211/// converter expects.
212///
213/// One derivation, used by every delivery an adapter renders and by the
214/// single-leaf shortcut a decomposed return takes. They drifted once while
215/// they were two — the shortcut was missing the field clone and handed `&F` to
216/// an `F` converter — which is why the walk is the registry's.
217pub fn reach_leaf_flat<L: DecomposedLeaf>(
218    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
219    leaf: &L,
220    path: &[PathStep],
221    base: TokenStream,
222    base_is_ref: bool,
223    consuming: bool,
224) -> TokenStream {
225    // An optional step BEFORE the last one needs a `match` whose `None` arm has
226    // somewhere to go. This derivation has none — it yields a plain Rust value,
227    // not a representation that can carry absence — so the shape is refused
228    // here rather than composed into code that cannot type-check in the
229    // consumer's crate.
230    //
231    // Asked of the leaf's OWN path, not of `path`. The caller may hand a
232    // suffix: `wrapper.rs` rebases onto a hoisted local, and `Hoisted::innermost`
233    // strips the prefix that bound it — including any optional step inside it.
234    // Checking the parameter would therefore pass exactly when the hoist is the
235    // conditional one, which is the case that cannot compose (an `Option<T>`
236    // local with a field read hung off it). The full path is what the shape
237    // question is about.
238    let own_path = leaf.reach();
239    assert!(
240        !own_path.iter().rev().skip(1).any(PathStep::is_optional),
241        "unfold: leaf `{}` reaches through an optional step but is \
242         delivered as a single return value, which has no `None` arm — this \
243         shape needs callback delivery",
244        leaf.name(),
245    );
246    // Whether what this leaf reaches is OURS, and so is moved rather than
247    // borrowed or cloned. The two leaf kinds say it differently:
248    //
249    // * an IDENTITY leaf carries the answer in its `out_ty` — the plan resolved
250    //   it to the owned type exactly when the value is the plan's to give away
251    //   (`place_is_owned`: an owned root, or a field of a CONSUMING value form),
252    //   and that is also what selected the owning converter, which boxes the
253    //   move rather than cloning a borrow;
254    // * a FIELD leaf's `out_ty` is the field type as written, owned either way,
255    //   so ownership is the enclosing form's: only a consuming one gives its
256    //   fields away.
257    //
258    // How to project that place is `steps_are_movable`'s question, and it is
259    // asked there rather than restated here. This used to spell it
260    // `all(is_plain_field)`, defending the restatement on the grounds that a
261    // trailing `Option` cannot reach return delivery anyway — true, and enforced
262    // in `single_return` (`core/unfold.rs`), which is precisely why a local
263    // restatement could disagree with the rule for as long as the invariant held
264    // somewhere else. `plan.rs` says two readings would drift and the
265    // disagreement would be a borrow handed to an owning converter; this is the
266    // second reading, removed.
267    let reached_is_ours = if leaf.identity() {
268        !matches!(
269            leaf.source().kind(),
270            prebindgen_flat::flat::TypeKind::Ref { .. }
271        )
272    } else {
273        consuming
274    };
275    if reached_is_ours && steps_are_movable(path) {
276        let segs: Vec<&syn::Ident> = path.iter().map(PathStep::ident).collect();
277        return quote!(#base #(.#segs)*);
278    }
279    let (e, lead) = project_leading_fields(&base, base_is_ref, path);
280    let e = fold_steps(qualify, &path[lead..], e, false);
281    if leaf.is_field_read() {
282        quote!((#e).clone())
283    } else {
284        e
285    }
286}
287/// Every value form on a plan, evaluated **once** and bound to a local
288/// (`__vf0`, `__vf1`, …), so a struct is built once per delivery rather than
289/// once per field. The bound prefixes come back with the statements, since
290/// reaching a leaf means starting from the innermost local it sits under.
291///
292/// Shared by both delivery paths — the multi-leaf encoder below and the
293/// single-leaf `Delivery::Return` shortcut in `emit/wrapper.rs`. The shortcut
294/// used to compose its reach straight off the raw value, which for a consuming
295/// value form emitted `f(&v)` against a by-value receiver: ill-typed Rust in
296/// the consumer's crate. One binder, so the two cannot disagree about what a
297/// hoist is or who owns it.
298pub struct Hoisted {
299    /// The `let __vfN = …;` bindings, outermost-first.
300    pub stmts: TokenStream,
301    /// Each hoist's path prefix and the local it was bound to.
302    bound: Vec<(Vec<PathStep>, syn::Ident)>,
303    /// Whether each bound hoist consumed the value it decomposed.
304    consuming: Vec<bool>,
305    /// Whether each bound local is `Option<TStruct>` rather than `TStruct` —
306    /// the hoist sits under an optional step, so the value form ran only where
307    /// the value was present. Its leaves cannot be emitted as independent
308    /// statements: they share ONE `match` on the local (see
309    /// [`encode_plan_leaves`]), taken by value, so a consuming form's fields
310    /// still move out inside the arm.
311    optional: Vec<bool>,
312}
313impl Hoisted {
314    /// Index of the innermost bound hoist `path` sits under, with that prefix
315    /// already consumed. `None` for a leaf under no value form at all — a
316    /// sibling `.field()` / `.field_self()`, which still reaches from the value
317    /// itself.
318    fn innermost(&self, path: &[PathStep]) -> Option<(usize, Vec<PathStep>)> {
319        self.bound
320            .iter()
321            .enumerate()
322            .filter(|(_, (p, _))| p.len() < path.len() && path.starts_with(p))
323            .max_by_key(|(_, (p, _))| p.len())
324            .map(|(i, (p, _))| (i, path[p.len()..].to_vec()))
325    }
326
327    /// The innermost bound local `path` sits under, with that prefix already
328    /// consumed, and whether that hoist gave its value away.
329    pub fn rebase(&self, path: &[PathStep]) -> Option<(syn::Ident, Vec<PathStep>, bool)> {
330        self.innermost(path)
331            .map(|(i, rest)| (self.bound[i].1.clone(), rest, self.consuming[i]))
332    }
333
334    /// The innermost **conditional** hoist `path` sits under: its index, the
335    /// local holding the `Option`, the name its `Some` arm binds, and the steps
336    /// left to reach the leaf from there. `None` when the leaf's innermost
337    /// hoist is unconditional (or there is none) — then [`Self::rebase`]
338    /// applies and the leaf is an ordinary standalone statement.
339    pub fn conditional(
340        &self,
341        path: &[PathStep],
342    ) -> Option<(usize, syn::Ident, syn::Ident, Vec<PathStep>)> {
343        let (i, rest) = self.innermost(path)?;
344        self.optional[i].then(|| (i, self.bound[i].1.clone(), format_ident!("__u{}", i), rest))
345    }
346
347    /// The local a hoist was bound to.
348    pub fn local(&self, i: usize) -> syn::Ident {
349        self.bound[i].1.clone()
350    }
351
352    /// Whether a hoist consumed the value it decomposed.
353    pub fn consumed(&self, i: usize) -> bool {
354        self.consuming[i]
355    }
356}
357/// Fold `path` over `base` the way an adapter's own gated reach does, but
358/// yielding an `Option<…>` rather than the adapter's absent value: the optional
359/// steps become a `map`/`and_then` chain, so an absent value short-circuits to
360/// `None` instead of to whatever that adapter uses for absence. `body` renders
361/// the innermost reached expression as a BARE value — the chain's last link
362/// wraps it.
363///
364/// The gated reach this mirrors is not in this crate. JniGen keeps its own, as
365/// `emit::delivery::reach_leaf`, because the absent value is its
366/// representation's — which is one of the walks this module was meant to end.
367///
368/// This is how a CONDITIONAL value form is bound — the accessor runs only where
369/// the value it decomposes is actually present.
370fn reach_optional(
371    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
372    path: &[PathStep],
373    base: TokenStream,
374    base_is_ref: bool,
375    depth: usize,
376    body: &dyn Fn(TokenStream) -> TokenStream,
377) -> TokenStream {
378    let (e, lead) = project_leading_fields(&base, base_is_ref, path);
379    match (lead..path.len()).find(|&i| path[i].is_optional()) {
380        None => body(fold_steps(qualify, &path[lead..], e, false)),
381        Some(k) => {
382            // Through the optional step INCLUSIVE: the same fold, so the
383            // borrow in front of it is the ordinary rule rather than a second
384            // statement of it.
385            let opt_e = fold_steps(qualify, &path[lead..=k], e, false);
386            let bind = format_ident!("__hb{}", depth);
387            // What the arm binds is the step's own value: an OWNED payload is a
388            // bare `T`, so composing the next step onto it directly would hand
389            // `T` to an accessor typed for `&T`. Say it is not a reference and
390            // let `project_leading_fields` borrow it; a borrowed payload is
391            // already one and passes through.
392            //
393            // With NO steps left the binding goes to `body` untouched — that is
394            // what lets a consuming value form MOVE an owned payload rather than
395            // borrow it straight back, so the terminal case stays "already a
396            // reference" whatever the payload is.
397            let rest = &path[k + 1..];
398            let inner = reach_optional(
399                qualify,
400                rest,
401                quote!(#bind),
402                rest.is_empty() || !path[k].yields_owned(),
403                depth + 1,
404                body,
405            );
406            // `map` when this is the LAST optional step (the body yields a bare
407            // value) and `and_then` when another follows (the recursion yields
408            // an `Option` that must not nest). The equivalent `match` reads the
409            // same but generated code runs through the consumer's lints, where
410            // `clippy::manual_map` is a denial.
411            let combinator = if rest.iter().any(PathStep::is_optional) {
412                format_ident!("and_then")
413            } else {
414                format_ident!("map")
415            };
416            quote! {
417                #opt_e.#combinator(|#bind| #inner)
418            }
419        }
420    }
421}
422pub fn bind_hoists(
423    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
424    hoists: &[crate::unfold::Hoist],
425    value: &TokenStream,
426    by_ref: bool,
427) -> Hoisted {
428    let mut out = Hoisted {
429        stmts: TokenStream::new(),
430        bound: Vec::new(),
431        consuming: Vec::new(),
432        optional: Vec::new(),
433    };
434    // Value forms COMPOSE, so each hoist is built from the longest hoist that
435    // is already a proper prefix of it (they arrive outermost-first), and from
436    // `value` otherwise.
437    for (i, h) in hoists.iter().enumerate() {
438        let local = format_ident!("__vf{}", i);
439        // A hoist under an optional step binds `Option<TStruct>`: the value
440        // form runs in the `Some` arm only. Core refuses to nest these, so the
441        // enclosing value is always the plan's own — no rebase to consider.
442        if h.prefix.iter().any(PathStep::is_optional) {
443            let (last, lead) = h
444                .prefix
445                .split_last()
446                .expect("a hoist prefix ends in its value-form call");
447            let consuming = h.consuming;
448            // The value form is handed the payload only when the optional step
449            // is the LAST thing before it; any step in between composes as a
450            // borrow, so what arrives is a reference either way.
451            let owned = lead.last().is_some_and(PathStep::yields_owned);
452            let expr = reach_optional(qualify, lead, value.clone(), by_ref, 0, &|reached| {
453                compose_value_form_call(qualify, last, reached, owned, consuming)
454            });
455            out.stmts.extend(quote! { let #local = #expr; });
456            out.bound.push((h.prefix.clone(), local));
457            out.consuming.push(h.consuming);
458            out.optional.push(true);
459            continue;
460        }
461        // Where the fold starts, and whether what it starts from is OWNED. The
462        // value form's own boundary is decided below, from what the fold ends
463        // up holding — never from where it began.
464        let (from, start, start_owned) = match out.rebase(&h.prefix) {
465            // A NESTED consuming form is handed the parent's field by MOVE: a
466            // hoisted value form is an owned struct and its fields are
467            // disjoint, so moving one out leaves every sibling leaf readable.
468            // `compose_step` borrows (`&(e).f`), so a plain field run to that
469            // field is projected here instead of going through it.
470            Some((outer, rest, _))
471                if h.consuming && rest[..rest.len() - 1].iter().all(PathStep::is_plain_field) =>
472            {
473                let lead = &rest[..rest.len() - 1];
474                let segs: Vec<&syn::Ident> = lead.iter().map(PathStep::ident).collect();
475                (h.prefix.len() - 1, quote!(#outer #(.#segs)*), true)
476            }
477            // Any other rebased hoist: project its own leading field run
478            // DIRECTLY off the parent local rather than reaching it through a
479            // borrow of the parent. A sibling hoist may already have moved a
480            // different field out — that is what a consuming value form does —
481            // and `&(&__vf0).wrapper` borrows the partially moved parent as a
482            // whole where `&__vf0.wrapper` is a disjoint borrow that survives.
483            // Same invariant `project_leading_fields` states for leaf reaches,
484            // and the same reason.
485            Some((outer, rest, _)) => {
486                let (e, lead) = project_leading_fields(&quote!(#outer), false, &rest);
487                (h.prefix.len() - rest.len() + lead, e, false)
488            }
489            None if by_ref => (0, value.clone(), false),
490            None => (0, value.clone(), true),
491        };
492        // Everything before the value form is an ordinary accessor chain.
493        let last = h.prefix.len() - 1;
494        let head = &h.prefix[from..last];
495        let e = fold_steps(qualify, head, start, start_owned);
496        let e_owned = head.last().map_or(start_owned, PathStep::yields_owned);
497        // The value form itself. A CONSUMING one takes its receiver BY VALUE —
498        // that is the move the whole declaration exists for — so it is handed
499        // what the fold holds when that is ours, and a clone when it is not:
500        // the same cost the borrowing form of the accessor would have paid,
501        // which keeps one declaration usable by both owned and `&T` returns.
502        // A borrowing one takes a reference, so an owned value is borrowed.
503        //
504        // Deciding this from the fold's RESULT rather than from its start is
505        // what lets a consuming form sit behind ordinary accessors: the chain
506        // in front borrows, the form itself still moves.
507        let expr = compose_value_form_call(qualify, &h.prefix[last], e, e_owned, h.consuming);
508        out.stmts.extend(quote! { let #local = #expr; });
509        out.bound.push((h.prefix.clone(), local));
510        out.consuming.push(h.consuming);
511        out.optional.push(false);
512    }
513    out
