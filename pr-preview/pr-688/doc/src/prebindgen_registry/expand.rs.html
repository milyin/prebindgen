1//! Constructor expansion — fold a value's construction into the wire
2//! signature of the function that consumes it, so the foreign side builds
3//! the value and calls the function in a single FFI crossing.
4//!
5//! A *constructor* is any `#[prebindgen]` function `f(p0, …) -> T` (or
6//! `-> Result<T, E>`) that builds a target type `T`. A type's input flatten
7//! (a type-level `expand_param!` `.variant*` list, or the per-fn
8//! `.expand_param(param, …)` override) replaces a parameter of that type —
9//! in the generated foreign signature only — with the constructor's inputs,
10//! flattened. The generated wrapper decodes those inputs, runs the
11//! constructor Rust-side (the **fold**), and passes the built value to the
12//! underlying call.
13//!
14//! * **One `Ctor` variant** (no identity): the parameter becomes `f`'s
15//!   parameters directly (no selector) — the plain "single" form.
16//! * **Two or more variants** (or an identity arm): the parameter becomes a
17//!   runtime selector (`i32`) plus one `Option`-wrapped input group per variant.
18//!   The identity variant passes an already-built `T` straight through.
19//!
20//! Everything here is **language-agnostic**: the fold is pure Rust and the
21//! per-leaf wire encode/decode is delegated to the adapter's existing
22//! converters. Resolution turns the declarations into [`FoldPlan`]s (stored on
23//! the registry, keyed by `(fn, param)`) and registers each leaf type as a
24//! required input so the resolver produces its converter. [`emit_fold`]
25//! emits the dispatch expression at the parameter-emission site.
26

50/// One variant of a constructor — a selector-dispatched alternative for the
51/// expanded parameter. A constructor with a single `Ctor` variant (and no
52/// `Identity`) is the degenerate "single" form: applied unconditionally with no
53/// selector. Two or more variants (or an `Identity` arm) get a runtime selector.
54#[derive(Clone)]
55pub enum Variant {
56    /// Build the target by calling this constructor function.
57    Ctor(syn::Ident),
58    /// Pass an already-built target value straight through.
59    Identity,
60}
61
62/// A type-level constructor declaration (`expand_param!(T).variant*`): the
63/// complete, ordered variant list for building `target` from flattened
64/// leaves. An immutable record — the variant order is the declaration order
65/// of the `variants` vector.
66#[derive(Clone)]
67pub struct ConstructorDecl {
68    /// The type being built, as an **identity**. Every use keyed it; none
69    /// spelled it.
70    pub target: TypeKey,
71    pub variants: Vec<Variant>,
72    /// Auto-`construct` every matching param of every declared fn. Always
73    /// `true` for type-level default (`expand_param!` `.variant*`) declarations.
74    pub default: bool,
75}
76
77/// How a construct declaration chooses the variants for a parameter.
78#[derive(Clone)]
79pub enum ExpandSel {
80    /// Use the target type's default constructor (error if none/ambiguous).
81    TopLevel,
82    /// Per-fn override (`.expand_param`): use exactly these build-from
83    /// variants (constructor fns and/or the identity/self arm).
84    Subset(Vec<Variant>),
85}
86
87/// A per-fn input expansion (`.expand_param(param, expand_param!(T)…)`) —
88/// construct `param` of `func` from the explicit variant list. Recorded as
89/// an explicit decl so the auto-`default` skips it; an identity-only list
90/// lowers to the skip-default plain form at resolution. Not related to the
91/// jnigen declaration-DSL type of the same name — this is the lowered core
92/// record.
93#[derive(Clone)]
94pub struct ExpandDecl {
95    pub func: syn::Ident,
96    pub param: syn::Ident,
97    /// The type the per-fn decl was declared for (`expand_param!(T)`) —
98    /// cross-checked against the named param's peeled type at resolution.
99    /// `None` for the internal `TopLevel` form (the type comes from the
100    /// param itself).
101    pub declared_target: Option<TypeKey>,
102    pub sel: ExpandSel,
103}
104
105/// Constructor / expansion declarations gathered from a language builder —
106/// an immutable record set: complete values, no build protocol. Declaration
107/// order is the vector order. Handed to the registry as
108/// [`Decompositions::expansions`](crate::Decompositions::expansions); empty
109/// or duplicate declarations are diagnosed at resolution (collected), not at
110/// construction.
111#[derive(Clone, Default)]
112pub struct Expansions {
113    pub constructors: Vec<ConstructorDecl>,
114    pub expands: Vec<ExpandDecl>,
115    /// `.skip_default_construct(param)` opt-outs: `(fn, param)` excluded from a
116    /// constructor `.default()` auto-apply — the lowered form of an
117    /// identity-only per-fn variant set (the plain handle, no selector).
118    pub skip_construct: std::collections::HashSet<(syn::Ident, syn::Ident)>,
119}

125/// Structural validation of the declaration records — empty variant lists
126/// and duplicate targets. Collects EVERY offender before failing, so a
127/// build surfaces all declaration problems at once.
128fn validate_declarations(exp: &Expansions) -> Result<(), ExpandError> {
129    let mut entries: Vec<ExpandDeclError> = Vec::new();
130    let mut ctor_targets: HashSet<String> = HashSet::new();
131    for c in &exp.constructors {
132        let target = c.target.as_str().to_string();
133        if c.variants.is_empty() {
134            entries.push(ExpandDeclError::EmptyConstructor {
135                target: target.clone(),
136            });
137        }
138        if !ctor_targets.insert(target.clone()) {
139            entries.push(ExpandDeclError::DuplicateConstructor { target });
140        }
141    }
142    let mut expand_keys: HashSet<(String, String)> = HashSet::new();
143    for ed in &exp.expands {
144        if let ExpandSel::Subset(v) = &ed.sel {
145            if v.is_empty() {
146                entries.push(ExpandDeclError::EmptySubset {
147                    func: ed.func.clone(),
148                    param: ed.param.clone(),
149                });
150            }
151        }
152        if !expand_keys.insert((ed.func.to_string(), ed.param.to_string())) {
153            entries.push(ExpandDeclError::DuplicateExpand {
154                func: ed.func.clone(),
155                param: ed.param.clone(),
156            });
157        }
158    }
159    if entries.is_empty() {
160        Ok(())
161    } else {
162        Err(ExpandError::InvalidDeclarations { entries })
163    }
164}

166/// Resolve every `.construct` declaration (explicit + `.default()`
167/// auto-applied) into a [`FoldPlan`], register each plan's leaf types as required
168/// inputs, and store the plans on the registry. `declared_fns` is the adapter's
169/// claimed `#[prebindgen]` fn set — the domain over which `.default()`
170/// constructors auto-apply.
171///
172/// Runs inside the builder's scan, before any conversion is built, so
173/// leaf converters resolve through the normal rank machinery.
174pub(crate) fn apply(
175    registry: &mut Registry,
176    exp: &Expansions,
177    declared_fns: &std::collections::HashSet<syn::Ident>,
178    accessor_fns: &std::collections::HashSet<syn::Ident>,
179    method_receivers: &std::collections::HashMap<syn::Ident, TypeKey>,
180) -> Result<(), ExpandError> {
181    validate_declarations(exp)?;
182    let mut done: HashSet<(String, String)> = HashSet::new();
183    let mut skip_construct = exp.skip_construct.clone();
184    for ed in &exp.expands {
185        // A `.fun_accessor` is never parameter-composed — an explicit
186        // `.construct(param)` on one is a build error.
187        if accessor_fns.contains(&ed.func) {
188            return Err(ExpandError::ConstructOnAccessor {
189                func: ed.func.clone(),
190            });
191        }
192        // Per-fn decl cross-check: the named param must exist and its peeled
193        // (`Option`/`&`) type must equal the decl's declared type — the
194        // typo guard for both coordinates of `.expand_param(name, decl)`.
195        if let Some(declared) = &ed.declared_target {
196            let param_ty = param_reading(registry, &ed.func, &ed.param)?;
197            let bare = constructed_value(&param_ty).key();
198            if bare != *declared {
199                return Err(ExpandError::ParamTypeMismatch {
200                    func: ed.func.clone(),
201                    param: ed.param.clone(),
202                    declared: declared.as_str().to_string(),
203                    actual: bare.as_str().to_string(),
204                });
205            }
206        }
207        // Identity-only variant set = the plain form: no selector, the param
208        // crosses as the bare value — lowered to the skip-default opt-out
209        // (the complete-set rule: "the set is {self}").
210        if let ExpandSel::Subset(v) = &ed.sel {
211            if matches!(v.as_slice(), [Variant::Identity]) {
212                skip_construct.insert((ed.func.clone(), ed.param.clone()));
213                done.insert((ed.func.to_string(), ed.param.to_string()));
214                continue;
215            }
216        }
217        process_expand(registry, exp, ed)?;
218        done.insert((ed.func.to_string(), ed.param.to_string()));
219    }
220
221    // `.default()` auto-apply: `construct` every matching param of every declared
222    // fn whose type peeled of `Option`/`&` equals a defaulted constructor target.
223    for c in &exp.constructors {
224        if !c.default {
225            continue;
226        }
227        let ckey = c.target.clone();
228        for func in declared_fns {
229            // Read accessors are excluded from the composer.
230            if accessor_fns.contains(func) {
231                continue;
232            }
233            let Some(params) = registry.flat().function(&func).map(|f| f.params.clone()) else {
234                continue;
235            };
236            // A method's receiver (first param of its class type) binds to `this`
237            // and is never input-flattened; skip exactly that one param.
238            let receiver_key = method_receivers.get(func);
239            let mut receiver_skipped = false;
240            for (pname, pty) in params.iter().map(|p| (p.name.clone(), p.ty.clone())) {
241                let bare_key = constructed_value(&pty).key();
242                if !receiver_skipped && receiver_key == Some(&bare_key) {
243                    receiver_skipped = true;
244                    continue;
245                }
246                if bare_key != ckey {
247                    continue;
248                }
249                if skip_construct.contains(&(func.clone(), pname.clone())) {
250                    continue;
251                }
252                if !done.insert((func.to_string(), pname.to_string())) {
253                    continue;
254                }
255                let ed = ExpandDecl {
256                    func: func.clone(),
257                    param: pname,
258                    declared_target: None,
259                    sel: ExpandSel::TopLevel,
260                };
261                process_expand(registry, exp, &ed)?;
262            }
263        }
264    }
265    Ok(())
266}

268/// `(name, type)` of each typed parameter.
269/// The **reading** of a declared function's parameter.
270///
271/// `Param::ty` is a `TypeRef` computed at parse time. Reaching into the item's
272/// `spell()` and digging the parameter out of `sig.inputs` — what these
273/// three sites used to do — re-derives a fact the model was already handing over,
274/// which is `origin` used for reasoning rather than for emission.
275fn param_reading(
276    registry: &Registry,
277    func: &syn::Ident,
278    param: &syn::Ident,
279) -> Result<prebindgen_flat::flat::TypeRef, ExpandError> {
280    registry
281        .flat()
282        .function(&func)
283        .ok_or_else(|| ExpandError::UnknownFunction(func.clone()))?
284        .params
285        .iter()
286        .find(|p| &p.name == param)
287        .map(|p| p.ty.clone())
288        .ok_or_else(|| ExpandError::UnknownParam(func.clone(), param.clone()))
289}

291/// Build + store the fold plan for one `.construct` declaration.
292fn process_expand(
293    registry: &mut Registry,
294    exp: &Expansions,
295    ed: &ExpandDecl,
296) -> Result<(), ExpandError> {
297    let param_ty = param_reading(registry, &ed.func, &ed.param)?;
298
299    // The boundary layers: `Option<&T>` → optional + by_ref, `Option<T>` →
300    // optional, `&T` → by_ref, and `target` is what is left under them.
301    let (optional, by_ref, target) = constructed_value_layers(&param_ty);
302    let target_key = target.key();
303
304    let variants = resolve_constructor(exp, registry, &target_key, ed)?;
305    let mut visited: HashSet<TypeKey> = HashSet::new();
306    let plan = build_plan(
307        exp,
308        registry,
309        ed,
310        optional,
311        by_ref,
312        &target,
313        &variants,
314        &mut visited,
315    )?;
316
317    // One callback per expanded parameter. A delivered value's site is named by
318    // the SOURCE parameter it arrived on, not by the leaf, so a second callback
319    // here would give two distinct positions one identity. Refused where the
320    // expansion is declared rather than left to collide when the sites are
321    // planned (#687 review).
322    if plan
323        .leaves
324        .iter()
325        .filter(|leaf| leaf.ty.callback_args().is_some())
326        .count()
327        > 1
328    {
329        return Err(ExpandError::TwoCallbacksInOneParam {
330            func: ed.func.clone(),
331            param: ed.param.clone(),
332        });
333    }
334    for leaf in &plan.leaves {
335        registry.require_input(&leaf.ty);
336    }
337    registry
338        .expansion_plans
339        .insert((ed.func.clone(), ed.param.clone()), plan);
340    Ok(())
341}

343/// Pick the constructor (its variants) for one `.expand`/`.expand_with`
344/// declaration. A constructor is keyed by its declared `target`; `TopLevel`
345/// requires it to be unique for the parameter's target type.
346fn resolve_constructor(
347    exp: &Expansions,
348    _registry: &Registry,
349    target_key: &TypeKey,
350    ed: &ExpandDecl,
351) -> Result<Vec<Variant>, ExpandError> {
352    match &ed.sel {
353        ExpandSel::Subset(variants) => Ok(variants.clone()),
354        // Unique per target: `ensure_default_constructor` dedups by type key.
355        ExpandSel::TopLevel => exp
356            .constructors
357            .iter()
358            .find(|c| c.target == *target_key)
359            .map(|c| c.variants.clone())
360            .ok_or_else(|| ExpandError::NoConstructor {
361                func: ed.func.clone(),
362                param: ed.param.clone(),
363                target: target_key.to_string(),
364            }),
365    }
366}

368/// Constructor signature: parameter `(name, type)` pairs and whether it is
369/// fallible (`-> Result<_, _>`). The produced (`Ok`) target type is *checked*
370/// here rather than returned — see below.
371///
372/// `expected` is the type the declaration is *for*, and the returned signature
373/// is one already proven to produce it. Taking it as a parameter rather than
374/// leaving the caller to check afterwards is the point: a declarator cannot
375/// reach a constructor's signature without saying what that constructor is
376/// supposed to build, so the check cannot be the thing a new declarator forgets
377/// (#223). The comparison is [`check_declared_target`], shared with the output
378/// side's accessor lookup.
379fn ctor_signature(
380    registry: &Registry,
381    func: &syn::Ident,
382    expected: &TypeKey,
383) -> Result<CtorSig, ExpandError> {
384    // Read off the element rather than re-walked from the signature: `params`
385    // and `ret` are the same facts, already decided once — including that an
386    // elided return and a written `-> ()` are one thing.
387    let f = registry
388        .flat()
389        .function(&func)
390        .ok_or_else(|| ExpandError::UnknownConstructor(func.clone()))?;
391
392    let params: Vec<(syn::Ident, prebindgen_flat::flat::TypeRef)> = f
393        .params
394        .iter()
395        .map(|p| (p.name.clone(), p.ty.clone()))
396        .collect();
397    // The model already read this return; `fallible_parts` is that reading, not a
398    // second look at the spelling.
399    let (target, fallible) = match f.ret.fallible_parts() {
400        Some((ok, _)) => (ok.key(), true),
401        None => (f.ret.key(), false),
402    };
403    check_declared_target(func, &target, expected)?;
404    Ok(CtorSig { params, fallible })
405}

407struct CtorSig {
408    /// Readings, not spellings: they come off `Function::params`, and a consumer
409    /// that needs the spelling takes it at the point it stores one.
410    params: Vec<(syn::Ident, prebindgen_flat::flat::TypeRef)>,
411    fallible: bool,
412}

414/// Build the [`FoldPlan`] for a chosen construction. A single `Ctor` variant
415/// (no identity) is the plain/unconditional form (no selector); anything else is
416/// selector-dispatched — so a "single" constructor and a 1-variant combined emit
417/// identical code.
418#[allow(clippy::too_many_arguments)]
419fn build_plan(
420    exp: &Expansions,
421    registry: &Registry,
422    ed: &ExpandDecl,
423    optional: bool,
424    by_ref: bool,
425    target: &prebindgen_flat::flat::TypeRef,
426    variants: &[Variant],
427    visited: &mut HashSet<TypeKey>,
428) -> Result<FoldPlan, ExpandError> {
429    let param = &ed.param;
430    let mut leaves: Vec<FoldLeaf> = Vec::new();
431
432    // Optional (`Option<T>`/`Option<&T>`) param. No recursion under `Optional`.
433    //  * single single-arg ctor → one nullable leaf (`Option<arg>`) decides
434    //    presence.
435    //  * single multi-arg ctor  → an explicit leading `present: bool` flag +
436    //    one plain (non-`Option`) leaf per arg. The flag keeps nullable
437    //    primitive args (e.g. an `Option<i32>` id) from boxing on the wire.
438    //  * combined (≥2 variants) → the same selector dispatch as a non-optional
439    //    param, with the selector ALSO encoding absence: `-1` = `None`,
440    //    `0..n-1` = the taken arm (no separate present flag — not-taken arms'
441    //    leaves are null exactly as in the non-optional selector case).
442    if optional {
443        let [Variant::Ctor(func)] = variants else {
444            // Combined-selector dispatch under `Optional`.
445            visited.insert(target.key());
446            let prefix = param.to_string();
447            let (selector, fold_variants) = build_core(
448                exp,
449                registry,
450                ed,
451                target,
452                variants,
453                by_ref,
454                &prefix,
455                &mut leaves,
456                visited,
457            )?;
458            visited.remove(&target.key());
459            return Ok(FoldPlan {
460                target: target.clone(),
461                by_ref,
462                shape: FoldShape::Optional((), Box::new(FoldShape::Base)),
463                leaves,
464                selector,
465                present: None,
466                variants: fold_variants,
467            });
468        };
469        let sig = ctor_signature(registry, func, &target.key())?;
470        if sig.params.len() == 1 {
471            let (_pn, pty) = &sig.params[0];
472            leaves.push(FoldLeaf {
473                name: param.clone(),
474                ty: pty.optional(),
475            });
476            return Ok(FoldPlan {
477                target: target.clone(),
478                by_ref,
479                shape: FoldShape::Optional((), Box::new(FoldShape::Base)),
480                leaves,
481                selector: None,
482                present: None,
483                variants: vec![FoldVariant {
484                    ctor: Some(func.clone()),
485                    fallible: sig.fallible,
486                    clone: false,
487                    inputs: vec![FoldArg::Leaf(0, false)],
488                }],
489            });
490        }
491        // Multi-arg: presence flag (leaf 0) + one flat leaf per ctor arg.
492        leaves.push(FoldLeaf {
493            name: ident(&format!("{}_present", param)),
494            // A presence flag no source wrote — placeless by construction.
495            ty: prebindgen_flat::flat::TypeRef::scalar(prebindgen_flat::flat::ScalarKind::Bool),
496        });
497        let prefix = param.to_string();
498        let mut inputs = Vec::new();
499        for (pname, pty) in &sig.params {
500            let name = ident(&format!("{}_{}", prefix, pname));
501            let arg = build_arg(
502                exp,
503                registry,
504                ed,
505                pty,
506                name,
507                /*dispatched=*/ false,
508                &mut leaves,
509                visited,
510            )?;
511            if matches!(arg, FoldArg::Build(_)) {
512                return Err(ExpandError::UnsupportedOptional {
513                    func: ed.func.clone(),
514                    param: ed.param.clone(),
515                    reason: "nested-buildable constructor arguments cannot be optional",
516                });
517            }
518            inputs.push(arg);
519        }
520        return Ok(FoldPlan {
521            target: target.clone(),
522            by_ref,
523            shape: FoldShape::Optional((), Box::new(FoldShape::Base)),
524            leaves,
525            selector: None,
526            present: Some(0),
527            variants: vec![FoldVariant {
528                ctor: Some(func.clone()),
529                fallible: sig.fallible,
530                clone: false,
531                inputs,
532            }],
533        });
534    }
535
536    // Non-optional: build the (possibly recursive) construct core. The target is
537    // on the cycle chain so a constructor parameter of the same type is rejected.
538    visited.insert(target.key());
539    let prefix = param.to_string();
540    let (selector, fold_variants) = build_core(
541        exp,
542        registry,
543        ed,
544        target,
545        variants,
546        by_ref,
547        &prefix,
548        &mut leaves,
549        visited,
550    )?;
551    visited.remove(&target.key());
552    Ok(FoldPlan {
553        target: target.clone(),
554        by_ref,
555        shape: FoldShape::Base,
556        leaves,
557        selector,
558        present: None,
559        variants: fold_variants,
560    })
561}

563/// Build a construct core (selector + dispatch arms) for `target` from its
564/// `variants`, appending wire leaves to `leaves`. Recursive: a constructor
565/// parameter whose type has its OWN default constructor is built as a nested
566/// [`FoldArg::Build`] (recursive input). Used by both the top-level [`build_plan`]
567/// and each nested build. `prefix` disambiguates leaf names across the tree.
568#[allow(clippy::too_many_arguments)]
569fn build_core(
570    exp: &Expansions,
571    registry: &Registry,
572    ed: &ExpandDecl,
573    target: &prebindgen_flat::flat::TypeRef,
574    variants: &[Variant],
575    by_ref: bool,
576    prefix: &str,
577    leaves: &mut Vec<FoldLeaf>,
578    visited: &mut HashSet<TypeKey>,
579) -> Result<(Option<usize>, Vec<FoldVariant>), ExpandError> {
580    if let [Variant::Ctor(func)] = variants {
581        // Single constructor — no selector; args passed directly (not Option-wrapped).
582        let sig = ctor_signature(registry, func, &target.key())?;
583        let np = sig.params.len();
584        let mut args = Vec::new();
585        for (pname, pty) in &sig.params {
586            let name = if np == 1 {
587                ident(prefix)
588            } else {
589                ident(&format!("{}_{}", prefix, pname))
590            };
591            args.push(build_arg(
592                exp, registry, ed, pty, name, false, leaves, visited,
593            )?);
594        }
595        Ok((
596            None,
597            vec![FoldVariant {
598                ctor: Some(func.clone()),
599                fallible: sig.fallible,
600                clone: false,
601                inputs: args,
602            }],
603        ))
604    } else {
605        // Combined — selector leaf, then `Option`-wrapped per-arm inputs.
606        let sel_idx = leaves.len();
607        leaves.push(FoldLeaf {
608            name: ident(&format!("{}_sel", prefix)),
609            // The selector, likewise composed and placeless.
610            ty: prebindgen_flat::flat::TypeRef::scalar(prebindgen_flat::flat::ScalarKind::I32),
611        });
612        let mut fold_variants: Vec<FoldVariant> = Vec::new();
613        for (vi, v) in variants.iter().enumerate() {
614            match v {
615                Variant::Ctor(func) => {
616                    let sig = ctor_signature(registry, func, &target.key())?;
617                    let np = sig.params.len();
618                    let mut args = Vec::new();
619                    for (pi, (_pname, pty)) in sig.params.iter().enumerate() {
620                        let name = if np == 1 {
621                            ident(&format!("{}_{}", prefix, vi))
622                        } else {
623                            ident(&format!("{}_{}_{}", prefix, vi, pi))
624                        };
625                        // `dispatched = true`: a combined arm's leaves are
626                        // `Option`-wrapped (selector presence). Recursive nesting
627                        // under a combined arm is rejected by `build_arg`.
628                        args.push(build_arg(
629                            exp, registry, ed, pty, name, true, leaves, visited,
630                        )?);
631                    }
632                    fold_variants.push(FoldVariant {
633                        ctor: Some(func.clone()),
634                        fallible: sig.fallible,
635                        clone: false,
636                        inputs: args,
637                    });
638                }
639                Variant::Identity => {
640                    let idx = leaves.len();
641                    let leaf_ty = if by_ref {
642                        target.borrowed().optional()
643                    } else {
644                        target.optional()
645                    };
646                    leaves.push(FoldLeaf {
647                        name: ident(&format!("{}_{}", prefix, vi)),
648                        ty: leaf_ty,
649                    });
650                    fold_variants.push(FoldVariant {
651                        ctor: None,
652                        fallible: false,
653                        clone: by_ref,
654                        inputs: vec![FoldArg::Leaf(idx, false)],
655                    });
656                }
657            }
658        }
659        Ok((Some(sel_idx), fold_variants))
660    }
661}

663/// Build one constructor-parameter input. If the parameter's (peeled) type has
664/// its own default constructor, recurse into a nested [`FoldArg::Build`]
665/// (recursive input); otherwise it is a flat wire [`FoldArg::Leaf`].
666#[allow(clippy::too_many_arguments)]
667fn build_arg(
668    exp: &Expansions,
669    registry: &Registry,
670    ed: &ExpandDecl,
671    pty: &prebindgen_flat::flat::TypeRef,
672    name: syn::Ident,
673    dispatched: bool,
674    leaves: &mut Vec<FoldLeaf>,
675    visited: &mut HashSet<TypeKey>,
676) -> Result<FoldArg, ExpandError> {
677    // The boundary layers down to the parameter's core type.
678    let (popt, pby_ref, bare) = constructed_value_layers(pty);
679    let key = bare.key();
680    // A default constructor for the parameter's type ⇒ recursive nested build.
681    let canon = exp
682        .constructors
683        .iter()
684        .find(|c| c.target == key && !c.variants.is_empty());
685    if let Some(c) = canon {
686        if dispatched {
687            return Err(ExpandError::UnsupportedRecursive {
688                func: ed.func.clone(),
689                reason: "recursive input under a selector-dispatched constructor variant",
690            });
691        }
692        if popt {
693            return Err(ExpandError::UnsupportedRecursive {
694                func: ed.func.clone(),
695                reason: "recursive input on an Option<…> parameter",
696            });
697        }
698        if !visited.insert(key.clone()) {
699            return Err(ExpandError::InputCycle {
700                ty: key.to_string(),
701            });
702        }
703        let variants = c.variants.clone();
704        let (selector, vars) = build_core(
705            exp,
706            registry,
707            ed,
708            &bare,
709            &variants,
710            pby_ref,
711            &name.to_string(),
712            leaves,
713            visited,
714        )?;
715        visited.remove(&key);
716        Ok(FoldArg::Build(Box::new(FoldBuild {
717            target: bare.clone(),
718            by_ref: pby_ref,
719            selector,
720            variants: vars,
721        })))
722    } else {
723        let idx = leaves.len();
724        // A dispatched (selector-presence) arm `Option`-wraps its leaves — but
725        // an argument that is itself `Option<…>` passes through with its own
726        // type: `None` is a legitimate value for the taken arm, and the wire
727        // cannot represent the double `Option` anyway. Marked `passthrough` so
728        // the emit side skips the selector-presence unwrap.
729        let passthrough = dispatched && popt;
730        leaves.push(FoldLeaf {
731            name,
732            ty: if dispatched && !passthrough {
733                pty.optional()
734            } else {
735                pty.clone()
736            },
737        });
738        Ok(FoldArg::Leaf(idx, passthrough))
739    }
740}

758/// Emit the fold expression for an expanded parameter. `leaf_locals` are the
759/// already-decoded Rust locals (1:1 with `plan.leaves`); `qualify` maps a
760/// constructor ident to its call path (e.g. prefixing the source module).
761///
762/// The returned expression has type `Result<<shaped> plan.target, String>`
763/// (`Result<Target>`, `Result<Option<Target>>`, …). The adapter routes its
764/// `Err(String)` through its own error channel. Folds the [`FoldShape`] layers
765/// top-down over one shared core construct — the value
766/// analog of how `Option<_>`/`Vec<_>` wrappers compose at the wire.
767pub fn emit_fold(
768    plan: &FoldPlan,
769    leaf_locals: &[syn::Ident],
770    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
771) -> syn::Expr {
772    fold_shape(&plan.shape, plan, leaf_locals, None, qualify)
773}
774
775/// Recurse over one [`FoldShape`] layer. `bound` is `Some(var)` when an
776/// enclosing `Optional`/`Iterable` layer has unwrapped the structured leaf and
777/// bound its element to `var` — the inner construct then builds from `var`
778/// instead of reading `leaf_locals`.
779fn fold_shape(
780    shape: &FoldShape,
781    plan: &FoldPlan,
782    leaf_locals: &[syn::Ident],
783    bound: Option<&syn::Ident>,
784    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
785) -> syn::Expr {
786    match shape {
787        FoldShape::Base => emit_core_construct(plan, leaf_locals, bound, qualify),
788        FoldShape::Optional((), inner) => {
789            if let Some(sidx) = plan.selector {
790                // Combined-selector dispatch under `Optional`: the selector
791                // ALSO encodes absence — `-1` = `None`, `0..n-1` = the taken
792                // arm (dispatched by the shared construct core; an out-of-range
793                // selector still hits its `Err` default arm).
794                let sel_local = &leaf_locals[sidx];
795                let inner_expr = emit_core_construct(plan, leaf_locals, None, qualify);
796                syn::parse_quote!(if #sel_local < 0 {
797                    ::core::result::Result::Ok(::core::option::Option::None)
798                } else {
799                    (#inner_expr).map(::core::option::Option::Some)
800                })
801            } else if let Some(pidx) = plan.present {
802                // Multi-arg: an explicit `present: bool` flag decides presence;
803                // the construct reads its plain arg leaves directly (`bound =
804                // None`), the flag leaf is consumed only by this `if`.
805                let present_local = &leaf_locals[pidx];
806                let inner_expr = emit_core_construct(plan, leaf_locals, None, qualify);
807                syn::parse_quote!(if #present_local {
808                    (#inner_expr).map(::core::option::Option::Some)
809                } else {
810                    ::core::result::Result::Ok(::core::option::Option::None)
811                })
812            } else {
813                // Single-arg: presence rides the sole shaped leaf's `Option`.
814                // The structured value is the enclosing bound var, or — at the
815                // top — that leaf's decoded local (`leaf_locals[0]`).
816                let value = bound.unwrap_or(&leaf_locals[0]);
817                let inner_ident = ident("__inner");
818                let inner_expr = fold_shape(inner, plan, leaf_locals, Some(&inner_ident), qualify);
819                syn::parse_quote!(match #value {
820                    ::core::option::Option::Some(#inner_ident) => {
821                        (#inner_expr).map(::core::option::Option::Some)
822                    }
823                    ::core::option::Option::None => {
824                        ::core::result::Result::Ok(::core::option::Option::None)
825                    }
826                })
827            }
828        }
829        FoldShape::Iterable(inner) => {
830            let value = bound.unwrap_or(&leaf_locals[0]);
831            let elem_ident = ident("__elem");
832            let inner_expr = fold_shape(inner, plan, leaf_locals, Some(&elem_ident), qualify);
833            syn::parse_quote!(
834                #value
835                    .into_iter()
836                    .map(|#elem_ident| #inner_expr)
837                    .collect::<::core::result::Result<::std::vec::Vec<_>, _>>()
838            )
839        }
840    }
841}

843/// Emit the innermost construct → `Result<Target, String>`. With `bound =
844/// Some(v)` (under an `Optional`/`Iterable` layer ⇒ single, single-arg ctor)
845/// the ctor is applied to `v`; with `bound = None` (top level) it reads the
846/// leaves — a single constructor (any arity) or a combined-selector dispatch.
847fn emit_core_construct(
848    plan: &FoldPlan,
849    leaf_locals: &[syn::Ident],
850    bound: Option<&syn::Ident>,
851    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
852) -> syn::Expr {
853    if let Some(v) = bound {
854        // Shaped construct: a single, single-arg constructor applied to the
855        // unwrapped element. (`apply` guarantees this shape — never identity,
856        // never combined, never multi-arg under a shape layer.)
857        let var = &plan.variants[0];
858        let func = var
859            .ctor
860            .as_ref()
861            .expect("shaped expansion is single-constructor (never identity)");
862        return ctor_call_result(&qualify(func), std::slice::from_ref(v), var.fallible);
863    }
864    emit_dispatch(plan.selector, &plan.variants, leaf_locals, qualify)
865}
866
867/// Emit a construct dispatch → `Result<Target, String>`: a single variant
868/// applied directly (no selector), or a `match` over the selector leaf. Shared
869/// by the top-level [`emit_core_construct`] and each nested [`emit_build`].
870fn emit_dispatch(
871    selector: Option<usize>,
872    variants: &[FoldVariant],
873    leaf_locals: &[syn::Ident],
874    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
875) -> syn::Expr {
876    match selector {
877        None => variant_result_expr(
878            &variants[0],
879            leaf_locals,
880            qualify,
881            /*dispatched=*/ false,
882        ),
883        Some(si) => {
884            let sel = &leaf_locals[si];
885            let arms: Vec<TokenStream> = variants
886                .iter()
887                .enumerate()
888                .map(|(vi, v)| {
889                    let lit = vi as i32;
890                    let body =
891                        variant_result_expr(v, leaf_locals, qualify, /*dispatched=*/ true);
892                    quote!(#lit => #body,)
893                })
894                .collect();
895            syn::parse_quote!({
896                match #sel {
897                    #(#arms)*
898                    __sel => ::core::result::Result::Err(::std::format!(
899                        "invalid constructor selector: {}",
900                        __sel
901                    )),
902                }
903            })
904        }
905    }
906}

908/// Emit a nested recursive-input build → `Result<SubTarget, String>` (the dual
909/// of [`emit_core_construct`] for a [`FoldArg::Build`] parameter).
910fn emit_build(
911    b: &FoldBuild,
912    leaf_locals: &[syn::Ident],
913    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
914) -> syn::Expr {
915    emit_dispatch(b.selector, &b.variants, leaf_locals, qualify)
916}

918/// Build a `Result<Target, String>` expression for one core variant. When
919/// `dispatched` (a combined-selector arm), the variant's input leaves are
920/// `Option<_>` — only the selected arm's inputs are present — so they are
921/// unwrapped (a missing input yields `Err`); otherwise they are passed
922/// directly. (This `Option`-ness is *selector presence*, distinct from
923/// [`FoldShape::Optional`], which is whole-param presence handled by the
924/// enclosing fold.)
925fn variant_result_expr(
926    v: &FoldVariant,
927    leaf_locals: &[syn::Ident],
928    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
929    dispatched: bool,
930) -> syn::Expr {
931    // A Leaf arg's decoded local. Identity arms and combined-dispatched arms are
932    // Leaf-only (recursive `Build` args appear only in a non-dispatched single
933    // constructor — `build_arg` rejects nesting under a dispatched variant).
934    let leaf = |a: &FoldArg| -> &syn::Ident {
935        match a {
936            FoldArg::Leaf(i, _) => &leaf_locals[*i],
937            FoldArg::Build(_) => {
938                unreachable!("recursive Build arg only in a non-dispatched single constructor")
939            }
940        }
941    };
942
943    match &v.ctor {
944        None => {
945            // Identity: the sole input is the target value (or a borrow of it
946            // that we clone, for `&T` consumers — preserving the caller's handle).
947            let loc = leaf(&v.inputs[0]);
948            // `&*__v` derefs through whatever the borrow leaf decoded to (a
949            // plain `&T`, or an adapter smart-pointer like jnigen's
950            // `OwnedObject<T>`) down to `T`, then clones — keeping the caller's
951            // handle alive without the core knowing the adapter's borrow type.
952            let some_val: syn::Expr = if v.clone {
953                syn::parse_quote!(::core::result::Result::Ok(::core::clone::Clone::clone(
954                    &*__v
955                )))
956            } else {
957                syn::parse_quote!(::core::result::Result::Ok(__v))
958            };
959            if dispatched {
960                syn::parse_quote!(match #loc {
961                    ::core::option::Option::Some(__v) => #some_val,
962                    ::core::option::Option::None => ::core::result::Result::Err(
963                        ::std::string::String::from("identity variant value missing")
964                    ),
965                })
966            } else if v.clone {
967                syn::parse_quote!(::core::result::Result::Ok(::core::clone::Clone::clone(&*#loc)))
968            } else {
969                syn::parse_quote!(::core::result::Result::Ok(#loc))
970            }
971        }
972        Some(func) => {
973            let path = qualify(func);
974            if dispatched {
975                // Combined arm — Leaf-only inputs. Selector-presence-wrapped
976                // inputs are unwrapped (missing ⇒ `Err`); **passthrough**
977                // inputs (constructor args that are themselves `Option<…>`)
978                // pass their decoded local directly — `None` is a legitimate
979                // value for the taken arm.
980                let mut wrapped_locals: Vec<&syn::Ident> = Vec::new();
981                let mut wrapped_binds: Vec<syn::Ident> = Vec::new();
982                let mut call_args: Vec<syn::Expr> = Vec::new();
983                for (i, a) in v.inputs.iter().enumerate() {
984                    let loc = leaf(a);
985                    if matches!(a, FoldArg::Leaf(_, true)) {
986                        call_args.push(syn::parse_quote!(#loc));
987                    } else {
988                        let b = ident(&format!("__p{}", i));
989                        wrapped_locals.push(loc);
990                        wrapped_binds.push(b.clone());
991                        call_args.push(syn::parse_quote!(#b));
992                    }
993                }
994                let call = ctor_call_result(&path, &call_args, v.fallible);
995                let missing = quote!(::core::result::Result::Err(::std::string::String::from(
996                    "constructor variant input missing"
997                )));
998                match wrapped_locals.len() {
999                    // All-passthrough arm: the selector alone decides; call directly.
1000                    0 => call,
1001                    1 => {
1002                        // `match a { Some(p0) => <call>, None => Err }`
1003                        let loc = wrapped_locals[0];
1004                        let p0 = &wrapped_binds[0];
1005                        syn::parse_quote!(match #loc {
1006                            ::core::option::Option::Some(#p0) => #call,
1007                            ::core::option::Option::None => #missing,
1008                        })
1009                    }
1010                    _ => {
1011                        // `match (a, b, …) { (Some(p0), Some(p1), …) => <call>, _ => Err }`
1012                        let some_pats: Vec<TokenStream> = wrapped_binds
1013                            .iter()
1014                            .map(|b| quote!(::core::option::Option::Some(#b)))
1015                            .collect();
1016                        syn::parse_quote!(match ( #(#wrapped_locals),* ) {
1017                            ( #(#some_pats),* ) => #call,
1018                            _ => #missing,
1019                        })
1020                    }
1021                }
1022            } else if v.inputs.iter().all(|a| matches!(a, FoldArg::Leaf(..))) {
1023                // Non-dispatched, flat (no recursion): call directly — identical
1024                // to the pre-recursion form.
1025                let args: Vec<&syn::Ident> = v.inputs.iter().map(&leaf).collect();
1026                ctor_call_result(&path, &args, v.fallible)
1027            } else {
1028                // Non-dispatched with ≥1 recursive `Build` arg: bind each arg
1029                // (Leaf = the decoded value; Build = the nested construct,
1030                // `?`-unwrapped) in an IIFE that provides the `Result` context.
1031                let mut stmts: Vec<TokenStream> = Vec::new();
1032                let mut args: Vec<TokenStream> = Vec::new();
1033                for (i, a) in v.inputs.iter().enumerate() {
1034                    let ai = ident(&format!("__a{}", i));
1035                    match a {
1036                        FoldArg::Leaf(li, _) => {
1037                            let loc = &leaf_locals[*li];
1038                            stmts.push(quote!(let #ai = #loc;));
1039                            args.push(quote!(#ai));
1040                        }
1041                        FoldArg::Build(b) => {
1042                            // Pin the nested build's error type to `String` so a
1043                            // non-fallible inner ctor's bare `Ok(..)` infers `E`.
1044                            let be = emit_build(b, leaf_locals, qualify);
1045                            stmts.push(quote!(
1046                                let #ai = {
1047                                    let __r: ::core::result::Result<_, ::std::string::String> = #be;
1048                                    __r?
1049                                };
1050                            ));
1051                            if b.by_ref {
1052                                args.push(quote!(&#ai));
1053                            } else {
1054                                args.push(quote!(#ai));
1055                            }
1056                        }
1057                    }
1058                }
1059                let call = ctor_call_result(&path, &args, v.fallible);
1060                syn::parse_quote!({
1061                    (|| -> ::core::result::Result<_, ::std::string::String> {
1062                        #(#stmts)*
1063                        #call
1064                    })()
1065                })
1066            }
1067        }
1068    }
1069}

1071/// `path(args…)` lifted to `Result<Target, String>` (mapping a fallible
1072/// constructor's error via `Display`).
1073fn ctor_call_result<I: quote::ToTokens>(path: &syn::Path, args: &[I], fallible: bool) -> syn::Expr {
1074    if fallible {
1075        syn::parse_quote!(#path( #(#args),* ).map_err(|__e| ::std::format!("{}", __e)))
1076    } else {
1077        syn::parse_quote!(::core::result::Result::Ok(#path( #(#args),* )))
1078    }
1079}

1085/// The value a constructor builds: `Option` off, then the borrow, and **nothing
1086/// else** — read off the model's classification rather than by taking the
1087/// spelling apart.
1088///
1089/// `Option<&T>`, `&T` and `T` all answer `T`, which is what every caller here
1090/// wants: they are matching a declared target, and a declaration names the type,
1091/// not the way a particular parameter happens to wrap it.
1092///
1093/// **`Vec<T>` answers `Vec<T>`, deliberately.** Expansion builds one value —
1094/// `FoldPlan`'s shape is `Base` or `Optional(Base)`, with no iterable arm — so
1095/// peeling a `Sequence` here would let a `Vec<T>` parameter match a `T`
1096/// constructor and emit a wrapper that reconstructs a single `T` and hands it to
1097/// a parameter expecting the collection. Leaving the `Sequence` on the core is
1098/// what makes that a non-match instead of a miscompile, and it is the reason this
1099/// is not [`TypeRef::layers`], which peels all three.
1100///
1101/// A type the grammar cannot express answers itself — the identity, not a
1102/// fallback classifier. Nothing reaching here can be one: every signature in play
1103/// was accepted by the frontend before the scan registered it.
1104fn constructed_value(reading: &prebindgen_flat::flat::TypeRef) -> &prebindgen_flat::flat::TypeRef {
1105    let after_opt = reading.optional_inner().unwrap_or(reading);
1106    after_opt.borrow_target().unwrap_or(after_opt)
1107}
1108
1109/// [`constructed_value`], plus which of the two layers were there.
1110fn constructed_value_layers(
1111    reading: &prebindgen_flat::flat::TypeRef,
1112) -> (bool, bool, prebindgen_flat::flat::TypeRef) {
1113    let optional = reading.optional_inner().is_some();
1114    let after_opt = reading.optional_inner().unwrap_or(reading);
1115    let by_ref = after_opt.borrow_target().is_some();
1116    let core = after_opt.borrow_target().unwrap_or(after_opt);
1117    // The core READING, not its spelling: the plan composes `Option<&T>` over
1118    // it, and composing from a reading keeps the kind and the syntax paired.
1119    (optional, by_ref, core.clone())
1120}
