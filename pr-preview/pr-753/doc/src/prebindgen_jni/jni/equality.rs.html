1//! Content-based `equals` / `hashCode` / `toString` for generated Kotlin
2//! classes with an **array-backed** property.
3//!
4//! A generated class mirrors a Rust type that derives `PartialEq`/`Eq`, so two
5//! values with equal contents must compare equal. Kotlin arrays compare by
6//! IDENTITY, which breaks that for every `Vec<u8>` or fixed-size primitive-array
7//! field that surfaces as a Kotlin array:
8//!
9//! ```text
10//! Timestamp(1uL, byteArrayOf(1,2,3)) == Timestamp(1uL, byteArrayOf(1,2,3))  // false
11//! ```
12//!
13//! Kotlin is inconsistent here rather than uniformly identity-based: a `data
14//! class`'s generated `hashCode`/`toString` DO special-case arrays
15//! (`contentHashCode` / `contentToString`), while its `equals` does not. So a
16//! broken value has an equal hash and an unequal `equals` — it lands in the
17//! right `HashMap` bucket and is then rejected. Both members are emitted here
18//! anyway, so the behavior is stated by the generated source rather than
19//! inherited from a compiler special case that only covers two of the three.
20
21use kotlin_codegen::{KtCode, KtFun, KtParam, KtType};
22
23/// Whether a Kotlin type carries an array anywhere inside it, and therefore
24/// compares by identity unless the generated operators dig in.
25///
26/// Recursive, because a container of arrays is just as identity-compared as a
27/// bare one: `List<ByteArray>` (from `Vec<Vec<u8>>`) inherits `ByteArray`'s
28/// `equals`, so two lists of equal chunks are unequal and `toString` renders
29/// `[[B@3830f1c0]`. A container of *classes* is fine — those already compare
30/// by value — so only an array at the bottom makes a property array-bearing.
31fn array_bearing(ty: &KtType) -> bool {
32    match ty {
33        KtType::Named { fqn, args, .. } => {
34            is_kotlin_array(fqn.rsplit('.').next().unwrap_or(fqn)) || args.iter().any(array_bearing)
35        }
36        KtType::Function { .. } => false,
37    }
38}

40/// Every Kotlin primitive array. All of them compare by identity, so a
41/// fixed-size Rust array field needs the content operators whatever its element
42/// type is — not only `[u8; N]`.
43fn is_kotlin_array(name: &str) -> bool {
44    crate::jni::wire_access::kotlin_array_descriptor(name).is_some()
45}
46
47/// The element type of a single-argument container (`List<T>` -> `T`).
48fn element_of(ty: &KtType) -> Option<&KtType> {
49    match ty {
50        KtType::Named { args, .. } if args.len() == 1 => Some(&args[0]),
51        _ => None,
52    }
53}

55/// `a == b` for one value of type `ty`, digging through containers.
56///
57/// `a`/`b` are Kotlin expressions. Nullability is handled per level: a
58/// `ByteArray?` rides `contentEquals`'s nullable-receiver overload, while a
59/// nullable container needs an explicit both-null / both-present test.
60fn eq_expr(a: &str, b: &str, ty: &KtType) -> String {
61    if !array_bearing(ty) {
62        return format!("{a} == {b}");
63    }
64    if element_of(ty).is_none() {
65        // The array itself.
66        return format!("{a}.contentEquals({b})");
67    }
68    let elem = element_of(ty).expect("checked above");
69    let inner = eq_expr("__x", "__y", elem);
70    let cmp = format!(
71        "{a}.size == {b}.size && {a}.indices.all {{ __i -> \
72         val __x = {a}[__i]; val __y = {b}[__i]; {inner} }}"
73    );
74    if ty.is_nullable() {
75        format!("(({a} == null && {b} == null) || ({a} != null && {b} != null && {cmp}))")
76    } else {
77        format!("({cmp})")
78    }
79}

81/// `hashCode` for one value of type `ty`, digging through containers. The
82/// container fold mirrors `Arrays.hashCode`'s 31-multiplier so a `List` and the
83/// array it came from agree.
84fn hash_expr(x: &str, ty: &KtType) -> String {
85    if !array_bearing(ty) {
86        return if ty.is_nullable() {
87            format!("({x}?.hashCode() ?: 0)")
88        } else {
89            format!("{x}.hashCode()")
90        };
91    }
92    match element_of(ty) {
93        None if ty.is_nullable() => format!("({x}?.contentHashCode() ?: 0)"),
94        None => format!("{x}.contentHashCode()"),
95        Some(elem) => {
96            let inner = hash_expr("__e", elem);
97            let fold = format!("{x}.fold(1) {{ __acc, __e -> 31 * __acc + {inner} }}");
98            if ty.is_nullable() {
99                format!("({x}?.let {{ __l -> {} }} ?: 0)", fold.replace(x, "__l"))
100            } else {
101                format!("({fold})")
102            }
103        }
104    }
105}

107/// `toString` rendering for one value of type `ty`, digging through containers
108/// so a nested array never prints as `[B@1a2b3c`.
109fn str_expr(x: &str, ty: &KtType) -> String {
110    if !array_bearing(ty) {
111        return format!("${{{x}}}");
112    }
113    match element_of(ty) {
114        None if ty.is_nullable() => format!("${{{x}?.contentToString()}}"),
115        None => format!("${{{x}.contentToString()}}"),
116        Some(elem) => {
117            let inner = str_expr("__e", elem);
118            let join = format!("{x}.joinToString(\", \", \"[\", \"]\") {{ __e -> \"{inner}\" }}");
119            if ty.is_nullable() {
120                format!("${{{x}?.let {{ __l -> {} }}}}", join.replace(x, "__l"))
121            } else {
122                format!("${{{join}}}")
123            }
124        }
125    }
126}

128/// The `equals` / `hashCode` / `toString` trio a class needs when any
129/// constructor property is array-backed.
130///
131/// `None` when none is — the compiler's own generation is already correct
132/// there, and emitting these would be pure churn on every existing class.
133///
134/// `props` is the class's constructor properties in declaration order, as
135/// `(kotlin_name, kotlin_type)`.
136pub(crate) fn content_equality_members(
137    class_name: &str,
138    props: &[(String, KtType)],
139) -> Option<Vec<KtFun>> {
140    if !props.iter().any(|(_, ty)| array_bearing(ty)) {
141        return None;
142    }
143
144    // `equals`: identity short-circuit, type check, then per-property
145    // comparison that digs through any container down to the arrays.
146    let comparisons: Vec<String> = props
147        .iter()
148        .map(|(name, ty)| eq_expr(name, &format!("other.{name}"), ty))
149        .collect();
150    let equals_body = KtCode::new()
151        .line("if (this === other) return true")
152        .line(format!("if (other !is {class_name}) return false"))
153        .line(format!("return {}", comparisons.join(" && ")));
154    let equals = KtFun::new("equals")
155        .modifier("override")
156        .param(KtParam::new("other", KtType::any().nullable()))
157        .returns(KtType::boolean())
158        .body(equals_body);
159
160    // `hashCode`: the standard 31-multiplier fold over the same per-property
161    // expressions, so equal values always agree.
162    let first = hash_expr(&props[0].0, &props[0].1);
163    let hash_body = if props.len() == 1 {
164        // A single property needs no accumulator — `var result` would draw a
165        // "never reassigned" warning in the generated source.
166        KtCode::new().line(format!("return {first}"))
167    } else {
168        let mut b = KtCode::new().line(format!("var result = {first}"));
169        for (name, ty) in &props[1..] {
170            b = b.line(format!("result = 31 * result + {}", hash_expr(name, ty)));
171        }
172        b.line("return result")
173    };
174    let hash_code = KtFun::new("hashCode")
175        .modifier("override")
176        .returns(KtType::int())
177        .body(hash_body);
178
179    // `toString`: an array at any depth would otherwise render as `[B@1a2b3c`.
180    let rendered: Vec<String> = props
181        .iter()
182        .map(|(name, ty)| format!("{name}={}", str_expr(name, ty)))
183        .collect();
184    let to_string = KtFun::new("toString")
185        .modifier("override")
186        .returns(KtType::string())
187        .expr_body(KtCode::new().line(format!("\"{class_name}({})\"", rendered.join(", "))));
188
189    Some(vec![equals, hash_code, to_string])
190}
