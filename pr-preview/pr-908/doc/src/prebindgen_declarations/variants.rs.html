1//! Recursive construction and matching of source enum payloads.

3use std::collections::BTreeMap;

5pub(crate) fn overrides<'a, T>(
6    entries: impl IntoIterator<Item = (&'a str, Vec<(&'a str, T)>)>,
7) -> BTreeMap<String, BTreeMap<String, T>> {
8    let mut variants = BTreeMap::new();
9    for (name, fields) in entries {
10        let mut members = BTreeMap::new();
11        for (field, tree) in fields {
12            assert!(
13                members.insert(field.into(), tree).is_none(),
14                "duplicate enum field override"
15            );
16        }
17        assert!(
18            variants.insert(name.into(), members).is_none(),
19            "duplicate enum variant override"
20        );
21    }
22    variants
23}
24
25fn validate_overrides<V, F, T>(
26    source: &prebindgen_flat::PayloadEnum<V, F>,
27    entries: &BTreeMap<String, BTreeMap<String, T>>,
28) -> Result<()> {
29    for (variant, fields) in entries {
30        let arm = source
31            .variants
32            .iter()
33            .find(|v| v.name.to_string().trim_start_matches("r#") == variant)
34            .ok_or_else(|| format!("unknown enum variant `{variant}`"))?;
35        for field in fields.keys() {
36            if !arm.fields.iter().any(|f| f.key() == *field) {
37                return Err(format!("unknown enum field `{variant}.{field}`"));
38            }
39        }
40    }
41    Ok(())
42}

45    #[allow(clippy::too_many_arguments)]
46    pub(crate) fn enum_input_recipe<C: Codec>(
47        &self,
48        flat: &Flat<C::Schema>,
49        codec: &C,
50        ty: &Type,
51        extra: &C::Extra,
52        entries: &BTreeMap<String, BTreeMap<String, Input<InputRequest>>>,
53        value: bool,
54        stack: &mut Vec<String>,
55    ) -> Result<Input> {
56        let source = flat.payload_enum(ty)?;
57        validate_overrides(source, entries)?;
58        let default = if value { Input::value() } else { Input::auto() };
59        let variants = source
60            .variants
61            .iter()
62            .map(|arm| {
63                let fields = entries.get(arm.name.to_string().trim_start_matches("r#"));
64                arm.fields
65                    .iter()
66                    .map(|field| {
67                        let recipe = fields.and_then(|f| f.get(&field.key())).unwrap_or(&default);
68                        let previous = self.input_role.get();
69                        if value {
70                            self.input_role.set(InputRole::EnumField);
71                        }
72                        let result = self.input(flat, codec, &field.ty, recipe, extra, stack);
73                        self.input_role.set(previous);
74                        result.map_err(|e| {
75                            format!("{}::{}.{}: {e}", type_key(ty), arm.name, field.key())
76                        })
77                    })
78                    .collect::<Result<Vec<_>>>()
79            })
80            .collect::<Result<Vec<_>>>()?;
81        Ok(Input::Enum { variants, value })
82    }
83    #[allow(clippy::too_many_arguments)]
84    pub(crate) fn enum_output_recipe<C: Codec>(
85        &self,
86        flat: &Flat<C::Schema>,
87        codec: &C,
88        ty: &Type,
89        extra: &C::Extra,
90        entries: &BTreeMap<String, BTreeMap<String, Output<OutputRequest>>>,
91        value: bool,
92        stack: &mut Vec<String>,
93        getter: bool,
94    ) -> Result<Output> {
95        let source = flat.payload_enum(ty)?;
96        validate_overrides(source, entries)?;
97        let default = if value {
98            Output::value()
99        } else {
100            Output::auto()
101        };
102        let variants = source
103            .variants
104            .iter()
105            .map(|arm| {
106                let fields = entries.get(arm.name.to_string().trim_start_matches("r#"));
107                arm.fields
108                    .iter()
109                    .map(|field| {
110                        let recipe = fields.and_then(|f| f.get(&field.key())).unwrap_or(&default);
111                        self.output(flat, codec, &field.ty, recipe, extra, stack, getter)
112                            .map_err(|e| {
113                                format!("{}::{}.{}: {e}", type_key(ty), arm.name, field.key())
114                            })
115                    })
116                    .collect::<Result<Vec<_>>>()
117            })
118            .collect::<Result<Vec<_>>>()?;
119        Ok(Output::Enum {
120            variants,
121            value,
122            mode: None,
123        })
124    }
125}

127impl<C: Codec> Compiler<'_, C> {
128    #[allow(clippy::too_many_arguments)]
129    pub(crate) fn enum_input(
130        &self,
131        ty: &Type,
132        recipes: &[Vec<Input>],
133        value: bool,
134        extra: &C::Extra,
135        active: &Condition,
136        names: Option<(&syn::Ident, &syn::Ident)>,
137    ) -> Result<CodecInput<C>> {
138        if matches!(ty, Type::Reference(_)) {
139            return Err("enum-field input constructs an owned value; select reconstruct() to borrow a temporary".into());
140        }
141        let source = self.flat.payload_enum(ty)?;
142        if source.variants.len() != recipes.len() {
143            return Err("wrong enum variant count".into());
144        }
145        i32::try_from(recipes.len()).map_err(|_| "too many enum variants")?;
146        let root = names.map(|(w, _)| w.clone()).unwrap_or_else(|| self.name());
147        let var = names.map(|(_, v)| v.clone()).unwrap_or_else(|| self.name());
148        let tag = format_ident!("{root}_tag");
149        let mut selector = Wire {
150            name: tag.clone(),
151            kind: self.codec.selector(extra)?,
152        };
153        let mut arms = vec![];
154        let mut cases = vec![];
155        let mut storage = TokenStream::new();
156        let mut fallible = self.codec.validates_inputs();
157        for (index, (arm, recipes)) in source.variants.iter().zip(recipes).enumerate() {
158            if arm.fields.len() != recipes.len() {
159                return Err("wrong enum field count".into());
160            }
161            let mut plans = vec![];
162            let mut arguments = vec![];
163            let mut children = vec![];
164            let condition = active.arm(tag.clone(), index as i32);
165            for (field_index, (field, recipe)) in arm.fields.iter().zip(recipes).enumerate() {
166                let wire = format_ident!("{root}_a{index}_f{field_index}");
167                let target = format_ident!("{var}_a{index}_f{field_index}");
168                let converted = format_ident!("{target}_value");
169                let previous = self.input_role.get();
170                if value {
171                    self.input_role.set(InputRole::EnumField);
172                }
173                let result = self.input_named(
174                    &field.ty,
175                    recipe,
176                    extra,
177                    &condition,
178                    Some((&wire, &converted)),
179                );
180                self.input_role.set(previous);
181                let mut child = result
182                    .map_err(|e| format!("{}::{}.{}: {e}", type_key(ty), arm.name, field.key()))?;
183                let name = format_ident!("{}", field.property());
184                if arm.fields.len() != 1 || field.key() != "0" {
185                    self.prefix(&mut child.node, ComponentSegment::Field(field.property()));
186                }
187                child.node = WireNode::Branch {
188                    controls: vec![],
189                    children: vec![child.node],
190                    meta: self.codec.group(GroupKind::Field(&name))?,
191                };
192                storage.extend(std::mem::take(&mut child.storage));
193                fallible |= child.fallible;
194                arguments.push(quote!(#target));
195                children.push(child.node.clone());
196                plans.push((target, child));
197            }
198            let constructed = source.spell(index, &arguments)?;
199            let mut body = quote!(#constructed);
200            for (target, child) in plans.iter().rev() {
201                body = child.bind(target, body)?;
202            }
203            let index = index as i32;
204            cases.push(quote!(#index=>{#body}));
205            let mut arm_node = WireNode::Branch {
206                controls: vec![],
207                children,
208                meta: self.codec.group(GroupKind::Fields)?,
209            };
210            self.prefix(
211                &mut arm_node,
212                ComponentSegment::Named(arm.name.to_string().trim_start_matches("r#").into()),
213            );
214            arms.push(arm_node);
215        }
216        let encoding = self.codec.enum_variants(EnumVariants {
217            flat: self.flat,
218            ty,
219            extra,
220            selector: &mut selector,
221            arms: &mut arms,
222            form: if value {
223                EnumForm::Value
224            } else {
225                EnumForm::Fields
226            },
227            direction: Direction::Input,
228        })?;
229        let invalid = if self.codec.validates_inputs() {
230            quote!(return Err("invalid enum tag".into()))
231        } else {
232            quote!(unsafe { ::core::hint::unreachable_unchecked() })
233        };
234        let mut plan = InputPlan::new(
235            WireNode::Branch {
236                controls: vec![selector],
237                children: arms,
238                meta: encoding.metadata,
239            },
240            quote!(let #var:#ty=match #tag {#(#cases,)* _=>#invalid};),
241            quote!(#var),
242        );
243        plan.storage = storage;
244        plan.fallible = fallible;
245        if value {
246            self.codec.input_enum(
247                self.flat,
248                ty,
249                extra,
250                root,
251                plan,
252                active,
253                self.input_role.get(),
254            )
255        } else {
256            Ok(plan)
257        }
258    }

260    #[allow(clippy::too_many_arguments)]
261    pub(crate) fn enum_output(
262        &self,
263        ty: &Type,
264        recipes: &[Vec<Output>],
265        grouped: bool,
266        mode: Option<ProjectionMode>,
267        extra: &C::Extra,
268        value: TokenStream,
269        borrowed: bool,
270        wire: Option<&syn::Ident>,
271    ) -> Result<CodecOutput<C>> {
272        let owned = mode.map_or(!borrowed, |m| m == ProjectionMode::Owned);
273        if owned && borrowed {
274            return Err("owned enum fields cannot consume a shared-reference occurrence".into());
275        }
276        let temporary_observation = !owned && !borrowed;
277        let observed = format_ident!(
278            "__enum_observed_{}",
279            wire.cloned().unwrap_or_else(|| self.name())
280        );
281        let prepare = if !owned && !borrowed {
282            quote!(let #observed=#value;)
283        } else {
284            TokenStream::new()
285        };
286        let value = if !owned && !borrowed {
287            quote!(&#observed)
288        } else {
289            value
290        };
291        let borrowed = !owned;
292        let source = self.flat.payload_enum(ty)?;
293        if source.variants.len() != recipes.len() {
294            return Err("wrong enum variant count".into());
295        }
296        i32::try_from(recipes.len()).map_err(|_| "too many enum variants")?;
297        let root = wire.cloned().unwrap_or_else(|| self.name());
298        let tag = format_ident!("{root}_tag");
299        let mut selector = Wire {
300            name: tag.clone(),
301            kind: self.codec.selector(extra)?,
302        };
303        let mut plans = vec![];
304        let mut patterns = vec![];
305        for (index, (arm, recipes)) in source.variants.iter().zip(recipes).enumerate() {
306            if arm.fields.len() != recipes.len() {
307                return Err("wrong enum field count".into());
308            }
309            let bindings = (0..arm.fields.len())
310                .map(|i| format_ident!("__{root}_a{index}_b{i}"))
311                .collect::<Vec<_>>();
312            patterns.push(source.spell(
313                index,
314                &bindings.iter().map(|b| quote!(#b)).collect::<Vec<_>>(),
315            )?);
316            let mut child = OutputPlan {
317                fallible: false,
318                context: Default::default(),
319                delivery: Default::default(),
320                commit: TokenStream::new(),
321                statements: TokenStream::new(),
322                values: vec![],
323                outcomes: prebindgen_flat::Outcomes {
324                    success: WireNode::Branch {
325                        controls: vec![],
326                        children: vec![],
327                        meta: self.codec.group(GroupKind::Fields)?,
328                    },
329                    source_error: None,
330                    binding_failure: None,
331                },
332            };
333            for (field_index, (field, recipe)) in arm.fields.iter().zip(recipes).enumerate() {
334                let binding = &bindings[field_index];
335                let wire = format_ident!("{root}_a{index}_f{field_index}");
336                let mut field_plan = self
337                    .with_copy_permission(mode == Some(ProjectionMode::Borrowed), || {
338                        self.with_temporary_receiver(temporary_observation, || {
339                            self.output_named(
340                                &field.ty,
341                                recipe,
342                                extra,
343                                quote!(#binding),
344                                borrowed,
345                                Some(&wire),
346                            )
347                        })
348                    })
349                    .map_err(|e| format!("{}::{}.{}: {e}", type_key(ty), arm.name, field.key()))?;
350                let name = format_ident!("{}", field.property());
351                if arm.fields.len() != 1 || field.key() != "0" {
352                    self.prefix(
353                        &mut field_plan.outcomes.success,
354                        ComponentSegment::Field(field.property()),
355                    );
356                }
357                field_plan.outcomes.success = WireNode::Branch {
358                    controls: vec![],
359                    children: vec![field_plan.outcomes.success],
360                    meta: self.codec.group(GroupKind::Field(&name))?,
361                };
362                append(&mut child, field_plan)?;
363            }
364            self.prefix(
365                &mut child.outcomes.success,
366                ComponentSegment::Named(arm.name.to_string().trim_start_matches("r#").into()),
367            );
368            plans.push(child);
369        }
370        let mut arms = plans
371            .iter_mut()
372            .map(|p| std::mem::take(&mut p.outcomes.success))
373            .collect::<Vec<_>>();
374        let encoding = self.codec.enum_variants(EnumVariants {
375            flat: self.flat,
376            ty,
377            extra,
378            selector: &mut selector,
379            arms: &mut arms,
380            form: if grouped {
381                EnumForm::Value
382            } else {
383                EnumForm::Fields
384            },
385            direction: Direction::Output,
386        })?;
387        if encoding.inactive.len() != plans.len() {
388            return Err("missing inactive enum arm values".into());
389        }
390        let mut defaults = vec![];
391        for (child, absent) in plans.iter().zip(encoding.inactive) {
392            let mut absent = absent
393                .into_iter()
394                .map(|(n, v)| (n.to_string(), v))
395                .collect::<BTreeMap<_, _>>();
396            defaults.push(
397                child
398                    .values
399                    .iter()
400                    .map(|(n, _)| {
401                        absent
402                            .remove(&n.to_string())
403                            .ok_or_else(|| format!("missing inactive enum value `{n}`"))
404                    })
405                    .collect::<Result<Vec<_>>>()?,
406            );
407            if !absent.is_empty() {
408                return Err("unexpected inactive enum output".into());
409            }
410        }
411        // One statically typed active-owner guard, with no boxing or per-arm
412        // Option storage. A failed later conversion drops the selected guard.
413        let guard_type = format_ident!("__PbEnumCommit_{root}");
414        let guard = format_ident!("__{root}_commit");
415        let cases = (0..plans.len())
416            .map(|i| format_ident!("A{i}"))
417            .collect::<Vec<_>>();
418        let parameters = (0..plans.len())
419            .map(|i| format_ident!("F{i}"))
420            .collect::<Vec<_>>();
421        let root_ref = &root;
422        let locals = plans
423            .iter()
424            .enumerate()
425            .flat_map(|(i, p)| {
426                (0..p.values.len()).map(move |j| format_ident!("__{root_ref}_a{i}_value{j}"))
427            })
428            .collect::<Vec<_>>();
429        let values = std::iter::once((tag.clone(), quote!(#tag)))
430            .chain(
431                plans
432                    .iter()
433                    .flat_map(|p| p.values.iter().map(|(n, _)| n.clone()))
434                    .zip(locals.iter().map(|n| quote!(#n))),
435            )
436            .collect();
437        let mut bodies = vec![];
438        let mut fallible = false;
439        for (index, (plan, pattern)) in plans.iter().zip(patterns).enumerate() {
440            let tag = index as i32;
441            let body = &plan.statements;
442            let commit = &plan.commit;
443            let case = &cases[index];
444            let fields = plans
445                .iter()
446                .enumerate()
447                .flat_map(|(i, p)| {
448                    if i == index {
449                        p.values.iter().map(|(_, v)| v.clone()).collect::<Vec<_>>()
450                    } else {
451                        defaults[i].clone()
452                    }
453                })
454                .collect::<Vec<_>>();
455            bodies.push(quote!(#pattern => { #body let values=(#(#fields,)*); (#tag,values,#guard_type::#case(move ||{#commit})) }));
456            fallible |= plan.fallible;
457        }
458        let statements = quote! {
459            #prepare
460            #[allow(non_camel_case_types)] enum #guard_type<#(#parameters),*> { #(#cases(#parameters)),* }
461            let (#tag,(#(#locals,)*),#guard)=match #value {#(#bodies,)*};
462        };
463        let commit = quote!(match #guard {#(#guard_type::#cases(action)=>action(),)*});
464        let plan = OutputPlan {
465            fallible,
466            context: Default::default(),
467            delivery: Default::default(),
468            statements,
469            commit,
470            values,
471            outcomes: prebindgen_flat::Outcomes {
472                success: WireNode::Branch {
473                    controls: vec![selector],
474                    children: arms,
475                    meta: encoding.metadata,
476                },
477                source_error: None,
478                binding_failure: None,
479            },
480        };
481        if grouped {
482            self.codec.output_enum(self.flat, ty, extra, root, plan)
483        } else {
484            Ok(plan)
485        }
486    }
