1//! What a binding declared, in terms neither language owns.
2//!
3//! A frontend turns its own declaration storage into these — one
4//! [`Declaration`] per thing the user asked for, inside the
5//! [`BindingRequests`](crate::BindingRequests) it hands the engine. The
6//! declaration is what the report accounts for; the request carries the target's
7//! configuration for it.
8
9use prebindgen_flat::flat::{Element, Flat, TypeKey};
10use serde::Serialize;
11
12/// What a [`Declaration`] declares: a function, a type, a constant, a callback
13/// or a conversion.
14///
15/// This is the coarse, language-neutral category — the five things any binding
16/// can be made of. It is deliberately coarser than an adapter's own vocabulary:
17/// `prebindgen-c` declares a type with `opaque_ptr` or `data_struct`, and
18/// `prebindgen-jni` with `ptr_class` or `data_class`, but all four produce a
19/// `Type`. The word the adapter used is kept separately in
20/// [`Declaration::representation`] and printed back verbatim.
21///
22/// It is bookkeeping, not a plan: what a declaration is planned from comes from
23/// its [`Origin`] — the captured item there is to work with — and the kind only
24/// picks between planners where one captured item backs two surfaces, as a
25/// Kotlin `val` read through a nullary function is planned as that function.
26/// Two things depend on the category:
27///
28/// * **Identity.** An [`Origin`] prints as `<kind>:<name>`, and the kind is
29///   what keeps the origins' several naming spaces apart: an origin may be a
30///   captured item's name, a type key, a callback's signature or a name the
31///   binding coined, so `type:Foo` and `conversion:Foo` are two declarations
32///   about one Rust type, and a reader of the report can tell which of them an
33///   entry belongs to.
34/// * **Report layout.** [`Report`](crate::Report) groups and sorts by it, so a
35///   report reads types first, then conversions, callbacks, constants and
36///   functions.
37///
38/// It says what the *target* language gets, not what the captured Rust source
39/// held — a Kotlin constant may be backed by a captured Rust function. That
40/// second question is the [`Origin`] variant.
41#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize)]
42#[serde(rename_all = "snake_case")]
43pub enum DeclarationKind {
44    /// A `#[prebindgen]` function the binding exports a wrapper for.
45    Function,
46    /// A `#[prebindgen]` type the binding gives a foreign representation.
47    Type,
48    /// A `#[prebindgen]` constant the binding exposes as a foreign constant.
49    Const,
50    /// A callback signature the binding exports as a foreign callable.
51    Callback,
52    /// A declared conversion between a Rust type and its wire form.
53    Conversion,
54}
55
56impl DeclarationKind {
57    /// The prefix an [`Origin`] prints with, and the report label.
58    pub fn as_str(self) -> &'static str {
59        match self {
60            DeclarationKind::Function => "fn",
61            DeclarationKind::Type => "type",
62            DeclarationKind::Const => "const",
63            DeclarationKind::Callback => "callback",
64            DeclarationKind::Conversion => "conversion",
65        }
66    }
67}

69/// What a declaration is named after, and what the engine plans it from.
70///
71/// A name alone says neither: `Sample` may be a captured type, a type key the
72/// binding coined for something the source never exported, or the name of a
73/// Kotlin constant built from an expression. The variant says which, and it
74/// says it once — a declaration's [`DeclarationKind`] and what its name must
75/// find in the captured source both follow from the
76/// variant instead of being stated beside it, so they cannot disagree and a
77/// pair that means nothing (a callback backed by a captured constant, a
78/// function the binding both defines and selects out of the source) cannot be
79/// written down.
80///
81/// [`generate`](crate::generate) routes on this: each planner is reached by its
82/// own variants and is handed the captured item they name, rather than a kind
83/// to decode and a name to look up.
84///
85/// It is also the declaration's identity — what an outcome is keyed by, what a
86/// [`Position`](crate::target::Position) is rooted at, what a target's
87/// [`Requirement`](crate::target::Requirement) resolves to. Stable across runs
88/// and across pipelines, so a report, a build script and a capability-selected
89/// test section can all name the same declaration: it is what the *source*
90/// calls the thing, not what the target does, and a rename on the foreign side
91/// must not silently retire a test's requirement. `<kind>:<name>` —
92/// `type:Stamp`, `fn:stamp_sum` — is how it prints and how the report writes
93/// it, and that spelling is a rendering: nothing reads an origin back out of
94/// it. Origins order the way they print — by kind, then by name — so a report
95/// sorted by origin reads as its ids read.
96#[derive(Clone, Debug, PartialEq, Eq, Hash)]
97pub enum Origin {
98    /// A captured `#[prebindgen]` function, exported as a foreign function.
99    Function(syn::Ident),
100    /// A foreign function the binding defines itself — `fun!(crate::x).sig(..)`
101    /// in the JNI frontend. Nothing in the captured source names it.
102    LocalFunction(String),
103    /// A captured `#[prebindgen]` constant, exposed as a foreign constant.
104    Const(syn::Ident),
105    /// A foreign constant read by calling a captured nullary function —
106    /// `constant!(X).fun(fun!(f))`. The target renders a constant; the engine
107    /// plans the function behind it.
108    ConstFromFunction(syn::Ident),
109    /// A foreign constant the binding computes rather than reads —
110    /// `constant!(X).expr(..)`. No captured item holds its value.
111    LocalConst(String),
112    /// A captured `#[prebindgen]` type, given a foreign representation.
113    Type(TypeKey),
114    /// A type the binding gives a representation although the source never
115    /// exported it — `String` crossing as an opaque handle.
116    LocalType(TypeKey),
117    /// A callback signature the binding exports as a foreign callable. No
118    /// captured item names one, so the signature is the name it goes by.
119    Callback(String),
120    /// A declared conversion between a Rust type and its wire form, defined by
121    /// the binding.
122    Conversion(TypeKey),
123}
124
125impl Origin {
126    /// What the target gets — see [`DeclarationKind`].
127    pub fn kind(&self) -> DeclarationKind {
128        match self {
129            Origin::Function(_) | Origin::LocalFunction(_) => DeclarationKind::Function,
130            Origin::Const(_) | Origin::ConstFromFunction(_) | Origin::LocalConst(_) => {
131                DeclarationKind::Const
132            }
133            Origin::Type(_) | Origin::LocalType(_) => DeclarationKind::Type,
134            Origin::Callback(_) => DeclarationKind::Callback,
135            Origin::Conversion(_) => DeclarationKind::Conversion,
136        }
137    }
138
139    /// The captured element this origin names, if the model holds it.
140    ///
141    /// Captured items live in one flat namespace holding functions, types and
142    /// constants, so the name alone finds any element; the variant is what says
143    /// whether the element found is the one the declaration meant. A constant
144    /// read through a function is looked up among the functions, which is what
145    /// separates [`Self::ConstFromFunction`] from [`Self::Const`]. A
146    /// binding-local origin names no captured item and so finds none — which
147    /// is not the same as a missing one, and [`Self::missing_from`] is the
148    /// question to ask about presence.
149    pub(crate) fn captured<'f>(&self, flat: &'f Flat) -> Option<&'f Element> {
150        let element = flat.element(&self.name())?;
151        matches!(
152            (self, element),
153            (
154                Origin::Function(_) | Origin::ConstFromFunction(_),
155                Element::Function(_)
156            ) | (Origin::Type(_), Element::Type(_))
157                | (Origin::Const(_), Element::Constant(_))
158        )
159        .then_some(element)
160    }
161
162    /// Whether the binding defines this itself: a callback signature, a
163    /// binding-local conversion helper, a function or constant of its own, or a
164    /// type the target represents although the source never exported it
165    /// (`String` as an opaque handle). Such an origin requires nothing of the
166    /// model.
167    pub fn is_binding_local(&self) -> bool {
168        matches!(
169            self,
170            Origin::LocalFunction(_)
171                | Origin::LocalConst(_)
172                | Origin::LocalType(_)
173                | Origin::Callback(_)
174                | Origin::Conversion(_)
175        )
176    }
177
178    /// Whether the model lacks what this origin must name.
179    ///
180    /// Naming the wrong kind — `.fun(fun!(x))` where the source captured
181    /// `const x` — counts as missing: it is an error in the binding, and the
182    /// engine fails the run over it instead of reporting a skip. False for a
183    /// binding-local origin.
184    pub(crate) fn missing_from(&self, flat: &Flat) -> bool {
185        !self.is_binding_local() && self.captured(flat).is_none()
186    }
187
188    /// The word a refusal uses for what this origin must name.
189    pub(crate) fn describe_captured(&self) -> &'static str {
190        match self {
191            Origin::Function(_) | Origin::ConstFromFunction(_) => "function",
192            Origin::Const(_) => "constant",
193            Origin::Type(_) => "type",
194            _ => "binding-local item",
195        }
196    }
197
198    /// The name it goes by — the part of the printed form after the kind.
199    pub fn name(&self) -> String {
200        match self {
201            Origin::Function(ident) | Origin::Const(ident) | Origin::ConstFromFunction(ident) => {
202                ident.to_string()
203            }
204            Origin::Type(key) | Origin::LocalType(key) | Origin::Conversion(key) => {
205                key.as_str().to_string()
206            }
207            Origin::LocalFunction(name) | Origin::LocalConst(name) | Origin::Callback(name) => {
208                name.clone()
209            }
210        }
211    }
212}

214impl Ord for Origin {
215    /// By kind, then by name — the printed order. A captured and a
216    /// binding-local origin of one kind and name print alike and are still
217    /// distinct, so the variant breaks that tie last, and `Ord` agrees with
218    /// `Eq`.
219    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
220        fn variant(origin: &Origin) -> u8 {
221            match origin {
222                Origin::Function(_) => 0,
223                Origin::LocalFunction(_) => 1,
224                Origin::Const(_) => 2,
225                Origin::ConstFromFunction(_) => 3,
226                Origin::LocalConst(_) => 4,
227                Origin::Type(_) => 5,
228                Origin::LocalType(_) => 6,
229                Origin::Callback(_) => 7,
230                Origin::Conversion(_) => 8,
231            }
232        }
233        self.kind()
234            .cmp(&other.kind())
235            .then_with(|| self.name().cmp(&other.name()))
236            .then_with(|| variant(self).cmp(&variant(other)))
237    }
238}
239
240impl PartialOrd for Origin {
241    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
242        Some(self.cmp(other))
243    }
244}
245
246impl std::fmt::Display for Origin {
247    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
248        write!(f, "{}:{}", self.kind().as_str(), self.name())
249    }
250}

252impl Serialize for Origin {
253    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
254        serializer.collect_str(self)
255    }
256}

258/// One declaration, as the report accounts for it.
259///
260/// Built by the adapter with [`Self::new`] and read back through the accessors.
261/// The [`Origin`] is both the run's key for it and what the engine plans it
262/// from; everything else about its identity — its kind, what it must find in
263/// the captured source, the name it goes by — is read back out of the origin,
264/// so nothing is stated twice and nothing can disagree.
265#[derive(Clone, Debug, PartialEq, Eq)]
266pub struct Declaration {
267    origin: Origin,
268    placement: String,
269    representation: String,
270}
271
272impl Serialize for Declaration {
273    /// Flat, and with the identity spelled out: the report carries `id` (the
274    /// origin as it prints), `kind` and `rust_origin` as separate columns, and
275    /// a reader of the JSON should not have to split the id to get at the last
276    /// two.
277    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
278        use serde::ser::SerializeStruct;
279        let mut entry = serializer.serialize_struct("Declaration", 5)?;
280        entry.serialize_field("id", &self.origin)?;
281        entry.serialize_field("kind", &self.origin.kind())?;
282        entry.serialize_field("rust_origin", &self.origin.name())?;
283        entry.serialize_field("placement", &self.placement)?;
284        entry.serialize_field("representation", &self.representation)?;
285        entry.end()
286    }
287}
288
289impl Declaration {
290    /// Declare `origin`, which the target places at `placement`.
291    pub fn new(
292        origin: Origin,
293        placement: impl Into<String>,
294        representation: impl Into<String>,
295    ) -> Self {
296        Declaration {
297            origin,
298            placement: placement.into(),
299            representation: representation.into(),
300        }
301    }
302
303    /// What this declaration is, what the engine plans it from, and what the
304    /// run keys it by — see [`Origin`].
305    pub fn origin(&self) -> &Origin {
306        &self.origin
307    }
308
309    /// Which kind of declaration it is.
310    pub fn kind(&self) -> DeclarationKind {
311        self.origin.kind()
312    }
313
314    /// What the Rust source calls it (`Calculator`, `calculator_new`), or the
315    /// signature for a callback that has no name of its own.
316    pub fn rust_origin(&self) -> String {
317        self.origin.name()
318    }
319
320    /// Where it is meant to land in the target language, spelled the way that
321    /// language spells it: `calculator_t`, `io.zenoh.jni.Session`.
322    pub fn placement(&self) -> &str {
323        &self.placement
324    }
325
326    /// The declarator that produced it (`opaque_ptr`, `data_class`, `fun`, …) —
327    /// the adapter's own word, printed back verbatim.
328    pub fn representation(&self) -> &str {
329        &self.representation
330    }
