1//! Declaration language and composition for preparing conversion slots.
2//!
3//! [`Declarations::fill`], [`bind_input`] and [`bind_output`] build conversions
4//! directly in a [`Bindings`] model. Recipes and codec input data are used only
5//! during construction; slots retain wires and code needed by wrapper writers.
6//! Generation is owned by the language generator;
7//! this crate owns only declaration resolution and slot preparation.
8//! Each [`Wire`] carries the adapter's own description in [`Wire::kind`].
9//! Reusable value and boxed-allocation leaves live in `prebindgen-conversions`.
10#![warn(missing_docs)]

36/// A recipe for converting boundary inputs into one source argument.
37///
38/// Binding expands the recipe into wires and Rust conversion code.
39#[derive(Clone, Debug)]
40pub enum Input {
41    /// Convert an optional payload, gating every conversion phase.
42    Optional(Box<Input>),
43    /// Convert the occurrence type itself with [`Codec::input_leaf`].
44    Leaf,
45    /// Construct a named record from children in field order; an empty list builds unit.
46    Fields(Vec<Input>),
47    /// Call a source function after preparing its arguments with the child recipes.
48    Constructor {
49        /// Unqualified source function name; its result must match the occurrence type.
50        function: String,
51        /// Recipes in the constructor's parameter order.
52        args: Vec<Input>,
53    },
54    /// Choose an owned input using an `i32` arm index followed by all arms' wires.
55    /// Inactive arms are neither validated nor acquired; indices are zero-based.
56    Selector(Vec<Input>),
57}
58impl Input {
59    /// Construct a record from field recipes, or unit from an empty list.
60    ///
61    /// `fields` holds one recipe per field of the source record, in declaration
62    /// order. Binding verifies the count against the record.
63    pub fn fields(fields: impl IntoIterator<Item = Input>) -> Self {
64        Self::Fields(fields.into_iter().collect())
65    }
66    /// Call a named source constructor with the supplied argument recipes.
67    ///
68    /// Binding checks its result type and argument count. Argument validation and
69    /// ownership acquisition precede the source call.
70    ///
71    /// # Arguments
72    ///
73    /// * `name` — unqualified name of a source function whose result type is the
74    ///   input's type.
75    /// * `args` — one recipe per parameter of that function, in order.
76    pub fn constructor(name: impl Into<String>, args: impl IntoIterator<Item = Input>) -> Self {
77        Self::Constructor {
78            function: name.into(),
79            args: args.into_iter().collect(),
80        }
81    }
82    /// Choose between owned representations of the same input type.
83    ///
84    /// `arms` holds one recipe per alternative; each must produce the input's
85    /// type. The caller passes the zero-based index of the chosen alternative in an
86    /// extra wire ahead of all arms' wires.
87    ///
88    /// Requires at least one arm. Each arm retains its own wires; this does not
89    /// merge equivalent wire parameters or generate destination-language overloads.
90    pub fn selector(arms: impl IntoIterator<Item = Input>) -> Self {
91        Self::Selector(arms.into_iter().collect())
92    }
93}
94/// A result conversion recipe, interpreted together with its source type and codec input data.
95///
96/// `Fields` observes fields; `OwnedFields` moves them. Explicit `Projected`
97/// leaves select borrowed or owned conversion independently of type contents.
98#[derive(Clone, Debug)]
99pub enum Output {
100    /// Convert an optional payload without preparing absent values.
101    Optional(Box<Output>),
102    /// Convert the occurrence type itself with [`Codec::output_leaf`].
103    Leaf,
104    /// Observe named fields in declaration order; an empty list represents unit.
105    Fields(Vec<Output>),
106    /// Move named fields into their output conversions.
107    OwnedFields(Vec<Output>),
108    /// Convert a leaf with an explicit ownership policy.
109    Projected(Projection),
110    /// Call a consuming accessor, then convert its owned result.
111    Into {
112        /// Source accessor taking exactly the owned occurrence type.
113        function: String,
114        /// Conversion of its owned result.
115        output: Box<Output>,
116    },
117    /// A shared-borrow accessor, followed by the output conversion of its result.
118    Getter {
119        /// Source getter taking exactly one shared reference to the current type.
120        function: String,
121        /// Conversion of the getter's owned result; borrowed results are unsupported.
122        output: Box<Output>,
123    },
124    /// Produce several views of one value. At most one child may consume it.
125    /// Observations run before consumption, while wire order follows this list.
126    Combine(Vec<Output>),
127}
128impl Output {
129    /// Observe record fields in declaration order; use an empty list for unit.
130    ///
131    /// `fields` holds one recipe per field of the source record, in declaration order.
132    pub fn fields(fields: impl IntoIterator<Item = Output>) -> Self {
133        Self::Fields(fields.into_iter().collect())
134    }
135    /// Observe a value through a source getter and convert the getter's result.
136    ///
137    /// The getter must take exactly `&T` for the current type `T`, and return an
138    /// owned value. Binding rejects mismatched signatures.
139    ///
140    /// # Arguments
141    ///
142    /// * `name` — unqualified name of the source getter.
143    /// * `output` — recipe for the getter's result.
144    pub fn getter(name: impl Into<String>, output: Output) -> Self {
145        Self::Getter {
146            function: name.into(),
147            output: Box::new(output),
148        }
149    }
150    /// Produce multiple outputs from one value.
151    ///
152    /// `parts` holds one recipe per view of the value; their wires appear in this
153    /// order. Binding rejects an empty list or more than one consuming part. The
154    /// framework uses [`Codec::consumes`] to order observations before a move.
155    pub fn combine(parts: impl IntoIterator<Item = Output>) -> Self {
156        Self::Combine(parts.into_iter().collect())
157    }
158    fn consumes<C: Codec>(&self, ty: &Type, extra: &C::Extra, codec: &C) -> bool {
159        match self {
160            Self::Optional(child) => child.consumes(option_inner(ty).unwrap_or(ty), extra, codec),
161            Self::Leaf => codec.consumes(ty, extra),
162            Self::Projected(p) => p.mode == ProjectionMode::Owned,
163            Self::OwnedFields(_) | Self::Into { .. } => true,
164            Self::Combine(p) => p.iter().any(|p| p.consumes(ty, extra, codec)),
165            _ => false,
166        }
167    }
168}

170/// Language-specific leaf conversions. Recursive composition stays in this declaration compiler.
171///
172/// A leaf is a source type the adapter converts itself: a value copied across
173/// the boundary or an object kept in Rust. `prebindgen-conversions` provides helpers
174/// for both representations. The
175/// adapter chooses by type; a recipe only says where the leaves are.
176///
177/// Input leaves must validate without consuming in `validation`, then
178/// acquire ownership without failure or source calls in `acquire`. Each acquired
179/// value needs an ordinary Rust owner so later failures drop it. `statements`
180/// can construct arguments from those owners. Every input wire must retain
181/// the supplied activation condition. Output leaves keep ownership in Rust
182/// locals until their infallible `commit` runs after successful delivery.
183///
184/// # Implementing a codec
185///
186/// This minimal codec uses Rust values as its carriers. It illustrates converter
187/// construction without a foreign ABI or object storage. A language adapter
188/// additionally writes wrappers and chooses native and destination-language
189/// carrier types for its wire kinds.
190///
191/// ```
192/// use prebindgen_flat::{Result, validate_bindings};
193/// use prebindgen_conversions::value;
194/// use prebindgen_declarations::{
195///     Codec, CodecInput, CodecOutput, Declarations,
196///     Bindings, fun, expand_param, expand_return, bind_input, Input,
197/// };
198/// use proc_macro2::TokenStream;
199/// use quote::quote;
200/// use syn::{Ident, Type};
201///
202/// use prebindgen_declarations::Condition;
203/// fn carrier(kind: value::Value) -> Type {
204///     match kind {
205///         value::Value::Scalar(p) => p.rust_type(),
206///         value::Value::Text => syn::parse_quote!(String),
207///         value::Value::Bytes => syn::parse_quote!(Vec<u8>),
208///     }
209/// }
210/// struct RustCarriers;
211/// impl Codec for RustCarriers {
212///     type Niche = ();
213///     type Group = ();
214///     type Extra = ();
215///     type Context = ();
216///     type Input = CodecInput<Self>;
217///     type Output = CodecOutput<Self>;
218///     type Wire = Type; // The Rust carrier type is the complete wire description.
219///
220///     fn optional_group(&self, _: Ident, _: Ident, _: Option<()>) {}
221///     fn field_group(&self, _: Ident) {}
222///     fn fields_group(&self) {}
223///     fn niche(&self, _: &Self::Wire) -> Option<()> { None }
224///     fn set_niche(&self, _: &mut Self::Wire, _: Option<()>) {}
225///     fn selector(&self, _: &()) -> Self::Wire { syn::parse_quote!(i32) }
226///     fn consumes(&self, _: &Type, _: &()) -> bool { false }
227///     fn input_leaf(
228///         &self, ty: &Type, _: &(), wire: Ident, var: Ident, active: &Condition,
229///     ) -> Result<CodecInput<Self>> {
230///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
231///         let decode = quote!(#wire.clone());
232///         value::input(ty, wire, var, &active.rust(), decode, carrier(kind))
233///     }
234///     fn output_leaf(
235///         &self, ty: &Type, _: &(), wire: Ident, v: TokenStream, borrowed: bool,
236///     ) -> Result<CodecOutput<Self>> {
237///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
238///         value::output(ty, wire, v, borrowed, |v| v, carrier(kind))
239///     }
240/// }
241///
242/// let source = syn::parse_file(r#"
243///     pub fn from_i32(value: i32) -> i64 { value as i64 }
244///     pub fn to_i32(value: &i64) -> i32 { *value as i32 }
245///     pub fn echo(value: i64) -> i64 { value }
246/// "#)?;
247/// let mut flat = Bindings::<Type>::parse(
248///     source.items.into_iter().map(|item| (item, Default::default())),
249/// )?;
250/// let defaults = Declarations::new(&flat, vec![
251///     expand_param!(i64).variant(fun!(from_i32)).variant_self().into(),
252///     expand_return!(i64).field(fun!(to_i32)).field_self().into(),
253/// ])?;
254/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
255/// let function = flat.function("echo")?;
256/// // Input: selector, constructor argument, direct value. Output: two values.
257/// assert_eq!(function.params[0].ty.slot.as_ref().unwrap().layout.len(), 3);
258/// assert_eq!(function.ret.slot.as_ref().unwrap().layout.len(), 2);
259/// // Replace just this parameter's converter; later default filling preserves it.
260/// bind_input(&mut flat, &RustCarriers, "echo", "value", &Input::Leaf, &())?;
261/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
262/// validate_bindings(&flat)?;
263/// let input = flat.function("echo")?.params[0].ty.slot.as_ref().unwrap();
264/// assert_eq!(input.layout.len(), 1);
265/// assert_eq!(prebindgen_flat::type_key(&input.layout[0].kind), "i64");
266/// assert!(input.resources.is_empty()); // No foreign ownership in this codec.
267/// # Ok::<(), Box<dyn std::error::Error>>(())
268/// ```
269pub trait Codec {
270    /// Adapter-owned description of an absent wire representation. Use `()` if unused.
271    /// The shared compiler never interprets this metadata.
272    type Niche: Clone;
273    /// Complete adapter-owned metadata for a parent node in the wire tree.
274    /// The shared compiler creates it through the group hooks below and never
275    /// interprets it. Use `()` when no group information is needed.
276    type Group: Clone;
277    /// Describe an optional value in the adapter's own metadata format.
278    /// `presence` is the normalized 0/1 binding used by payload conditions.
279    /// `discriminator` names a descendant wire; `niche` is its reserved absent
280    /// representation, or `None` when a separate flag is the first child.
281    /// With a flag, `presence` and `discriminator` name that same wire.
282    fn optional_group(
283        &self,
284        presence: syn::Ident,
285        discriminator: syn::Ident,
286        niche: Option<Self::Niche>,
287    ) -> Self::Group;
288    /// Describe one named source field, whose subtree may contain any number of wires.
289    fn field_group(&self, name: syn::Ident) -> Self::Group;
290    /// Describe a record or unit value's ordered collection of field subtrees.
291    /// The shared helpers check field-name uniqueness before calling this hook.
292    fn fields_group(&self) -> Self::Group;
293
294    /// Input data borrowed while building one occurrence's conversion; use `()` if unused.
295    /// The same value is passed through nested conversions. It is not stored in Flat;
296    /// hooks retain only the data needed for emission in their code and wires.
297    type Extra;
298    /// Context supplied by the language generator when emitting stored templates.
299    type Context: prebindgen_conversions::TemplateContext + 'static;
300    /// Final self-contained input converter installed after preparation.
301    type Input: prebindgen_flat::Conversion + From<CodecInput<Self>>;
302    /// Final self-contained output converter installed after preparation.
303    type Output: prebindgen_flat::Conversion + From<CodecOutput<Self>>;
304    /// Preserve a logical path in private preparation metadata when the codec needs it.
305    /// The final converter decides how to use this mapping; Flat never interprets it.
306    /// Called from the innermost component outward; prepend each segment to its path.
307    fn prefix_component(&self, _wire: &mut Self::Wire, _segment: ComponentSegment) {}
308
309    /// Complete description of one wire, including its type and adapter metadata.
310    /// Grouping belongs to parent nodes; no wire kind contains child wires.
311    type Wire: prebindgen_flat::WireKind + Clone + std::fmt::Debug;
312    /// Describe an `i32` selector tag or optional presence flag.
313    ///
314    /// Selector arm indices start at zero; optional flags use 0 and 1. A codec
315    /// may advertise excluded negative values as niches for enclosing options.
316    /// A newly added flag retains its own niches; inactive payload wires do not.
317    /// `extra` is the occurrence data of the conversion that contains the selector.
318    fn selector(&self, extra: &Self::Extra) -> Self::Wire;
319    /// Whether a leaf output consumes the current source value.
320    ///
321    /// This must agree with [`Self::output_leaf`]. It governs move ordering and
322    /// double-consumption rejection in [`Output::Combine`], not input ownership.
323    /// A non-consuming leaf receives a shared reference with `borrowed` set.
324    ///
325    /// # Arguments
326    ///
327    /// * `ty` — owned source type of the value being delivered.
328    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`].
329    fn consumes(&self, ty: &Type, extra: &Self::Extra) -> bool;
330    /// Prepare a leaf input of source type `ty` using occurrence data `extra`.
331    ///
332    /// All fallible checks belong in [`InputPlan::validation`]; do not
333    /// consume values there. Return a generation error when this type or access
334    /// mode is unsupported.
335    ///
336    /// # Arguments
337    ///
338    /// * `ty` — the source parameter type at this position, as written: `T`,
339    ///   `&T` or `&mut T`.
340    /// * `extra` — occurrence data of the conversion, as passed to [`bind_input`]
341    ///   or [`Declarations::fill`].
342    /// * `wire` — fresh identifier for the boundary parameter this leaf adds.
343    ///   A leaf that needs more wires or locals derives their names from it.
344    /// * `var` — fresh identifier for the Rust local that holds the converted value.
345    ///   [`InputPlan::argument`] normally refers to it.
346    /// * `active` — the selector arms under which this leaf is used. Copy it to every
347    ///   returned wire and resource, and guard validation and acquisition with it.
348    fn input_leaf(
349        &self,
350        ty: &Type,
351        extra: &Self::Extra,
352        wire: syn::Ident,
353        var: syn::Ident,
354        active: &Condition,
355    ) -> Result<CodecInput<Self>>;
356    /// Convert a leaf with explicit ownership, independently of its usual policy.
357    ///
358    /// The compiler supplies either an owned value or a shared reference according
359    /// to `projection.mode`. The default supports leaves without field overrides.
360    fn output_projected(
361        &self,
362        ty: &Type,
363        extra: &Self::Extra,
364        wire: syn::Ident,
365        value: TokenStream,
366        projection: &Projection,
367    ) -> Result<CodecOutput<Self>> {
368        if !projection.fields.is_empty() {
369            return Err("this output converter does not support field projection overrides".into());
370        }
371        let mut payload = ty;
372        while let Some(inner) = option_inner(payload) {
373            payload = inner;
374        }
375        if projection.mode == ProjectionMode::Owned && matches!(payload, Type::Reference(_)) {
376            return Err("owned projection cannot consume a reference payload".into());
377        }
378        self.output_leaf(
379            ty,
380            extra,
381            wire,
382            value,
383            projection.mode == ProjectionMode::Borrowed,
384        )
385    }
386    /// The unused representation currently available on this wire.
387    /// Initialize it when creating the wire and return the state last written by
388    /// `set_niche`. Never offer a representation already used by an inner option
389    /// or accepted as a present leaf payload. It may describe one value or a set.
390    fn niche(&self, wire: &Self::Wire) -> Option<Self::Niche>;
391    /// Update the unused representation stored in a wire's own description.
392    /// The compiler calls this after reserving a niche or disabling a conditional wire.
393    /// Codecs must preserve this update in their wire description. A codec with
394    /// no niches can implement this as a no-op and always return `None` from `niche`.
395    fn set_niche(&self, wire: &mut Self::Wire, niche: Option<Self::Niche>);
396    /// Recognize presence without decoding the payload. Called on original inner metadata.
397    /// The expression must be total and side-effect-free for every initialized wire
398    /// value, including inactive placeholders; it must not dereference raw pointers.
399    /// It returns false for every representation reserved by this niche.
400    fn is_present(
401        &self,
402        _wire: &Self::Wire,
403        _niche: &Self::Niche,
404        _value: TokenStream,
405    ) -> Result<TokenStream> {
406        Err("niche presence testing is unsupported by this codec".into())
407    }
408    /// Remaining excluded representation after reserving `niche` for one optional layer.
409    /// The compiler stores the result in the wire through `set_niche`. Returning
410    /// `None` exhausts this domain; enclosing options then use another wire or a flag.
411    fn remaining_niche(&self, _wire: &Self::Wire, _niche: &Self::Niche) -> Option<Self::Niche> {
412        None
413    }
414    /// Inert wire expression used for an absent payload, optionally using a niche.
415    fn absent_value(
416        &self,
417        _wire: &Self::Wire,
418        _niche: Option<&Self::Niche>,
419        _input: bool,
420    ) -> Result<TokenStream> {
421        Err("optional values are unsupported by this codec".into())
422    }
423    /// Complete a composed parameter plan before it is stored in a slot.
424    /// Called once per bound parameter, including records and selectors.
425    fn finish_input(&self, _plan: &mut CodecInput<Self>, _extra: &Self::Extra) -> Result<()> {
426        Ok(())
427    }
428    /// Complete a composed result plan, including unit and multi-wire results.
429    /// Install call-level delivery here, after all output expressions are known.
430    fn finish_output(&self, _plan: &mut CodecOutput<Self>, _extra: &Self::Extra) -> Result<()> {
431        Ok(())
432    }
433    /// Prepare delivery of the Rust expression `value` as one or more wires.
434    ///
435    /// Keep unpublished allocations under Rust ownership and transfer them only in
436    /// [`OutputPlan::commit`]. Return a generation error for unsupported
437    /// conversions. Install final delivery in [`Codec::finish_output`], rather
438    /// than on nested leaf plans; recursive composition rejects leaf delivery.
439    ///
440    /// # Arguments
441    ///
442    /// * `ty` — the owned source type of the value.
443    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`]
444    ///   or [`Declarations::fill`].
445    /// * `wire` — fresh identifier for the delivered value. A leaf that needs more
446    ///   wires or locals derives their names from it.
447    /// * `value` — Rust expression producing the value to deliver.
448    /// * `borrowed` — `true` when `value` is a shared reference to `ty`, such as a
449    ///   record field observed in place; do not move out of it. `false` when
450    ///   `value` is owned.
451    fn output_leaf(
452        &self,
453        ty: &Type,
454        extra: &Self::Extra,
455        wire: syn::Ident,
456        value: TokenStream,
457        borrowed: bool,
458    ) -> Result<CodecOutput<Self>>;
459}
460/// A flat model whose slots contain only conversion data for wrapper emission.
461///
462/// Unbound source helpers have `None` slots. Bound functions need `Some` for their
463/// result and every parameter; [`prebindgen_flat::validate_bindings`] checks this before emission.
464/// Record fields retain only names and source types. Recipes and [`Codec::Extra`]
465/// are construction inputs, not stored in this model.
466///
467/// Output conversion data cannot be attached to an input position:
468/// ```compile_fail
469/// use prebindgen_declarations::Bindings;
470/// fn swap(flat: &mut Bindings<syn::Type>) {
471///     let function = flat.function_mut("foo").unwrap();
472///     function.params[0].ty.slot = function.ret.slot.clone();
473/// }
474/// ```
475/// Nor can a recipe be stored in a slot:
476/// ```compile_fail
477/// use prebindgen_declarations::{Bindings, Input};
478/// fn configure(flat: &mut Bindings<syn::Type>) {
479///     flat.function_mut("foo").unwrap().params[0].ty.slot = Some(Input::Leaf);
480/// }
481/// ```
482pub type Bindings<W, C = (), G = ()> = Flat<InputPlan<W, C, G>, OutputPlan<W, C, G>>;
483
484/// Build and attach a converter to a named function parameter.
485///
486/// Resolves the occurrence's source type from `flat`, checks recipe compatibility,
487/// and recursively composes wires and code. Nested conversions use the supplied
488/// `extra`, not data from referenced constructors or getters. Recipes and `extra`
489/// are borrowed only for this call. A failure leaves the existing slot unchanged.
490/// Rebinding an occurrence replaces its converter without changing other slots.
491/// Generated names are scoped by parameter position, so separate calls cannot
492/// collide within the wrapper. Keep the function signature fixed after binding.
493/// Check completeness with [`prebindgen_flat::validate_bindings`].
494///
495/// # Arguments
496///
497/// * `flat` — the model holding the function; the parameter's slot is replaced.
498/// * `codec` — the adapter's leaf conversions.
499/// * `function` — unqualified name of the source function.
500/// * `parameter` — name of the parameter, as written in the source.
501/// * `recipe` — how to build the argument from boundary inputs.
502/// * `extra` — occurrence data passed to every codec call for this conversion.
503///
504/// Input binding rejects output recipes:
505/// ```compile_fail
506/// use prebindgen_declarations::{bind_input, Codec, Output};
507/// use prebindgen_flat::Flat;
508/// fn wrong<C: Codec>(flat: &mut Flat<C::Input, C::Output>, codec: &C, extra: &C::Extra) {
509///     bind_input(flat, codec, "foo", "x", &Output::Leaf, extra).unwrap();
510/// }
511/// ```
512pub fn bind_input<C: Codec>(
513    flat: &mut Flat<C::Input, C::Output>,
514    codec: &C,
515    function: &str,
516    parameter: &str,
517    recipe: &Input,
518    extra: &C::Extra,
519) -> Result<()> {
520    let f = flat.function(function)?;
521    let index = f
522        .params
523        .iter()
524        .position(|p| p.name == parameter)
525        .ok_or_else(|| format!("`{function}`: unknown parameter `{parameter}`"))?;
526    let mut conversion = Compiler::new(flat, codec, format!("p{index}")).input(
527        &f.params[index].ty.ty,
528        recipe,
529        extra,
530        &Condition::default(),
531    )?;
532    codec.finish_input(&mut conversion, extra)?;
533    conversion.layout.validate()?;
534    flat.function_mut(function)?.params[index].ty.slot = Some(conversion.into());
535    Ok(())
536}

538/// Build and attach a converter to a function result, replacing any existing one.
539///
540/// Uses the same construction and failure rules as [`bind_input`]. Output code
541/// uses [`prebindgen_conversions::source_result_expr`] for the source call's result local.
542/// The recipe is checked for compatible types and signatures, and for repeated
543/// consumption of the same value. Unit results use `Output::fields([])`.
544///
545/// # Arguments
546///
547/// * `flat` — the model holding the function; the result's slot is replaced.
548/// * `codec` — the adapter's leaf conversions.
549/// * `function` — unqualified name of the source function.
550/// * `recipe` — how to deliver the result as boundary outputs.
551/// * `extra` — occurrence data passed to every codec call for this conversion.
552///
553/// Output binding rejects input recipes:
554/// ```compile_fail
555/// use prebindgen_declarations::{bind_output, Codec, Input};
556/// use prebindgen_flat::Flat;
557/// fn wrong<C: Codec>(flat: &mut Flat<C::Input, C::Output>, codec: &C, extra: &C::Extra) {
558///     bind_output(flat, codec, "foo", &Input::Leaf, extra).unwrap();
559/// }
560/// ```
561pub fn bind_output<C: Codec>(
562    flat: &mut Flat<C::Input, C::Output>,
563    codec: &C,
564    function: &str,
565    recipe: &Output,
566    extra: &C::Extra,
567) -> Result<()> {
568    let mut conversion = Compiler::new(flat, codec, "ret".into()).output(
569        &flat.function(function)?.ret.ty,
570        recipe,
571        extra,
572        prebindgen_conversions::source_result_expr(),
573        false,
574    )?;
575    codec.finish_output(&mut conversion, extra)?;
576    conversion.validate_value_count()?;
577    conversion.layout.validate()?;
578    flat.function_mut(function)?.ret.slot = Some(conversion.into());
579    Ok(())
580}
581
582struct Compiler<'a, C: Codec> {
583    flat: &'a Flat<C::Input, C::Output>,
584    codec: &'a C,
585    prefix: String,
586    next: std::cell::Cell<usize>,
587}
588impl<'a, C: Codec> Compiler<'a, C> {
589    fn new(flat: &'a Flat<C::Input, C::Output>, codec: &'a C, prefix: String) -> Self {
590        Self {
591            flat,
592            codec,
593            prefix,
594            next: std::cell::Cell::new(0),
595        }
596    }
597    fn name(&self) -> syn::Ident {
598        let next = self.next.get();
599        let n = format_ident!("{}_w{next}", self.prefix);
600        self.next.set(next + 1);
601        n
602    }
603    fn prefix(&self, wires: &mut WireLayout<C::Wire, C::Group>, segment: ComponentSegment) {
604        for wire in wires {
605            self.codec.prefix_component(&mut wire.kind, segment.clone());
606        }
607    }
608    fn input(
609        &self,
610        ty: &Type,
611        recipe: &Input,
612        extra: &C::Extra,
613        active: &Condition,
614    ) -> Result<CodecInput<C>> {
615        let var = self.name();
616        let mut wires = WireLayout::default();
617        let mut validation = TokenStream::new();
618        let mut acquire = TokenStream::new();
619        let mut resources = Vec::new();
620        let mut acknowledgement = None;
621        let mut statements = TokenStream::new();
622        let argument = quote!(#var);
623        match recipe {
624            Input::Optional(inner) => {
625                let ty = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
626                let tag = self.name();
627                let child = self.input(ty, inner, extra, &active.arm(tag.clone(), 1))?;
628                return optional_input(self.codec, extra, tag, var, active, child);
629            }
630            Input::Leaf => {
631                let w = self.name();
632                return self.codec.input_leaf(ty, extra, w, var, active);
633            }
634            Input::Fields(children) => {
635                if type_key(ty) == "()" && children.is_empty() {
636                    wires.enclose(self.codec.fields_group());
637                    return Ok(InputPlan {
638                        context: Default::default(),
639                        acknowledgement,
640                        layout: wires,
641                        validation,
642                        acquire,
643                        resources,
644                        statements,
645                        argument: quote!(()),
646                    });
647                }
648                let r = self.flat.record(ty)?;
649                if r.fields.len() != children.len() {
650                    return Err(format!(
651                        "`{}`: expected {} field converters, got {}",
652                        type_key(ty),
653                        r.fields.len(),
654                        children.len()
655                    ));
656                }
657                return record_input(
658                    self.codec,
659                    ty,
660                    var,
661                    r.fields.iter().zip(children).map(|(field, child)| {
662                        Ok((
663                            field.name.clone(),
664                            self.input(&field.ty, child, extra, active)?,
665                        ))
666                    }),
667                );
668            }
669            Input::Constructor { function, args } => {
670                let f = self.flat.function(function)?;
671                if !same_type(&f.ret.ty, ty) {
672                    return Err(format!(
673                        "constructor `{function}` returns `{}`, expected `{}`",
674                        type_key(&f.ret.ty),
675                        type_key(ty)
676                    ));
677                }
678                if f.params.len() != args.len() {
679                    return Err(format!("constructor `{function}`: wrong argument count"));
680                }
681                let mut call_args = Vec::new();
682                for (param, child) in f.params.iter().zip(args) {
683                    let mut p = self.input(&param.ty.ty, child, extra, active)?;
684                    self.prefix(
685                        &mut p.layout,
686                        ComponentSegment::Argument(param.name.to_string()),
687                    );
688                    validation.extend(p.validation);
689                    acquire.extend(p.acquire);
690                    resources.extend(p.resources);
691                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
692                    statements.extend(p.statements);
693                    wires.extend(p.layout);
694                    call_args.push(p.argument);
695                }
696                let path = &f.path;
697                statements.extend(quote!(let #var:#ty=#path(#(#call_args),*);));
698            }
699            Input::Selector(arms) => {
700                if arms.is_empty() {
701                    return Err("an input selector needs at least one alternative".into());
702                }
703                if matches!(ty, Type::Reference(_)) {
704                    return Err("selector results must be owned in this subset".into());
705                }
706                let tag = self.name();
707                wires.push(Wire {
708                    name: tag.clone(),
709                    kind: {
710                        let mut kind = self.codec.selector(extra);
711                        self.codec
712                            .prefix_component(&mut kind, ComponentSegment::Selector);
713                        kind
714                    },
715                });
716                let count = arms.len() as i32;
717                let invalid = active.and(quote!(!(0..#count).contains(&#tag)));
718                validation.extend(quote!(if #invalid {return Err("invalid selector".into());}));
719                let mut cases = Vec::new();
720                for (i, arm) in arms.iter().enumerate() {
721                    let mut p = self.input(ty, arm, extra, &active.arm(tag.clone(), i as i32))?;
722                    self.prefix(&mut p.layout, ComponentSegment::Alternative(i));
723                    for wire in p.layout.iter_mut() {
724                        self.codec.set_niche(&mut wire.kind, None);
725                    }
726                    validation.extend(p.validation);
727                    acquire.extend(p.acquire);
728                    resources.extend(p.resources);
729                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
730                    let body = p.statements;
731                    let value = p.argument;
732                    wires.extend(p.layout);
733                    let i = i as i32;
734                    cases.push(quote!(#i=>{#body #value}));
735                }
736                statements = quote!(let #var:#ty=match #tag{#(#cases,)* _=>unreachable!("validated selector")};);
737            }
738        }
739        Ok(InputPlan {
740            context: Default::default(),
741            acknowledgement,
742            layout: wires,
743            validation,
744            acquire,
745            resources,
746            statements,
747            argument,
748        })
749    }
750    fn output(
751        &self,
752        ty: &Type,
753        recipe: &Output,
754        extra: &C::Extra,
755        value: TokenStream,
756        borrowed: bool,
757    ) -> Result<CodecOutput<C>> {
758        let mut out = OutputPlan {
759            context: Default::default(),
760            delivery: Default::default(),
761            commit: TokenStream::new(),
762            layout: Default::default(),
763            statements: TokenStream::new(),
764            values: Vec::new(),
765        };
766        match recipe {
767            Output::Optional(inner) => {
768                let ty = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
769                let tag = self.name();
770                let payload = self.name();
771                let child = self.output(ty, inner, extra, quote!(#payload), borrowed)?;
772                return optional_output(self.codec, extra, tag, payload, value, borrowed, child);
773            }
774            Output::Leaf => {
775                let w = self.name();
776                return self.codec.output_leaf(ty, extra, w, value, borrowed);
777            }
778            Output::Projected(projection) => {
779                if borrowed && projection.mode == ProjectionMode::Owned {
780                    return Err(
781                        "owned projection cannot consume a shared-reference occurrence".into(),
782                    );
783                }
784                let value = if !borrowed && projection.mode == ProjectionMode::Borrowed {
785                    quote!(&(#value))
786                } else {
787                    value
788                };
789                return self
790                    .codec
791                    .output_projected(ty, extra, self.name(), value, projection);
792            }
793            Output::Fields(children) | Output::OwnedFields(children) => {
794                let owned = matches!(recipe, Output::OwnedFields(_));
795                if owned && borrowed {
796                    return Err("owned fields cannot consume a shared-reference occurrence".into());
797                }
798                if type_key(ty) == "()" && children.is_empty() {
799                    out.layout.enclose(self.codec.fields_group());
800                    return Ok(out);
801                }
802                let r = self.flat.record(ty)?;
803                if r.fields.len() != children.len() {
804                    return Err("wrong output field count".into());
805                }
806                let convert = |index: usize, value| {
807                    self.output(&r.fields[index].ty, &children[index], extra, value, !owned)
808                };
809                return if owned {
810                    record_output_owned(
811                        self.codec,
812                        self.name(),
813                        value,
814                        r.fields.iter().map(|f| f.name.clone()),
815                        convert,
816                    )
817                } else {
818                    record_output(
819                        self.codec,
820                        self.name(),
821                        value,
822                        r.fields.iter().map(|f| f.name.clone()),
823                        convert,
824                    )
825                };
826            }
827            Output::Getter { function, output } | Output::Into { function, output } => {
828                let consuming = matches!(recipe, Output::Into { .. });
829                if consuming && borrowed {
830                    return Err(
831                        "consuming accessor cannot consume a shared-reference occurrence".into(),
832                    );
833                }
834                let f = self.flat.function(function)?;
835                let expected: Type = if consuming {
836                    ty.clone()
837                } else {
838                    syn::parse_quote!(&#ty)
839                };
840                if f.params.len() != 1 || !same_type(&f.params[0].ty.ty, &expected) {
841                    return Err(format!(
842                        "getter `{function}` must take exactly `{}`",
843                        type_key(&expected)
844                    ));
845                }
846                if matches!(f.ret.ty, Type::Reference(_)) {
847                    return Err("borrowed getter results are outside this subset".into());
848                }
849                let v = self.name();
850                let path = &f.path;
851                let arg = if borrowed || consuming {
852                    value
853                } else {
854                    quote!(&#value)
855                };
856                out.statements = quote!(let #v=#path(#arg););
857                let mut child = self.output(&f.ret.ty, output, extra, quote!(#v), false)?;
858                self.prefix(
859                    &mut child.layout,
860                    ComponentSegment::Getter(function.clone()),
861                );
862                append(&mut out, child)?;
863            }
864            Output::Combine(parts) => {
865                if parts.is_empty() {
866                    return Err("an output combination needs at least one part".into());
867                }
868                if parts
869                    .iter()
870                    .filter(|p| p.consumes(ty, extra, self.codec))
871                    .count()
872                    > 1
873                {
874                    return Err("a combination cannot consume the same value twice".into());
875                }
876                let mut bodies = Vec::new();
877                for (index, child) in parts.iter().enumerate() {
878                    let consuming = child.consumes(ty, extra, self.codec);
879                    let arg = if borrowed || consuming {
880                        value.clone()
881                    } else {
882                        quote!(&#value)
883                    };
884                    let mut p = self.output(ty, child, extra, arg, borrowed || !consuming)?;
885                    self.prefix(&mut p.layout, ComponentSegment::View(index));
886                    bodies.push((consuming, std::mem::take(&mut p.statements)));
887                    append(&mut out, p)?;
888                }
889                // Wire order follows the declaration. Observations execute before the move.
890                bodies.sort_by_key(|(consumes, _)| *consumes);
891                for (_, body) in bodies {
892                    out.statements.extend(body);
893                }
894            }
895        }
896        Ok(out)
897    }
898}
899fn append<W: prebindgen_flat::WireKind, C, M>(
900    to: &mut OutputPlan<W, C, M>,
901    from: OutputPlan<W, C, M>,
902) -> Result<()> {
903    if !from.delivery.statements.is_empty()
904        || from.delivery.result.to_string() != quote!(()).to_string()
905    {
906        return Err("composed output delivery must be installed by Codec::finish_output".into());
907    }
908    to.commit.extend(from.commit);
909    to.layout.extend(from.layout);
910    to.statements.extend(from.statements);
911    to.values.extend(from.values);
912    Ok(())
913}

933fn merge_acknowledgement(
934    target: &mut Option<TokenStream>,
935    incoming: Option<TokenStream>,
936) -> Result<()> {
937    if let Some(code) = incoming {
938        if let Some(previous) = target {
939            if previous.to_string() != code.to_string() {
940                return Err("conflicting input acknowledgements".into());
941            }
942        } else {
943            *target = Some(code);
944        }
945    }
946    Ok(())
947}
