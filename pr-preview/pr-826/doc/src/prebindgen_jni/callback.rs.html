8/// Prepared callback signature and the independent output boundary of its arguments.
9///
10/// A callback crosses into Rust as one JVM object. Its Rust-to-Kotlin argument
11/// converters are retained here for later invocations, which can outlive the
12/// function accepting that callback.
13#[derive(Clone, Debug)]
14pub struct JniCallback {
15    package: String,
16    name: String,
17    arguments: Vec<Argument>,
18}
19#[derive(Clone, Debug)]
20struct Argument {
21    source_type: syn::Type,
22    value: TokenStream,
23    output: JniOutput,
24    scoped: bool,
25}
26pub(crate) fn declared_outputs(
27    flat: &Bindings,
28    outputs: &[Output<OutputRequest>],
29) -> Result<BTreeSet<String>> {
30    outputs
31        .iter()
32        .filter_map(|tree| match tree {
33            Output::Request(OutputRequest::ForType { ty, .. }) => {
34                Some(flat.resolve_type(ty).map(|ty| type_key(&ty)))
35            }
36            _ => None,
37        })
38        .collect()
39}
40/// Preserve direct callback slots while constructors compile their own occurrences.
41pub(crate) fn prepare(
42    flat: &mut Bindings,
43    defaults: &Declarations,
44    functions: &[FunctionDecl],
45    explicit_outputs: &BTreeSet<String>,
46) -> Result<()> {
47    for declaration in functions {
48        let params = flat.function(&declaration.name)?.params.clone();
49        for (index, parameter) in params.into_iter().enumerate() {
50            if callback_signature(&parameter.ty.ty)?.is_none() {
51                continue;
52            }
53            if declaration.params.contains_key(&parameter.name.to_string()) {
54                return Err("source callback parameters use their argument output expansions, not expand_param".into());
55            }
56            if parameter.ty.slot.is_some() {
57                continue;
58            }
59            let plan = input(
60                flat,
61                defaults,
62                explicit_outputs,
63                &parameter.ty.ty,
64                format_ident!("p{index}_callback"),
65                format_ident!("p{index}_callback_value"),
66                &prebindgen_declarations::Condition::default(),
67            )?;
68            flat.function_mut(&declaration.name)?.params[index].ty.slot = Some(plan.into());
69        }
70    }
71    Ok(())
72}

74/// Compile one callback occurrence, including a source constructor's argument.
75pub(crate) fn input(
76    flat: &Bindings,
77    defaults: &Declarations,
78    explicit_outputs: &BTreeSet<String>,
79    ty: &syn::Type,
80    wire: syn::Ident,
81    var: syn::Ident,
82    active: &prebindgen_declarations::Condition,
83) -> Result<InputPlan<JniWire>> {
84    let signature = callback_signature(ty)?.ok_or("expected source callback")?;
85    let name = if signature.arguments.is_empty() {
86        "VoidCallback".into()
87    } else {
88        format!(
89            "{}Callback",
90            signature
91                .arguments
92                .iter()
93                .map(subject_name)
94                .collect::<Vec<_>>()
95                .join("")
96        )
97    };
98    validate_name(&name)?;
99    let package = signature
100        .arguments
101        .first()
102        .and_then(|ty| {
103            flat.type_info(subject_type(ty))
104                .ok()?
105                .metadata
106                .class
107                .as_ref()
108                .map(|c| c.pkg.clone())
109        })
110        .unwrap_or_else(|| flat.metadata.package.clone());
111    if foreign_name_exists(flat, &format!("{package}.{name}")) {
112        return Err("duplicate foreign callback interface name".into());
113    }
114    let mut arguments = vec![];
115    for (i, source_type) in signature.arguments.into_iter().enumerate() {
116        let binding = format_ident!("callback_source_{i}");
117        let (ty, borrowed) = match &source_type {
118            syn::Type::Reference(r) if r.mutability.is_none() => match &*r.elem {
119                syn::Type::Slice(_) => (source_type.clone(), false),
120                syn::Type::Path(p) if p.path.is_ident("str") => (source_type.clone(), false),
121                ty => (ty.clone(), true),
122            },
123            syn::Type::Reference(_) => {
124                return Err("mutable callback arguments are unsupported".into())
125            }
126            _ => (source_type.clone(), false),
127        };
128        let value = quote!(#binding);
129        if contains_result(&ty) {
130            return Err(
131                "Result-valued callback arguments require an explicit value representation".into(),
132            );
133        }
134        let recipe = defaults.output_recipe(flat, &data::DataCodec::default(), &ty, &())?;
135        let scoped = !explicit_outputs.contains(&type_key(&ty))
136            && !explicit_outputs.contains(&type_key(subject_type(&ty)));
137        let output = prebindgen_declarations::compile_callback_argument(
138            flat,
139            &data::DataCodec::default(),
140            &ty,
141            &recipe,
142            i,
143            borrowed,
144            &(),
145        )?;
146        sequence::validate_buffered(&output.output_outcomes().success)?;
147        reject_sequence_handles(&output.output_outcomes().success, false)?;
148        arguments.push(Argument {
149            source_type,
150            value,
151            output,
152            scoped,
153        });
154    }
155    let callback = Rc::new(JniCallback {
156        package,
157        name,
158        arguments,
159    });
160    callback.input(wire, var, active)
161}
162impl JniCallback {
163    pub(crate) fn fqn(&self) -> String {
164        format!("{}.{}", self.package, self.name)
165    }

167    fn input(
168        self: &Rc<Self>,
169        wire: syn::Ident,
170        var: syn::Ident,
171        active: &prebindgen_declarations::Condition,
172    ) -> Result<InputPlan<JniWire>> {
173        prebindgen_conversions::validate_wire_names(self.arguments.iter().flat_map(|arg| {
174            prebindgen_flat::wire_definitions(std::iter::once(
175                &arg.output.output_outcomes().success,
176            ))
177        }))?;
178        let env = prebindgen_conversions::template_parameter("environment");
179        let retained = format_ident!("{wire}_retained");
180        let environment = format_ident!("callback_environment");
181        let context = JniContext {
182            environment: environment.clone(),
183            acknowledgement: format_ident!("unused_ack"),
184            domain_output: format_ident!("unused_domain"),
185            is_domain_error: false,
186            output_committed: quote!(),
187        };
188        let mut descriptor = String::from("(");
189        let mut encode = TokenStream::new();
190        let mut commits = TokenStream::new();
191        let mut values = vec![];
192        let mut source_bindings = TokenStream::new();
193        let mut prepares = vec![];
194        let mut params = vec![];
195        let mut slots = 0;
196        for (index, argument) in self.arguments.iter().enumerate() {
197            let source = format_ident!("callback_value_{index}");
198            let param = format_ident!("callback_source_{index}");
199            let ty = &argument.source_type;
200            params.push(quote!(#param: #ty));
201            let value = &argument.value;
202            source_bindings.extend(quote!(let #source = #value;));
203            let fields = group::callback_parameters(&argument.output.output_outcomes().success);
204            let assignments =
205                prebindgen_conversions::match_output_fields(fields, &argument.output.rust.values)?;
206            for (position, (field, value)) in assignments.into_iter().enumerate() {
207                descriptor.push_str(field.kind.jvm_descriptor());
208                slots += field.kind.parameter_slots();
209                let local = format_ident!("callback_java_{index}_{position}");
210                let code = field.kind.output_argument(value, &local);
211                encode.extend(render(&context, &code.statements, &source)?);
212                values.push(render(&context, &code.value, &source)?);
213            }
214            commits.extend(argument.output.code_fragment_slot(
215                OutputFragment::Commit {
216                    context: &context,
217                    source: &source,
218                },
219                &syn::parse_quote!(()),
220            )?);
221            prepares.push((argument, source));
222        }
223        descriptor.push_str(")V");
224        if slots > 254 {
225            return Err("source callback exceeds JVM parameter limit".into());
226        }
227        let capacity = i32::try_from(values.len() + 16).map_err(|_| "too many callback fields")?;
228        let mut invoke = quote!({
229            #encode
230            let arguments = [#(#values),*];
231            #commits
232            unsafe { #retained.invoke(#environment, &arguments) }
233        });
234        for (argument, source) in prepares.into_iter().rev() {
235            invoke = argument.output.code_fragment_slot(
236                OutputFragment::Prepare {
237                    context: &context,
238                    source: &source,
239                    next: invoke,
240                },
241                &syn::parse_quote!(::prebindgen_runtime::Result<()>),
242            )?;
243        }
244        let (registration, activate) = if active.0.is_empty() {
245            (
246                quote!(let #retained = ::prebindgen_jni_runtime::Callback::new(&mut #env, &#wire, #descriptor)?;),
247                quote!(),
248            )
249        } else {
250            let condition = active.rust();
251            (
252                quote!(let #retained = if #condition {Some(::prebindgen_jni_runtime::Callback::new(&mut #env, &#wire, #descriptor)?)} else {None};),
253                // Optional/selector construction executes only on its validated active path.
254                quote!(let #retained = #retained.expect("validated active callback");),
255            )
256        };
257        Ok(InputPlan {
258            fallible: true,
259            context: Default::default(),
260            node: WireNode::Leaf(vec![Wire {
261                name: wire.clone(),
262                kind: JniWire::Callback {
263                    callback: self.clone(),
264                },
265            }]),
266            acknowledgement: None,
267            acquire: quote!(),
268            resources: vec![],
269            validation: registration,
270            statements: quote!(#activate #[allow(unused_mut)] let #var = move |#(#params),*| {
271                #retained.with_env(#capacity, |#environment| { let mut #environment = #environment; #source_bindings #invoke });
272            };),
273            argument: quote!(#var),
274        })
275    }

277    pub(crate) fn adapter(&self, parameter: &str, context: &KotlinContext<'_>) -> Result<String> {
278        let mut raw = vec![];
279        let mut outputs = vec![];
280        let mut offset = 0;
281        for (index, argument) in self.arguments.iter().enumerate() {
282            let prefix = format!("callbackOutput{index}_");
283            let ctx = KotlinContext {
284                result: "parts",
285                output_prefix: &prefix,
286                cleanup_failure: Some("callbackCleanupFailure"),
287                ..*context
288            };
289            let mut output = argument.output.kotlin(&ctx)?;
290            let fields = group::callback_parameters(&argument.output.output_outcomes().success)
291                .collect::<Vec<_>>();
292            for (i, field) in fields.iter().enumerate() {
293                let name = format!("raw{}", offset + i);
294                raw.push(format!("{name}: {}", field.kind.kotlin_type()));
295            }
296            for text in output.delivered.iter_mut().chain([
297                &mut output.output_locals,
298                &mut output.materialize,
299                &mut output.cleanup,
300            ]) {
301                for i in (0..fields.len()).rev() {
302                    *text = text.replace(&format!("parts[{i}]"), &format!("raw{}", offset + i));
303                }
304            }
305            offset += fields.len();
306            outputs.push((argument.scoped, output));
307        }
308        let mut locals = String::from("        var callbackCleanupFailure: Throwable? = null\n");
309        let mut materialize = String::new();
310        let mut cleanup = String::new();
311        let mut scoped_cleanup = String::new();
312        let mut delivered = vec![];
313        for (argument, (scoped, output)) in outputs.iter().enumerate() {
314            locals.push_str(&output.output_locals);
315            materialize.push_str(&output.materialize);
316            cleanup.push_str(&output.cleanup);
317            if *scoped {
318                scoped_cleanup.push_str(&output.cleanup);
319            }
320            for (value, (ty, expression)) in output
321                .result_types
322                .iter()
323                .zip(&output.delivered)
324                .enumerate()
325            {
326                let name = format!("callbackArgument{argument}_{value}");
327                locals.push_str(&argument_local(&name, ty));
328                materialize.push_str(&format!("            {name} = {expression}\n"));
329                delivered.push(name);
330            }
331        }
332        let reconstruction = format!("try {{\n{materialize}        }} catch (failure: Throwable) {{\n{cleanup}            callbackCleanupFailure?.let {{ failure.addSuppressed(it) }}\n            throw failure\n        }}");
333        let call = format!("{parameter}.run({})", delivered.join(", "));
334        let body = if scoped_cleanup.is_empty() {
335            format!("{reconstruction}\n        {call}")
336        } else {
337            format!("var callbackFailure: Throwable? = null\n        var callbackReady = false\n        try {{\n            {reconstruction}\n            callbackReady = true\n            {call}\n        }} catch (failure: Throwable) {{\n            callbackFailure = failure\n            throw failure\n        }} finally {{\n            if (callbackReady) {{\n{scoped_cleanup}                callbackCleanupFailure?.let {{ cleanup -> if (callbackFailure == null) throw cleanup else callbackFailure.addSuppressed(cleanup) }}\n            }}\n        }}")
338        };
339        Ok(format!("if ({parameter} == null) null else object {{\n    @Suppress(\"UNUSED_PARAMETER\", \"UNUSED_VARIABLE\", \"UNCHECKED_CAST\", \"UNNECESSARY_NOT_NULL_ASSERTION\", \"USELESS_CAST\", \"VARIABLE_WITH_REDUNDANT_INITIALIZER\")\n    fun run({}) {{\n{locals}        {body}\n    }}\n}}", raw.join(", ")))
340    }
341}
342fn render(context: &JniContext, code: &TokenStream, source: &syn::Ident) -> Result<TokenStream> {
343    let code = prebindgen_conversions::render_template(code, &|name| {
344        Ok(if name == "source_result" {
345            quote!(#source)
346        } else {
347            prebindgen_conversions::template_parameter(name)
348        })
349    })?;
350    context.render(&code)
351}
352pub(crate) fn files(
353    flat: &Bindings,
354    classes: &BTreeMap<String, KotlinClass>,
355    files: &mut BTreeMap<String, String>,
356) -> Result<()> {
357    for function in flat.functions.values().filter(|f| f.metadata.is_some()) {
358        for wire in function.input_wire_definitions()? {
359            let JniWire::Callback { callback } = &wire.kind else {
360                continue;
361            };
362            let context = KotlinContext {
363                classes,
364                result: "parts",
365                failure: "failure",
366                cleanup_failure: None,
367                output_prefix: "output",
368            };
369            let mut params = vec![];
370            let mut names = BTreeSet::new();
371            for (i, arg) in callback.arguments.iter().enumerate() {
372                let output = arg.output.kotlin(&context)?;
373                output.callback_parameter_declarations()?;
374                for (j, ty) in output.result_types.iter().enumerate() {
375                    let mut name = output
376                        .result_names
377                        .get(j)
378                        .and_then(|n| n.clone())
379                        .unwrap_or_else(|| format!("arg{i}_{j}"));
380                    while !names.insert(name.clone()) {
381                        name.push('_');
382                    }
383                    params.push(format!("`{name}`: {ty}"));
384                }
385            }
386            let text = format!("package {}\n\n/** Invoked by Rust, possibly from a native worker thread. */\npublic fun interface {} {{\n    public fun run({})\n}}\n", callback.package, callback.name, params.join(", "));
387            let path = format!(
388                "{}/{}.kt",
389                callback.package.replace('.', "/"),
390                callback.name
391            );
392            if let Some(existing) = files.get(&path) {
393                if existing != &text {
394                    return Err("callback interface signature collision".into());
395                }
396            } else {
397                files.insert(path, text);
398            }
399        }
400    }
401    Ok(())
402}

404fn subject_type(ty: &syn::Type) -> &syn::Type {
405    match ty {
406        syn::Type::Reference(r) => subject_type(&r.elem),
407        _ => prebindgen_flat::option_inner(ty)
408            .map(subject_type)
409            .unwrap_or(ty),
410    }
411}
412fn subject_name(ty: &syn::Type) -> String {
413    match ty {
414        syn::Type::Reference(r) => subject_name(&r.elem),
415        syn::Type::Slice(s) => format!("{}List", subject_name(&s.elem)),
416        _ => {
417            if let Some(inner) = prebindgen_flat::option_inner(ty) {
418                return format!("Optional{}", subject_name(inner));
419            }
420            if let Some(inner) = prebindgen_flat::vec_inner(ty) {
421                return format!("{}Vec", subject_name(inner));
422            }
423            match ty {
424                syn::Type::Path(p) => p.path.segments.last().unwrap().ident.to_string(),
425                _ => "Unit".into(),
426            }
427        }
428    }
429}

431fn contains_result(ty: &syn::Type) -> bool {
432    prebindgen_flat::result_parts(ty).is_some()
433        || prebindgen_flat::option_inner(ty).is_some_and(contains_result)
434        || prebindgen_flat::sequence_parts(ty).is_some_and(|(ty, _)| contains_result(ty))
435        || matches!(ty,syn::Type::Reference(r) if contains_result(&r.elem))
436}
437fn reject_sequence_handles(node: &kotlin::JniNode, repeated: bool) -> Result<()> {
438    match node {
439        WireNode::Leaf(wires) => {
440            if repeated
441                && wires
442                    .iter()
443                    .any(|w| matches!(w.kind, JniWire::Handle { .. }))
444            {
445                return Err("handle-containing sequence callback arguments are unsupported".into());
446            }
447        }
448        WireNode::Chain { meta, child, .. } => {
449            reject_sequence_handles(child, repeated || matches!(meta, JniChain::Sequence(_)))?
450        }
451        WireNode::Branch { children, .. } => {
452            for child in children {
453                reject_sequence_handles(child, repeated)?;
454            }
455        }
456    }
457    Ok(())
458}

460fn argument_local(name: &str, ty: &str) -> String {
461    let default = if ty.ends_with('?') {
462        Some("null")
463    } else {
464        match ty.strip_prefix("kotlin.").unwrap_or(ty) {
465            "Boolean" => Some("false"),
466            "Byte" => Some("0.toByte()"),
467            "Short" => Some("0.toShort()"),
468            "Int" => Some("0"),
469            "Long" => Some("0L"),
470            "UByte" => Some("0.toUByte()"),
471            "UShort" => Some("0.toUShort()"),
472            "UInt" => Some("0U"),
473            "ULong" => Some("0UL"),
474            "Float" => Some("0.0f"),
475            "Double" => Some("0.0"),
476            "Char" => Some("'\\u0000'"),
477            _ => None,
478        }
479    };
480    match default {
481        Some(default) => format!("        var {name}: {ty} = {default}\n"),
482        None => format!("        lateinit var {name}: {ty}\n"),
483    }
