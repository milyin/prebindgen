8/// Prepared callback signature and the independent output boundary of its arguments.
9///
10/// A callback crosses into Rust as one JVM object. Its Rust-to-Kotlin argument
11/// converters are retained here for later invocations, which can outlive the
12/// function accepting that callback.
13#[derive(Clone, Debug)]
14pub struct JniCallback {
15    package: String,
16    name: String,
17    arguments: Vec<Argument>,
18    return_value: Option<CallbackReturn>,
19}
20#[derive(Clone, Debug)]
21struct CallbackReturn {
22    package: String,
23    source_type: syn::Type,
24    input: JniInput,
25}
26#[derive(Clone, Debug)]
27struct Argument {
28    source_type: syn::Type,
29    value: TokenStream,
30    output: JniOutput,
31    scoped: bool,

33/// Retain source callback occurrences before filtering to exported functions/helpers.
34pub(crate) fn source_parameters(
35    items: &[(syn::Item, prebindgen::SourceLocation)],
36) -> Vec<syn::Type> {
37    items
38        .iter()
39        .filter_map(|(item, _)| match item {
40            // Generic parameters introduce a different type scope; they are not
41            // concrete callback occurrences in the supported flat source model.
42            syn::Item::Fn(function) if function.sig.generics.params.is_empty() => Some(function),
43            _ => None,
44        })
45        .flat_map(|function| &function.sig.inputs)
46        .filter_map(|parameter| match parameter {
47            syn::FnArg::Typed(parameter)
48                if matches!(parameter.ty.as_ref(), syn::Type::ImplTrait(_)) =>
49            {
50                Some((*parameter.ty).clone())
51            }
52            _ => None,
53        })
54        .collect()
55}

57/// A declaration configures a source occurrence, rather than inventing a signature.
58pub(crate) fn validate_source_occurrence(
59    flat: &Bindings,
60    declaration: &prebindgen_declarations::CallbackInput,
61    sources: &[syn::Type],
62) -> Result<()> {
63    let expected = callback_signature(&declaration.ty)?
64        .ok_or("callback declaration requires impl Fn(...) -> R")?;
65    let matches = sources.iter().any(|source| {
66        let Ok(Some(actual)) = callback_signature(source) else {
67            return false;
68        };
69        actual.arguments.len() == expected.arguments.len()
70            && actual
71                .arguments
72                .iter()
73                .zip(&expected.arguments)
74                .all(|(actual, expected)| {
75                    flat.resolve_type(actual)
76                        .is_ok_and(|actual| prebindgen_flat::same_type(&actual, expected))
77                })
78            && flat
79                .resolve_type(&actual.return_type)
80                .is_ok_and(|actual| prebindgen_flat::same_type(&actual, &expected.return_type))
81    });
82    if !matches {
83        let name = declaration.foreign.as_deref().unwrap_or("<unnamed>");
84        return Err(format!(
85            "callback declaration `{name}` for `{}` does not match any source function parameter; check the callback argument and return types",
86            type_key(&declaration.ty)
87        ));
88    }
89    Ok(())
90}

92pub(crate) fn declared_outputs(
93    flat: &Bindings,
94    outputs: &[Output<OutputRequest>],
95) -> Result<BTreeSet<String>> {
96    outputs
97        .iter()
98        .filter_map(|tree| match tree {
99            Output::Request(OutputRequest::ForType { ty, .. }) => {
100                Some(flat.resolve_type(ty).map(|ty| type_key(&ty)))
101            }
102            _ => None,
103        })
104        .collect()
105}
106/// Compile one callback occurrence, including a source constructor's argument.
107pub(crate) fn input(
108    flat: &Bindings,
109    configuration: &prebindgen_declarations::CallbackInput,
110    explicit_outputs: &BTreeSet<String>,
111    ty: &syn::Type,
112    wire: syn::Ident,
113    var: syn::Ident,
114    active: &prebindgen_declarations::Condition,
115) -> Result<InputPlan<JniWire>> {
116    let signature = callback_signature(ty)?.ok_or("expected source callback")?;
117    let returns_unit = matches!(&signature.return_type, syn::Type::Tuple(t) if t.elems.is_empty());
118    if returns_unit
119        && !matches!(configuration.return_value.as_ref(), Input::Fields(fields) if fields.is_empty())
120    {
121        return Err("unit callback returns require an empty unit input representation".into());
122    }
123    let mut name = if signature.arguments.is_empty() {
124        "VoidCallback".into()
125    } else {
126        format!(
127            "{}Callback",
128            signature
129                .arguments
130                .iter()
131                .map(subject_name)
132                .collect::<Vec<_>>()
133                .join("")
134        )
135    };
136    if !returns_unit {
137        name = format!(
138            "{}To{}Callback",
139            name.strip_suffix("Callback").unwrap(),
140            subject_name(&signature.return_type)
141        );
142    }
143    if let Some(selected) = &configuration.foreign {
144        name = selected.clone();
145    }
146    validate_class_name(&name)?;
147    let package = configuration.namespace.clone().unwrap_or_else(|| {
148        signature
149            .arguments
150            .first()
151            .and_then(|ty| {
152                flat.type_info(subject_type(ty))
153                    .ok()?
154                    .metadata
155                    .class
156                    .as_ref()
157                    .map(|c| c.pkg.clone())
158            })
159            .unwrap_or_else(|| flat.metadata.package.clone())
160    });
161    validate_package(&package)?;
162    if configuration.arguments.len() != signature.arguments.len() {
163        return Err("prepared callback has the wrong argument count".into());
164    }
165    if foreign_name_exists(flat, &format!("{package}.{name}")) {
166        return Err("duplicate foreign callback interface name".into());
167    }
168    let return_value = if returns_unit {
169        None
170    } else {
171        if has_reference(&signature.return_type) {
172            return Err(
173                "callback return values must be owned; borrowed returns are unsupported".into(),
174            );
175        }
176        let codec = data::DataCodec::default();
177        let input = prebindgen_declarations::compile_callback_return(
178            flat,
179            &codec,
180            &signature.return_type,
181            &configuration.return_value,
182            &(),
183        )?;
184        Some(CallbackReturn {
185            package: flat.metadata.package.clone(),
186            source_type: signature.return_type,
187            input,
188        })
189    };
190    let mut arguments = vec![];
191    for (i, source_type) in signature.arguments.into_iter().enumerate() {
192        let binding = format_ident!("callback_source_{i}");
193        let (ty, borrowed) = match &source_type {
194            syn::Type::Reference(r) if r.mutability.is_none() => match &*r.elem {
195                syn::Type::Slice(_) => (source_type.clone(), false),
196                syn::Type::Path(p) if p.path.is_ident("str") => (source_type.clone(), false),
197                ty => (ty.clone(), true),
198            },
199            syn::Type::Reference(_) => {
200                return Err("mutable callback arguments are unsupported".into())
201            }
202            _ => (source_type.clone(), false),
203        };
204        let value = quote!(#binding);
205        if contains_result(&ty) {
206            return Err(
207                "Result-valued callback arguments require an explicit value representation".into(),
208            );
209        }
210        let recipe = &configuration.arguments[i];
211        let scoped = !configuration.explicit_arguments.contains(&i)
212            && !explicit_outputs.contains(&type_key(&ty))
213            && !explicit_outputs.contains(&type_key(subject_type(&ty)));
214        let output = prebindgen_declarations::compile_callback_argument(
215            flat,
216            &data::DataCodec::default(),
217            &ty,
218            recipe,
219            i,
220            borrowed,
221            &(),
222        )?;
223        sequence::validate_buffered(&output.output_outcomes().success)?;
224        arguments.push(Argument {
225            source_type,
226            value,
227            output,
228            scoped,
229        });
230    }
231    let callback = Rc::new(JniCallback {
232        package,
233        name,
234        arguments,
235        return_value,
236    });
237    callback.input(wire, var, active)
238}

240    pub(crate) fn bulk_nodes(&self) -> Vec<(bool, &kotlin::JniNode)> {
241        let mut nodes = self
242            .arguments
243            .iter()
244            .map(|a| (false, &a.output.output_outcomes().success))
245            .collect::<Vec<_>>();
246        if let Some(value) = &self.return_value {
247            nodes.push((true, value.input.input_node()));
248        }
249        nodes
250    }

256    fn input(
257        self: &Rc<Self>,
258        wire: syn::Ident,
259        var: syn::Ident,
260        active: &prebindgen_declarations::Condition,
261    ) -> Result<InputPlan<JniWire>> {
262        prebindgen_conversions::validate_wire_names(self.arguments.iter().flat_map(|arg| {
263            prebindgen_flat::wire_definitions(std::iter::once(
264                &arg.output.output_outcomes().success,
265            ))
266        }))?;
267        let env = prebindgen_conversions::template_parameter("environment");
268        let retained = format_ident!("{wire}_retained");
269        let environment = format_ident!("callback_environment");
270        let context = JniContext {
271            environment: environment.clone(),
272            acknowledgement: format_ident!("unused_ack"),
273            domain_output: format_ident!("unused_domain"),
274            return_sink: None,
275            is_domain_error: false,
276            output_committed: quote!(),
277        };
278        let mut descriptor = String::from("(");
279        let mut encode = TokenStream::new();
280        let mut commits = TokenStream::new();
281        let mut values = vec![];
282        let mut source_bindings = TokenStream::new();
283        let mut prepares = vec![];
284        let mut params = vec![];
285        let mut slots = 0;
286        for (index, argument) in self.arguments.iter().enumerate() {
287            let source = format_ident!("callback_value_{index}");
288            let param = format_ident!("callback_source_{index}");
289            let ty = &argument.source_type;
290            params.push(quote!(#param: #ty));
291            let value = &argument.value;
292            source_bindings.extend(quote!(let #source = #value;));
293            let fields = group::callback_parameters(&argument.output.output_outcomes().success);
294            let assignments =
295                prebindgen_conversions::match_output_fields(fields, &argument.output.rust.values)?;
296            for (position, (field, value)) in assignments.into_iter().enumerate() {
297                descriptor.push_str(field.kind.jvm_descriptor());
298                slots += field.kind.parameter_slots();
299                let local = format_ident!("callback_java_{index}_{position}");
300                let code = field.kind.output_argument(value, &local);
301                encode.extend(render(&context, &code.statements, &source)?);
302                values.push(render(&context, &code.value, &source)?);
303            }
304            commits.extend(argument.output.code_fragment_slot(
305                OutputFragment::Commit {
306                    context: &context,
307                    source: &source,
308                },
309                &syn::parse_quote!(()),
310            )?);
311            prepares.push((argument, source));
312        }
313        descriptor.push(')');
314        descriptor.push_str(
315            &self
316                .return_value
317                .as_ref()
318                .map(|r| r.descriptor())
319                .unwrap_or_else(|| "V".into()),
320        );
321        if slots > 254 {
322            return Err("source callback exceeds JVM parameter limit".into());
323        }
324        let return_references =
325            self.return_value
326                .as_ref()
327                .map_or(0, |result| match result.fields().as_slice() {
328                    [] => 0,
329                    [field] => usize::from(field.kind.is_reference()),
330                    fields => fields.len() + 1,
331                });
332        let capacity = i32::try_from(values.len() + return_references + 16)
333            .map_err(|_| "too many callback fields")?;
334        let call = if let Some(result) = &self.return_value {
335            result.rust_invoke(&context, &retained, &environment)?
336        } else {
337            quote!(unsafe { #retained.invoke(#environment, &arguments) })
338        };
339        let mut invoke = quote!({
340            #encode
341            let arguments = [#(#values),*];
342            #commits
343            #call
344        });
345        for (argument, source) in prepares.into_iter().rev() {
346            invoke = argument.output.code_fragment_slot(
347                OutputFragment::Prepare {
348                    context: &context,
349                    source: &source,
350                    next: invoke,
351                },
352                &syn::parse_quote!(::prebindgen_runtime::Result<()>),
353            )?;
354        }
355        let callback_name = self.fqn();
356        let (registration, activate) = if active.0.is_empty() {
357            (
358                quote!(let #retained = ::prebindgen_jni_runtime::Callback::new_named(&mut #env, &#wire, #descriptor, #callback_name)?;),
359                quote!(),
360            )
361        } else {
362            let condition = active.rust();
363            (
364                quote!(let #retained = if #condition {Some(::prebindgen_jni_runtime::Callback::new_named(&mut #env, &#wire, #descriptor, #callback_name)?)} else {None};),
365                // Optional/selector construction executes only on its validated active path.
366                quote!(let #retained = #retained.expect("validated active callback");),
367            )
368        };
369        let invocation = if self.return_value.is_some() {
370            quote!(#retained.with_env_result(#capacity, |#environment| { let mut #environment = #environment; #source_bindings #invoke }).unwrap_or_else(|error| panic!("callback failed: {}", error)))
371        } else {
372            quote!(#retained.with_env(#capacity, |#environment| { let mut #environment = #environment; #source_bindings #invoke });)
373        };
374        Ok(InputPlan {
375            fallible: true,
376            context: Default::default(),
377            node: WireNode::Leaf(vec![Wire {
378                name: wire.clone(),
379                kind: JniWire::Callback {
380                    callback: self.clone(),
381                },
382            }]),
383            acknowledgement: None,
384            acquire: quote!(),
385            resources: vec![],
386            validation: registration,
387            statements: quote!(#activate # [allow(unused_mut, clippy::collapsible_if)] let #var = move |#(#params),*| {
388                #invocation
389            };),
390            argument: quote!(#var),
391        })
392    }

394    pub(crate) fn adapter(&self, parameter: &str, context: &KotlinContext<'_>) -> Result<String> {
395        let mut raw = vec![];
396        let mut outputs = vec![];
397        let mut offset = 0;
398        for (index, argument) in self.arguments.iter().enumerate() {
399            let prefix = format!("callbackOutput{index}_");
400            let ctx = KotlinContext {
401                result: "parts",
402                output_prefix: &prefix,
403                cleanup_failure: Some("callbackCleanupFailure"),
404                ..*context
405            };
406            let mut output = argument.output.kotlin(&ctx)?;
407            let fields = group::callback_parameters(&argument.output.output_outcomes().success)
408                .collect::<Vec<_>>();
409            for (i, field) in fields.iter().enumerate() {
410                let name = format!("raw{}", offset + i);
411                raw.push(format!("{name}: {}", field.kind.kotlin_type()));
412            }
413            for text in output.delivered.iter_mut().chain([
414                &mut output.output_locals,
415                &mut output.materialize,
416                &mut output.cleanup,
417            ]) {
418                for i in (0..fields.len()).rev() {
419                    *text = text.replace(&format!("parts[{i}]"), &format!("raw{}", offset + i));
420                }
421            }
422            offset += fields.len();
423            outputs.push((argument.scoped, output));
424        }
425        let mut locals = String::from("        var callbackCleanupFailure: Throwable? = null\n");
426        let mut materialize = String::new();
427        let mut cleanup = String::new();
428        let mut scoped_cleanup = String::new();
429        let mut delivered = vec![];
430        for (argument, (scoped, output)) in outputs.iter().enumerate() {
431            locals.push_str(&output.output_locals);
432            materialize.push_str(&output.materialize);
433            cleanup.push_str(&output.cleanup);
434            if *scoped {
435                scoped_cleanup.push_str(&output.cleanup);
436            }
437            for (value, (ty, expression)) in output
438                .result_types
439                .iter()
440                .zip(&output.delivered)
441                .enumerate()
442            {
443                let name = format!("callbackArgument{argument}_{value}");
444                locals.push_str(&argument_local(&name, ty));
445                materialize.push_str(&format!("            {name} = {expression}\n"));
446                delivered.push(name);
447            }
448        }
449        let reconstruction = format!("try {{\n{materialize}        }} catch (failure: Throwable) {{\n{cleanup}            callbackCleanupFailure?.let {{ if (failure !== it) failure.addSuppressed(it) }}\n            throw failure\n        }}");
450        let mut call = format!("{parameter}.run({})", delivered.join(", "));
451        let (return_type, finish_return) = if let Some(result) = &self.return_value {
452            locals.push_str(&argument_local(
453                "callbackReturned",
454                &result.kotlin_type(context)?,
455            ));
456            call = format!("callbackReturned = {call}");
457            if !scoped_cleanup.is_empty() && !result.input.kotlin(context)?.consumed.is_empty() {
458                locals.push_str("var callbackReturnReady = false\n");
459                call.push_str("; callbackReturnReady = true");
460            }
461            (
462                format!(": {}", result.raw_kotlin_type()),
463                result.kotlin_encode("callbackReturned", context)?,
464            )
465        } else {
466            (String::new(), String::new())
467        };
468        let body = if scoped_cleanup.is_empty() {
469            format!("{reconstruction}\n        {call}")
470        } else {
471            format!("var callbackFailure: Throwable? = null\n        var callbackReady = false\n        try {{\n            {reconstruction}\n            callbackReady = true\n            {call}\n        }} catch (failure: Throwable) {{\n            callbackFailure = failure\n            throw failure\n        }} finally {{\n            if (callbackReady) {{\n{scoped_cleanup}                callbackCleanupFailure?.let {{ cleanup -> if (callbackFailure == null) throw cleanup else if (callbackFailure !== cleanup) callbackFailure.addSuppressed(cleanup) }}\n            }}\n        }}")
472        };
473        let body = if let Some(result) = &self.return_value {
474            if !scoped_cleanup.is_empty() {
475                if let Some(cleanup) = result.kotlin_cleanup("callbackReturned", context)? {
476                    format!("try {{ {body} }} catch (failure: Throwable) {{ if (callbackReturnReady) {{ {cleanup} }}; throw failure }}")
477                } else {
478                    body
479                }
480            } else {
481                body
482            }
483        } else {
484            body
485        };
486        let body = format!("{body}\n{finish_return}");
487        Ok(format!("if ({parameter} == null) null else object {{\n    @Suppress(\"UNUSED_PARAMETER\", \"UNUSED_VARIABLE\", \"UNCHECKED_CAST\", \"UNNECESSARY_NOT_NULL_ASSERTION\", \"USELESS_CAST\", \"VARIABLE_WITH_REDUNDANT_INITIALIZER\")\n    fun run({}){return_type} {{\n{locals}        {body}\n    }}\n}}", raw.join(", ")))
488    }
489}
490fn has_reference(ty: &syn::Type) -> bool {
491    matches!(ty, syn::Type::Reference(_))
492        || prebindgen_flat::option_inner(ty).is_some_and(has_reference)
493        || prebindgen_flat::sequence_parts(ty).is_some_and(|(t, _)| has_reference(t))
494}
495impl CallbackReturn {
496    fn fields(&self) -> Vec<&Wire<JniWire>> {
497        group::argument_parameters(self.input.input_node()).collect()
498    }
499    fn descriptor(&self) -> String {
500        match self.fields().as_slice() {
501            [field] => field.kind.jvm_descriptor().into(),
502            _ => "[Ljava/lang/Object;".into(),
503        }
504    }
505    fn raw_kotlin_type(&self) -> String {
506        match self.fields().as_slice() {
507            [field] => field.kind.kotlin_type().into(),
508            _ => "Array<Any?>".into(),
509        }
510    }
511    fn kotlin_type(&self, context: &KotlinContext<'_>) -> Result<String> {
512        let input = self.input.kotlin(context)?;
513        if input.parameters.len() != 1 {
514            return Err("callback return conversion must expose exactly one Kotlin value".into());
515        }
516        Ok(input.parameters[0]
517            .split_once(": ")
518            .ok_or("invalid callback return parameter")?
519            .1
520            .into())
521    }
522    fn kotlin_encode(&self, call: &str, context: &KotlinContext<'_>) -> Result<String> {
523        self.kotlin_type(context)?;
524        let input = self.input.kotlin(context)?;
525        let name = input.parameters[0].split_once(": ").unwrap().0;
526        let binding = format!("val {name} = {call}\n");
527        let mut setup = String::new();
528        for binding in input.setup {
529            setup.push_str(&format!("val {} = {}\n", binding.name, binding.expression));
530        }
531        let raw = match input.arguments.as_slice() {
532            [value] => value.clone(),
533            values => format!("kotlin.arrayOf<kotlin.Any?>({})", values.join(", ")),
534        };
535        let mut body = format!(
536            "val callbackReturnWire = {raw}\nval __pb_consume: ({}.NativeHandle) -> Unit = {{ it.markConsumed() }}\n{}\ncallbackReturnWire",
537            context.package,
538            input.consumed.join("; ")
539        );
540        if !input.locks.is_empty() || !input.lock_groups.is_empty() {
541            let prefix = &self.package;
542            if input.lock_groups.is_empty() && input.locks.len() <= 3 {
543                // Kotlin evaluates all handle arguments before acquiring monitors.
544                // Reuse the ordinary-call overloads without allocating a lock list.
545                body = format!(
546                    "{prefix}.withHandleLocks({}) {{ {body} }}",
547                    input.locks.join(", ")
548                );
549            } else {
550                let handles = crate::kotlin::lock_list(&input.locks, &input.lock_groups);
551                body = format!("{prefix}.withHandleLocks({handles}) {{ {body} }}");
552            }
553        }
554        body = format!("{setup} {body}");
555        if let Some(cleanup) = self.kotlin_cleanup(name, context)? {
556            body = format!(
557                "try {{ {body} }} catch (failure: Throwable) {{ {cleanup}; throw failure }}"
558            );
559        }
560        Ok(format!("return kotlin.run {{ {binding} {body} }}"))
561    }
562    fn kotlin_cleanup(&self, value: &str, context: &KotlinContext<'_>) -> Result<Option<String>> {
563        if self.input.kotlin(context)?.consumed.is_empty() {
564            return Ok(None);
565        }
566        let prefix = &self.package;
567        let collect = returned_handle_cleanup(self.input.input_node(), value, prefix, &mut 0)?;
568        Ok(Some(format!("val returnedOwners = kotlin.collections.ArrayList<{prefix}.NativeHandle>(); val returnedHandles = kotlin.collections.ArrayList<{prefix}.NativeHandle>(); {collect}\nval released = kotlin.collections.HashSet<Long>(); for (owner in returnedOwners) {{ if (owner.isClosed) continue; try {{ if (released.add(owner.address)) owner.close() else owner.markConsumed() }} catch (cleanup: Throwable) {{ if (failure !== cleanup) failure.addSuppressed(cleanup) }} }}; for (handle in returnedHandles) {{ if (handle.address in released) handle.markConsumed() }}")))
569    }
570    fn ownership_guard(&self, context: &JniContext) -> Result<(TokenStream, TokenStream)> {
571        let fields = self.fields();
572        if let [field] = fields.as_slice() {
573            if let JniWire::Handle { handle, .. } = &field.kind {
574                if handle.access == Access::Move {
575                    let name = &field.name;
576                    let ty = &handle.ty;
577                    return Ok((
578                        quote!(let _callback_return_owner=unsafe {::prebindgen_runtime::TransferredBox::new(#name as *mut ::prebindgen_jni_runtime::Allocation<#ty>, __prebindgen_input_consumed)}?;),
579                        quote!(),
580                    ));
581                }
582            }
583        }
584        let scan = receive_return_owners(self.input.input_node(), context)?;
585        if scan.is_empty() {
586            return Ok((quote!(), quote!()));
587        }
588        Ok((
589            quote!(let mut callback_return_owners=::prebindgen_runtime::CallbackReturnAllocations::tracking(__prebindgen_input_consumed); #scan callback_return_owners.check()?;),
590            quote!(),
591        ))
592    }
593    fn rust_invoke(
594        &self,
595        context: &JniContext,
596        retained: &syn::Ident,
597        env: &syn::Ident,
598    ) -> Result<TokenStream> {
599        let mut input = self.input.clone();
600        let cache = crate::bulk::callback_cache(input.input_node(), context)?;
601        input.rust.validation = crate::bulk::cached_reads(&input.rust.validation)?;
602        let fields = self.fields();
603        let result = format_ident!("callback_result");
604        let (owners, commit_owners) = self.ownership_guard(context)?;
605        let ownership_reads = quote!(#owners #cache);
606        let (return_type, decode, late_decode) = match fields.as_slice() {
607            [field] => {
608                let name = &field.name;
609                let read = field.kind.read_callback_return(&result);
610                (
611                    field.kind.callback_return_type(),
612                    quote!(let #name = #read;),
613                    quote!(),
614                )
615            }
616            _ => {
617                let mut reads = Vec::new();
618                let mut late_reads = Vec::new();
619                for (i, field) in fields.iter().enumerate() {
620                    let name = &field.name;
621                    let read = field
622                        .kind
623                        .read_array_element(&quote!(callback_result_rows), i);
624                    let code = quote!(let #name=#read?;);
625                    if mentions_value(&ownership_reads, name) {
626                        reads.push(code);
627                    } else {
628                        late_reads.push(code);
629                    }
630                }
631                (
632                    quote!(::prebindgen_jni_runtime::jni::signature::ReturnType::Object),
633                    quote!(
634                        let callback_result_rows = ::prebindgen_jni_runtime::jni::objects::JObjectArray::from(#result.l().map_err(|e|e.to_string())?);
635                        #(#reads)*
636                    ),
637                    quote!(#(#late_reads)*),
638                )
639            }
640        };
641        let target = format_ident!("callback_return_value");
642        let ty = &self.source_type;
643        let expected = syn::parse_quote!(::prebindgen_runtime::Result<#ty>);
644        let mut body = input.code_fragment_slot(
645            InputFragment::Construct {
646                context,
647                target: &target,
648                next: quote!(Ok(#target)),
649            },
650            &expected,
651        )?;
652        body = input.code_fragment_slot(
653            InputFragment::Acquire {
654                context,
655                next: body,
656            },
657            &expected,
658        )?;
659        body = quote!({ #commit_owners #body });
660        let checks = input.code_fragment_slot(
661            InputFragment::CheckInternalAliases { context },
662            &syn::parse_quote!(()),
663        )?;
664        if checks.to_string() != quote!(()).to_string() {
665            body = quote!({let _: () = #checks; #body});
666        }
667        body = input.code_fragment_slot(
668            InputFragment::Validate {
669                context,
670                next: body,
671            },
672            &expected,
673        )?;
674        let mut alias_check = prebindgen_conversions::staged::ParameterAliasCheck::default();
675        alias_check.include_parameter(context, &input.rust.resources)?;
676        let aliases = alias_check.initialize()?;
677        context.render(&quote!({
678            let __pb_input_consumed=::core::cell::Cell::new(0usize);
679            let __prebindgen_input_consumed=&__pb_input_consumed;
680            #aliases
681            let #result = unsafe { #retained.invoke_return(#env, #return_type, &arguments) }?;
682            #decode
683            #cache
684            #owners
685            #late_decode
686            #body
687        }))
688    }
689}
690// Only read ownership-bearing fields and their controls during the guard scan.
691// Fallible reads of ordinary fields belong after the guard has been installed.
692// Traverse the configured Kotlin shape on the error path. The wire tree keeps
693// source field names and ownership, so borrowed fields are never reclaimed.
694fn returned_handle_cleanup(
695    node: &crate::kotlin::JniNode,
696    value: &str,
697    prefix: &str,
698    next: &mut usize,
699) -> Result<String> {
700    if !crate::kotlin::input_handles(node, false) {
701        return Ok(String::new());
702    }
703    let local = format!("returnedValue{}", *next);
704    *next += 1;
705    Ok(match node {
706        WireNode::Leaf(wires) => {
707            let owned = wires.iter().any(
708                |w| matches!(&w.kind,JniWire::Handle{handle,..} if handle.access==Access::Move),
709            );
710            format!("({value} as? {prefix}.NativeHandle)?.let {{ {local} -> returnedHandles.add({local}); {} }}\n",if owned{format!("returnedOwners.add({local})")}else{String::new()})
711        }
712        WireNode::Chain {
713            meta: crate::JniChain::Sequence(_),
714            child,
715            ..
716        } => {
717            let body = returned_handle_cleanup(child, &local, prefix, next)?;
718            format!("({value} as? Iterable<*>)?.forEach {{ {local} -> {body} }}\n")
719        }
720        WireNode::Chain { child, .. } => returned_handle_cleanup(child, value, prefix, next)?,
721        WireNode::Branch {
722            meta: crate::JniBranch::Group(crate::JniGroup::Fields(Some(data))),
723            children,
724            ..
725        } => {
726            let mut body = String::new();
727            for (field, child) in crate::data::fields(children)? {
728                let expr = format!("{local}.`{}`", field.to_string().trim_start_matches("r#"));
729                body.push_str(&returned_handle_cleanup(child, &expr, prefix, next)?);
730            }
731            format!(
732                "({value} as? {})?.let {{ {local} -> {body} }}\n",
733                data.class_name
734            )
735        }
736        WireNode::Branch { children, .. } => {
737            let mut body = String::new();
738            for child in children {
739                body.push_str(&returned_handle_cleanup(child, value, prefix, next)?);
740            }
741            body
742        }
743    })
744}

746fn mentions_value(code: &TokenStream, name: &syn::Ident) -> bool {
747    code.clone().into_iter().any(|token| match token {
748        proc_macro2::TokenTree::Ident(value) => value == *name,
749        proc_macro2::TokenTree::Group(group) => mentions_value(&group.stream(), name),
750        _ => false,
751    })
752}
753pub(crate) fn receive_return_owners(
754    node: &kotlin::JniNode,
755    context: &JniContext,
756) -> Result<TokenStream> {
757    Ok(match node {
758        WireNode::Chain {meta:JniChain::Bulk(layout),controls,child} => crate::bulk::scan_bulk(layout,controls,child,context)?,
759        WireNode::Chain {meta:JniChain::Sequence(JniSequence::Bulk),controls,child} => crate::bulk::scan_rows(controls,child,context)?,
760        WireNode::Leaf(fields)=>fields.iter().filter_map(|field| {
761            if let JniWire::Handle {handle,..}=&field.kind {
762                if handle.access==Access::Move {let name=&field.name;let ty=&handle.ty;return Some(quote!(unsafe {callback_return_owners.receive(#name as *mut ::prebindgen_jni_runtime::Allocation<#ty>);}));}
763            }
764            None
765        }).collect(),
766        WireNode::Chain { meta:JniChain::Optional(optional),child,..}=> {
767            let scan=receive_return_owners(child,context)?;
768            if scan.is_empty(){return Ok(scan);}
769            let name=optional.discriminator();
770            let present=if let Some(niche)=optional.niche(){let field=group::argument_parameters(child).find(|w|w.name==*name).ok_or("missing callback return optional discriminator")?; niche.present(&field.kind,quote!(#name))} else {quote!(#name==1)};
771            quote!(if #present {#scan})
772        }
773        WireNode::Chain {meta:JniChain::Sequence(_),controls,child}=> {
774            let scan=receive_return_owners(child,context)?;
775            if scan.is_empty(){return Ok(scan);}
776            let carrier=&controls[0].name;let row=format_ident!("{carrier}_ownership_row");
777            let fields=group::argument_parameters(child).collect::<Vec<_>>();let count=fields.len();let capacity=i32::try_from(count+8).map_err(|_|"too many returned fields")?;
778            let reads=fields.into_iter().enumerate().filter(|(_,field)|mentions_value(&scan,&field.name)).map(|(index,field)|{let name=&field.name;let read=field.kind.read_array_element(&quote!(#row),index);quote!(let #name=#read?;)});
779            let env=&context.environment;
780            let code=quote!(if !#carrier.is_null() {let length=#env.get_array_length(&#carrier).map_err(|e|e.to_string())? as usize;for index in 0..length {
781                let frame=unsafe {::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;
782                let #row=::prebindgen_jni_runtime::row(&mut #env,&#carrier,index,#count)?;
783                #(#reads)* #scan drop(frame);
784            }});
785            context.render(&code)?
786        }
787        WireNode::Branch {meta:JniBranch::Alternatives,controls,children}=> {
788            let selector=&controls[0].name;let scans=children.iter().enumerate().map(|(i,child)|{let scan=receive_return_owners(child,context)?;let i=i as i32;Ok(quote!(if #selector==#i {#scan}))}).collect::<Result<Vec<_>>>()?;quote!(#(#scans)*)
789        }
790        WireNode::Branch {children,..}=>children.iter().map(|child|receive_return_owners(child,context)).collect::<Result<Vec<_>>>()?.into_iter().collect(),
791    })
792}
793fn render(context: &JniContext, code: &TokenStream, source: &syn::Ident) -> Result<TokenStream> {
794    let code = prebindgen_conversions::render_template(code, &|name| {
795        Ok(if name == "source_result" {
796            quote!(#source)
797        } else {
798            prebindgen_conversions::template_parameter(name)
799        })
800    })?;
801    context.render(&code)
802}
803pub(crate) fn files(
804    flat: &Bindings,
805    classes: &BTreeMap<String, KotlinClass>,
806    files: &mut BTreeMap<String, String>,
807) -> Result<()> {
808    let mut ownership_by_name = BTreeMap::new();
809    let mut wires = Vec::new();
810    for input in &flat.metadata.callbacks {
811        wires.extend(prebindgen_flat::wire_definitions(std::iter::once(
812            input.input_node(),
813        )));
814    }
815    for function in flat.functions.values().filter(|f| f.metadata.is_some()) {
816        wires.extend(function.input_wire_definitions()?);
817    }
818    for wire in wires {
819        let JniWire::Callback { callback } = &wire.kind else {
820            continue;
821        };
822        let context = KotlinContext {
823            package: &flat.metadata.package,
824            classes,
825            result: "parts",
826            failure: "failure",
827            cleanup_failure: None,
828            output_prefix: "output",
829        };
830        let mut params = vec![];
831        let mut names = BTreeSet::new();
832        for (i, arg) in callback.arguments.iter().enumerate() {
833            let output = arg.output.kotlin(&context)?;
834            output.callback_parameter_declarations()?;
835            for (j, ty) in output.result_types.iter().enumerate() {
836                let mut name = output
837                    .result_names
838                    .get(j)
839                    .and_then(|n| n.clone())
840                    .unwrap_or_else(|| format!("arg{i}_{j}"));
841                while !names.insert(name.clone()) {
842                    name.push('_');
843                }
844                params.push(format!("`{name}`: {ty}"));
845            }
846        }
847        let return_type = callback
848            .return_value
849            .as_ref()
850            .map(|r| r.kotlin_type(&context).map(|ty| format!(": {ty}")))
851            .transpose()?
852            .unwrap_or_default();
853        let contract = if let Some(result) = &callback.return_value {
854            let ownership = if result.input.kotlin(&context)?.consumed.is_empty() {
855                ""
856            } else {
857                " Returned handles transfer ownership to Rust."
858            };
859            format!(" Failure to invoke or convert the returned value raises a Rust panic on the invoking thread; an uncaught worker panic terminates that worker.{ownership}")
860        } else {
861            String::new()
862        };
863        let ownership = callback
864            .arguments
865            .iter()
866            .map(|arg| {
867                let handles = prebindgen_flat::wire_definitions(std::iter::once(
868                    &arg.output.output_outcomes().success,
869                ))
870                .any(|wire| matches!(wire.kind, JniWire::Handle { .. }));
871                handles && !arg.scoped
872            })
873            .collect::<Vec<_>>();
874        if let Some(previous) = ownership_by_name.insert(callback.fqn(), ownership.clone()) {
875            if previous != ownership {
876                return Err("callback interface ownership collision; use callback!(...).name(...) for a distinct interface".into());
877            }
878        }
879        let text = format!("package {}\n\n/** Invoked by Rust, possibly from a native worker thread.{contract} */\npublic fun interface {} {{\n    public fun run({}){return_type}\n}}\n", callback.package, callback.name, params.join(", "));
880        let path = format!(
881            "{}/{}.kt",
882            callback.package.replace('.', "/"),
883            callback.name
884        );
885        if let Some(existing) = files.get(&path) {
886            if existing != &text {
887                return Err("callback interface signature collision; use callback!(...).name(...) for a distinct interface".into());
888            }
889        } else {
890            files.insert(path, text);
891        }
892    }
893    Ok(())
894}

896fn subject_type(ty: &syn::Type) -> &syn::Type {
897    match ty {
898        syn::Type::Reference(r) => subject_type(&r.elem),
899        _ => prebindgen_flat::option_inner(ty)
900            .map(subject_type)
901            .unwrap_or(ty),
902    }
903}
904fn subject_name(ty: &syn::Type) -> String {
905    match ty {
906        syn::Type::Reference(r) => subject_name(&r.elem),
907        syn::Type::Slice(s) => format!("{}List", subject_name(&s.elem)),
908        _ => {
909            if let Some(inner) = prebindgen_flat::option_inner(ty) {
910                return format!("Optional{}", subject_name(inner));
911            }
912            if let Some(inner) = prebindgen_flat::vec_inner(ty) {
913                return format!("{}Vec", subject_name(inner));
914            }
915            match ty {
916                syn::Type::Path(p) => p.path.segments.last().unwrap().ident.to_string(),
917                _ => "Unit".into(),
918            }
919        }
920    }
921}

923fn contains_result(ty: &syn::Type) -> bool {
924    prebindgen_flat::result_parts(ty).is_some()
925        || prebindgen_flat::option_inner(ty).is_some_and(contains_result)
926        || prebindgen_flat::sequence_parts(ty).is_some_and(|(ty, _)| contains_result(ty))
927        || matches!(ty,syn::Type::Reference(r) if contains_result(&r.elem))
928}
929fn argument_local(name: &str, ty: &str) -> String {
930    let default = if ty.ends_with('?') {
931        Some("null")
932    } else {
933        let kotlin_type = ty.strip_prefix("kotlin.").unwrap_or(ty);
934        crate::wire::jvm::primitive_local_default(kotlin_type).or(match kotlin_type {
935            "UByte" => Some("0.toUByte()"),
936            "UShort" => Some("0.toUShort()"),
937            "UInt" => Some("0U"),
938            "ULong" => Some("0UL"),
939            "Char" => Some("'\\u0000'"),
940            _ => None,
941        })
942    };
943    match default {
944        Some(default) => format!("        var {name}: {ty} = {default}\n"),
945        None => format!("        lateinit var {name}: {ty}\n"),
946    }
