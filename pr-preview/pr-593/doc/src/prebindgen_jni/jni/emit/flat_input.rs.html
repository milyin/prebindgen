14/// Takes the **element**, not the `syn::ItemStruct` it was parsed from (#289):
15/// `flat::Field::ty` is already a `TypeRef`, so every peel below is the model's
16/// answer rather than a last-path-segment test on tokens that had a reading one
17/// level up. Same move `build_flat_struct_node` made for the flatten path; this
18/// is the whole-object `.jobject_input()` decoder.
19#[derive(Clone)]
20pub(crate) struct JObjectStructInputPlan {
21    shape: flat::Struct,
22    source_module: syn::Path,
23    fields: Vec<JObjectStructFieldPlan>,
24}

26impl JObjectStructInputPlan {
27    /// Every converter the decoder calls, one per property.
28    pub(crate) fn calls(&self, out: &mut Vec<prebindgen_registry::write::ArtifactKey>) {
29        for field in &self.fields {
30            field.kind.calls(out);
31        }
32    }

35#[derive(Clone)]
36struct JObjectStructFieldPlan {
37    name: syn::Ident,
38    property: String,
39    error: String,
40    sum_payload: bool,
41    kind: JObjectStructFieldKind,

44impl JObjectStructFieldKind {
45    /// The pipeline this property decodes through.
46    fn calls(&self, out: &mut Vec<prebindgen_registry::write::ArtifactKey>) {
47        match self {
48            Self::Handle { pipeline, .. }
49            | Self::Unsigned64 { pipeline, .. }
50            | Self::Enum { pipeline, .. }
51            | Self::Primitive { pipeline, .. }
52            | Self::IntoObject { pipeline, .. }
53            | Self::Object { pipeline, .. } => pipeline.calls(out),
54        }
55    }
56}

91/// Freeze the explicit whole-`JObject` struct decoder without assembling Rust
92/// source. Every branch below selects JVM property policy from Flat readings
93/// and already-compiled registry fragments. The final writer alone constructs
94/// bindings, child calls, and the source struct literal.
95pub(crate) fn build_jobject_struct_input_plan(
96    ext: &Declarations,
97    s: &flat::Struct,
98    registry: &impl Conversions,
99) -> Option<JObjectStructInputPlan> {
100    let struct_name = s.name.to_string();
101    let mut fields = Vec::new();
102    for field in &s.fields {
103        let name = field.name.clone()?;
104        let property = kotlin_property_name(&name);
105        let error = format!("{struct_name}.{property}: {{}}");
106        fields.push(build_jobject_property_plan(
107            ext, &field.ty, name, property, error, false,
108        )?);
109    }
110    Some(JObjectStructInputPlan {
111        shape: s.clone(),
112        source_module: struct_module_path(ext, registry, &s.name),
113        fields,
114    })
115}

117fn build_jobject_property_plan(
118    ext: &Declarations,
119    reading: &TypeRef,
120    name: syn::Ident,
121    property: String,
122    error: String,
123    sum_payload: bool,
124) -> Option<JObjectStructFieldPlan> {
125    let optional = reading.optional_inner().is_some();
126    let inner = reading.optional_inner().unwrap_or(reading);
127    let entry = ext.in_frag(reading)?;
128    let wire = entry.destination().clone();
129    let projection = entry.projection();
130    let whole = entry.pipeline(
131        prebindgen_registry::recipe::Direction::Construct,
132        prebindgen_registry::recipe::Mode::Owned,
133    );
134    let kind = if let Some(projection) = projection
135        .as_ref()
136        .filter(|projection| matches!(&projection.kind, ProjectionKind::Handle))
137    {
138        JObjectStructFieldKind::Handle {
139            descriptor: format!("L{};", handle_field_fqn(ext, projection).replace('.', "/")),
140            pipeline: whole,
141        }
142    } else if let Some(projection) = projection
143        .as_ref()
144        .filter(|projection| !sum_payload && matches!(&projection.kind, ProjectionKind::Unsigned64))
145    {
146        let niche = optional
147            && matches!(
148                projection.strategy,
149                FoldStrategy::Optional(NullableKind::Niche, _)
150            );
151        let pipeline = if optional && !niche {
152            ext.in_frag(inner)?.pipeline(
153                prebindgen_registry::recipe::Direction::Construct,
154                prebindgen_registry::recipe::Mode::Owned,
155            )
156        } else {
157            whole
158        };
159        JObjectStructFieldKind::Unsigned64 {
160            optional,
161            wrap_some: optional && !niche,
162            pipeline,
163        }
164    } else if ext.is_kotlin_enum_reading(inner) {
165        let fqn = match inner.unwrapped().kind() {
166            flat::TypeKind::Named { id, .. } => id.ident(),
167            _ => None,
168        }
169        .and_then(|ident| ext.kotlin_fqn(&TypeKey::from_ident(&ident)))?;
170        let pipeline = if optional {
171            ext.in_frag(inner)?.pipeline(
172                prebindgen_registry::recipe::Direction::Construct,
173                prebindgen_registry::recipe::Mode::Owned,
174            )
175        } else {
176            whole
177        };
178        JObjectStructFieldKind::Enum {
179            descriptor: format!("L{};", fqn.replace('.', "/")),
180            optional,
181            pipeline,
182        }
183    } else {
184        match jni_field_access(&wire) {
185            Some((descriptor, accessor, false)) => JObjectStructFieldKind::Primitive {
186                wire,
187                descriptor: descriptor.to_string(),
188                accessor,
189                pipeline: whole,
190            },
191            Some((descriptor, _, true)) => JObjectStructFieldKind::IntoObject {
192                wire,
193                descriptor: descriptor.to_string(),
194                pipeline: whole,
195            },
196            None => {
197                let descriptor = ext
198                    .in_frag(inner)
199                    .and_then(|entry| jni_field_access(entry.destination()))
200                    .and_then(|(descriptor, _, is_object)| {
201                        if is_object {
202                            Some(descriptor.to_string())
203                        } else {
204                            box_descriptor_for_primitive(descriptor).map(str::to_string)
205                        }
206                    })
207                    .or_else(|| {
208                        match inner.unwrapped().kind() {
209                            flat::TypeKind::Named { id, .. } => id.ident(),
210                            _ => None,
211                        }
212                        .and_then(|ident| {
213                            ext.kotlin_fqn(&TypeKey::from_ident(&ident))
214                                .map(|fqn| format!("L{};", fqn.replace('.', "/")))
215                        })
216                    })
217                    .or_else(|| {
218                        inner
219                            .sequence_elem()
220                            .is_some()
221                            .then(|| "Ljava/util/List;".to_string())
222                    })
223                    .unwrap_or_else(|| "Ljava/lang/Object;".to_string());
224                JObjectStructFieldKind::Object {
225                    descriptor,
226                    pipeline: whole,
227                }
228            }
229        }
230    };
231    Some(JObjectStructFieldPlan {
232        name,
233        property,
234        error,
235        sum_payload,
236        kind,
237    })
238}

240impl JObjectStructInputPlan {
241    pub(crate) fn render(&self, emit: &prebindgen_registry::RustWriter) -> syn::Expr {
242        let mut preludes = Vec::new();
243        let mut inits = Vec::new();
244        for field in &self.fields {
245            preludes.push(field.render(quote!(v), emit));
246            let name = &field.name;
247            inits.push(quote!(#name));
248        }
249        let module = &self.source_module;
250        let ident = &self.shape.name;
251        let ctor = emit.shape_struct(&self.shape, quote!(#module::#ident), &inits);
252        syn::parse_quote!({
253            #(#preludes)*
254            #ctor
255        })
256    }
257}

259impl JObjectStructFieldPlan {
260    fn render(&self, receiver: TokenStream, emit: &prebindgen_registry::RustWriter) -> TokenStream {
261        let name = &self.name;
262        let property = &self.property;
263        let error = &self.error;
264        let raw = if self.sum_payload {
265            format_ident!("{}_raw", name)
266        } else {
267            format_ident!("__{}_raw", name)
268        };
269        let object = if self.sum_payload {
270            format_ident!("{}_obj", name)
271        } else {
272            format_ident!("__{}_jobj", name)
273        };
274        match &self.kind {
275            JObjectStructFieldKind::Handle {
276                descriptor,
277                pipeline,
278            } => {
279                let decode = pipeline.invoke_converter(quote!(env), quote!(#raw), name, emit);
280                quote! {
281                    let #object: jni::objects::JObject = env.get_field(#receiver, #property, #descriptor)
282                        .and_then(|val| val.l())
283                        .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))?;
284                    let #raw: jni::sys::jlong = if #object.is_null() {
285                        0
286                    } else {
287                        env.call_method(&#object, "peek", "()J", &[])
288                            .and_then(|val| val.j())
289                            .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))?
290                    };
291                    let #name = #decode;
292                }
293            }
294            JObjectStructFieldKind::Unsigned64 {
295                optional,
296                wrap_some,
297                pipeline,
298            } => {
299                let decode = pipeline.invoke_converter(quote!(env), quote!(#raw), name, emit);
300                if *optional {
301                    let decoded = if *wrap_some {
302                        quote!(::core::option::Option::Some(#decode))
303                    } else {
304                        quote!(#decode)
305                    };
306                    quote! {
307                        let #object: jni::objects::JObject = env
308                            .get_field(#receiver, #property, "Lkotlin/ULong;")
309                            .and_then(|val| val.l())
310                            .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))?;
311                        let #name = if #object.is_null() {
312                            ::core::option::Option::None
313                        } else {
314                            let #raw: jni::sys::jlong = env
315                                .call_method(&#object, "unbox-impl", "()J", &[])
316                                .and_then(|val| val.j())
317                                .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))?;
318                            #decoded
319                        };
320                    }
321                } else {
322                    quote! {
323                        let #raw: jni::sys::jlong = env
324                            .get_field(#receiver, #property, "J")
325                            .and_then(|val| val.j())
326                            .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))?;
327                        let #name = #decode;
328                    }
329                }
330            }
331            JObjectStructFieldKind::Enum {
332                descriptor,
333                optional,
334                pipeline,
335            } => {
336                let decode = pipeline.invoke_converter(quote!(env), quote!(#raw), name, emit);
337                if *optional {
338                    quote! {
339                        let #object: jni::objects::JObject = env.get_field(#receiver, #property, #descriptor)
340                            .and_then(|val| val.l())
341                            .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))?;
342                        let #name = if #object.is_null() {
343                            ::core::option::Option::None
344                        } else {
345                            let #raw: jni::sys::jint = env.call_method(&#object, "getValue", "()I", &[])
346                                .and_then(|val| val.i())
347                                .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))?;
348                            ::core::option::Option::Some(#decode)
349                        };
350                    }
351                } else {
352                    quote! {
353                        let #object: jni::objects::JObject = env.get_field(#receiver, #property, #descriptor)
354                            .and_then(|val| val.l())
355                            .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))?;
356                        let #raw: jni::sys::jint = env.call_method(&#object, "getValue", "()I", &[])
357                            .and_then(|val| val.i())
358                            .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))?;
359                        let #name = #decode;
360                    }
361                }
362            }
363            JObjectStructFieldKind::Primitive {
364                wire,
365                descriptor,
366                accessor,
367                pipeline,
368            } => {
369                let decode = pipeline.invoke_converter(quote!(env), quote!(#raw), name, emit);
370                quote! {
371                    let #raw: #wire = env.get_field(#receiver, #property, #descriptor)
372                        .and_then(|val| val.#accessor())
373                        .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))? as _;
374                    let #name = #decode;
375                }
376            }
377            JObjectStructFieldKind::IntoObject {
378                wire,
379                descriptor,
380                pipeline,
381            } => {
382                let decode = pipeline.invoke_converter(quote!(env), quote!(#raw), name, emit);
383                quote! {
384                    let #object: jni::objects::JObject = env.get_field(#receiver, #property, #descriptor)
385                        .and_then(|val| val.l())
386                        .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))?;
387                    let #raw: #wire = #object.into();
388                    let #name = #decode;
389                }
390            }
391            JObjectStructFieldKind::Object {
392                descriptor,
393                pipeline,
394            } => {
395                let decode = pipeline.invoke_converter(quote!(env), quote!(#raw), name, emit);
396                quote! {
397                    let #raw: jni::objects::JObject = env.get_field(#receiver, #property, #descriptor)
398                        .and_then(|val| val.l())
399                        .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!(#error, e)))?;
400                    let #name = #decode;
401                }
402            }
403        }
404    }
405}

407/// Whole-object **input** decode for a `sealed_class` sum: a `JObject` of the
408/// sealed interface → the Rust enum, by `instanceof` dispatch over the nested
409/// variant classes and a property read per payload.
410///
411/// This is the counterpart of a nested data class's own input converter, not
412/// a second mechanism: a sum reached as a **field** sits inside a parent
413/// `JObject` that is already being read field-by-field, so the parent's
414/// generic field branch just delegates here — the same delegation
415/// `Annotated.payload` uses. The tag-gated *flattened* form is the separate
416/// parameter path.
417///
418/// The output direction deliberately has no such converter: a sum crosses
419/// Rust → Kotlin **flattened, always** (`PlanFieldKind::Sum`), which is the
420/// design's rejected-alternative note about per-crossing JVM objects. Reading
421/// one field out of a `JObject` the caller already handed us costs nothing
422/// extra, so the asymmetry is real rather than an oversight.
423/// Takes the **element**, not the `syn::ItemEnum` it was parsed from (#289):
424/// `Alternative::fields` carries a `TypeRef` per payload, so the property read
425/// below asks the model instead of peeling tokens. It also retires the two zips
426/// this used to run — a `SumSpec` derived from the item, paired back against the
427/// item it came from — because `Alternative` already is that pairing.
428#[derive(Clone)]
429pub(crate) struct JObjectSumInputPlan {
430    shape: flat::Variant,
431    source_module: syn::Path,
432    enum_name: String,
433    alternatives: Vec<JObjectSumAlternativePlan>,
434}
435
436#[derive(Clone)]
437struct JObjectSumAlternativePlan {
438    shape: flat::Alternative,
439    jvm_class: String,
440    fields: Vec<JObjectSumFieldPlan>,
441}
442
443#[derive(Clone)]
444struct JObjectSumFieldPlan {
445    shape: flat::Field,
446    property: JObjectStructFieldPlan,
447}
448
449impl JObjectSumInputPlan {
450    /// Every converter the decoder calls, one per property of every
451    /// alternative.
452    pub(crate) fn calls(&self, out: &mut Vec<prebindgen_registry::write::ArtifactKey>) {
453        for field in self
454            .alternatives
455            .iter()
456            .flat_map(|alternative| &alternative.fields)
457        {
458            field.property.kind.calls(out);
459        }
460    }
461}

463pub(crate) fn build_jobject_sum_input_plan(
464    ext: &Declarations,
465    v: &flat::Variant,
466    registry: &impl Conversions,
467) -> Option<JObjectSumInputPlan> {
468    let key = TypeKey::from_ident(&v.name);
469    let cfg = ext.types.get(&key)?;
470    let sum_cfg = cfg.sum()?;
471    let iface_fqn = cfg.name_spec.as_ref().map(|s| ext.fqn_of(s))?;
472    let iface_path = iface_fqn.replace('.', "/");
473    let enum_name = v.name.to_string();
474    let mut alternatives = Vec::new();
475    for alt in &v.alternatives {
476        let kotlin_name = ext.sum_variant_class_name(sum_cfg, &alt.name);
477        let jvm_class = format!("{iface_path}${kotlin_name}");
478        let mut fields = Vec::new();
479        for field in &alt.fields {
480            let property = crate::jni::struct_plan::sum_field_prop_name(&field.member());
481            let bind = format_ident!("__p_{}", property);
482            let error = format!("{enum_name}.{kotlin_name}.{property}: {{}}");
483            fields.push(JObjectSumFieldPlan {
484                shape: field.clone(),
485                property: build_jobject_property_plan(ext, &field.ty, bind, property, error, true)?,
486            });
487        }
488        alternatives.push(JObjectSumAlternativePlan {
489            shape: alt.clone(),
490            jvm_class,
491            fields,
492        });
493    }
494    Some(JObjectSumInputPlan {
495        shape: v.clone(),
496        source_module: ext.fn_module(registry, &v.name),
497        enum_name,
498        alternatives,
499    })
500}

502impl JObjectSumInputPlan {
503    pub(crate) fn render(&self, emit: &prebindgen_registry::RustWriter) -> syn::Expr {
504        let source_module = &self.source_module;
505        let enum_ident = &self.shape.name;
506        let enum_name = &self.enum_name;
507        let mut arms = Vec::new();
508        for alternative in &self.alternatives {
509            let vident = &alternative.shape.name;
510            let jvm_class = &alternative.jvm_class;
511            let mut preludes = Vec::new();
512            let mut inits = Vec::new();
513            for field in &alternative.fields {
514                preludes.push(field.property.render(quote!(__obj), emit));
515                let bind = &field.property.name;
516                inits.push(field.shape.bind(&quote!(#bind)));
517            }
518            let ctor = emit.shape_alternative(
519                &alternative.shape,
520                quote!(#source_module::#enum_ident::#vident),
521                &inits,
522            );
523            arms.push(quote! {
524                if env.is_instance_of(__obj, #jvm_class)
525                    .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(
526                        format!(concat!(#enum_name, ": instanceof ", #jvm_class, ": {}"), e)))?
527                {
528                    #(#preludes)*
529                    return ::core::result::Result::Ok(#ctor);
530                }
531            });
532        }
533        let no_match = format!("{enum_name}: value is not one of its declared variants");
534        let null_msg = format!("{enum_name}: null value where a variant was required");
535        syn::parse_quote!({
536            let __obj = v;
537            (|| -> ::core::result::Result<#source_module::#enum_ident, __JniErr> {
538                if __obj.is_null() {
539                    return ::core::result::Result::Err(
540                        <__JniErr as ::core::convert::From<String>>::from(#null_msg.to_string()),
541                    );
542                }
543                #(#arms)*
544                ::core::result::Result::Err(
545                    <__JniErr as ::core::convert::From<String>>::from(#no_match.to_string()),
546                )
547            })()?
548        })
549    }
550}

552// ──────────────────────────────────────────────────────────────────────
553// Struct input flattening (pass a data_class param as its leaf fields)
554// ──────────────────────────────────────────────────────────────────────
555
556/// One flattened leaf of a struct **input** param. The mirror of
557/// [`EncSlot`] for the input direction: instead of reading the field with
558/// `env.get_field(...)` out of a single `JObject`, the leaf crosses the JNI
559/// boundary as its own wrapper parameter, and the three coordinated sites — the
560/// native wrapper signature, the `JNINative` extern declaration and the Kotlin
561/// call-site destructure — read it so they cannot drift in order, type, or
562/// nullability.
563///
564/// One JNI parameter a flattened `data_class` occupies, as the site names it.
565///
566/// A projection of the fragment's [`Wire`](crate::jni::compile::Wire) and
567/// nothing more: the wire says what crosses and how Kotlin reaches it, and this
568/// pairs that with the one thing a wire cannot know — the parameter name the
569/// site hangs it off.
570pub(crate) struct FlatLeaf {
571    /// Native wrapper parameter ident — also the decode source.
572    pub native_ident: syn::Ident,
573    /// Kotlin `external fun` parameter name (camelCase).
574    pub kt_name: String,
575    /// The wire itself.
576    pub wire: crate::jni::compile::Wire,
577}
578
579impl FlatLeaf {
580    /// This wire as one parameter of the site named `param`.
581    fn of(param: &syn::Ident, wire: &crate::jni::compile::Wire) -> Self {
582        let native = format!("{param}_{}", wire.path.replace('.', "_"));
583        Self {
584            native_ident: format_ident!("{native}"),
585            kt_name: snake_to_camel(&native),
586            wire: wire.clone(),
587        }
588    }
589
590    /// Native wire type, lifetime-annotated for object wires.
591    pub fn native_wire_ty(&self) -> TokenStream {
592        annotate_jobject_with_lifetime(&self.wire.ty, "a").to_token_stream()
593    }
594
595    /// Kotlin `external fun` parameter type (incl. a trailing `?`).
596    pub fn kt_wire_ty(&self) -> &str {
597        &self.wire.kt_ty
598    }
599
600    /// Per-field input converter operation (`None` for a synthetic gate or tag).
601    pub fn conv(&self) -> Option<&prebindgen_registry::OperationId> {
602        self.wire.conv()
603    }
604
605    /// The complete conversion, stages included, or `None` for a gate or tag.
606    pub fn entry(&self) -> Option<&prebindgen_registry::ConverterImpl<KotlinMeta>> {
607        self.wire.entry.as_ref()
608    }
609
610    /// Whether this is a synthetic `…Present: Boolean` gate.
611    pub fn is_present_flag(&self) -> bool {
612        self.wire.is_present_flag()
613    }
614
615    /// Whether the handle access can be null, either because the field itself
616    /// is optional or because an optional ancestor gates it.
617    pub fn handle_nullable(&self) -> bool {
618        self.wire.handle_nullable
619    }
620
621    /// Kotlin call-site destructure expression feeding this leaf, rooted at
622    /// `base` — the object expression at this call site (the camelCase param
623    /// name, `this` for a promoted receiver, `__e` for the vec-build loop
624    /// variable).
625    pub fn kt_access(&self, base: &str) -> String {
626        self.wire.access.render(base)
627    }
628
629    /// The Kotlin expression yielding the handle this leaf carries, rooted at
630    /// `base` — the thing the lock scaffold locks and `markConsumed()`s.
631    /// `None` when the leaf is not a handle.
632    pub fn kt_handle_target(&self, base: &str) -> Option<String> {
633        self.wire
634            .handle_target
635            .as_ref()
636            .map(|walk| crate::jni::compile::reached(base, walk))
637    }
638
639    /// Native call argument for this leaf. Handle pointers are bound under
640    /// the unified lock scaffold; every other leaf is read directly from the
641    /// Kotlin object graph.
642    pub fn kt_call_arg(&self, base: &str) -> String {
643        if self.wire.handle_target.is_some() {
644            format!("{}_ptr", self.kt_name)
645        } else {
646            self.kt_access(base)
647        }
648    }
649}

651/// The outer layer a flattened call argument restores after its registry chain
652/// returns the value represented by the crossing.
653///
654/// Product/Optional/Choice construction belongs to the registry chain. The
655/// renderer only adds the source call's borrow; wrappers outside that borrow
656/// still belong afterwards (`Box<&S>` -> `Box::new(&value)`).
657pub(crate) struct RebuildTarget {
658    /// Wrappers over the borrow, if there is one — the `Box` of `Box<&S>`.
659    arg: Vec<&'static str>,
660    borrow: RebuildBorrow,
661}
662
663#[derive(Clone, Copy)]
664enum RebuildBorrow {
665    None,
666    Outer { mutable: bool },
667    OptionalInner { mutable: bool },
668}
669
670impl RebuildTarget {
671    /// Put back the wrappers over the **borrow** — the `Box` of `Box<&S>`,
672    /// which goes on after the call site has added its `&`.
673    ///
674    /// A no-op when the parameter is not a borrow: wrappers at or below the
675    /// crossing are already restored by the registry chain.
676    pub fn call_arg(&self, ident: &syn::Ident) -> TokenStream {
677        let value = match self.borrow {
678            RebuildBorrow::None => quote!(#ident),
679            RebuildBorrow::Outer { mutable: false } => quote!(&#ident),
680            RebuildBorrow::Outer { mutable: true } => quote!(&mut #ident),
681            RebuildBorrow::OptionalInner { mutable: false } => quote!(#ident.as_ref()),
682            RebuildBorrow::OptionalInner { mutable: true } => quote!(#ident.as_mut()),
683        };
684        if !matches!(self.borrow, RebuildBorrow::Outer { .. }) {
685            return value;
686        }
687        crate::jni::trait_impl::build_through_wrappers(&self.arg, value)
688            .expect("every layer was checked buildable when the plan was built")
689    }
690
691    fn mutable_binding(&self) -> bool {
692        matches!(
693            self.borrow,
694            RebuildBorrow::Outer { mutable: true } | RebuildBorrow::OptionalInner { mutable: true }
695        )
696    }
697}
698
699/// Descend an outer borrow, an `Option`, or the borrow inside an `Option` to the
700/// struct a specialized lowering will **rebuild**, keeping each layer's reading
701/// so its spelling can be restored.
702///
703/// One function because it is one rule, and the layers are checked **on the way
704/// down**: an erasure sits outside the layer it wraps, so `Box<&S>` classifies
705/// as `Ref` and interpreting `kind` before asking would discard the `Box`.
706///
707/// The only refusal left is a wrapper the adapter cannot **build** — `Cow`, by
708/// policy rather than by impossibility (see its `WRAPPER_OPS` recipe). A wrapped
709/// spelling that declines here keeps the general converter path, which is
710/// correct if less direct.
711fn rebuildable_target(arg: &TypeRef) -> Option<(RebuildTarget, &TypeRef)> {
712    // A probe per layer: "can this spelling be rebuilt at all", asked before the
713    // peel that would hide it. The token is irrelevant — only the `Option` is.
714    let buildable = |t: &TypeRef| build_through_erased_wrappers(t, quote!(__probe)).map(|_| ());
715    buildable(arg)?;
716    let (borrow, t1) = match arg.unwrapped().kind() {
717        flat::TypeKind::Ref { mutable, inner, .. } => {
718            (RebuildBorrow::Outer { mutable: *mutable }, inner.as_ref())
719        }
720        _ => (RebuildBorrow::None, arg),
721    };
722    buildable(t1)?;
723    let mut inner = t1.optional_inner().unwrap_or(t1);
724    // The chain must also be able to rebuild the crossing's inner spelling.
725    buildable(inner)?;
726    let borrow = if matches!(borrow, RebuildBorrow::None) {
727        match inner.kind() {
728            flat::TypeKind::Ref {
729                mutable,
730                inner: target,
731                ..
732            } if arg.erased_wrappers().is_empty() => {
733                inner = target.as_ref();
734                buildable(inner)?;
735                RebuildBorrow::OptionalInner { mutable: *mutable }
736            }
737            _ => borrow,
738        }
739    } else {
740        borrow
741    };
742    // Only wrappers outside the borrow are kept. Inner wrappers are already
743    // part of the registry chain's source policy.
744    Some((
745        RebuildTarget {
746            arg: arg.erased_wrappers(),
747            borrow,
748        },
749        inner,
750    ))
751}

753/// A flattened plan for one struct input parameter. Built once by
754/// [`build_flat_input_plan`] and consumed by all three codegen sites.
755pub(crate) struct FlatInputPlan {
756    pub leaves: Vec<FlatLeaf>,
757    /// Registry-composed source converter over those leaves. Planning verifies
758    /// its presence and leaf arity; cross-artifact and runtime tests verify that
759    /// the ordered leaves have the same meaning on both sides of JNI. There is
760    /// no adapter-side reconstruction path.
761    pub chain: crate::jni::compile::ComposedChain,
762    /// Source identity retained only for the deliberately non-recursive Vec
763    /// push helper, which constructs one simple element literal.
764    pub struct_module: syn::Path,
765    pub struct_ident: syn::Ident,
766    /// Whether any Product child is itself composed. The specialized Vec
767    /// helper only understands a Product of terminal leaves.
768    pub contains_composed_child: bool,
769    /// The layer readings the rebuild has to satisfy — carried rather than
770    /// re-derived at the emission sites, so the descent is stated once.
771    pub target: RebuildTarget,
772}

774// `impl_into_target` lived here: it extracted `S` from an `impl Into<S> + …`
775// spelling for `build_flat_input_plan`'s struct-target peel. It is gone because
776// that peel now takes a reading, and the model REFUSES `impl Trait` that is not
777// the callback form (`UnsupportedTypeReason::DisallowedImplTrait`) — so a
778// parameter spelled `impl Into<S>` never becomes a `TypeRef` and never reached
779// the call. `cargo check` confirmed it dead rather than the reasoning alone.
780// jnigen's actual `impl Into<…>` support is elsewhere: plugin wrapper exts build
781// a converter artifact identity via `Declarations::input_converter_name`,
782// which never consults this.
783// `flat_probe_inner` lived here: it peeled `&` then `Option` off a SPELLING to
784// reach the type an enum probe should ask about. Its last caller now asks
785// `is_kotlin_enum_reading`, whose `enum_probe` peels the same two layers off the
786// model — so `Box<Priority>` probes as `Priority` where this answered about the
787// wrapper (#289).
788
789/// Kotlin literal that fills a leaf slot when its `Option<struct>` parent is
790/// absent (the `present` flag tells Rust to ignore it). `None` for nullable
791/// leaves, which simply ride a JVM `null`. Mirrors
792/// [`primitive_default_for_descriptor`] on the Rust side.
793pub(crate) fn kt_leaf_default(sig: &str, nullable: bool) -> Option<String> {
794    if nullable {
795        return None;
796    }
797    Some(
798        match sig {
799            "Z" => "false",
800            "B" | "S" | "I" => "0",
801            "C" => "'\\u0000'",
802            "J" => "0L",
803            "F" => "0.0f",
804            "D" => "0.0",
805            "Ljava/lang/String;" => "\"\"",
806            other => {
807                // An inert primitive-array slot is an EMPTY array of its own
808                // type, never null — the slot is non-nullable in the factory.
809                if let Some(n) = crate::jni::wire_access::kotlin_array_of_descriptor(other) {
810                    return Some(format!("{n}(0)"));
811                }
812                "null"
813            }
814        }
815        .to_string(),
816    )
817}

843pub(crate) fn wire_kotlin_type(entry: &prebindgen_registry::ConverterImpl<KotlinMeta>) -> String {
844    if let Some(p) = JniPrim::from_wire(&entry.destination) {
845        return p.kotlin_type().to_string();
846    }
847    if let syn::Type::Path(tp) = &entry.destination {
848        if let Some(last) = tp.path.segments.last() {
849            return match last.ident.to_string().as_str() {
850                "JString" => "String".to_string(),
851                "JByteArray" => "ByteArray".to_string(),
852                _ => entry
853                    .metadata
854                    .kotlin_name
855                    .as_ref()
856                    .map(ToString::to_string)
857                    .unwrap_or_else(|| "Any".to_string()),
858            };
859        }
860    }
861    if matches!(entry.destination, syn::Type::Ptr(_)) {
862        "Long".to_string()
863    } else {
864        entry
865            .metadata
866            .kotlin_name
867            .as_ref()
868            .map(ToString::to_string)
869            .unwrap_or_else(|| "Any".to_string())
870    }
871}

873pub(crate) fn build_flat_input_plan(
874    ext: &Declarations,
875    registry: &impl Conversions,
876    param_name: &syn::Ident,
877    arg: &TypeRef,
878) -> Result<Option<FlatInputPlan>, FlatInputError> {
879    // 1. Resolve the struct target through `&` and `Option<…>` — off the model,
880    //    keeping each layer's reading so the rebuild can restore its spelling.
881    let Some((target, inner)) = rebuildable_target(arg) else {
882        return Ok(None);
883    };
884    // `impl Into<S>` is NOT peeled here, and cannot be: the model refuses
885    // `impl Trait` that is not the callback form (`DisallowedImplTrait`), so a
886    // parameter spelled that way never becomes a reading and never reaches this
887    // function.
888    // The name off the classification, not off the last path segment: `Box<S>`
889    // IS `S` here, and taking the spelling apart would answer about the wrapper.
890    let flat::TypeKind::Named { id, .. } = inner.unwrapped().kind() else {
891        return Ok(None);
892    };
893    let Some(name) = id.ident() else {
894        return Ok(None);
895    };
896    // The ELEMENT, not the item it was parsed from: its fields already carry
897    // readings, which is the whole of #289.
898    let Some(st) = registry.flat().struct_type(&name) else {
899        return Ok(None);
900    };
901    // The DECLARATION is keyed by the type, not by the spelling: a
902    // `Box<Payload>` parameter is a `Payload` to Kotlin and must find
903    // `Payload`'s data-class declaration.
904    let key = inner.stripped_key();
905    let Some(cfg) = ext.types.get(&key) else {
906        return Ok(None);
907    };
908    if cfg.special_decl() || cfg.name_spec.is_none() || cfg.jobject_input {
909        return Ok(None);
910    }
911    // Identity / pass-through guard: the resolved param must decode to the
912    // struct itself, not an opaque handle / value projection (`projection`
913    // present) and not a multi-source / non-identity `impl Into<S>` (which
914    // surfaces as `"Any"` Dispatch or a foreign source type). The resolved
915    // param's Kotlin type (compared by short name, since metadata carries the
916    // FQN) must equal the struct's data-class name.
917    let Some(entry) = ext.in_frag(arg) else {
918        return Ok(None);
919    };
920    if entry.metadata.projection.is_some() {
921        return Ok(None);
922    }
923    let dc_short = cfg
924        .name_spec
925        .as_ref()
926        .map(|s| ext.fqn_of(s))
927        .map(|fqn| fqn.rsplit('.').next().unwrap_or(&fqn).to_string())
928        .unwrap_or_else(|| name.to_string());
929    let entry_short = entry
930        .metadata
931        .kotlin_name
932        .as_ref()
933        .and_then(|t| t.simple_name());
934    if entry_short != Some(dc_short.as_str()) {
935        return Ok(None);
936    }
937    drop(entry);
938
939    // 2. The parameters this crossing occupies, as the recipe states them. What
940    //    used to be walked and named here is composed once per crossing, so
941    //    the only thing left for the site to say is which parameter each wire
942    //    hangs off.
943    let Some(wires) = ext.wires_of(arg) else {
944        return Ok(None);
945    };
946    let leaves: Vec<FlatLeaf> = wires.iter().map(|w| FlatLeaf::of(param_name, w)).collect();
947    let chain = ext
948        .composed_chain(arg, prebindgen_registry::recipe::Direction::Construct)
949        .filter(|chain| chain.layout.leaf_count() == leaves.len())
950        .ok_or_else(|| {
951            flat_error(
952                &key,
953                &param_name.to_string(),
954                "the registry-composed layout arity does not match the declared JNI wires",
955            )
956        })?;
957    chain.activate();
958    let contains_composed_child = match &chain.layout {
959        crate::jni::compile::JLayout::Product(parts) => {
960            parts.iter().any(crate::jni::compile::JLayout::is_composed)
961        }
962        _ => true,
963    };
964    Ok(Some(FlatInputPlan {
965        leaves,
966        chain,
967        struct_module: struct_module_path(ext, registry, &st.name),
968        struct_ident: st.name.clone(),
969        contains_composed_child,
970        target,
971    }))
972}

974/// Render native reconstruction for a [`FlatInputPlan`] through its registry
975/// chain. Building the Product/Optional/Choice tree is entirely a registry
976/// responsibility; this site only supplies the JNI wire leaves.
977/// Failures route through `signal_error` and return the function `on_err`
978/// sentinel. Returns the prelude and call argument (`arg` or `&arg`).
979pub(crate) fn render_flat_input_decode(
980    plan: &FlatInputPlan,
981    arg_ident: &syn::Ident,
982    on_err: &TokenStream,
983    emit: &prebindgen_registry::RustWriter,
984) -> (TokenStream, TokenStream) {
985    let leaves: Vec<syn::Ident> = plan
986        .leaves
987        .iter()
988        .map(|leaf| leaf.native_ident.clone())
989        .collect();
990    let intermediate = plan.chain.layout.expression(&leaves);
991    let converter = emit.operation_ident("jni", &plan.chain.operation);
992    let binding = plan.target.mutable_binding().then(|| quote!(mut));
993    let prelude = quote! {
994        let #binding #arg_ident = match #converter(&mut env, #intermediate) {
995            ::core::result::Result::Ok(__value) => __value,
996            ::core::result::Result::Err(__error) => {
997                signal_binding_error(
998                    &mut env,
999                    &__error_sink,
1000                    &__SINK_MID,
1001                    __SINK_FQN,
1002                    __SINK_DESCR,
1003                    &__error.to_string(),
1004                );
1005                return #on_err;
1006            }
1007        };
1008    };
1009    (prelude, plan.target.call_arg(arg_ident))
1010}
