1//! What a frontend states before planning: the whole of what planning reads.
2//!
3//! A frontend turns its user's configuration into one [`Binding`] — the
4//! carriers generated Rust may hold a value in, the representations values
5//! cross as, the rules saying which values take which representation, and the
6//! outputs to expose — and hands it to [`generate`](crate::generate) with its
7//! [`Target`]. The registry plans from the binding and the source model alone:
8//! no target code runs until the plan is complete. So the configuration can be
9//! checked before planning, printed, and diffed.

19/// Which binding issued an id. Every [`Binding`] draws a fresh one, and each id
20/// it issues carries it, so an id handed to another binding is recognised as
21/// foreign instead of naming whatever sits at the same index there.
22#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
23struct Issuer(u64);
24
25impl Issuer {
26    fn fresh() -> Self {
27        use std::sync::atomic::{AtomicU64, Ordering};
28        static NEXT: AtomicU64 = AtomicU64::new(0);
29        Issuer(NEXT.fetch_add(1, Ordering::Relaxed))
30    }
31}

73/// One of a target's few wire types: a kind of Rust type a value may cross
74/// the boundary in, which is what everything that holds a value — an
75/// aggregate's members, a wrapper's parameters — states its acceptance in.
76///
77/// Every target states the same three things about each of its classes, and
78/// the registry reads them the same way for every target.
79pub trait WireClass: Clone + Eq + std::hash::Hash + std::fmt::Debug {
80    /// Every class of the target: what a holder accepting anything accepts.
81    fn all() -> Vec<Self>;
82
83    /// How a refusal names it: `pointer`, in `unsupported.c.member.pointer`.
84    ///
85    /// It is part of a capability code, which a build script may match and a
86    /// report groups skips by, so it is the class's identity in text: unique
87    /// among the classes [`all`](Self::all) returns — [`generate`](crate::generate)
88    /// refuses a target with two classes of one name — and kept stable when the
89    /// target changes, as a capability code is.
90    fn name(&self) -> &'static str;
91
92    /// The Rust type a carrier of this class is. A class naming one exact
93    /// type is written as that type — `i64`, `jni::sys::jlong`. A class whose
94    /// carriers are types the target declares itself writes `_` where the
95    /// declared type's name goes: `_` for a `repr(C)` struct, `*mut _` for a
96    /// pointer to an incomplete one. [`WireType::declared`] fills it in.
97    fn rust(&self) -> syn::Type;
98}

100/// A Rust type generated code may hold a value in at the boundary.
101///
102/// Its Rust type is its class's, so a carrier and its class cannot disagree:
103/// it is built with [`WireType::exact`] for a class naming one type, and with
104/// [`WireType::declared`], given the declared type's name, for one that
105/// leaves the name to the target.
106#[derive(Clone, Debug, PartialEq, Eq, Hash)]
107pub struct WireType<C, M> {
108    rust: syn::Type,
109    /// Which of the adapter's few wire types it is — what a holder's
110    /// [`Accepts`] is stated in.
111    pub class: C,
112    /// For an aggregate, which wire types its members may be; `None` for a
113    /// carrier with no members. A `Product` needs a carrier with members.
114    pub members: Option<Accepts<C>>,
115    /// What the target's writers need to know of it.
116    pub meta: M,
117}
118
119impl<C: WireClass, M> WireType<C, M> {
120    /// A carrier of a class naming one exact Rust type: `i64`, a `jlong`.
121    ///
122    /// Panics if the class leaves a declared type's name to fill in: that is
123    /// the frontend calling the wrong constructor.
124    pub fn exact(class: C, members: Option<Accepts<C>>, meta: M) -> Self {
125        let rust = class.rust();
126        assert!(
127            !holds_placeholder(&rust),
128            "{class:?} carriers are types the target declares; name one with `WireType::declared`"
129        );
130        WireType {
131            rust,
132            class,
133            members,
134            meta,
135        }
136    }
137
138    /// A carrier of a class whose Rust type is one the target declares, named
139    /// `name`: the class's type with `name` in place of `_` — `*mut Ledger`
140    /// for `*mut _`.
141    ///
142    /// Panics if the class names one exact type, which has no name to fill in.
143    pub fn declared(class: C, name: syn::Ident, members: Option<Accepts<C>>, meta: M) -> Self {
144        let mut rust = class.rust();
145        assert!(
146            holds_placeholder(&rust),
147            "{class:?} carriers are one exact type; build one with `WireType::exact`"
148        );
149        Name(name).visit_type_mut(&mut rust);
150        WireType {
151            rust,
152            class,
153            members,
154            meta,
155        }
156    }
157
158    /// The Rust type: `i64`, `*mut ledger_t`, `Stamp`, `JObject<'_>`.
159    pub fn rust(&self) -> &syn::Type {
160        &self.rust
161    }
162}

164/// Whether a class's type leaves a declared type's name to fill in.
165fn holds_placeholder(ty: &syn::Type) -> bool {
166    struct Finds(bool);
167    impl syn::visit_mut::VisitMut for Finds {
168        fn visit_type_infer_mut(&mut self, _: &mut syn::TypeInfer) {
169            self.0 = true;
170        }
171    }
172    let mut finds = Finds(false);
173    finds.visit_type_mut(&mut ty.clone());
174    finds.0
175}
176
177/// Fills a class's `_` with the declared type's name.
178struct Name(syn::Ident);
179
180impl syn::visit_mut::VisitMut for Name {
181    fn visit_type_mut(&mut self, ty: &mut syn::Type) {
182        match ty {
183            syn::Type::Infer(_) => {
184                let name = &self.0;
185                *ty = syn::parse_quote!(#name);
186            }
187            _ => syn::visit_mut::visit_type_mut(self, ty),
188        }
189    }
190}

192/// Which wire types a holder may hold: an aggregate's members, a wrapper's
193/// parameters or its return.
194///
195/// The registry checks each placement once the plan has worked out what is
196/// placed there, and refuses the value it would put anywhere else.
197#[derive(Clone, Debug, PartialEq, Eq, Hash)]
198pub struct Accepts<C> {
199    pub classes: Vec<C>,
200}
201
202impl<C: WireClass> Accepts<C> {
203    pub fn of(classes: impl IntoIterator<Item = C>) -> Self {
204        Accepts {
205            classes: classes.into_iter().collect(),
206        }
207    }
208
209    /// Every class of the target.
210    pub fn any() -> Self {
211        Self::of(C::all())
212    }
213
214    pub fn holds(&self, class: &C) -> bool {
215        self.classes.contains(class)
216    }
217}

219/// How the values a conversion rule covers cross.
220///
221/// It names its carriers and operations and states no type for either: a
222/// codec reads its carrier and produces the source type, a `read` reads the
223/// product's carrier and produces the part's, whatever carrier that part
224/// resolves to. The registry works them out when it plans a value, and feeds
225/// them to the writer.
226// A binding holds a few dozen of these, each once, so the unboxed codecs cost
227// nothing worth an indirection.
228#[allow(clippy::large_enum_variant)]
229#[derive(Clone, Debug, PartialEq, Eq, Hash)]
230pub enum Representation<Op> {
231    /// The whole value, one operation each way: a scalar, a handle, a
232    /// fieldless enum. Each direction has a carrier of its own, because some
233    /// values are received in a form they are not returned in — a C enum
234    /// arrives as `MaybeUninit` of itself.
235    Terminal {
236        /// `None`: the value never crosses into Rust.
237        into_rust: Option<Codec<Op>>,
238        /// `None`: the value never crosses out of Rust.
239        out_of_rust: Option<Codec<Op>>,
240        /// How the foreign side gives a held value back without converting
241        /// it — a handle's typed drop. A type output with a release exports it
242        /// as a wrapper of its own.
243        release: Option<Operation<Op>>,
244    },
245    /// The parts of a relation, carried together in one carrier, into Rust.
246    Product {
247        via: Via,
248        /// The aggregate or object holding the parts. Must have members.
249        carrier: CarrierId,
250        /// One part out of the carrier, applied once per part.
251        read: Operation<Op>,
252    },
253    /// A foreign callable, into Rust as an `impl Fn(..)`: the registry builds
254    /// the closure, whose every call converts the arguments out of Rust and
255    /// hands them to `invoke`.
256    Callback {
257        /// What holds the callable on the foreign side: a C closure struct, a
258        /// JVM object. Its members are the arguments' carriers.
259        carrier: CarrierId,
260        /// Applied once to the carrier, where the callable enters Rust: what
261        /// every call will need, such as a reference the JVM keeps alive. Its
262        /// result is moved into the closure.
263        capture: Operation<Op>,
264        /// Applied on every call to what `capture` produced and the
265        /// arguments' carriers, in order. It produces nothing.
266        invoke: Operation<Op>,
267        /// What a call does when converting an argument or invoking fails:
268        /// the closure returns nothing, so no caller can take the failure.
269        /// Reporting inside a call has no runtime context to use.
270        routes: Vec<FailureRoute<Op>>,
271    },
272    /// A representation the target does not lower, refused by name.
273    Unsupported(Unsupported),
274}

291/// One operation, as the binding states it.
292#[derive(Clone, Debug, PartialEq, Eq, Hash)]
293pub struct Operation<Op> {
294    pub implementation: Implementation<Op>,
295    /// Runtime contexts it needs, by name: `jni.env`. A function form's
296    /// context parameters supply them.
297    pub context: Vec<String>,
298    /// Its failure category and error type, if it can fail.
299    pub failure: Option<Failure>,
300}
301
302impl<Op> Operation<Op> {
303    /// A standard operation, with the failure it has by definition.
304    pub fn standard(op: StandardOp) -> Self {
305        let failure = match &op {
306            StandardOp::FromRaw
307            | StandardOp::EnumIn {
308                invalid: Some(_), ..
309            } => Some(Failure::binding_message()),
310            _ => None,
311        };
312        Operation {
313            implementation: Implementation::Standard(op),
314            context: Vec::new(),
315            failure,
316        }
317    }
318
319    /// One of the target's own operations, infallible until stated
320    /// otherwise.
321    pub fn target(op: Op) -> Self {
322        Operation {
323            implementation: Implementation::Target(op),
324            context: Vec::new(),
325            failure: None,
326        }
327    }
328
329    /// The same, needing the runtime context `name`.
330    pub fn context(mut self, name: impl Into<String>) -> Self {
331        self.context.push(name.into());
332        self
333    }
334
335    /// The same, failing in `category` with an error of type `error`.
336    pub fn fails(mut self, category: FailureCategory, error: syn::Type) -> Self {
337        self.failure = Some(Failure {
338            category,
339            error: Box::new(error),
340        });
341        self
342    }
343}

370/// An operation the registry writes itself, because writing it means naming a
371/// source type, which only the registry can do.
372///
373/// None states a type: where it is used says which. What differs per target —
374/// the carrier an address is cast to, what a value is carried as — is the
375/// carrier's and the arms'.
376#[derive(Clone, Debug, PartialEq, Eq, Hash)]
377pub enum StandardOp {
378    /// The result *is* the operand: the carrier and the source value are one
379    /// Rust value. Renders nothing at all.
380    Identity,
381    /// Read the part's member of the aggregate the carrier is: `arg0.secs`.
382    ReadMember,
383    /// Hand an owned source value to the foreign side as an address:
384    /// `Box::into_raw(Box::new(v)) as <carrier>`. The foreign side owns the
385    /// allocation from here on.
386    IntoRaw,
387    /// Take back what [`StandardOp::IntoRaw`] handed out:
388    /// `*Box::from_raw(v as *mut <source>)`. A null address is a binding
389    /// failure carrying a `String`; the handle is consumed either way.
390    FromRaw,
391    /// Drop what [`StandardOp::IntoRaw`] handed out without converting it. A
392    /// null address releases nothing, as `free(NULL)` does. Produces no value.
393    Release,
394    /// Take a value of a fieldless source enum to what carries it: `match v {
395    /// source::Op::Add => <carried>, … }`, one arm per value.
396    EnumOut { values: Vec<crate::target::EnumArm> },
397    /// The reverse: a carried value back to the source enum.
398    ///
399    /// With `invalid` set the match ends in a default arm that fails with
400    /// that message, formatted with the value — the carrier can hold
401    /// something no value of the enum names. With `bits` set the match is on
402    /// the carrier's bits read as that integer type: C's `MaybeUninit<op_t>`
403    /// holds whatever `int` the caller passed. The carrier must be exactly as
404    /// large as `bits`, and initialized, which only the caller can promise, so
405    /// the registry makes a wrapper that reads one `unsafe`.
406    EnumIn {
407        values: Vec<crate::target::EnumArm>,
408        invalid: Option<String>,
409        bits: Option<Box<syn::Type>>,
410    },
411}

502/// From an output's root to one value inside it.

562/// Everything planning reads from a binding.
563///
564/// The ids it issues are valid only in it. Every method taking an id panics if
565/// the id was issued by another binding: that is a frontend bug, found where
566/// the id is handed over rather than as a wrong plan.
567pub struct Binding<T: Target> {
568    issuer: Issuer,
569    carriers: Vec<CarrierOf<T>>,
570    representations: Vec<RepresentationOf<T>>,
571    rules: Vec<(Scope, ReprId)>,
572    outputs: Vec<(Declaration, OutputFormOf<T>)>,
573}
574
575impl<T: Target> Default for Binding<T> {
576    fn default() -> Self {
577        Binding {
578            issuer: Issuer::fresh(),
579            carriers: Vec::new(),
580            representations: Vec::new(),
581            rules: Vec::new(),
582            outputs: Vec::new(),
583        }
584    }
585}
586
587impl<T: Target> Binding<T> {
588    pub fn new() -> Self {
589        Self::default()
590    }
591
592    /// A carrier generated Rust may use. Declaring an equal carrier again
593    /// returns the same id.
594    pub fn carrier(&mut self, carrier: CarrierOf<T>) -> CarrierId {
595        let index = match self.carriers.iter().position(|known| *known == carrier) {
596            Some(index) => index,
597            None => {
598                self.carriers.push(carrier);
599                self.carriers.len() - 1
600            }
601        };
602        CarrierId {
603            issuer: self.issuer,
604            index,
605        }
606    }
607
608    /// One way a type crosses. Declaring an equal representation again
609    /// returns the same id.
610    ///
611    /// Panics if a carrier it names was issued by another binding.
612    pub fn representation(&mut self, representation: RepresentationOf<T>) -> ReprId {
613        let carriers: Vec<CarrierId> = match &representation {
614            Representation::Terminal {
615                into_rust,
616                out_of_rust,
617                ..
618            } => into_rust
619                .iter()
620                .chain(out_of_rust)
621                .map(|codec| codec.carrier)
622                .collect(),
623            Representation::Product { carrier, .. } | Representation::Callback { carrier, .. } => {
624                vec![*carrier]
625            }
626            Representation::Unsupported(_) => Vec::new(),
627        };
628        for carrier in carriers {
629            self.check(carrier.issuer, "carrier");
630        }
631        let index = match self
632            .representations
633            .iter()
634            .position(|known| *known == representation)
635        {
636            Some(index) => index,
637            None => {
638                self.representations.push(representation);
639                self.representations.len() - 1
640            }
641        };
642        ReprId {
643            issuer: self.issuer,
644            index,
645        }
646    }
647
648    /// The values `scope` covers cross as `representation`.
649    ///
650    /// Panics if `representation`, or the output `scope` names, was issued by
651    /// another binding.
652    pub fn rule(&mut self, scope: Scope, representation: ReprId) {
653        self.check(representation.issuer, "representation");
654        if let Scope::At(output, _) = &scope {
655            self.check(output.issuer, "output");
656        }
657        self.rules.push((scope, representation));
658    }
659
660    /// Expose `declaration` in `form`. The id is how a rule addresses a value
661    /// inside the output.
662    ///
663    /// Panics if the representation a type form names was issued by another
664    /// binding.
665    pub fn output(&mut self, declaration: Declaration, form: OutputFormOf<T>) -> OutputId {
666        if let OutputForm::Type { representation, .. } = &form {
667            self.check(representation.issuer, "representation");
668        }
669        self.outputs.push((declaration, form));
670        self.output_id(self.outputs.len() - 1)
671    }
672
673    /// The carrier `id` names. Panics if another binding issued `id`.
674    pub fn carrier_of(&self, id: CarrierId) -> &CarrierOf<T> {
675        self.check(id.issuer, "carrier");
676        &self.carriers[id.index]
677    }
678
679    /// The representation `id` names. Panics if another binding issued `id`.
680    pub fn representation_of(&self, id: ReprId) -> &RepresentationOf<T> {
681        self.check(id.issuer, "representation");
682        &self.representations[id.index]
683    }
684
685    /// Every rule, in the order the frontend stated them.
686    pub fn rules(&self) -> &[(Scope, ReprId)] {
687        &self.rules
688    }
689
690    /// Every output, in the order the frontend stated them; an output's
691    /// position here is its id's.
692    pub fn outputs(&self) -> &[(Declaration, OutputFormOf<T>)] {
693        &self.outputs
694    }
695
696    /// The declaration and form recorded as one output. Panics if another
697    /// binding issued `id`.
698    pub fn output_of(&self, id: OutputId) -> &(Declaration, OutputFormOf<T>) {
699        self.check(id.issuer, "output");
700        &self.outputs[id.index]
701    }
702
703    /// The form recorded with one output. Panics if another binding issued
704    /// `id`.
705    pub fn form_of(&self, id: OutputId) -> &OutputFormOf<T> {
706        &self.output_of(id).1
707    }
708
709    /// The id of the output at `index` in [`outputs`](Self::outputs).
710    pub(crate) fn output_id(&self, index: usize) -> OutputId {
711        OutputId {
712            issuer: self.issuer,
713            index,
714        }
715    }
716
717    /// An id this binding issued exists in it, since nothing is ever removed:
718    /// the issuer is all there is to check.
719    fn check(&self, issuer: Issuer, kind: &str) {
720        assert!(
721            issuer == self.issuer,
722            "a {kind} id issued by another binding was handed to this one"
723        );
724    }
725}
726
727impl<T: Target> std::fmt::Display for Binding<T> {
728    /// One line per carrier, representation, rule and output, a function
729    /// form's details on the lines under its output, with the target's own
730    /// vocabulary in its `Debug` form. It prints every field planning reads,
731    /// which makes it the diagnostic to diff when two builds of one binding
732    /// generate differently. It is not a proof that two bindings are equal:
733    /// the target's classes, metadata and operations print only as much as
734    /// their `Debug` tells apart.
735    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
736        for (index, carrier) in self.carriers.iter().enumerate() {
737            let members = match &carrier.members {
738                Some(accepts) => format!("members {}", classes(&accepts.classes)),
739                None => "no members".to_string(),
740            };
741            writeln!(
742                f,
743                "carrier  c{index}  {}  {}  {members}  {:?}",
744                tokens(&carrier.rust),
745                carrier.class.name(),
746                carrier.meta
747            )?;
748        }
749        for (index, representation) in self.representations.iter().enumerate() {
750            let described = match representation {
751                Representation::Terminal {
752                    into_rust,
753                    out_of_rust,
754                    release,
755                } => {
756                    let codec = |codec: &Option<Codec<T::Op>>| match codec {
757                        Some(codec) => format!("{} {}", codec.carrier, operation(&codec.operation)),
758                        None => "-".to_string(),
759                    };
760                    format!(
761                        "terminal  in: {}  out: {}{}",
762                        codec(into_rust),
763                        codec(out_of_rust),
764                        match release {
765                            Some(release) => format!("  release: {}", operation(release)),
766                            None => String::new(),
767                        }
768                    )
769                }
770                Representation::Product { via, carrier, read } => {
771                    format!("product  {carrier}  {via:?}  read: {}", operation(read))
772                }
773                Representation::Callback {
774                    carrier,
775                    capture,
776                    invoke,
777                    routes,
778                } => format!(
779                    "callback  {carrier}  capture: {}  invoke: {}  routes: [{}]",
780                    operation(capture),
781                    operation(invoke),
782                    routes.iter().map(route).collect::<Vec<_>>().join("; ")
783                ),
784                Representation::Unsupported(reason) => {
785                    format!("unsupported  {}: {}", reason.capability, reason.explanation)
786                }
787            };
788            writeln!(f, "repr     r{index}  {described}")?;
789        }
790        for (scope, representation) in &self.rules {
791            let scope = match scope {
792                Scope::Type(key) => format!("type {}", key.as_str()),
793                Scope::At(output, path) => {
794                    format!("at {} {path}", self.outputs[output.index].0)
795                }
796            };
797            writeln!(f, "rule     {scope}  {representation}")?;
798        }
799        for (declaration, form) in &self.outputs {
800            match form {
801                OutputForm::Type {
802                    representation,
803                    release,
804                    meta,
805                } => {
806                    writeln!(f, "output   {declaration}  type {representation}  {meta:?}")?;
807                    if let Some(release) = release {
808                        write_form(f, "release", release)?;
809                    }
810                }
811                OutputForm::Function { form, meta } => {
812                    writeln!(f, "output   {declaration}  function  {meta:?}")?;
813                    write_form(f, "form", form)?;
814                }
815                OutputForm::Unsupported(reason) => writeln!(
816                    f,
817                    "output   {declaration}  unsupported {}: {}",
818                    reason.capability, reason.explanation
819                )?,
820            }
821        }
822        Ok(())
823    }
824}

826/// Classes as a refusal names them: `[i64, pointer]`.
827fn classes<C: WireClass>(classes: &[C]) -> String {
828    let names: Vec<&str> = classes.iter().map(WireClass::name).collect();
829    format!("[{}]", names.join(", "))
830}
831
832fn tokens(item: &impl quote::ToTokens) -> String {
833    item.to_token_stream().to_string()
834}
835
836/// An operation on one line: what it is, the contexts it needs and the failure
837/// it can raise.
838fn operation<Op: std::fmt::Debug>(operation: &Operation<Op>) -> String {
839    let mut text = format!("{:?}", operation.implementation);
840    if !operation.context.is_empty() {
841        text += &format!(" needs {}", operation.context.join(", "));
842    }
843    if let Some(failure) = &operation.failure {
844        text += &format!(
845            " fails {} {}",
846            failure.category.as_str(),
847            tokens(&*failure.error)
848        );
849    }
850    text
851}
852
853fn terminal(terminal: &Terminal) -> String {
854    match terminal {
855        Terminal::Return(value) => format!("return {}", tokens(value)),
856        Terminal::Abort => "abort".to_string(),
857    }
858}
859
860/// A function form, indented under the output it belongs to: its signature on
861/// one line, its acceptance on the next, then one line per route.
862fn write_form<Op: std::fmt::Debug, C: WireClass>(
863    f: &mut std::fmt::Formatter<'_>,
864    label: &str,
865    form: &FunctionForm<Op, C>,
866) -> std::fmt::Result {
867    let context: Vec<String> = form
868        .context
869        .iter()
870        .map(|param| {
871            format!(
872                "{}{}: {}{}",
873                if param.mutable { "mut " } else { "" },
874                param.name,
875                tokens(&param.ty),
876                match &param.supplies {
877                    Some(name) => format!(" supplies {name}"),
878                    None => String::new(),
879                }
880            )
881        })
882        .collect();
883    let inputs: Vec<String> = form.inputs.iter().map(ToString::to_string).collect();
884    let attrs: Vec<String> = form.attrs.iter().map(tokens).collect();
885    writeln!(
886        f,
887        "         {label}  {}extern {:?} {}  context [{}]  inputs [{}]{}",
888        if form.unsafety { "unsafe " } else { "" },
889        form.abi,
890        form.symbol,
891        context.join(", "),
892        inputs.join(", "),
893        match attrs.is_empty() {
894            true => String::new(),
895            false => format!("  attrs [{}]", attrs.join(" ")),
896        }
897    )?;
898    writeln!(
899        f,
900        "         {label}  params {}  ret {}",
901        classes(&form.params.classes),
902        classes(&form.ret.classes)
903    )?;
904    for one in &form.routes {
905        writeln!(f, "         {label}  route {}", route(one))?;
906    }
907    Ok(())
908}
909
910/// A failure route on one line: its category, what reports it, and how it
911/// ends.
912fn route<Op: std::fmt::Debug>(route: &FailureRoute<Op>) -> String {
913    let report = match &route.report {
914        Some(report) => format!(
915            "report {} by {}, if that fails {}, then ",
916            tokens(&report.error),
917            operation(&report.operation),
918            terminal(&route.on_report_failure)
919        ),
920        None => String::new(),
921    };
922    format!(
923        "{}: {report}{}",
924        route.category.as_str(),
925        terminal(&route.terminate)
926    )
927}
