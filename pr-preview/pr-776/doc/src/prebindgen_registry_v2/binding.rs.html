1//! What a frontend states before planning: the whole of what planning reads.
2//!
3//! A frontend turns its user's configuration into one [`Binding`] — the
4//! wire types generated Rust may hold a value in, the representations values
5//! cross as, the rules saying which values take which representation, and the
6//! outputs to expose — and hands it to [`generate`](crate::generate) with its
7//! [`Target`]. The registry plans from the binding and the source model alone:
8//! no target code runs until the plan is complete. So the configuration can be
9//! checked before planning, printed, and diffed.
10

18/// Which binding issued an id. Every [`Binding`] draws a fresh one, and each id
19/// it issues carries it, so an id handed to another binding is recognised as
20/// foreign instead of naming whatever sits at the same index there.
21#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
22struct Issuer(u64);
23
24impl Issuer {
25    fn fresh() -> Self {
26        use std::sync::atomic::{AtomicU64, Ordering};
27        static NEXT: AtomicU64 = AtomicU64::new(0);
28        Issuer(NEXT.fetch_add(1, Ordering::Relaxed))
29    }
30}

72/// What kind of wire type a value crosses in: C's `I64`, `Pointer`,
73/// `Aggregate`; JNI's `Long`, `Int`, `Handle`, `Object`.
74///
75/// A kind carries no data, so what a target can do with a wire type — which
76/// parts it can have, what a wrapper can take or return — is stated per kind
77/// and cannot depend on any one wire type's name or metadata.
78pub trait WireKind: Copy + Eq + std::hash::Hash + std::fmt::Debug + 'static {
79    /// Every kind of the target.
80    const ALL: &'static [Self];
81
82    /// How a refusal names it: `pointer`, in `unsupported.c.member.pointer`.
83    ///
84    /// It is part of a capability code, which a build script may match and a
85    /// report groups skips by, so it is the kind's identity in text: unique
86    /// among [`ALL`](Self::ALL) — [`generate`](crate::generate) refuses a
87    /// target with two kinds of one name — and kept stable when the target
88    /// changes, as a capability code is.
89    fn name(self) -> &'static str;
90
91    /// The kinds a wire type of this kind can have as its parts — a C struct's
92    /// members, a JVM object's properties, a callback's arguments. Empty for a
93    /// kind that has no parts.
94    fn parts(self) -> &'static [Self] {
95        &[]
96    }
97}

99/// The type of a value on the boundary, as both sides see it: its Rust type,
100/// and what the foreign side reads it as. `i64` and a Kotlin `Long`; `*mut
101/// Ledger` and the C handle `Ledger *`; a `JObject` and an `example.Stamp`.
102///
103/// A target's wire types are its own enum, one variant per kind, each holding
104/// the data only that kind needs: C's name for a struct it declares, a Kotlin
105/// class's name. Two wire types of one Rust type differ when the foreign side
106/// reads them differently — a `jlong` holding a number and one holding an
107/// address — so equality is the whole value's.
108pub trait WireType: Clone + Eq + std::hash::Hash + std::fmt::Debug {
109    type Kind: WireKind;
110
111    fn kind(&self) -> Self::Kind;
112
113    /// The Rust type: `i64`, `*mut Ledger`, `Stamp`, `jni::objects::JObject<'_>`.
114    fn rust(&self) -> syn::Type;
115}

117/// The kind of a target's wire types.
118pub type WireKindOf<T> = <<T as Target>::WireType as WireType>::Kind;

120/// How the values a conversion rule covers cross.
121///
122/// It names its wire types and operations and states no type for either: a
123/// codec reads its wire type and produces the source type, a `read` reads the
124/// product's wire type and produces the part's, whatever wire type that part
125/// resolves to. The registry works them out when it plans a value, and feeds
126/// them to the writer.
127// A binding holds a few dozen of these, each once, so the unboxed codecs cost
128// nothing worth an indirection.
129#[allow(clippy::large_enum_variant)]
130#[derive(Clone, Debug, PartialEq, Eq, Hash)]
131pub enum Representation<Op> {
132    /// The whole value, one operation each way: a scalar, a handle, a
133    /// fieldless enum. Each direction has a wire type of its own, because some
134    /// values are received in a form they are not returned in — a C enum
135    /// arrives as `MaybeUninit` of itself.
136    Terminal {
137        /// `None`: the value never crosses into Rust.
138        into_rust: Option<Codec<Op>>,
139        /// `None`: the value never crosses out of Rust.
140        out_of_rust: Option<Handout<Op>>,
141    },
142    /// The parts of a relation, carried together in one wire type, into Rust.
143    Product {
144        via: Via,
145        /// The aggregate or object holding the parts: a wire type of a kind
146        /// that has parts.
147        wire_type: WireTypeId,
148        /// One part out of the wire type, applied once per part.
149        read: Operation<Op>,
150    },
151    /// A foreign callable, into Rust as an `impl Fn(..)`: the registry builds
152    /// the closure, whose every call converts the arguments out of Rust and
153    /// hands them to `invoke`.
154    Callback {
155        /// What holds the callable on the foreign side: a C closure struct, a
156        /// JVM object. Its parts are the arguments' wire types.
157        wire_type: WireTypeId,
158        /// Applied once to the wire type, where the callable enters Rust: what
159        /// every call will need, such as a reference the JVM keeps alive. Its
160        /// result is moved into the closure.
161        capture: Operation<Op>,
162        /// Applied on every call to what `capture` produced and the
163        /// arguments' wire types, in order. It produces nothing.
164        invoke: Operation<Op>,
165        /// What a call does when converting an argument or invoking fails:
166        /// the closure returns nothing, so no caller can take the failure.
167        /// Reporting inside a call has no runtime context to use.
168        routes: Vec<FailureRoute<Op>>,
169    },
170    /// A representation the target does not lower, refused by name.
171    Unsupported(Unsupported),
172}

182/// The out-of-Rust direction of a [`Representation::Terminal`]: the
183/// conversion, and — for a value the foreign side holds and owes back — how it
184/// gives one back unconverted.
185#[derive(Clone, Debug, PartialEq, Eq, Hash)]
186pub struct Handout<Op> {
187    pub codec: Codec<Op>,
188    /// Frees what `codec` handed out, taking it in `codec`'s wire type: a
189    /// handle's typed drop. Its presence is what makes the representation a
190    /// handle; a type output exposing one exports it as a wrapper of its own.
191    pub release: Option<Operation<Op>>,
192}
193
194impl<Op> Handout<Op> {
195    /// A value handed out and owned by the foreign side from then on.
196    pub fn owned(codec: Codec<Op>) -> Self {
197        Handout {
198            codec,
199            release: None,
200        }
201    }
202}

211/// One operation, as the binding states it: which one, and nothing else.
212///
213/// What it needs and what it can raise are facts of the operation itself —
214/// the registry's own table for a standard one, the target's
215/// [`TargetOp`] for one of its own — so no use of an operation restates them
216/// and no two uses can disagree.
217#[derive(Clone, Debug, PartialEq, Eq, Hash)]
218pub enum Operation<Op> {
219    /// One the registry writes itself.
220    Standard(StandardOp),
221    /// One of the target's own, which its writer writes.
222    Target(Op),
223}
224
225impl<Op: TargetOp> Operation<Op> {
226    /// The runtime contexts it needs besides its operand, by name:
227    /// `"jni.env"`. A standard operation needs none.
228    pub fn contexts(&self) -> &'static [&'static str] {
229        match self {
230            Operation::Standard(_) => &[],
231            Operation::Target(op) => op.contexts(),
232        }
233    }
234
235    /// Its failure category and error type, if it can fail.
236    pub fn failure(&self) -> Option<Failure> {
237        match self {
238            Operation::Standard(op) => op.failure(),
239            Operation::Target(op) => op.failure(),
240        }
241    }
242}

244/// An operation only a target's language has — a JVM getter call, a throw —
245/// and the two facts about it the registry plans with. Both are the
246/// operation's own, the same wherever it is applied.
247pub trait TargetOp: Clone + Eq + std::hash::Hash + std::fmt::Debug {
248    /// The runtime contexts it needs besides its operand, by name: `"jni.env"`.
249    /// A function form's context parameters supply them, and nothing supplies
250    /// one inside a callback's call.
251    fn contexts(&self) -> &'static [&'static str] {
252        &[]
253    }
254
255    /// Its failure category and error type, if it can fail.
256    fn failure(&self) -> Option<Failure> {
257        None
258    }
259}

261/// How an operation fails.
262#[derive(Clone, Debug, PartialEq, Eq, Hash)]
263pub struct Failure {
264    pub category: FailureCategory,
265    pub error: Box<syn::Type>,
266}
267
268impl Failure {
269    /// A binding failure carrying a `String` message, which is what every
270    /// standard operation that can fail raises.
271    pub fn binding_message() -> Self {
272        Failure {
273            category: FailureCategory::Binding,
274            error: Box::new(syn::parse_quote!(String)),
275        }
276    }
277}

279/// An operation the registry writes itself, because writing it means naming a
280/// source type, which only the registry can do.
281///
282/// None states a type: where it is used says which. What differs per target —
283/// the wire type an address is cast to, what a value is carried as — is the
284/// wire type's and the arms'.
285#[derive(Clone, Debug, PartialEq, Eq, Hash)]
286pub enum StandardOp {
287    /// The result *is* the operand: the wire type and the source value are one
288    /// Rust value. Renders nothing at all.
289    Identity,
290    /// Read the part's member of the aggregate the wire type is: `arg0.secs`.
291    ReadMember,
292    /// Hand an owned source value to the foreign side as an address:
293    /// `Box::into_raw(Box::new(v)) as <wire type>`. The foreign side owns the
294    /// allocation from here on.
295    IntoRaw,
296    /// Take back what [`StandardOp::IntoRaw`] handed out:
297    /// `*Box::from_raw(v as *mut <source>)`. A null address is a binding
298    /// failure carrying a `String`; the handle is consumed either way.
299    FromRaw,
300    /// Drop what [`StandardOp::IntoRaw`] handed out without converting it. A
301    /// null address releases nothing, as `free(NULL)` does. Produces no value.
302    Release,
303    /// Take a value of a fieldless source enum to what carries it: `match v {
304    /// source::Op::Add => <carried>, … }`, one arm per value.
305    EnumOut { values: Vec<crate::target::EnumArm> },
306    /// The reverse: a carried value back to the source enum.
307    ///
308    /// With `invalid` set the match ends in a default arm that fails with
309    /// that message, formatted with the value — the wire type can hold
310    /// something no value of the enum names. With `bits` set the match is on
311    /// the wire type's bits read as that integer type: C's `MaybeUninit<op_t>`
312    /// holds whatever `int` the caller passed. The wire type must be exactly as
313    /// large as `bits`, and initialized, which only the caller can promise, so
314    /// the registry makes a wrapper that reads one `unsafe`.
315    EnumIn {
316        values: Vec<crate::target::EnumArm>,
317        invalid: Option<String>,
318        bits: Option<Box<syn::Type>>,
319    },
320}
321
322impl StandardOp {
323    /// The failure a standard operation has by definition: taking back a null
324    /// handle, and reading a number no value of an enum has where the enum
325    /// says how to refuse one — each a binding failure carrying a message.
326    pub fn failure(&self) -> Option<Failure> {
327        match self {
328            StandardOp::FromRaw
329            | StandardOp::EnumIn {
330                invalid: Some(_), ..
331            } => Some(Failure::binding_message()),
332            _ => None,
333        }
334    }
335}

337/// How one source function is exported: everything about its wrapper that is
338/// not a planned value.
339#[derive(Clone, Debug, PartialEq, Eq, Hash)]
340pub struct FunctionForm<Op> {
341    /// The `extern` string: `"C"`, `"system"`.
342    pub abi: String,
343    /// The symbol the wrapper is exported under.
344    pub symbol: String,
345    /// Wrapper parameters no source parameter feeds, before those that do:
346    /// the JNI environment, the receiver.
347    pub context: Vec<ContextParam>,
348    /// The names of the wrapper parameters that carry the inputs, in source
349    /// order — the source's own names for a function, one name for a

351    pub inputs: Vec<syn::Ident>,
352    /// One route per failure category the conversions can raise. A category
353    /// with no route makes the export unsupported.
354    pub routes: Vec<FailureRoute<Op>>,
355    /// Attributes beyond `#[no_mangle]`, which the writer owns.
356    pub attrs: Vec<syn::Attribute>,
357    /// Whether the wrapper is an `unsafe fn`.
358    pub unsafety: bool,
359}

393/// What one output is.
394#[derive(Clone, Debug, PartialEq, Eq, Hash)]
395pub enum OutputForm<Op, O> {
396    /// One representation of a type, exposed: its wire types' Rust
397    /// declarations, and its foreign declaration. The representation is also
398    /// the rule at the output's root, so two outputs of one type each plan as
399    /// declared.
400    Type {
401        representation: ReprId,
402        /// The wrapper releasing a handed-out value, for a representation
403        /// that has a release.
404        release: Option<FunctionForm<Op>>,
405        meta: O,
406    },
407    /// A source function, exported through a wrapper of this form.
408    Function { form: FunctionForm<Op>, meta: O },
409    /// A declarator the target does not lower, refused by name.
410    Unsupported(Unsupported),
411}

413/// Which values a conversion rule covers.
414#[derive(Clone, Debug, PartialEq, Eq, Hash)]
415pub enum Scope {

422/// From an output's root to one value inside it.
423#[derive(Clone, Debug, Default, PartialEq, Eq, Hash)]
424pub struct ValuePath(pub Vec<Step>);
425
426impl ValuePath {
427    pub fn root() -> Self {
428        ValuePath(Vec::new())
429    }
430
431    /// This path, one step further.
432    pub fn then(&self, step: Step) -> Self {
433        let mut steps = self.0.clone();
434        steps.push(step);
435        ValuePath(steps)
436    }
437}
438
439impl std::fmt::Display for ValuePath {
440    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
441        let steps: Vec<String> = self.0.iter().map(Step::to_string).collect();
442        write!(f, "{}", steps.join("."))
443    }
444}

446/// One step of a [`ValuePath`], named as the source names it.
447#[derive(Clone, Debug, PartialEq, Eq, Hash)]
448pub enum Step {
449    /// A source function's parameter, by the name the source gives it.
450    Param(String),
451    /// A source function's result.
452    Return,
453    /// A field of the struct the current value is read through: its name, or
454    /// its position for a tuple field.
455    Field(String),
456    /// An argument of the callback the current value is, by position: the
457    /// value Rust hands the foreign callable, out of Rust.
458    Arg(usize),
459}
460
461impl std::fmt::Display for Step {
462    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
463        match self {
464            Step::Param(name) => write!(f, "param {name}"),
465            Step::Return => write!(f, "return"),
466            Step::Field(name) => write!(f, "field {name}"),
467            Step::Arg(index) => write!(f, "arg {index}"),
468        }
469    }
470}

479/// Everything planning reads from a binding.
480///
481/// The ids it issues are valid only in it. Every method taking an id panics if
482/// the id was issued by another binding: that is a frontend bug, found where
483/// the id is handed over rather than as a wrong plan.
484pub struct Binding<T: Target> {
485    issuer: Issuer,
486    wire_types: Vec<T::WireType>,
487    representations: Vec<RepresentationOf<T>>,
488    rules: Vec<(Scope, ReprId)>,
489    outputs: Vec<(Declaration, OutputFormOf<T>)>,
490}
491
492impl<T: Target> Default for Binding<T> {
493    fn default() -> Self {
494        Binding {
495            issuer: Issuer::fresh(),
496            wire_types: Vec::new(),
497            representations: Vec::new(),
498            rules: Vec::new(),
499            outputs: Vec::new(),
500        }
501    }
502}
503
504impl<T: Target> Binding<T> {
505    pub fn new() -> Self {
506        Self::default()
507    }
508
509    /// A wire type generated Rust may use. Declaring an equal wire type again
510    /// returns the same id.
511    pub fn wire_type(&mut self, wire_type: T::WireType) -> WireTypeId {
512        let index = match self.wire_types.iter().position(|known| *known == wire_type) {
513            Some(index) => index,
514            None => {
515                self.wire_types.push(wire_type);
516                self.wire_types.len() - 1
517            }
518        };
519        WireTypeId {
520            issuer: self.issuer,
521            index,
522        }
523    }
524
525    /// One way a type crosses. Declaring an equal representation again
526    /// returns the same id.
527    ///
528    /// Panics if a wire type it names was issued by another binding.
529    pub fn representation(&mut self, representation: RepresentationOf<T>) -> ReprId {
530        let wire_types: Vec<WireTypeId> = match &representation {
531            Representation::Terminal {
532                into_rust,
533                out_of_rust,
534            } => into_rust
535                .iter()
536                .chain(out_of_rust.as_ref().map(|handout| &handout.codec))
537                .map(|codec| codec.wire_type)
538                .collect(),
539            Representation::Product { wire_type, .. }
540            | Representation::Callback { wire_type, .. } => {
541                vec![*wire_type]
542            }
543            Representation::Unsupported(_) => Vec::new(),
544        };
545        for wire_type in wire_types {
546            self.check(wire_type.issuer, "wire type");
547        }
548        let index = match self
549            .representations
550            .iter()
551            .position(|known| *known == representation)
552        {
553            Some(index) => index,
554            None => {
555                self.representations.push(representation);
556                self.representations.len() - 1
557            }
558        };
559        ReprId {
560            issuer: self.issuer,
561            index,
562        }
563    }
564
565    /// The values `scope` covers cross as `representation`.
566    ///
567    /// Panics if `representation`, or the output `scope` names, was issued by
568    /// another binding.
569    pub fn rule(&mut self, scope: Scope, representation: ReprId) {
570        self.check(representation.issuer, "representation");
571        if let Scope::At(output, _) = &scope {
572            self.check(output.issuer, "output");
573        }
574        self.rules.push((scope, representation));
575    }
576
577    /// Expose `declaration` in `form`. The id is how a rule addresses a value
578    /// inside the output.
579    ///
580    /// Panics if the representation a type form names was issued by another
581    /// binding.
582    pub fn output(&mut self, declaration: Declaration, form: OutputFormOf<T>) -> OutputId {
583        if let OutputForm::Type { representation, .. } = &form {
584            self.check(representation.issuer, "representation");
585        }
586        self.outputs.push((declaration, form));
587        self.output_id(self.outputs.len() - 1)
588    }
589
590    /// The wire type `id` names. Panics if another binding issued `id`.
591    pub fn wire_type_of(&self, id: WireTypeId) -> &T::WireType {
592        self.check(id.issuer, "wire type");
593        &self.wire_types[id.index]
594    }
595
596    /// The representation `id` names. Panics if another binding issued `id`.
597    pub fn representation_of(&self, id: ReprId) -> &RepresentationOf<T> {
598        self.check(id.issuer, "representation");
599        &self.representations[id.index]
600    }
601
602    /// Every rule, in the order the frontend stated them.
603    pub fn rules(&self) -> &[(Scope, ReprId)] {
604        &self.rules
605    }
606
607    /// Every output, in the order the frontend stated them; an output's
608    /// position here is its id's.
609    pub fn outputs(&self) -> &[(Declaration, OutputFormOf<T>)] {
610        &self.outputs
611    }
612
613    /// The declaration and form recorded as one output. Panics if another
614    /// binding issued `id`.
615    pub fn output_of(&self, id: OutputId) -> &(Declaration, OutputFormOf<T>) {
616        self.check(id.issuer, "output");
617        &self.outputs[id.index]
618    }
619
620    /// The form recorded with one output. Panics if another binding issued
621    /// `id`.
622    pub fn form_of(&self, id: OutputId) -> &OutputFormOf<T> {
623        &self.output_of(id).1
624    }
625
626    /// The id of the output at `index` in [`outputs`](Self::outputs).
627    pub(crate) fn output_id(&self, index: usize) -> OutputId {
628        OutputId {
629            issuer: self.issuer,
630            index,
631        }
632    }
633
634    /// An id this binding issued exists in it, since nothing is ever removed:
635    /// the issuer is all there is to check.
636    fn check(&self, issuer: Issuer, kind: &str) {
637        assert!(
638            issuer == self.issuer,
639            "a {kind} id issued by another binding was handed to this one"
640        );
641    }
642}

644impl<T: Target> std::fmt::Display for Binding<T> {
645    /// One line per wire type, representation, rule and output, a function
646    /// form's details on the lines under its output, with the target's own
647    /// vocabulary in its `Debug` form. It prints every field planning reads,
648    /// which makes it the diagnostic to diff when two builds of one binding
649    /// generate differently. It is not a proof that two bindings are equal:
650    /// the target's wire types and operations print only as much as their
651    /// `Debug` tells apart.
652    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
653        for (index, wire_type) in self.wire_types.iter().enumerate() {
654            writeln!(
655                f,
656                "wire     w{index}  {}  {}  {:?}",
657                tokens(&wire_type.rust()),
658                wire_type.kind().name(),
659                wire_type
660            )?;
661        }
662        for (index, representation) in self.representations.iter().enumerate() {
663            let described = match representation {
664                Representation::Terminal {
665                    into_rust,
666                    out_of_rust,
667                } => {
668                    let codec = |codec: Option<&Codec<T::Op>>| match codec {
669                        Some(codec) => {
670                            format!("{} {}", codec.wire_type, operation(&codec.operation))
671                        }
672                        None => "-".to_string(),
673                    };
674                    format!(
675                        "terminal  in: {}  out: {}{}",
676                        codec(into_rust.as_ref()),
677                        codec(out_of_rust.as_ref().map(|handout| &handout.codec)),
678                        match out_of_rust
679                            .as_ref()
680                            .and_then(|handout| handout.release.as_ref())
681                        {
682                            Some(release) => format!("  release: {}", operation(release)),
683                            None => String::new(),
684                        }
685                    )
686                }
687                Representation::Product {
688                    via,
689                    wire_type,
690                    read,
691                } => {
692                    format!("product  {wire_type}  {via:?}  read: {}", operation(read))
693                }
694                Representation::Callback {
695                    wire_type,
696                    capture,
697                    invoke,
698                    routes,
699                } => format!(
700                    "callback  {wire_type}  capture: {}  invoke: {}  routes: [{}]",
701                    operation(capture),
702                    operation(invoke),
703                    routes.iter().map(route).collect::<Vec<_>>().join("; ")
704                ),
705                Representation::Unsupported(reason) => {
706                    format!("unsupported  {}: {}", reason.capability, reason.explanation)
707                }
708            };
709            writeln!(f, "repr     r{index}  {described}")?;
710        }
711        for (scope, representation) in &self.rules {
712            let scope = match scope {
713                Scope::Type(key) => format!("type {}", key.as_str()),
714                Scope::At(output, path) => {
715                    format!("at {} {path}", self.outputs[output.index].0)
716                }
717            };
718            writeln!(f, "rule     {scope}  {representation}")?;
719        }
720        for (declaration, form) in &self.outputs {
721            match form {
722                OutputForm::Type {
723                    representation,
724                    release,
725                    meta,
726                } => {
727                    writeln!(f, "output   {declaration}  type {representation}  {meta:?}")?;
728                    if let Some(release) = release {
729                        write_form(f, "release", release)?;
730                    }
731                }
732                OutputForm::Function { form, meta } => {
733                    writeln!(f, "output   {declaration}  function  {meta:?}")?;
734                    write_form(f, "form", form)?;
735                }
736                OutputForm::Unsupported(reason) => writeln!(
737                    f,
738                    "output   {declaration}  unsupported {}: {}",
739                    reason.capability, reason.explanation
740                )?,
741            }
742        }
743        Ok(())
744    }
745}
746
747fn tokens(item: &impl quote::ToTokens) -> String {
748    item.to_token_stream().to_string()
749}
750
751/// An operation on one line: what it is, the contexts it needs and the failure
752/// it can raise.
753fn operation<Op: TargetOp>(operation: &Operation<Op>) -> String {
754    let mut text = format!("{operation:?}");
755    if !operation.contexts().is_empty() {
756        text += &format!(" needs {}", operation.contexts().join(", "));
757    }
758    if let Some(failure) = &operation.failure() {
759        text += &format!(
760            " fails {} {}",
761            failure.category.as_str(),
762            tokens(&*failure.error)
763        );
764    }
765    text
766}
767
768fn terminal(terminal: &Terminal) -> String {
769    match terminal {
770        Terminal::Return(value) => format!("return {}", tokens(value)),
771        Terminal::Abort => "abort".to_string(),
772    }
773}

775/// A function form, indented under the output it belongs to: its signature on
776/// one line, then one line per route.
777fn write_form<Op: TargetOp>(
778    f: &mut std::fmt::Formatter<'_>,
779    label: &str,
780    form: &FunctionForm<Op>,
781) -> std::fmt::Result {
782    let context: Vec<String> = form
783        .context
784        .iter()
785        .map(|param| {
786            format!(
787                "{}{}: {}{}",
788                if param.mutable { "mut " } else { "" },
789                param.name,
790                tokens(&param.ty),
791                match &param.supplies {
792                    Some(name) => format!(" supplies {name}"),
793                    None => String::new(),
794                }
795            )
796        })
797        .collect();
798    let inputs: Vec<String> = form.inputs.iter().map(ToString::to_string).collect();
799    let attrs: Vec<String> = form.attrs.iter().map(tokens).collect();
800    writeln!(
801        f,
802        "         {label}  {}extern {:?} {}  context [{}]  inputs [{}]{}",
803        if form.unsafety { "unsafe " } else { "" },
804        form.abi,
805        form.symbol,
806        context.join(", "),
807        inputs.join(", "),
808        match attrs.is_empty() {
809            true => String::new(),
810            false => format!("  attrs [{}]", attrs.join(" ")),
811        }
812    )?;
813    for one in &form.routes {
814        writeln!(f, "         {label}  route {}", route(one))?;
815    }
816    Ok(())
817}
818
819/// A failure route on one line: its category, what reports it, and how it
820/// ends.
821fn route<Op: TargetOp>(route: &FailureRoute<Op>) -> String {
822    let report = match &route.report {
823        Some(report) => format!(
824            "report {} by {}, if that fails {}, then ",
825            tokens(&report.error),
826            operation(&report.operation),
827            terminal(&route.on_report_failure)
828        ),
829        None => String::new(),
830    };
831    format!(
832        "{}: {report}{}",
833        route.category.as_str(),
834        terminal(&route.terminate)
835    )
836}
