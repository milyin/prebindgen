1//! One finished v2 generation run: what was planned, what was emitted, and
2//! what a missing capability left out.
3//!
4//! The run owns its whole pipeline — parsing, resolution, plans, emission and
5//! the report. It reads the same model and the same declarations v1 does and
6//! shares nothing else with it: no `Registry`, no fragment store, no resolver,
7//! no emitter. That independence is the point of #719, and a dependency edge is
8//! how it would be lost. [`crate::generate`] is the entry point; this is its
9//! result.
10

23/// The engine's name wherever a run identifies itself.
24pub const PIPELINE: &str = "v2";

26/// A finished v2 run: the model and binding it read, what it generated, and
27/// what a missing capability left out.
28///
29/// Immutable. Every writer is a pure emission over it, so they can run in any
30/// order, or not at all.
31pub struct Generation<T: Target> {
32    flat: Flat,
33    /// The target this was generated for, as [`Target::NAME`] spells it —
34    /// stamped into the generated file.
35    target: &'static str,
36    binding: Binding<T>,
37    skipped: Vec<(Declaration, Skip)>,
38    unused_rules: Vec<(Scope, ReprId)>,
39    values: Vec<ValuePlan>,
40    retained: Vec<Retained>,
41    functions: Vec<FunctionPlan<T::Op>>,
42    rust: String,
43}
44
45impl<T: Target> std::fmt::Debug for Generation<T> {
46    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
47        write!(
48            f,
49            "Generation({} target, {} function(s), {} skipped)",
50            self.target,
51            self.functions.len(),
52            self.skipped.len()
53        )
54    }
55}

118        &self.flat

121    /// The `#[cfg]` conditions the captured item behind `declaration` was
122    /// written under, spelled as the source wrote them — empty in the ordinary
123    /// case.
124    ///
125    /// Text rather than tokens, because the registry is what puts a condition
126    /// on the Rust it emits. What is left for a foreign writer is the
127    /// declaration in *its* language, which usually cannot state a condition
128    /// at all; saying so in that declaration's documentation is the most it
129    /// can do.
130    pub fn item_conditions(&self, declaration: &Declaration) -> Vec<String> {
131        declaration
132            .captured(&self.flat)
133            .map(|element| crate::emit::Writer.conditions(Conditioned::Item(element)))
134            .unwrap_or_default()
135            .iter()
136            .map(|condition| prebindgen_flat::close_up(&condition.to_string()))
137            .collect()
138    }
139
140    /// What the binding asked for and did not get, in the order it asked.
141    ///
142    /// One entry per declaration a missing capability stopped, with the
143    /// [`Skip`] naming the capability, a readable sentence, and the path from
144    /// the declaration to the value that could not be lowered. A declaration
145    /// absent from this list was generated.
146    ///
147    /// One entity may be declared more than once, so the same [`Declaration`]
148    /// can appear here beside an emitted one: what the binding declared them
149    /// as is what told them apart, and that is the target's own vocabulary,
150    /// which this engine does not speak.
151    pub fn skipped(&self) -> &[(Declaration, Skip)] {
152        &self.skipped
153    }
154
155    /// The type rules no planned value was covered by, in the order the
156    /// binding recorded them. Not an error: a type rule is a default for many
157    /// values, and a scalar table covers kinds a binding may never mention.
158    pub fn unused_rules(&self) -> &[(Scope, ReprId)] {
159        &self.unused_rules
160    }

162    /// Write the generated Rust file.
163    ///
164    /// Stamped with the pipeline and target that produced it, so a file
165    /// included by mistake from the other engine's output root says so on its
166    /// first line rather than by a link error. The file is always valid Rust:
167    /// a consumer's `include!` compiles whether this run generated anything or
168    /// not.
169    pub fn write_rust(&self, out_path: impl AsRef<Path>) -> Result<PathBuf, EngineError> {
170        // A relative path is resolved against `OUT_DIR`, as v1 resolves one: a
171        // build script that writes `"perftest.rs"` must not land a file in the
172        // crate root under one engine and in `OUT_DIR` under the other.
173        let out_path = out_path.as_ref();
174        let out_path = if out_path.is_relative() {
175            PathBuf::from(std::env::var("OUT_DIR").map_err(|_| {
176                std::io::Error::new(
177                    std::io::ErrorKind::NotFound,
178                    "a relative output path needs OUT_DIR, which only a build script sets",
179                )
180            })?)
181            .join(out_path)
182        } else {
183            out_path.to_path_buf()
184        };
185        if let Some(parent) = out_path.parent() {
186            std::fs::create_dir_all(parent)?;
187        }
188        let contents = format!(
189            "// Generated by prebindgen {PIPELINE} ({} target) — do not edit.\n{}",
190            self.target, self.rust
191        );
192        std::fs::write(&out_path, contents)?;
193        Ok(out_path)
194    }
195}

197/// Check the binding's outputs against the model before anything is planned.
198///
199/// A declaration is a statement of intent, so a declared item the source never
200/// captured is an error and not a capability question — the same rule v1 holds.
201///
202/// Each declaration says which of the three captured kinds its name must find —
203/// which is not always its own kind, since a Kotlin `val` may be backed by a
204/// nullary function. Looking a function up among the constants reported a typo
205/// that was not there; looking it up in the whole namespace let a
206/// `.fun(fun!(x))` naming a captured `const x` through as a capability skip.
207///
208/// An output is the declaration *and* the form recorded with it, so one
209/// entity declared twice is two outputs and is allowed; the same entity
210/// declared twice in the same form is the binding saying one thing twice, and
211/// is refused — the two would be one foreign declaration emitted from two
212/// plans.
213pub(crate) fn check_declarations<K: Clone + Eq + std::hash::Hash>(
214    declared: &[(Declaration, K)],
215    flat: &Flat,
216) -> Result<(), EngineError> {
217    let missing: Vec<_> = declared
218        .iter()
219        .map(|(declaration, _)| declaration)
220        .filter(|declaration| declaration.missing_from(flat))
221        .cloned()
222        .collect();
223    if !missing.is_empty() {
224        return Err(EngineError::DeclaredNotFound { entries: missing });
225    }
226
227    let mut seen = std::collections::HashSet::new();
228    let mut repeated: Vec<_> = declared
229        .iter()
230        .filter(|output| !seen.insert(*output))
231        .map(|(declaration, _)| declaration.clone())
232        .collect();
233    repeated.sort();
234    repeated.dedup();
235    if !repeated.is_empty() {
236        return Err(EngineError::DuplicateDeclaration { entries: repeated });
237    }
238    Ok(())
239}

241/// Whether a struct field was written under a `#[cfg]` the capture reader
242/// could not answer.
243///
244/// A frontend asks while it builds a binding: a target that cannot state a
245/// condition on its own side — Kotlin has none — refuses such a struct rather
246/// than promise a member the library has only sometimes. Reading the
247/// condition itself stays the registry's, which puts it on the Rust it emits.
248pub fn field_is_conditional(field: &prebindgen_flat::flat::Field) -> bool {
249    !crate::emit::Writer
250        .conditions(Conditioned::Field(field))
251        .is_empty()
252}
