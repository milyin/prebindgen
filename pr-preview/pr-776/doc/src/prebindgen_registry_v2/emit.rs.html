1//! The common Rust writer: one component, both targets.
2//!
3//! It owns the shape of the generated Rust — the locals and their order, the
4//! branches on failure, the construction of the source value, the call, the
5//! return — and allocates every temporary from the plan, so two operations
6//! rendered into one wrapper cannot collide over a name. A target contributes
7//! one expression per operation of its own ([`Target::write_operation`]) and
8//! the declarations its wire types need ([`Target::write_wire_type`]); it
9//! contributes no control flow.
10//!
11//! Source types and struct shapes are generated through Flat's emission
12//! capability, never from retained syntax and never spelled by an adapter.
13

41/// How generated code names each entity: the module it is reached through.
42///
43/// A captured item is reached through the source module the frontend
44/// configured for the run. An item the binding defines itself is reached
45/// where the binding said it is — `crate::helpers::x` — which its origin
46/// records as its crate stamp, since for a binding-local item that is what
47/// the stamp means. One value carries both so the writer asks one question.
48pub(crate) struct Reach {
49    /// Name → module, for every entity that is not reached through the
50    /// default. The shape [`RustEmitter::emit_source_type`] takes.
51    modules: HashMap<String, syn::Path>,
52    default: syn::Path,
53}
54
55impl Reach {
56    pub(crate) fn new(flat: &Flat, default: syn::Path) -> Self {
57        let modules = flat
58            .elements()
59            .filter_map(|element| Some((element.name()?, element)))
60            .filter(|(name, _)| flat.is_binding_local(*name))
61            .filter_map(|(name, element)| {
62                let module = element.location().crate_name.as_deref()?;
63                Some((name.to_string(), syn::parse_str(module).ok()?))
64            })
65            .collect();
66        Reach { modules, default }
67    }
68
69    /// The module the entity with this name is reached through.
70    fn of(&self, name: &str) -> &syn::Path {
71        self.modules.get(name).unwrap_or(&self.default)
72    }
73}

75/// The whole generated Rust file, as text.
76#[allow(clippy::too_many_arguments)]
77pub(crate) fn render<T: Target>(
78    flat: &Flat,
79    target: &T,
80    binding: &Binding<T>,
81    reach: &Reach,
82    retained: &[Retained],
83    nodes: &[ValuePlan],
84    primitives: &[Applied<T::Op>],
85    functions: &[FunctionPlan<T::Op>],
86) -> String {
87    // The guards come first, and they are emitted whatever else this run
88    // retained. Each one is a compile-time assertion the capture reader
89    // injected — today, that the source crate's features match the set the
90    // capture was filtered by — and a binding that dropped it would compile
91    // happily against a source crate it disagrees with. It belongs to no
92    // declaration, so nothing in retention decides whether to keep it.
93    let guards = flat.guards().map(|guard| Writer.guard(guard));
94    let wire_types = wire_types(flat, target, binding, retained, nodes);
95    // A helper an operation needs is emitted once, before the wrappers, in
96    // the order the wrappers first need it.
97    let mut helpers: Vec<Artifact> = Vec::new();
98    let wrappers: Vec<TokenStream> = functions
99        .iter()
100        .map(|function| {
101            wrapper(
102                flat,
103                target,
104                binding,
105                reach,
106                primitives,
107                function,
108                &mut helpers,
109            )
110        })
111        .collect();
112    let helpers = helpers.iter().map(|helper| &helper.rust);
113    let tokens = quote! {
114        #(#guards)*
115        #(#wire_types)*
116        #(#helpers)*
117        #(#wrappers)*
118    };
119    match syn::parse2::<syn::File>(tokens.clone()) {
120        Ok(file) => prettyplease::unparse(&file),
121        // Unparseable output is a defect in this writer, and hiding it behind
122        // pretty printing would only move the compiler error somewhere less
123        // useful.
124        Err(_) => tokens.to_string(),
125    }
126}

128/// The declarations the wire types of every retained type need, each once, in
129/// the order the types were declared.
130///
131/// A wire type's declaration exists only where the source item its type names
132/// does — the `repr(C)` mirror of a conditional struct is itself conditional —
133/// so every item the target writes carries that item's conditions.
134fn wire_types<T: Target>(
135    flat: &Flat,
136    target: &T,
137    binding: &Binding<T>,
138    retained: &[Retained],
139    nodes: &[ValuePlan],
140) -> Vec<TokenStream> {
141    let mut written: std::collections::HashSet<WireTypeId> = std::collections::HashSet::new();
142    let mut items = Vec::new();
143    for values in retained {
144        let OutputForm::Type { representation, .. } = binding.form_of(values.output) else {
145            continue;
146        };
147        let conditions = values
148            .declaration
149            .captured(flat)
150            .map(|element| Writer.conditions(Conditioned::Item(element)))
151            .unwrap_or_default();
152        let unit = values
153            .declaration
154            .entity_name()
155            .and_then(|name| flat.unit_enum(&name));
156        let root = values.inputs.first().map(|root| &nodes[root.0]);
157        let used: Vec<(WireTypeId, bool)> = match binding.representation_of(*representation) {
158            Representation::Terminal {
159                into_rust,
160                out_of_rust,
161                ..
162            } => into_rust
163                .iter()
164                .chain(out_of_rust.iter())
165                .map(|codec| (codec.wire_type, false))
166                .collect(),
167            Representation::Product { wire_type, .. }
168            | Representation::Callback { wire_type, .. } => {
169                vec![(*wire_type, true)]
170            }
171            Representation::Unsupported(_) => Vec::new(),
172        };
173        for (wire_type, with_parts) in used {
174            if !written.insert(wire_type) {
175                continue;
176            }
177            let parts = match (with_parts, root) {
178                (true, Some(root)) => root
179                    .relation
180                    .parts()
181                    .iter()
182                    .zip(&root.children)
183                    .map(|(part, child)| (part, binding.wire_type_of(nodes[child.0].wire_type)))
184                    .collect(),
185                _ => Vec::new(),
186            };
187            let feed = WireTypeFeed {
188                wire_type: binding.wire_type_of(wire_type),
189                parts,
190                unit,
191            };
192            for item in target.write_wire_type(&feed) {
193                items.push(quote!(#(#conditions)* #item));
194            }
195        }
196    }
197    items
198}

200/// One exported function.
201fn wrapper<T: Target>(
202    flat: &Flat,
203    target: &T,
204    binding: &Binding<T>,
205    reach: &Reach,
206    primitives: &[Applied<T::Op>],
207    function: &FunctionPlan<T::Op>,
208    helpers: &mut Vec<Artifact>,
209) -> TokenStream {
210    // Parameters are named by the form, because a convention that requires an
211    // environment operand has to say what it is called. Everything else is
212    // named here, and around those names: a temporary that shadowed a live
213    // parameter would compile and read the wrong value.
214    // Reserved by their canonical spelling: `r#v0` and `v0` are one name to
215    // Rust, so reserving the raw form would leave the plain one free to shadow
216    // it.
217    let mut taken: HashSet<String> = function
218        .params
219        .iter()
220        .map(|(_, p)| unraw(&p.name))
221        .collect();
222    let ok_binding = free_name("value", &mut taken);
223    let error_binding = free_name("error", &mut taken);
224    let mut scribe = Scribe {
225        flat,
226        target,
227        binding,
228        reach,
229        primitives,
230        helpers,
231        names: HashMap::new(),
232        taken,
233        next_local: 0,
234        ok_binding,
235        error_binding,
236    };
237    let params: Vec<TokenStream> = function
238        .params
239        .iter()
240        .map(|(id, param)| {
241            scribe.names.insert(*id, param.name.clone());
242            let name = &param.name;
243            let ty = &param.ty;
244            let mutable = param.mutable.then(|| quote!(mut));
245            quote!(#mutable #name: #ty)
246        })
247        .collect();
248
249    // A wrapper names source items — the function it calls, and every struct it
250    // constructs — and compiles only where all of them exist. So it inherits
251    // each one's condition: a `#[cfg]` the capture reader could not answer and
252    // rewrote back onto the item. Rust conjoins repeated `#[cfg]` attributes on
253    // one item, so carrying them side by side settles the wrapper's own
254    // condition without anything here reading what they say. Two items carrying
255    // the same condition state it once, which conjunction makes cosmetic and
256    // review makes worth doing.
257    //
258    // The instructions are where a wrapper spells a source item: its signature
259    // carries wire types only, and the one operation that spells a source type —
260    // a handle taken back or released, cast to `*mut source::Ledger` — is a
261    // standard one the registry writes, so it is found here too.
262    let mut conditions = Vec::new();
263    let mut conditioned: HashSet<String> = HashSet::new();
264    for step in &function.instrs {
265        let named = match &step.instr {
266            Instr::Construct { name, .. } => name.clone(),
267            Instr::Call { function, .. } => function.clone(),
268            Instr::Apply { primitive, .. } => {
269                let applied = &primitives[primitive.0];
270                match &applied.operation {
271                    Operation::Standard(StandardOp::FromRaw | StandardOp::Release) => {
272                        match applied.subject.kind() {
273                            prebindgen_flat::flat::TypeKind::Named { id, .. } => id.name.clone(),
274                            _ => continue,
275                        }
276                    }
277                    _ => continue,
278                }
279            }
280            // What a call converts leaves Rust, and names nothing a condition
281            // is on that the wrapper's call does not name too.
282            Instr::Closure { .. } => continue,
283        };
284        // Planning names only items the model declares.
285        let Some(element) = flat.element(named.as_str()) else {
286            continue;
287        };
288        for condition in Writer.conditions(Conditioned::Item(element)) {
289            if conditioned.insert(condition.to_string()) {
290                conditions.push(condition);
291            }
292        }
293    }
294
295    let frame = Frame {
296        routes: &function.routes,
297        contexts: function
298            .params
299            .iter()
300            .filter_map(|(_, param)| match &param.role {
301                ParamRole::Context(name) => Some((name.clone(), param.name.clone())),
302                _ => None,
303            })
304            .collect(),
305    };
306    let statements = scribe.statements(&function.instrs, &frame);
307
308    let tail = match function.result {
309        Some(result) => {
310            let name = &scribe.names[&result];
311            quote!(#name)
312        }
313        None => quote!(),
314    };
315    let symbol = format_ident!("{}", function.symbol);
316    let abi = &function.abi;
317    let attrs = &function.attrs;
318    // A wire type whose bits are read as an integer holds whatever the caller
319    // put in it. A foreign caller passes a number, which the match checks;
320    // safe Rust could pass storage never initialized, which no check can
321    // look at. So the wrapper is `unsafe`, and says what it is owed.
322    let reads_bits = function.instrs.iter().any(|step| match &step.instr {
323        Instr::Apply { primitive, .. } => matches!(
324            primitives[primitive.0].operation,
325            Operation::Standard(StandardOp::EnumIn { bits: Some(_), .. })
326        ),
327        _ => false,
328    });
329    let unsafety = (function.unsafety || reads_bits).then(|| quote!(unsafe));
330    let safety = reads_bits.then(|| {
331        quote! {
332            /// # Safety
333            ///
334            /// Every argument that carries an enum as its bits must be
335            /// initialized. The wrapper reads those bits as an integer and
336            /// refuses a number no value has, but reading storage that was
337            /// never initialized is undefined behaviour.
338        }
339    });
340    let ret = function
341        .ret
342        .as_ref()
343        .map(|ty| quote!(-> #ty))
344        .unwrap_or_default();
345    quote! {
346        #safety
347        #(#conditions)*
348        #[no_mangle]
349        #(#attrs)*
350        pub #unsafety extern #abi fn #symbol(#(#params),*) #ret {
351            #(#statements)*
352            #tail
353        }
354    }
355}

357/// Where statements are rendered: the routes a failure there takes, and the
358/// names the runtime contexts go by. A wrapper's body is one frame, and the
359/// body of a callback's closure is another, with the callback's routes and no
360/// contexts.
361struct Frame<'a, Op> {
362    routes: &'a [FailureRoute<Op>],
363    contexts: HashMap<String, syn::Ident>,
364}
365
366impl<Op> Frame<'_, Op> {
367    fn context(&self, name: &str) -> syn::Ident {
368        self.contexts
369            .get(name)
370            .cloned()
371            .expect("a missing runtime context is refused before assembly")
372    }
373}

375/// Renders the statements of one wrapper, and allocates every name in it.
376struct Scribe<'a, T: Target> {
377    flat: &'a Flat,
378    target: &'a T,
379    binding: &'a Binding<T>,
380    reach: &'a Reach,
381    primitives: &'a [Applied<T::Op>],
382    helpers: &'a mut Vec<Artifact>,
383    names: HashMap<ValueId, syn::Ident>,
384    taken: HashSet<String>,
385    /// Temporaries are numbered by definition order, which is why the same
386    /// operation used twice cannot collide with itself.
387    next_local: usize,
388    ok_binding: syn::Ident,
389    error_binding: syn::Ident,
390}
391
392impl<T: Target> Scribe<'_, T> {
393    /// A fresh name for `id`, one nothing in this wrapper uses.
394    fn local(&mut self, id: ValueId) -> syn::Ident {
395        let name = loop {
396            let candidate = format!("v{}", self.next_local);
397            self.next_local += 1;
398            if self.taken.insert(candidate.clone()) {
399                break format_ident!("{candidate}");
400            }
401        };
402        self.names.insert(id, name.clone());
403        name
404    }
405
406    fn statements(&mut self, instrs: &[Stmt], frame: &Frame<'_, T::Op>) -> Vec<TokenStream> {
407        let mut statements = Vec::new();
408        for step in instrs {
409            // What a field's condition holds the statement to: a member
410            // declared only sometimes is read only sometimes, and the
411            // initializer that consumes the read exists only then. The
412            // statement is one `let`, so the attribute goes in front of it.
413            let guarded = &step.conditions;
414            let statement = self.statement(&step.instr, frame);
415            statements.push(quote!(#(#guarded)* #statement));
416        }
417        statements
418    }
419
420    fn statement(&mut self, instr: &Instr, frame: &Frame<'_, T::Op>) -> TokenStream {
421        match instr {
422            Instr::Apply {
423                primitive,
424                operands,
425                result,
426            } => {
427                let applied = &self.primitives[primitive.0];
428                let operand_names: Vec<syn::Ident> = operands
429                    .iter()
430                    .map(|operand| match operand {
431                        Operand::Value(id) => self.names[id].clone(),
432                        Operand::Context(name) => frame.context(name),
433                    })
434                    .collect();
435                let expression = operation(
436                    self.target,
437                    self.binding,
438                    self.reach,
439                    applied,
440                    &operand_names,
441                    self.helpers,
442                );
443                let Some(failure) = &applied.failure else {
444                    return match result {
445                        Some(result) => {
446                            let name = self.local(*result);
447                            quote!(let #name = #expression;)
448                        }
449                        None => quote!(#expression;),
450                    };
451                };
452                let error = error_pattern(frame.routes, failure.category, &self.error_binding);
453                let route = failure_arm(
454                    self.target,
455                    frame,
456                    failure.category,
457                    &self.error_binding,
458                    self.helpers,
459                );
460                match result {
461                    Some(result) => {
462                        let name = self.local(*result);
463                        let ok = &self.ok_binding;
464                        quote! {
465                            let #name = match #expression {
466                                Ok(#ok) => #ok,
467                                Err(#error) => { #route }
468                            };
469                        }
470                    }
471                    None => quote! {
472                        if let Err(#error) = #expression { #route }
473                    },
474                }
475            }
476            Instr::Construct {
477                name,
478                parts,
479                result,
480            } => {
481                let item = self
482                    .flat
483                    .struct_type(name.as_str())
484                    .expect("a construction names a struct the model declares");
485                let ident = &item.name;
486                let module = self.reach.of(&ident.to_string());
487                let head = quote!(#module::#ident);
488                // An initializer carries its own field's condition: the value
489                // it names was bound by a statement under the same one, so a
490                // field the source does not have is neither read nor filled in.
491                //
492                // Read from the model here, while the statements got theirs
493                // from `Part::conditions` at plan time. The two agree because
494                // both read the same field.
495                let bound: Vec<TokenStream> = item
496                    .fields
497                    .iter()
498                    .zip(parts)
499                    .map(|(field, value)| {
500                        let conditions = Writer.conditions(Conditioned::Field(field));
501                        let bound = field.bind(&self.names[value]);
502                        quote!(#(#conditions)* #bound)
503                    })
504                    .collect();
505                let value = Writer.shape_struct(item, head, &bound);
506                let name = self.local(*result);
507                quote!(let #name = #value;)
508            }
509            Instr::Call {
510                function: callee,
511                args,
512                result,
513            } => {
514                let module = self.reach.of(callee);
515                let callee = format_ident!("{callee}");
516                let args: Vec<syn::Ident> =
517                    args.iter().map(|value| self.names[value].clone()).collect();
518                let call = quote!(#module::#callee(#(#args),*));
519                // A source function returning nothing produces no value, and a
520                // `let` over it would name one nobody can use.
521                match result {
522                    Some(result) => {
523                        let name = self.local(*result);
524                        quote!(let #name = #call;)
525                    }
526                    None => quote!(#call;),
527                }
528            }
529            Instr::Closure {
530                representation,
531                params,
532                instrs,
533                result,
534                ..
535            } => {
536                let Representation::Callback { routes, .. } =
537                    self.binding.representation_of(*representation)
538                else {
539                    unreachable!("a closure is built for a callback representation")
540                };
541                let params: Vec<TokenStream> = params
542                    .iter()
543                    .map(|(id, ty)| {
544                        let name = self.local(*id);
545                        let ty =
546                            Writer.emit_source_type(ty, &self.reach.modules, &self.reach.default);
547                        quote!(#name: #ty)
548                    })
549                    .collect();
550                // Inside a call no runtime context exists: planning refused
551                // any operation there that asks for one.
552                let inner = Frame {
553                    routes,
554                    contexts: HashMap::new(),
555                };
556                let body = self.statements(instrs, &inner);
557                let name = self.local(*result);
558                quote! {
559                    let #name = move |#(#params),*| {
560                        #(#body)*
561                    };
562                }
563            }
564        }
565    }
566}

568/// Keep `written`'s helpers, each once.
569fn keep(helpers: &mut Vec<Artifact>, written: Written) -> TokenStream {
570    for helper in written.helpers {
571        if !helpers.iter().any(|kept| kept.name == helper.name) {
572            helpers.push(helper);
573        }
574    }
575    written.text
576}

578/// One operation's expression: the registry writes its own, the target writes
579/// its own from what the plan feeds it.
580fn operation<T: Target>(
581    target: &T,
582    binding: &Binding<T>,
583    reach: &Reach,
584    applied: &Applied<T::Op>,
585    operands: &[syn::Ident],
586    helpers: &mut Vec<Artifact>,
587) -> TokenStream {
588    // The handle operations spell a source type, which is what makes them the
589    // registry's: an adapter has no way to, and no business doing it.
590    let source_type = |ty| Writer.emit_source_type(ty, &reach.modules, &reach.default);
591    let wire_rust = |slot: Option<Slot>| match slot {
592        Some(Slot::WireType(wire_type)) => {
593            let ty = binding.wire_type_of(wire_type).rust();
594            quote!(#ty)
595        }
596        _ => source_type(&applied.subject),
597    };
598    let value = &operands[0];
599    match &applied.operation {
600        Operation::Standard(StandardOp::Identity) => quote!(#value),
601        Operation::Standard(StandardOp::ReadMember) => {
602            let member = applied
603                .part
604                .as_ref()
605                .expect("a member read is applied to a part")
606                .member();
607            quote!(#value.#member)
608        }
609        Operation::Standard(StandardOp::IntoRaw) => {
610            let rust = wire_rust(applied.result);
611            quote!(Box::into_raw(Box::new(#value)) as #rust)
612        }
613        Operation::Standard(StandardOp::FromRaw) => {
614            let ty = source_type(&applied.subject);
615            let message = format!("null `{}` handle", applied.subject.key());
616            quote! {
617                ::core::ptr::NonNull::new(#value as *mut #ty)
618                    .map(|handle| unsafe { *Box::from_raw(handle.as_ptr()) })
619                    .ok_or_else(|| String::from(#message))
620            }
621        }
622        Operation::Standard(StandardOp::EnumOut { values }) => {
623            let ty = source_type(&applied.subject);
624            let arms = values.iter().map(|arm| {
625                let pattern = enum_value(&ty, arm);
626                let carried = &arm.carried;
627                quote!(#pattern => #carried)
628            });
629            quote!(match #value { #(#arms),* })
630        }
631        Operation::Standard(StandardOp::EnumIn {
632            values,
633            invalid,
634            bits,
635        }) => {
636            let value = match bits {
637                Some(bits) => quote!(unsafe { ::core::mem::transmute_copy::<_, #bits>(&(#value)) }),
638                None => quote!(#value),
639            };
640            let ty = source_type(&applied.subject);
641            let arms = values.iter().map(|arm| {
642                let carried = &arm.carried;
643                let constructed = enum_value(&ty, arm);
644                match invalid {
645                    Some(_) => quote!(#carried => ::core::result::Result::Ok(#constructed)),
646                    None => quote!(#carried => #constructed),
647                }
648            });
649            match invalid {
650                Some(message) => quote! {
651                    match #value {
652                        #(#arms,)*
653                        other => ::core::result::Result::Err(::std::format!(#message, other)),
654                    }
655                },
656                None => quote!(match #value { #(#arms),* }),
657            }
658        }
659        Operation::Standard(StandardOp::Release) => {
660            let ty = source_type(&applied.subject);
661            quote! {
662                drop(::core::ptr::NonNull::new(#value as *mut #ty)
663                    .map(|handle| unsafe { Box::from_raw(handle.as_ptr()) }))
664            }
665        }
666        Operation::Target(op) => {
667            let fed = |slot: Slot| match slot {
668                Slot::Source => Fed::Source(&applied.subject),
669                Slot::WireType(wire_type) => Fed::WireType(binding.wire_type_of(wire_type)),
670                Slot::Captured => Fed::Captured,
671            };
672            // After the value and the contexts come the arguments a callback's
673            // `invoke` is handed; its `capture` is told their wire types only.
674            let named = &operands[1 + applied.contexts.len()..];
675            let feed = OperationFeed {
676                value: Some((value.clone(), fed(applied.value))),
677                contexts: applied
678                    .contexts
679                    .iter()
680                    .map(|name| name.to_string())
681                    .zip(operands[1..].iter().cloned())
682                    .collect(),
683                error: None,
684                result: applied.result.map(fed),
685                part: applied.part.as_ref(),
686                args: applied
687                    .args
688                    .iter()
689                    .enumerate()
690                    .map(|(index, wire_type)| {
691                        (named.get(index).cloned(), binding.wire_type_of(*wire_type))
692                    })
693                    .collect(),
694            };
695            keep(helpers, target.write_operation(op, &feed))
696        }
697    }
698}

700/// One value of a fieldless enum, spelled as the source declared it.
701///
702/// A pattern and a constructor are the same text for a fieldless value, so
703/// this serves both sides of the two enum operations. `Add` stays `Add`, and
704/// `Add()` and `Mul {}` keep their delimiters — the model's speller decides,
705/// from the shape it captured.
706fn enum_value(ty: &TokenStream, arm: &crate::target::EnumArm) -> TokenStream {
707    let name = &arm.name;
708    arm.shape.spell_fieldless(quote!(#ty::#name))
709}

711/// The pattern a failure arm binds the error with: the name when a reporter
712/// reads it, `_` when the route only terminates.
713fn error_pattern<Op>(
714    routes: &[FailureRoute<Op>],
715    category: FailureCategory,
716    error: &syn::Ident,
717) -> TokenStream {
718    let reported = routes
719        .iter()
720        .any(|route| route.category == category && route.report.is_some());
721    if reported {
722        quote!(#error)
723    } else {
724        quote!(_)
725    }
726}

728/// What one failure category does at this boundary: report if the form has an
729/// operation for it, then terminate.
730fn failure_arm<T: Target>(
731    target: &T,
732    frame: &Frame<'_, T::Op>,
733    category: FailureCategory,
734    error: &syn::Ident,
735    helpers: &mut Vec<Artifact>,
736) -> TokenStream {
737    let route = frame
738        .routes
739        .iter()
740        .find(|route| route.category == category)
741        .expect("an unrouted failure category is refused before assembly");
742    let report = route.report.as_ref().map(|report| {
743        let contexts: Vec<(String, syn::Ident)> = report
744            .operation
745            .contexts()
746            .iter()
747            .map(|name| (name.to_string(), frame.context(name)))
748            .collect();
749        let expression = match &report.operation {
750            Operation::Target(op) => {
751                let feed = OperationFeed::<T> {
752                    value: None,
753                    contexts,
754                    error: Some(error.clone()),
755                    result: None,
756                    part: None,
757                    args: Vec::new(),
758                };
759                keep(helpers, target.write_operation(op, &feed))
760            }
761            // A standard operation converts a value, and a report has none to
762            // convert; a binding that stated one is refused before assembly.
763            Operation::Standard(_) => {
764                unreachable!("a reporting operation is the target's own")
765            }
766        };
767        let on_failure = terminal(&route.on_report_failure);
768        match report.operation.failure() {
769            None => quote!(#expression;),
770            Some(_) => quote! {
771                if #expression.is_err() { #on_failure }
772            },
773        }
774    });
775    let terminate = terminal(&route.terminate);
776    quote! {
777        #report
778        #terminate
779    }
780}
781
782fn terminal(terminal: &Terminal) -> TokenStream {
783    match terminal {
784        Terminal::Abort => quote!(std::process::abort();),
785        Terminal::Return(expr) => quote!(return #expr;),
786    }
787}

789/// An identifier's canonical spelling: `r#type` and `type` name one binding.
790fn unraw(ident: &syn::Ident) -> String {
791    ident.to_string().trim_start_matches("r#").to_string()
792}

794/// A name nothing in this wrapper already uses.
795fn free_name(preferred: &str, taken: &mut std::collections::HashSet<String>) -> syn::Ident {
796    if taken.insert(preferred.to_string()) {
797        return format_ident!("{preferred}");
798    }
799    for suffix in 1.. {
800        let candidate = format!("{preferred}_{suffix}");
801        if taken.insert(candidate.clone()) {
802            return format_ident!("{candidate}");
803        }
804    }
805    unreachable!("the search above only ends by returning")
806}
