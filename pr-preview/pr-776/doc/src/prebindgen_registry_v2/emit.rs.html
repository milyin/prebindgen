1//! The common Rust writer: one component, both targets.
2//!
3//! It owns the shape of the generated Rust — the locals and their order, the
4//! branches on failure, the construction of the source value, the call, the
5//! return — and allocates every temporary from the plan, so two operations
6//! rendered into one wrapper cannot collide over a name. A target contributes
7//! one expression per operation of its own ([`Target::write_operation`]) and
8//! the declarations its carriers need ([`Target::write_carrier`]); it
9//! contributes no control flow.
10//!
11//! Source types and struct shapes are generated through Flat's emission
12//! capability, never from retained syntax and never spelled by an adapter.
13

40/// How generated code names each entity: the module it is reached through.
41///
42/// A captured item is reached through the source module the frontend
43/// configured for the run. An item the binding defines itself is reached
44/// where the binding said it is — `crate::helpers::x` — which its origin
45/// records as its crate stamp, since for a binding-local item that is what
46/// the stamp means. One value carries both so the writer asks one question.
47pub(crate) struct Reach {
48    /// Name → module, for every entity that is not reached through the
49    /// default. The shape [`RustEmitter::emit_source_type`] takes.
50    modules: HashMap<String, syn::Path>,
51    default: syn::Path,
52}
53
54impl Reach {
55    pub(crate) fn new(flat: &Flat, default: syn::Path) -> Self {
56        let modules = flat
57            .elements()
58            .filter_map(|element| Some((element.name()?, element)))
59            .filter(|(name, _)| flat.is_binding_local(*name))
60            .filter_map(|(name, element)| {
61                let module = element.location().crate_name.as_deref()?;
62                Some((name.to_string(), syn::parse_str(module).ok()?))
63            })
64            .collect();
65        Reach { modules, default }
66    }
67
68    /// The module the entity with this name is reached through.
69    fn of(&self, name: &str) -> &syn::Path {
70        self.modules.get(name).unwrap_or(&self.default)
71    }
72}

74/// The whole generated Rust file, as text.
75#[allow(clippy::too_many_arguments)]
76pub(crate) fn render<T: Target>(
77    flat: &Flat,
78    target: &T,
79    binding: &Binding<T>,
80    reach: &Reach,
81    retained: &[Retained],
82    nodes: &[ValuePlan],
83    primitives: &[Applied<T::Op>],
84    functions: &[FunctionPlan<T::Op>],
85) -> String {
86    // The guards come first, and they are emitted whatever else this run
87    // retained. Each one is a compile-time assertion the capture reader
88    // injected — today, that the source crate's features match the set the
89    // capture was filtered by — and a binding that dropped it would compile
90    // happily against a source crate it disagrees with. It belongs to no
91    // declaration, so nothing in retention decides whether to keep it.
92    let guards = flat.guards().map(|guard| Writer.guard(guard));
93    let carriers = carriers(flat, target, binding, retained, nodes);
94    // A helper an operation needs is emitted once, before the wrappers, in
95    // the order the wrappers first need it.
96    let mut helpers: Vec<Artifact> = Vec::new();
97    let wrappers: Vec<TokenStream> = functions
98        .iter()
99        .map(|function| {
100            wrapper(
101                flat,
102                target,
103                binding,
104                reach,
105                primitives,
106                function,
107                &mut helpers,
108            )
109        })
110        .collect();
111    let helpers = helpers.iter().map(|helper| &helper.rust);
112    let tokens = quote! {
113        #(#guards)*
114        #(#carriers)*
115        #(#helpers)*
116        #(#wrappers)*
117    };
118    match syn::parse2::<syn::File>(tokens.clone()) {
119        Ok(file) => prettyplease::unparse(&file),
120        // Unparseable output is a defect in this writer, and hiding it behind
121        // pretty printing would only move the compiler error somewhere less
122        // useful.
123        Err(_) => tokens.to_string(),
124    }
125}

127/// The declarations the carriers of every retained type need, each once, in
128/// the order the types were declared.
129///
130/// A carrier's declaration exists only where the source item its type names
131/// does — the `repr(C)` mirror of a conditional struct is itself conditional —
132/// so every item the target writes carries that item's conditions.
133fn carriers<T: Target>(
134    flat: &Flat,
135    target: &T,
136    binding: &Binding<T>,
137    retained: &[Retained],
138    nodes: &[ValuePlan],
139) -> Vec<TokenStream> {
140    let mut written: std::collections::HashSet<CarrierId> = std::collections::HashSet::new();
141    let mut items = Vec::new();
142    for values in retained {
143        let OutputForm::Type { representation, .. } = binding.form_of(values.output) else {
144            continue;
145        };
146        let conditions = values
147            .declaration
148            .captured(flat)
149            .map(|element| Writer.conditions(Conditioned::Item(element)))
150            .unwrap_or_default();
151        let unit = values
152            .declaration
153            .entity_name()
154            .and_then(|name| flat.unit_enum(&name));
155        let root = values.inputs.first().map(|root| &nodes[root.0]);
156        let used: Vec<(CarrierId, bool)> = match binding.representation_of(*representation) {
157            Representation::Terminal {
158                into_rust,
159                out_of_rust,
160                ..
161            } => into_rust
162                .iter()
163                .chain(out_of_rust.iter())
164                .map(|codec| (codec.carrier, false))
165                .collect(),
166            Representation::Product { carrier, .. } | Representation::Callback { carrier, .. } => {
167                vec![(*carrier, true)]
168            }
169            Representation::Unsupported(_) => Vec::new(),
170        };
171        for (carrier, with_members) in used {
172            if !written.insert(carrier) {
173                continue;
174            }
175            let members = match (with_members, root) {
176                (true, Some(root)) => root
177                    .relation
178                    .parts()
179                    .iter()
180                    .zip(&root.children)
181                    .map(|(part, child)| (part, binding.carrier_of(nodes[child.0].carrier)))
182                    .collect(),
183                _ => Vec::new(),
184            };
185            let feed = CarrierFeed {
186                carrier: binding.carrier_of(carrier),
187                members,
188                unit,
189            };
190            for item in target.write_carrier(&feed) {
191                items.push(quote!(#(#conditions)* #item));
192            }
193        }
194    }
195    items
196}

198/// One exported function.
199fn wrapper<T: Target>(
200    flat: &Flat,
201    target: &T,
202    binding: &Binding<T>,
203    reach: &Reach,
204    primitives: &[Applied<T::Op>],
205    function: &FunctionPlan<T::Op>,
206    helpers: &mut Vec<Artifact>,
207) -> TokenStream {
208    // Parameters are named by the form, because a convention that requires an
209    // environment operand has to say what it is called. Everything else is
210    // named here, and around those names: a temporary that shadowed a live
211    // parameter would compile and read the wrong value.
212    // Reserved by their canonical spelling: `r#v0` and `v0` are one name to
213    // Rust, so reserving the raw form would leave the plain one free to shadow
214    // it.
215    let mut taken: HashSet<String> = function
216        .params
217        .iter()
218        .map(|(_, p)| unraw(&p.name))
219        .collect();
220    let ok_binding = free_name("value", &mut taken);
221    let error_binding = free_name("error", &mut taken);
222    let mut scribe = Scribe {
223        flat,
224        target,
225        binding,
226        reach,
227        primitives,
228        helpers,
229        names: HashMap::new(),
230        taken,
231        next_local: 0,
232        ok_binding,
233        error_binding,
234    };
235    let params: Vec<TokenStream> = function
236        .params
237        .iter()
238        .map(|(id, param)| {
239            scribe.names.insert(*id, param.name.clone());
240            let name = &param.name;
241            let ty = &param.ty;
242            let mutable = param.mutable.then(|| quote!(mut));
243            quote!(#mutable #name: #ty)
244        })
245        .collect();
246
247    // A wrapper names source items — the function it calls, and every struct it
248    // constructs — and compiles only where all of them exist. So it inherits
249    // each one's condition: a `#[cfg]` the capture reader could not answer and
250    // rewrote back onto the item. Rust conjoins repeated `#[cfg]` attributes on
251    // one item, so carrying them side by side settles the wrapper's own
252    // condition without anything here reading what they say. Two items carrying
253    // the same condition state it once, which conjunction makes cosmetic and
254    // review makes worth doing.
255    //
256    // The instructions are where a wrapper spells a source item: its signature
257    // carries carriers only, and the one operation that spells a source type —
258    // a handle taken back or released, cast to `*mut source::Ledger` — is a
259    // standard one the registry writes, so it is found here too.
260    let mut conditions = Vec::new();
261    let mut conditioned: HashSet<String> = HashSet::new();
262    for step in &function.instrs {
263        let named = match &step.instr {
264            Instr::Construct { name, .. } => name.clone(),
265            Instr::Call { function, .. } => function.clone(),
266            Instr::Apply { primitive, .. } => {
267                let applied = &primitives[primitive.0];
268                match &applied.implementation {
269                    Implementation::Standard(StandardOp::FromRaw | StandardOp::Release) => {
270                        match applied.subject.kind() {
271                            prebindgen_flat::flat::TypeKind::Named { id, .. } => id.name.clone(),
272                            _ => continue,
273                        }
274                    }
275                    _ => continue,
276                }
277            }
278            // What a call converts leaves Rust, and names nothing a condition
279            // is on that the wrapper's call does not name too.
280            Instr::Closure { .. } => continue,
281        };
282        // Planning names only items the model declares.
283        let Some(element) = flat.element(named.as_str()) else {
284            continue;
285        };
286        for condition in Writer.conditions(Conditioned::Item(element)) {
287            if conditioned.insert(condition.to_string()) {
288                conditions.push(condition);
289            }
290        }
291    }
292
293    let frame = Frame {
294        routes: &function.routes,
295        contexts: function
296            .params
297            .iter()
298            .filter_map(|(_, param)| match &param.role {
299                ParamRole::Context(name) => Some((name.clone(), param.name.clone())),
300                _ => None,
301            })
302            .collect(),
303    };
304    let statements = scribe.statements(&function.instrs, &frame);
305
306    let tail = match function.result {
307        Some(result) => {
308            let name = &scribe.names[&result];
309            quote!(#name)
310        }
311        None => quote!(),
312    };
313    let symbol = format_ident!("{}", function.symbol);
314    let abi = &function.abi;
315    let attrs = &function.attrs;
316    // A carrier whose bits are read as an integer holds whatever the caller
317    // put in it. A foreign caller passes a number, which the match checks;
318    // safe Rust could pass storage never initialized, which no check can
319    // look at. So the wrapper is `unsafe`, and says what it is owed.
320    let reads_bits = function.instrs.iter().any(|step| match &step.instr {
321        Instr::Apply { primitive, .. } => matches!(
322            primitives[primitive.0].implementation,
323            Implementation::Standard(StandardOp::EnumIn { bits: Some(_), .. })
324        ),
325        _ => false,
326    });
327    let unsafety = (function.unsafety || reads_bits).then(|| quote!(unsafe));
328    let safety = reads_bits.then(|| {
329        quote! {
330            /// # Safety
331            ///
332            /// Every argument that carries an enum as its bits must be
333            /// initialized. The wrapper reads those bits as an integer and
334            /// refuses a number no value has, but reading storage that was
335            /// never initialized is undefined behaviour.
336        }
337    });
338    let ret = function
339        .ret
340        .as_ref()
341        .map(|ty| quote!(-> #ty))
342        .unwrap_or_default();
343    quote! {
344        #safety
345        #(#conditions)*
346        #[no_mangle]
347        #(#attrs)*
348        pub #unsafety extern #abi fn #symbol(#(#params),*) #ret {
349            #(#statements)*
350            #tail
351        }
352    }
353}

355/// Where statements are rendered: the routes a failure there takes, and the
356/// names the runtime contexts go by. A wrapper's body is one frame, and the
357/// body of a callback's closure is another, with the callback's routes and no
358/// contexts.
359struct Frame<'a, Op> {
360    routes: &'a [FailureRoute<Op>],
361    contexts: HashMap<String, syn::Ident>,
362}
363
364impl<Op> Frame<'_, Op> {
365    fn context(&self, name: &str) -> syn::Ident {
366        self.contexts
367            .get(name)
368            .cloned()
369            .expect("a missing runtime context is refused before assembly")
370    }
371}

373/// Renders the statements of one wrapper, and allocates every name in it.
374struct Scribe<'a, T: Target> {
375    flat: &'a Flat,
376    target: &'a T,
377    binding: &'a Binding<T>,
378    reach: &'a Reach,
379    primitives: &'a [Applied<T::Op>],
380    helpers: &'a mut Vec<Artifact>,
381    names: HashMap<ValueId, syn::Ident>,
382    taken: HashSet<String>,
383    /// Temporaries are numbered by definition order, which is why the same
384    /// operation used twice cannot collide with itself.
385    next_local: usize,
386    ok_binding: syn::Ident,
387    error_binding: syn::Ident,
388}
389
390impl<T: Target> Scribe<'_, T> {
391    /// A fresh name for `id`, one nothing in this wrapper uses.
392    fn local(&mut self, id: ValueId) -> syn::Ident {
393        let name = loop {
394            let candidate = format!("v{}", self.next_local);
395            self.next_local += 1;
396            if self.taken.insert(candidate.clone()) {
397                break format_ident!("{candidate}");
398            }
399        };
400        self.names.insert(id, name.clone());
401        name
402    }
403
404    fn statements(&mut self, instrs: &[Stmt], frame: &Frame<'_, T::Op>) -> Vec<TokenStream> {
405        let mut statements = Vec::new();
406        for step in instrs {
407            // What a field's condition holds the statement to: a member
408            // declared only sometimes is read only sometimes, and the
409            // initializer that consumes the read exists only then. The
410            // statement is one `let`, so the attribute goes in front of it.
411            let guarded = &step.conditions;
412            let statement = self.statement(&step.instr, frame);
413            statements.push(quote!(#(#guarded)* #statement));
414        }
415        statements
416    }
417
418    fn statement(&mut self, instr: &Instr, frame: &Frame<'_, T::Op>) -> TokenStream {
419        match instr {
420            Instr::Apply {
421                primitive,
422                operands,
423                result,
424            } => {
425                let applied = &self.primitives[primitive.0];
426                let operand_names: Vec<syn::Ident> = operands
427                    .iter()
428                    .map(|operand| match operand {
429                        Operand::Value(id) => self.names[id].clone(),
430                        Operand::Context(name) => frame.context(name),
431                    })
432                    .collect();
433                let expression = operation(
434                    self.target,
435                    self.binding,
436                    self.reach,
437                    applied,
438                    &operand_names,
439                    self.helpers,
440                );
441                let Some(failure) = &applied.failure else {
442                    return match result {
443                        Some(result) => {
444                            let name = self.local(*result);
445                            quote!(let #name = #expression;)
446                        }
447                        None => quote!(#expression;),
448                    };
449                };
450                let error = error_pattern(frame.routes, failure.category, &self.error_binding);
451                let route = failure_arm(
452                    self.target,
453                    frame,
454                    failure.category,
455                    &self.error_binding,
456                    self.helpers,
457                );
458                match result {
459                    Some(result) => {
460                        let name = self.local(*result);
461                        let ok = &self.ok_binding;
462                        quote! {
463                            let #name = match #expression {
464                                Ok(#ok) => #ok,
465                                Err(#error) => { #route }
466                            };
467                        }
468                    }
469                    None => quote! {
470                        if let Err(#error) = #expression { #route }
471                    },
472                }
473            }
474            Instr::Construct {
475                name,
476                parts,
477                result,
478            } => {
479                let item = self
480                    .flat
481                    .struct_type(name.as_str())
482                    .expect("a construction names a struct the model declares");
483                let ident = &item.name;
484                let module = self.reach.of(&ident.to_string());
485                let head = quote!(#module::#ident);
486                // An initializer carries its own field's condition: the value
487                // it names was bound by a statement under the same one, so a
488                // field the source does not have is neither read nor filled in.
489                //
490                // Read from the model here, while the statements got theirs
491                // from `Part::conditions` at plan time. The two agree because
492                // both read the same field.
493                let bound: Vec<TokenStream> = item
494                    .fields
495                    .iter()
496                    .zip(parts)
497                    .map(|(field, value)| {
498                        let conditions = Writer.conditions(Conditioned::Field(field));
499                        let bound = field.bind(&self.names[value]);
500                        quote!(#(#conditions)* #bound)
501                    })
502                    .collect();
503                let value = Writer.shape_struct(item, head, &bound);
504                let name = self.local(*result);
505                quote!(let #name = #value;)
506            }
507            Instr::Call {
508                function: callee,
509                args,
510                result,
511            } => {
512                let module = self.reach.of(callee);
513                let callee = format_ident!("{callee}");
514                let args: Vec<syn::Ident> =
515                    args.iter().map(|value| self.names[value].clone()).collect();
516                let call = quote!(#module::#callee(#(#args),*));
517                // A source function returning nothing produces no value, and a
518                // `let` over it would name one nobody can use.
519                match result {
520                    Some(result) => {
521                        let name = self.local(*result);
522                        quote!(let #name = #call;)
523                    }
524                    None => quote!(#call;),
525                }
526            }
527            Instr::Closure {
528                representation,
529                params,
530                instrs,
531                result,
532                ..
533            } => {
534                let Representation::Callback { routes, .. } =
535                    self.binding.representation_of(*representation)
536                else {
537                    unreachable!("a closure is built for a callback representation")
538                };
539                let params: Vec<TokenStream> = params
540                    .iter()
541                    .map(|(id, ty)| {
542                        let name = self.local(*id);
543                        let ty =
544                            Writer.emit_source_type(ty, &self.reach.modules, &self.reach.default);
545                        quote!(#name: #ty)
546                    })
547                    .collect();
548                // Inside a call no runtime context exists: planning refused
549                // any operation there that asks for one.
550                let inner = Frame {
551                    routes,
552                    contexts: HashMap::new(),
553                };
554                let body = self.statements(instrs, &inner);
555                let name = self.local(*result);
556                quote! {
557                    let #name = move |#(#params),*| {
558                        #(#body)*
559                    };
560                }
561            }
562        }
563    }
564}

566/// Keep `written`'s helpers, each once.
567fn keep(helpers: &mut Vec<Artifact>, written: Written) -> TokenStream {
568    for helper in written.helpers {
569        if !helpers.iter().any(|kept| kept.name == helper.name) {
570            helpers.push(helper);
571        }
572    }
573    written.text
574}

576/// One operation's expression: the registry writes its own, the target writes
577/// its own from what the plan feeds it.
578fn operation<T: Target>(
579    target: &T,
580    binding: &Binding<T>,
581    reach: &Reach,
582    applied: &Applied<T::Op>,
583    operands: &[syn::Ident],
584    helpers: &mut Vec<Artifact>,
585) -> TokenStream {
586    // The handle operations spell a source type, which is what makes them the
587    // registry's: an adapter has no way to, and no business doing it.
588    let source_type = |ty| Writer.emit_source_type(ty, &reach.modules, &reach.default);
589    let carrier_type = |slot: Option<Slot>| match slot {
590        Some(Slot::Carrier(carrier)) => {
591            let ty = binding.carrier_of(carrier).rust();
592            quote!(#ty)
593        }
594        _ => source_type(&applied.subject),
595    };
596    let value = &operands[0];
597    match &applied.implementation {
598        Implementation::Standard(StandardOp::Identity) => quote!(#value),
599        Implementation::Standard(StandardOp::ReadMember) => {
600            let member = applied
601                .part
602                .as_ref()
603                .expect("a member read is applied to a part")
604                .member();
605            quote!(#value.#member)
606        }
607        Implementation::Standard(StandardOp::IntoRaw) => {
608            let carrier = carrier_type(applied.result);
609            quote!(Box::into_raw(Box::new(#value)) as #carrier)
610        }
611        Implementation::Standard(StandardOp::FromRaw) => {
612            let ty = source_type(&applied.subject);
613            let message = format!("null `{}` handle", applied.subject.key());
614            quote! {
615                ::core::ptr::NonNull::new(#value as *mut #ty)
616                    .map(|handle| unsafe { *Box::from_raw(handle.as_ptr()) })
617                    .ok_or_else(|| String::from(#message))
618            }
619        }
620        Implementation::Standard(StandardOp::EnumOut { values }) => {
621            let ty = source_type(&applied.subject);
622            let arms = values.iter().map(|arm| {
623                let pattern = enum_value(&ty, arm);
624                let carried = &arm.carried;
625                quote!(#pattern => #carried)
626            });
627            quote!(match #value { #(#arms),* })
628        }
629        Implementation::Standard(StandardOp::EnumIn {
630            values,
631            invalid,
632            bits,
633        }) => {
634            let value = match bits {
635                Some(bits) => quote!(unsafe { ::core::mem::transmute_copy::<_, #bits>(&(#value)) }),
636                None => quote!(#value),
637            };
638            let ty = source_type(&applied.subject);
639            let arms = values.iter().map(|arm| {
640                let carried = &arm.carried;
641                let constructed = enum_value(&ty, arm);
642                match invalid {
643                    Some(_) => quote!(#carried => ::core::result::Result::Ok(#constructed)),
644                    None => quote!(#carried => #constructed),
645                }
646            });
647            match invalid {
648                Some(message) => quote! {
649                    match #value {
650                        #(#arms,)*
651                        other => ::core::result::Result::Err(::std::format!(#message, other)),
652                    }
653                },
654                None => quote!(match #value { #(#arms),* }),
655            }
656        }
657        Implementation::Standard(StandardOp::Release) => {
658            let ty = source_type(&applied.subject);
659            quote! {
660                drop(::core::ptr::NonNull::new(#value as *mut #ty)
661                    .map(|handle| unsafe { Box::from_raw(handle.as_ptr()) }))
662            }
663        }
664        Implementation::Target(op) => {
665            let fed = |slot: Slot| match slot {
666                Slot::Source => Fed::Source(&applied.subject),
667                Slot::Carrier(carrier) => Fed::Carrier(binding.carrier_of(carrier)),
668                Slot::Captured => Fed::Captured,
669            };
670            // After the value and the contexts come the arguments a callback's
671            // `invoke` is handed; its `capture` is told their carriers only.
672            let named = &operands[1 + applied.contexts.len()..];
673            let feed = OperationFeed {
674                value: Some((value.clone(), fed(applied.value))),
675                contexts: applied
676                    .contexts
677                    .iter()
678                    .cloned()
679                    .zip(operands[1..].iter().cloned())
680                    .collect(),
681                error: None,
682                result: applied.result.map(fed),
683                part: applied.part.as_ref(),
684                args: applied
685                    .args
686                    .iter()
687                    .enumerate()
688                    .map(|(index, carrier)| {
689                        (named.get(index).cloned(), binding.carrier_of(*carrier))
690                    })
691                    .collect(),
692            };
693            keep(helpers, target.write_operation(op, &feed))
694        }
695    }
696}

698/// One value of a fieldless enum, spelled as the source declared it.
699///
700/// A pattern and a constructor are the same text for a fieldless value, so
701/// this serves both sides of the two enum operations. `Add` stays `Add`, and
702/// `Add()` and `Mul {}` keep their delimiters — the model's speller decides,
703/// from the shape it captured.
704fn enum_value(ty: &TokenStream, arm: &crate::target::EnumArm) -> TokenStream {
705    let name = &arm.name;
706    arm.shape.spell_fieldless(quote!(#ty::#name))
707}

709/// The pattern a failure arm binds the error with: the name when a reporter
710/// reads it, `_` when the route only terminates.
711fn error_pattern<Op>(
712    routes: &[FailureRoute<Op>],
713    category: FailureCategory,
714    error: &syn::Ident,
715) -> TokenStream {
716    let reported = routes
717        .iter()
718        .any(|route| route.category == category && route.report.is_some());
719    if reported {
720        quote!(#error)
721    } else {
722        quote!(_)
723    }
724}
725
726/// What one failure category does at this boundary: report if the form has an
727/// operation for it, then terminate.
728fn failure_arm<T: Target>(
729    target: &T,
730    frame: &Frame<'_, T::Op>,
731    category: FailureCategory,
732    error: &syn::Ident,
733    helpers: &mut Vec<Artifact>,
734) -> TokenStream {
735    let route = frame
736        .routes
737        .iter()
738        .find(|route| route.category == category)
739        .expect("an unrouted failure category is refused before assembly");
740    let report = route.report.as_ref().map(|report| {
741        let contexts: Vec<(String, syn::Ident)> = report
742            .operation
743            .context
744            .iter()
745            .map(|name| (name.clone(), frame.context(name)))
746            .collect();
747        let expression = match &report.operation.implementation {
748            Implementation::Target(op) => {
749                let feed = OperationFeed::<T> {
750                    value: None,
751                    contexts,
752                    error: Some(error.clone()),
753                    result: None,
754                    part: None,
755                    args: Vec::new(),
756                };
757                keep(helpers, target.write_operation(op, &feed))
758            }
759            // A standard operation converts a value, and a report has none to
760            // convert; a binding that stated one is refused before assembly.
761            Implementation::Standard(_) => {
762                unreachable!("a reporting operation is the target's own")
763            }
764        };
765        let on_failure = terminal(&route.on_report_failure);
766        match report.operation.failure {
767            None => quote!(#expression;),
768            Some(_) => quote! {
769                if #expression.is_err() { #on_failure }
770            },
771        }
772    });
773    let terminate = terminal(&route.terminate);
774    quote! {
775        #report
776        #terminate
777    }
778}
779
780fn terminal(terminal: &Terminal) -> TokenStream {
781    match terminal {
782        Terminal::Abort => quote!(std::process::abort();),
783        Terminal::Return(expr) => quote!(return #expr;),
784    }
785}

787/// An identifier's canonical spelling: `r#type` and `type` name one binding.
788fn unraw(ident: &syn::Ident) -> String {
789    ident.to_string().trim_start_matches("r#").to_string()
790}

792/// A name nothing in this wrapper already uses.
793fn free_name(preferred: &str, taken: &mut std::collections::HashSet<String>) -> syn::Ident {
794    if taken.insert(preferred.to_string()) {
795        return format_ident!("{preferred}");
796    }
797    for suffix in 1.. {
798        let candidate = format!("{preferred}_{suffix}");
799        if taken.insert(candidate.clone()) {
800            return format_ident!("{candidate}");
801        }
802    }
803    unreachable!("the search above only ends by returning")
804}
