1//! Optional expression protocol for stored plans used by the C and JNI adapters.
2//! Scoped requests keep prepared temporaries alive while evaluating a continuation.
3//! Other generators can define entirely different request enums.
4use crate::{InputPlan, OutputPlan, TemplateContext};

6use proc_macro2::TokenStream;
7use quote::quote;

9/// Convert one input, then run `next` while the value and its backing storage live.
10/// Takes must be reported in wire traversal order before later failures/unwinds.
11/// Incorrect access metadata or omitted reporting can make caller/callback cleanup
12/// free an acquired owner twice. Custom conversion authors must uphold this safety
13/// contract; arbitrary emitted Rust cannot be proved correct from another inventory.
14/// Place `next` exactly once as an unlabelled tail block, without a new closure.
15pub struct InputFragment<'a, C> {
16    /// Concrete bindings and ownership reporting for this boundary.
17    pub context: &'a C,
18    /// Caller-selected Rust argument name.
19    pub target: &'a syn::Ident,
20    /// Already-rendered remaining call code.
21    pub next: TokenStream,
22}
23/// Slot requests understood by stored output plans.
24pub enum OutputFragment<'a, C> {
25    /// Prepare output temporaries, then evaluate `next` while they remain alive.
26    Prepare {
27        /// Bindings used to render prepared templates.
28        context: &'a C,
29        /// Caller-selected source-result binding.
30        source: &'a syn::Ident,
31        /// Expression evaluated inside the scope introduced by this request.
32        next: TokenStream,
33    },
34    /// Deliver output, then evaluate `next` with delivery bindings still in scope.
35    Deliver {
36        /// Bindings used to render prepared templates.
37        context: &'a C,
38        /// Caller-selected source-result binding.
39        source: &'a syn::Ident,
40        /// Expression evaluated inside the scope introduced by this request.
41        next: TokenStream,
42    },
43    /// Unit expression committing ownership after successful delivery.
44    Commit {
45        /// Bindings used to render prepared templates.
46        context: &'a C,
47        /// Caller-selected source-result binding.
48        source: &'a syn::Ident,
49    },
50    /// Wrapper's final result expression.
51    Result {
52        /// Bindings used to render prepared templates.
53        context: &'a C,
54        /// Caller-selected source-result binding.
55        source: &'a syn::Ident,
56    },
57}
58fn expression(code: TokenStream) -> Result<TokenStream> {
59    syn::parse2::<syn::Expr>(code.clone())
60        .map_err(|e| format!("conversion did not emit an expression: {e}"))?;
61    Ok(code)
62}

77impl<W: WireKind, C: TemplateContext + 'static, N, B> InputPlan<W, C, N, B> {
78    /// Emit the complete conversion using caller-specialized carrier reads.
79    pub fn emit_code(
80        &self,
81        request: InputFragment<'_, C>,
82        code: &TokenStream,
83    ) -> Result<TokenStream> {
84        let code = crate::emission::bind_target(code, request.target)?;
85        let code = request.context.render(&code)?;
86        let storage = request.context.render(&self.storage)?;
87        let body = continue_with(&code, request.next)?;
88        expression(scope(storage, body)?)
89    }
90}

105impl<W: WireKind, C: TemplateContext + 'static, N, B> OutputPlan<W, C, N, B> {
106    /// Emit the output phases with an explicitly selected delivery protocol.
107    /// Preparation and ownership commit still use this plan's original storage.
108    pub fn emit_with_delivery(
109        &self,
110        mode: OutputFragment<'_, C>,
111        delivery: &crate::Delivery,
112    ) -> Result<TokenStream> {
113        expression(match mode {
114            OutputFragment::Prepare {
115                context,
116                source,
117                next,
118            } => {
119                let code = crate::emission::render_output(context, &self.statements, source)?;
120                scope(code, next)?
121            }
122            OutputFragment::Deliver {
123                context,
124                source,
125                next,
126            } => {
127                let code = crate::emission::render_output(context, &delivery.statements, source)?;
128                scope(code, next)?
129            }
130            OutputFragment::Commit { context, source } => {
131                let code = crate::emission::render_output(context, &self.commit, source)?;
132                quote!({ #code })
133            }
134            OutputFragment::Result { context, source } => {
135                crate::emission::render_output(context, &delivery.result, source)?
136            }
137        })
138    }
139}
140/// Compose complete input conversions, the source call and output delivery.
141/// `result_type` is the wrapper result type inside `Result<_, String>`.
142///
143/// Adapters establish their boundary receipt before entering this expression.
144/// Other generators may compose their own request protocols outside Flat.
145pub fn emit_call<I, O, C: TemplateContext + 'static, M>(
146    function: &Function<I, O, M>,
147    context: &C,
148    source: &syn::Ident,
149    result_type: &syn::Type,
150    parameter_names: &[syn::Ident],
151) -> Result<TokenStream>
152where
153    for<'a> I: Conversion<SlotFragment<'a> = InputFragment<'a, C>>,
154    for<'a> O: Conversion<SlotFragment<'a> = OutputFragment<'a, C>>,
155{
156    let inputs = function.input_conversions()?;
157    let output = function.output_conversion()?;
158    let unit = syn::parse_quote!(());
159    let expected = syn::parse_quote!(::core::result::Result<#result_type, ::std::string::String>);
160    let continuation = crate::template_parameter("continuation");
161    let path = &function.path;
162    let prepare = output.code_fragment_slot(
163        OutputFragment::Prepare {
164            context,
165            source,
166            next: continuation.clone(),
167        },
168        &expected,
169    )?;
170    let deliver = output.code_fragment_slot(
171        OutputFragment::Deliver {
172            context,
173            source,
174            next: continuation,
175        },
176        &expected,
177    )?;
178    let commit = output.code_fragment_slot(OutputFragment::Commit { context, source }, &unit)?;
179    let result =
180        output.code_fragment_slot(OutputFragment::Result { context, source }, result_type)?;
181    let mut used = parameter_names
182        .iter()
183        .map(ToString::to_string)
184        .collect::<std::collections::BTreeSet<_>>();
185    reserve(
186        &quote!(#path #source #prepare #deliver #commit #result),
187        &mut used,
188    );
189    let mut fresh = |base: String| {
190        let mut name = base;
191        while !used.insert(name.clone()) {
192            name.push('_');
193        }
194        syn::Ident::new(&name, proc_macro2::Span::call_site())
195    };
196    let args: Vec<_> = (0..inputs.len())
197        .map(|i| fresh(format!("__prebindgen_argument_{i}")))
198        .collect();
199    let delivered = continue_with(&deliver, quote!({let _: () = #commit; Ok(#result)}))?;
200    let mut body = continue_with(&prepare, delivered)?;
201    body = scope(quote!(let #source=#path(#(#args),*);), body)?;
202    for (input, target) in inputs.iter().zip(&args).rev() {
203        body = input.code_fragment_slot(
204            InputFragment {
205                context,
206                target,
207                next: body,
208            },
209            &expected,
210        )?;
211    }
212    Ok(body)
213}
214/// Replace exactly one continuation without rendering or re-scanning inserted code.
215/// This operation can compose prepared templates before adapter bindings exist.
216pub fn continue_with(code: &TokenStream, next: TokenStream) -> Result<TokenStream> {
217    fn walk(code: &TokenStream, next: &TokenStream, count: &mut usize) -> Result<TokenStream> {
218        let tokens: Vec<_> = code.clone().into_iter().collect();
219        let mut out = TokenStream::new();
220        let mut i = 0;
221        while i < tokens.len() {
222            if matches!(&tokens[i],proc_macro2::TokenTree::Ident(id) if id=="__prebindgen_emit")
223                && matches!(tokens.get(i+1),Some(proc_macro2::TokenTree::Punct(p)) if p.as_char()=='!')
224                && matches!(tokens.get(i+2),Some(proc_macro2::TokenTree::Group(g)) if g.stream().to_string()=="continuation")
225            {
226                *count += 1;
227                out.extend(next.clone());
228                i += 3;
229                continue;
230            }
231            match &tokens[i] {
232                proc_macro2::TokenTree::Group(g) => {
233                    let mut group =
234                        proc_macro2::Group::new(g.delimiter(), walk(&g.stream(), next, count)?);
235                    group.set_span(g.span());
236                    out.extend([proc_macro2::TokenTree::Group(group)]);
237                }
238                token => out.extend([token.clone()]),
239            }
240            i += 1;
241        }
242        Ok(out)
243    }
244    let mut count = 0;
245    let out = walk(code, &next, &mut count)?;
246    if count != 1 {
247        return Err(format!(
248            "input/output conversion must contain exactly one continuation, found {count}"
249        ));
250    }
251    flatten_tail(out)
252}
253/// Place code inside an enclosing backing-storage scope.
254pub(crate) fn enclose(storage: TokenStream, body: TokenStream) -> Result<TokenStream> {
255    scope(storage, body)
256}

258// Scoped requests use a tail continuation. Merge unlabelled tail blocks to keep
259// generated syntax shallow for functions with hundreds of wire parameters.
260// Statements retain their order, and locals still surround the continuation.
261fn scope(prefix: TokenStream, next: TokenStream) -> Result<TokenStream> {
262    flatten_tail(quote!({ #prefix #next }))
263}
264fn flatten_tail(code: TokenStream) -> Result<TokenStream> {
265    let mut expr: syn::Expr = syn::parse2(code).map_err(|e| e.to_string())?;
266    if let syn::Expr::Block(outer) = &mut expr {
267        while let Some(syn::Stmt::Expr(syn::Expr::Block(inner), None)) = outer.block.stmts.last() {
268            if inner.label.is_some() || !inner.attrs.is_empty() {
269                break;
270            }
271            let Some(syn::Stmt::Expr(syn::Expr::Block(inner), None)) = outer.block.stmts.pop()
272            else {
273                unreachable!()
274            };
275            outer.block.stmts.extend(inner.block.stmts);
276        }
277    }
278    Ok(quote!(#expr))
279}
280fn reserve(code: &TokenStream, used: &mut std::collections::BTreeSet<String>) {
281    for token in code.clone() {
282        match token {
283            proc_macro2::TokenTree::Ident(name) => {
284                used.insert(name.to_string());
285            }
286            proc_macro2::TokenTree::Group(group) => reserve(&group.stream(), used),
287            _ => {}
288        }
289    }
290}
