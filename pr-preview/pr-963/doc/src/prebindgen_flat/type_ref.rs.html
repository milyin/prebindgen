1//! Opaque references issued for entries already present in a Flat type table.
2use crate::{same_type, type_key, Flat, Result, Schema, TypeInfo};
3use std::sync::Arc;
4use syn::Type;
5
6/// A reference to an existing entry in [`Flat::types`].
7///
8/// Only [`Flat::type_ref`] creates references. Cloning a reference preserves its
9/// table key; it does not copy the source definition or its metadata. References
10/// remain usable after cloning a Flat or mapping its attachments, provided the
11/// type-table keys are preserved. Removing or renaming an entry makes lookup fail.
12/// These are table references, not globally unique Rust type identities.
13///
14/// ```
15/// use prebindgen_flat::Flat;
16/// let source = syn::parse_file("pub struct Point { pub x: i32 }")?;
17/// let flat = Flat::<()>::parse(source.items.into_iter().map(|item| (item, Default::default())))?;
18/// let point = flat.type_ref(&syn::parse_quote!(Point))?;
19/// assert_eq!(flat.get_type(&point)?.ty, syn::parse_quote!(crate::Point));
20/// # Ok::<(), Box<dyn std::error::Error>>(())
21/// ```
22///
23/// Fields are private; a reference cannot be constructed from a table key.
24///
25/// ```compile_fail
26/// use prebindgen_flat::TypeRef;
27/// let forged = TypeRef { key: "Point".into() };
28/// ```
29///
30/// There is no unchecked string conversion.
31///
32/// ```compile_fail
33/// use prebindgen_flat::TypeRef;
34/// let forged: TypeRef = "Point".into();
35/// ```
36///
37/// Nor is there a default reference.
38///
39/// ```compile_fail
40/// use prebindgen_flat::TypeRef;
41/// let forged = TypeRef::default();
42/// ```
43#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
44pub struct TypeRef {
45    key: Arc<str>,
46}
47
48impl TypeRef {
49    /// Borrow the actual table key, for consumers borrowing [`Flat::types`].
50    /// This permits lookup without exposing a way to construct a reference.
51    pub fn key(&self) -> &str {
52        &self.key
53    }
54}
55
56impl<M: Schema> Flat<M> {
57    /// Issue a reference to an existing canonical type entry.
58    ///
59    /// Declared short names and supported type spellings are resolved first.
60    /// This does not register a type: even built-ins and composites require an
61    /// existing entry, created for example by [`Flat::type_metadata_mut`].
62    pub fn type_ref(&self, ty: &Type) -> Result<TypeRef> {
63        let ty = if matches!(ty, Type::ImplTrait(_)) {
64            self.resolve_parameter_type(ty)?
65        } else {
66            self.resolve_type(ty)?
67        };
68        let (key, _) = self
69            .types
70            .iter()
71            .find(|(_, entry)| same_type(&entry.ty, &ty))
72            .ok_or_else(|| format!("unregistered source type `{}`", type_key(&ty)))?;
73        Ok(TypeRef {
74            key: Arc::from(key.as_str()),
75        })
76    }
77
78    /// Borrow an entry directly by a Flat-issued reference.
79    #[allow(clippy::type_complexity)] // Keep the schema attachment roles explicit.
80    pub fn get_type(
81        &self,
82        reference: &TypeRef,
83    ) -> Result<&TypeInfo<M::Type, M::Field, M::Variant, M::Record, M::Enum, M::PayloadEnum>> {
84        self.types
85            .get(reference.key())
86            .ok_or_else(|| format!("missing source type entry `{}`", reference.key()))
87    }
88
89    /// Mutably borrow an entry directly by a Flat-issued reference.
90    #[allow(clippy::type_complexity)] // Keep the schema attachment roles explicit.
91    pub fn get_type_mut(
92        &mut self,
93        reference: &TypeRef,
94    ) -> Result<&mut TypeInfo<M::Type, M::Field, M::Variant, M::Record, M::Enum, M::PayloadEnum>>
95    {
96        self.types
97            .get_mut(reference.key())
98            .ok_or_else(|| format!("missing source type entry `{}`", reference.key()))
99    }
100}

102#[cfg(test)]
103mod tests {
104    use super::*;
105
106    fn source() -> Flat<()> {
107        let items = syn::parse_file("pub struct Point {pub x:i32}")
108            .unwrap()
109            .items;
110        Flat::parse(items.into_iter().map(|item| (item, Default::default()))).unwrap()
111    }
112
113    #[test]
114    fn references_identify_the_actual_entry_across_clones_and_mutation() {
115        let mut flat = source();
116        let reference = flat.type_ref(&syn::parse_quote!(Point)).unwrap();
117        assert_eq!(
118            reference,
119            flat.type_ref(&syn::parse_quote!(crate::Point)).unwrap()
120        );
121        assert!(std::ptr::eq(
122            flat.get_type(&reference).unwrap(),
123            &flat.types["Point"]
124        ));
125        flat.get_type_mut(&reference).unwrap().definition = crate::TypeDefinition::Opaque;
126        let copied = flat.clone();
127        assert!(matches!(
128            copied.get_type(&reference).unwrap().definition,
129            crate::TypeDefinition::Opaque
130        ));
131    }
132
133    #[test]
134    fn issuance_requires_registration_and_deleted_references_fail_lookup() {
135        let mut flat = source();
136        assert!(flat.type_ref(&syn::parse_quote!(Missing)).is_err());
137        assert!(flat.type_ref(&syn::parse_quote!(u64)).is_err());
138        assert!(flat.type_ref(&syn::parse_quote!(Vec<Point>)).is_err());
139        assert_eq!(flat.types.len(), 1);
140        flat.type_metadata_mut(&syn::parse_quote!(Vec<Point>))
141            .unwrap();
142        let reference = flat.type_ref(&syn::parse_quote!(Vec<Point>)).unwrap();
143        assert_eq!(
144            flat.get_type(&reference).unwrap().ty,
145            syn::parse_quote!(Vec<crate::Point>)
146        );
147        flat.types.remove(reference.key()).unwrap();
148        assert!(flat.get_type(&reference).is_err());
149        assert!(flat.get_type_mut(&reference).is_err());
150    }
