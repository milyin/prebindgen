6/// Prepared typed domain-error payload and its Kotlin handler. This is owned by
7/// a function's Result output converter, never by a parallel type registry.
8#[derive(Clone, Debug)]
9pub struct JniDomainError {
10    pub(crate) handler: String,
11    output: JniOutputConverter,
12    // An explicit on_error mapper precedes the replaceable handler payload.
13    // Its prefix is prepared by the same compiler as ordinary Convert nodes.
14    mapping: Option<(syn::Ident, TokenStream)>,
15}
16impl JniDomainError {
17    /// Ordinary converter producing the arguments of this error's Kotlin handler.
18    pub fn output(&self) -> &JniOutputConverter {
19        &self.output
20    }
21    /// Fully qualified Kotlin error-handler interface.
22    pub fn handler(&self) -> &str {
23        &self.handler
24    }
25    pub(crate) fn replace_output(&mut self, output: JniOutputConverter) -> Result<()> {
26        if output.output_outcomes().source_error.is_some() {
27            return Err("a domain-error payload cannot open another domain-error channel".into());
28        }
29        if group::error_wire_shape(&self.output.output_outcomes().success)?
30            != group::error_wire_shape(&output.output_outcomes().success)?
31        {
32            return Err("domain-error converter differs from the declared error wires".into());
33        }
34        sequence::validate_buffered(&output.output_outcomes().success)?;
35        self.output = output;
36        Ok(())
37    }
38    pub(crate) fn throws(&self, exception: Option<&str>) -> String {
39        exception.map_or_else(String::new, |class| {
40            format!("@kotlin.jvm.Throws({}::class)\n", class.replace('$', "."))
41        })
42    }
43    pub(crate) fn parameter(&self, ret: &str, exception: Option<&str>) -> String {
44        let default = exception.map_or_else(String::new, |class| {
45            format!(
46                " = {} {{ message -> throw {}(message) }}",
47                self.handler,
48                class.replace('$', ".")
49            )
50        });
51        format!("onError: {}<{ret}>{default}", self.handler)
52    }
53    fn prepare(&self, context: &JniContext, source: &syn::Ident) -> Result<TokenStream> {
54        let mut context = context.clone();
55        context.is_domain_error = true;
56        context.return_delivery = if context.return_sink.is_some() {
57            JniReturnDelivery::Capture
58        } else {
59            JniReturnDelivery::Packet
60        };
61        let reported = &context.domain_output;
62        context.output_committed = if context.return_sink.is_some() {
63            quote!(#reported = Some(std::ptr::null_mut());)
64        } else {
65            quote!(#reported = Some(result.as_raw());)
66        };
67        returns::prepare(&self.output, &mut context)?;
68        let (payload, prefix) = if let Some((value, preparation)) = &self.mapping {
69            (
70                value,
71                prebindgen_conversions::render_output(&context, preparation, source)?,
72            )
73        } else {
74            (source, TokenStream::new())
75        };
76        let expected = syn::parse_quote!(::prebindgen_runtime::Result<()>);
77        let commit = self.output.code_fragment_slot(
78            OutputFragment::Commit {
79                context: &context,
80                source: payload,
81            },
82            &syn::parse_quote!(()),
83        )?;
84        let deliver = self.output.code_fragment_slot(
85            OutputFragment::Deliver {
86                context: &context,
87                source: payload,
88                next: quote!({let _:() = #commit; Ok(())}),
89            },
90            &expected,
91        )?;
92        let prepare = self.output.code_fragment_slot(
93            OutputFragment::Prepare {
94                context: &context,
95                source: payload,
96                next: deliver,
97            },
98            &expected,
99        )?;
100        Ok(quote!((|| -> ::prebindgen_runtime::Result<()> {#prefix #prepare})()))
101    }
102}

104#[derive(Clone, Debug)]
105struct ResultOutput {
106    success: JniOutputConverter,
107    error: JniDomainError,
108    success_source: syn::Ident,
109    failure_source: syn::Ident,
110    outcomes: Outcomes<JniOutputWireNode>,
111}
112impl ResultOutput {
113    fn new(
114        success: JniOutputConverter,
115        error: JniDomainError,
116        success_source: syn::Ident,
117        failure_source: syn::Ident,
118    ) -> Result<Self> {
119        if success.output_outcomes().source_error.is_some()
120            || error.output.output_outcomes().source_error.is_some()
121        {
122            return Err("nested Result completion channels require result_as_value".into());
123        }
124        let mut outcomes = success.output_outcomes().clone();
125        outcomes.source_error = Some(error.output.output_outcomes().success.clone());
126        Ok(Self {
127            success,
128            error,
129            success_source,
130            failure_source,
131            outcomes,
132        })
133    }
134}
135impl Conversion for ResultOutput {
136    type Wire = JniWire;
137    type Chain = JniOutputWireChain;
138    type Branch = JniOutputWireBranch;
139    type Annotation = JniOutputWireAnnotation;
140    type Container = JniOutputWireContainer;
141    type SlotFragment<'a> = OutputFragment<'a, JniContext>;
142    fn code_fragment_slot(
143        &self,
144        request: Self::SlotFragment<'_>,
145        expected: &syn::Type,
146    ) -> Result<TokenStream> {
147        let payload = &self.success_source;
148        Ok(match request {
149            OutputFragment::Prepare {
150                context,
151                source,
152                next,
153            } => {
154                let failure = &self.failure_source;
155                let error = self.error.prepare(context, failure)?;
156                let prepared = self.success.code_fragment_slot(
157                    OutputFragment::Prepare {
158                        context,
159                        source: payload,
160                        next,
161                    },
162                    expected,
163                )?;
164                quote!({let #payload = match #source {::core::result::Result::Ok(value)=>value,::core::result::Result::Err(#failure)=>{let _:()=(#error)?;return Err(String::new());}};#prepared})
165            }
166            OutputFragment::Deliver { context, next, .. } => self.success.code_fragment_slot(
167                OutputFragment::Deliver {
168                    context,
169                    source: payload,
170                    next,
171                },
172                expected,
173            )?,
174            OutputFragment::Commit { context, .. } => self.success.code_fragment_slot(
175                OutputFragment::Commit {
176                    context,
177                    source: payload,
178                },
179                expected,
180            )?,
181            OutputFragment::Result { context, .. } => self.success.code_fragment_slot(
182                OutputFragment::Result {
183                    context,
184                    source: payload,
185                },
186                expected,
187            )?,
188        })
189    }
190}
191impl OutputConversion for ResultOutput {

193        &self.outcomes
194    }

229pub(crate) fn compile(
230    cx: &Compiler<'_, JniCodec>,
231    ty: &syn::Type,
232    tree: &JniOutput,
233    value: TokenStream,
234    borrowed: bool,
235    reference_return: bool,
236) -> Result<Option<JniOutputConverter>> {
237    let OutputNode::Branch {
238        operation:
239            JniOutputBranch::Result {
240                handler,
241                preserve_message_mapper,
242            },
243        children,
244    } = tree
245    else {
246        return Ok(None);
247    };
248    if borrowed {
249        return Err("Result error routing requires an owned Result occurrence".into());
250    }
251    let (ok, err) =
252        prebindgen_flat::result_parts(ty).ok_or("Result recipe requires Result<T, E>")?;
253    let [success, error] = children.as_slice() else {
254        return Err("Result completion needs Ok and Err conversions".into());
255    };
256    let success = cx
257        .output_root(ok, success, &(), value.clone(), false, reference_return)
258        .map_err(|e| format!("Result Ok: {e}"))?;
259    let (error, mapping) = (|| {
260        if *preserve_message_mapper {
261            let OutputNode::Unary {
262                operation: JniOutputUnary::Convert { conversion },
263                child,
264            } = error
265            else {
266                return Err("mapped Result error requires its resolved value conversion".into());
267            };
268            if prebindgen_flat::type_key(&conversion.representation) != "String" {
269                return Err("a preserved Result message mapper must produce String".into());
270            }
271            let (local, preparation, _) =
272                cx.prepare_output_conversion(err, conversion, value, false)?;
273            Ok((
274                cx.output_root(
275                    &conversion.representation,
276                    child,
277                    &(),
278                    prebindgen_conversions::source_result_expr(),
279                    false,
280                    true,
281                )?,
282                Some((local, preparation)),
283            ))
284        } else {
285            Ok((cx.output_root(err, error, &(), value, false, true)?, None))
286        }
287    })()
288    .map_err(|e: String| format!("Result Err: {e}"))?;
289    let error = JniDomainError {
290        handler: handler.clone(),
291        output: error,
292        mapping,
293    };
294    Ok(Some(JniOutputConverter::new(ResultOutput::new(
295        success,
296        error,
297        cx.name(),
298        cx.name(),
299    )?)))
300}

306    #[test]
307    fn occurrence_diagnostics_name_result_branches_before_completion() {
308        let flat: JniBindings = Flat::parse(
309            syn::parse_file("pub fn read()->Result<i64,i64>{}")
310                .unwrap()
311                .items
312                .into_iter()
313                .map(|i| (i, Default::default())),
314        )
315        .unwrap();
316        let value = || OutputNode::Leaf(crate::JniOutputLeafOp::Value);
317        for (children, branch) in [
318            (vec![OutputNode::None, value()], "Ok"),
319            (vec![value(), OutputNode::None], "Err"),
320            (vec![OutputNode::None, OutputNode::None], "Ok"),
321        ] {
322            let tree = OutputNode::Branch {
323                operation: JniOutputBranch::Result {
324                    handler: "test.ErrorHandler".into(),
325                    preserve_message_mapper: false,
326                },
327                children,
328            };
329            let error = prebindgen_declarations::compile_output(
330                &flat,
331                &JniCodec::raw(),
332                "read",
333                &tree,
334                &(),
335            )
336            .err()
337            .unwrap();
338            println!("{error}");
339            assert_eq!(
340                error,
341                format!("function `read` result: Result {branch}: None output requires unit")
342            );
343        }
344    }

346    #[test]
347    fn reference_error_mappers_preserve_e_and_shared_e_arguments() {
348        let calls = regex::Regex::new(r"crate::message\((&?)[a-zA-Z0-9_]+\)").unwrap();
349        for error in ["&str", "&i64", "&mut i64"] {
350            for shared in [false, true] {
351                for default in [false, true] {
352                    for success in ["i64", "Vec<i64>"] {
353                        let parameter = if shared {
354                            format!("&{error}")
355                        } else {
356                            error.into()
357                        };
358                        let source = syn::parse_file(&format!(
359                            "pub fn read(lifetime: &str)->Result<{success},{error}>{{}} \
360                         pub fn message(error: {parameter})->String{{}}"
361                        ))
362                        .unwrap();
363                        let mut builder = JniGen::builder()
364                            .items(
365                                source
366                                    .items
367                                    .into_iter()
368                                    .map(|item| (item, Default::default())),
369                            )
370                            .set_package_prefix("test");
371                        let mut function = fun!(read);
372                        if success == "Vec<i64>" {
373                            function = function.callback_return();
374                        }
375                        if default {
376                            builder =
377                                builder.error_mapper(syn::parse_str(error).unwrap(), fun!(message));
378                        } else {
379                            function = function.on_error(fun!(message));
380                        }
381                        let generated = builder.package(package!().fun(function)).build().unwrap();
382                        let arguments: Vec<_> = calls.captures_iter(&generated.rust).collect();
383                        assert_eq!(arguments.len(), if success == "i64" { 1 } else { 2 });
384                        for argument in arguments {
385                            assert_eq!(&argument[1], if shared { "&" } else { "" });
386                        }
387                    }
388                }
389            }
390        }
391    }

394    fn same_handler_name_requires_the_complete_kotlin_signature_to_match() {
395        let source = syn::parse_file(
396            "pub fn first()->Result<i64,i64>{} pub fn second()->Result<i64,String>{}",
397        )
398        .unwrap();
399        let mut flat = JniGen::builder()
400            .items(
401                source
402                    .items
403                    .into_iter()
404                    .map(|item| (item, Default::default())),
405            )
406            .set_package_prefix("test")
407            .package(package!().fun(fun!(first)).fun(fun!(second)))
408            .prepare()
409            .unwrap();
410        let name = flat.function("first").unwrap().metadata[0]
411            .call()
412            .unwrap()
413            .output
414            .domain_error()
415            .unwrap()
416            .handler

418        let output = &mut flat.function_mut("second").unwrap().metadata[0]
