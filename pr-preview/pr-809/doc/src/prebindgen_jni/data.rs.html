6/// A Kotlin data class whose fields are copied across the JNI boundary.
7///
8/// Supports owned public named-field records containing scalars, `String`,
9/// `Vec<u8>`, and other declared data classes. Handles, references, empty records,
10/// and recursive records are rejected. No native allocation is owned by this class.
11///
12/// ```
13/// use prebindgen_jni::{JniGen, data_class, package, fun};
14/// let source = syn::parse_file("pub struct Point { pub x: f64, pub y: f64 } pub fn echo(p: Point) -> Point { p }")?;
15/// let generated = JniGen::builder()
16///     .items(source.items.into_iter().map(|item| (item, Default::default())))
17///     .set_package_prefix("example")
18///     .package(package!().class(data_class!(Point)).fun(fun!(echo)))
19///     .build()?;
20/// assert!(generated.kotlin["example/Point.kt"].contains("data class Point"));
21/// assert!(generated.kotlin["example/Functions.kt"].contains(": example.Point"));
22/// # Ok::<(), Box<dyn std::error::Error>>(())
23/// ```
24#[derive(Clone)]
25pub struct DataClassDecl {
26    /// Source record type, resolved against the selected flat model.
27    pub ty: syn::Type,
28    /// Literal Kotlin short name, or the source type's last path segment.
29    pub name: Option<String>,
30}
31impl DataClassDecl {
32    /// Declare a data class for the source record `ty`.
33    pub fn new(ty: syn::Type) -> Self {
34        Self { ty, name: None }
35    }
36    /// Set a literal Kotlin short class name, validated during build.
37    pub fn name(mut self, name: impl Into<String>) -> Self {
38        self.name = Some(name.into());
39        self
40    }
41}
42/// A class declaration placed in a [`PackageDecl`].
43#[derive(Clone)]
44pub enum ClassDecl {
45    /// An owning native handle.
46    Ptr(PtrClassDecl),
47    /// A value record copied between Kotlin and Rust.
48    Data(DataClassDecl),
49}
50impl From<PtrClassDecl> for ClassDecl {
51    fn from(value: PtrClassDecl) -> Self {
52        Self::Ptr(value)
53    }
54}
55impl From<DataClassDecl> for ClassDecl {
56    fn from(value: DataClassDecl) -> Self {
57        Self::Data(value)
58    }
59}
60impl ClassDecl {
61    pub(crate) fn ty(&self) -> &syn::Type {
62        match self {
63            Self::Ptr(c) => &c.ty,
64            Self::Data(c) => &c.ty,
65        }
66    }
67    pub(crate) fn name(&self) -> &Option<String> {
68        match self {
69            Self::Ptr(c) => &c.name,
70            Self::Data(c) => &c.name,
71        }
72    }
73}

75/// Declare a Kotlin value record from Rust type syntax.
76///
77/// Returns [`DataClassDecl`]. The source type need not be in scope. Invalid type
78/// syntax panics; unsupported record fields are diagnosed during build.
79macro_rules! data_class {($($ty:tt)+)=>{$crate::DataClassDecl::new($crate::__type(stringify!($($ty)+)))};}

87#[derive(Clone, Debug)]
88struct DataField {
89    name: syn::Ident,
90    ty: syn::Type,
91    shape: Shape,
92}
93#[derive(Clone, Debug)]
94enum Shape {
95    Value(Value),
96    Record(Rc<RecordShape>),
97}
98impl Shape {
99    fn kotlin_type(&self) -> String {
100        match self {
101            Self::Value(Value::Scalar(Primitive::I128 | Primitive::U128)) => {
102                "java.math.BigInteger".into()
103            }
104            Self::Value(Value::Scalar(p)) => format!("kotlin.{}", scalar::kotlin(*p)),
105            Self::Value(Value::Text) => "kotlin.String".into(),
106            Self::Value(Value::Bytes) => "kotlin.ByteArray".into(),
107            Self::Record(r) => r.fqn.clone(),
108        }
109    }
110    fn slots(&self) -> usize {
111        match self {
112            Self::Value(Value::Scalar(p)) => scalar::parameter_slots(*p),
113            _ => 1,
114        }
115    }
116}
117/// Private preparation metadata connecting a flattened value wire to a data class.
118///
119/// Constructed by the JNI declaration compiler; wrapper writers see only its
120/// underlying transport through [`JniWire`].
121#[derive(Clone, Debug)]
122pub struct DataWire {
123    pub(crate) leaf: JniKind,
124    pub(crate) root: String,
125    pub(crate) record: Rc<RecordShape>,
126    pub(crate) path: Vec<String>,
127    pub(crate) condition: String,
128}
129impl JniKind {
130    pub(crate) fn leaf(&self) -> &Self {
131        match self {
132            Self::Data(d) => d.leaf.leaf(),
133            _ => self,
134        }
135    }

138pub(crate) struct DataCodec {
139    records: BTreeMap<String, Rc<RecordShape>>,

141impl DataCodec {
142    pub(crate) fn new<I, O>(flat: &Flat<I, O>, classes: &BTreeMap<String, Class>) -> Result<Self> {
143        fn resolve<I, O>(
144            flat: &Flat<I, O>,
145            classes: &BTreeMap<String, Class>,
146            key: &str,
147            stack: &mut Vec<String>,
148            records: &mut BTreeMap<String, Rc<RecordShape>>,
149        ) -> Result<Rc<RecordShape>> {
150            if stack.iter().any(|k| k == key) {
151                return Err(format!("recursive data class `{key}`"));
152            }
153            if let Some(r) = records.get(key) {
154                return Ok(r.clone());
155            }
156            let class = &classes[key];
157            let record = flat
158                .record(&class.ty)
159                .map_err(|_| format!("data class `{key}` requires a named-field record"))?;
160            if record.fields.is_empty() {
161                return Err(format!("data class `{key}` must have at least one field"));
162            }
163            stack.push(key.into());
164            let mut fields = Vec::new();
165            let mut getters = BTreeSet::from(["getClass".to_string()]);
166            for f in &record.fields {
167                let name = f.name.to_string();
168                let name = name.trim_start_matches("r#");
169                validate_name(name)
170                    .map_err(|e| format!("data class `{key}` field `{name}`: {e}"))?;
171                // Distinct Rust names such as `x` and `X` can generate the
172                // same JVM accessor even though Kotlin properties are distinct.
173                let getter = if name.starts_with("is")
174                    && name.len() > 2
175                    && !name.as_bytes()[2].is_ascii_lowercase()
176                {
177                    name.to_string()
178                } else {
179                    format!("get{}{}", name[..1].to_ascii_uppercase(), &name[1..])
180                };
181                if !getters.insert(getter.clone()) {
182                    return Err(format!(
183                        "data class `{key}` field `{name}` has a conflicting JVM getter `{getter}`"
184                    ));
185                }
186                let field_key = type_key(&f.ty);
187                let shape = if matches!(f.ty, syn::Type::Reference(_)) {
188                    return Err(format!(
189                        "data class `{key}` field `{name}` cannot be a reference"
190                    ));
191                } else if let Some(value) = Value::of(&f.ty) {
192                    Shape::Value(value)
193                } else if classes.get(&field_key).is_some_and(|c| c.data) {
194                    Shape::Record(resolve(flat, classes, &field_key, stack, records)?)
195                } else {
196                    return Err(format!("data class `{key}` field `{name}` has unsupported type `{field_key}`; expected a scalar, String, Vec<u8>, or declared data class"));
197                };
198                fields.push(DataField {
199                    name: f.name.clone(),
200                    ty: f.ty.clone(),
201                    shape,
202                });
203            }
204            // Kotlin's generated copy$default also needs a receiver, default masks,
205            // and a marker. Check that method, not just the primary constructor.
206            let slots = fields.iter().map(|f| f.shape.slots()).sum::<usize>()
207                + fields.len().div_ceil(32)
208                + 2;
209            if slots > 255 {
210                return Err(format!("data class `{key}` copy method needs {slots} JVM parameter slots; the limit is 255"));
211            }
212            stack.pop();
213            let shape = Rc::new(RecordShape {
214                ty: class.ty.clone(),
215                fqn: class.fqn(),
216                fields,
217            });
218            records.insert(key.into(), shape.clone());
219            Ok(shape)
220        }
221        let mut records = BTreeMap::new();
222        for (key, class) in classes {
223            if class.data {
224                resolve(flat, classes, key, &mut Vec::new(), &mut records)?;
225            }
226        }
227        Ok(Self { records })
228    }
229    fn record(&self, ty: &syn::Type) -> Result<Option<&Rc<RecordShape>>> {
230        if let syn::Type::Reference(r) = ty {
231            if self.records.contains_key(&type_key(&r.elem)) {
232                return Err(format!(
233                    "data class `{}` supports owned occurrences only",
234                    type_key(&r.elem)
235                ));
236            }
237        }
238        Ok(self.records.get(&type_key(ty)))
239    }
240    pub(crate) fn render(&self, class: &Class) -> Result<String> {
241        let shape = &self.records[&type_key(&class.ty)];
242        let fields = shape
243            .fields
244            .iter()
245            .map(|f| {
246                format!(
247                    "    public val `{}`: {}",
248                    field_name(f),
249                    f.shape.kotlin_type()
250                )
251            })
252            .collect::<Vec<_>>()
253            .join(",\n");
254        let mut out = format!(
255            "package {}\n\npublic data class {}(\n{fields}\n)",
256            class.pkg, class.name
257        );
258        if shape
259            .fields
260            .iter()
261            .any(|f| matches!(f.shape, Shape::Value(Value::Bytes)))
262        {
263            let equal = shape
264                .fields
265                .iter()
266                .map(|f| {
267                    let n = field_name(f);
268                    match f.shape {
269                        Shape::Value(Value::Bytes) => {
270                            format!("this.`{n}`.contentEquals(other.`{n}`)")
271                        }
272                        Shape::Value(Value::Scalar(Primitive::F32 | Primitive::F64)) => {
273                            format!("this.`{n}`.compareTo(other.`{n}`) == 0")
274                        }
275                        _ => format!("this.`{n}` == other.`{n}`"),
276                    }
277                })
278                .collect::<Vec<_>>()
279                .join(" &&\n            ");
280            let hashes = shape
281                .fields
282                .iter()
283                .map(|f| {
284                    let n = field_name(f);
285                    if matches!(f.shape, Shape::Value(Value::Bytes)) {
286                        format!("this.`{n}`.contentHashCode()")
287                    } else {
288                        format!("this.`{n}`.hashCode()")
289                    }
290                })
291                .collect::<Vec<_>>();
292            let mut hash = hashes[0].clone();
293            for h in &hashes[1..] {
294                hash = format!("31 * ({hash}) + {h}");
295            }
296            let display = shape
297                .fields
298                .iter()
299                .map(|f| {
300                    let n = field_name(f);
301                    let suffix = if matches!(f.shape, Shape::Value(Value::Bytes)) {
302                        ".contentToString()"
303                    } else {
304                        ""
305                    };
306                    format!("{n}=${{this.`{n}`{suffix}}}")
307                })
308                .collect::<Vec<_>>()
309                .join(", ");
310            out.push_str(&format!(" {{\n    override fun equals(other: kotlin.Any?): kotlin.Boolean =\n        this === other || (other is {} &&\n            {equal})\n    override fun hashCode(): kotlin.Int = {hash}\n    override fun toString(): kotlin.String = \"{}({display})\"\n}}", class.name, class.name));
311        }
312        out.push('\n');
313        Ok(out)
314    }
315}

317    field.name.to_string().trim_start_matches("r#").into()

320impl RecordShape {
321    fn input(
322        &self,
323        w: &syn::Ident,
324        v: &syn::Ident,
325        active: &Condition,
326    ) -> Result<InputPlan<JniKind>> {
327        let mut out = InputPlan {
328            context: Default::default(),
329            acknowledgement: None,
330            wires: vec![],
331            validation: quote!(),
332            acquire: quote!(),
333            resources: vec![],
334            statements: quote!(),
335            argument: quote!(#v),
336        };
337        let mut fields = Vec::new();
338        for (i, f) in self.fields.iter().enumerate() {
339            let fw = format_ident!("{w}_f{i}");
340            let fv = format_ident!("{v}_f{i}");
341            let child = match &f.shape {
342                Shape::Value(_) => JniCodec.input_leaf(&f.ty, &(), fw, fv, active)?,
343                Shape::Record(r) => r.input(&fw, &fv, active)?,
344            };
345            debug_assert!(
346                child.acquire.is_empty()
347                    && child.resources.is_empty()
348                    && child.acknowledgement.is_none(),
349                "data fields must remain value-only"
350            );
351            out.wires.extend(child.wires);
352            out.validation.extend(child.validation);
353            out.statements.extend(child.statements);
354            let name = &f.name;
355            let arg = child.argument;
356            fields.push(quote!(#name: #arg));
357        }
358        let ty = &self.ty;
359        out.statements
360            .extend(quote!(let #v: #ty = #ty { #(#fields),* };));
361        Ok(out)
362    }
363    fn output(&self, w: &syn::Ident, value: TokenStream) -> Result<OutputPlan<JniKind>> {
364        let mut out = OutputPlan {
365            context: Default::default(),
366            delivery: Default::default(),
367            commit: quote!(),
368            wires: vec![],
369            statements: quote!(),
370            values: vec![],
371        };
372        // Bind even a non-place expression once before observing its fields.
373        let record = format_ident!("{w}_record");
374        out.statements.extend(quote!(let #record = &(#value);));
375        // Always observe fields. Owned results remain Rust-owned until delivery,
376        // and the same data leaf can safely appear in multiple output views.
377        for (i, f) in self.fields.iter().enumerate() {
378            let fw = format_ident!("{w}_f{i}");
379            let name = &f.name;
380            let field = quote!(&#record.#name);
381            let child = match &f.shape {
382                Shape::Value(_) => JniCodec.output_leaf(&f.ty, &(), fw, field, true)?,
383                Shape::Record(r) => r.output(&fw, field)?,
384            };
385            debug_assert!(
386                child.commit.is_empty(),
387                "data fields cannot transfer native ownership"
388            );
389            out.wires.extend(child.wires);
390            out.statements.extend(child.statements);
391            out.values.extend(child.values);
392        }
393        Ok(out)
394    }
395    fn paths(&self, path: &[String], paths: &mut Vec<Vec<String>>) {
396        for f in &self.fields {
397            let mut path = path.to_vec();
398            path.push(field_name(f));
399            match &f.shape {
400                Shape::Value(_) => paths.push(path),
401                Shape::Record(r) => r.paths(&path, paths),
402            }
403        }
404    }
405    fn mark(
406        self: &Rc<Self>,
407        wires: &mut [prebindgen_flat::Wire<JniKind>],
408        root: String,
409        condition: String,
410    ) {
411        let mut paths = Vec::new();
412        self.paths(&[], &mut paths);
413        assert_eq!(wires.len(), paths.len());
414        for (wire, path) in wires.iter_mut().zip(paths) {
415            wire.kind = JniKind::Data(Box::new(DataWire {
416                leaf: wire.kind.clone(),
417                root: root.clone(),
418                record: self.clone(),
419                path,
420                condition: condition.clone(),
421            }));
422        }
423    }
424    pub(crate) fn construct(&self, values: &mut impl Iterator<Item = String>) -> String {
425        let fields = self
426            .fields
427            .iter()
428            .map(|f| match &f.shape {
429                Shape::Value(_) => values.next().expect("one expression per record leaf"),
430                Shape::Record(r) => r.construct(values),
431            })
432            .collect::<Vec<_>>();
433        format!("{}({})", self.fqn, fields.join(", "))
434    }
435    pub(crate) fn leaf_count(&self) -> usize {
436        self.fields
437            .iter()
438            .map(|f| match &f.shape {
439                Shape::Value(_) => 1,
440                Shape::Record(r) => r.leaf_count(),
441            })
442            .sum()
443    }
444}
445impl Codec for DataCodec {
446    type Extra = ();
447    type Wire = JniKind;
448    type Context = JniContext;
449    type Input = JniInput;
450    type Output = JniOutput;
451    fn selector(&self, extra: &()) -> JniKind {
452        JniCodec.selector(extra)
453    }
454    fn consumes(&self, ty: &syn::Type, extra: &()) -> bool {
455        !self.records.contains_key(&type_key(ty)) && JniCodec.consumes(ty, extra)
456    }
457    fn finish_input(&self, plan: &mut InputPlan<JniKind>, extra: &()) -> Result<()> {
458        JniCodec.finish_input(plan, extra)
459    }
460    fn finish_output(&self, plan: &mut OutputPlan<JniKind>, extra: &()) -> Result<()> {
461        JniCodec.finish_output(plan, extra)
462    }
463    fn input_leaf(
464        &self,
465        ty: &syn::Type,
466        extra: &(),
467        wire: syn::Ident,
468        var: syn::Ident,
469        active: &Condition,
470    ) -> Result<InputPlan<JniKind>> {
471        let Some(record) = self.record(ty)? else {
472            return JniCodec.input_leaf(ty, extra, wire, var, active);
473        };
474        let mut plan = record.input(&wire, &var, active)?;
475        record.mark(&mut plan.wires, wire.to_string(), kotlin_condition(active));
476        Ok(plan)
477    }
478    fn output_leaf(
479        &self,
480        ty: &syn::Type,
481        extra: &(),
482        wire: syn::Ident,
483        value: TokenStream,
484        borrowed: bool,
485    ) -> Result<OutputPlan<JniKind>> {
486        let Some(record) = self.record(ty)? else {
487            return JniCodec.output_leaf(ty, extra, wire, value, borrowed);
488        };
489        let mut plan = record.output(&wire, value)?;
490        record.mark(&mut plan.wires, wire.to_string(), "true".into());
491        Ok(plan)
492    }
