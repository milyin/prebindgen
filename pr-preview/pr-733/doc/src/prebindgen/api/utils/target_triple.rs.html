10/// TargetTriple is a small utility around `target_lexicon::Triple` with helpers
11/// to access parts and to convert into Rust cfg tokens.
12#[derive(Clone, Debug, PartialEq, Eq, Hash)]
13pub struct TargetTriple {
14    triple: String,
15    arch: Option<String>,
16    vendor: Option<String>,
17    os: Option<String>,
18    env: Option<String>,
19}
20
21fn extract_cfg_condition(s: &str, name: &str) -> Option<String> {
22    let prefix = format!("{}=\"", name);
23    if s.starts_with(&prefix) && s.ends_with('"') {
24        let s = &s[prefix.len()..s.len() - 1];
25        if s.is_empty() {
26            None
27        } else {
28            Some(s.to_string())
29        }
30    } else {
31        None
32    }
33}
34
35impl TargetTriple {
36    /// Parse from a string like "aarch64-apple-darwin".
37    pub fn parse(s: impl Into<String>) -> Result<Self, String> {
38        let triple = s.into();
39
40        // Fast path: check global cache first.
41        if let Some(cached) = TARGET_TRIPLE_CACHE
42            .get_or_init(|| RwLock::new(HashMap::new()))
43            .read()
44            .ok()
45            .and_then(|m| m.get(&triple).cloned())
46        {
47            return Ok(cached);
48        }
49        // run `rustc --print cfg --target <target_triple>` and capture the output
50        let output = std::process::Command::new("rustc")
51            .args(["--print", "cfg", "--target", &triple])
52            .output()
53            .map_err(|e| format!("Failed to run rustc: {e}"))?;
54
55        if !output.status.success() {
56            return Err(format!("rustc failed with status: {}", output.status));
57        }
58
59        let stdout = String::from_utf8_lossy(&output.stdout);
60        let mut arch = None;
61        let mut vendor = None;
62        let mut os = None;
63        let mut env = None;
64
65        for line in stdout.lines() {
66            if let Some(a) = extract_cfg_condition(line, "target_arch") {
67                arch = Some(a);
68            } else if let Some(v) = extract_cfg_condition(line, "target_vendor") {
69                vendor = Some(v);
70            } else if let Some(o) = extract_cfg_condition(line, "target_os") {
71                os = Some(o);
72            } else if let Some(e) = extract_cfg_condition(line, "target_env") {
73                env = Some(e);
74            }
75        }
76
77        let parsed = Self {
78            triple: triple.clone(),
79            arch,
80            vendor,
81            os,
82            env,
83        };
84
85        // Store in cache for subsequent calls.
86        if let Some(lock) = TARGET_TRIPLE_CACHE.get() {
87            if let Ok(mut map) = lock.write() {
88                map.insert(triple, parsed.clone());
89            }
90        }
91
92        Ok(parsed)
93    }
94
95    /// Get the architecture as a canonical string used by Rust cfg target_arch.
96    pub fn arch(&self) -> Option<&str> {
97        self.arch.as_deref()
98    }
99
100    /// Get the vendor as string used by Rust cfg target_vendor.
101    pub fn vendor(&self) -> Option<&str> {
102        self.vendor.as_deref()
103    }
104
105    /// Get the operating system as string used by Rust cfg target_os.
106    pub fn os(&self) -> Option<&str> {
107        self.os.as_deref()
108    }
109
110    /// Get the environment as string used by Rust cfg target_env (may be "unknown").
111    pub fn env(&self) -> Option<&str> {
112        self.env.as_deref()
113    }
114
115    /// Build a cfg expression TokenStream like:
116    /// all(target_arch = "aarch64", target_vendor = "apple", target_os = "macos", target_env = "gnu")
117    /// Omits target_env when unknown/empty.
118    pub fn to_cfg_tokens(&self) -> TokenStream {
119        let mut parts: Vec<TokenStream> = Vec::with_capacity(4);
120        if let Some(ref arch) = self.arch {
121            let arch = LitStr::new(arch, proc_macro2::Span::call_site());
122            parts.push(quote! { target_arch = #arch });
123        }
124        if let Some(ref vendor) = self.vendor {
125            let vendor = LitStr::new(vendor, proc_macro2::Span::call_site());
126            parts.push(quote! { target_vendor = #vendor });
127        }
128        if let Some(ref os) = self.os {
129            let os = LitStr::new(os, proc_macro2::Span::call_site());
130            parts.push(quote! { target_os = #os });
131        }
132        if let Some(ref env) = self.env {
133            let env = LitStr::new(env, proc_macro2::Span::call_site());
134            parts.push(quote! { target_env = #env });
135        }
136        if parts.is_empty() {
137            quote! { true }
138        } else if parts.len() == 1 {
139            parts.remove(0)
140        } else {
141            quote! { all( #(#parts),* ) }
142        }
143    }
144}

146impl std::str::FromStr for TargetTriple {
147    type Err = String;
148
149    fn from_str(s: &str) -> Result<Self, Self::Err> {
150        TargetTriple::parse(s)
151    }
152}
153
154/// Allow quoting a TargetTriple directly, yielding its cfg tokens.
155impl quote::ToTokens for TargetTriple {
156    fn to_tokens(&self, tokens: &mut TokenStream) {
157        tokens.extend(self.to_cfg_tokens());
158    }
159}
