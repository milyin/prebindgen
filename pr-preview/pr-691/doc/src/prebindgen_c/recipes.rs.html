1//! Cbindgen's per-type policies, lowered into the registry's recipe vocabulary.
2//!
3//! A build script writes `.opaque_ptr()`, `.data_struct()`, `.enum_type()`,
4//! `.tagged_union()` or `.repr_c_struct()`, each saying what C shape one Rust
5//! type takes. This turns those into recipes in a [`Recipes`] table: the shared
6//! statement of **which parts** a value gets across in, with nothing about the
7//! C wire in it.
8//!
9//! There is one recipe per declared type and direction, and every crossing nobody
10//! declared takes the recipe the registry derives from its kind. So the chain of
11//! `or_else` guesses this replaced — try a custom conversion, then a handle,
12//! then a data struct, then an enum, … — is now a lookup, and the answer is
13//! whatever the build script said rather than whichever guess fired first.
14

16    flat::{Flat, Type, TypeRef},
17    recipe::{
18        Arm, Construct, Constructing, Deconstruct, Deconstructing, Direction, Reach, RecipeError,
19        RecipeName, Recipes,
20    },
21};

25/// The recipe a type with no parts takes: the adapter emits the conversion itself.
26fn whole() -> RecipeName {
27    RecipeName::new("whole")
28}
29
30/// The recipe a type crossing field by field takes.
31pub(crate) fn parts() -> RecipeName {
32    RecipeName::new("parts")
33}
34
35/// The recipe a value takes as a **tagged union's payload**, where it rides
36/// differently from both of the readings below.
37///
38/// A `Box<T>` or `Option<Box<T>>` over a declared `opaque_ptr` rides in the
39/// union as a bare `*mut t_t` the C side owns. That is neither the whole-value
40/// reading — a handle parameter is spelled `T` and reclaimed from its own
41/// pointer — nor a struct field's.
42///
43/// The recipe is filed on the **stripped** crossing, because `Crossing::key` peels
44/// `Box`: `Box<Blob>` and `Blob` share one crossing and are told apart by the
45/// site that picks between their recipes, and by the fragment, which is keyed by
46/// the spelling.
47pub(crate) fn payload() -> RecipeName {
48    RecipeName::new("payload")
49}
50
51/// The recipe a value takes **inside a `data_struct`'s mirror**, where its wire
52/// differs from the one it takes on its own.
53///
54/// One crossing read two ways is two recipes, and the site picks — which is what
55/// `declare_default` and `Ask::Recipe` exist for. Two types need it.
56///
57/// `bool`: a field arrives from C by value and may hold any byte until it is
58/// normalised, so it crosses as `MaybeUninit<bool>`, while a `bool` returned
59/// from Rust is already one of two values and passes through.
60///
61/// `String`: a field decodes a null pointer to an empty string, where a
62/// `String` parameter refuses one. Both readings were in the hand-written field
63/// walk; the recipe is what makes the difference visible.
64pub(crate) fn in_field() -> RecipeName {
65    RecipeName::new("field")
66}

68impl CbindgenBuilder {
69    /// Every recipe this binding's declarations state.
70    ///
71    /// A type declared but absent from the model is skipped rather than
72    /// refused: the scan already reports it, and reporting it twice in
73    /// different words helps nobody.
74    pub(crate) fn recipes(&self, model: &Flat) -> Result<Recipes, Vec<RecipeError>> {
75        let mut recipes = Recipes::builder();
76        // The crossings a second recipe will also land on, so the loop below knows
77        // to make the whole-value one the default rather than leaving two recipes
78        // with no answer between them.
79        //
80        // Two sources. A payload recipe shares a crossing with its handle's own,
81        // because `Crossing::key` peels `Box` — right for lookup, and it means
82        // `Blob` and `Box<Blob>` are one crossing. And `bool` and `String` each
83        // get a field recipe unconditionally, which collides with a whole-value
84        // recipe a binding declared itself.
85        let mut shared: std::collections::HashSet<TypeKey> = self
86            .boxed_payloads(model)
87            .iter()
88            .map(|t| t.borrow_target().unwrap_or(t).stripped_key())
89            .collect();
90        for builtin in [syn::parse_quote!(bool), syn::parse_quote!(String)] {
91            if let Ok(ty) = model.classify(&builtin) {
92                shared.insert(ty.key());
93            }
94        }
95        // Every per-type policy, and every `convert!`-declared conversion. The
96        // second matters as much as the first: a conversion may be declared on
97        // a type the registry would otherwise read as an arity layer, and
98        // `convert!(Option<Duration> => ..)` means the adapter emits that
99        // optional's conversion itself rather than wrapping `Duration`'s.
100        let mut declared: Vec<(TypeKey, Origin<syn::Type>)> = self
101            .declared_types()
102            .into_iter()
103            .chain(
104                self.convert_decls
105                    .iter()
106                    .map(|d| (d.key().clone(), d.rust_type().clone())),
107            )
108            .collect();
109        declared.sort_by(|a, b| a.0.as_str().cmp(b.0.as_str()));
110        declared.dedup_by(|a, b| a.0 == b.0);
111        let declared_keys: std::collections::HashSet<TypeKey> =
112            declared.iter().map(|(k, _)| k.clone()).collect();
113
114        for (_key, origin) in declared {
115            // The declarator's own tokens, re-parsed: a declared type may be
116            // an alias the model holds as an `Extern`, which carries no reading
117            // of its own to borrow.
118            let Ok(spelled) = syn::parse2::<syn::Type>(origin.declared_spelling()) else {
119                continue;
120            };
121            let Ok(ty) = model.classify(&spelled) else {
122                continue;
123            };
124            // A tagged union is a tag plus a union, rebuilt one arm at a time.
125            // Neither the tag nor the selector is a crossing, so neither is
126            // declared: how the C side is told which arm is live is the
127            // adapter's business.
128            if self.tagged_unions.contains_key(&_key) {
129                let Some(alternatives) = alternative_field_counts(model, &_key) else {
130                    continue;
131                };
132                let out: Vec<Arm<Deconstruct>> = alternatives
133                    .iter()
134                    .enumerate()
135                    .map(|(alternative, count)| Arm {
136                        alternative,
137                        op: Deconstruct::Fields((0..*count).map(Reach::Field).collect()),
138                    })
139                    .collect();
140                let into: Vec<Arm<Construct>> = alternatives
141                    .iter()
142                    .enumerate()
143                    .map(|(alternative, _)| Arm {
144                        alternative,
145                        op: Construct::Fields,
146                    })
147                    .collect();
148                recipes
149                    .declare(ty.clone(), parts(), Deconstructing::Choice { arms: out })
150                    .declare(ty, parts(), Constructing::Choice { arms: into });
151                continue;
152            }
153            // A `data_struct` converts each field and reassembles the
154            // converted fields into one C struct — many parts, one wire value,
155            // which is the pair #450 keeps apart. Everything else declared here
156            // is one wire value with nothing inside it that crosses on its own:
157            // an opaque handle, a value-opaque mirror, a C enum.
158            if let Some(count) = self
159                .data
160                .contains_key(&_key)
161                .then(|| field_count(model, &_key))
162            {
163                let Some(count) = count else { continue };
164                let reaches: Vec<Reach> = (0..count).map(Reach::Field).collect();
165                recipes
166                    .declare(
167                        ty.clone(),
168                        parts(),
169                        Deconstructing::Product(Deconstruct::Fields(reaches)),
170                    )
171                    .declare(ty, parts(), Constructing::Product(Construct::Fields));
172                continue;
173            }
174            if shared.contains(&ty.borrow_target().unwrap_or(&ty).stripped_key()) {
175                recipes
176                    .declare_default(ty.clone(), whole(), Deconstructing::Atomic)
177                    .declare_default(ty, whole(), Constructing::Atomic);
178            } else {
179                recipes
180                    .declare(ty.clone(), whole(), Deconstructing::Atomic)
181                    .declare(ty, whole(), Constructing::Atomic);
182            }
183        }
184        // The payload recipes, on the stripped crossing each `Box`-over-handle
185        // shares with its own handle type. Only the types a union arm actually
186        // carries.
187        for ty in self.boxed_payloads(model) {
188            // The handle's own recipe is already filed on this crossing, so one of
189            // the two has to say which a site gets when it names neither.
190            recipes
191                .declare(ty.clone(), payload(), Deconstructing::Atomic)
192                .declare(ty, payload(), Constructing::Atomic);
193        }
194
195        // The field recipes. Declared for every binding rather than only where a
196        // struct has such a field: a recipe for a crossing nobody uses is inert,
197        // and the alternative is walking every declared struct twice to find
198        // out.
199        //
200        // **Always**, whether or not the type's whole-value recipe was declared
201        // above. The two are independent readings — the hand-written walks kept
202        // them so — and a binding may declare the whole-value one itself:
203        // `convert!` refuses a builtin, but `opaque_ptr(String)` is accepted and
204        // `out_terminal` has an arm for it. Suppressing the field recipe there
205        // stranded every string field, because `bindings` asks each one for it.
206        let boolean = model
207            .classify(&syn::parse_quote!(bool))
208            .expect("bool is a scalar the model always classifies");
209        // A whole-value recipe the loop already filed stays as it is and only has
210        // to become the default; otherwise it is declared here.
211        if !declared_keys.contains(&boolean.key()) {
212            recipes
213                .declare_default(boolean.clone(), whole(), Deconstructing::Atomic)
214                .declare_default(boolean.clone(), whole(), Constructing::Atomic);
215        }
216        recipes
217            .declare(boolean.clone(), in_field(), Deconstructing::Atomic)
218            .declare(boolean, in_field(), Constructing::Atomic);
219        if let Ok(string) = model.classify(&syn::parse_quote!(String)) {
220            // Output only ever has one reading, so `String` gets a second recipe in
221            // the constructing direction alone.
222            if !declared_keys.contains(&string.key()) {
223                recipes.declare_default(string.clone(), whole(), Constructing::Atomic);
224            }
225            recipes.declare(string, in_field(), Constructing::Atomic);
226        }
227
228        recipes.build(model)
229    }

231    /// Which recipe a union payload of this type takes, and in which directions,
232    /// where it is not the type's own default.
233    ///
234    /// Every `Box`-over-handle a union arm carries, keyed once.
235    fn boxed_payloads(&self, model: &Flat) -> Vec<TypeRef> {
236        let mut payloads: Vec<TypeRef> = Vec::new();
237        for (key, _) in sorted_by_key(&self.tagged_unions) {
238            for fields in alternatives_of(model, key).unwrap_or_default() {
239                for field in fields {
240                    if self.declared_opaque_payload_inner(&field.ty).is_some()
241                        || r_boxed_inner(&field.ty).is_some()
242                    {
243                        payloads.push(field.ty.clone());
244                    }
245                }
246            }
247        }
248        payloads.sort_by_key(|t| t.key().as_str().to_owned());
249        payloads.dedup_by_key(|t| t.key().as_str().to_owned());
250        payloads
251    }

253    /// `bool` and `String` read as they do inside a struct; a `Box`-over-handle
254    /// needs [`payload`], the one reading neither of the other two covers.
255    fn payload_reading(&self, fty: &TypeRef) -> Option<(RecipeName, &'static [Direction])> {
256        use prebindgen_registry::flat::{ScalarKind, TypeKind};
257        if self.declared_opaque_payload_inner(fty).is_some() || r_boxed_inner(fty).is_some() {
258            return Some((payload(), &[Direction::Construct, Direction::Deconstruct]));
259        }
260        match fty.unwrapped().kind() {
261            TypeKind::Scalar(ScalarKind::Bool) => {
262                Some((in_field(), &[Direction::Construct, Direction::Deconstruct]))
263            }
264            TypeKind::String => Some((in_field(), &[Direction::Construct])),
265            _ => None,
266        }
267    }

269    /// Which recipe each part of a declared product takes, where it is not the
270    /// part type's own default.
271    ///
272    /// Only `bool` inside a `data_struct` today — see [`in_field`].
273    pub(crate) fn bindings(
274        &self,
275        model: &Flat,
276        recipes: &Recipes,
277    ) -> Result<prebindgen_registry::recipe::Bindings, Vec<RecipeError>> {
278        // `recipe::Origin` is which declaration asked; `flat::Origin` is a
279        // captured item's own syntax. Both are in play here, so the one that
280        // arrives by glob keeps its name.
281        use prebindgen_registry::recipe::{Ask, Bindings, Crossing, Origin as Asked, Site};
282
283        let mut bound = Bindings::builder();
284
285        // C hands a fallible function's ok value back through an out-parameter
286        // and its error through the return, so what crosses at the **return**
287        // site is the ok arm alone. The error arm is its own site, which the
288        // walk enumerates because a `Result` has two arms whatever reads it.
289        let mut fallible: Vec<&syn::Ident> = model
290            .functions()
291            .filter(|function| function.ret.fallible_parts().is_some())
292            .map(|function| &function.name)
293            .collect();
294        fallible.sort_by_key(|name| name.to_string());
295        for name in fallible {
296            let Some((ok, _)) = model
297                .function(name)
298                .and_then(|function| function.ret.fallible_parts())
299            else {
300                continue;
301            };
302            bound.bind(
303                Site {
304                    owner: name.clone(),
305                    role: prebindgen_registry::recipe::Role::Return,
306                },
307                Crossing::new(
308                    ok.clone(),
309                    prebindgen_registry::recipe::Direction::Deconstruct,
310                ),
311                Ask::Default,
312                Asked::Function,
313            );
314        }
315        // A union's payload reads like a struct's field for `bool` and
316        // `String`, and needs a reading of its own where it is a `Box` over a
317        // declared handle — see [`payload`].
318        for (key, cfg) in sorted_by_key(&self.tagged_unions) {
319            let Ok(spelled) = syn::parse2::<syn::Type>(cfg.rust_type.declared_spelling()) else {
320                continue;
321            };
322            let Ok(ty) = model.classify(&spelled) else {
323                continue;
324            };
325            let Some(alternatives) = alternatives_of(model, key) else {
326                continue;
327            };
328            // Every arm numbers its parts from zero, so a payload is addressed
329            // by its alternative as well as its index — `part 0` alone names
330            // one part per arm.
331            for (arm, fields) in alternatives.iter().enumerate() {
332                for (index, field) in fields.iter().enumerate() {
333                    let Some((recipe, directions)) = self.payload_reading(&field.ty) else {
334                        continue;
335                    };
336                    for &direction in directions {
337                        let of = Crossing::new(ty.clone(), direction);
338                        let row = of.row(parts());
339                        bound.bind(
340                            Site::arm_part(&row, Some(arm), index),
341                            Crossing::new(field.ty.clone(), direction),
342                            Ask::Recipe(recipe.clone()),
343                            Asked::Part,
344                        );
345                    }
346                }
347            }
348        }
349        let mut declared: Vec<(TypeKey, Origin<syn::Type>)> = self
350            .data
351            .iter()
352            .map(|(k, c)| (k.clone(), c.rust_type.clone()))
353            .collect();
354        declared.sort_by(|a, b| a.0.as_str().cmp(b.0.as_str()));
355        for (key, origin) in declared {
356            let Ok(spelled) = syn::parse2::<syn::Type>(origin.declared_spelling()) else {
357                continue;
358            };
359            let Ok(ty) = model.classify(&spelled) else {
360                continue;
361            };
362            let Some(fields) = struct_fields(model, &key) else {
363                continue;
364            };
365            for (index, field) in fields.iter().enumerate() {
366                let kind = field.ty.unwrapped().kind();
367                let is_bool = matches!(
368                    kind,
369                    prebindgen_registry::flat::TypeKind::Scalar(
370                        prebindgen_registry::flat::ScalarKind::Bool
371                    )
372                );
373                let is_string = matches!(kind, prebindgen_registry::flat::TypeKind::String);
374                if !is_bool && !is_string {
375                    continue;
376                }
377                // A `String` reads differently only on the way in.
378                let directions: &[Direction] = if is_bool {
379                    &[Direction::Construct, Direction::Deconstruct]
380                } else {
381                    &[Direction::Construct]
382                };
383                for &direction in directions {
384                    let of = Crossing::new(ty.clone(), direction);
385                    let row = of.row(parts());
386                    bound.bind(
387                        Site::part(&row, index),
388                        Crossing::new(field.ty.clone(), direction),
389                        Ask::Recipe(in_field()),
390                        Asked::Part,
391                    );
392                }
393            }
394        }
395        bound.build(recipes)
396    }

399/// The fields the model gives this declared struct.
400fn struct_fields(model: &Flat, key: &TypeKey) -> Option<Vec<prebindgen_registry::flat::Field>> {
401    match model.declared_type(&key.ident()?)? {
402        Type::Struct(s) => Some(s.fields.clone()),
403        _ => None,
404    }
405}

407/// The alternatives of this declared sum, with their payload fields.
408fn alternatives_of(
409    model: &Flat,
410    key: &TypeKey,
411) -> Option<Vec<Vec<prebindgen_registry::flat::Field>>> {
412    match model.declared_type(&key.ident()?)? {
413        Type::Variant(v) => Some(v.alternatives.iter().map(|a| a.fields.clone()).collect()),
414        _ => None,
415    }
416}
417
418/// How many fields each alternative of this declared sum carries.
419fn alternative_field_counts(model: &Flat, key: &TypeKey) -> Option<Vec<usize>> {
420    match model.declared_type(&key.ident()?)? {
421        Type::Variant(v) => Some(v.alternatives.iter().map(|a| a.fields.len()).collect()),
422        _ => None,
423    }
424}
425
426/// How many fields the model gives this declared struct.
427fn field_count(model: &Flat, key: &TypeKey) -> Option<usize> {
428    Some(struct_fields(model, key)?.len())
