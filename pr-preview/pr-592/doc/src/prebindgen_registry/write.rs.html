1//! Rust file emission for the resolved `Registry`.
2//!
3//! `write_rust` takes the [`Assembly`] the adapter compiled — the frozen set
4//! of final artifacts the file is made of; renders each with the writer-owned
5//! [`crate::RustWriter`], surrounds them with the adapter's prerequisites and
6//! prebindgen's own anonymous feature-check consts, and hands the assembled
7//! file to `Destination::write`.
8//!
9//! The artifacts arrive from the adapter rather than being collected from the
10//! registry, which is what lets one crossing contribute more than one function
11//! — or one that occupies more than a single wire value.
12//!
13//! This module is `pub`, so **every `pub` item in it is public API of the
14//! crate**. [`RustArtifact`] is the deliberately narrow late-rendering seam for
15//! out-of-crate adapters. Every artifact carries registry-owned semantic
16//! identity; final Rust names are never used as planning identity.
17

19    collections::{BTreeSet, HashMap},
20    path::{Path, PathBuf},

25/// Errors surfaced by the file-emission phase.
26///
27/// Binding validation is NOT here — it runs once in
28/// [`RegistryBuilder::build`](crate::RegistryBuilder::build)
29/// (see [`Prebindgen::validate_resolved`]), so an invalid binding fails
30/// before a built generator exists and never reaches a writer.
31/// Emission-integrity checks still run here because reachability is finalized
32/// while per-item output is assembled.
33#[derive(Debug)]
34pub enum WriteError {
35    /// A reachable artifact calls one that does not reach the file, either
36    /// because reachability filtering dropped it or because it was never
37    /// planned. Proven from the artifacts' own edges, before anything renders.
38    UnreachedDependency {
39        /// `(caller, unreached callee)` identities, sorted and de-duplicated.
40        edges: Vec<(String, String)>,
41    },
42}
43
44impl std::fmt::Display for WriteError {
45    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
46        match self {
47            WriteError::UnreachedDependency { edges } => {
48                write!(f, "generated artifacts call artifacts the file omits:")?;
49                for (caller, callee) in edges {
50                    write!(f, "\n  - {caller} calls {callee}")?;
51                }
52                write!(
53                    f,
54                    "\nartifact reachability or dependency planning is incomplete"
55                )
56            }
57        }
58    }
59}
60
61impl std::error::Error for WriteError {}

63/// Identity of one final artifact in an [`Assembly`].
64///
65/// A private converter is identified by the registry operation it implements;
66/// every other final artifact — a helper the converter bodies call, an
67/// exported wrapper, a type mirror — is identified by the adapter-scoped
68/// [`ArtifactId`](crate::generation::ArtifactId) its plan was recorded under.
69#[derive(Clone, Debug, PartialEq, Eq, Hash)]
70pub enum ArtifactKey {
71    /// A private converter, identified by its registry operation.
72    Operation(crate::generation::OperationId),
73    /// Any other final artifact, identified by its plan's artifact identity.
74    Artifact(crate::generation::ArtifactId),
75}
76
77impl std::fmt::Display for ArtifactKey {
78    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
79        match self {
80            ArtifactKey::Operation(operation) => operation.fmt(f),
81            ArtifactKey::Artifact(artifact) => artifact.fmt(f),
82        }
83    }
84}

86/// One final artifact of the generated Rust file.
87///
88/// Resolution decides what the artifact is and what it depends on; this trait
89/// is the one thing the artifact still owes the writer — the Rust items it
90/// becomes. [`Self::render`] runs at the final writing boundary, with the
91/// [`crate::RustWriter`] used for the rest of final Rust emission and with no
92/// access to a live [`Registry`], so an artifact cannot resume resolution
93/// while the file is being assembled.
94pub trait RustArtifact {
95    /// Semantic identity, owned by the registry rather than by the Rust
96    /// symbol final emission will choose.
97    fn key(&self) -> ArtifactKey;
98
99    /// Whether the generated adapter surface reaches this artifact.
100    /// Validation-only artifacts return false: they are still rendered so
101    /// that a caller of one can be reported, but they do not reach the file.
102    fn reachable(&self) -> bool {
103        true
104    }
105
106    /// The identities this artifact's items answer for.
107    ///
108    /// Defaults to its own — the common case, where one artifact renders the
109    /// items of one identity. An artifact that renders another identity's item
110    /// too says so here, which is how a dependency on that identity is
111    /// satisfied. The C callback artifact does exactly that: it renders the
112    /// Invoke helper of a converter operation whose own fragment deliberately
113    /// carries no artifact.
114    fn provides(&self) -> Vec<ArtifactKey> {
115        vec![self.key()]
116    }
117
118    /// The artifacts this one's body calls.
119    ///
120    /// Deliberately without a default: an artifact that calls nothing says so
121    /// by returning an empty vector, and one that gains a call has to answer
122    /// the question again. These edges are what proves the file complete —
123    /// see [`WriteError::UnreachedDependency`] — so an unanswered one is a
124    /// silently weaker check rather than a compile error.
125    fn calls(&self) -> Vec<ArtifactKey>;
126
127    /// Materialize the artifact's Rust items.
128    fn render(&self, emit: &crate::RustWriter) -> Vec<syn::Item>;
129}

131/// The frozen set of final artifacts a generated Rust file is assembled from.
132///
133/// An assembly is ordered: artifacts reach the file in the order the adapter
134/// added them, which for converters is the registry-owned dependency order of
135/// the fragments they came from. It holds one artifact per [`ArtifactKey`], so
136/// sharing never depends on the Rust symbol final emission allocates.
137pub struct Assembly<A> {
138    artifacts: Vec<A>,
139    /// Every identity the held artifacts answer for, including the ones an
140    /// artifact provides beyond its own key.
141    provided: HashMap<ArtifactKey, usize>,
142}
143
144impl<A: RustArtifact> Assembly<A> {
145    /// The artifacts, in emission order, including the ones no reachable
146    /// artifact calls. The writer skips those; they are held so that
147    /// [`Self::reaches`] can tell "planned but unreached" from "never
148    /// planned".
149    pub fn artifacts(&self) -> impl ExactSizeIterator<Item = &A> {
150        self.artifacts.iter()
151    }
152
153    /// Whether an artifact answering for this identity reaches the file.
154    ///
155    /// Reachability is read here rather than when the assembly was frozen: an
156    /// adapter may reach an artifact after adding it, and the file is what
157    /// settles the answer.
158    pub fn reaches(&self, key: &ArtifactKey) -> bool {
159        self.provided
160            .get(key)
161            .is_some_and(|position| self.artifacts[*position].reachable())
162    }
163}

165/// Collection phase preceding a frozen [`Assembly`].
166pub struct AssemblyBuilder<A> {
167    positions: HashMap<ArtifactKey, usize>,
168    artifacts: Vec<A>,
169}
170
171impl<A> Default for AssemblyBuilder<A> {
172    fn default() -> Self {
173        Self {
174            positions: HashMap::new(),
175            artifacts: Vec::new(),
176        }
177    }
178}
179
180impl<A: RustArtifact> AssemblyBuilder<A> {
181    /// Start an empty assembly.
182    pub fn new() -> Self {
183        Self::default()
184    }
185
186    /// Add one final artifact.
187    ///
188    /// An identity already present is kept once. A reachable artifact replaces
189    /// an unreachable one already held under the same identity, which is what
190    /// lets an adapter add a dormant artifact before the parent that reaches
191    /// it has been planned.
192    ///
193    /// Two *reachable* artifacts sharing one identity are a planning error,
194    /// not a choice to make: keeping either would drop an item the file needs,
195    /// and the reader would meet it as a missing name in the generated crate
196    /// rather than as a diagnostic here. Converters are exempt — one
197    /// registry operation is legitimately reached from several sites, and
198    /// de-duplicating those is what this method is for.
199    pub fn artifact(&mut self, artifact: A) -> &mut Self {
200        match self.positions.get(&artifact.key()).copied() {
201            Some(position) => {
202                if let ArtifactKey::Artifact(id) = artifact.key() {
203                    assert!(
204                        !(self.artifacts[position].reachable() && artifact.reachable()),
205                        "two reachable artifacts share the identity {id}"
206                    );
207                }
208                if !self.artifacts[position].reachable() && artifact.reachable() {
209                    self.artifacts[position] = artifact;
210                }
211            }
212            None => {
213                self.positions.insert(artifact.key(), self.artifacts.len());
214                self.artifacts.push(artifact);
215            }
216        }
217        self
218    }
219
220    /// Freeze the assembly.
221    pub fn build(self) -> Assembly<A> {
222        let mut provided = self.positions;
223        for (position, artifact) in self.artifacts.iter().enumerate() {
224            for key in artifact.provides() {
225                provided.insert(key, position);
226            }
227        }
228        Assembly {
229            artifacts: self.artifacts,
230            provided,
231        }
232    }
233}

235impl<A: RustArtifact> FromIterator<A> for Assembly<A> {
236    fn from_iter<I: IntoIterator<Item = A>>(artifacts: I) -> Self {
237        let mut builder = AssemblyBuilder::new();
238        for artifact in artifacts {
239            builder.artifact(artifact);
240        }
241        builder.build()
242    }
243}

245/// Check an assembly's edges against what its artifacts render: every call to
246/// another artifact's item is declared by [`RustArtifact::calls`], and every
247/// identity [`RustArtifact::provides`] claims is one the artifact defines.
248///
249/// Test support, like [`RustWriter::for_test`](crate::RustWriter::for_test).
250/// Emission proves the file complete from the edges alone, which is only worth
251/// anything if the edges name every call — and the one place the calls
252/// themselves are visible is the rendered output. So the adapters' own test
253/// suites run this over the bindings they build: production reasons from the
254/// edges, and this checks the edges against what rendering actually emits.
255///
256/// `namespace` is the adapter's operation namespace, the one it passes to
257/// [`RustWriter::operation_ident`](crate::RustWriter::operation_ident).
258///
259/// # Panics
260///
261/// Naming the caller, the callee and the undeclared call, if a rendered body
262/// calls a converter its artifact did not declare.
263#[cfg(any(test, feature = "testing"))]
264pub fn assert_edges_cover_rendered_calls<A: RustArtifact>(
265    assembly: &Assembly<A>,
266    emit: &crate::RustWriter,
267    namespace: &str,
268) {
269    // What every artifact actually renders, which is the ground truth both
270    // halves are checked against: a call resolves to an artifact by the name
271    // that artifact defines, whether the identity is an operation or an
272    // adapter-scoped artifact. A runtime helper is called by name like any
273    // converter.
274    let rendered: Vec<(ArtifactKey, BTreeSet<String>, BTreeSet<String>)> = assembly
275        .artifacts()
276        .map(|artifact| {
277            let mut called = CalledIdents(BTreeSet::new());
278            let mut defined = BTreeSet::new();
279            for mut item in artifact.render(emit) {
280                if let syn::Item::Fn(function) = &item {
281                    defined.insert(function.sig.ident.to_string());
282                }
283                syn::visit_mut::VisitMut::visit_item_mut(&mut called, &mut item);
284            }
285            (artifact.key(), defined, called.0)
286        })
287        .collect();
288    let mut definer: HashMap<&str, &ArtifactKey> = HashMap::new();
289    for (key, defined, _) in &rendered {
290        for name in defined {
291            definer.insert(name.as_str(), key);
292        }
293    }
294    let mut provider: HashMap<ArtifactKey, usize> = HashMap::new();
295    for (position, artifact) in assembly.artifacts().enumerate() {
296        for key in artifact.provides() {
297            provider.insert(key, position);
298        }
299    }
300
301    for (artifact, (key, defined, called)) in assembly.artifacts().zip(&rendered) {
302        // Every identity claimed must be one this artifact actually defines,
303        // or claiming it would grant reachability to an item no one renders.
304        for claimed in artifact.provides() {
305            if let ArtifactKey::Operation(operation) = &claimed {
306                let ident = emit.operation_ident(namespace, operation).to_string();
307                assert!(
308                    defined.contains(&ident),
309                    "{key} claims to provide {claimed}, but renders no `{ident}`"
310                );
311            }
312        }
313        if !artifact.reachable() {
314            continue;
315        }
316        let mut allowed: BTreeSet<&str> = defined.iter().map(String::as_str).collect();
317        for callee in artifact.calls() {
318            // Resolved the way emission resolves it: through whichever
319            // artifact answers for that identity, which is not always the one
320            // whose own key it is.
321            allowed.extend(
322                provider
323                    .get(&callee)
324                    .into_iter()
325                    .flat_map(|position| rendered[*position].1.iter().map(String::as_str)),
326            );
327        }
328        for name in called {
329            if allowed.contains(name.as_str()) {
330                continue;
331            }
332            if let Some(callee) = definer.get(name.as_str()) {
333                panic!(
334                    "{key} calls {callee} as `{name}`, which it does not declare as a dependency"
335                );
336            }
337        }
338    }
339}

341/// Every function name called in the visited items.
342#[cfg(any(test, feature = "testing"))]
343struct CalledIdents(BTreeSet<String>);
344
345#[cfg(any(test, feature = "testing"))]
346impl syn::visit_mut::VisitMut for CalledIdents {
347    fn visit_expr_call_mut(&mut self, call: &mut syn::ExprCall) {
348        if let syn::Expr::Path(path) = call.func.as_ref() {
349            if let Some(segment) = path.path.segments.last() {
350                self.0.insert(segment.ident.to_string());
351            }
352        }
353        syn::visit_mut::visit_expr_call_mut(self, call);
354    }
355}

357/// Emit a resolved registry, assembling the generated Rust file from a frozen
358/// [`Assembly`].
359///
360/// The assembly is what the adapter's compilation produced: one final artifact
361/// per item the file will contain, already deduplicated and in registry-owned
362/// dependency order. Handing artifacts over instead of a converter table is
363/// what frees an adapter to emit a conversion the table could not hold —
364/// several functions for one crossing, or one occupying more than a single
365/// wire value.
366///
367/// `out_path` may be relative (resolved against `OUT_DIR` by prebindgen) or
368/// absolute. Returns the path actually written.
369pub fn write_rust<P: AsRef<Path>, E: Prebindgen, A: RustArtifact>(
370    registry: &Registry,
371    ext: &E,
372    assembly: &Assembly<A>,
373    out_path: P,
374) -> Result<PathBuf, WriteError> {
375    // Validation already ran ONCE in the generator's `build` — a built generator
376    // (the only source of a resolved registry) is valid by construction, so
377    // this writer does no binding resolution. It does validate the assembled
378    // private call graph before handing the file to the destination.
379    // The capability, minted here and nowhere else in this function's reach.
380    // Every artifact below is handed a borrow; nothing else in the pipeline is.
381    // See `prebindgen_flat::flat::emit` for what that buys and what it
382    // deliberately does not.
383    let emit = crate::RustWriter::new(registry, ext.source_module());
384    let mut items: Vec<syn::Item> = Vec::new();
385
386    // Dependency completeness, proven from the artifacts' own edges before
387    // anything is rendered: whatever a reachable artifact calls must itself
388    // reach the file.
389    let mut unreached: BTreeSet<(String, String)> = BTreeSet::new();
390    for artifact in assembly.artifacts().filter(|artifact| artifact.reachable()) {
391        for callee in artifact.calls() {
392            if !assembly.reaches(&callee) {
393                unreached.insert((artifact.key().to_string(), callee.to_string()));
394            }
395        }
396    }
397    if !unreached.is_empty() {
398        return Err(WriteError::UnreachedDependency {
399            edges: unreached.into_iter().collect(),
400        });
401    }
402
403    // Only what reaches the file renders. An unreachable artifact used to be
404    // rendered anyway, for its function names alone, so that a caller of one
405    // could be named; the edges above answer that question without it.
406    for artifact in assembly.artifacts().filter(|artifact| artifact.reachable()) {
407        items.extend(artifact.render(&emit));
408    }
409
410    // 2. Anonymous consts, verbatim. Last, and in stream order. These are
411    //    prebindgen's own injected feature checks, identical for every
412    //    adapter and named by none, so they are written here rather than
413    //    planned as an artifact of one adapter's assembly.
414    for guard in registry.flat().guards() {
415        items.push(syn::Item::Const(emit.guard(guard)));
416    }
417
418    let dest: Destination = items.into_iter().collect();
419    Ok(dest.write(out_path))
420}
