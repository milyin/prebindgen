12/// Peel the layers that never change whether a value's core is a Kotlin enum,
13/// **off the model**: a borrow and an optional, in any nesting. So `&Priority`,
14/// `Priority`, `Option<Priority>` and `Option<&Priority>` all probe as
15/// `Priority` — letting nullable enum params (`Option<enum>`) wire as the raw
16/// `Int` discriminant plus an allocated niche, instead of leaking the enum
17/// object to the Rust converter.
18///
19/// A **run is not peeled**. `Vec<Priority>` is a `List<Priority>`, not an enum,
20/// so this is deliberately not [`TypeRef::layer_stack`], which strips the
21/// sequence layer too.
22///
23/// Borrowing rather than composing is not a shortcut: every layer of a reading
24/// already holds the next as a `TypeRef` of its own, so there is nothing to
25/// mint — which is also why this needs no registry. What it returns spells
26/// itself (`spell()`) and classifies itself (`kind`), and the two cannot
27/// disagree.
28pub(crate) fn enum_probe(reading: &TypeRef) -> &TypeRef {
29    let mut cur = reading;
30    while let Some(inner) = cur.borrow_target().or_else(|| cur.optional_inner()) {
31        cur = inner;
32    }
33    cur
34}

36// The bottom-up layer fold is the shared `prebindgen_registry::shape::fold_shape`
37// (its `on_optional` receives the layer's `&NullableKind` + the wrapped
38// `&FoldStrategy`, so callers can special-case e.g. a `Niche` layer sitting
39// directly over the `Base` leaf). Used by the **type-name** folds
40// (`handle_kt_type` / `projection_wire_return`). The **expression** folds
41// (`render_handle_close` / `fold_projection_wrap`) are deliberately *not*
42// expressed through it: they fold the other direction (threading a `receiver` /
43// fresh lambda variable top-down rather than combining a bottom-up result), so
44// a shared combinator would obscure rather than simplify them.
45use prebindgen_registry::shape::fold_shape;

47/// The Kotlin type for a closeable handle reached through the folded
48/// [`FoldStrategy`] layers, given the leaf typed-handle type (e.g.
49/// `ZKeyExpr`): `Direct → ZKeyExpr`, `Nullable(inner) → <inner>?`,
50/// `Iterable(inner) → List<<inner>>`.
51pub(crate) fn handle_kt_type(strategy: &FoldStrategy, leaf: &KtType) -> KtType {
52    fold_shape(
53        strategy,
54        &|| leaf.clone(),
55        // The declared Kotlin projection type is `T?` regardless of how null
56        // is represented over the wire — the wrap fold and the wire-return
57        // helper read the kind to handle the wire shape separately.
58        &|inner, _kind, _inner_strategy| inner.nullable(),
59        &|inner| KtType::generic("List", [inner]),
60    )
61}

63/// Typed Kotlin leaf of a projection. Declared handle projections
64/// take their configured class FQN; the built-in `u64` projection is Kotlin's
65/// stable unsigned scalar type.
66pub(crate) fn projection_leaf_kt(ext: &Declarations, proj: &Projection) -> Option<KtType> {
67    match proj.kind {
68        ProjectionKind::Handle => ext.kotlin_fqn(&proj.leaf_key).map(KtType::cls),
69        ProjectionKind::Unsigned64 => Some(KtType::cls("ULong")),
70    }
71}

73/// Wrap one raw projection leaf into its typed Kotlin form.
74pub(crate) fn projection_wrap_expr(kind: &ProjectionKind, short: &str, raw: &str) -> String {
75    match kind {
76        ProjectionKind::Handle => handle_from_raw(short, raw),
77        ProjectionKind::Unsigned64 => format!("{raw}.toULong()"),

81/// For a projection (handle / unsigned) **struct field**,
82/// compute the `(wire_param_type, wrap_expr)` the data class's `fromParts`
83/// factory uses: the wire param type matches the leaf wire
84/// the late struct-output codec passes (handle → `Long` jlong sentinel), and the wrap
85/// reconstructs the typed value in JVM
86/// bytecode (`Short(arg)`, with null mapped from the `0L` sentinel for handles
87/// or the declared invalid `Long` for a bounded unsigned representation; JVM
88/// null remains the fallback for non-niche value projections). Only the
89/// `Direct` and `Nullable{Direct}` shapes a scalar projection field can take
90/// are supported — a collection
91/// (`Vec<projection>`) field is rejected (matching the struct bridge's
92/// scalar-only guard).
93pub(crate) fn factory_projection_wire_wrap(
94    proj: &crate::jni::Projection,
95    short: &str,
96    name: &str,
97) -> (KtType, String) {
98    use prebindgen_registry::shape::Shape::*;
99
100    use crate::jni::{NullableKind, ProjectionKind::*};
101    let direct = |kind: &crate::jni::ProjectionKind| match kind {
102        Handle => (KtType::long(), handle_from_raw(short, name)),
103        Unsigned64 => (KtType::long(), format!("{name}.toULong()")),
104    };
105    match &proj.strategy {
106        Base => direct(&proj.kind),
107        Optional(nullable, inner) => {
108            if !matches!(**inner, Base) {
109                panic!(
110                    "factory_projection_wire_wrap: only `Nullable<Direct>` projection struct \
111                     fields are supported (field `{name}`)"
112                );
113            }
114            match proj.kind {
115                // Handle null rides the `0L` jlong sentinel. The present case
116                // mints the handle through `handle_from_raw` for the same
117                // reason the `Base` arm above does: a handle's constructor is
118                // private, and the factory is the only way in (#430).
119                Handle => (
120                    KtType::long(),
121                    format!(
122                        "if ({name} == 0L) null else {}",
123                        handle_from_raw(short, name)
124                    ),
125                ),
126                Unsigned64 => match nullable {
127                    // A bounded unsigned representation reserves an invalid
128                    // raw value for `None`, so the factory receives primitive
129                    // `Long` and restores the nullable semantic property.
130                    NullableKind::Niche => {
131                        let sentinel = projection_leaf_sentinel(proj).unwrap_or_else(|| {
132                            panic!(
133                                "factory_projection_wire_wrap: niche unsigned field `{name}` \
134                                 has no declared sentinel"
135                            )
136                        });
137                        (
138                            KtType::long(),
139                            format!("if ({name} == {sentinel}) null else {name}.toULong()"),
140                        )
141                    }
142                    // Plain `Option<u64>` has no spare bit pattern, so its
143                    // primitive wire is boxed and JVM null represents `None`.
144                    NullableKind::Boxed => {
145                        (KtType::long().nullable(), format!("{name}?.toULong()"))
146                    }
147                },
148            }
149        }
150        Iterable(_) => panic!(
151            "factory_projection_wire_wrap: collection (`Vec<projection>`) struct fields are not \
152             supported by the fromParts factory (field `{name}`)"
153        ),
154    }
155}

157/// True for the Kotlin types that map to JVM **primitives** (never null over
158/// the JNI boundary). Used to decide which flattened `Option<nested>` leaf
159/// params must be made nullable in the parent factory signature.
160pub(crate) fn is_kotlin_primitive_ty(t: &KtType) -> bool {
161    !t.is_nullable()
162        && t.leaf_name().is_some_and(|n| {
163            matches!(
164                n,
165                "Long" | "Int" | "Boolean" | "Double" | "Float" | "Byte" | "Short" | "Char"
166            )
167        })
168}

174/// Recursively build the Kotlin `fromParts` factory for a data class — the
175/// mirror of the native `flatten_struct_encode` (in the [`jni`](super)
176/// module). Both walk the same [`build_struct_plan`], so the leaf order and
177/// slot types agree by construction.
178/// Returns `(params, reconstruct, mints_handle)`:
179/// * `params` — the flattened `(name, kotlin_type)` list (one per transitive
180///   leaf wire; nested data-class fields are inlined, `Option<nested>` prepends
181///   a `…__present: Boolean` flag). Order/types match the native call's JVM
182///   descriptor positionally.
183/// * `reconstruct` — the Kotlin expression building this struct:
184///   `Class(<part per constructor field>)`, where a nested field reconstructs
185///   via `Child.fromParts(<child param names>)` (`if (present) … else null` when
186///   optional) and a leaf reconstructs with its wrap.
187/// * `mints_handle` — whether any transitive leaf is an opaque handle, i.e.
188///   whether this factory takes a raw native pointer and so needs the
189///   raw-pointer guard (see [`plan_mints_handle`]).
190#[allow(clippy::too_many_arguments)]
191pub(crate) fn flatten_struct_factory(
192    ext: &Declarations,
193    registry: &Registry,
194    s: &prebindgen_registry::flat::Struct,
195    prefix: &str,
196    class_name: &str,
197    imports: &mut BTreeSet<String>,
198    depth: usize,
199) -> Option<StructFactory> {
200    let plan = ext.struct_plan(registry, s, depth)?;
201    let (params, reconstruct) = factory_from_plan(&plan, prefix, class_name, imports)?;
202    Some((params, reconstruct, plan_mints_handle(&plan)))
203}

205/// Walk a [`StructPlan`] emitting the Kotlin `fromParts` side: the flattened
206/// factory params and the reconstruct expression.
207fn factory_from_plan(
208    plan: &StructPlan,
209    prefix: &str,
210    class_name: &str,
211    imports: &mut BTreeSet<String>,
212) -> Option<(Vec<(String, KtType)>, String)> {
213    let mut params: Vec<(String, KtType)> = Vec::new();
214    let mut parts: Vec<String> = Vec::new();
215
216    for f in &plan.fields {
217        let camel = mangle_kotlin_ident(&kt_snake_to_camel(&f.fname.to_string()));
218        let base = if prefix.is_empty() {
219            camel.clone()
220        } else {
221            format!("{prefix}_{camel}")
222        };
223        let (p, part) = factory_field(&f.kind, &base, imports)?;
224        params.extend(p);
225        parts.push(part);
226    }
227
228    let reconstruct = format!("{class_name}({})", parts.join(", "));
229    Some((params, reconstruct))
230}

232/// Declare the `fromParts` slots of ONE value position and the expression
233/// that rebuilds it — the Kotlin mirror of
234/// [`encode_field`](super::encode_field). A sum's payloads go through here
235/// too, so a payload and a struct field of the same type declare the same
236/// slot on both sides.
237fn factory_field(
238    kind: &PlanFieldKind,
239    base: &str,
240    imports: &mut BTreeSet<String>,
241) -> Option<(Vec<(String, KtType)>, String)> {
242    let mut params: Vec<(String, KtType)> = Vec::new();
243    let mut parts: Vec<String> = Vec::new();
244    let base = base.to_string();
245    {
246        let f_kind = kind;
247        match f_kind {
248            // Projection leaf (handle / `ULong`).
249            PlanFieldKind::Projection { proj, fqn, .. } => {
250                let short = register_fqn(fqn, imports);
251                let (wire_ty, wrap) = factory_projection_wire_wrap(proj, &short, &base);
252                params.push((base.clone(), wire_ty));
253                parts.push(wrap);
254            }
255            // Enum leaf → `Int`, rebuilt via `Enum.fromInt(i)` (raw text: the
256            // enum's short name, its FQN collected as a body import).
257            PlanFieldKind::Enum { kotlin, .. } => {
258                let short = register_fqn(kotlin.leaf_name()?, imports);
259                params.push((base.clone(), KtType::int()));
260                parts.push(format!("{short}.fromInt({base})"));
261            }
262            // `Option<enum>` leaf: prefer the primitive niche selected by the
263            // registry; retain nullable boxed-Int rendering as the fallback.
264            PlanFieldKind::OptionEnum { kotlin, niche, .. } => {
265                let short = register_fqn(kotlin.leaf_name()?, imports);
266                match niche {
267                    Some(niche) => {
268                        params.push((base.clone(), KtType::int()));
269                        parts.push(format!(
270                            "if ({base} == {niche}) null else {short}.fromInt({base})"
271                        ));
272                    }
273                    None => {
274                        params.push((base.clone(), KtType::int().nullable()));
275                        parts.push(format!("{base}?.let {{ {short}.fromInt(it) }}"));
276                    }
277                }
278            }
279            // Data-carrying enum — the tag slot plus every variant's group
280            // side by side, rebuilt by an inlined `when` (no JNI crossing,
281            // exactly like a nested data class).
282            PlanFieldKind::Sum {
283                kotlin_fqn,
284                optional,
285                variants,
286                ..
287            } => {
288                let iface_short = register_fqn(kotlin_fqn, imports);
289                let flag = format!("{base}__present");
290                if *optional {
291                    params.push((flag.clone(), KtType::boolean()));
292                }
293                params.push((format!("{base}__tag"), KtType::int()));
294
295                let mut arms: Vec<String> = Vec::new();
296                for (tag, v) in variants.iter().enumerate() {
297                    // Each group's slots are declared once, in variant order,
298                    // and are INERT whenever another variant is live — so the
299                    // nullability rule below applies to every group, not just
300                    // to an optional one.
301                    let mut fwd: Vec<String> = Vec::new();
302                    for pf in &v.fields {
303                        let slot = format!("{base}_{}", pf.slot);
304                        let (group_params, part) = factory_field(&pf.kind, &slot, imports)?;
305                        fwd.push(nullable_group_part(&mut params, group_params, part));
306                    }
307                    let ctor = if v.fields.is_empty() {
308                        format!("{iface_short}.{}", v.kotlin_name)
309                    } else {
310                        format!("{iface_short}.{}({})", v.kotlin_name, fwd.join(", "))
311                    };
312                    arms.push(format!("{tag} -> {ctor}"));
313                }
314                let when = format!(
315                    "when ({base}__tag) {{ {}; else -> throw IllegalArgumentException(\"{}: \
316                     invalid tag ${base}__tag\") }}",
317                    arms.join("; "),
318                    iface_short,
319                );
320                parts.push(if *optional {
321                    format!("if ({flag}) {when} else null")
322                } else {
323                    when
324                });
325            }
326            // Nested data-class field — inline its leaves and reconstruct via
327            // the child's own `fromParts` (in bytecode, no JNI crossing).
328            PlanFieldKind::Nested {
329                optional,
330                child_fqn,
331                plan: child,
332            } => {
333                let child_fqn = child_fqn.as_ref()?;
334                let child_short = register_fqn(child_fqn, imports);
335                let (child_params, _child_reconstruct) =
336                    factory_from_plan(child, &base, &child_short, imports)?;
337                let child_names = child_params
338                    .iter()
339                    .map(|(n, _)| n.clone())
340                    .collect::<Vec<_>>()
341                    .join(", ");
342                if !*optional {
343                    params.extend(child_params);
344                    parts.push(format!("{child_short}.fromParts({child_names})"));
345                } else {
346                    // `Option<nested>`: the parent receives default-null object wires
347                    // for the child's leaves when absent (the native `None` arm), so
348                    // every object-typed child param must be NULLABLE in the parent
349                    // signature. Inside the `if (present)` guard the values are
350                    // non-null again, so forward them to the child's (non-null)
351                    // `fromParts` with `!!`. Primitive params (Long/Int/Boolean)
352                    // can't be null and are forwarded as-is; already-nullable params
353                    // stay nullable.
354                    let flag = format!("{base}__present");
355                    let mut fwd_names: Vec<String> = Vec::with_capacity(child_params.len());
356                    params.push((flag.clone(), KtType::boolean()));
357                    for (n, t) in &child_params {
358                        if is_kotlin_primitive_ty(t) || t.is_nullable() {
359                            params.push((n.clone(), t.clone()));
360                            fwd_names.push(n.clone());
361                        } else {
362                            params.push((n.clone(), t.clone().nullable()));
363                            fwd_names.push(format!("{n}!!"));
364                        }
365                    }
366                    parts.push(format!(
367                        "if ({flag}) {child_short}.fromParts({}) else null",
368                        fwd_names.join(", ")
369                    ));
370                }
371            }
372            // Leaf primitive / object (string, byte array, Vec, …) — forwarded
373            // unchanged to the constructor. Full-FQN param type; the
374            // render-time `ImportSet` shortens it.
375            PlanFieldKind::Leaf {
376                kotlin, nullable, ..
377            } => {
378                let ty = if *nullable {
379                    kotlin.clone().nullable()
380                } else {
381                    kotlin.clone()
382                };
383                params.push((base.clone(), ty));
384                parts.push(base);
385            }
386        }
387    }
388
389    debug_assert_eq!(
390        parts.len(),
391        1,
392        "factory_field must yield exactly one reconstruct expression"
393    );
394    Some((params, parts.remove(0)))
395}

397/// Declare one variant group's slots and return the expression that forwards
398/// them into the variant constructor.
399///
400/// A group is **inert** whenever another variant is live, and the Rust
401/// encoder fills an inert object slot with `JObject::null()`
402/// ([`primitive_default_for_descriptor`](super::primitive_default_for_descriptor)).
403/// A JVM `null` handed to a non-null Kotlin parameter throws inside the
404/// intrinsic null-check, before any generated code runs — so every
405/// object-shaped group slot must be declared nullable and forwarded `!!`
406/// inside its own (live) arm, where it is non-null again. Primitives take
407/// their `0`/`false` default and need no such treatment.
408///
409/// This is the N=1 `Option<nested>` rule (see the `Nested { optional }` arm)
410/// generalized to N groups; both call it so the two paths cannot drift.
411fn nullable_group_part(
412    params: &mut Vec<(String, KtType)>,
413    group_params: Vec<(String, KtType)>,
414    part: String,
415) -> String {
416    let mut part = part;
417    for (n, t) in group_params {
418        if is_kotlin_primitive_ty(&t) || t.is_nullable() {
419            params.push((n, t));
420        } else {
421            // The reconstruct expression references the slot by name; the
422            // live arm re-asserts non-nullness there.
423            part = replace_ident(&part, &n, &format!("{n}!!"));
424            params.push((n, t.nullable()));
425        }
426    }
427    part
428}

430/// Replace whole-identifier occurrences of `from` in a rendered Kotlin
431/// expression. Whole-identifier: a match must not be flanked by identifier
432/// characters, so rewriting `x` never touches `x_2` or `ax` — slot names of
433/// one group share a prefix (`exact_v0`, `exact_v01`), and a substring
434/// replace would corrupt them.
435///
436/// The left-hand character is taken from what has already been **written**,
437/// never from the remaining input: `rest` advances past a rejected occurrence,
438/// so by the next candidate the character preceding it can already have been
439/// consumed. Re-slicing `rest` there loses it and turns two adjacent
440/// occurrences (`"axx"`, rewriting `x`) into one accepted match.
441fn replace_ident(haystack: &str, from: &str, to: &str) -> String {
442    let is_ident_char = |c: char| c == '_' || c.is_alphanumeric();
443    let mut out = String::with_capacity(haystack.len());
444    let mut rest = haystack;
445    while let Some(pos) = rest.find(from) {
446        out.push_str(&rest[..pos]);
447        let before_ok = out.chars().next_back().is_none_or(|c| !is_ident_char(c));
448        let after = &rest[pos + from.len()..];
449        let after_ok = after.chars().next().is_none_or(|c| !is_ident_char(c));
450        if before_ok && after_ok {
451            out.push_str(to);
452        } else {
453            out.push_str(from);
454        }
455        rest = after;
456    }
457    out.push_str(rest);
458    out
459}

461/// Render the Kotlin `close()` expression for a handle `receiver` through
462/// the folded [`FoldStrategy`] layers. Fresh lambda variable per nesting
463/// level avoids `it` shadowing; the common single-layer cases are
464/// special-cased for readable output (`x?.close()`, `x.forEach { it.close() }`).
465pub(crate) fn render_handle_close(strategy: &crate::jni::FoldStrategy, receiver: &str) -> String {
466    use prebindgen_registry::shape::Shape::*;
467    fn go(strategy: &crate::jni::FoldStrategy, receiver: &str, depth: usize) -> String {
468        match strategy {
469            Base => format!("{receiver}.close()"),
470            // The Kotlin-side receiver is already nullable (`handle_kt_type`
471            // emits `T?` for both niche and boxed kinds), so `?.close()` covers
472            // both wire representations.
473            Optional(_, inner) => match &**inner {
474                Base => format!("{receiver}?.close()"),
475                _ => {
476                    let v = format!("e{depth}");
477                    format!("{receiver}?.let {{ {v} -> {} }}", go(inner, &v, depth + 1))
478                }
479            },
480            Iterable(inner) => {
481                let v = format!("e{depth}");
482                format!(
483                    "{receiver}.forEach {{ {v} -> {} }}",
484                    go(inner, &v, depth + 1)
485                )
486            }
487        }
488    }
489    go(strategy, receiver, 0)
490}

492/// Fold the projection wrap call `W(receiver)` through the
493/// [`FoldStrategy`] layers:
494/// * `Direct`         → `W(x)`
495/// * `Nullable{Boxed}` → `x?.let { W(it) }` (JVM-null at the wire)
496/// * `Nullable{Niche}` over a primitive wire (e.g. `jlong`) →
497///   `x.let { if (it == <sentinel>) null else W(it) }`
498/// * `Nullable{Niche}` over an object wire (e.g. `JByteArray`) →
499///   `x?.let { W(it) }` (the wire is already a nullable reference)
500/// * `Iterable`       → `x.map { W(it) }`
501///
502/// `niche_sentinel` is the Kotlin literal to compare against for the
503/// `Niche+primitive` arm (e.g. `"0L"` for `jlong`-wired handles). When the
504/// wire is object-shaped the sentinel is unused — `null` is the wire-level
505/// representation and `?.let` is a no-cost null check.
506pub(crate) fn fold_projection_wrap(
507    strategy: &crate::jni::FoldStrategy,
508    receiver: &str,
509    kind: &crate::jni::ProjectionKind,
510    wrap_class: &str,
511    niche_sentinel: Option<&str>,
512) -> String {
513    use prebindgen_registry::shape::Shape::*;
514
515    use crate::jni::NullableKind;
516    fn go(
517        s: &crate::jni::FoldStrategy,
518        r: &str,
519        kind: &crate::jni::ProjectionKind,
520        w: &str,
521        sentinel: Option<&str>,
522        depth: usize,
523    ) -> String {
524        match s {
525            Base => projection_wrap_expr(kind, w, r),
526            Optional(nullable_kind, inner) => match (nullable_kind, &**inner) {
527                // Primitive-wired niche → can't carry null on the wire, so
528                // compare against the sentinel and synthesize null on the
529                // Kotlin side.
530                (NullableKind::Niche, Base) if sentinel.is_some() => {
531                    let s = sentinel.unwrap();
532                    let wrapped = projection_wrap_expr(kind, w, "it");
533                    format!("{r}.let {{ if (it == {s}) null else {wrapped} }}")
534                }
535                // Object-wired niche or fully boxed Nullable → `?.let { W(it) }`.
536                (_, Base) => {
537                    let wrapped = projection_wrap_expr(kind, w, "it");
538                    format!("{r}?.let {{ {wrapped} }}")
539                }
540                // Deeper nesting. The niche/boxed distinction is only
541                // observable at the outermost layer covering a `Direct`
542                // leaf; intermediate layers (nullable-of-iterable etc.)
543                // can keep the simple form because Kotlin's `?.` chain
544                // already represents the layered null.
545                _ => {
546                    let v = format!("e{depth}");
547                    format!(
548                        "{r}?.let {{ {v} -> {} }}",
549                        go(inner, &v, kind, w, sentinel, depth + 1)
550                    )
551                }
552            },
553            Iterable(inner) => match &**inner {
554                Base => {
555                    let wrapped = projection_wrap_expr(kind, w, "it");
556                    format!("{r}.map {{ {wrapped} }}")
557                }
558                _ => {
559                    let v = format!("e{depth}");
560                    format!(
561                        "{r}.map {{ {v} -> {} }}",
562                        go(inner, &v, kind, w, sentinel, depth + 1)
563                    )
564                }
565            },
566        }
567    }
568    go(strategy, receiver, kind, wrap_class, niche_sentinel, 0)
569}

571/// JNI extern's declared Kotlin wire-return for a projection. The leaf wire
572/// is the inner converter's destination Kotlin name — `Long` for both
573/// projection kinds (a handle's pointer, a `ULong`'s raw bit pattern). The
574/// fold honours
575/// [`NullableKind`] so the declared wire matches the runtime ABI:
576/// `Niche+primitive` keeps the layer non-nullable on the wire (the sentinel
577/// represents null); `Niche+object` and `Boxed` add `?`.
578pub(crate) fn projection_wire_return(proj: &crate::jni::Projection) -> KtType {
579    use crate::jni::{FoldStrategy, NullableKind, ProjectionKind};
580    let (inner_wire, inner_is_primitive) = match proj.kind {
581        ProjectionKind::Handle => (KtType::long(), true),
582        ProjectionKind::Unsigned64 => (KtType::long(), true),
583    };
584    fold_shape(
585        &proj.strategy,
586        &|| inner_wire.clone(),
587        &|inner, kind, inner_strategy| {
588            // A niche layer over a primitive wire keeps the wire non-nullable —
589            // the sentinel value is the null representation. Object-wired niches
590            // and full-boxed Nullables both add `?` (JVM null on the reference).
591            match (kind, inner_strategy) {
592                (NullableKind::Niche, FoldStrategy::Base) if inner_is_primitive => inner,
593                _ => inner.nullable(),
594            }
595        },
596        &|inner| KtType::generic("List", [inner]),
597    )
598}

600/// Kotlin null-sentinel literal for the *leaf wire* of a projection. Read
601/// at the wrapper-body call site and forwarded to [`fold_projection_wrap`];
602/// `None` when the leaf wire has no primitive null sentinel, where
603/// `?.let { }` covers the JVM-null case directly.
604pub(crate) fn projection_leaf_sentinel(proj: &crate::jni::Projection) -> Option<String> {
605    if let Some(sentinel) = proj.niche_sentinels.first() {
606        return Some(sentinel.clone());
607    }
608    use crate::jni::ProjectionKind;
609    let leaf_wire: syn::Type = match proj.kind {
610        ProjectionKind::Handle => syn::parse_quote!(jni::sys::jlong),
611        // No niche exists for `u64`; `Option<u64>` uses the boxed path, so a
612        // primitive sentinel must never be synthesized.
613        ProjectionKind::Unsigned64 => return None,
614    };
615    kotlin_null_sentinel(&leaf_wire).map(|s| s.to_string())
616}

618/// The sentinel a **wrap** should test, given the projection and whether an
619/// ancestor makes the leaf nullable — the one rule both derivations of that
620/// wrap read (`unfold_leaf_kt` in `render.rs`, `leaf_iface_param` in
621/// `iface.rs`), so it cannot drift between them again (#142).
622///
623/// A sentinel is the leaf's **own** `None` representation, so it belongs to the
624/// leaf's own type and to nothing above it. Two independent facts meet here and
625/// only the first one answers:
626///
627/// * the leaf's type carries a niche — `Option<Duration>` over a bounded
628///   `convert!`, whose strategy is `Optional(Niche, _)`. Its `None` IS the
629///   sentinel, and the test stays whatever the ancestor does.
630/// * an **ancestor** can be absent (a conditional value form, an
631///   `Option<sum>`/`Option<nested>` field). That widens the wire — the Rust
632///   side boxes any nullable leaf, see
633///   [`leaf_is_prim`](crate::jni::emit::leaf_is_prim) — and `?.` alone carries
634///   it. It grants no sentinel.
635///
636/// [`projection_leaf_sentinel`] answers off `niche_sentinels`, which
637/// `attach_domain_sentinels` puts on the **bare** type's converter as well as
638/// the `Option` one. Asking it without the `Base` check therefore handed a
639/// sentinel to a leaf that has no niche encoding at all, splicing
640/// `?.let { if (it == -1L) … }` into a wrap whose own encoder can never emit
641/// `-1`. Harmless at runtime — the value is outside the declared range — and
642/// wrong on its face.
643pub(crate) fn wrap_sentinel(proj: &crate::jni::Projection, nullable: bool) -> Option<String> {
644    // `Base` means the leaf has no `Option` of its own: whatever absence it can
645    // express is the ancestor's, and `?.` already expresses that.
646    if nullable && matches!(proj.strategy, crate::jni::FoldStrategy::Base) {
647        return None;
648    }
649    projection_leaf_sentinel(proj)
650}

652/// Kotlin literal for the null-sentinel of a primitive wire — used by
653/// [`fold_projection_wrap`] when a `Niche` layer covers a primitive wire and
654/// can't carry JVM null. Mirrors `jni_field_access`'s primitive descriptors.
655/// Returns `None` for object-shaped wires (where JVM null *is* the null
656/// representation and `?.let` is the right pattern).
657pub(crate) fn kotlin_null_sentinel(wire: &syn::Type) -> Option<&'static str> {
658    let (_, _, is_object) = crate::jni::wire_access::jni_field_access(wire)?;
659    if is_object {
660        return None;
661    }
662    let syn::Type::Path(tp) = wire else {
663        return None;
664    };
665    let last = tp.path.segments.last()?;
666    Some(match last.ident.to_string().as_str() {
667        "jlong" => "0L",
668        "jint" | "jshort" | "jbyte" | "jchar" => "0",
669        "jfloat" => "0.0f",
670        "jdouble" => "0.0",
671        "jboolean" => "false",
672        _ => return None,
673    })
674}

676/// Shorten a class FQN to its simple name for use in **raw body text** (a
677/// `fromParts` reconstruct fragment: `Child.fromParts(…)`, `Enum.fromInt(…)`,
678/// `ZenohId(bytes)`), registering the FQN into `used` — the body `Code`'s own
679/// import list. A non-dotted name (a Kotlin builtin like `String`) needs no
680/// import and passes through. Signature types are NOT shortened this way —
681/// those are full-FQN `KtType`s in the AST, shortened + imported by the
682/// render-time `ImportSet`.
683pub(crate) fn register_fqn(fqn: &str, used: &mut BTreeSet<String>) -> String {
684    if fqn.contains('.') {
685        used.insert(fqn.to_string());
686        fqn.rsplit('.').next().unwrap_or(fqn).to_string()
687    } else {
688        fqn.to_string()
689    }
690}

692#[cfg(test)]
693mod replace_ident_tests {
694    use super::replace_ident;
695
696    /// The property the function exists for: a slot name is rewritten only as a
697    /// whole identifier, so the `!!` never lands on a longer name that merely
698    /// starts (or ends) with it.
699    #[test]
700    fn only_whole_identifiers_are_rewritten() {
701        assert_eq!(replace_ident("x", "x", "x!!"), "x!!");
702        assert_eq!(replace_ident("x_2", "x", "x!!"), "x_2");
703        assert_eq!(replace_ident("ax", "x", "x!!"), "ax");
704        assert_eq!(
705            replace_ident("Reading.Exact(exact_v0)", "exact_v0", "exact_v0!!"),
706            "Reading.Exact(exact_v0!!)"
707        );
708        // A prefix of a longer slot name of the same group is left alone.
709        assert_eq!(
710            replace_ident("f(exact_v0, exact_v01)", "exact_v0", "exact_v0!!"),
711            "f(exact_v0!!, exact_v01)"
712        );
713    }
714
715    /// A rejected occurrence must not consume the context of the next one: with
716    /// the left character re-sliced out of the remaining input, `"axx"` rewrote
717    /// its second `x` — the first having been dropped from `rest` along with the
718    /// `a` that disqualified it.
719    #[test]
720    fn a_rejected_match_keeps_its_left_context() {
721        assert_eq!(replace_ident("axx", "x", "x!!"), "axx");
722        assert_eq!(replace_ident("xxx", "x", "x!!"), "xxx");
723        assert_eq!(replace_ident("x ax", "x", "x!!"), "x!! ax");
724        // Every occurrence of a genuinely standalone identifier is rewritten.
725        assert_eq!(replace_ident("x + x", "x", "x!!"), "x!! + x!!");
726    }
