1//! What the generated C-facing Rust file is assembled from.
2//!
3//! Each final artifact is planned when the C generation plan is frozen and
4//! rendered at the final writing boundary, so nothing it needs is looked up in
5//! a live registry while the file is assembled. A wrapper — the exported
6//! `extern "C"` function one declared `#[prebindgen]` function becomes — reads
7//! its own frozen state plus the boundary sites of its own function, which the
8//! same frozen plan holds.
9
10use std::rc::Rc;

12use prebindgen_registry::{
13    generation::GenerationPlan,
14    write::{ArtifactKey, RustArtifact},
15};

19/// One final artifact of the generated Rust file.
20pub(crate) enum CFinalArtifact {
21    /// A private converter, carrying one value across the boundary.
22    Converter(Box<crate::chain::CFunction>),
23    /// The exported wrapper for one declared `#[prebindgen]` function.
24    Wrapper(Box<CWrapper>),
25    /// One captured constant, re-stated as an alias to its source.
26    Const(Box<CConst>),
27    /// The memory helpers the generated layer hands `char*` and array blocks
28    /// to C through.
29    Memory(Box<CMemory>),
30    /// The array builder, which copies a `Vec<T>` into a C block.
31    ArrayBuilder,
32    /// One `#[repr(C)]` mirror of a declared data struct.
33    DataStruct(Box<CDataStruct>),
34    /// One `#[repr(C)]` mirror of a declared fieldless enum.
35    Enum(Box<CEnum>),
36    /// One reserved representation value a generated sum-type ABI uses.
37    DomainConstant(Box<CDomainConstant>),
38    /// One artifact the registry generation plan already holds: an opaque
39    /// handle, a value-opaque, a tagged union, or a callback.
40    Planned(Box<CPlanned>),
41}
42
43impl RustArtifact for CFinalArtifact {
44    fn key(&self) -> ArtifactKey {
45        match self {
46            Self::Converter(converter) => converter.key(),
47            Self::Wrapper(wrapper) => wrapper.key(),
48            Self::Const(constant) => constant.key(),
49            Self::Memory(memory) => memory.key(),
50            Self::ArrayBuilder => array_builder_key(),
51            Self::DataStruct(item) => item.key(),
52            Self::Enum(item) => item.key(),
53            Self::DomainConstant(item) => item.key(),
54            Self::Planned(item) => item.key(),
55        }
56    }
57
58    fn provides(&self) -> Vec<ArtifactKey> {
59        match self {
60            Self::Planned(item) => item.provides(),
61            _ => vec![self.key()],
62        }
63    }
64
65    fn calls(&self) -> Vec<ArtifactKey> {
66        match self {
67            Self::Converter(converter) => converter.calls(),
68            Self::Wrapper(wrapper) => wrapper.calls(),
69            Self::Const(constant) => constant.calls(),
70            Self::Memory(memory) => memory.calls(),
71            // The block it fills is malloc'd, and freed by the universal freer.
72            Self::ArrayBuilder => vec![memory_key()],
73            Self::DataStruct(item) => item.calls(),
74            Self::Enum(item) => item.calls(),
75            Self::DomainConstant(item) => item.calls(),
76            Self::Planned(item) => item.calls(),
77        }
78    }
79
80    fn render(&self, emit: &prebindgen_registry::RustWriter) -> Vec<syn::Item> {
81        match self {
82            Self::Converter(converter) => converter.render(emit),
83            Self::Wrapper(wrapper) => wrapper.render(emit),
84            Self::Const(constant) => constant.render(emit),
85            Self::Memory(memory) => memory.render(emit),
86            Self::ArrayBuilder => CMemory::render_array_builder(),
87            Self::DataStruct(item) => item.render(emit),
88            Self::Enum(item) => item.render(emit),
89            Self::DomainConstant(item) => item.render(emit),
90            Self::Planned(item) => item.render(emit),
91        }
92    }
93}

95/// One declared function's exported wrapper, frozen at the end of resolution.
96pub(crate) struct CWrapper {
97    /// The plan the wrapper's own boundary sites are read from.
98    generation: Rc<GenerationPlan<crate::compile::CRepresentation>>,
99    /// The `#[prebindgen]` function this wrapper exports.
100    function: prebindgen_registry::flat::Function,
101    /// The `#[no_mangle]` symbol the wrapper is exported under.
102    symbol: syn::Ident,
103    /// Path the wrapper calls the source function by.
104    call_path: syn::Path,
105    /// Whether `.panic()` was declared, which is what allows a fallible
106    /// conversion in a function that returns no `Result`.
107    panic: bool,
108    /// Per parameter, the resource its argument names and how the call uses
109    /// it — `None` for a parameter that names no single owned resource. The
110    /// alias preflight compares these; declaring which types are opaque is a
111    /// binding question, so it is answered while planning.
112    alias_slots: Vec<Option<(TypeKey, AliasAccess)>>,
113}
114
115impl CWrapper {
116    /// Plan the wrapper for one declared function.
117    pub(crate) fn new(
118        decls: &CbindgenBuilder,
119        generation: &Rc<GenerationPlan<crate::compile::CRepresentation>>,
120        function: &prebindgen_registry::flat::Function,
121    ) -> Self {
122        let name = &function.name;
123        if let Some((_, err)) = function.ret.fallible_parts() {
124            let err_key = err.key();
125            assert!(
126                decls.error.contains(&err_key),
127                "Cbindgen: function `{}` returns `Result<_, {}>` but `{}` is not a \
128                 declared error type — add `.data_struct({}).error()`",
129                name,
130                err_key,
131                err_key,
132                err_key,
133            );
134        }
135        Self {
136            generation: Rc::clone(generation),
137            function: function.clone(),
138            symbol: decls.fn_symbol(name),
139            call_path: decls.src_fn(name),
140            panic: decls.functions.get(name).is_some_and(|cfg| cfg.panic),
141            alias_slots: function
142                .params
143                .iter()
144                .map(|param| decls.alias_slot_of(&param.ty))
145                .collect(),
146        }
147    }

149    /// One boundary site of this wrapper's own function, when the function has
150    /// one in that role. A void return and an infallible function have no
151    /// return or error site.
152    fn site_if_planned(
153        &self,
154        role: prebindgen_registry::recipe::Role,
155    ) -> Option<&prebindgen_registry::generation::SitePlan<crate::compile::CRepresentation>> {
156        let id = prebindgen_registry::generation::SiteId::new(prebindgen_registry::recipe::Site {
157            owner: self.function.name.clone(),
158            role,
159        });
160        self.generation.site(&id)
161    }
162
163    /// One boundary site of this wrapper's own function.
164    fn site(
165        &self,
166        role: prebindgen_registry::recipe::Role,
167    ) -> &prebindgen_registry::generation::SitePlan<crate::compile::CRepresentation> {
168        let id = prebindgen_registry::generation::SiteId::new(prebindgen_registry::recipe::Site {
169            owner: self.function.name.clone(),
170            role,
171        });
172        let site = self
173            .generation
174            .site(&id)
175            .unwrap_or_else(|| panic!("C generation plan has no site {id}"));
176        assert!(
177            matches!(
178                site.cleanup(),
179                prebindgen_registry::generation::Cleanup::None
180            ),
181            "ordinary C sites cannot carry a deferred cleanup operation"
182        );
183        site
184    }

186    /// Assemble this wrapper's `#[no_mangle] extern "C"` function.
187    fn render_fn(&self, emit: &prebindgen_registry::RustWriter) -> syn::ItemFn {
188        let f = &self.function;
189        let orig = &f.name;
190        let call_path = &self.call_path;
191        let sym = &self.symbol;
192
193        // The ELEMENT: a signature is a parameter list and a return, both
194        // already classified. An elided return is `TypeKind::Unit`, which is
195        // the `ReturnType::Default` arm this used to write.
196
197        let has_fallible_input = f.params.iter().enumerate().any(|(index, _)| {
198            self.site(prebindgen_registry::recipe::Role::Param { index })
199                .abi()
200                .payload()
201                .failure()
202                == prebindgen_registry::generation::Failure::Fallible
203        });
204
205        // Peel an outer `Result<_, E>`; `value_ty` is the success/return value.
206        // Off `TypeKind::Fallible`, where `result_parts` found the `Result` in a
207        // path first — and both sides come back as readings, so everything
208        // downstream of here reads too.
209        let (value_ty, err_reading) = match f.ret.fallible_parts() {
210            Some((ok, e)) => (ok, Some(e)),
211            None => (&f.ret, None),
212        };
213        let value_site = (!matches!(value_ty.kind(), TypeKind::Unit))
214            .then(|| self.site(prebindgen_registry::recipe::Role::Return));
215        let has_fallible_output = value_site.is_some_and(|site| {
216            site.abi().payload().failure() == prebindgen_registry::generation::Failure::Fallible
217        });
218
219        // Error wiring: that the error type is declared via `.error()` was
220        // checked when the wrapper was planned.
221        let err_bits = err_reading.map(|err| {
222            let site = self.site(prebindgen_registry::recipe::Role::Error);
223            let crate::compile::CValue::Direct {
224                wire, converter, ..
225            } = site.abi().payload()
226            else {
227                panic!("C error site must have one direct wire");
228            };
229            (
230                wire.clone(),
231                converter.ident(emit),
232                emit.emit_source_type(err),
233            )
234        });
235
236        // No `Result` channel ⇒ a fallible input must be declared `.panic()`.
237        if err_reading.is_none() {
238            assert!(
239                !(has_fallible_input || has_fallible_output) || self.panic,
240                "Cbindgen: function `{}` has a fallible binding conversion but does not \
241                 return `Result`; add \
242                 `.panic()` after its `.function(...)` declaration to allow aborting \
243                 on the internal error, or change its signature",
244                orig,
245            );
246        }
247
248        // Structural lowering of the (present/ok) value, then the null-niche rule:
249        //   * Result + a free pointer niche  → NULL marks `Err` (value in-band);
250        //   * Result without a free niche     → `bool` status, value to out-params;
251        //   * no Result                       → field 0 is the C return, rest out.
252        let shape = value_site.map_or(
253            FrozenValueLayout {
254                fields: Vec::new(),
255                niches: Niches::empty(),
256            },
257            |site| FrozenValueLayout {
258                fields: site.abi().payload().fields(),
259                niches: site.abi().payload().effective_niches(),
260            },
261        );
262        let result_slot = shape.niches.clone().carve().map(|(slot, _)| slot);
263        let result_in_band = err_reading.is_some() && result_slot.is_some();
264        let field0_is_return = result_in_band || err_reading.is_none();
265
266        // Partition fields into the (optional) C return value + out-parameters,
267        // and pick C names for the out-params (see `out_param_name`).
268        let mut targets: Vec<TokenStream> = Vec::new();
269        let mut out_fields: Vec<&WireField> = Vec::new();
270        // `field0_wire` is the wire of the value's primary field when that field
271        // is carried by the C return slot (modes A/D); `None` for mode B and unit.
272        let field0_wire: Option<syn::Type> = if field0_is_return {
273            shape.fields.first().map(|f| f.wire.clone())
274        } else {
275            None
276        };
277        if field0_is_return {
278            if !shape.fields.is_empty() {
279                targets.push(quote!(__ret));
280                out_fields.extend(shape.fields[1..].iter());
281            }
282        } else {
283            out_fields.extend(shape.fields.iter());
284        }
285        let prefixed = out_fields.iter().any(|wf| wf.suffix.is_empty());
286        let out_names: Vec<syn::Ident> = out_fields
287            .iter()
288            .map(|wf| out_param_name(wf.suffix, prefixed))
289            .collect();
290        for name in &out_names {
291            targets.push(quote!(*#name));
292        }
293        let out_param_decls: Vec<TokenStream> = out_fields
294            .iter()
295            .zip(&out_names)
296            .map(|(wf, name)| {
297                let wire = &wf.wire;
298                quote!(#name: *mut #wire)
299            })
300            .collect();
301
302        // C wrapper return type: the payload's field 0 (modes A/D), `bool` status
303        // (mode B), or `void` (a unit value with no `Result`).
304        let c_return: Option<syn::Type> = if field0_is_return {
305            field0_wire.clone()
306        } else {
307            Some(syn::parse_quote!(bool))
308        };
309
310        // Input decode: route a fallible-input failure to the error out-param
311        // (with the wrapper's fail value) when there is a `Result`, else panic.
312        let fail_return = if result_in_band {
313            let slot = result_slot.as_ref().expect("in-band result has a niche");
314            let value = &slot.value;
315            quote!(#value)
316        } else {
317            quote!(false)
318        };
319        let mut planned_routes = f
320            .params
321            .iter()
322            .enumerate()
323            .filter_map(|(index, _)| {
324                self.site(prebindgen_registry::recipe::Role::Param { index })
325                    .failure_route()
326            })
327            .chain(
328                value_site
329                    .into_iter()
330                    .filter_map(|site| site.failure_route()),
331            );
332        let planned_route = planned_routes.next();
333        assert!(
334            planned_routes.all(|route| Some(route) == planned_route),
335            "C function sites disagree on their frozen failure route"
336        );
337        match planned_route {
338            Some(crate::compile::CFailureRoute::Panic) => {
339                assert!(
340                    err_bits.is_none(),
341                    "panic route attached to a Result function"
342                );
343            }
344            Some(crate::compile::CFailureRoute::Error(error_site)) => {
345                assert!(
346                    err_bits.is_some(),
347                    "error route attached without a Result channel"
348                );
349                let expected = prebindgen_registry::generation::SiteId::new(
350                    prebindgen_registry::recipe::Site {
351                        owner: orig.clone(),
352                        role: prebindgen_registry::recipe::Role::Error,
353                    },
354                );
355                assert_eq!(
356                    error_site, &expected,
357                    "failure route names the wrong error site"
358                );
359            }
360            None => {}
361        }
362        let input_route = match &err_bits {
363            Some((_, e_conv, e_ty_src)) => ErrRoute::Result {
364                e_conv,
365                e_ty_src: e_ty_src.clone(),
366                fail_return: fail_return.clone(),
367            },
368            None => ErrRoute::Panic,
369        };
370        let (in_params, decodes, call_args) = self.planned_inputs(&input_route, emit);
371        let call = quote!(#call_path(#(#call_args),*));
372
373        let e_param = err_bits
374            .as_ref()
375            .map(|(err_wire, _, _)| quote!(e: *mut #err_wire));
376        let ret_arrow = c_return.as_ref().map(|w| quote!(-> #w));
377
378        // Assemble the body per the three structural modes.
379        let body = match (&err_bits, field0_is_return) {
380            // No `Result`: straight-line. `void` when there are no fields.
381            (None, _) => {
382                if let Some(field0_wire) = field0_wire.as_ref() {
383                    let enc = value_site.map_or_else(TokenStream::new, |site| {
384                        site.abi()
385                            .payload()
386                            .encode(quote!(__v), &targets, &input_route, emit)
387                    });
388                    quote!(
389                        #(#decodes)*
390                        let __v = #call;
391                        let __ret: #field0_wire;
392                        #enc
393                        __ret
394                    )
395                } else {
396                    quote!( #(#decodes)* #call; )
397                }
398            }
399            // `Result` with a free niche: value in-band, NULL marks `Err`.
400            (Some((_, e_conv, _)), true) => {
401                let field0_wire = field0_wire.as_ref().expect("in-band ⇒ pointer return");
402                let null = &result_slot
403                    .as_ref()
404                    .expect("in-band result has a niche")
405                    .value;
406                let enc = value_site.map_or_else(TokenStream::new, |site| {
407                    site.abi()
408                        .payload()
409                        .encode(quote!(__v), &targets, &input_route, emit)
410                });
411                quote!(
412                    #(#decodes)*
413                    match #call {
414                        ::core::result::Result::Ok(__v) => { let __ret: #field0_wire; #enc __ret }
415                        ::core::result::Result::Err(__err) => {
416                            if !e.is_null() { *e = #e_conv(__err); }
417                            #null
418                        }
419                    }
420                )
421            }
422            // `Result` without a free niche: `bool` status, value to out-params.
423            (Some((_, e_conv, _)), false) => {
424                let enc = value_site.map_or_else(TokenStream::new, |site| {
425                    site.abi()
426                        .payload()
427                        .encode(quote!(__v), &targets, &input_route, emit)
428                });
429                quote!(
430                    #(#decodes)*
431                    match #call {
432                        ::core::result::Result::Ok(__v) => { #enc true }
433                        ::core::result::Result::Err(__err) => {
434                            if !e.is_null() { *e = #e_conv(__err); }
435                            false
436                        }
437                    }
438                )
439            }
440        };
441
442        syn::parse_quote! {
443            #[no_mangle]
444            #[allow(non_snake_case, unused_mut, unused_variables, unused_unsafe, dead_code)]
445            pub unsafe extern "C" fn #sym(
446                #(#in_params,)*
447                #(#out_param_decls,)*
448                #e_param
449            ) #ret_arrow {
450                #body
451            }
452        }
453    }
454
455    /// The runtime **alias preflight**: reject a call whose arguments name the
456    /// same resource in a combination that would reconstruct or invalidate it
457    /// twice, *before* any conversion runs.
458    ///
459    /// `z_combine(primary: ZThing, fallback: ZThing)` is a supported
460    /// declaration; called as `z_combine(x, x)` it reaches `Box::from_raw`
461    /// twice on one allocation. So is `f(a: ZThing, b: &ZThing)` — the borrow
462    /// dangles the moment the consume takes ownership — and `f(a: &mut T, b:
463    /// &T)`, where the exclusive reference is not exclusive at all. Rejecting
464    /// the *declaration* is not an option: these are shapes that ship today, so
465    /// removing them would be a regression that a later stage would have to
466    /// undo.
467    ///
468    /// Emitted whenever a call has **at least one `Consume` or
469    /// `ExclusiveBorrow`** and **any other active access in the same resource
470    /// domain**. Stated that way rather than as "two or more consumed
471    /// parameters", which would skip exactly the two mixed cases above.
472    ///
473    /// A NULL pointer names no resource — two NULLs are not an alias — and is
474    /// rejected by each converter's own null check, so the comparison skips it.
475    ///
476    /// Shared/shared is *not* rejected: two `&T` to one resource is legal Rust
477    /// and legal C.
478    fn alias_preflight(&self, route: &ErrRoute) -> Option<TokenStream> {
479        let slots: Vec<(syn::Ident, TypeKey, AliasAccess)> = self
480            .function
481            .params
482            .iter()
483            .zip(&self.alias_slots)
484            .filter_map(|(param, slot)| {
485                slot.as_ref()
486                    .map(|(key, access)| (param.name.clone(), key.clone(), *access))
487            })
488            .collect();
489
490        let on_err = route_message(route);
491        let mut checks: Vec<TokenStream> = Vec::new();
492        for i in 0..slots.len() {
493            for j in (i + 1)..slots.len() {
494                let (a, a_key, a_access) = &slots[i];
495                let (b, b_key, b_access) = &slots[j];
496                if a_key != b_key {
497                    continue;
498                }
499                // At least one side must be exclusive about the resource.
500                if matches!(a_access, AliasAccess::Shared)
501                    && matches!(b_access, AliasAccess::Shared)
502                {
503                    continue;
504                }
505                let msg = format!(
506                    "aliasing arguments: `{a}` ({}) and `{b}` ({}) are the same `{}` — a \
507                     consumed or exclusively-borrowed resource may not be named twice in one call",
508                    a_access.describe(),
509                    b_access.describe(),
510                    a_key.as_str(),
511                );
512                checks.push(quote!(
513                    if !(#a as *const ()).is_null() && (#a as *const ()) == (#b as *const ()) {
514                        let __msg = ::std::string::String::from(#msg);
515                        #on_err
516                    }
517                ));
518            }
519        }
520        (!checks.is_empty()).then(|| quote!(#(#checks)*))
521    }

523    /// Build the wire param list, per-input decode statements, and call-site
524    /// argument expressions. Fallible inputs (converter returns `Result<_,
525    /// String>`) route their `Err(msg)` per `route`; infallible inputs decode
526    /// directly.
527    fn planned_inputs(
528        &self,
529        route: &ErrRoute<'_>,
530        emit: &prebindgen_registry::RustWriter,
531    ) -> (Vec<TokenStream>, Vec<TokenStream>, Vec<TokenStream>) {
532        let mut params = Vec::new();
533        let mut decodes: Vec<TokenStream> = self.alias_preflight(route).into_iter().collect();
534        let mut call_args = Vec::new();
535
536        for (index, param) in self.function.params.iter().enumerate() {
537            let ident = &param.name;
538            let site = self.site(prebindgen_registry::recipe::Role::Param { index });
539            match site.abi().payload() {
540                crate::compile::CValue::BorrowedInput {
541                    element,
542                    wire,
543                    reinterpret,
544                } => {
545                    let len_id = format_ident!("{}_len", ident);
546                    let source = emit.emit_source_type(element);
547                    params.push(quote!(#ident: #wire));
548                    params.push(quote!(#len_id: usize));
549                    let from_parts = if *reinterpret {
550                        quote!(::core::slice::from_raw_parts(
551                            #ident as *const #source,
552                            #len_id,
553                        ))
554                    } else {
555                        quote!(::core::slice::from_raw_parts(#ident, #len_id))
556                    };
557                    decodes.push(quote!(
558                        let #ident: &[#source] = if #ident.is_null() {
559                            &[]
560                        } else {
561                            #from_parts
562                        };
563                    ));
564                }
565                crate::compile::CValue::Direct {
566                    wire, converter, ..
567                } => {
568                    let conv = converter.ident(emit);
569                    params.push(quote!(#ident: #wire));
570                    if converter.fallible() {
571                        let on_err = route_message(route);
572                        decodes.push(quote!(
573                            let #ident = match #conv(#ident) {
574                                ::core::result::Result::Ok(__v) => __v,
575                                ::core::result::Result::Err(__msg) => { #on_err }
576                            };
577                        ));
578                    } else {
579                        decodes.push(quote!(let #ident = #conv(#ident);));
580                    }
581                }
582                _ => panic!("C input site has an output-only ABI plan"),
583            }
584            call_args.push(quote!(#ident));
585        }
586
587        (params, decodes, call_args)
588    }
589}

591impl RustArtifact for CWrapper {
592    fn calls(&self) -> Vec<ArtifactKey> {
593        // One site per parameter, plus the return and the error channel when
594        // the function has them — the same sites the body decodes and encodes
595        // through.
596        let mut calls = Vec::new();
597        let roles = (0..self.function.params.len())
598            .map(|index| prebindgen_registry::recipe::Role::Param { index })
599            .chain([
600                prebindgen_registry::recipe::Role::Return,
601                prebindgen_registry::recipe::Role::Error,
602            ]);
603        for role in roles {
604            if let Some(site) = self.site_if_planned(role) {
605                site.abi().payload().calls(&mut calls);
606            }
607        }
608        calls
609    }
610
611    fn key(&self) -> ArtifactKey {
612        ArtifactKey::Artifact(
613            prebindgen_registry::generation::ArtifactId::new("c-wrapper", self.symbol.to_string())
614                .expect("an exported symbol is a non-empty artifact name"),
615        )
616    }
617
618    fn render(&self, emit: &prebindgen_registry::RustWriter) -> Vec<syn::Item> {
619        vec![syn::Item::Fn(self.render_fn(emit))]
620    }
621}

623/// One captured `#[prebindgen]` constant, re-stated in the generated file as
624/// an alias to the source item.
625///
626/// The initializer is never copied: it may name source-crate internals, and
627/// an alias keeps a constant with a non-portable initializer valid here.
628/// cbindgen cannot evaluate a path initializer, so an aliased constant does
629/// not surface as a `#define` in the C header.
630pub(crate) struct CConst {
631    /// The constant as the model holds it.
632    constant: prebindgen_registry::flat::Constant,
633    /// Module the source constant is reached through. `None` when the binding
634    /// declared no source module, which emits nothing.
635    source_module: Option<syn::Path>,
636}
637
638impl CConst {
639    /// Plan the alias for one captured constant.
640    pub(crate) fn new(
641        decls: &CbindgenBuilder,
642        constant: &prebindgen_registry::flat::Constant,
643    ) -> Self {
644        Self {
645            constant: constant.clone(),
646            source_module: decls.source_module.clone(),
647        }
648    }
649}

653        Vec::new()
654    }
655
656    fn key(&self) -> ArtifactKey {

658            prebindgen_registry::generation::ArtifactId::new(
659                "c-const",
660                self.constant.name.to_string(),

668            .as_ref()

671    }

757/// The memory helpers: the C allocator the generated layer calls, the raw
758/// C-string block builder, the universal freer C calls back, and — when a
759/// `Vec<T>` return hands out a block — the array builder.
760///
761/// Planned only when the layer actually hands memory to C, which is also when
762/// a declared `.free_memory_function` becomes required.
763pub(crate) struct CMemory {
764    /// The declared freer's exported symbol.
765    free_ident: syn::Ident,
766}
767
768impl CMemory {
769    /// Plan the memory helpers for a binding that hands memory to C.
770    ///
771    /// Whether it does is not asked here: the artifacts that allocate say so
772    /// through their own dependencies, and the caller plans this when one of
773    /// them names it.
774    pub(crate) fn new(decls: &CbindgenBuilder) -> Self {
775        let Some(free_fn) = &decls.free_fn else {
776            panic!(
777                "Cbindgen: the generated layer hands C memory it must free — a \
778                 `char*` block (a `String` return or a `String` data-struct \
779                 field) or an array block (a `Vec` returned or delivered to a \
780                 callback) — but no memory-freeing function is declared: add \
781                 `.free_memory_function(\"z_free\")`"
782            )
783        };
784        Self {
785            free_ident: format_ident!("{}", free_fn),
786        }
787    }
788
789    /// Copy a `Vec<W>` into a C-`malloc`'d block of `W` and return
790    /// `(ptr, len)` (empty ⇒ `(NULL, 0)`). The block is freed C-side via the
791    /// `z_free_array` macro (per-element drop + the universal freer).
792    fn render_array_builder() -> Vec<syn::Item> {
793        vec![syn::parse_quote!(
794            #[allow(non_snake_case, dead_code)]
795            pub(crate) unsafe fn __cbg_alloc_array<W>(v: ::std::vec::Vec<W>) -> (*mut W, usize) {
796                let n = v.len();
797                if n == 0 {
798                    return (::core::ptr::null_mut(), 0);
799                }
800                let p = malloc(n.wrapping_mul(::core::mem::size_of::<W>())) as *mut W;
801                if p.is_null() {
802                    return (::core::ptr::null_mut(), 0);
803                }
804                for (i, e) in v.into_iter().enumerate() {
805                    ::core::ptr::write(p.add(i), e);
806                }
807                (p, n)
808            }
809        )]
810    }
811}
812
813impl RustArtifact for CMemory {
814    fn calls(&self) -> Vec<ArtifactKey> {
815        Vec::new()
816    }
817
818    fn key(&self) -> ArtifactKey {
819        memory_key()
820    }
821
822    fn render(&self, _emit: &prebindgen_registry::RustWriter) -> Vec<syn::Item> {
823        let free_ident = &self.free_ident;
824        // C allocator (linked from the C runtime; no crate dependency).
825        let items: Vec<syn::Item> = vec![
826            syn::parse_quote!(
827                extern "C" {
828                    fn malloc(size: usize) -> *mut ::core::ffi::c_void;
829                    fn free(ptr: *mut ::core::ffi::c_void);
830                }
831            ),
832            // Raw, destructor-free C-string block. `CString::new` drops interior
833            // NULs so the terminator marks the true end for C consumers.
834            syn::parse_quote!(
835                #[allow(non_snake_case, dead_code)]
836                pub(crate) fn __cbg_alloc_cstr(
837                    s: ::std::string::String,
838                ) -> *mut ::core::ffi::c_char {
839                    let c = ::std::ffi::CString::new(s).unwrap_or_default();
840                    let bytes = c.as_bytes_with_nul();
841                    unsafe {
842                        let p = malloc(bytes.len()) as *mut u8;
843                        if p.is_null() {
844                            return ::core::ptr::null_mut();
845                        }
846                        ::core::ptr::copy_nonoverlapping(bytes.as_ptr(), p, bytes.len());
847                        p as *mut ::core::ffi::c_char
848                    }
849                }
850            ),
851            // Universal raw memory freer: type-agnostic C `free`, no length, no
852            // destructor (NULL-safe via C `free`).
853            syn::parse_quote!(
854                #[no_mangle]
855                #[allow(non_snake_case, unused_variables)]
856                pub unsafe extern "C" fn #free_ident(p: *mut ::core::ffi::c_void) {
857                    free(p);
858                }
859            ),
860        ];
861        items
862    }
863}

865/// One declared data struct's `#[repr(C)]` mirror.
866///
867/// The mirror is a layout fact: each field is stated in its wire form, which
868/// is decided while planning, when the binding's declarations and the model
869/// are both available.
870pub(crate) struct CDataStruct {
871    /// The C-facing struct name.
872    c_struct: syn::Ident,
873    /// Field name and wire type, in source order.
874    fields: Vec<(syn::Ident, syn::Type)>,
875}
876
877impl CDataStruct {
878    /// Plan the mirrors of every declared data struct that crosses.
879    pub(crate) fn all(decls: &CbindgenBuilder, registry: &Registry) -> Vec<Self> {
880        let mut mirrors = Vec::new();
881        for (key, _cfg) in sorted_by_key(&decls.data) {
882            let Some(reading) = registry.reading(key) else {
883                continue;
884            };
885            if decls.in_frag(&reading).is_none() && decls.out_frag(&reading).is_none() {
886                continue;
887            }
888            let Some(fields) = decls.struct_fields(registry, &reading.key()) else {
889                continue;
890            };
891            mirrors.push(Self {
892                c_struct: decls.c_type_ident(&reading.key()),
893                fields: fields
894                    .iter()
895                    .map(|(name, ty)| {
896                        let wire = decls.data_field_wire(ty).unwrap_or_else(|| {
897                            panic!(
898                                "Cbindgen: field `{}` of data struct `{}` has unsupported type `{}`",
899                                name,
900                                type_short(&reading.key()),
901                                ty
902                            )
903                        });
904                        (name.clone(), wire)
905                    })
906                    .collect(),
907            });
908        }
909        mirrors
910    }
911}
912
913impl RustArtifact for CDataStruct {
914    fn calls(&self) -> Vec<ArtifactKey> {
915        Vec::new()
916    }
917
918    fn key(&self) -> ArtifactKey {
919        artifact_id("c-data-struct", self.c_struct.to_string())
920    }
921
922    fn render(&self, _emit: &prebindgen_registry::RustWriter) -> Vec<syn::Item> {
923        let c_struct = &self.c_struct;
924        let fields = self
925            .fields
926            .iter()
927            .map(|(name, wire)| quote!(pub #name: #wire));
928        vec![syn::parse_quote!(
929            #[repr(C)]
930            #[allow(non_camel_case_types)]
931            pub struct #c_struct {
932                #(#fields,)*
933            }
934        )]
935    }
936}

938/// One declared fieldless enum's `#[repr(C)]` mirror.
939///
940/// Each discriminant is re-stated **as written** — `= 0x07` stays `0x07` —
941/// which is what keeps every value C already accepted: a `const` or
942/// `cfg`-driven expression, and anything the source's own `repr` admits.
943/// Resolving each to a number would narrow that to what `i64` and a literal
944/// can express, for no gain, since cbindgen re-reads this as Rust source.
945///
946/// That is why the model's own values are retained here and spelled by the
947/// writer, rather than the mirror being spelled while planning.
948pub(crate) struct CEnum {
949    /// The C-facing enum name.
950    c_name: syn::Ident,
951    /// The source enum's values, in declaration order.
952    values: Vec<prebindgen_registry::flat::EnumValue>,
953}
954
955impl CEnum {
956    /// Plan the mirrors of every declared fieldless enum that crosses.
957    pub(crate) fn all(decls: &CbindgenBuilder, registry: &Registry) -> Vec<Self> {
958        let mut mirrors = Vec::new();
959        for (key, _cfg) in sorted_by_key(&decls.enums) {
960            let Some(reading) = registry.reading(key) else {
961                continue;
962            };
963            if decls.in_frag(&reading).is_none() && decls.out_frag(&reading).is_none() {
964                continue;
965            }
966            let Some(item) = unit_enum(registry, &reading.key()) else {
967                continue;
968            };
969            mirrors.push(Self {
970                c_name: decls.c_type_ident(&reading.key()),
971                values: item.values.clone(),
972            });
973        }
974        mirrors
975    }
976}
977
978impl RustArtifact for CEnum {
979    fn calls(&self) -> Vec<ArtifactKey> {
980        Vec::new()
981    }
982
983    fn key(&self) -> ArtifactKey {
984        artifact_id("c-enum", self.c_name.to_string())
985    }
986
987    fn render(&self, emit: &prebindgen_registry::RustWriter) -> Vec<syn::Item> {
988        let c_name = &self.c_name;
989        let variants = self.values.iter().map(|value| {
990            let id = &value.name;
991            match emit.discriminant(value) {
992                Some(expr) => quote!(#id = #expr),
993                None => quote!(#id),
994            }
995        });
996        vec![syn::parse_quote!(
997            #[repr(C)]
998            #[derive(Copy, Clone, Debug, Eq, PartialEq)]
999            #[allow(non_camel_case_types)]
1000            pub enum #c_name {
1001                #(#variants),*
1002            }
1003        )]
1004    }
1005}
