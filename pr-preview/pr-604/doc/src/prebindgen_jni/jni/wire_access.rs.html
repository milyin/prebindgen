10/// Map a JNI wire type to `(jvm_field_descriptor, JValue_accessor_ident, is_object)`.
11///
12/// Primitive types (`jlong`, `jint`, …) set `is_object = false` and the
13/// accessor names the `.j()` / `.i()` / … `JValue` variant.
14///
15/// Object types (`JString`, `JByteArray`, …) set `is_object = true`; the
16/// caller uses `.l()` to get a `JObject` and then `.into()` to cast to the
17/// wire type.
18/// The Kotlin primitive arrays and their JVM descriptors.
19///
20/// ONE table, read forwards (Kotlin type → descriptor, for interface signatures
21/// and field slots) and backwards (descriptor → empty literal, for inert
22/// flattened slots). Four separate copies of this knowledge existed before
23/// fixed-size arrays needed all of them, and only some got updated.
24pub(crate) const KOTLIN_PRIM_ARRAYS: [(&str, &str); 8] = [
25    ("BooleanArray", "[Z"),
26    ("ByteArray", "[B"),
27    ("CharArray", "[C"),
28    ("ShortArray", "[S"),
29    ("IntArray", "[I"),
30    ("LongArray", "[J"),
31    ("FloatArray", "[F"),
32    ("DoubleArray", "[D"),
33];
34
35/// JVM descriptor of a Kotlin primitive array (`ByteArray` → `[B`).
36pub(crate) fn kotlin_array_descriptor(name: &str) -> Option<&'static str> {
37    KOTLIN_PRIM_ARRAYS
38        .iter()
39        .find(|(n, _)| *n == name)
40        .map(|(_, d)| *d)
41}
42
43/// The Kotlin primitive array a JVM array descriptor names (`[B` → `ByteArray`).
44pub(crate) fn kotlin_array_of_descriptor(descr: &str) -> Option<&'static str> {
45    KOTLIN_PRIM_ARRAYS
46        .iter()
47        .find(|(_, d)| *d == descr)
48        .map(|(n, _)| *n)
49}

51/// The JNI **reference** (object-shaped) wire types: the `J*` handles plus the
52/// eight primitive arrays.
53///
54/// ONE list. It drives the JVM field descriptor, the wire→`JObject` cast, and
55/// the lifetime annotation on a generated converter's signature — three sites
56/// that each kept their own copy until fixed-size arrays added a wire to all of
57/// them and only two got updated.
58pub(crate) fn is_jni_reference_wire(ty: &syn::Type) -> bool {
59    let syn::Type::Path(tp) = ty else {
60        return false;
61    };
62    let Some(last) = tp.path.segments.last() else {
63        return false;
64    };
65    matches!(
66        last.ident.to_string().as_str(),
67        "JObject"
68            | "JString"
69            | "JClass"
70            | "JBooleanArray"
71            | "JByteArray"
72            | "JCharArray"
73            | "JShortArray"
74            | "JIntArray"
75            | "JLongArray"
76            | "JFloatArray"
77            | "JDoubleArray"
78    )
79}

81pub(crate) fn jni_field_access(jni_type: &syn::Type) -> Option<(&'static str, syn::Ident, bool)> {
82    if let Some(p) = JniPrim::from_wire(jni_type) {
83        return Some((p.descriptor(), format_ident!("{}", p.unbox_getter()), false));
84    }
85    let syn::Type::Path(tp) = jni_type else {
86        return None;
87    };
88    let sig = match tp.path.segments.last()?.ident.to_string().as_str() {
89        "JString" => "Ljava/lang/String;",
90        // The eight primitive-array wires, one per `JniPrim` scalar — a
91        // fixed-size Rust array crosses as the matching Kotlin primitive array
92        // (see `prim_array`).
93        "JBooleanArray" => "[Z",
94        "JByteArray" => "[B",
95        "JCharArray" => "[C",
96        "JShortArray" => "[S",
97        "JIntArray" => "[I",
98        "JLongArray" => "[J",
99        "JFloatArray" => "[F",
100        "JDoubleArray" => "[D",
101        _ => return None,
102    };
103    Some((sig, format_ident!("l"), true))
104}

106/// Map a JNI **primitive** field descriptor to the descriptor of its
107/// `java.lang.*` box class — the JVM slot an `Option`-boxed primitive leaf
108/// occupies ([`box_helper_for_wire`] produces the boxed value; this names its
109/// type in a method signature).
110pub(crate) fn box_descriptor_for_primitive(sig: &str) -> Option<&'static str> {
111    JniPrim::from_descriptor(sig).map(JniPrim::box_descriptor)
112}

114/// Map a JNI **primitive** wire type to the `prebindgen::lang` cached-boxing
115/// runtime helper that boxes it into its `java.lang.*` wrapper. Used when a
116/// primitive leaf must be delivered through an *erased* `Object`-typed call
117/// (e.g. a Kotlin function type's `invoke`). The helpers resolve the box class
118/// and its static `valueOf` once per process — boxing inline with
119/// `env.new_object("java/lang/Integer", …)` would re-run `FindClass` +
120/// `GetMethodID` on every delivery, which dominates the callback hot path.
121/// Returns `None` for object wires (`JString`/`JByteArray`/`JObject` are
122/// already objects).
123pub(crate) fn box_helper_for_wire(wire: &syn::Type) -> Option<syn::Ident> {
124    JniPrim::from_wire(wire).map(|p| format_ident!("box_{}", p.wire_name()))
125}
