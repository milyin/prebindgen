5pub(crate) fn transport(
6    wire: syn::Ident,
7    length: syn::Ident,
8    element: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
9    values: &[(syn::Ident, TokenStream)],
10) -> Result<SequenceOutput<JniWire, crate::JniChain>> {
11    let env = prebindgen_conversions::template_parameter("environment");
12    let method = format_ident!("{wire}_callback_method");
13    let class = format_ident!("{wire}_callback_class");
14    let frame = format_ident!("{wire}_callback_frame");
15    let mut descriptor = String::from("(");
16    let mut encode = TokenStream::new();
17    let mut args = Vec::new();
18    let mut needs_frame = false;
19    let slots: usize = crate::group::callback_parameters(element)
20        .map(|w| w.kind.parameter_slots())
21        .sum();
22    if slots > 254 {
23        return Err("sequence callback exceeds JVM parameter limit".into());
24    }
25    for (i, (field, value)) in prebindgen_conversions::match_output_fields(
26        crate::group::callback_parameters(element),
27        values,
28    )?
29    .into_iter()
30    .enumerate()
31    {
32        descriptor.push_str(field.kind.jvm_descriptor());
33        let local = format_ident!("{wire}_callback_object_{i}");
34        let argument = field.kind.output_argument(value, &local);
35        // This is the same classification used by output_argument to allocate
36        // a JVM object. Primitive wire values, including handles, create none.
37        needs_frame |= field.kind.is_reference();
38        encode.extend(argument.statements);
39        args.push(argument.value);
40    }
41    descriptor.push_str(")V");
42    let (enter, leave) = if needs_frame {
43        let capacity = i32::try_from(crate::group::callback_parameters(element).len() + 8)
44            .map_err(|_| "too many callback fields")?;
45        (
46            quote!(let #frame=unsafe{::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;),
47            quote!(drop(#frame);),
48        )
49    } else {
50        (TokenStream::new(), TokenStream::new())
51    };
52    let commit_element = prebindgen_conversions::template_parameter("commit_element");
53    Ok(SequenceOutput {
54        controls: vec![prebindgen_flat::Wire {
55            name: wire.clone(),
56            // Zero means present, even when no rows arrived. The enclosing
57            // option reserves -1; both values use cached boxes in packet mode.
58            kind: JniWire::selector(),
59        }],
60        commit_each: true,
61        fallible: true,
62        metadata: JniChain::Sequence(JniSequence::Callback),
63        begin: quote! {
64            // Every accepted row belongs to the Kotlin collector, including
65            // when a later row fails. Publish this before any row can arrive.
66            #env.set_boolean_array_region(&__pb_output_accepted,0,&[1]).map_err(|e|e.to_string())?;
67            let #class=#env.get_object_class(&__pb_callback).map_err(|e|e.to_string())?;
68            let #method=#env.get_method_id(&#class,"run",#descriptor).map_err(|e|e.to_string())?;
69            // An extra upcall costs more than small-list growth. Keep short
70            // captures on the existing JNI path and reserve only larger lists,
71            // once before delivery and never inside the row loop.
72            if #length >= 64 {
73                let capacity=<i32 as ::core::convert::TryFrom<_>>::try_from(#length).map_err(|_| "collection exceeds JVM capacity".to_string())?;
74                let (_, reserve)=__PB_SEQUENCE_CAPACITY.get().ok_or_else(|| "collection capacity hint is not initialized".to_string())?;
75                unsafe {#env.call_method_unchecked(&__pb_callback,*reserve,::prebindgen_jni_runtime::jni::signature::ReturnType::Primitive(::prebindgen_jni_runtime::jni::signature::Primitive::Void),&[::prebindgen_jni_runtime::jni::sys::jvalue{i:capacity}])}.map_err(|e|e.to_string())?;
76            }
77            #env.delete_local_ref(#class).map_err(|e|e.to_string())?;
78        },
79        enter,
80        store: quote! {
81            #encode
82            // The method was resolved on this callback object with exactly these
83            // JVM wire types. The generated bridge only retains raw fields.
84            unsafe {#env.call_method_unchecked(&__pb_callback,#method,::prebindgen_jni_runtime::jni::signature::ReturnType::Primitive(::prebindgen_jni_runtime::jni::signature::Primitive::Void),&[#(#args),*])}.map_err(|e|e.to_string())?;
85            #commit_element
86            #leave
87        },
88        // Row payloads crossed through run; only scalar presence crosses back.
89        finish: quote!(let #wire=0i32;),
90        values: vec![(wire.clone(), quote!(#wire))],
91        commit: quote!(),
92    })
93}

95/// Retain only raw wire fields while JNI and handle locks are active. A small
96/// immutable record avoids primitive boxes and generic row arrays without
97/// copying wide rows when the collector grows. Logical projections can share
98/// the same physical JVM record type.
99pub(crate) fn capture(
100    element: &kotlin::JniNode,
101    names: &mut names::Names,
102) -> Result<(String, String)> {
103    let fields = crate::group::callback_parameters(element).collect::<Vec<_>>();
104    let signature = fields
105        .iter()
106        .map(|w| w.kind.jvm_descriptor())
107        .collect::<String>();
108    let label = fields
109        .iter()
110        .take(3)
111        .map(|w| w.kind.kotlin_type().trim_end_matches('?'))
112        .collect::<Vec<_>>()
113        .join("_");
114    let label = if fields.is_empty() {
115        "Empty".into()
116    } else {
117        format!("{label}_{}", fields.len())
118    };
119    let kotlin_types = fields
120        .iter()
121        .map(|w| w.kind.kotlin_type())
122        .collect::<Vec<_>>()
123        .join(",");
124    let name = names.name("SequenceRow", &label, &[&signature, &kotlin_types])?;
125    let fields = fields
126        .iter()
127        .enumerate()
128        .map(|(i, w)| format!("@JvmField val c{i}: {}", w.kind.kotlin_type()))
129        .collect::<Vec<_>>()
130        .join(", ");
131    Ok((
132        name.clone(),
133        format!("    internal class {name}({fields})\n"),
134    ))
135}

142pub(crate) fn fold(
143    layout: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
144    context: &KotlinContext<'_>,
145) -> Result<Fold> {
146    let optional = crate::sequence::callback_optional(layout)?;
147    let (carrier, element, _) =
148        crate::sequence::return_sequence(layout).ok_or("missing sequence callback layout")?;
149    let position = crate::group::result_packet_fields(layout)
150        .position(|w| w.name == carrier.name)
151        .unwrap();
152    let row = "__pb_fold_row";
153    let ctx = KotlinContext {
154        result: row,
155        output_prefix: "foldElement",
156        cleanup_failure: None,
157        ..*context
158    };
159    // Lexical row cleanup keeps partially constructed wrappers in scope. Only
160    // dynamic nested collections need the shared owner journal; fixed fields
161    // can use their existing locals without allocating a journal or a lambda.
162    fn nested_owners(node: &kotlin::JniNode) -> bool {
163        match node {
164            prebindgen_flat::WireNode::Chain {
165                meta: JniChain::Sequence(_),
166                child,
167                ..
168            } => kotlin::input_handles(child, false),
169            prebindgen_flat::WireNode::Chain { child, .. } => nested_owners(child),
170            prebindgen_flat::WireNode::Branch { children, .. } => {
171                children.iter().any(nested_owners)
172            }
173            _ => false,
174        }
175    }
176    let fields = crate::group::callback_parameters(element).collect::<Vec<_>>();
177    let fields_read = (0..fields.len())
178        .map(|i| format!("__pb_fold_row.c{i}"))
179        .collect::<Vec<_>>();
180    let mut unread = kotlin::output_kotlin(element, &ctx)?;
181    returns::remap(&mut unread, row, &fields, 0, &fields_read);
182    let shared = nested_owners(element);
183    let owners = "__pb_foldOwners";
184    let child = if shared {
185        let mut child = kotlin::output_kotlin_scoped(element, &ctx, Some(owners))?;
186        returns::remap(&mut child, row, &fields, 0, &fields_read);
187        child
188    } else {
189        unread.clone()
190    };
191    let present = format!("{}[{position}] as Int == 0", context.result);
192    let raw = "__pb_rows";
193    let types = child.callback_type_parameters()?;
194    let args = child.delivered.join(", ");
195    let failure = context.failure;
196    let resources = !unread.cleanup.is_empty();
197    let mut locals = String::from("var __pb_accumulator: A? = null\n");
198    let mut initialize = String::new();
199    let mut row = format!("{}\ncurrent = fold(current, {args})\n", child.materialize);
200    let mut discard = String::new();
201    let mut cleanup = String::new();
202    if resources {
203        locals.push_str("var __pb_processed = 0\n");
204        let release = if shared {
205            locals.push_str(&format!(
206                "var {owners}: {}.CollectionOwners? = null\n",
207                context.package
208            ));
209            initialize = format!(
210                "if ({raw}.size != 0) {owners} = {}.CollectionOwners()\n",
211                context.package
212            );
213            row = format!(
214                "{}\n{owners}!!.complete = true\ncurrent = fold(current, {args})\n",
215                child.materialize
216            );
217            format!(
218                "{owners}?.close({failure})\nif ({owners}?.complete != true) {{\n{}\n}}",
219                child.cleanup
220            )
221        } else {
222            child.cleanup.clone()
223        };
224        let delivered = if shared {
225            format!("{owners}!!.release()\n")
226        } else {
227            String::new()
228        };
229        // Account for the current row before cleanup, so a secondary cleanup
230        // failure cannot make the outer failure scan destroy it a second time.
231        row = format!("try {{\n{row}}} catch ({failure}: Throwable) {{\n__pb_processed++\ntry {{\n{release}\n}} catch (cleanup: Throwable) {{ if ({failure} !== cleanup) {failure}.addSuppressed(cleanup) }}\nthrow {failure}\n}}\n__pb_processed++\n{delivered}");
232        discard = format!("fun __pb_discard({failure}: Throwable, start: Int = 0) {{\nfor (__pb_rowIndex in start until {raw}.size) {{\ntry {{\nval __pb_fold_row = {raw}[__pb_rowIndex]\n{}\n{}\n}} catch (cleanup: Throwable) {{ if ({failure} !== cleanup) {failure}.addSuppressed(cleanup) }}\n}}\n}}\n", unread.output_locals, unread.cleanup);
233        cleanup = format!("if ({present}) __pb_discard({failure}, __pb_processed)\n");
234    }
235    Ok(Fold {
236        parameter: format!("fold: (A, {types}) -> A"),
237        discard,
238        output: KotlinOutput {
239            result_names: vec![None],
240            delivered: vec![if optional { "__pb_accumulator".into() } else { "__pb_accumulator as A".into() }],
241            result_types: vec![if optional { "A?".into() } else { "A".into() }],
242            output_locals: locals,
243            materialize: format!("if ({present}) {{\n{initialize}var current = initial\nfor (__pb_rowIndex in 0 until {raw}.size) {{\nval __pb_fold_row = {raw}[__pb_rowIndex]\n{}\n{row}}}\n__pb_accumulator = current\n}}\n", child.output_locals),
244            cleanup,
245        },
246    })
247}

249/// One reservation method shared by every generated collector. Register the
250/// actual interface class once, avoiding application-class lookups on workers
251/// and method-name/signature allocations on each large return.
252pub(crate) fn emit_capacity(
253    prefix: &str,
254    rust: &mut TokenStream,
255    native: &mut String,
256    names: &mut names::Names,
257) -> Result<()> {
258    let method = names.name("init", "SequenceCapacity", &["collection capacity hint"])?;
259    let symbol = symbol(prefix, &method);
260    rust.extend(quote! {
261        static __PB_SEQUENCE_CAPACITY: ::std::sync::OnceLock<(
262            ::prebindgen_jni_runtime::jni::objects::GlobalRef,
263            ::prebindgen_jni_runtime::jni::objects::JMethodID,
264        )> = ::std::sync::OnceLock::new();
265        #[no_mangle]
266        pub extern "system" fn #symbol(mut env: ::prebindgen_jni_runtime::jni::JNIEnv<'_>, _class: ::prebindgen_jni_runtime::jni::objects::JClass<'_>, capacity: ::prebindgen_jni_runtime::jni::objects::JClass<'_>) {
267            if let Err(error)=::prebindgen_runtime::boundary(|| {
268                if __PB_SEQUENCE_CAPACITY.get().is_none() {
269                    let reserve=env.get_method_id(&capacity,"reserve","(I)V").map_err(|e|e.to_string())?;
270                    let class=env.new_global_ref(&capacity).map_err(|e|e.to_string())?;
271                    let _=__PB_SEQUENCE_CAPACITY.set((class,reserve));
272                }
273                let (class,_)=__PB_SEQUENCE_CAPACITY.get().expect("capacity method initialized");
274                if !env.is_same_object(class.as_obj(),&capacity).map_err(|e|e.to_string())? {
275                    return Err("collection capacity interface registered with a different class".into());
276                }
277                Ok(())
278            }) { ::prebindgen_jni_runtime::fail(&mut env,&error); }
279        }
280    });
281    native.push_str(&format!("    internal interface SequenceCapacity {{ fun reserve(capacity: Int) }}\n    @JvmStatic private external fun {method}(capacity: Class<*>)\n    init {{ {method}(SequenceCapacity::class.java) }}\n"));
282    Ok(())
