16/// One generated Kotlin `enum class` source — variants in
17/// SCREAMING_SNAKE_CASE, each carrying the Rust discriminant as a
18/// `val value: Int`, plus a `fromInt(value: Int)` companion. Mirrors
19/// the hand-written `io.zenoh.qos.Priority` shape so adapter code that
20/// already speaks the `.value` / `.fromInt(...)` idiom keeps working.
21pub(crate) fn build_enum_class(
22    class_name: &str,
23    item_enum: &prebindgen_registry::flat::Enum,
24) -> KtClass {
25    // Same discriminant source of truth the Rust `jint → variant` decode
26    // uses, so Kotlin `value(N)` and the generated decode agree — and it is the
27    // model's, which is where "same" stops needing to be maintained.
28    let entries: Vec<KtEnumEntry> = item_enum
29        .discriminant_values()
30        .unwrap_or_else(|name| {
31            panic!(
32                "enum `{}` variant `{name}` has a non-literal discriminant; use a literal \
33                 integer value (e.g. `= 1`) or an implicit discriminant",
34                item_enum.name
35            )
36        })
37        .into_iter()
38        .map(|(ident, value)| {
39            KtEnumEntry::with_args(
40                mangle_kotlin_ident(&crate::util::camel_to_screaming_snake(&ident.to_string())),
41                value.to_string(),
42            )
43        })
44        .collect();
45
46    let framework_line = format!(
47        "JVM-side surface for the native Rust `{}` enum.",
48        item_enum.name
49    );
50    let enum_kdoc = item_enum
51        .docs()
52        .map(|d| format!("{d}\n\n{framework_line}"))
53        .unwrap_or(framework_line);
54    let mut class = KtClass::enum_(class_name)
55        .vis(KtVis::Public)
56        .kdoc(enum_kdoc)
57        .ctor_param(
58            KtCtorParam::new("value", KtType::int())
59                .val()
60                .vis(KtVis::Public),
61        );
62    for e in entries {
63        class = class.entry(e);
64    }
65    // `@JvmStatic` exposes `fromInt` as a real static method on the enum
66    // class itself (rather than only on the `Companion` nested class). The
67    // generated struct-encoder calls it via `env.call_static_method`,
68    // which wouldn't find a companion-only method.
69    class.companion(
70        KtCompanion::new().vis(KtVis::Public).member(
71            KtFun::new("fromInt")
72                .vis(KtVis::Public)
73                .annotation("JvmStatic")
74                .param(KtParam::new("value", KtType::int()))
75                .returns(KtType::cls(class_name))
76                .expr_body(KtCode::new().line("entries.first { it.value == value }")),
77        ),
78    )
79}

81/// Build the Kotlin `data class` declaration for a `data_class`-declared
82/// Rust struct. The class is **self-contained**: property/factory-param types
83/// are full-FQN `KtType`s (the render-time `ImportSet` shortens + imports
84/// them), and the `fromParts` factory's raw-text class references carry their
85/// imports on the factory body `Code`.
86pub(crate) fn build_data_class(
87    ext: &Declarations,
88    class_name: &str,
89    item_struct: &prebindgen_registry::flat::Struct,
90    registry: &Registry,
91) -> KtClass {
92    // A tuple struct is an `Extern` in the model, never a `Struct`, so every
93    // field here is named by construction.
94    let fields_named = &item_struct.fields;
95
96    // The class declaration is derived from ONE classification of the fields,
97    // shared with the sealed-class emitter rather than repeated here. Deriving
98    // it separately — a classification with its own rules — is what let a
99    // property's type disagree with the parameter that fills it (#156). The
100    // parameter itself comes from the decomposition, so what these two must
101    // agree about is the value, not the slot.
102    let plan = ext
103        .struct_plan(registry.flat(), item_struct, 0)
104        .unwrap_or_else(|| {
105            panic!(
106                "data class `{}`: could not classify every field for the fromParts bridge. Each \
107             field needs a resolved OUTPUT converter (that direction declares the slot the \
108             encoder fills) AND the Kotlin metadata that converter carries — a `kotlin_name`, \
109             or a registered class for a projection leaf",
110                item_struct.name
111            )
112        });
113
114    let mut ctor_params: Vec<KtCtorParam> = Vec::new();
115    // Property (name, type) pairs, for the content-equality members an
116    // array-backed property needs — see [`equality::content_equality_members`].
117    let mut equality_props: Vec<(String, KtType)> = Vec::new();
118    // Track per-field destructible (name, folded close strategy) so the
119    // bottom emitter can produce a matching `close()` body for each.
120    let mut destructible_fields: Vec<(String, crate::jni::FoldStrategy)> = Vec::new();
121    for (field, pf) in fields_named.iter().zip(&plan.fields) {
122        let field_ident = field.name.as_ref().unwrap_or_else(|| {
123            panic!(
124                "render_data_class_source: struct `{}` has an unnamed field in named-fields context",
125                item_struct.name
126            )
127        });
128        let kotlin_field_name = kotlin_property_name(field_ident);
129        let owner = format!("{}.{}", item_struct.name, field_ident);
130
131        // The declaration reads ONE direction — output — because that is the
132        // direction the class hands values OUT in, and the direction
133        // `build_struct_plan` classifies exclusively (`flat_input.rs` is the
134        // input side's own). A property whose type came from whichever
135        // direction happened to resolve, while the slot carrying it came from
136        // the other, is how the declaration drifted.
137        //
138        // The property and the slot are different questions — one per field,
139        // one per leaf — and the slots are the decomposition's now. What must
140        // still agree is that they describe the same value, which is what
141        // reading one direction for both is for.
142        //
143        // A projection visible only on the INPUT side is exactly that drift
144        // made reachable: the old walk emitted a typed handle property (plus
145        // `AutoCloseable`) while the plan classified the same field as an
146        // ordinary leaf, so the property and its own factory parameter
147        // disagreed. Reject it at the declaration instead.
148        if !matches!(pf.kind, PlanFieldKind::Projection { .. }) {
149            if let Some(proj) = ext
150                .in_frag(&field.ty)
151                .and_then(|e| e.metadata.projection.clone())
152            {
153                panic!(
154                    "data class field `{owner}`: the INPUT converter projects this field as `{:?}` \
155                     but the OUTPUT converter does not, so the property type and its own \
156                     `fromParts` parameter would disagree. The output direction declares the \
157                     bridge, so give the field an output converter with the same projection — or \
158                     drop the input-side projection",
159                    proj.kind
160                );
161            }
162        }
163
164        let property_type = pf.kind.property_type(&owner);
165        equality_props.push((kotlin_field_name.clone(), property_type.clone()));
166        ctor_params.push(KtCtorParam::new(&kotlin_field_name, property_type).val());
167        if let Some(strategy) = pf.kind.destructible() {
168            destructible_fields.push((kotlin_field_name, strategy));
169        }
170    }
171
172    // `fromParts` companion factory, from the struct's decomposition — the
173    // same leaves the native encode fills, so nested data-class fields are
174    // inlined as their leaf wires and native builds the whole object graph with
175    // ONE `call_static_method`. Its raw-text class references
176    // (`Child.fromParts`, `Enum.fromInt`, projection wraps) use short names; the
177    // FQNs they need are collected here and attached to the factory body `Code`
178    // below.
179    let mut factory_imports: BTreeSet<String> = BTreeSet::new();
180    let (factory_params, factory_reconstruct, factory_mints_handle) = flatten_struct_factory(
181        ext,
182        registry,
183        item_struct,
184        "",
185        class_name,
186        &mut factory_imports,
187        0,
188    )
189    .unwrap_or_else(|| {
190        panic!("render_data_class_source: could not build fromParts factory for `{class_name}`")
191    });
192
193    // Kotlin requires a `data class` to declare at least one constructor
194    // property, so the model takes the first one at construction. A declared
195    // `data_class` with no fields could never have rendered as valid Kotlin.
196    let mut ctor_params = ctor_params.into_iter();
197    let first = ctor_params.next().unwrap_or_else(|| {
198        panic!(
199            "data class `{class_name}`: Rust struct `{}` has no fields — a Kotlin `data class` \
200             must declare at least one constructor property",
201            item_struct.name
202        )
203    });
204    let mut class = KtClass::data(class_name, first).vis(KtVis::Public);
205    if let Some(doc) = item_struct.docs() {
206        class = class.kdoc(doc);
207    }
208    for p in ctor_params {
209        class = class.ctor_param(p);
210    }
211    // Array-backed properties compare by identity in Kotlin, so a class with
212    // one gets explicit content-based operators (the Rust type derives `Eq`).
213    for m in crate::jni::equality::content_equality_members(class_name, &equality_props)
214        .into_iter()
215        .flatten()
216    {
217        class = class.member(m);
218    }
219    // Supertype clause: a data class with a destructible native-handle field
220    // implements `AutoCloseable`; otherwise no supertype.
221    if !destructible_fields.is_empty() {
222        class = class.implements(KtType::cls("AutoCloseable"));
223        // `close()` walks every destructible field via its folded close
224        // strategy. `JNINativeHandle.close()` is idempotent
225        // (Cleaner.Cleanable.clean() invokes exactly once), so calling
226        // this multiple times — or alongside the cleaner's own firing on
227        // GC — is safe. NOTE: `data class` copy() shares the handle
228        // reference between copies; if you intend to close independently,
229        // don't copy this class.
230        let mut body = KtCode::new();
231        for (fname, strategy) in &destructible_fields {
232            body = body.line(render_handle_close(strategy, fname));
233        }
234        class = class.member(KtFun::new("close").modifier("override").body(body));
235    }
236    // `fromParts` factory: the late native struct codec makes ONE
237    // `call_static_method` passing the whole graph's flattened leaf wires;
238    // this factory reassembles it (incl. nested `Child.fromParts(...)`) in
239    // JVM bytecode. `public`, not `internal`: an `internal` fun is mangled
240    // to `fromParts$<module>`, unresolvable by native (`NoSuchMethodError`).
241    // The factory body's raw-text class references (short names) carry their
242    // imports on this `Code`, so the whole `data class` is self-contained.
243    let mut factory_body = KtCode::new().line(factory_reconstruct);
244    for fqn in factory_imports {
245        factory_body = factory_body.import(fqn);
246    }
247    // Guarded only when a leaf actually is a raw pointer: a `fromParts` over
248    // plain scalars and byte arrays cannot forge anything, and marking it
249    // would delete a safe factory from Java and make unrelated Kotlin
250    // consumers opt into a raw-pointer contract it does not have.
251    let factory = KtFun::new("fromParts");
252    let mut factory = if factory_mints_handle {
253        ext.mark_unsafe(factory)
254    } else {
255        factory
256    }
257    .vis(KtVis::Public)
258    .annotation("JvmStatic")
259    .returns(KtType::cls(class_name))
260    .expr_body(factory_body);
261    for (name, ty) in &factory_params {
262        factory = factory.param(KtParam::new(name, ty.clone()));
263    }
264    class = class.companion(KtCompanion::new().vis(KtVis::Public).member(factory));
265    class
266}

268/// Render one typed-handle Kotlin source file. Pure-shell form (with a
269/// method hook that appends `ViaJNI` to methods of the handle class):
270///
271/// ```kotlin
272/// public class JNIFoo(initialPtr: Long) : NativeHandle(initialPtr) {
273///     public fun free() = free { freePtrViaJNI(it) }
274///     private external fun freePtrViaJNI(ptr: Long)
275/// }
276/// ```
277///
278/// When `promoted_functions` is non-empty, one extra instance method is
279/// appended per `#[prebindgen]` fn — the matching opaque first param
280/// (Rust type-key = `promoted_rust_key`) is dropped from the Kotlin
281/// signature, and its `withPtr` / `consume` wrapper uses the
282/// inherited [`NativeHandle`] scope.
283///
284/// The free-pointer extern name is built as
285/// `<mangle_method(package, class, "freePtr")>`. Kotlin/JVM's JNI name mangler binds it
286/// to the matching `Java_<pkg>_<class>_<mangled-freePtr>`
287/// extern on the Rust side (the auto-generated destructor).
288pub(crate) fn build_typed_handle(
289    ext: &Declarations,
290    registry: &Registry,
291    class_name: &str,
292    rust_doc_name: &str,
293    key: &TypeKey,
294    imports: &mut BTreeSet<String>,
295) -> KtClass {
296    // The typed handle is a pure shell — `ptr` slot + `close()`/`take()` +
297    // the `freePtr` extern. All functions are emitted as flat free functions
298    // in their namespace package; nothing is promoted onto the class.
299    //
300    // Every typed handle extends the shared `NativeHandle` base, which owns
301    // the `@Volatile` pointer slot (`ptr`) and its monitor — that common
302    // supertype is what lets `render_wrapper_fn` collect a `List<NativeHandle>`
303    // and lock it in one pointer-sorted, deadlock-safe pass. The subclass keeps
304    // its own type-specific `close()`/`take()`/`freePtr`.
305    let class_fqn = ext
306        .types
307        .get(key)
308        .and_then(|cfg| cfg.name_spec.as_ref())
309        .map(|spec| ext.fqn_of(spec))
310        .unwrap_or_else(|| class_name.to_string());
311    let (class_package, final_class_name) = class_fqn
312        .rsplit_once('.')
313        .unwrap_or(("", class_fqn.as_str()));
314    let free_extern = ext.mangle_method(class_package, final_class_name, "freePtr");
315    let gc_managed = ext
316        .types
317        .get(key)
318        .and_then(|cfg| cfg.opaque())
319        .is_some_and(|o| o.gc_managed);
320    let base_short = if gc_managed {
321        "GcNativeHandle"
322    } else {
323        "NativeHandle"
324    };
325    let base_fqn = if ext.package.is_empty() {
326        base_short.to_string()
327    } else {
328        format!("{}.{base_short}", ext.package)
329    };
330    let members = ext.class_members.get(key).map(Vec::as_slice).unwrap_or(&[]);
331    if !members.is_empty() && !ext.package.is_empty() {
332        imports.insert(format!("{}.{}", ext.package, ext.jni_native_class_name()));
333    }
334
335    // Companion object: the `@JvmStatic external fun freePtr(ptr: Long)` called
336    // by `close()`, plus one **factory** member per `.constructor(f, name)`
337    // (a free wrapper — no receiver — returning the class).
338    let mut companion = KtCompanion::new()
339        .vis(KtVis::Public)
340        .member(
341            // `@JvmSynthetic`: a `@JvmStatic external fun` is a public static
342            // native method, so `Storage.freePtr(0xdeadbeefL)` from Java would
343            // free an address of the caller's choosing.
344            KtFun::new(free_extern.clone())
345                .annotation("JvmStatic")
346                .annotation(JVM_SYNTHETIC)
347                .external()
348                .param(KtParam::new("ptr", KtType::long())),
349        )
350        // The replacement for the raw-pointer constructor. `internal` +
351        // `@JvmSynthetic` is reachable from generated Kotlin and from neither
352        // another Kotlin module nor Java; the constructor itself is `private`,
353        // because a constructor can be neither hidden by `@JvmSynthetic`
354        // (Kotlin rejects the target) nor by `internal` (still public on the
355        // JVM). Nothing on the Rust side constructs a handle.
356        .member(
357            internal_fun(HANDLE_FACTORY)
358                .kdoc(
359                    "Wrap a pointer a generated native call returned. Passing anything \
360                     else — a literal, a stale pointer, one belonging to another \
361                     handle — is undefined behaviour, which is why this is not part \
362                     of the public API.",
363                )
364                .param(KtParam::new("initialPtr", KtType::long()))
365                .returns(KtType::cls(class_name))
366                .expr_body(KtCode::new().line(format!("{class_name}(initialPtr)"))),
367        );
368    for m in members.iter().filter(|m| m.kind == MemberKind::Constructor) {
369        if let Some(item_fn) = registry.flat().function(&m.rust_ident) {
370            if let Some(f) = render_wrapper_fn(
371                ext,
372                item_fn,
373                registry,
374                Some(ext.effective_method_name(key, m).as_str()),
375                None,
376            ) {
377                for ov in render_param_overloads(ext, item_fn, registry, &f) {
378                    companion = companion.member(ov);
379                }
380                companion = companion.member(f);
381            }
382        }
383    }
384
385    // KDoc: the Rust struct's `///` prose first, framework line after.
386    let framework_line = format!("Typed handle for a native Zenoh `{rust_doc_name}`.");
387    let class_kdoc = source_item_doc(registry, key)
388        .map(|d| format!("{d}\n\n{framework_line}"))
389        .unwrap_or(framework_line);
390    // Consumer interfaces (`.implements`) and the generated `<Name>Api`
391    // interface (`.interface()`) are attached by `apply_class_interface` in
392    // `write_typed_handles` after the class body is built.
393    let mut class = KtClass::class_(class_name)
394        .vis(KtVis::Public)
395        // The class is public; minting one from a raw `Long` is not. `private`
396        // rather than `internal`, because an internal constructor is still a
397        // public JVM constructor — `new Storage(0xdeadbeefL).close()` compiled
398        // from Java. Every generated call site goes through the companion's
399        // `fromRawPtr` instead (see `handle_from_raw`).
400        .ctor_vis(KtVis::Private)
401        .kdoc(class_kdoc)
402        .ctor_param(KtCtorParam::new("initialPtr", KtType::long()));
403    class = if gc_managed {
404        // GC-managed lifecycle: the pointer lives in the inherited atomic
405        // cell; every release path settles the once-only untagged→tagged
406        // ticket via `releaseCell`, and the registered Cleaner action frees
407        // the box only if no other path won first. `clean()` on the explicit
408        // paths is eager deregistration (the action then no-ops on the
409        // already-tagged cell).
410        if !ext.package.is_empty() {
411            imports.insert(format!("{}.releaseCell", ext.package));
412            imports.insert(format!("{}.registerGcHandle", ext.package));
413        }
414        class
415            .extends(KtType::cls(base_fqn), Some("initialPtr"))
416            .member(
417                KtProperty::val("__cleanable")
418                    .vis(KtVis::Private)
419                    .initializer(format!("registerGcHandle(this) {{ {free_extern}(it) }}")),
420            )
421            .member(
422                KtFun::new("close")
423                    .annotation("Synchronized")
424                    .modifier("override")
425                    .body(
426                        KtCode::new()
427                            .line("val p = releaseCell(cell)")
428                            .line(format!("if (p != 0L) {free_extern}(p)"))
429                            .line("__cleanable?.clean()"),
430                    ),
431            )
432            .member(
433                KtFun::new("take")
434                    .vis(KtVis::Public)
435                    .annotation("Synchronized")
436                    .returns(KtType::cls(class_name))
437                    .body(
438                        KtCode::new()
439                            .line("val p = releaseCell(cell)")
440                            .line("__cleanable?.clean()")
441                            .line(format!(
442                                "return {}",
443                                handle_from_raw(class_name, "if (p != 0L) p else cell.get()")
444                            )),
445                    ),
446            )
447    } else {
448        class
449            .extends(KtType::cls(base_fqn), Some("initialPtr"))
450            .member(
451                KtFun::new("close")
452                    .annotation("Synchronized")
453                    .modifier("override")
454                    .body(
455                        KtCode::new()
456                            .line("val p = ptr")
457                            .blk("if (p != 0L && (p and 1L) == 0L) {", |c| {
458                                c.line("ptr = p or 1L").line(format!("{free_extern}(p)"))
459                            }),
460                    ),
461            )
462            // Transfer ownership of the native pointer into a fresh handle,
463            // leaving this one empty. Lets a callback receiver retain a handle
464            // that the framework would otherwise `close()` when the callback
465            // returns.
466            .member(
467                KtFun::new("take")
468                    .vis(KtVis::Public)
469                    .annotation("Synchronized")
470                    .returns(KtType::cls(class_name))
471                    .body(
472                        KtCode::new()
473                            .line("val p = ptr")
474                            .line("ptr = p or 1L")
475                            .line(format!("return {}", handle_from_raw(class_name, "p"))),
476                    ),
477            )
478    };
479    let mut class = class.companion(companion);
480
481    // Promoted instance methods: each `.method(f)` becomes an instance method
482    // (receiver bound to `this`), delegating to the same centralized
483    // `JNINative` extern as a free wrapper would.
484    for m in members.iter().filter(|m| m.kind == MemberKind::Method) {
485        if let Some(item_fn) = registry.flat().function(&m.rust_ident) {
486            if let Some(f) = render_wrapper_fn(
487                ext,
488                item_fn,
489                registry,
490                Some(ext.effective_method_name(key, m).as_str()),
491                Some(key),
492            ) {
493                for ov in render_param_overloads(ext, item_fn, registry, &f) {
494                    class = class.member(ov);
495                }
496                class = class.member(f);
497            }
498        }
499    }
500    class
501}

510/// The JNINative `external fun <method>(…): <wire-return>` for one bound
511/// function, as a `KtFun` (the AST renderer shortens types + collects imports
512/// + wraps long signatures uniformly). Wire level (matches the Rust extern):
513///   * opaque-handle (Borrow/Consume) → jlong → `Long`
514///   * `enum_class`                  → jint  → `Int` (call passes `.value`)
515///   * `Any` (impl-Into Dispatch)     → JObject → `Any`
516///   * everything else                → the entry's high-level Kotlin type
517///
518/// Opaque returns become `Long`; every other return uses [`classify_return`]'s
519/// `kt_return` (Unit is no return type). `None` if a param's converter isn't
520/// resolved. Full-FQN types throughout — no derivation-time shortening.
521pub(crate) fn render_extern_decl(
522    ext: &Declarations,
523    f: &prebindgen_registry::flat::Function,
524    registry: &Registry,
525) -> Option<KtFun> {
526    // The name and wire params come straight off the lowered plan — the
527    // same classification the Rust extern and the Kotlin call site consume,
528    // so the three sites agree on arity, types, and symbol by construction.
529    let fplan = ext.fn_plan(registry, f).ok()?;
530    let jni_call = &fplan.jni_method;
531    let mut params: Vec<KtParam> = Vec::new();
532    for leaf in fplan.leaves() {
533        for native in leaf.native.iter() {
534            params.push(KtParam::new(
535                native.kt_name.clone(),
536                native.kt_wire.clone()?,
537            ));
538        }
539    }
540    // Output (data) expansion: a **callback** delivery appends the lambda(s)
541    // before the error sink and returns the erased `Any?`. A **return**
542    // delivery appends nothing and returns the real converted wire below.
543    if let FnOutputPlan::Unfold(u) = &fplan.output {
544        if u.iterable_fold {
545            // `acc` is the unbounded accumulator `A` (may be nullable) → `Any?`;
546            // `fold` is the non-null adapter callback.
547            params.push(KtParam::new("acc", KtType::any().nullable()));
548            params.push(KtParam::new("fold", KtType::any()));
549        } else {
550            params.push(KtParam::new("build", KtType::any()));
551        }
552    }
553    // Trailing error-sink callbacks — the binding channel (`errorSink`) always,
554    // then the typed domain channel (`domainSink`) for a fallible-typed fn. Both
555    // erased to `Any` (JObject) on the wire; the wrapper passes a capture for
556    // each. A domain plan ⇒ `error_plans` has this fn.
557    params.push(KtParam::new("errorSink", KtType::any()));
558    if fplan.error.is_some() {
559        params.push(KtParam::new("domainSink", KtType::any()));
560    }
561
562    let wire_return: Option<KtType> = match &fplan.output {
563        FnOutputPlan::Unfold(_) => Some(KtType::any().nullable()),
564        FnOutputPlan::Value(v) => {
565            // The plan classified the declared surface once — `convert_out_ty`
566            // for a `convert_output` (Return), else the function's own return.
567            let (kt_return, projection) = render_return_surface(&v.surface)?;
568            // JNI extern's wire return: projections state their folded wire;
569            // otherwise a primitive destination comes directly from the
570            // frozen Rust-side conversion plan. This ordering matters for
571            // nested optionals: enum niches can collapse every Optional layer
572            // to one `jint`, although the deliberately lossy public surface is
573            // still `Priority?`. Object wires retain their declared reference
574            // type so the wrapper receives the useful boxed/static type.
575            match &projection {
576                Some(p) => Some(projection_wire_return(p)),
577                None if JniPrim::from_wire(v.pipeline.wire()).is_some() => {
578                    kotlin_for_wire(v.pipeline.wire())
579                }
580                None if v.is_enum => Some(KtType::int()),
581                None if v.is_option_enum => Some(KtType::int().nullable()),
582                None => kt_return,
583            }
584        }
585    };
586
587    let mut fun = KtFun::new(jni_call).external();
588    for p in params {
589        fun = fun.param(p);
590    }
591    if let Some(rt) = wire_return {
592        fun = fun.returns(rt);
593    }
594    Some(fun)
595}

597struct Param {
598    kt_name: String,
599    kt_type: KtType,
600    mode: ParamMode,
601    /// `true` when the param's Rust type is a `enum_class`-declared enum: the
602    /// high-level Kotlin signature uses the typed enum (`Priority`), but the
603    /// underlying JNI `external fun` declares the param as `Int` (jint wire).
604    /// The wrapper bridges the two by passing `<name>.value` at the call site.
605    as_enum_value: bool,
606    /// Primitive sentinel used when an optional enum parameter is absent.
607    enum_niche: Option<String>,

610enum ParamMode {
611    Borrow,  // &T opaque-handle → withPtr
612    Consume, // T  opaque-handle → consume
613    /// `Option<&T>` / `Option<&mut T>` opaque-handle → `withPtrOrZero`.
614    /// Nullable typed-handle param; the wrapper runs the body under the read
615    /// lock when the handle is non-null and with `0L` when null. The Rust
616    /// converter materializes `Option<OwnedObject<T>>` and the call site uses
617    /// `.as_deref()` / `.as_deref_mut()`.
618    BorrowNullable,
619    /// By-value `Option<T>` opaque-handle → nullable consume. Same as
620    /// [`Self::Consume`] but the typed param is nullable (`T?`): `0L` when
621    /// absent, and the handle's `ptr` slot is nulled after the call only when
622    /// present. The Rust converter consumes the `Box` to `Option<T>`.
623    ConsumeNullable,
624    PassThrough,
625    /// Kotlin `ULong` projected to its raw JNI `Long` bit pattern.
626    Unsigned64 {
627        niche: Option<String>,
628    },
629    /// Flattenable `data_class` param: the high-level Kotlin signature keeps the
630    /// typed object, but the `JNINative` call destructures it into the leaf
631    /// access expressions (no `JObject` crosses, so the Rust side skips
632    /// `env.get_field(...)`). The strings are the per-leaf call-site
633    /// expressions in plan order.
634    FlattenStruct {
635        accesses: Vec<String>,
636        /// Owned handles nested anywhere below the flattened root. They join
637        /// the wrapper's one sorted lock set and are marked consumed after
638        /// the native call, exactly like top-level by-value handles.
639        handles: Vec<Opaque>,
640    },
641    /// `&[T]` / `Vec<T>` of a flattenable data_class `T`: the public Kotlin
642    /// signature keeps `List<T>`, but the wrapper allocates a Rust-side `Vec<T>`
643    /// (opaque `jlong` handle), pushes each element's decoupled leaves in a loop
644    /// via `<base>Push`, passes the handle to the extern, then frees it in a
645    /// `finally`. No `JObject` list crosses, so the Rust side skips per-element
646    /// `env.get_field(...)`. `base` is the helper method base (`payloadVec`);
647    /// `elem_accesses` are the per-element leaf push expressions rooted at the
648    /// loop variable (`__e.id`, `__e.seq`, …), in plan order.
649    VecBuild {
650        base: String,
651        elem_accesses: Vec<String>,
652    },
653    /// `impl Fn(args)` callback param: typed Kotlin lambda over the flattened
654    /// leaves of each arg's callback plan (whole arg when plan-less), erased to
655    /// `Any` at the extern tier — the same shape as the unfold `build`/`onError`
656    /// lambdas. `call_arg` is the call-site expression.
657    Callback {
658        call_arg: String,
659    },
660    /// Bare `Option<primitive>` / `Option<enum>` param decomposed into a
661    /// `(present: Boolean, value: <prim>)` pair so no boxed `java.lang.*`
662    /// crosses (and the Rust side does no `intValue()` unboxing). The public
663    /// Kotlin signature keeps `T?`; the call site passes `present_expr`
664    /// (`<name> != null`) then `value_expr` (`<name> ?: 0` / `<name>?.value ?:
665    /// 0`). See [`crate::jni::compile::OptionalPairPlan`].
666    OptionalPair {
667        present_expr: String,
668        value_expr: String,
669    },
670}

709/// Build a single top-level (free-function) wrapper as a [`KtFun`].
710/// Returns `None` if the function has a parameter whose Kotlin type isn't
711/// registered (in that case we skip the function rather than panicking — the
712/// legacy `JNINative.kt` retains the unwrapped external fun so callers still
713/// have an escape hatch).
714///
715/// Every `#[prebindgen]` function is emitted as a flat namespaced free function
716/// — opaque-handle parameters are ordinary `NativeHandle` params, locked via the
717/// per-call `withSortedHandleLocks` scaffold.
718///
719/// When `receiver_key` is `Some(class_key)` the function is emitted as an
720/// **instance method** of that class: the first parameter whose (peeled) Rust
721/// type equals `class_key` is dropped from the signature and bound to `this`
722/// (the inherited `NativeHandle` scope for a `ptr_class` — `this.ptr` + lock).
723/// The JNINative extern/call is
724/// unchanged (keyed on the Rust ident), so only the Kotlin wrapper relocates.
725/// The Kotlin surface of a wrapper: the assembled `KtFun` with every
726/// parameter/return type in place but **no body**, plus the emission
727/// internals [`render_wrapper_fn`] needs to fill that body. One derivation of
728/// the overload surface, shared by emission (which adds the body) and
729/// [`validate_symbols`](crate::jni::validate_symbols)
730/// (which erases `fun` to a JVM signature), so the emitted overload and the
731/// validated one cannot drift (issue #89).
732pub(crate) struct WrapperSurface {
733    /// The wrapper with its full signature and an empty body. The validator
734    /// reads this; [`render_wrapper_fn`] fills the body and adds the KDoc.
735    pub fun: KtFun,
736    // Emission-only internals — computed while assembling the signature,
737    // consumed by `render_wrapper_fn`; opaque to the validator.
738    params: Vec<Param>,
739    out: OutputPlan,
740    sink: ErrorSink,
741    jni_call: String,
742    /// FQNs the wrapper **body** references by short name (extension `asRaw`,
743    /// hoisted singletons, the error-capture holder). Signature-type imports
744    /// are NOT here — those are full-FQN `KtType`s in `fun`, collected by the
745    /// render-time `ImportSet`. `render_wrapper_fn` attaches these to the
746    /// body `Code`, so the emitted `KtFun` is self-contained; the validator
747    /// ignores them.
748    body_imports: BTreeSet<String>,
749}

751/// Whether an error handler may return null when it cannot manufacture the
752/// value a failed call was meant to produce.
753#[derive(Clone, Copy)]
754enum RecoveryReturn {
755    /// Public wrappers: reference-shaped returns become nullable. Types whose
756    /// nullable form has a different JVM representation (primitives and
757    /// ULong) keep their declared type.
758    NullableReferences,
759    /// Constant helpers always install a throwing handler themselves, so they
760    /// retain the constant's declared type instead of leaking nullability into
761    /// the public property.
762    Declared,
763}
764
765/// The type returned by both error handlers and, consequently, by the wrapper
766/// itself. A handler may decline to fabricate a reference result by returning
767/// null; primitive-shaped results keep their existing unboxed contract.
768fn recovery_return_type(out: &OutputPlan, policy: RecoveryReturn) -> KtType {
769    let declared = out.kt_return.clone().unwrap_or_else(KtType::unit);
770    if matches!(policy, RecoveryReturn::Declared) {
771        return declared;
772    }
773    let generics: Vec<String> = out.generic.iter().cloned().collect();
774    nullable_recovery_type(declared, &generics)
775}
776
777fn nullable_recovery_type(declared: KtType, generics: &[String]) -> KtType {
778    if declared.is_nullable() || declared == KtType::unit() {
779        return declared;
780    }
781    let nullable = declared.clone().nullable();
782    if crate::jni::symbols::erase_kt_type(generics, &declared)
783        == crate::jni::symbols::erase_kt_type(generics, &nullable)
784    {
785        nullable
786    } else {
787        declared
788    }
789}

801    registry: &Registry,
802    kotlin_name_override: Option<&str>,
803    receiver_key: Option<&TypeKey>,
804) -> Option<WrapperSurface> {
805    build_wrapper_surface_with_recovery(
806        ext,
807        f,
808        registry,
809        kotlin_name_override,
810        receiver_key,

815fn build_wrapper_surface_with_recovery(
816    ext: &Declarations,
817    f: &prebindgen_registry::flat::Function,
818    registry: &Registry,
819    kotlin_name_override: Option<&str>,
820    receiver_key: Option<&TypeKey>,
821    recovery: RecoveryReturn,
822) -> Option<WrapperSurface> {
823    let mut body_imports = BTreeSet::new();
824    let fplan = ext.fn_plan(registry, f).ok()?;
825    // The Kotlin extern in `JNINative` is keyed on the Rust ident (the
826    // plan's `jni_method`). The per-entry `.name("...")` override only
827    // changes the *user-facing* Kotlin wrapper name; the JNI call still has
828    // to hit the one extern that the Rust extern actually emits.
829    let kt_name = match kotlin_name_override {
830        Some(n) => n.to_string(),
831        None => kt_snake_to_camel(&f.name.to_string()),
832    };
833    let jni_call = fplan.jni_method.clone();
834    let (params, receiver_idx) = classify_params(&fplan, &mut body_imports, receiver_key)?;
835    let out = classify_output(ext, &fplan, &mut body_imports)?;
836    let r_ty = recovery_return_type(&out, recovery);
837    let sink = error_sink_parts(&fplan, &mut body_imports, &r_ty)?;
838
839    let mut fun = KtFun::new(&kt_name).vis(KtVis::Public);
840    if let Some(g) = &out.generic {
841        fun = fun.generic(g);
842    }
843    for (i, p) in params.iter().enumerate() {
844        // The receiver param is bound to `this` — not a rendered parameter.
845        if Some(i) == receiver_idx {
846            continue;
847        }
848        fun = fun.param(KtParam::new(&p.kt_name, p.kt_type.clone()));
849    }
850    // The error callbacks — **required**: the generated code never throws; the
851    // consumer decides how a failure surfaces (e.g. by throwing its own type).
852    // Every wrapper takes the binding handler; a fallible-typed fn additionally
853    // takes the domain handler, ordered `onBindingError, onError` so the DOMAIN
854    // `onError` is the natural trailing lambda. When an output-expansion
855    // builder/fold lambda exists it must stay the **trailing** lambda, so the
856    // error params go *before* it — but *after* any non-lambda `builder_lead`
857    // (`acc: A`), which is passed positionally.
858    let mut err_params = vec![KtParam::new(&sink.binding_param, sink.binding_type.clone())];
859    if let Some(d) = &sink.domain {
860        err_params.push(KtParam::new("onError", d.onerr_type.clone()));
861    }
862    if let Some((bp_name, bp_ty)) = &out.builder_param {
863        if let Some((lead_name, lead_ty)) = &out.builder_lead {
864            fun = fun.param(KtParam::new(lead_name, lead_ty.clone()));
865        }
866        for ep in err_params {
867            fun = fun.param(ep);
868        }
869        fun = fun.param(KtParam::new(bp_name, bp_ty.clone()));
870    } else {
871        for ep in err_params {
872            fun = fun.param(ep);
873        }
874    }
875    if out.cast_return {
876        fun = fun.annotation("Suppress(\"UNCHECKED_CAST\")");
877    }
878    if out.kt_return.is_some() {
879        fun = fun.returns(r_ty);
880    }
881    Some(WrapperSurface {
882        fun,
883        params,
884        out,
885        sink,
886        jni_call,
887        body_imports,
888    })
889}

908fn render_wrapper_fn_with_recovery(
909    ext: &Declarations,
910    f: &prebindgen_registry::flat::Function,
911    registry: &Registry,
912    kotlin_name_override: Option<&str>,
913    receiver_key: Option<&TypeKey>,
914    recovery: RecoveryReturn,
915) -> Option<KtFun> {
916    let surface = build_wrapper_surface_with_recovery(
917        ext,
918        f,
919        registry,
920        kotlin_name_override,
921        receiver_key,
922        recovery,
923    )?;
924    let WrapperSurface {
925        mut fun,
926        params,
927        out,
928        sink,
929        jni_call,
930        mut body_imports,
931    } = surface;
932    // KDoc: the Rust fn's `///` prose first, then generated notes for every
933    // position an expansion reshaped away from the Rust signature (N1).
934    // Emission-only — the validator skips it.
935    let fplan = ext.fn_plan(registry, f).ok()?;
936    if let Some(doc) = wrapper_kdoc(f, &fplan) {
937        fun = fun.kdoc(doc);
938    }
939    // Collect the opaque-handle params so we can scaffold pointer-ordered
940    // synchronized blocks around them.
941    let opaques = collect_opaques(&params);
942    let is_unit = fun.ret.is_none();
943    let body_expr = build_native_call(ext, &jni_call, &params, &out, &sink);
944    let return_mode = if is_unit {
945        BodyReturn::Unit
946    } else {
947        BodyReturn::Value(build_success_return(ext, &out, "__ret"))
948    };
949    // `render_body` extends `body_imports` with the body's own references; all
950    // of them are attached to the body `Code`, so the emitted `KtFun` carries
951    // its own imports (signature imports ride the FQN types via the render-time
952    // `ImportSet`). The wrapper is thus a self-contained decl — callers no
953    // longer thread a per-file import set through it.
954    let mut body = render_body(
955        ext,
956        &params,
957        &opaques,
958        &sink,
959        &body_expr,
960        &return_mode,
961        &mut body_imports,
962    );
963    for fqn in body_imports {
964        body = body.import(fqn);
965    }
966    Some(fun.body(body))
967}

1092    ext: &Declarations,
1093    registry: &Registry,
1094    mut helper: KtFun,
1095    val_name: String,
1096    kdoc: String,
1097    imports: &mut BTreeSet<String>,
1098) -> Option<(KtFun, KtProperty)> {
1099    helper.vis = KtVis::Private;
1100    let helper_name = helper.name.clone();
1101    // A constant always carries a value type; a helper with no return would
1102    // mean the type never resolved — skip like an unresolvable fn.
1103    let val_ty = helper.ret.clone()?;
1104    let spec = ext.iface_spec(registry, &SpecKey::JniErrorHandler)?;
1105    imports.insert(spec.fqn());
1106    let init = format!(
1107        "{helper_name}(JniErrorHandler {{ je -> error(je ?: \"const {val_name}: JNI getter failed\") }})"
1108    );
1109    let prop = KtProperty::val(&val_name)
1110        .ty(val_ty)
1111        .vis(KtVis::Public)
1112        .delegate(format!("lazy {{ {init} }}"))
1113        .kdoc(kdoc);
1114    Some((helper, prop))
1115}

1176/// Type every effective input of the lowered [`JniFunctionPlan`] into a
1177/// [`Param`] (Kotlin name/type + call-site [`ParamMode`]). The crossing-form
1178/// operation comes from the plan; this site only renders each
1179/// [`KotlinParamOp`] into its Kotlin surface and call form. Returns the params
1180/// plus the index of the
1181/// instance-method receiver (the first param whose peeled type matches
1182/// `receiver_key`), which is bound to `this` and dropped from the signature.
1183fn classify_params(
1184    fplan: &JniFunctionPlan,
1185    imports: &mut BTreeSet<String>,
1186    receiver_key: Option<&TypeKey>,
1187) -> Option<(Vec<Param>, Option<usize>)> {
1188    let mut receiver_idx: Option<usize> = None;
1189    let mut params: Vec<Param> = Vec::new();
1190    for leaf in fplan.leaves() {
1191        let mut name = leaf.kt_name.clone();
1192
1193        // Instance-method receiver: the first parameter whose peeled Rust type
1194        // is the owning class binds to `this` (so `this_ptr`/`this.ptr`/lock or
1195        // `this.bytes` fall out of the normal param handling) and is dropped
1196        // from the rendered signature.
1197        if receiver_idx.is_none() {
1198            if let Some(rk) = receiver_key {
1199                if &peel_receiver_key(&leaf.reading) == rk {
1200                    receiver_idx = Some(params.len());
1201                    name = "this".to_string();
1202                }
1203            }
1204        }
1205
1206        // `impl Fn(args)` param: a generated typed `fun interface`
1207        // (`<ArgShorts>Callback`) whose `run` parameters are the flattened
1208        // leaves of each arg's callback plan (the arg whole when plan-less).
1209        // The extern receives it erased (`Any`) and the native trampoline
1210        // calls the typed `run`, so no call-site adapter exists.
1211        // Lambda-literal call sites SAM-convert unchanged.
1212        if let KotlinParamOp::Callback { iface, .. } = &leaf.kotlin {
1213            let spec = iface.as_deref()?;
1214            let kt_type = spec.kt_ref(vec![]);
1215            // The extern receives the RAW twin: the generated `asRaw()`
1216            // proxy (built once per registration) wraps raw leaves into the
1217            // typed objects the user's interface declares.
1218            let call_arg = if spec.needs_raw() {
1219                imports.insert(format!("{}.asRaw", spec.package));
1220                format!("{name}.asRaw()")
1221            } else {
1222                name.clone()
1223            };
1224            params.push(Param {
1225                kt_name: name.clone(),
1226                kt_type,
1227                mode: ParamMode::Callback { call_arg },
1228                as_enum_value: false,
1229                enum_niche: None,
1230            });
1231            continue;
1232        }
1233
1234        // Typed surface: the projection's Kotlin FQN for opaque handles /
1235        // value projections (any `Option<_>` layer is nullable purely on the
1236        // typed-wrapper surface — the handle wire stays `jlong` with `0` =
1237        // absent), else the resolved entry's Kotlin name. `None` (unresolved
1238        // name) skips the wrapper — the escape-hatch path.
1239        let kt_type_raw = leaf.kt_public.clone()?;
1240
1241        // Map the plan's crossing form to the Kotlin call-site mode.
1242        let mode = match &leaf.kotlin {
1243            KotlinParamOp::VecBuild { helpers, .. } => {
1244                // Slice/Vec of a flattenable data_class: build the Rust-side
1245                // Vec by pushing each element's leaves, pass the handle (see
1246                // the body direction + `build_vec_build_helper_items`). The
1247                // high-level signature stays `List<T>` (registered below).
1248                let elem_accesses = helpers
1249                    .plan
1250                    .leaves
1251                    .iter()
1252                    .filter(|l| !l.is_present_flag())
1253                    .map(|l| l.kt_access("__e"))
1254                    .collect();
1255                ParamMode::VecBuild {
1256                    base: helpers.base.clone(),
1257                    elem_accesses,
1258                }
1259            }
1260            KotlinParamOp::OptionalPair(sp) => {
1261                // Bare `Option<primitive>` / `Option<enum>`: cross as a
1262                // `(present, value)` pair (no boxed object). The high-level
1263                // signature keeps `T?`; only the call-site args split in two.
1264                let present_expr = format!("{name} != null");
1265                let value_expr = if sp.is_enum {
1266                    format!("{name}?.value ?: {}", sp.value_kt_zero)
1267                } else {
1268                    format!("{name} ?: {}", sp.value_kt_zero)
1269                };
1270                ParamMode::OptionalPair {
1271                    present_expr,
1272                    value_expr,
1273                }
1274            }
1275            KotlinParamOp::FlattenStruct(plan) => {
1276                let handles = plan
1277                    .leaves
1278                    .iter()
1279                    .filter_map(|leaf| {
1280                        // Through the leaf's own access template rather than a
1281                        // reconstructed `<name>.<field>`, so the lock target is
1282                        // whatever expression the leaf actually reads from.
1283                        // No live case needs it yet — `build_flat_sum_field`
1284                        // returns `None` for a projection payload, so a
1285                        // tag-gated handle leaf cannot reach here — but the
1286                        // template is the leaf's own answer to "where did this
1287                        // come from", and deriving it twice is how the two
1288                        // drift apart.
1289                        let target = leaf.kt_handle_target(&name)?;
1290                        let consume_null = if leaf.handle_nullable() {
1291                            format!("{target}?.markConsumed()")
1292                        } else {
1293                            format!("{target}.markConsumed()")
1294                        };
1295                        Some(Opaque {
1296                            name: leaf.kt_name.clone(),
1297                            target,
1298                            consume_null: Some(consume_null),
1299                            nullable: leaf.handle_nullable(),
1300                            // A flattened leaf carries no Kotlin class name of
1301                            // its own, so its domain is unknown and it compares
1302                            // against every other handle — the fail-safe
1303                            // direction (an extra `ptr` comparison, never a
1304                            // missed alias).
1305                            domain: None,
1306                        })
1307                    })
1308                    .collect();
1309                ParamMode::FlattenStruct {
1310                    accesses: plan.leaves.iter().map(|l| l.kt_call_arg(&name)).collect(),
1311                    handles,
1312                }
1313            }
1314            KotlinParamOp::Handle { mode, .. } => match mode {
1315                HandleMode::Borrow => ParamMode::Borrow,
1316                HandleMode::Consume => ParamMode::Consume,
1317                HandleMode::BorrowNullable => ParamMode::BorrowNullable,
1318                HandleMode::ConsumeNullable => ParamMode::ConsumeNullable,
1319            },
1320            KotlinParamOp::Unsigned64 { niche } => ParamMode::Unsigned64 {
1321                niche: niche.clone(),
1322            },
1323            KotlinParamOp::Plain => ParamMode::PassThrough,
1324            KotlinParamOp::Callback { .. } => unreachable!("callback params handled above"),
1325        };
1326
1327        // Full-FQN surface type — the render-time `ImportSet` shortens it and
1328        // collects the import from the AST (issue #89 follow-up: decouple the
1329        // signature from the derivation-time import side-effect). No manual
1330        // registration here.
1331        let kt_type = if leaf.optional {
1332            kt_type_raw.nullable()
1333        } else {
1334            kt_type_raw
1335        };
1336        params.push(Param {
1337            kt_name: name,
1338            kt_type,
1339            mode,
1340            as_enum_value: leaf.as_enum_value,
1341            enum_niche: leaf.enum_niche.clone(),
1342        });
1343    }
1344    Some((params, receiver_idx))
1345}

1347/// Classify the output side into an [`OutputPlan`].
1348///
1349/// Output (data) expansion: the return value is delivered to a caller
1350/// callback per shape:
1351///   * `Decompose`/`Optional` (M1–M3): decompose into leaves → `build:
1352///     (L0, …) -> R` once; `<R>`, returns `R` / `R?`.
1353///   * `Iterable` (M4 whole / M5 decomposed): per element, fold
1354///     `(acc, leaves…) -> acc`; `<A>`, returns `A`, threads the accumulator.
1355///
1356/// Each leaf is delivered with its final Kotlin type; a leaf whose typed form
1357/// can't be constructed Rust-side makes the wrapper install an **adapter** that
1358/// applies the Kotlin-side projection wrap before the user callback. Leaves
1359/// with no such projection ⇒ the callback is passed directly (M1–M4
1360/// unchanged).
1361fn classify_output(
1362    ext: &Declarations,
1363    fplan: &JniFunctionPlan,
1364    imports: &mut BTreeSet<String>,
1365) -> Option<OutputPlan> {
1366    let unfold = fplan.unfold.as_ref();
1367    // `builder_param` is the trailing **lambda** param (build / fold) as a
1368    // `(name, function-type)` pair. For the `Iterable` shape, the non-lambda
1369    // accumulator (`acc: A`) goes in `builder_lead` — it must precede
1370    // `onError` (a defaulted param) so the positional-`acc` call stays valid;
1371    // the trailing `fold` lambda follows.
1372    let mut builder_param: Option<(String, KtType)> = None;
1373    let mut builder_lead: Option<(String, KtType)> = None;
1374    let mut generic: Option<String> = None;
1375    // Extra call-site args injected before `__sink` (e.g. `build`/adapter, or
1376    // `acc` + the fold callback/adapter for `Iterable`).
1377    let mut unfold_call_args: Vec<String> = Vec::new();
1378
1379    let (kt_return, projection) = if let FnOutputPlan::Value(v) = &fplan.output {
1380        // `convert_output` (Return) and plain returns: the wrapper returns
1381        // the value directly — the plan classified the declared surface once
1382        // (`convert_out_ty` for a convert, the signature's own output
1383        // otherwise). No callback param, no generic, no extra call args; the
1384        // extern returns the real wire and `build_call` applies the
1385        // projection wrap (handle) below.
1386        render_return_surface(&v.surface)?
1387    } else if matches!(&fplan.output, FnOutputPlan::Unfold(u) if u.fixed_builder) {
1388        let FnOutputPlan::Unfold(u) = &fplan.output else {
1389            unreachable!()
1390        };
1391        let plan = unfold?;
1392        // Synthesized by-value `data_class` delivery via a **fixed, hoisted
1393        // singleton** — the wrapper takes no caller `build`/`fold` param and is
1394        // not generic over `R`/`A`. The native side still receives the singleton
1395        // as an erased `Any` and calls its cached `run`, so the whole delivery
1396        // machinery is reused; only the Rust-side object construction is gone.
1397        // The concrete return is the data class (`T` / `Option<T>`) or, for a
1398        // `Vec<data_class>` fold, a `List<Class>` composed on the Kotlin side.
1399        // The concrete element/return Kotlin type. For a decomposed `data_class`
1400        // builder/fold it is the data class (`plan.source`'s registered FQN); for
1401        // a **whole-element leaf** fold (`plan.element` set — String / handle)
1402        // `plan.source` (e.g. `String`) has no class FQN, so take the
1403        // element's typed view from the folder interface's element param instead.
1404        // Full-FQN class type: the render-time `ImportSet` shortens it (and
1405        // handles simple-name collisions) when it renders the return type. The
1406        // `ArrayList<…>` body arg below uses the short name — the class is
1407        // imported via that return type, so the short name resolves.
1408        let class_ty = if u.whole_element {
1409            let spec = u.iface.as_deref()?;
1410            spec.params[1].typed.clone()
1411        } else {
1412            let class_fqn = ext.kotlin_fqn(&plan.source.key()).map(|s| s.to_string())?;
1413            KtType::cls(class_fqn)
1414        };
1415        let class_short = kt_type_short(&class_ty);
1416        if u.iterable_fold {
1417            // `Vec<data_class>` fold: allocate an `ArrayList<Class>` accumulator,
1418            // pass the hoisted **folder-appender** singleton as `fold` (it
1419            // rebuilds each element via `fromParts` and appends it), and return
1420            // the threaded accumulator as `List<Class>` (`?`-nullable for an
1421            // `Option<Vec<…>>` return — `None` yields a null list). Per element
1422            // only the raw leaves cross — no Java object is built on the Rust side.
1423            let spec = u.iface.as_deref()?;
1424            let holder = spec.singleton_holder_name();
1425            let field = crate::jni::SINGLETON_FIELD;
1426            imports.insert(spec.singleton_holder_fqn());
1427            unfold_call_args.push(format!("ArrayList<{class_short}>()"));
1428            unfold_call_args.push(format!("{holder}.{field}"));
1429            let list_ty = KtType::generic("List", [class_ty]);
1430            let kt = if u.optional {
1431                list_ty.nullable()
1432            } else {
1433                list_ty
1434            };
1435            (Some(kt), None)
1436        } else {
1437            // Scalar: the hoisted `__<Name>Builder` singleton calls `fromParts`;
1438            // the wrapper returns the concrete class (`?`-nullable for `Option`).
1439            let spec = u.iface.as_deref()?;
1440            let singleton = format!("__{}", spec.raw_name());
1441            imports.insert(format!("{}.{singleton}", spec.package));
1442            unfold_call_args.push(singleton);
1443            let kt = if u.optional {
1444                class_ty.nullable()
1445            } else {
1446                class_ty
1447            };
1448            (Some(kt), None)
1449        }
1450    } else if let (FnOutputPlan::Unfold(u), Some(_)) = (&fplan.output, unfold) {
1451        // The builder / fold params are generated typed `fun interface`s
1452        // (`<Source>Builder<out R>` / `<Element>Folder<A>`); the native side
1453        // calls their typed `run` with raw jvalues (no call-site adapter).
1454        // Lambda-literal call sites SAM-convert unchanged.
1455        generic = u.generic.map(str::to_string);
1456        // An `Iterable` fold — bare or `Optional`-wrapped — folds with `<A>`
1457        // (`acc` lead + `fold` lambda). The wrapped form returns `A?`: `None`
1458        // skips the fold and delivers null (matching the scalar `R?` and the
1459        // fixed path's null `List`), `Some(empty)` returns `acc` unchanged.
1460        if u.generic == Some("A") {
1461            let spec = u.iface.as_deref()?;
1462            builder_lead = Some(("acc".to_string(), KtType::var_("A")));
1463            builder_param = Some(("fold".to_string(), spec.kt_ref(vec![KtType::var_("A")])));
1464            unfold_call_args.push("acc".to_string());
1465            if spec.needs_raw() {
1466                imports.insert(format!("{}.asRaw", spec.package));
1467                unfold_call_args.push("fold.asRaw()".to_string());
1468            } else {
1469                unfold_call_args.push("fold".to_string());
1470            }
1471            let kt = if u.optional {
1472                KtType::var_("A").nullable()
1473            } else {
1474                KtType::var_("A")
1475            };
1476            (Some(kt), None)
1477        } else {
1478            let spec = u.iface.as_deref()?;
1479            builder_param = Some(("build".to_string(), spec.kt_ref(vec![KtType::var_r()])));
1480            if spec.needs_raw() {
1481                imports.insert(format!("{}.asRaw", spec.package));
1482                unfold_call_args.push("build.asRaw()".to_string());
1483            } else {
1484                unfold_call_args.push("build".to_string());
1485            }
1486            let kt = if u.optional {
1487                KtType::var_r().nullable()
1488            } else {
1489                KtType::var_r()
1490            };
1491            (Some(kt), None)
1492        }
1493    } else {
1494        unreachable!("FnOutputPlan is either Value or Unfold-with-plan")
1495    };
1496    // enum_class returns cross the JNI wire as jint → Kotlin `Int` (`Int?`
1497    // niche-backed for `Option<enum>`) — so `build_call` can wrap the result
1498    // with `fromInt`. The plan's probes run over the convert-peeled declared type;
1499    // the wrapper surface keeps the historical `unfold.is_none()` mask
1500    // (`Value` ∧ ¬`is_convert` ⟺ no unfold plan).
1501    let (is_enum_return, is_option_enum_return, enum_niches) = match &fplan.output {
1502        FnOutputPlan::Value(v) if !v.is_convert => {
1503            (v.is_enum, v.is_option_enum, v.enum_niches.clone())
1504        }
1505        _ => (false, false, Vec::new()),
1506    };
1507
1508    Some(OutputPlan {
1509        kt_return,
1510        projection,
1511        builder_param,
1512        builder_lead,
1513        generic,
1514        unfold_call_args,
1515        cast_return: matches!(&fplan.output, FnOutputPlan::Unfold(_)),
1516        is_enum_return,
1517        is_option_enum_return,
1518        enum_niches,
1519    })
1520}

1522/// Build the raw JNINative call expression. Every param maps to exactly one
1523/// call arg (or several, for a flattened data_class); the output plan's extra
1524/// args and the trailing `__cap` follow. Kotlin-side result projections are
1525/// deliberately deferred to [`build_success_return`], after the native error
1526/// captures have been checked.
1527fn build_native_call(
1528    ext: &Declarations,
1529    jni_call: &str,
1530    params: &[Param],
1531    out: &OutputPlan,
1532    sink: &ErrorSink,
1533) -> String {
1534    let mut args: Vec<String> = Vec::with_capacity(params.len());
1535    for p in params.iter() {
1536        // Flattened data_class param expands into multiple call args
1537        // (the leaf destructure expressions, in plan order).
1538        if let ParamMode::FlattenStruct { accesses, .. } = &p.mode {
1539            args.extend(accesses.iter().cloned());
1540            continue;
1541        }
1542        // VecBuild param: the extern receives the `jlong` Vec handle the
1543        // wrapper body allocated and filled (`__vec_<name>`), not the `List`.
1544        if let ParamMode::VecBuild { .. } = &p.mode {
1545            args.push(format!("__vec_{}", p.kt_name));
1546            continue;
1547        }
1548        // An Optional pair expands into two call args: the present flag
1549        // and the value-or-zero expression (in that order).
1550        if let ParamMode::OptionalPair {
1551            present_expr,
1552            value_expr,
1553        } = &p.mode
1554        {
1555            args.push(present_expr.clone());
1556            args.push(value_expr.clone());
1557            continue;
1558        }
1559        let arg = match &p.mode {
1560            ParamMode::Borrow
1561            | ParamMode::Consume
1562            | ParamMode::BorrowNullable
1563            | ParamMode::ConsumeNullable => format!("{}_ptr", p.kt_name),
1564            ParamMode::Unsigned64 { niche } => {
1565                if p.kt_type.is_nullable() {
1566                    match niche {
1567                        Some(niche) => format!("{}?.toLong() ?: {}", p.kt_name, niche),
1568                        None => format!("{}?.toLong()", p.kt_name),
1569                    }
1570                } else {
1571                    format!("{}.toLong()", p.kt_name)
1572                }
1573            }
1574            ParamMode::PassThrough => {
1575                if p.as_enum_value {
1576                    // Enum → its `Int` discriminant for the extern. Nullable
1577                    // enum (`Enum?`) uses `?.value` so it stays `Int?`.
1578                    if p.kt_type.is_nullable() {
1579                        match &p.enum_niche {
1580                            Some(niche) => format!("{}?.value ?: {}", p.kt_name, niche),
1581                            None => format!("{}?.value", p.kt_name),
1582                        }
1583                    } else {
1584                        format!("{}.value", p.kt_name)
1585                    }
1586                } else {
1587                    p.kt_name.clone()
1588                }
1589            }
1590            // Callback lambda → the param itself (the extern takes the
1591            // erased `Any`).
1592            ParamMode::Callback { call_arg } => call_arg.clone(),
1593            ParamMode::FlattenStruct { .. } => {
1594                unreachable!("FlattenStruct expanded before the single-arg match")
1595            }
1596            ParamMode::VecBuild { .. } => {
1597                unreachable!("VecBuild expanded before the single-arg match")
1598            }
1599            ParamMode::OptionalPair { .. } => {
1600                unreachable!("OptionalPair expanded before the single-arg match")
1601            }
1602        };
1603        args.push(arg);
1604    }
1605    // Output expansion: the builder / (acc, fold) cross just before the
1606    // error callback.
1607    args.extend(out.unfold_call_args.iter().cloned());
1608    // Trailing error-sink captures: `__bcap` (binding) always, then `__dcap`
1609    // (typed domain error) for a fallible-typed fn — each records its channel's
1610    // args and sets a flag (no throw on the Rust upcall). The wrapper reads them
1611    // after the native call returns (see the body below).
1612    args.push("__bcap".to_string());
1613    if sink.domain.is_some() {
1614        args.push("__dcap".to_string());
1615    }
1616    format!(
1617        "{}.{jni_call}({})",
1618        ext.jni_native_class_name(),
1619        args.join(", ")
1620    )
1621}

1623/// Transform a successful raw native return into the public Kotlin return.
1624/// This expression is emitted only after binding/domain captures have been
1625/// checked, so a native failure placeholder can never reach an enum lookup,
1626/// value projection, or erased-result cast.
1627fn build_success_return(ext: &Declarations, out: &OutputPlan, raw: &str) -> String {
1628    if let Some(p) = &out.projection {
1629        // Fold the wrap through the projection strategy. The wrap class is
1630        // the projection leaf's typed short name (a Handle's typed-handle
1631        // class, or `ULong`). The sentinel is the Kotlin
1632        // null-representation literal for the leaf wire — used only by
1633        // the `Niche+primitive` arm of `fold_projection_wrap`.
1634        let leaf_fqn = ext
1635            .kotlin_fqn(&p.leaf_key)
1636            .unwrap_or_else(|| p.leaf_key.to_string());
1637        let short = leaf_fqn.rsplit('.').next().unwrap_or(&leaf_fqn).to_string();
1638        let sentinel = projection_leaf_sentinel(p);
1639        fold_projection_wrap(&p.strategy, raw, &p.kind, &short, sentinel.as_deref())
1640    } else if out.is_enum_return {
1641        let enum_kt = out
1642            .kt_return
1643            .as_ref()
1644            .expect("enum return has a Kotlin type");
1645        format!("{enum_kt}.fromInt({raw})")
1646    } else if out.is_option_enum_return {
1647        // `kt_return` renders nullable (`Priority?`); the companion lives
1648        // on the non-null class name.
1649        let enum_kt = out
1650            .kt_return
1651            .as_ref()
1652            .expect("Option<enum> return has a Kotlin type")
1653            .to_string();
1654        let enum_kt = enum_kt.trim_end_matches('?');
1655        if out.enum_niches.is_empty() {
1656            format!("{raw}?.let {{ {enum_kt}.fromInt(it) }}")
1657        } else {
1658            let absent = out
1659                .enum_niches
1660                .iter()
1661                .map(|niche| format!("{raw} == {niche}"))
1662                .collect::<Vec<_>>()
1663                .join(" || ");
1664            format!("if ({absent}) null else {enum_kt}.fromInt({raw})")
1665        }
1666    } else if out.cast_return {
1667        let cast_kt = out
1668            .kt_return
1669            .as_ref()
1670            .expect("callback delivery returns R/A");
1671        format!("{raw} as {}", kt_type_short(cast_kt))
1672    } else {
1673        raw.to_string()
1674    }
1675}

1677/// The opaque-handle params (Borrow/Consume modes) — the set the lock

1681        .iter()
1682        .flat_map(|p| {

1702                target,

1711/// Error-callback wiring — the two independent channels ([`ErrorSink`]). The
1712/// **binding** handler `JniErrorHandler<R>.run(je: String?)` is always present
1713/// (its captured message slot is `__bcap.ze0`, since the capture is uniform).
1714/// A fallible fn with a typed error also gets the **domain** handler
1715/// `<Src>Handler<R>.run(ze…)` — no `je`, called only on `Err(E)`. Each is a
1716/// separate SAM param; the wrapper passes a per-thread capture to the extern,
1717/// then after the native call redispatches to whichever channel fired.
1718fn error_sink_parts(
1719    fplan: &JniFunctionPlan,
1720    imports: &mut BTreeSet<String>,
1721    r_ty: &KtType,
1722) -> Option<ErrorSink> {
1723    let ifaces = fplan.onerror_iface.as_ref()?;
1724    let binding_spec = &ifaces.binding;
1725    let binding_type = binding_spec.kt_ref(vec![r_ty.clone()]);
1726    imports.insert(binding_spec.capture_fqn());
1727    let binding_capture_short = binding_spec.capture_name();
1728
1729    // The typed domain channel exists only for a fallible fn with an error
1730    // plan; when present, both the interface spec and the error plan are.
1731    let domain = if let Some(domain_spec) = &ifaces.domain {
1732        let error_plan = fplan
1733            .error
1734            .as_ref()
1735            .expect("domain handler ⇒ frozen error plan");
1736        // Per ze leaf: (raw capture Kotlin type, raw→typed wrap). The CAPTURE
1737        // is the raw twin (what the native side calls); the user's handler is
1738        // the TYPED interface — the redispatch wraps each raw slot.
1739        let ze_info: Vec<(KtType, crate::jni::WrapKind)> = domain_spec
1740            .params
1741            .iter()
1742            .map(|p| {
1743                if let Some(fqn) = p.wrap.class_fqn() {
1744                    imports.insert(fqn.to_string());
1745                }
1746                (p.raw.clone(), p.wrap.clone())
1747            })
1748            .collect();
1749        debug_assert_eq!(ze_info.len(), error_plan.leaves.len());
1750        imports.insert(domain_spec.capture_fqn());
1751        // The domain redispatch args — the decomposed leaves only (no `je`).
1752        // The native side always fills the raw ze on `Err`, so non-null slots
1753        // are asserted `!!` then wrapped raw → typed.
1754        let call_args = ze_info
1755            .iter()
1756            .enumerate()
1757            .map(|(i, (raw, wrap))| {
1758                if raw.is_nullable() {
1759                    wrap.wrap_expr(&format!("__dcap.ze{i}"), true)
1760                } else {
1761                    wrap.wrap_expr(&format!("__dcap.ze{i}!!"), false)
1762                }
1763            })
1764            .collect::<Vec<_>>()
1765            .join(", ");
1766        Some(DomainSink {
1767            onerr_type: domain_spec.kt_ref(vec![r_ty.clone()]),
1768            capture_short: domain_spec.capture_name(),
1769            call_args,
1770        })
1771    } else {
1772        None
1773    };
1774
1775    // When there is a domain channel, the binding handler is named
1776    // `onBindingError` and the domain one is `onError`; otherwise the binding
1777    // handler is the sole `onError`.
1778    let binding_param = if domain.is_some() {
1779        "onBindingError".to_string()
1780    } else {
1781        "onError".to_string()
1782    };
1783    Some(ErrorSink {
1784        binding_param,
1785        binding_type,
1786        binding_capture_short,
1787        binding_call_arg: "__bcap.ze0".to_string(),
1788        domain,
1789    })
1790}

1792/// Pre-lock **alias preflight**: reject a call whose handle arguments name the
1793/// same native resource in a combination that would consume or invalidate it
1794/// twice — before the lock, and before any conversion.
1795///
1796/// `zCombine(primary: ZThing, fallback: ZThing)` is a supported declaration;
1797/// called as `zCombine(x, x)` it hands one allocation to two consuming
1798/// converters. `zAbsorb(a: ZThing, b: ZThing?)` mixing a consume with a borrow
1799/// is the same defect: the borrow dangles the moment the consume takes
1800/// ownership. Emitted whenever the call has **at least one consumed handle**
1801/// and **any other handle**.
1802///
1803/// Comparison is on `ptr`, the resource's own address, not on the declared
1804/// Kotlin type — which is what makes `T` and `T?` (and a handle nested inside a
1805/// flattened `data_class`) compare in the same domain without enumerating the
1806/// spellings. Two distinct live handles cannot share an address, so this has no
1807/// false positives; `0L` names no resource and is skipped.
1808///
1809/// A binding failure, so it routes through the same channel a closed handle
1810/// does — a function-level return, never a throw.
1811fn render_alias_preflight(opaques: &[Opaque], binding_param: &str, is_unit: bool) -> KtCode {
1812    let mut guards = KtCode::new();
1813    if opaques.len() < 2 || !opaques.iter().any(|o| o.consume_null.is_some()) {
1814        return guards;
1815    }
1816    let ptr_of = |o: &Opaque| {
1817        if o.nullable {
1818            format!("({t}?.ptr ?: 0L)", t = o.target)
1819        } else {
1820            format!("{t}.ptr", t = o.target)
1821        }
1822    };
1823    for i in 0..opaques.len() {
1824        for j in (i + 1)..opaques.len() {
1825            let (a, b) = (&opaques[i], &opaques[j]);
1826            // Two borrows of one resource are legal; at least one side must be
1827            // taking it away.
1828            if a.consume_null.is_none() && b.consume_null.is_none() {
1829                continue;
1830            }
1831            // Two handles of different classes are different allocations, so
1832            // their pointers can never be equal — checking them would be dead
1833            // code in every wrapper that mixes handle types. An unknown domain
1834            // compares against everything.
1835            if let (Some(da), Some(db)) = (&a.domain, &b.domain) {
1836                if da != db {
1837                    continue;
1838                }
1839            }
1840            let (pa, pb) = (ptr_of(a), ptr_of(b));
1841            let msg = format!(
1842                "\"Aliasing arguments: '{}' and '{}' are the same native resource; a consumed \
1843                 handle may not be passed twice in one call.\"",
1844                a.name, b.name,
1845            );
1846            let cond = format!("{pa} != 0L && {pa} == {pb}");
1847            guards = if is_unit {
1848                guards.wline(format!(
1849                    "if ({cond}) {{ {binding_param}.run({msg}); return }}"
1850                ))
1851            } else {
1852                guards.wline(format!("if ({cond}) return {binding_param}.run({msg})"))
1853            };
1854        }
1855    }
1856    guards
1857}

1859/// Pre-lock closed-handle guards: a racy-but-safe `isClosed()` check before
1860/// the lock. A closed handle is a **binding** failure, so it routes to
1861/// `<binding_param>.run("Operation on a closed native handle.")` (function-level
1862/// return; no throw). Racy: a close between this check and the native call is
1863/// caught by the Rust-side converter guard (the tag bit survives the race),
1864/// which routes through the same binding channel.
1865fn render_prelock_guards(opaques: &[Opaque], binding_param: &str, is_unit: bool) -> KtCode {
1866    const CLOSED_MSG: &str = "\"Operation on a closed native handle.\"";
1867    let mut guards = KtCode::new();
1868    for o in opaques {
1869        let cond = if o.nullable {
1870            format!("{t}?.isClosed() == true", t = o.target)
1871        } else {
1872            format!("{t}.isClosed()", t = o.target)
1873        };
1874        guards = if is_unit {
1875            guards.wline(format!(
1876                "if ({cond}) {{ {binding_param}.run({CLOSED_MSG}); return }}"
1877            ))
1878        } else {
1879            guards.wline(format!(
1880                "if ({cond}) return {binding_param}.run({CLOSED_MSG})"
1881            ))
1882        };
1883    }
1884    guards
1885}

1887/// The call in statement position, `bind`-prefixed (`""` / `"val __ret = "`),
1888/// wrapped in a consume `try/finally` when any handle is consumed.
1889fn render_value_stmt(bind: &str, body_expr: &str, opaques: &[Opaque]) -> KtCode {
1890    let consume_stmts: Vec<&str> = opaques
1891        .iter()
1892        .filter_map(|o| o.consume_null.as_deref())
1893        .collect();
1894    if consume_stmts.is_empty() {
1895        KtCode::new().wline(format!("{bind}{body_expr}"))
1896    } else {
1897        let mut fin = KtCode::new();
1898        for s in consume_stmts {
1899            fin = fin.line(s);
1900        }
1901        KtCode::new().try_finally(bind, KtCode::new().wline(body_expr), fin)
1902    }
1903}

1905/// The core call statement: `bind` + a single Kotlin **expression**
1906/// evaluating to the call's result. Handle params contribute
1907/// pointer-binding statements and a deadlock-safe `withSortedHandleLocks`
1908/// acquisition; the whole thing is expression-shaped (via `run { … }` where
1909/// statements are needed) so the caller can bind it to `__ret`, rethrow a
1910/// captured sink error, then return.
1911fn render_core_stmt(
1912    ext: &Declarations,
1913    opaques: &[Opaque],
1914    body_expr: &str,
1915    imports: &mut BTreeSet<String>,
1916    bind: &str,
1917) -> KtCode {
1918    // Under-lock pointer reads. The closed-handle check is done pre-lock
1919    // (`prelock_guards`, → `onError`); these just bind the ptr the call
1920    // passes. A handle closed after the guard carries the tag bit (odd
1921    // value), which the Rust-side converter guard rejects — never
1922    // dereferenced.
1923    let mut ptr_binds = KtCode::new();
1924    for o in opaques {
1925        ptr_binds = if o.nullable {
1926            ptr_binds.line(format!(
1927                "val {n}_ptr = {t}?.ptr ?: 0L",
1928                n = o.name,
1929                t = o.target
1930            ))
1931        } else {
1932            ptr_binds.line(format!("val {n}_ptr = {t}.ptr", n = o.name, t = o.target))
1933        };
1934    }
1935
1936    if opaques.is_empty() {
1937        // No handles — the call expression stands alone.
1938        render_value_stmt(bind, body_expr, opaques)
1939    } else if !ext.emit_handle_locks {
1940        // Lock-free mode: ptr binds then the value, wrapped as an expression.
1941        KtCode::new().blk(format!("{bind}run {{"), |c| {
1942            c.push(ptr_binds)
1943                .push(render_value_stmt("", body_expr, opaques))
1944        })
1945    } else {
1946        // Fast path: a statically-known, small (1–3), all-non-null handle set.
1947        // Pass the handles positionally to the allocation-free fixed-arity
1948        // `withSortedHandleLocks` overload. Otherwise build a `List` and use
1949        // the recursive overload.
1950        //
1951        // Deliberate trade-off (#68): ANY nullable handle takes the `List`
1952        // path, even when the present/absent split could reach a fixed-arity
1953        // overload (`if (h != null) withSortedHandleLocks(a, h) {…} else …`).
1954        // The small-list allocation is benchmark-noise, and the branch would
1955        // duplicate the whole call body per arm — longer generated code for
1956        // no measured gain. All three overloads stay: each is exercised by
1957        // all-non-null wrappers (the common case).
1958        let fixed_arity = !opaques.iter().any(|o| o.nullable) && (1..=3).contains(&opaques.len());
1959        if !ext.package.is_empty() {
1960            imports.insert(format!("{}.withSortedHandleLocks", ext.package));
1961            if !fixed_arity {
1962                imports.insert(format!("{}.NativeHandle", ext.package));
1963            }
1964        }
1965        if fixed_arity {
1966            let targets = opaques
1967                .iter()
1968                .map(|o| o.target.as_str())
1969                .collect::<Vec<_>>()
1970                .join(", ");
1971            KtCode::new().blk(format!("{bind}withSortedHandleLocks({targets}) {{"), |c| {
1972                c.push(ptr_binds)
1973                    .push(render_value_stmt("", body_expr, opaques))
1974            })
1975        } else {
1976            let mut adds = KtCode::new();
1977            for o in opaques {
1978                adds = if o.nullable {
1979                    adds.line(format!("{t}?.let {{ __locks.add(it) }}", t = o.target))
1980                } else {
1981                    adds.line(format!("__locks.add({t})", t = o.target))
1982                };
1983            }
1984            KtCode::new().blk(format!("{bind}run {{"), |c| {
1985                c.line("val __locks = ArrayList<NativeHandle>()")
1986                    .push(adds)
1987                    .blk("withSortedHandleLocks(__locks) {", |l| {
1988                        l.push(ptr_binds)
1989                            .push(render_value_stmt("", body_expr, opaques))
1990                    })
1991            })
1992        }
1993    }
1994}

1996/// Assemble the wrapper's body text: pre-lock guards, the per-thread error
1997/// capture, the (possibly Vec-building) core call, the post-call failure
1998/// redispatch, and the return.
1999///
2000/// No throw from the binding: the wrapper installs a **capture** the extern
2001/// invokes on `Err` (a SAM literal of the same handler interface — the
2002/// native side calls its typed `run`), then — after the native call —
2003/// calls the user's `onError.run` and returns its `R` if a failure was
2004/// recorded. A pre-lock closed-handle guard short-circuits straight to
2005/// `onError.run` (it can't reach the capture).
2006/// Slice/Vec params built as Rust-side `Vec` handles: allocate the handle
2007/// before the lock, fill it by pushing each element's leaves, and free it in
2008/// a `finally` (always — the target wrapper either borrows the boxed `Vec` or
2009/// `mem::take`s it, leaving an empty `Vec` to drop). The transient handle is
2010/// not a `NativeHandle`, so it never joins the lock set.
2011enum BodyReturn {
2012    Unit,
2013    Value(String),
2014}
2015
2016fn render_body(
2017    ext: &Declarations,
2018    params: &[Param],
2019    opaques: &[Opaque],
2020    sink: &ErrorSink,
2021    body_expr: &str,
2022    return_mode: &BodyReturn,
2023    imports: &mut BTreeSet<String>,
2024) -> KtCode {
2025    let is_unit = matches!(return_mode, BodyReturn::Unit);
2026    let vec_build: Vec<(&String, &String, &Vec<String>)> = params
2027        .iter()
2028        .filter_map(|p| match &p.mode {
2029            ParamMode::VecBuild {
2030                base,
2031                elem_accesses,
2032            } => Some((&p.kt_name, base, elem_accesses)),
2033            _ => None,
2034        })
2035        .collect();
2036    // The capture is a per-thread reusable holder (zero allocation): the
2037    // extern writes its `@JvmField` slots via `run`, the wrapper reads
2038    // them after the (synchronous) call. `acquire()` resets the slots.
2039    let mut b = render_alias_preflight(opaques, &sink.binding_param, is_unit)
2040        .push(render_prelock_guards(opaques, &sink.binding_param, is_unit))
2041        .line(format!(
2042            "val __bcap = {}.acquire()",
2043            sink.binding_capture_short
2044        ));
2045    if let Some(d) = &sink.domain {
2046        b = b.line(format!("val __dcap = {}.acquire()", d.capture_short));
2047    }
2048    // Post-call redispatch: at most one channel fires (the extern signals
2049    // binding OR domain, then returns), so check binding first, then domain.
2050    let mut failed_checks: Vec<String> = vec![format!(
2051        "if (__bcap.failed) return {}.run({})",
2052        sink.binding_param, sink.binding_call_arg
2053    )];
2054    if let Some(d) = &sink.domain {
2055        failed_checks.push(format!(
2056            "if (__dcap.failed) return onError.run({})",
2057            d.call_args
2058        ));
2059    }
2060    let bind = if is_unit { "" } else { "val __ret = " };
2061    if vec_build.is_empty() {
2062        b = b.push(render_core_stmt(ext, opaques, body_expr, imports, bind));
2063        for chk in &failed_checks {
2064            b = b.wline(chk);
2065        }
2066    } else {
2067        let native = ext.jni_native_class_name();
2068        for (name, base, _) in &vec_build {
2069            let new_m = crate::jni::vec_helper_method_name(ext, base, "New");
2070            b = b.wline(format!("val __vec_{name} = {native}.{new_m}({name}.size)"));
2071        }
2072        // `try { fill…; <core> } finally { free… }`: Kotlin `try` is an
2073        // expression, so for a non-unit fn `__ret` binds to the core call
2074        // (the block's last expression). A push runs no JVM upcall, so the
2075        // loop needs no per-element failure check.
2076        let mut fill = KtCode::new();
2077        for (name, base, accesses) in &vec_build {
2078            let push_m = crate::jni::vec_helper_method_name(ext, base, "Push");
2079            let args = std::iter::once(format!("__vec_{name}"))
2080                .chain(accesses.iter().cloned())
2081                .collect::<Vec<_>>()
2082                .join(", ");
2083            fill = fill.blk(format!("for (__e in {name}) {{"), |c| {
2084                c.wline(format!("{native}.{push_m}({args})"))
2085            });
2086        }
2087        let mut free = KtCode::new();
2088        for (name, base, _) in &vec_build {
2089            let free_m = crate::jni::vec_helper_method_name(ext, base, "Free");
2090            free = free.wline(format!("{native}.{free_m}(__vec_{name})"));
2091        }
2092        let core = render_core_stmt(ext, opaques, body_expr, imports, "");
2093        b = b.try_finally(bind, fill.push(core), free);
2094        for chk in &failed_checks {
2095            b = b.wline(chk);
2096        }
2097    }
2098    b = match return_mode {
2099        BodyReturn::Unit => b,
2100        BodyReturn::Value(expr) => b.line(format!("return {expr}")),
2101    };
2102    b
2103}

2105/// The Kotlin typing of one delivered lambda leaf: `(builder_kt, wire_kt,
2106/// wrap, is_value_projection)` — the type the *user's* lambda sees, the type the
2107/// extern delivers, and the expression rebuilding the former from the latter
2108/// (`pk` is the adapter's parameter name; passthrough unless the leaf carries a
2109/// value projection that can't be built Rust-side).
2110/// Shared by the unfold builder/fold lambda and the callback lambda params.
2111pub(crate) fn unfold_leaf_kt(
2112    ext: &Declarations,
2113    out_ty: &prebindgen_registry::flat::TypeRef,
2114    nullable: bool,
2115    pk: &str,
2116) -> Option<(KtType, String, String, bool)> {
2117    let proj = ext
2118        .out_frag(out_ty)
2119        .and_then(|e| e.metadata.projection.clone());
2120    let is_value_projection = proj
2121        .as_ref()
2122        .map(|p| matches!(p.kind, crate::jni::ProjectionKind::Unsigned64))
2123        .unwrap_or(false);
2124    // builder_kt: enum → Int; otherwise the normal classified type
2125    // (handle class / String / ByteArray / Long …).
2126    let builder_kt = if ext.is_kotlin_enum_reading(out_ty) {
2127        KtType::int()
2128    } else {
2129        classify_return(ext, out_ty)?.0?
2130    };
2131    let (mut wire_kt, wrap) = if is_value_projection {
2132        let p = proj.as_ref().unwrap();
2133        // Wrap class = the projection leaf's typed short name — NOT
2134        // `builder_kt` (which is `Short?` for an `Option<…>` leaf and would
2135        // leak the `?` into the constructor call).
2136        let leaf_fqn = ext
2137            .kotlin_fqn(&p.leaf_key)
2138            .unwrap_or_else(|| p.leaf_key.to_string());
2139        let short = leaf_fqn.rsplit('.').next().unwrap_or(&leaf_fqn).to_string();
2140        // The sentinel is the leaf's OWN `None`, so an absence it inherits from
2141        // an ancestor grants it none — `wrap_sentinel` is that rule, shared with
2142        // `leaf_iface_param`, which derives the same wrap (#142).
2143        let sentinel = wrap_sentinel(p, nullable);
2144        let mut wrap = fold_projection_wrap(&p.strategy, pk, &p.kind, &short, sentinel.as_deref());
2145        // A `nullable` leaf (an `Option` nesting step on its path) makes the
2146        // wire nullable even when the strategy itself is `Direct` — guard the
2147        // wrap so a null wire stays null instead of feeding the constructor.
2148        if nullable && matches!(p.strategy, crate::jni::FoldStrategy::Base) {
2149            let inner = projection_wrap_expr(&p.kind, &short, "it");
2150            wrap = format!("{pk}?.let {{ {inner} }}");
2151        }
2152        // Raw-text wire for the callback/builder descriptor — `KtType`'s
2153        // Display matches the historical string (`Long`/`ByteArray`/`List<…>`).
2154        (projection_wire_return(p).to_string(), wrap)
2155    } else {
2156        (builder_kt.to_string(), pk.to_string())
2157    };
2158    let builder_kt = if nullable {
2159        wire_kt.push('?');
2160        builder_kt.nullable()
2161    } else {
2162        builder_kt
2163    };
2164    Some((builder_kt, wire_kt, wrap, is_value_projection))
2165}

2340/// Caller-facing notes for every boundary position an expansion reshaped:
2341/// expanded params (what to pass instead of the Rust argument), decomposed
2342/// returns (what the builder/fold receives), and error decompositions
2343/// (what `onError` receives). Reads the same frozen function plan the report
2344/// uses.
2345fn shape_notes(fplan: &JniFunctionPlan) -> Option<String> {
2346    let mut notes: Vec<String> = Vec::new();
2347
2348    let mut plans: Vec<(&syn::Ident, &prebindgen_registry::expand::FoldPlan)> = fplan
2349        .params
2350        .iter()
2351        .filter_map(|param| match &param.form {
2352            ParamForm::Expanded { plan, .. } => Some((&param.ident, plan.fold())),
2353            ParamForm::Single(_) => None,
2354        })
2355        .collect();
2356    plans.sort_by_key(|(p, _)| p.to_string());
2357    for (param, plan) in plans {
2358        let target = plan.target.to_string();
2359        let arms: Vec<String> = plan
2360            .variants
2361            .iter()
2362            .map(|v| match &v.ctor {
2363                Some(c) => format!("its `{c}` inputs"),
2364                None => format!("an existing `{target}`"),
2365            })
2366            .collect();
2367        let leaf_names: Vec<String> = plan
2368            .leaves
2369            .iter()
2370            .map(|l| snake_to_camel(&l.name.to_string()))
2371            .collect();
2372        let how = if plan.selector.is_some() {
2373            if plan.produces_option() {
2374                format!(
2375                    "pass EITHER {} — the selector chooses the arm, `-1` = absent",
2376                    arms.join(" OR ")
2377                )
2378            } else {
2379                format!(
2380                    "pass EITHER {} — the selector chooses the arm",
2381                    arms.join(" OR ")
2382                )
2383            }
2384        } else {
2385            arms.join(" / ").to_string()
2386        };
2387        notes.push(format!(
2388            "Parameter `{param}` is the Rust `{target}` argument, expanded: {how} \
2389             (crosses as `{}`).",
2390            leaf_names.join("`, `")
2391        ));
2392    }
2393
2394    if let Some(plan) = &fplan.unfold {
2395        let source = plan.source.to_string();
2396        let leaves: Vec<&str> = plan.leaves.iter().map(|l| l.name.as_str()).collect();
2397        match plan.delivery {
2398            prebindgen_registry::unfold::Delivery::Callback if !leaves.is_empty() => {
2399                notes.push(format!(
2400                    "The Rust `{source}` result is delivered decomposed: the builder \
2401                     callback receives (`{}`).",
2402                    leaves.join("`, `")
2403                ));
2404            }
2405            prebindgen_registry::unfold::Delivery::Return => {
2406                notes.push(format!(
2407                    "The Rust `{source}` result is converted and returned as a single value."
2408                ));
2409            }
2410            _ => {}
2411        }
2412    }
2413
2414    if let Some(plan) = &fplan.error {
2415        let source = plan.source.to_string();
2416        let leaves: Vec<&str> = plan.leaves.iter().map(|l| l.name.as_str()).collect();
2417        notes.push(format!(
2418            "On a domain error `onError` receives the decomposed Rust `{source}` error \
2419             (`{}`); a binding/system failure goes to `onBindingError` instead.",
2420            leaves.join("`, `")
2421        ));
2422    }
2423
2424    if notes.is_empty() {
2425        None
2426    } else {
2427        Some(notes.join("\n"))
2428    }
2429}
