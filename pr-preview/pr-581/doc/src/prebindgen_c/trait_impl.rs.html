10/// What an emitter asks instead of the converter table.
11///
12/// [`CbindgenBuilder::compiled`] holds every fragment this binding compiled,
13/// keyed by crossing. A fragment answers for a crossing that occupies several
14/// wire values; the converter table's single `destination` cannot.
15impl CbindgenBuilder {
16    /// The fragment for `ty` crossing in the given direction, if one compiled.
17    ///
18    /// Shares the fragment rather than copying it: the store is read while
19    /// compilation is still writing to it, so a borrow cannot be held across
20    /// the next write.
21    pub(crate) fn frag(
22        &self,
23        ty: &TypeRef,
24        direction: Direction,
25    ) -> Option<std::rc::Rc<crate::compile::CFrag>> {
26        self.compiled.borrow().fragment(&ty.key(), direction)
27    }
28
29    /// The fragment that builds a Rust `ty` out of C parts.
30    pub(crate) fn in_frag(&self, ty: &TypeRef) -> Option<std::rc::Rc<crate::compile::CFrag>> {
31        self.frag(ty, Direction::Construct)
32    }
33
34    /// The fragment that takes a Rust `ty` apart into C parts.
35    pub(crate) fn out_frag(&self, ty: &TypeRef) -> Option<std::rc::Rc<crate::compile::CFrag>> {
36        self.frag(ty, Direction::Deconstruct)
37    }

44    /// Select one ordinary whole-value input operation without spelling the
45    /// source type. The retained [`TypeRef`] is materialized only by the final
46    /// Rust renderer.
47    pub(crate) fn in_terminal(
48        &self,
49        ty: &TypeRef,
50        registry: &impl Conversions,
51    ) -> Option<crate::chain::InputTerminalPlan> {
52        use crate::chain::InputTerminalOperation;
53
54        let plan = |wire, operation| crate::chain::InputTerminalPlan {
55            source: ty.clone(),
56            wire,
57            operation,
58        };
59        let key = ty.key();
60
61        // Opaque handle, by-value consume: `*Box::from_raw(v)` — fallible
62        // (null handle → message).
63        if self.opaque.contains_key(&key) {
64            let c_struct = self.c_type_ident(&key);
65            return Some(plan(
66                syn::parse_quote!(*mut #c_struct),
67                InputTerminalOperation::OwnedHandle {
68                    null_message: format!("null {} handle passed by value", type_short(&key)),
69                },
70            ));
71        }
72
73        // Inline-opaque, by-`*mut` consume. Ownership-specific write-back is a
74        // Flat/adapter semantic decision and is safe to freeze as wire-side
75        // statements before the source type is spelled.
76        if let Some(opaque) = self.value_opaque_ty_of(&key).cloned() {
77            let writeback = self.value_opaque_writeback_plan(registry, &key)?;
78            return Some(plan(
79                syn::parse_quote!(*mut #opaque),
80                InputTerminalOperation::ValueOpaque {
81                    opaque,
82                    writeback,
83                    null_message: format!("null {} value passed by value", type_short(&key)),
84                },
85            ));
86        }
87
88        // A C enum arrives through `MaybeUninit` so its discriminant can be
89        // validated before any restricted Rust enum value is materialized.
90        if self.enums.contains_key(&key) {
91            let e = unit_enum(registry, &key)?;
92            let c_name = self.c_type_ident(&key);
93            let c_name_string = c_name.to_string();
94            return Some(plan(
95                syn::parse_quote!(::core::mem::MaybeUninit<#c_name>),
96                InputTerminalOperation::Enum {
97                    c_name,
98                    variants: e.values.iter().map(|value| value.name.clone()).collect(),
99                    invalid_message: format!("invalid discriminant {{}} for `{c_name_string}`"),
100                    size_message: format!(
101                        "`{c_name_string}`: a #[repr(C)] enum must have the size of a C `int`"
102                    ),
103                    align_message: format!(
104                        "`{c_name_string}`: a #[repr(C)] enum must have the alignment of a C `int`"
105                    ),
106                },
107            ));
108        }
109
110        if r_is_string(ty) {
111            return Some(plan(
112                syn::parse_quote!(*const ::core::ffi::c_char),
113                InputTerminalOperation::String,
114            ));
115        }
116        if r_is_str(ty) {
117            return Some(plan(
118                syn::parse_quote!(*const ::core::ffi::c_char),
119                InputTerminalOperation::StrMarker,
120            ));
121        }
122        if r_is_bool(ty) {
123            return Some(plan(bool_wire(), InputTerminalOperation::Bool));
124        }
125        if r_is_scalar(ty) {
126            return Some(plan(scalar_ty(ty)?, InputTerminalOperation::Scalar));
127        }
128        None
129    }

131    /// The mirror field idents to null in a by-value consume's gravestone write-back,
132    /// for a `repr_c_struct` (`generate_mirror`) whose owned-pointer fields are all
133    /// **nullable** (`Option<Box<T>>`). `Some(idents)` (possibly empty — a pure
134    /// scalar/enum mirror needs no write-back) enables the cheap field-nulling path;
135    /// `None` (not a generate_mirror, or a bare `Box<T>` field whose NULL would be an
136    /// invalid `Box`) forces the full `gravestone()` write.
137    fn nullable_owned_ptr_fields(
138        &self,
139        registry: &impl Conversions,
140        key: &TypeKey,
141    ) -> Option<Vec<syn::Ident>> {
142        let cfg = self.value_opaque.get(key)?;
143        if !cfg.generate_mirror {
144            return None;
145        }
146        let mut idents = Vec::new();
147        for (fname, fty) in self.struct_fields(registry, key)? {
148            // An owned-pointer field is one whose mirror wire is a raw pointer
149            // (`Option<Box<T>>` / `Box<T>` → `*mut t_t`); scalars/enums are not.
150            if matches!(self.mirror_field_wire(fty), Some(syn::Type::Ptr(_))) {
151                // Bare `Box<T>`: cannot be nulled (an invalid `Box`).
152                fty.optional_inner()?;
153                idents.push(fname);
154            }
155        }
156        Some(idents)
157    }

159    /// The semantic write-back policy shared by late input rendering and the
160    /// emitted public `_take` helper.
161    fn value_opaque_writeback_plan(
162        &self,
163        registry: &impl Conversions,
164        key: &TypeKey,
165    ) -> Option<crate::chain::ValueOpaqueWriteback> {
166        use crate::chain::ValueOpaqueWriteback;
167
168        let cfg = self.value_opaque.get(key)?;
169        if cfg.generate_mirror {
170            match self.nullable_owned_ptr_fields(registry, key) {
171                // No owned-pointer fields ⇒ plain data, nothing to clean up.
172                Some(fields) if fields.is_empty() => Some(ValueOpaqueWriteback::None),
173                // All owned-pointer fields nullable ⇒ null them in place (drop-safe).
174                Some(fields) => Some(ValueOpaqueWriteback::NullFields(fields)),
175                // Bare `Box<T>` field ⇒ a NULL would be an invalid `Box`; full gravestone.
176                None => Some(ValueOpaqueWriteback::Gravestone),
177            }
178        } else {
179            // Non-mirror opaque: the consumer chose the kind explicitly.
180            match cfg.kind {
181                OpaqueKind::Owned => Some(ValueOpaqueWriteback::Gravestone),
182                OpaqueKind::Data => Some(ValueOpaqueWriteback::None),
183            }
184        }
185    }

187    /// Whether the auto-generated `Gravestone` impl is needed for a `repr_c_struct`
188    /// mirror: only when its consume/`_take` write-back uses `gravestone()` — i.e. it
189    /// has a bare `Box<T>` owned-pointer field (a null `Box` is invalid). Nullable
190    /// (`Option<Box<T>>`) mirrors null in place and need no `Gravestone`/`Default`;
191    /// non-mirror owned types get their `Gravestone` impl from the consumer.
192    fn mirror_needs_gravestone_impl(&self, registry: &Registry, key: &TypeKey) -> bool {
193        match self.value_opaque.get(key) {
194            Some(cfg) if cfg.generate_mirror => {
195                self.nullable_owned_ptr_fields(registry, key).is_none()
196            }
197            _ => false,
198        }
199    }

201    /// Compile every site of every exported function.
202    ///
203    /// The second half of #450 reaching real positions. A **recipe** is what a
204    /// type's conversion is, and both adapters have compiled those since
205    /// #455/#457; a **site** is one position in the generated API, and this is
206    /// where they start being asked about.
207    ///
208    /// What it buys is that the contracts a site owns are asked at all. Which
209    /// answers count as failures is the target's, through
210    /// [`Compile::tolerates`] — and Cbindgen answers `Borrowed` everywhere,
211    /// because a `*const T` is C's own non-owning pointer and its zero-copy
212    /// accessors return one. So the validity check passes here by policy
213    /// rather than by luck, and the composition contracts — a part's type and
214    /// how it is held — are what actually run.
215    ///
216    /// The returned plans become the immutable site store consumed by ordinary
217    /// wrappers and callback artifacts alike. Every callback argument is a root
218    /// deconstruction site, so its ABI and encoder are the same frozen payload
219    /// an ordinary return consumes.
220    fn compile_sites<'v>(
221        &'v self,
222        compiler: &mut prebindgen_registry::recipe::Compiler<
223            '_,
224            crate::compile::CCompile<'v, Registry>,
225        >,
226        registry: &'v Registry,
227    ) -> Result<CPlanParts, String> {
228        use prebindgen_registry::{
229            generation::{ArtifactId, ArtifactInput, ArtifactPlan},
230            recipe::{Crossing, Direction, Role, Site},
231        };
232
233        struct PendingCallback {
234            callback: crate::chain::CallbackArtifact,
235            inputs: Vec<ArtifactInput>,
236        }
237
238        let mut adapter = crate::compile::CCompile {
239            gen: self,
240            registry,
241        };
242        let mut plans = Vec::new();
243        let mut callbacks = std::collections::BTreeMap::<String, PendingCallback>::new();
244        let declared = self.declared_functions();
245        let mut names: Vec<&syn::Ident> = declared.iter().collect();
246        names.sort_by_key(|i| i.to_string());
247        for name in names {
248            let Some(f) = registry.flat().function(name) else {
249                continue;
250            };
251            for (index, param) in f.params.iter().enumerate() {
252                let site = Site {
253                    owner: name.clone(),
254                    role: Role::Param { index },
255                };
256                let crossing = Crossing::new(param.ty.clone(), Direction::Construct);
257                if let Some(plan) = compiler
258                    .site(&mut adapter, site, crossing)
259                    .map_err(|e| e.to_string())?
260                {
261                    plans.push(plan);
262                }
263
264                let Some(args) = param.ty.callback_args() else {
265                    continue;
266                };
267                let key: CallbackKey = args.iter().map(|arg| arg.key()).collect();
268                let cfg = self.callbacks.get(&key).ok_or_else(|| {
269                    format!(
270                        "Cbindgen: callback parameter {} of `{name}` has no callback declaration",
271                        index
272                    )
273                })?;
274                let mut arguments = Vec::new();
275                for (arg_index, arg) in args.iter().enumerate() {
276                    let site = Site {
277                        owner: name.clone(),
278                        role: Role::CallbackArg {
279                            param: index,
280                            arg: arg_index,
281                        },
282                    };
283                    let crossing = Crossing::new(arg.clone(), Direction::Deconstruct);
284                    let plan = compiler
285                        .site(&mut adapter, site, crossing)
286                        .map_err(|e| e.to_string())?
287                        .ok_or_else(|| {
288                            format!(
289                                "Cbindgen: callback argument {arg_index} of parameter {index} in \
290                                 `{name}` was omitted"
291                            )
292                        })?;
293                    let zero_copy_element = self.callback_slice_elem_wire_type_of(arg);
294                    let takeable = cfg.takeable.contains(&arg_index);
295                    if zero_copy_element.is_none() && !plan.abi().payload().has_abi() {
296                        return Err(format!(
297                            "Cbindgen: callback argument `{arg}` has no C ABI — deliver its parts \
298                             as separate callback arguments instead"
299                        ));
300                    }
301                    if takeable
302                        && !matches!(plan.abi().payload(), crate::compile::CValue::Direct { .. })
303                    {
304                        return Err(format!(
305                            "Cbindgen: takeable callback argument `{arg}` must have one direct C wire"
306                        ));
307                    }
308                    arguments.push(crate::chain::CallbackArgument {
309                        site: plan.id().clone(),
310                        value: plan.abi().payload().clone(),
311                        zero_copy_element,
312                        takeable,
313                    });
314                    plans.push(plan);
315                }
316
317                let inputs = arguments
318                    .iter()
319                    .map(|argument| ArtifactInput::Site {
320                        site: argument.site.clone(),
321                        slots: argument.value.slots(),
322                    })
323                    .collect();
324
325                let callback_name = self.callback_c_name(&key);
326                let callback = crate::chain::CallbackArtifact::new(
327                    self.callback_c_ident(&key),
328                    crate::compile::callback_operation(&param.ty),
329                    param.ty.clone(),
330                    args.to_vec(),
331                    arguments,
332                );
333                match callbacks.entry(callback_name.clone()) {
334                    std::collections::btree_map::Entry::Vacant(entry) => {
335                        entry.insert(PendingCallback { callback, inputs });
336                    }
337                    std::collections::btree_map::Entry::Occupied(mut entry) => {
338                        if entry.get().callback.signature() != callback.signature() {
339                            return Err(format!(
340                                "Cbindgen: callback name `{callback_name}` resolves to incompatible C ABIs"
341                            ));
342                        }
343                        entry.get_mut().inputs.extend(inputs);
344                    }
345                }
346            }
347            // A `Result`'s arms are their own sites; the whole `Result` is not
348            // a value C ever holds, so it is not one.
349            let returns: Vec<(Role, &prebindgen_registry::flat::TypeRef)> =
350                match f.ret.fallible_parts() {
351                    Some((ok, err)) => vec![(Role::Return, ok), (Role::Error, err)],
352                    None => vec![(Role::Return, &f.ret)],
353                };
354            for (role, ty) in returns {
355                if matches!(ty.kind(), prebindgen_registry::flat::TypeKind::Unit) {
356                    continue;
357                }
358                let site = Site {
359                    owner: name.clone(),
360                    role,
361                };
362                let crossing = Crossing::new(ty.clone(), Direction::Deconstruct);
363                if let Some(plan) = compiler
364                    .site(&mut adapter, site, crossing)
365                    .map_err(|e| e.to_string())?
366                {
367                    plans.push(plan);
368                }
369            }
370        }
371        let artifacts = callbacks
372            .into_iter()
373            .map(|(name, pending)| {
374                let id = ArtifactId::new("c-callback", name).map_err(|e| e.to_string())?;
375                Ok(ArtifactPlan::new(
376                    id,
377                    Vec::new(),
378                    pending.inputs,
379                    crate::chain::CArtifact::Callback(pending.callback),
380                ))
381            })
382            .collect::<Result<Vec<_>, String>>()?;
383        Ok(CPlanParts {
384            sites: plans,
385            artifacts,
386        })
387    }

389    /// Which alternative of `key` these parts came from, by matching their
390    /// field types against the model's.
391    ///
392    /// Only for a refusal message. `Compile::fields` is handed one arm's parts
393    /// without being told which arm — the driver numbers parts per alternative
394    /// and `choice` is where the `Alternative` arrives — so naming the arm the
395    /// way the declaration writes it means finding it again. A refusal that
396    /// says `Odd::Many` points at the line to change; one that says only the
397    /// type does not.
398    pub(crate) fn union_arm_name(
399        &self,
400        key: &TypeKey,
401        registry: &impl Conversions,
402        parts: &[(
403            prebindgen_registry::recipe::Part<'_>,
404            &crate::compile::CFrag,
405        )],
406    ) -> Option<String> {
407        let variant = match registry.flat().declared_type(&key.ident()?)? {
408            prebindgen_registry::flat::Type::Variant(v) => v,
409            _ => return None,
410        };
411        variant
412            .alternatives
413            .iter()
414            .find(|a| {
415                a.fields.len() == parts.len()
416                    && a.fields
417                        .iter()
418                        .zip(parts)
419                        .all(|(f, (p, _))| f.ty.key() == p.ty.key())
420            })
421            .map(|a| a.name.to_string())
422    }

424    /// Retain a tagged-union pointer payload without spelling its Rust type.
425    ///
426    /// The C wire owns one heap allocation. A source payload that already says
427    /// `Box<T>` transfers that box directly; a bare declared handle `T` is
428    /// boxed or unboxed at the boundary. Optionality is represented by NULL,
429    /// while a NULL non-optional input is a binding error because it can be
430    /// observed after a union arm has already been dropped.
431    pub(crate) fn payload_plan(
432        &self,
433        fty: &TypeRef,
434        direction: Direction,
435    ) -> Option<crate::chain::PayloadPlan> {
436        let optional = fty.optional_inner().is_some();
437        let (wire, source_inner, boxed, short) =
438            if let Some(inner) = self.declared_opaque_payload_inner(fty) {
439                let c = self.c_type_ident(&inner);
440                (
441                    syn::parse_quote!(*mut #c),
442                    fty.optional_inner().unwrap_or(fty).clone(),
443                    false,
444                    type_short(&inner),
445                )
446            } else {
447                let inner = r_boxed_inner(fty)?;
448                let c = self.c_type_ident(&inner.key());
449                (
450                    syn::parse_quote!(*mut #c),
451                    inner.clone(),
452                    true,
453                    type_short(&inner.key()),
454                )
455            };
456        Some(crate::chain::PayloadPlan {
457            source: fty.clone(),
458            source_inner,
459            wire,
460            direction,
461            optional,
462            boxed,
463            null_message: format!(
464                "null payload for `{short}` (a non-optional payload cannot be NULL — the \
465                 union may already have been dropped)"
466            ),
467        })
468    }

470    /// `String` **input from a `data_struct`'s mirror**: a null `char *`
471    /// decodes to an empty string rather than refusing.
472    ///
473    /// The second reading a `String` has, and the reason it needs a recipe of its
474    /// own. A `String` **parameter** is a pointer the caller chose to pass, so
475    /// a null one is a caller error and the ordinary input-terminal plan says so. A `String`
476    /// **field** shares a struct with every other field, and refusing it would
477    /// make the whole struct's decode fallible — so a function taking such a
478    /// struct by value would need a `Result` return or `.panic()`, for a field
479    /// it may not even read.
480    ///

482    /// hand-written field walk had; stating it as a recipe is what makes it

489            return None;
490        }
491        Some(crate::chain::InputTerminalPlan {
492            source: ty.clone(),
493            wire: syn::parse_quote!(*const ::core::ffi::c_char),
494            operation: crate::chain::InputTerminalOperation::StringField,
495        })

521/// Per-section [`CbindgenBuilder::prerequisites`] emitters. Each returns the runtime-
522/// support items for one concern; the trait method concatenates them in order,
523/// so the emitted preamble is identical to the former single function.
524impl CbindgenBuilder {
525    /// C allocator extern + raw C-string allocator + the universal memory freer.
526    /// Emitted when the layer hands `char*`/array memory to C. Panics if such
527    /// memory is produced but no `.free_memory_function` is declared.
528    fn prereq_alloc_free(&self, registry: &Registry, produces_array: bool) -> Vec<syn::Item> {
529        let mut items: Vec<syn::Item> = Vec::new();
530        if !(self.needs_free(registry) || produces_array) {
531            return items;
532        }
533        let free_ident = match &self.free_fn {
534            Some(name) => format_ident!("{}", name),
535            None => panic!(
536                "Cbindgen: the generated layer hands `char*` string memory to C \
537                 (a `String` return or a `String` data-struct field) but no \
538                 memory-freeing function is declared — add \
539                 `.free_memory_function(\"z_free\")`"
540            ),
541        };
542        // C allocator (linked from the C runtime; no crate dependency).
543        items.push(syn::parse_quote!(
544            extern "C" {
545                fn malloc(size: usize) -> *mut ::core::ffi::c_void;
546                fn free(ptr: *mut ::core::ffi::c_void);
547            }
548        ));
549        // Raw, destructor-free C-string block. `CString::new` drops interior
550        // NULs so the terminator marks the true end for C consumers.
551        items.push(syn::parse_quote!(
552            #[allow(non_snake_case, dead_code)]
553            pub(crate) fn __cbg_alloc_cstr(s: ::std::string::String) -> *mut ::core::ffi::c_char {
554                let c = ::std::ffi::CString::new(s).unwrap_or_default();
555                let bytes = c.as_bytes_with_nul();
556                unsafe {
557                    let p = malloc(bytes.len()) as *mut u8;
558                    if p.is_null() {
559                        return ::core::ptr::null_mut();
560                    }
561                    ::core::ptr::copy_nonoverlapping(bytes.as_ptr(), p, bytes.len());
562                    p as *mut ::core::ffi::c_char
563                }
564            }
565        ));
566        // Universal raw memory freer: type-agnostic C `free`, no length, no
567        // destructor (NULL-safe via C `free`).
568        items.push(syn::parse_quote!(
569            #[no_mangle]
570            #[allow(non_snake_case, unused_variables)]
571            pub unsafe extern "C" fn #free_ident(p: *mut ::core::ffi::c_void) {
572                free(p);
573            }
574        ));
575        items
576    }

578    /// Array builder: copy a `Vec<W>` into a C-`malloc`'d block of `W` and
579    /// return `(ptr, len)` (empty ⇒ `(NULL, 0)`). The block is freed C-side
580    /// via the `z_free_array` macro (per-element drop + the universal freer).
581    fn prereq_array_builder(&self, produces_array: bool) -> Vec<syn::Item> {
582        let mut items: Vec<syn::Item> = Vec::new();
583        if !produces_array {
584            return items;
585        }
586        items.push(syn::parse_quote!(
587            #[allow(non_snake_case, dead_code)]
588            pub(crate) unsafe fn __cbg_alloc_array<W>(v: ::std::vec::Vec<W>) -> (*mut W, usize) {
589                let n = v.len();
590                if n == 0 {
591                    return (::core::ptr::null_mut(), 0);
592                }
593                let p = malloc(n.wrapping_mul(::core::mem::size_of::<W>())) as *mut W;
594                if p.is_null() {
595                    return (::core::ptr::null_mut(), 0);
596                }
597                for (i, e) in v.into_iter().enumerate() {
598                    ::core::ptr::write(p.add(i), e);
599                }
600                (p, n)
601            }
602        ));
603        items
604    }

620    /// Freeze source-dependent C declaration families as registry artifacts.
621    ///
622    /// Planning retains source TypeRefs and target-owned wire syntax. Only the
623    /// final artifact renderer may ask Emit to spell a source type.
624    fn type_artifact_plans(
625        &self,
626        registry: &Registry,
627    ) -> Result<
628        Vec<prebindgen_registry::generation::ArtifactPlan<crate::compile::CRepresentation>>,
629        String,
630    > {
631        use prebindgen_registry::generation::{ArtifactId, ArtifactPlan};
632
633        let mut artifacts = Vec::new();
634
635        for (key, _cfg) in sorted_by_key(&self.opaque) {
636            let Some(reading) = registry.reading(key) else {
637                continue;
638            };
639            let dependencies = self.artifact_fragment_inputs(&reading);
640            if dependencies.is_empty() {
641                continue;
642            }
643            artifacts.push(ArtifactPlan::new(
644                ArtifactId::new("c-opaque-handle", key.as_str()).map_err(|e| e.to_string())?,
645                Vec::new(),
646                dependencies,
647                crate::chain::CArtifact::OpaqueHandle(crate::chain::OpaqueHandleArtifact {
648                    source: reading,
649                    c_struct: self.c_type_ident(key),
650                    drop_ident: self.destructor_symbol(key),
651                }),
652            ));
653        }
654
655        let takeable_keys = self.takeable_type_keys();
656        let mut values: Vec<(&TypeKey, &ValueOpaqueCfg)> = self.value_opaque.iter().collect();
657        values.sort_by(|a, b| a.0.as_str().cmp(b.0.as_str()));
658        for (key, cfg) in values {
659            let Some(reading) = registry.reading(key) else {
660                continue;
661            };
662            let dependencies = self.artifact_fragment_inputs(&reading);
663            if dependencies.is_empty() {
664                continue;
665            }
666            let mirror = if cfg.generate_mirror {
667                let fields = self.struct_fields(registry, key).unwrap_or_else(|| {
668                    panic!(
669                        "Cbindgen::repr_c_struct: `{}` is not a named struct",
670                        type_short(key)
671                    )
672                });
673                let restricted = self.restricted_validity_fields(registry, key);
674                if !restricted.is_empty() && !cfg.assume_c_field_validity {
675                    let listed: Vec<String> = restricted
676                        .iter()
677                        .map(|(fname, reason)| format!("  `{fname}`: {reason}"))
678                        .collect();
679                    panic!(
680                        "Cbindgen::repr_c_struct: `{}` crosses C's memory by whole-struct \
681                         reinterpret, but these fields have restricted-validity Rust types:\n\
682                         {}\n\
683                         A C caller can write a byte outside those domains, and the reinterpret \
684                         materialises it with no hook to normalise or validate it first (#170, \
685                         #158). Move the field to a `data_struct` (per-field wires), pass it as \
686                         a separate parameter, or widen it to an integer. If this binding's C \
687                         side is trusted to write only in-domain bytes — or never hands the \
688                         mirror back at all — acknowledge it with `.assume_c_field_validity()`.",
689                        type_short(key),
690                        listed.join("\n"),
691                    );
692                }
693                Some(crate::chain::ValueOpaqueMirror {
694                    ident: self.c_type_ident(key),
695                    fields: fields
696                        .iter()
697                        .map(|(name, ty)| {
698                            let wire = self.mirror_field_wire(ty).unwrap_or_else(|| {
699                                panic!(
700                                    "Cbindgen::repr_c_struct: field `{}` of `{}` has unsupported \
701                                     type `{}` (expected a scalar, a declared `enum_type`, or an \
702                                     opaque pointer `Option<Box<T>>`/`Box<T>` with `T` an `opaque_ptr`)",
703                                    name,
704                                    type_short(key),
705                                    ty
706                                )
707                            });
708                            (name.clone(), wire)
709                        })
710                        .collect(),
711                    gravestone: self.mirror_needs_gravestone_impl(registry, key),
712                })
713            } else {
714                None
715            };
716            let take = takeable_keys
717                .contains(key)
718                .then(|| crate::chain::ValueOpaqueTake {
719                    ident: self.take_symbol(key),
720                    writeback: self
721                        .value_opaque_writeback_plan(registry, key)
722                        .expect("value-opaque declaration has a write-back policy"),
723                });
724            artifacts.push(ArtifactPlan::new(
725                ArtifactId::new("c-value-opaque", key.as_str()).map_err(|e| e.to_string())?,
726                Vec::new(),
727                dependencies,
728                crate::chain::CArtifact::ValueOpaque(crate::chain::ValueOpaqueArtifact {
729                    source: reading,
730                    opaque: cfg.opaque.clone(),
731                    mirror,
732                    drop_ident: self.destructor_symbol(key),
733                    take,
734                }),
735            ));
736        }
737        Ok(artifacts)
738    }

740    /// Freeze one owning payload's recursive release policy without spelling
741    /// the Rust type used by a boxed-pointer cleanup.
742    fn payload_cleanup_plan(
743        &self,
744        fty: &TypeRef,
745        registry: &Registry,
746    ) -> Result<crate::chain::PayloadCleanup, String> {
747        use prebindgen_registry::generation::ArtifactId;
748
749        if r_is_string(fty) {
750            return Ok(crate::chain::PayloadCleanup::AllocatedString);
751        }
752        if self.tagged_union_has_drop(fty, registry) {
753            return Ok(crate::chain::PayloadCleanup::NestedUnion {
754                artifact: ArtifactId::new("c-tagged-union", fty.key().as_str())
755                    .map_err(|e| e.to_string())?,
756                drop_ident: self.destructor_symbol(&fty.key()),
757            });
758        }
759        let owning = self.owning_data_struct_fields(fty, registry);
760        if !owning.is_empty() {
761            let mut fields = Vec::with_capacity(owning.len());
762            for (name, ty) in owning {
763                fields.push((name, self.payload_cleanup_plan(ty, registry)?));
764            }
765            return Ok(crate::chain::PayloadCleanup::Fields(fields));
766        }
767        Ok(crate::chain::PayloadCleanup::BoxedPointer {
768            source: Box::new(r_boxed_inner(fty).unwrap_or(fty).clone()),
769        })
770    }

772    /// Freeze every reached tagged-union declaration and typed destructor.
773    fn tagged_union_artifact_plans(
774        &self,
775        registry: &Registry,
776    ) -> Result<
777        Vec<prebindgen_registry::generation::ArtifactPlan<crate::compile::CRepresentation>>,
778        String,
779    > {
780        use prebindgen_registry::generation::{ArtifactId, ArtifactPlan};
781
782        let mut artifacts = Vec::new();
783        for (key, _cfg) in sorted_by_key(&self.tagged_unions) {
784            let Some(reading) = registry.reading(key) else {
785                continue;
786            };
787            let inputs = self.artifact_fragment_inputs(&reading);
788            if inputs.is_empty() {
789                continue;
790            }
791            let Some(sum) = payload_enum(registry, key) else {
792                continue;
793            };
794            let mut prerequisites = Vec::new();
795            let mut arms = Vec::with_capacity(sum.alternatives.len());
796            for alternative in &sum.alternatives {
797                let mut fields = Vec::with_capacity(alternative.fields.len());
798                for field in &alternative.fields {
799                    let wire = self.payload_wire_of(key, &alternative.name, field);
800                    let cleanup = if self.payload_wire_owns(&field.ty, &wire, registry) {
801                        let cleanup = self.payload_cleanup_plan(&field.ty, registry)?;
802                        cleanup.prerequisites(&mut prerequisites);
803                        Some(cleanup)
804                    } else {
805                        None
806                    };
807                    fields.push(crate::chain::TaggedUnionFieldArtifact { wire, cleanup });
808                }
809                arms.push(crate::chain::TaggedUnionArmArtifact {
810                    alternative: alternative.clone(),
811                    fields,
812                });
813            }
814            prerequisites.sort();
815            prerequisites.dedup();
816            let drop_ident = arms
817                .iter()
818                .any(|arm| arm.fields.iter().any(|field| field.cleanup.is_some()))
819                .then(|| self.destructor_symbol(key));
820            artifacts.push(ArtifactPlan::new(
821                ArtifactId::new("c-tagged-union", key.as_str()).map_err(|e| e.to_string())?,
822                prerequisites,
823                inputs,
824                crate::chain::CArtifact::TaggedUnion(crate::chain::TaggedUnionArtifact {
825                    c_name: self.c_type_ident(key),
826                    arms,
827                    drop_ident,
828                }),
829            ));
830        }
831        Ok(artifacts)
832    }

834    /// Data structs: `#[repr(C)]` mirror only. Heap (`String`) fields are
835    /// `char*` raw blocks the C user releases individually via the
836    /// `free_memory_function` — no per-struct destructor.
837    fn prereq_data_structs(&self, registry: &Registry) -> Vec<syn::Item> {
838        let mut items: Vec<syn::Item> = Vec::new();
839        for (key, _cfg) in sorted_by_key(&self.data) {
840            let Some(reading) = registry.reading(key) else {
841                continue;
842            };
843            if self.in_frag(&reading).is_none() && self.out_frag(&reading).is_none() {
844                continue;
845            }
846            let Some(fields) = self.struct_fields(registry, &reading.key()) else {
847                continue;
848            };
849            let c_struct = self.c_type_ident(&reading.key());
850            let mut field_defs: Vec<TokenStream> = Vec::new();
851            for (fname, fty) in &fields {
852                let wire = self.data_field_wire(fty).unwrap_or_else(|| {
853                    panic!(
854                        "Cbindgen: field `{}` of data struct `{}` has unsupported type `{}`",
855                        fname,
856                        type_short(&reading.key()),
857                        fty
858                    )
859                });
860                field_defs.push(quote!(pub #fname: #wire));
861            }
862            items.push(syn::parse_quote!(
863                #[repr(C)]
864                #[allow(non_camel_case_types)]
865                pub struct #c_struct {
866                    #(#field_defs,)*
867                }
868            ));
869        }
870        items
871    }

873    /// Enums: `#[repr(C)]` mirror — variant idents with each discriminant
874    /// **re-emitted verbatim**, exactly as the source wrote it.
875    ///
876    /// Deliberately NOT routed through the shared
877    /// [`enum_discriminant_values`](prebindgen_registry::types_util::enum_discriminant_values).
878    /// That helper resolves each variant to a concrete `i64`, which is what an
879    /// adapter needs when it must *know the number* — JniGenBuilder's `jint` decode
880    /// and the Kotlin `value(N)` constants. This mirror needs no number: it is
881    /// Rust source that cbindgen re-reads, so passing the expression through
882    /// keeps every discriminant C already accepted — a `const` or `cfg`-driven
883    /// expression, and any value the source's own `repr` admits, including
884    /// ones outside `i64`. Resolving here would narrow that domain to what
885    /// `i64` and a literal can express, for no gain.
886    ///
887    /// The two adapters therefore agree on the *rule* (Rust's own assignment
888    /// order, which the shared helper encodes) while differing on what they
889    /// need from it — a number versus a spelling.
890    fn prereq_enums(
891        &self,
892        registry: &Registry,
893        emit: &prebindgen_registry::RustWriter,
894    ) -> Vec<syn::Item> {
895        let mut items: Vec<syn::Item> = Vec::new();
896        for (key, _cfg) in sorted_by_key(&self.enums) {
897            let Some(reading) = registry.reading(key) else {
898                continue;
899            };
900            if self.in_frag(&reading).is_none() && self.out_frag(&reading).is_none() {
901                continue;
902            }
903            let Some(e) = unit_enum(registry, &reading.key()) else {
904                continue;
905            };
906            let cname = self.c_type_ident(&reading.key());
907            // The C mirror re-states the discriminant **as written** — `= 0x07`
908            // stays `0x07` — which is the one consumer `EnumValue`'s retained
909            // syntax exists for, and the model's own docs name it.
910            let variants = e.values.iter().map(|v| {
911                let id = &v.name;
912                match emit.discriminant(v) {
913                    Some(expr) => quote!(#id = #expr),
914                    None => quote!(#id),
915                }
916            });
917            items.push(syn::parse_quote!(
918                #[repr(C)]
919                #[derive(Copy, Clone, Debug, Eq, PartialEq)]
920                #[allow(non_camel_case_types)]
921                pub enum #cname {
922                    #(#variants),*
923                }
924            ));
925        }
926        items
927    }

929    /// The wire of one payload field, or a generation error naming the
930    /// offending variant field and the supported set.
931    fn payload_wire_of(&self, key: &TypeKey, variant: &syn::Ident, field: &Field) -> syn::Type {
932        self.payload_field_wire(&field.ty).unwrap_or_else(|reason| {
933            panic!(
934                "Cbindgen::tagged_union: payload `{}::{}{}` of type `{}` cannot cross: {}",
935                type_short(key),
936                variant,
937                match &field.name {
938                    Some(n) => format!(".{n}"),
939                    None => String::new(),
940                },
941                field.ty,
942                reason,
943            )
944        })
945    }

949    /// State this binding into `registry` — see `JniGenBuilder::declare_into`.
950    ///
951    /// Push, not pull: the build script calls this, and the registry never
952    /// calls back. cbindgen declares no selective const surface (its
953    /// `on_const` policy generates a source-module alias for every captured
954    /// const) and no decompositions.
955    /// Binding-local fns declared by `convert!(..).local(..)`.
956    fn collect_local_functions(&self) -> Vec<(syn::ItemFn, String)> {
957        let mut result = Vec::new();
958        let mut seen = HashMap::<syn::Ident, String>::new();
959        for (ident, path, sig) in self.convert_decls.iter().flat_map(|decl| decl.locals()) {
960            let origin = prebindgen_registry::decl::local_path_prefix(path);
961            let mut sig = sig.clone();
962            sig.ident = ident.clone();
963            let signature = quote!(#origin #sig).to_string();
964            match seen.get(ident) {
965                Some(previous) if previous == &signature => continue,
966                Some(_) => panic!(
967                    "binding-local conversion fn `{ident}` is declared with two different signatures"
968                ),
969                None => {
970                    seen.insert(ident.clone(), signature);
971                }
972            }
973            let item: syn::ItemFn = syn::parse_quote!(#sig { unimplemented!() });
974            result.push((item, origin));
975        }
976        result
977    }

979    /// State this binding into `registry`, then resolve it — see
980    /// `JniGenBuilder::build`.
981    /// Read the source, resolve every crossing, and hand back the binding —
982    /// see `JniGenBuilder::build`.
983    pub fn build(self) -> Result<Cbindgen, prebindgen_registry::WriteRustError> {
984        let flat = self
985            .sources
986            .clone()
987            .build()
988            .map_err(prebindgen_registry::ScanError::from)?;

993    /// [`Self::build`] over a registry described elsewhere — the test seam.
994    pub(crate) fn build_with(
995        mut self,
996        registry: prebindgen_registry::RegistryBuilder,
997    ) -> Result<Cbindgen, prebindgen_registry::WriteRustError> {
998        let declared = self.declare_into(registry)?.validate_with(&self)?;
999        // A second holding of the model: `convert_with` consumes the builder,
1000        // and the table outlives that call.
1001        let model = declared.flat().clone();
1002        let recipes = self.recipes(&model).map_err(|errors| {
1003            prebindgen_registry::ScanError::AdapterInvariant {
1004                message: errors
1005                    .iter()
1006                    .map(|e| e.to_string())
1007                    .collect::<Vec<_>>()
1008                    .join("; "),
1009            }
1010        })?;
1011        let bindings = self.bindings(&model, &recipes).map_err(|errors| {
1012            prebindgen_registry::ScanError::AdapterInvariant {
1013                message: errors
1014                    .iter()
1015                    .map(|e| e.to_string())
1016                    .collect::<Vec<_>>()
1017                    .join("; "),
1018            }
1019        })?;
1020        // The compiler resumes from the fragments produced while registry
1021        // conversion resolution ran. It is cloned here as a map of `Rc`s`; the
1022        // immutable generation plan below becomes the only rendering input.
1023        let registry = declared
1024            .convert_with(|crossing, built| {
1025                let mut compiler = prebindgen_registry::recipe::Compiler::resume(
1026                    &model,
1027                    &recipes,
1028                    &bindings,
1029                    self.compiled.borrow().clone(),
1030                );
1031                let conv = self.compile_crossing(&mut compiler, crossing, built);
1032                *self.compiled.borrow_mut() = compiler.finish();
1033                // The conversion stays here; what the registry gets back is
1034                // which other crossings this one delegates to, which is what
1035                // its reachability walk needs.
1036                conv.map(|c| prebindgen_registry::Answer::over(c.subs))
1037            })?
1038            .build()?;
1039        // Freeze every ordinary and callback position plus the callback artifacts
1040        // that consume those sites.
1041        let CPlanParts {
1042            sites,
1043            mut artifacts,
1044        } = {
1045            let mut compiler = prebindgen_registry::recipe::Compiler::resume(
1046                &model,
1047                &recipes,
1048                &bindings,
1049                self.compiled.borrow().clone(),
1050            );
1051            let parts = self
1052                .compile_sites(&mut compiler, &registry)
1053                .map_err(|message| prebindgen_registry::ScanError::AdapterInvariant { message })?;
1054            *self.compiled.borrow_mut() = compiler.finish();
1055            parts
1056        };
1057        artifacts.extend(
1058            self.type_artifact_plans(&registry)
1059                .map_err(|message| prebindgen_registry::ScanError::AdapterInvariant { message })?,
1060        );
1061        artifacts.extend(
1062            self.tagged_union_artifact_plans(&registry)
1063                .map_err(|message| prebindgen_registry::ScanError::AdapterInvariant { message })?,
1064        );
1065        let mut generation = prebindgen_registry::generation::GenerationPlanBuilder::new();
1066        for fragment in self.compiled.borrow().fragments() {
1067            generation.fragment(fragment.freeze());
1068        }
1069        for site in sites {
1070            generation.site(site);
1071        }
1072        for artifact in artifacts {
1073            generation.artifact(artifact);
1074        }
1075        self.generation = Some(generation.build().map_err(|errors| {
1076            prebindgen_registry::ScanError::AdapterInvariant {
1077                message: errors.to_string(),
1078            }
1079        })?);
1080        self.validate_resolved(&registry)
1081            .map_err(|message| prebindgen_registry::ScanError::AdapterInvariant { message })?;
1082        Ok(Cbindgen {
1083            gen: self,
1084            registry,
1085        })
1086    }

1088    /// Build the conversion for one crossing by asking the table which recipe it
1089    /// takes and the driver to compile that recipe.
1090    ///
1091    /// `None` records a gap, exactly as the chain of guesses this replaced did:
1092    /// whether the gap matters is the registry's call, and its report names the
1093    /// crossing.
1094    fn compile_crossing<'v, R: Conversions>(
1095        &'v self,
1096        compiler: &mut prebindgen_registry::recipe::Compiler<'_, crate::compile::CCompile<'v, R>>,
1097        crossing: &Crossing,
1098        built: &'v R,
1099    ) -> Option<crate::compile::CFrag> {
1100        let (dir, key) = crossing;
1101        // The reading the scan already took for this crossing, fetched by the
1102        // key the crossing IS.
1103        let ty = built.reading(key)?;
1104        let direction = *dir;
1105        let mut adapter = crate::compile::CCompile {
1106            gen: self,
1107            registry: built,
1108        };
1109        let crossing = prebindgen_registry::recipe::Crossing::new(ty, direction);
1110        let fragment = compiler.crossing(&mut adapter, &crossing).ok()?;
1111        Some((*fragment).clone())
1112    }

1115        &self,
1116        mut registry: RegistryBuilder,
1117    ) -> Result<RegistryBuilder, prebindgen_registry::ScanError> {
1118        for (item_fn, origin) in self.collect_local_functions() {
1119            registry = registry.local_function(item_fn, origin)?;
1120        }
1121        for ident in self.declared_functions() {
1122            registry = registry.export(&ident);
1123        }
1124        for ident in self.helper_functions() {
1125            registry = registry.reference(&ident);
1126        }
1127        for ty in self.declared_types().into_values() {
1128            registry = registry.export_type(ty);
1129        }
1130        Ok(registry)
1131    }

1135    /// Report what this binding left unclaimed. Here because it is the
1136    /// earliest generator-owned hook that sees the model, and it runs exactly
1137    /// where the registry used to print these itself. Moves into
1138    /// `CbindgenBuilder::generate` once that exists (prebindgen#251 phase E).
1139    ///
1140    /// `consts: None` — cbindgen has no selective const mechanism, so every
1141    /// captured const reaches its source-alias policy and none is ever a skip.
1142    fn validate(&self, binding: &Building<'_>) -> Result<(), String> {
1143        let mut functions = self.declared_functions();
1144        functions.extend(self.helper_functions());
1145        prebindgen_registry::warn_unclaimed(
1146            binding.flat(),
1147            &prebindgen_registry::Claimed {
1148                functions,
1149                // The report asks what was *claimed*, which is a set of
1150                // identities — the declarations' spellings are the scan's
1151                // business, not this one's.
1152                types: self.declared_types().into_keys().collect(),
1153                consts: None,
1154                ignored_functions: self.ignored_functions(),
1155                ignored_types: self.ignored_types(),
1156                ..Default::default()
1157            },
1158        );
1159        Ok(())
1160    }

1162    // Consts have no declaration mechanism here (`declared_consts` stays
1163    // `None`), so every indexed const re-emits through the default
1164    // `on_const` — a path-alias against this source module, keeping consts
1165    // with non-portable initializers valid in the generated file. (cbindgen
1166    // cannot evaluate a path initializer, so aliased consts don't surface
1167    // as `#define`s in the C header.)

1170    }

1172    // ── Structural type resolution ──────────────────────────────────────
1173    // The adapter peels `ty` itself: a rank-0 terminal category, else a
1174    // wrapper shape (`Option<_>`, `&`/`&mut`/`&[_]`/`&str`). See `in_wrappers`
1175    // / `out_wrappers`.
1176
1177    fn prerequisites(
1178        &self,
1179        registry: &Registry,
1180        emit: &prebindgen_registry::RustWriter,
1181    ) -> Vec<syn::Item> {
1182        // C-string data memory (string returns + `String` fields of data structs)
1183        // is malloc'd raw and freed by the single universal `free_memory_function`.
1184        // Array returns (`Vec<T>`) also hand out a malloc'd block freed via the
1185        // same function (per element through the `z_free_array` macro), so the
1186        // allocator/freer prelude is needed for them too. Each section's emitter
1187        // lives in the `impl CbindgenBuilder` block above; order is significant.
1188        let produces_array = self.produces_array(registry);
1189        let mut items: Vec<syn::Item> = Vec::new();
1190        items.extend(self.prereq_alloc_free(registry, produces_array));
1191        items.extend(self.prereq_array_builder(produces_array));
1192        items.extend(crate::chain::render_artifacts(
1193            self.generation
1194                .as_ref()
1195                .expect("C generation plan was not frozen"),
1196            "c-opaque-handle",
1197            emit,
1198        ));
1199        items.extend(self.prereq_data_structs(registry));
1200        items.extend(crate::chain::render_artifacts(

1279/// Output-direction terminal categories: the shapes that cross whole, reached
1280/// from the `atomic` hook.
1281impl CbindgenBuilder {
1282    pub(crate) fn out_terminal(
1283        &self,
1284        ty: &TypeRef,
1285        registry: &impl Conversions,
1286    ) -> Option<crate::chain::OutputTerminalPlan> {
1287        let plan = |wire, operation| crate::chain::OutputTerminalPlan {
1288            source: ty.clone(),
1289            wire,
1290            operation,
1291        };
1292        // Unit return: trivial converter so `()` (and `Result<(), _>`) resolves.
1293        // Never actually called — void-returning wrappers ignore it, and
1294        // `emit_fallible_wrapper` special-cases `Result<(), E>` to drop the
1295        // out-param entirely (it exists only to satisfy the resolver).
1296        if matches!(ty.kind(), TypeKind::Unit) {
1297            return Some(plan(
1298                syn::parse_quote!(()),
1299                crate::chain::OutputTerminalOperation::Unit,
1300            ));
1301        }
1302
1303        // `String` output: a `malloc`'d `char*` raw block freed via the
1304        // `free_memory_function`. A `String` explicitly declared `opaque_ptr`
1305        // (held by C as `string_t *`) opts out — the opaque-handle branch below
1306        // owns it then (mirroring the input side, where owned-handle selection wins).
1307        if r_is_string(ty) && !self.opaque.contains_key(&ty.key()) {
1308            return Some(plan(
1309                syn::parse_quote!(*mut ::core::ffi::c_char),
1310                crate::chain::OutputTerminalOperation::String,
1311            ));
1312        }
1313
1314        // FFI-safe scalar (`bool`, integers, floats): identity pass-through.
1315        if r_is_scalar(ty) {
1316            let spelled = scalar_ty(ty)?;
1317            return Some(plan(spelled, crate::chain::OutputTerminalOperation::Scalar));
1318        }
1319
1320        let key = ty.key();
1321
1322        // Opaque handle output: `Box::into_raw` → the bare `*mut #c_struct` handle.
1323        if self.opaque.contains_key(&key) {
1324            let c_struct = self.c_type_ident(&ty.key());
1325            return Some(plan(
1326                syn::parse_quote!(*mut #c_struct),
1327                crate::chain::OutputTerminalOperation::OwnedHandle { c_struct },
1328            ));
1329        }
1330
1331        // Opaque error output (e.g. `ZError`): not a by-value struct — marshal it
1332        // to a malloc'd `char*` message via the recorded accessor `fn(&E) ->
1333        // String`. The error out-param of a `Result<_, E>` wrapper is thus
1334        // `char **e`. Freed by the universal `free_memory_function`.
1335        if let Some(msg_fn) = self.opaque_errors.get(&key) {
1336            return Some(plan(
1337                syn::parse_quote!(*mut ::core::ffi::c_char),
1338                crate::chain::OutputTerminalOperation::OpaqueError {
1339                    message_path: self.src_fn(msg_fn),
1340                },
1341            ));
1342        }
1343
1344        // Value-opaque output: move the Rust value's bytes into the opaque
1345        // counterpart, by value (no Box). Size/align equality is asserted at the
1346        // type's emission site (fail-closed).
1347        if let Some(opaque) = self.value_opaque_ty_of(&ty.key()) {
1348            let opaque = opaque.clone();
1349            return Some(plan(
1350                opaque,
1351                crate::chain::OutputTerminalOperation::ValueOpaque,
1352            ));
1353        }
1354
1355        // Enum output: `match` the source enum to the C enum.
1356        if self.enums.contains_key(&key) {
1357            let e = unit_enum(registry, &ty.key())?;
1358            let cname = self.c_type_ident(&ty.key());
1359            return Some(plan(
1360                syn::parse_quote!(#cname),
1361                crate::chain::OutputTerminalOperation::Enum {
1362                    c_name: cname,
1363                    variants: e.values.iter().map(|value| value.name.clone()).collect(),
1364                },
1365            ));
1366        }
1367
1368        None
1369    }
1370}

1372/// Structural wrapper-shape resolvers (the post-rank-machinery surface). Each
1373/// peels `ty`'s outermost layer and composes the inner's converter; `subs`
1374/// lists the immediate inner(s) it looked up.
1375impl CbindgenBuilder {
1376    /// `&[E]` slice **input** retained without a rendered converter body.
1377    /// The frozen site plan owns the two-parameter (`*const E_wire`, `usize`)
1378    /// ABI and zero-copy decode.
1379    pub(crate) fn in_slice_plan(&self, ty: &TypeRef) -> Option<crate::chain::SliceInputPlan> {
1380        let e = r_shared_slice_elem(ty)?;
1381        // #170, the slice instance. The two-param lowering builds the
1382        // `&[E]` zero-copy from C's own block, so there is nowhere to
1383        // normalise the bytes: `&[bool]` would materialise every
1384        // element's restricted domain at once. `MaybeUninit<bool>` is
1385        // not a fix here — the callee wants `&[bool]`, and rebuilding
1386        // the block would silently drop the zero-copy contract this
1387        // path exists for. Rejected until a raw-wire lowering exists.
1388        if r_is_bool(e) {
1389            panic!(
1390                "Cbindgen: `&[bool]` cannot cross IN from C. A `bool` slice is \
1391                 reinterpreted zero-copy from the caller's block, so a byte outside \
1392                 `{{0, 1}}` would become a Rust `bool` with no chance to normalise it \
1393                 (#170). Take the flags as an integer slice, or wrap them in a declared \
1394                 `opaque_ptr` handle."
1395            );
1396        }
1397        let scalar = scalar_ty(e);
1398        let wire: syn::Type = match &scalar {
1399            Some(e_ty) => syn::parse_quote!(*const #e_ty),
1400            None => {
1401                let counterpart = self.value_opaque_ty_of(&e.key())?.clone();
1402                syn::parse_quote!(*const #counterpart)
1403            }
1404        };
1405        Some(crate::chain::SliceInputPlan {
1406            element: e.clone(),
1407            wire,
1408            reinterpret: scalar.is_none(),
1409        })
1410    }

1412    /// Retain an input or output borrow without spelling its Rust referent.
1413    pub(crate) fn borrow_plan(
1414        &self,
1415        ty: &TypeRef,
1416        direction: Direction,
1417    ) -> Option<crate::chain::BorrowPlan> {
1418        use crate::chain::{BorrowOperation, BorrowPlan};
1419
1420        // `mutable` off the `Ref` itself, NOT `is_exclusive_borrow`: that
1421        // reading deliberately answers `false` for `&mut MaybeUninit<_>` — an
1422        // out-param slot is not an exclusive borrow OF A VALUE — and these arms
1423        // ask the syntactic question, "did the source write `&mut`".
1424        let TypeKind::Ref {
1425            mutable: rf_mut,
1426            inner: rf_inner,
1427            ..
1428        } = ty.kind()
1429        else {
1430            return None;
1431        };
1432        // The borrow's target, as a reading — every use below is its identity
1433        // or its source path, both of which the model answers.
1434        let elem = rf_inner;
1435
1436        let plan = |source_inner, wire, operation, null_message| BorrowPlan {
1437            source_inner,
1438            wire,
1439            operation,
1440            null_message,
1441        };
1442
1443        if direction == Direction::Deconstruct {
1444            if *rf_mut {
1445                return None;
1446            }
1447            let key = elem.key();
1448            let wire_ty: syn::Type = if self.opaque.contains_key(&key) {
1449                let c_struct = self.c_type_ident(&key);
1450                syn::parse_quote!(#c_struct)
1451            } else {
1452                self.value_opaque_ty_of(&key)?.clone()
1453            };
1454            return Some(plan(
1455                elem.as_ref().clone(),
1456                syn::parse_quote!(*const #wire_ty),
1457                BorrowOperation::SharedOutput,
1458                String::new(),
1459            ));
1460        }
1461
1462        // `&str`: borrow a UTF-8 C string directly from the caller.
1463        if !*rf_mut && r_is_str(rf_inner) {
1464            return Some(plan(
1465                elem.as_ref().clone(),
1466                syn::parse_quote!(*const ::core::ffi::c_char),
1467                BorrowOperation::StrInput,
1468                "null pointer passed for str argument".to_owned(),
1469            ));
1470        }
1471        // `&mut T` (mutable borrow). Three sub-cases, all wiring to a `*mut` of the
1472        // wire (the C memory IS the Rust value for a value-opaque mirror — asserted
1473        // layout-identical — so the cast is sound; `&mut` is a borrow, no gravestone).
1474        if *rf_mut {
1475            // `&mut MaybeUninit<X>` (X value-opaque): out-param into uninitialized
1476            // memory. Rust writes via the `MaybeUninit` (no drop of the garbage slot).
1477            // `TypeKind::Uninit` is the form `maybe_uninit_inner` matched by
1478            // reading a path's tail ident.
1479            if let prebindgen_registry::flat::TypeKind::Uninit(inner) = elem.kind() {
1480                let op = self.value_opaque_ty_of(&inner.key())?.clone();
1481                let short = type_short(&inner.key());
1482                return Some(plan(
1483                    inner.as_ref().clone(),
1484                    syn::parse_quote!(*mut #op),
1485                    BorrowOperation::MutableUninitInput,
1486                    format!("null {short} pointer"),
1487                ));
1488            }
1489            // `&mut` opaque handle, or `&mut` value-opaque: both reinterpret the C
1490            // pointer as a mutable Rust reference. The wire is the handle's C struct
1491            // or the value-opaque mirror.
1492            let wire_ty: syn::Type = if self.opaque.contains_key(&elem.key()) {
1493                let c_struct = self.c_type_ident(&elem.key());
1494                syn::parse_quote!(#c_struct)
1495            } else {
1496                self.value_opaque_ty_of(&elem.key())?.clone()
1497            };
1498            let short = type_short(&elem.key());
1499            return Some(plan(
1500                elem.as_ref().clone(),
1501                syn::parse_quote!(*mut #wire_ty),
1502                BorrowOperation::MutableInput,
1503                format!("null {short} pointer"),
1504            ));
1505        }
1506        // `&T` (shared borrow) of an opaque handle or value-opaque type.
1507        let key1 = elem.key();
1508        let wire_ty: syn::Type = if self.opaque.contains_key(&key1) {
1509            let c_struct = self.c_type_ident(&elem.key());
1510            syn::parse_quote!(#c_struct)
1511        } else {
1512            self.value_opaque_ty_of(&elem.key())?.clone()
1513        };
1514        let short = type_short(&elem.key());
1515        Some(plan(
1516            elem.as_ref().clone(),
1517            syn::parse_quote!(*const #wire_ty),
1518            BorrowOperation::SharedInput,
1519            format!("null {short} pointer"),
1520        ))
1521    }

1523    /// A typed Optional, Sequence, or Result output marker.
1524    ///
1525    /// Its frozen C value or function site owns the real multi-leaf ABI; the
1526    /// marker retains semantic identity and dependency edges for emission.
1527    pub(crate) fn out_marker_plan(
1528        &self,
1529        operation: crate::chain::MarkerOperation,
1530        subject: &TypeRef,
1531    ) -> Option<crate::chain::MarkerPlan> {
1532        let subs = match operation {
1533            crate::chain::MarkerOperation::ChoiceArm => return None,
1534            crate::chain::MarkerOperation::Optional | crate::chain::MarkerOperation::Sequence => {
1535                vec![subject.clone()]
1536            }
1537            crate::chain::MarkerOperation::Result => {
1538                let (ok, err) = subject.fallible_parts()?;
1539                vec![ok.clone(), err.clone()]
1540            }
1541        };
1542        Some(crate::chain::MarkerPlan { operation, subs })
1543    }
