5/// Last-segment ident of a `TypeKey` — e.g. `"Publisher<'static>"` →
6/// `"Publisher"`, `"AdvancedSubscriber<()>"` → `"AdvancedSubscriber"`. Used by
7/// the structured builders ([`Declarations::ptr_class`],
8/// [`Declarations::data_class`]) to derive a default Kotlin class name from
9/// the Rust type-key. Panics for non-path types (e.g. closures, references) —
10/// the per-kind `*_name_mangle` closures see only path-shaped
11/// shorts. For verbatim Kotlin expressions on non-path types, use a
12/// scalar / generic type wrapper.
13pub(crate) fn rust_short_name(key: &TypeKey) -> String {
14    rust_short_name_opt(key).unwrap_or_else(|| {
15        panic!(
16            "rust_short_name: cannot derive Kotlin name from type-key `{}` — \
17             only path-shaped types are supported here",
18            key.as_str()
19        )
20    })
21}
22
23/// Fallible variant of [`rust_short_name`] — returns `None` for
24/// non-path types instead of panicking. Used by
25/// [`Declarations::note_wrapper_registration`] which is called for rank-0
26/// wrapper patterns including non-path shapes like `()` where there
27/// is no Kotlin short name to derive.
28pub(crate) fn rust_short_name_opt(key: &TypeKey) -> Option<String> {
29    key.short_name()
30}

32/// Add the generated function lifetime to every JNI reference in an
33/// intermediate type, including references nested in registry-owned tuples.
34pub(crate) fn annotate_jobject_with_lifetime(ty: &syn::Type, life: &str) -> syn::Type {
35    if let syn::Type::Tuple(tuple) = ty {
36        let mut tuple = tuple.clone();
37        for elem in &mut tuple.elems {
38            *elem = annotate_jobject_with_lifetime(elem, life);
39        }
40        return syn::Type::Tuple(tuple);
41    }
42    if let syn::Type::Path(tp) = ty {
43        if let Some(last) = tp.path.segments.last() {
44            if crate::jni::wire_access::is_jni_reference_wire(ty)
45                && matches!(last.arguments, syn::PathArguments::None)
46            {
47                let mut new = tp.clone();
48                if let Some(last) = new.path.segments.last_mut() {
49                    let lt =
50                        syn::Lifetime::new(&format!("'{}", life), proc_macro2::Span::call_site());
51                    last.arguments =
52                        syn::PathArguments::AngleBracketed(syn::AngleBracketedGenericArguments {
53                            colon2_token: None,
54                            lt_token: syn::token::Lt::default(),
55                            args: syn::punctuated::Punctuated::from_iter(std::iter::once(
56                                syn::GenericArgument::Lifetime(lt),
57                            )),
58                            gt_token: syn::token::Gt::default(),
59                        });
60                }
61                return syn::Type::Path(new);
62            }
63        }
64    }
65    ty.clone()
66}

72// `pat_match` lived here — `ty.to_token_stream().to_string() == pat` — and was
73// how the converter selector decided what a type WAS: rebuild a wildcard
74// pattern from the spelling, render it to a string, compare. That made the
75// answer depend on how Rust happened to spell the type, so `Box<Option<T>>`
76// reconstructed as `Box<_>`, matched nothing, and got no converter at all
77// (#270). Dispatch reads `TypeKind` now; nothing needs it.

79/// INPUT: wire → rust. Format `<wire_id>_to_<rust_id>_<hash>` (including
80/// `impl Fn(...)` lambda converters — the legacy
81/// `process_kotlin_<Name>_callback` naming is gone with the fun-interface
82/// subsystem).
83pub(crate) fn input_name(rust: &TokenStream, wire: &syn::Type) -> syn::Ident {
84    let rust_id = sanitize_for_ident(&rust.to_string());
85    let wire_id = wire_short(wire);
86    let h = hash_pair(rust, wire);
87    let s = format!("{}_to_{}_{:08x}", wire_id, rust_id, h & 0xffff_ffff);
88    syn::Ident::new(&s, Span::call_site())
89}

91/// OUTPUT: rust → wire. Format `<rust_id>_to_<wire_id>_<hash>`.
92pub(crate) fn output_name(rust: &TokenStream, wire: &syn::Type) -> syn::Ident {
93    let rust_id = sanitize_for_ident(&rust.to_string());
94    let wire_id = wire_short(wire);
95    let h = hash_pair(rust, wire);
96    let s = format!("{}_to_{}_{:08x}", rust_id, wire_id, h & 0xffff_ffff);
97    syn::Ident::new(&s, Span::call_site())
98}

100pub(crate) fn sanitize_for_ident(s: &str) -> String {
101    // Special-case the empty tuple — the all-punctuation token stream
102    // would sanitize to a meaningless fallback. `unit` is recognisable.
103    if s.trim() == "()" {
104        return "unit".to_string();
105    }
106    let mut out = String::with_capacity(s.len());
107    let mut prev_underscore = false;
108    for c in s.chars() {
109        if c.is_ascii_alphanumeric() {
110            out.push(c);
111            prev_underscore = false;
112        } else if !prev_underscore {
113            out.push('_');
114            prev_underscore = true;
115        }
116    }
117    while out.starts_with('_') {
118        out.remove(0);
119    }
120    while out.ends_with('_') {
121        out.pop();
122    }
123    if out.is_empty() {
124        out.push_str("ty");
125    }
126    if out.chars().next().is_some_and(|c| c.is_ascii_digit()) {
127        out.insert(0, '_');
128    }
129    out
130}

132pub(crate) fn wire_short(wire: &syn::Type) -> String {
133    if let syn::Type::Path(tp) = wire {
134        if let Some(last) = tp.path.segments.last() {
135            return sanitize_for_ident(&last.ident.to_string());
136        }
137    }
138    sanitize_for_ident(&wire.to_token_stream().to_string())
139}

141pub(crate) fn hash_pair(rust: &TokenStream, wire: &syn::Type) -> u64 {
142    hash_name_pair(&rust.to_string(), wire)
143}

145/// Hash the canonical source name and intermediate type used in a converter
146/// identifier. Late plans pass a `TypeKey` string so naming never requires
147/// access to Rust syntax; legacy rendered converters pass their token spelling.
148pub(crate) fn hash_name_pair(source: &str, wire: &syn::Type) -> u64 {
149    use std::{
150        collections::hash_map::DefaultHasher,
151        hash::{Hash, Hasher},
152    };
153    let mut h = DefaultHasher::new();
154    source.hash(&mut h);
155    "::".hash(&mut h);
156    wire.to_token_stream().to_string().hash(&mut h);
157    h.finish()
158}
