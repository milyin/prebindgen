1//! Rust file emission for the resolved `Registry`.
2//!
3//! `write_rust` takes the [`Assembly`] the adapter compiled — the frozen set
4//! of final artifacts the file is made of; renders each with the writer-owned
5//! [`crate::RustWriter`], surrounds them with the adapter's prerequisites and
6//! prebindgen's own anonymous feature-check consts, and hands the assembled
7//! file to `Destination::write`.
8//!
9//! The artifacts arrive from the adapter rather than being collected from the
10//! registry, which is what lets one crossing contribute more than one function
11//! — or one that occupies more than a single wire value.
12//!
13//! This module is `pub`, so **every `pub` item in it is public API of the
14//! crate**. [`RustArtifact`] is the deliberately narrow late-rendering seam for
15//! out-of-crate adapters. Every artifact carries registry-owned semantic
16//! identity; final Rust names are never used as planning identity.
17
18use std::{
19    collections::{BTreeSet, HashMap},
20    path::{Path, PathBuf},
21};

25/// Errors surfaced by the file-emission phase.
26///
27/// Binding validation is NOT here — it runs once in
28/// [`RegistryBuilder::build`](crate::RegistryBuilder::build)
29/// (see [`Prebindgen::validate_resolved`]), so an invalid binding fails
30/// before a built generator exists and never reaches a writer.
31/// Emission-integrity checks still run here because reachability is finalized
32/// while per-item output is assembled.
33#[derive(Debug)]
34pub enum WriteError {
35    /// Generated code calls a planned private converter whose function was
36    /// removed by reachability filtering.
37    UnrenderedConverterCalls {
38        /// `(caller, missing converter)` pairs, sorted and de-duplicated.
39        calls: Vec<(String, String)>,
40    },
41}
42
43impl std::fmt::Display for WriteError {
44    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
45        match self {
46            WriteError::UnrenderedConverterCalls { calls } => {
47                write!(
48                    f,
49                    "generated code calls private converters that were not rendered:"
50                )?;
51                for (caller, converter) in calls {
52                    write!(f, "\n  - `{caller}` calls `{converter}`")?;
53                }
54                write!(
55                    f,
56                    "\nconverter reachability or dependency planning is incomplete"
57                )
58            }
59        }
60    }
61}
62
63impl std::error::Error for WriteError {}

65/// Identity of one final artifact in an [`Assembly`].
66///
67/// A private converter is identified by the registry operation it implements;
68/// every other final artifact — a helper the converter bodies call, an
69/// exported wrapper, a type mirror — is identified by the adapter-scoped
70/// [`ArtifactId`](crate::generation::ArtifactId) its plan was recorded under.
71#[derive(Clone, Debug, PartialEq, Eq, Hash)]
72pub enum ArtifactKey {
73    /// A private converter, identified by its registry operation.
74    Operation(crate::generation::OperationId),
75    /// Any other final artifact, identified by its plan's artifact identity.
76    Artifact(crate::generation::ArtifactId),
77}
78
79impl std::fmt::Display for ArtifactKey {
80    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
81        match self {
82            ArtifactKey::Operation(operation) => operation.fmt(f),
83            ArtifactKey::Artifact(artifact) => artifact.fmt(f),
84        }
85    }
86}

88/// One final artifact of the generated Rust file.
89///
90/// Resolution decides what the artifact is and what it depends on; this trait
91/// is the one thing the artifact still owes the writer — the Rust items it
92/// becomes. [`Self::render`] runs at the final writing boundary, with the
93/// [`crate::RustWriter`] used for the rest of final Rust emission and with no
94/// access to a live [`Registry`], so an artifact cannot resume resolution
95/// while the file is being assembled.
96pub trait RustArtifact {
97    /// Semantic identity, owned by the registry rather than by the Rust
98    /// symbol final emission will choose.
99    fn key(&self) -> ArtifactKey;
100
101    /// Whether the generated adapter surface reaches this artifact.
102    /// Validation-only artifacts return false: they are still rendered so
103    /// that a caller of one can be reported, but they do not reach the file.
104    fn reachable(&self) -> bool {
105        true
106    }
107
108    /// Materialize the artifact's Rust items.
109    fn render(&self, emit: &crate::RustWriter) -> Vec<syn::Item>;
110}

112/// The frozen set of final artifacts a generated Rust file is assembled from.
113///
114/// An assembly is ordered: artifacts reach the file in the order the adapter
115/// added them, which for converters is the registry-owned dependency order of
116/// the fragments they came from. It holds one artifact per [`ArtifactKey`], so
117/// sharing never depends on the Rust symbol final emission allocates.
118pub struct Assembly<A> {
119    artifacts: Vec<A>,
120}
121
122impl<A: RustArtifact> Assembly<A> {
123    /// The artifacts, in emission order. Includes the unreachable ones — the
124    /// writer renders those to report anything that still calls them.
125    pub fn artifacts(&self) -> impl ExactSizeIterator<Item = &A> {
126        self.artifacts.iter()
127    }
128}
129
130/// Collection phase preceding a frozen [`Assembly`].
131pub struct AssemblyBuilder<A> {
132    positions: HashMap<ArtifactKey, usize>,
133    artifacts: Vec<A>,
134}
135
136impl<A> Default for AssemblyBuilder<A> {
137    fn default() -> Self {
138        Self {
139            positions: HashMap::new(),
140            artifacts: Vec::new(),
141        }
142    }
143}
144
145impl<A: RustArtifact> AssemblyBuilder<A> {
146    /// Start an empty assembly.
147    pub fn new() -> Self {
148        Self::default()
149    }
150
151    /// Add one final artifact.
152    ///
153    /// An identity already present is kept once. A reachable artifact replaces
154    /// an unreachable one already held under the same identity, which is what
155    /// lets an adapter add a dormant artifact before the parent that reaches
156    /// it has been planned.
157    ///
158    /// Two *reachable* artifacts sharing one identity are a planning error,
159    /// not a choice to make: keeping either would drop an item the file needs,
160    /// and the reader would meet it as a missing name in the generated crate
161    /// rather than as a diagnostic here. Converters are exempt — one
162    /// registry operation is legitimately reached from several sites, and
163    /// de-duplicating those is what this method is for.
164    pub fn artifact(&mut self, artifact: A) -> &mut Self {
165        match self.positions.get(&artifact.key()).copied() {
166            Some(position) => {
167                if let ArtifactKey::Artifact(id) = artifact.key() {
168                    assert!(
169                        !(self.artifacts[position].reachable() && artifact.reachable()),
170                        "two reachable artifacts share the identity {id}"
171                    );
172                }
173                if !self.artifacts[position].reachable() && artifact.reachable() {
174                    self.artifacts[position] = artifact;
175                }
176            }
177            None => {
178                self.positions.insert(artifact.key(), self.artifacts.len());
179                self.artifacts.push(artifact);
180            }
181        }
182        self
183    }
184
185    /// Freeze the assembly.
186    pub fn build(self) -> Assembly<A> {
187        Assembly {
188            artifacts: self.artifacts,
189        }
190    }
191}

193impl<A: RustArtifact> FromIterator<A> for Assembly<A> {
194    fn from_iter<I: IntoIterator<Item = A>>(artifacts: I) -> Self {
195        let mut builder = AssemblyBuilder::new();
196        for artifact in artifacts {
197            builder.artifact(artifact);
198        }
199        builder.build()
200    }
201}

203/// Emit a resolved registry, assembling the generated Rust file from a frozen
204/// [`Assembly`].
205///
206/// The assembly is what the adapter's compilation produced: one final artifact
207/// per item the file will contain, already deduplicated and in registry-owned
208/// dependency order. Handing artifacts over instead of a converter table is
209/// what frees an adapter to emit a conversion the table could not hold —
210/// several functions for one crossing, or one occupying more than a single
211/// wire value.
212///
213/// `out_path` may be relative (resolved against `OUT_DIR` by prebindgen) or
214/// absolute. Returns the path actually written.
215pub fn write_rust<P: AsRef<Path>, E: Prebindgen, A: RustArtifact>(
216    registry: &Registry,
217    ext: &E,
218    assembly: &Assembly<A>,
219    out_path: P,
220) -> Result<PathBuf, WriteError> {
221    // Validation already ran ONCE in the generator's `build` — a built generator
222    // (the only source of a resolved registry) is valid by construction, so
223    // this writer does no binding resolution. It does validate the assembled
224    // private call graph before handing the file to the destination.
225    // The capability, minted here and nowhere else in this function's reach.
226    // Every artifact below is handed a borrow; nothing else in the pipeline is.
227    // See `prebindgen_flat::flat::emit` for what that buys and what it
228    // deliberately does not.
229    let emit = crate::RustWriter::new(registry, ext.source_module());
230    let mut items: Vec<syn::Item> = Vec::new();
231
232    // Every artifact renders, including the ones the adapter surface does not
233    // reach: an unreachable artifact still contributes its function names, so
234    // a caller that survived reachability filtering can be reported below.
235    let rendered: Vec<(bool, Vec<syn::Item>)> = assembly
236        .artifacts()
237        .map(|artifact| (artifact.reachable(), artifact.render(&emit)))
238        .collect();
239    let converter_names: BTreeSet<String> = rendered
240        .iter()
241        .flat_map(|(_, artifact_items)| artifact_items)
242        .filter_map(|item| match item {
243            syn::Item::Fn(function) => Some(function.sig.ident.to_string()),
244            _ => None,
245        })
246        .collect();
247    for (reachable, artifact_items) in rendered {
248        if reachable {
249            items.extend(artifact_items);
250        }
251    }
252
253    // 2. Anonymous consts, verbatim. Last, and in stream order. These are
254    //    prebindgen's own injected feature checks, identical for every
255    //    adapter and named by none, so they are written here rather than
256    //    planned as an artifact of one adapter's assembly.
257    for guard in registry.flat().guards() {
258        items.push(syn::Item::Const(emit.guard(guard)));
259    }
260
261    validate_converter_calls(&mut items, &converter_names)?;
262    let dest: Destination = items.into_iter().collect();
263    Ok(dest.write(out_path))
264}

266/// Refuse a generated file whose rendered functions still call a private
267/// converter plan removed by reachability filtering.
268///
269/// This is deliberately an integrity check for planned converter functions,
270/// not a general unresolved-name check. Calls to missing prerequisites or
271/// arbitrary external functions remain rustc's responsibility.
272fn validate_converter_calls(
273    items: &mut [syn::Item],
274    candidates: &BTreeSet<String>,
275) -> Result<(), WriteError> {
276    let rendered: BTreeSet<String> = items
277        .iter()
278        .filter_map(|item| match item {
279            syn::Item::Fn(function) => Some(function.sig.ident.to_string()),
280            _ => None,
281        })
282        .collect();
283    let mut visitor = ConverterCallValidator {
284        candidates,
285        rendered: &rendered,
286        caller: None,
287        calls: BTreeSet::new(),
288    };
289    for item in items {
290        syn::visit_mut::VisitMut::visit_item_mut(&mut visitor, item);
291    }
292    if visitor.calls.is_empty() {
293        Ok(())
294    } else {
295        Err(WriteError::UnrenderedConverterCalls {
296            calls: visitor.calls.into_iter().collect(),
297        })
298    }
299}
300
301struct ConverterCallValidator<'a> {
302    candidates: &'a BTreeSet<String>,
303    rendered: &'a BTreeSet<String>,
304    caller: Option<String>,
305    calls: BTreeSet<(String, String)>,
306}
307
308impl syn::visit_mut::VisitMut for ConverterCallValidator<'_> {
309    fn visit_item_fn_mut(&mut self, function: &mut syn::ItemFn) {
310        let previous = self.caller.replace(function.sig.ident.to_string());
311        syn::visit_mut::visit_item_fn_mut(self, function);
312        self.caller = previous;
313    }
314
315    fn visit_expr_call_mut(&mut self, call: &mut syn::ExprCall) {
316        if let syn::Expr::Path(path) = call.func.as_ref() {
317            if let Some(segment) = path.path.segments.last() {
318                let name = segment.ident.to_string();
319                if self.candidates.contains(&name) && !self.rendered.contains(&name) {
320                    self.calls.insert((
321                        self.caller
322                            .clone()
323                            .unwrap_or_else(|| "<generated item>".to_string()),
324                        name,
325                    ));
326                }
327            }
328        }
329        syn::visit_mut::visit_expr_call_mut(self, call);
330    }
331}
