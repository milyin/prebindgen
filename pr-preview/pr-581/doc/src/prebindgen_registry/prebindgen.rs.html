1//! `Prebindgen` — what a generator still hands the emitter.
2//!
3//! One method per `#[prebindgen]` item kind an adapter still emits per item
4//! (`on_struct`, `on_variant`, `on_enum`, `on_const`) returning typed Rust
5//! items to emit, plus the items they depend on (`prerequisites`) and two
6//! invariant checks. An exported wrapper is not among them: it is an artifact
7//! of the adapter's frozen assembly.
8//!
9//! **Conversion is not here.** A generator builds those itself, one per
10//! crossing, inside `RegistryBuilder::convert_with` — so there is no
11//! `on_input_type`, no deferral, and no fixed-point loop retrying until it
12//! converges.
13//!
14//! [`ConverterImpl::converter`] is the stable identity of the wire-facing
15//! converter. Its executable artifact is retained separately by the adapter's
16//! frozen generation plan; this shared registry carrier never stores rendered
17//! Rust syntax.
18
19use crate::{generation::OperationId, niches::Niches, registry::Registry};

26/// One link in a converter's [stage chain](`ConverterImpl::pre_stages`) —
27/// a value-inspecting step that sits between the rust value the
28/// `#[prebindgen]` fn yields/receives and the wire-facing
29/// [`ConverterImpl::converter`].
30///
31/// Each stage is a fallible `In → Result<Out, Err>` function. The core
32/// pipeline only ever orders [`Self::converter`]; how a
33/// stage's `Err` arm is surfaced to the foreign side — throw an exception,
34/// return an error code, set `errno`, … — is entirely up to the
35/// destination-language adapter and is described by [`Self::metadata`].
36#[derive(Clone)]
37pub struct Stage<M = ()> {
38    /// Stable identity of this stage's separately retained executable
39    /// artifact.
40    pub converter: OperationId,
41    /// Adapter-specific extras for this stage — the same type as the owning
42    /// converter's ([`ConverterImpl::metadata`]). The core never
43    /// inspects this; the adapter's emitter reads it to decide how the
44    /// stage's `Err` arm is surfaced (e.g. a JNI adapter stores the JVM
45    /// exception class and `throw_*` fn to call here; a C adapter might
46    /// store the error-code sentinel). Defaults to `()`.
47    pub metadata: M,
48}

50/// Result of resolving one converter — the wire (destination) type the rest
51/// of the registry sees, plus the identity of its separately retained
52/// executable artifact.
53///
54/// `converter` is registry-owned semantic identity. Composed parents retain
55/// that identity directly; they never recompute it from a Rust type or wire
56/// type. The writer turns it into a private Rust symbol only through
57/// [`crate::RustWriter`] during final file assembly.
58#[derive(Clone)]
59pub struct ConverterImpl<M = ()> {
60    /// Wire/destination type. Other converters that ask "what's the wire
61    /// form of this rust type?" read this. The actual function may return
62    /// a wrapped form (e.g. an adapter's own `Result`-like envelope) — that
63    /// is the adapter's internal calling convention; `destination` is the
64    /// value the wire carries on success.
65    pub destination: syn::Type,
66    /// Stable identity of the **wire-facing** stage. The adapter's frozen
67    /// generation plan owns its parameter list, return type, body, and other
68    /// Rust emission policy.
69    /// For input direction this is the FIRST stage in execution order
70    /// (it takes the wire); for output direction this is the LAST stage
71    /// (it produces the wire).
72    pub converter: OperationId,
73    /// **Rust-side** stages that compose with [`Self::converter`] to form
74    /// the full conversion chain. Default empty — a 1-stage converter
75    /// is just `function`.
76    ///
77    /// Order is rust-side-first → function-side-last. Concretely:
78    /// * **Input** (wire → rust): chain runs `wire → function →
79    ///   pre_stages[0] → pre_stages[1] → … → pre_stages[N-1] → rust`.
80    /// * **Output** (rust → wire): chain runs `rust → pre_stages[N-1] →
81    ///   … → pre_stages[1] → pre_stages[0] → function → wire`.
82    ///
83    /// Each stage is fallible; how its `Err` arm is surfaced is adapter
84    /// specific and carried in [`Stage::metadata`].
85    pub pre_stages: Vec<Stage<M>>,
86    /// Bit-patterns the wire type can represent but this converter never
87    /// produces (output) and rejects (input). Wrapper handlers like
88    /// `Option<_>` consume one slot for their own discriminant and
89    /// re-export the rest — see [`Niches`] for the cascade model.
90    /// Default is empty (no niche optimisation).
91    pub niches: Niches,
92    /// Adapter-specific extras carried alongside the converter. Filled by
93    /// the same handler that produces `destination` / `function` / `niches`,
94    /// and read by the adapter's language-side emitters off the fragment this
95    /// conversion belongs to. Set this where you build the converter, not in a
96    /// side channel.
97    pub metadata: M,
98    /// Inner crossings this converter composed from — the ones the adapter
99    /// looked up to build a wrapper (`Option<X>` → `[X]`, `Result<T,E>` →
100    /// `[T, E]`, `&T` → `[&T]`). Empty for a terminal converter (scalar, opaque
101    /// handle, string) and for a callback's own converter (callback args are
102    /// cross-direction — their required-ness flows through the registry's
103    /// type-graph edges, not here).
104    ///
105    /// This is what an adapter reports back as an
106    /// [`Answer`](crate::Answer), and what `propagate_required` walks to mark
107    /// reachable crossings required.
108    ///
109    /// **Identities, not spellings.** They are looked up and walked, never
110    /// emitted. An adapter that composed the inner type keys it
111    /// (`TypeKey::from_type`); one that read it off the model asks the reading
112    /// (`TypeRef::key`), and names no escape to do it.
113    pub subs: Vec<crate::registry::TypeKey>,
114}

116/// What an emitter asks of a conversion it holds.
117impl<M> ConverterImpl<M> {
118    /// Semantic identity of the wire-facing converter function.
119    pub fn converter_id(&self) -> &OperationId {
120        &self.converter
121    }
122
123    /// Wire type this conversion carries on success.
124    pub fn wire_type(&self) -> &syn::Type {
125        &self.destination
126    }
127
128    /// Rust-side stages in input execution order, after the wire-facing
129    /// converter has decoded the wire value.
130    pub fn input_stage_order(&self) -> impl Iterator<Item = (usize, &Stage<M>)> {
131        self.pre_stages.iter().enumerate().rev()
132    }
133
134    /// Rust-side stages in output execution order, before the wire-facing
135    /// converter encodes the final wire value.
136    pub fn output_stage_order(&self) -> impl Iterator<Item = (usize, &Stage<M>)> {
137        self.pre_stages.iter().enumerate()
138    }
139}

141/// The single extension point of the pipeline: implement this trait once per
142/// **destination language** (C/cbindgen, JNI/Kotlin, Swift, Python, …) to teach
143/// the language-agnostic [`Registry`] how that language represents Rust types
144/// on the wire and what wrapper code to emit.
145///
146/// The trait has no language-specific concepts of its own, and — since the
147/// registry stopped asking it questions — one job left: **per-item emission**.
148/// The file emitter calls `on_struct` / `on_variant` / `on_enum` / `on_const`
149/// to produce the per-item output, plus `prerequisites` and the two
150/// `validate` hooks for adapter invariants. An exported wrapper is not among
151/// them: it is an artifact of the adapter's frozen assembly.
152///
153/// What used to be here and is not any more: which items to build, how
154/// composites decompose, and the wire form of each type. A generator states the
155/// first two into the builder (`RegistryBuilder::export`,
156/// `RegistryBuilder::decompose`) and answers the third inside
157/// `RegistryBuilder::convert_with` — so nothing in core calls back to ask. Moving emission out too is what would delete this
158/// trait entirely (prebindgen#251 phase E).
159///
160/// Anything language-specific the rest of the pipeline must carry — a JNI
161/// adapter's Kotlin class names and exception info, a C adapter's header names,
162/// etc. — rides in [`ConverterImpl::metadata`], and is read back by the
163/// adapter's own emitter off the fragment that conversion belongs to. The
164/// registry neither stores it nor names its type.
165///
166/// # The rule an adapter must obey
167///
168/// "Analyze the Flat model; generate Rust only" tells an adapter where every
169/// decision comes from. Final source-type output is an inert fragment produced
170/// by [`RustWriter`](crate::RustWriter), never a typed view of retained syntax. It is silent
171/// on the question adapters actually face — what the **destination language**
172/// ends up seeing.
173/// That one has its own answer:
174///
175/// > **Same `kind` ⇒ same destination-language type.** The *wire* is the
176/// > generator's to choose, and may differ per spelling.
177///
178/// The weaker-sounding half is the important one. It is tempting to write "same
179/// `kind` ⇒ same wire", and that is **false** — prebindgen's own adapters
180/// violate it deliberately:
181///
182/// | Rust | `kind` | Kotlin type | wire |
183/// |---|---|---|---|
184/// | `&[Payload]` | `Ref(Slice)` | `List<Payload>` | `Long` — a handle to a Rust-side `Vec` |
185/// | `Vec<Box<Payload>>` | `Vec(Boxed)` | `List<Payload>` | `JObject` — a Java `List<Payload>` |
186///
187/// Two wires, one surface. Choosing a wire is exactly the generator's job, and
188/// the destination-language wrapper absorbs the difference; a caller cannot
189/// tell. What a caller *can* tell — and what
190/// [`unwrapped`](crate::flat::TypeRef::unwrapped) exists to prevent — is the
191/// **type** changing because the source spelled a `Box`.
192///
193/// The rule scopes to **converted** positions: those where a converter stands
194/// between the Rust value and the destination and is therefore free to bridge.
195/// It cannot apply to a **layout mirror**, where the destination type is
196/// reinterpreted from the source struct's bytes and is a *layout* fact rather
197/// than a surface choice — there `Box<T>` (a pointer) genuinely is a different
198/// destination type from `T` (inline), the spelling is load-bearing by
199/// construction, and no erasure can apply. The C adapter's `repr_c_struct` is
200/// the one such position in-tree, and its own documentation carries that half.
201///
202/// Reusing a mirror's spelling test in a converted position is how the rule
203/// gets broken (prebindgen#230, #292).
204pub trait Prebindgen {
205    /// Rust items the adapter's emitted converters depend on (helper
206    /// structs, type aliases, runtime-support code). Emitted at the top
207    /// of the destination file, before all auto-generated converters.
208    ///
209    /// Default: none. Wrapper adapters that compose a base adapter should
210    /// forward to or extend the base's `prerequisites()`. The resolved
211    /// `registry` is supplied so prerequisites can be gated on what the
212    /// (feature-aware) scan actually contains — e.g. emitting a
213    /// per-opaque-handle item only for handles a scanned `#[prebindgen]`
214    /// fn references.
215    fn prerequisites(&self, _registry: &Registry, _emit: &crate::RustWriter) -> Vec<syn::Item> {
216        Vec::new()
217    }
218
219    // ── Declaration queries ────────────────────────────────────────
220
221    /// Adapter-invariant checks that need registry **signatures** — the
222    /// earliest they can run (decl objects are built before any source is
223    /// read). Called by `RegistryBuilder::validate_with` right after the declaration
224    /// scan (so a missing fn has already hard-errored; validate sees only
225    /// indexed items) and before plan application. An `Err` aborts the
226    /// resolve as `ScanError::AdapterInvariant` with the message verbatim
227    /// — e.g. jnigen rejects a `.fun()` member whose target has no
228    /// receiver parameter of the class type.
229    ///
230    /// Default: no checks.
231    fn validate(&self, _binding: &crate::registry::Building<'_>) -> Result<(), String> {
232        Ok(())
233    }
234
235    /// Post-**resolve** validation boundary — the counterpart of
236    /// [`Self::validate`] that sees the fully resolved registry (converters,
237    /// plans, metadata). Every artifact writer calls it before writing
238    /// anything, so an invalid binding fails cleanly — with every problem
239    /// reported at once — instead of panicking midway after a sibling
240    /// artifact already reached disk. Deterministic over `(self, registry)`;
241    /// it runs once per write call, which keeps artifact writes
242    /// order-independent.
243    ///
244    /// Default: no checks.
245    fn validate_resolved(&self, _registry: &Registry) -> Result<(), String> {
246        Ok(())
247    }
248
249    /// Absolute path under which the source crate's items are reachable
250    /// from the generated file (e.g. `zenoh_flat`), for adapters that
251    /// qualify emitted references against one. An adapter may also use it in
252    /// its required [`Self::on_const`] policy to generate a path-alias to the
253    /// source item instead of copying initializer tokens. Default: `None`.
254    fn source_module(&self) -> Option<&syn::Path> {
255        None
256    }
257
258    // ── Item methods ───────────────────────────────────────────────
259    //
260    // Each takes the **element**, not the `syn` item it was parsed from.
261    //
262    // The element is the model's own node: its types are `TypeRef`s, already
263    // classified. An adapter handed one therefore cannot ask what a type means
264    // and be told "no reading" — the question a `&syn::ItemFn` forced it to ask
265    // the registry, and which answered wrongly for a type that never entered
266    // the pipeline (#275). Final Rust source types are generated by `RustWriter` from
267    // those model facts; retained type syntax is unavailable here.
268
269    /// Per-struct emission. Typically empty for languages that get
270    /// everything they need from auto-generated converters.
271    fn on_struct(
272        &self,
273        s: &prebindgen_flat::flat::Struct,
274        registry: &Registry,
275        emit: &crate::RustWriter,
276    ) -> Vec<syn::Item>;
277
278    /// Per-sum emission — an `enum` whose alternatives carry payloads.
279    ///
280    /// Separate from [`Self::on_enum`] because the model separates them: the
281    /// two are numbered differently and consumed as different constructs. An
282    /// adapter with nothing to say about one shape returns an empty vector, as
283    /// both in-tree adapters do for both.
284    fn on_variant(
285        &self,
286        v: &prebindgen_flat::flat::Variant,
287        registry: &Registry,
288        emit: &crate::RustWriter,
289    ) -> Vec<syn::Item>;
290
291    /// Per-enum emission — the fieldless shape, a named set of integers.
292    fn on_enum(
293        &self,
294        e: &prebindgen_flat::flat::Enum,
295        registry: &Registry,
296        emit: &crate::RustWriter,
297    ) -> Vec<syn::Item>;
298
299    /// Per-const emission. There is deliberately no default: Flat models the
300    /// const's name, type and documentation, but not an arbitrary Rust
301    /// initializer expression. Each adapter must therefore state its value
302    /// policy explicitly, commonly by generating a path-alias to a source
303    /// module. Initializer tokens are never copied or parsed.
304    ///
305    /// A const reaching here is always named: prebindgen's own injected feature
306    /// checks are [`Guard`](prebindgen_flat::flat::Guard)s, not consts, so this
307    /// never has to recognise one.
308    fn on_const(
309        &self,
310        c: &prebindgen_flat::flat::Constant,
311        registry: &Registry,
312        emit: &crate::RustWriter,
313    ) -> Vec<syn::Item>;
