5/// Return expansion declaration specialized for this adapter.

14/// ```
15pub type DeclCExpandReturn = prebindgen_declarations::DeclExpandReturn<
16    COutputConfig,
17    DeclCFunction,
18    std::convert::Infallible,
19>;
20/// Named-field declaration specialized for this adapter.

22/// Start input expansion using this adapter's node constructors.
23pub fn expand_param(ty: Type) -> DeclCExpandParam {
24    prebindgen_declarations::DeclExpandParam::new(
25        ty,
26        |function: DeclCFunction| CInputConfig::call(function.reference_name(false)),
27        || CInputConfig::Leaf(CInputLeafOp::Value),
28        |ty, alternatives, extension| {
29            let input = alternatives.build(CInputConfig::selector, CInputConfig::auto, |child| {
30                CInputConfig::Unary {
31                    operation: CInputUnary::Temporary,
32                    child: Box::new(child),
33                }
34            });
35            let _ = extension;
36            CInputConfig::for_type(ty, input)
37        },
38    )
39}
40/// Start return expansion using this adapter's node constructors.
41pub fn expand_return(ty: Type) -> DeclCExpandReturn {
42    prebindgen_declarations::DeclExpandReturn::new(
43        ty,
44        |function| {
45            prebindgen_declarations::getter_view(
46                function,
47                |name| COutputConfig::getter(name, COutputConfig::auto()),
48                |name, child| COutputConfig::Unary {
49                    operation: COutputUnary::Named { name },
50                    child: Box::new(child),
51                },
52            )
53        },
54        || COutputConfig::Leaf(COutputLeafOp::Value),
55        |policy| COutputConfig::Leaf(COutputLeafOp::Projected(policy)),
56        super::configuration::owned_field_view,
57        |ty, views, extension| {
58            let output = COutputConfig::combine(views);
59            let _ = extension;
60            COutputConfig::for_type(ty, output)
61        },
62    )
63}
64/// Decompose a source accessor's returned record into named field views.
65pub fn fields(function: syn::Ident) -> DeclCFields {
66    prebindgen_declarations::DeclFields::new(
67        function,
68        |output| match output {
69            COutputConfig::Unary {
70                operation: COutputUnary::ConfigForType { ty },
71                child: output,
72            } if matches!(output.as_ref(),COutputConfig::Branch {operation:COutputBranch::Combine,children} if children.is_empty()) => {
73                COutputConfig::for_type(*ty, COutputConfig::Leaf(COutputLeafOp::Discard))
74            }
75            output => output,
76        },
77        |function, overrides, _extension| {
78            let (fields, children) = overrides.into_iter().unzip();
79            COutputConfig::getter(
80                function,
81                COutputConfig::Branch {
82                    operation: COutputBranch::ConfigNamedFields(CNamedFields {
83                        fields,
84                        mode: prebindgen_declarations::ProjectionMode::Borrowed,
85                    }),
86                    children,
87                },
88            )
89        },
90    )
91}

99impl prebindgen_declarations::CallbackInput<CInputConfig, COutputConfig> for CInputConfigOps {

103            CInputConfig::auto,
104            |children| COutputConfig::Branch {

107            },
108            |metadata, domain, codomain| CInputConfig::Arrow {

112            },
113            CInputConfig::for_type,
114        )
115    }
