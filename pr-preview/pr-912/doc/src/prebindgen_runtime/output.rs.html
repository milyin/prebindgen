1//! Ownership of initialized rows before a foreign collection result is delivered.
2use core::{marker::PhantomData, mem::MaybeUninit};
3
4/// Final row storage that reclaims owned wire fields if output conversion fails.
5///
6/// Rows are `Copy` foreign representations; their owned fields need a generated
7/// destructor rather than Rust's ordinary field destruction. Storage is allocated
8/// once at its final length. Completed rows need no separate ownership queue.
9pub struct OutputRows<T: Copy> {
10    data: *mut MaybeUninit<T>,
11    length: usize,
12    initialized: usize,
13    release: unsafe fn(&T),
14    marker: PhantomData<T>,
15}
16impl<T: Copy> OutputRows<T> {
17    /// Allocate final storage and register destruction of completed rows.
18    ///
19    /// # Safety
20    /// `release` must reclaim exactly the additional resources owned by a
21    /// completed row, including nested row allocations, and skip borrowed fields.
22    /// It must contain destructor panics per field so other fields are reclaimed.
23    /// The generated initializer must transfer each row's resources exactly once
24    /// before calling [`Self::commit_row`].
25    #[inline]
26    pub unsafe fn new(length: usize, release: unsafe fn(&T)) -> Self {
27        // Retain a raw pointer, not a Box: moving the guard must not retag storage
28        // after its C pointer has been exposed.
29        let data = Box::into_raw(Box::<[T]>::new_uninit_slice(length)).cast();
30        Self {
31            data,
32            length,
33            initialized: 0,
34            release,
35            marker: PhantomData,
36        }
37    }
38    /// Store the next row before transferring its resources to this guard.
39    ///
40    /// # Safety
41    /// Fewer than `len()` rows must have completed. The row's resources must
42    /// remain owned by the initializer until [`Self::commit_row`]. Do not change
43    /// completed rows or retain references to this storage across a write.
44    #[inline]
45    pub unsafe fn write_next(&mut self, row: T) {
46        debug_assert!(self.initialized < self.length);
47        self.data.add(self.initialized).write(MaybeUninit::new(row));
48    }
49    /// Take cleanup responsibility for the row most recently written.
50    ///
51    /// # Safety
52    /// The next row must have been written, and all its owned resources must
53    /// have transferred to this guard. No fallible or unwinding operation may
54    /// separate the resource transfer from this call. Commit each row once.
55    #[inline]
56    pub unsafe fn commit_row(&mut self) {
57        debug_assert!(self.initialized < self.length);
58        self.initialized += 1;
59    }
60    /// Address of final row storage. Only completed rows may be read.
61    #[inline]
62    pub fn as_ptr(&self) -> *const T {
63        self.data.cast()
64    }
65    /// Allocated element count, including rows not yet initialized.
66    #[inline]
67    pub fn len(&self) -> usize {
68        self.length
69    }
70    /// Whether this output has no rows.
71    #[inline]
72    pub fn is_empty(&self) -> bool {
73        self.length == 0
74    }
75    /// Relinquish completed storage and its fields to the foreign recipient.
76    /// Its generated destructor must free the row allocation as `Box<[T]>`.
77    #[inline]
78    pub fn commit(self) {
79        debug_assert_eq!(self.initialized, self.length);
80        core::mem::forget(self);
81    }
82}
83impl<T: Copy> Drop for OutputRows<T> {
84    #[inline]
85    fn drop(&mut self) {
86        // Recover ownership first; derive row references from this Box after
87        // recovery, rather than from the earlier raw pointer's provenance.
88        let rows =
89            unsafe { Box::from_raw(core::ptr::slice_from_raw_parts_mut(self.data, self.length)) };
90        for row in &rows[..self.initialized] {
91            let _ = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| unsafe {
92                (self.release)(row.assume_init_ref());
93            }));
94        }
95    }
96}

98#[cfg(test)]
99mod tests {
100    use super::*;
101    use std::sync::{
102        atomic::{AtomicUsize, Ordering},
103        Arc,
104    };
105
106    struct Resource {
107        id: usize,
108        drops: Arc<[AtomicUsize; 8]>,
109        panic: bool,
110    }
111    impl Drop for Resource {
112        fn drop(&mut self) {
113            self.drops[self.id].fetch_add(1, Ordering::SeqCst);
114            assert!(!self.panic, "injected resource destructor panic");
115        }
116    }
117    #[derive(Clone, Copy)]
118    struct Row {
119        first: *mut Resource,
120        second: *mut Resource,
121    }
122    unsafe fn release(row: &Row) {
123        for pointer in [row.first, row.second] {
124            let _ = crate::boundary(|| {
125                if !pointer.is_null() {
126                    drop(Box::from_raw(pointer));
127                }
128                Ok(())
129            });
130        }
131    }
132    fn counts() -> Arc<[AtomicUsize; 8]> {
133        Arc::new(std::array::from_fn(|_| AtomicUsize::new(0)))
134    }
135    fn resource(drops: &Arc<[AtomicUsize; 8]>, id: usize, panic: bool) -> Box<Resource> {
136        Box::new(Resource {
137            id,
138            drops: drops.clone(),
139            panic,
140        })
141    }
142    #[test]
143    fn incomplete_rows_remain_with_the_initializer() {
144        let drops = counts();
145        let mut rows = unsafe { OutputRows::new(3, release) };
146        unsafe {
147            rows.write_next(Row {
148                first: Box::into_raw(resource(&drops, 0, false)),
149                second: Box::into_raw(resource(&drops, 1, false)),
150            });
151            rows.commit_row();
152        }
153        let mut pending = resource(&drops, 2, false);
154        unsafe {
155            rows.write_next(Row {
156                first: &mut *pending,
157                second: core::ptr::null_mut(),
158            });
159        }
160        drop(rows);
161        assert_eq!(drops[0].load(Ordering::SeqCst), 1);
162        assert_eq!(drops[1].load(Ordering::SeqCst), 1);
163        assert_eq!(drops[2].load(Ordering::SeqCst), 0);
164        drop(pending);
165        assert_eq!(drops[2].load(Ordering::SeqCst), 1);
166    }
167    #[test]
168    fn successful_delivery_and_guard_moves_preserve_the_exported_pointer() {
169        let drops = counts();
170        let mut rows = unsafe { OutputRows::new(1, release) };
171        let pointer = rows.as_ptr();
172        fn moved(rows: OutputRows<Row>) -> OutputRows<Row> {
173            rows
174        }
175        rows = moved(rows);
176        unsafe {
177            rows.write_next(Row {
178                first: Box::into_raw(resource(&drops, 0, false)),
179                second: core::ptr::null_mut(),
180            });
181            rows.commit_row();
182        }
183        assert_eq!(rows.len(), 1);
184        assert!(!rows.is_empty());
185        rows.commit();
186        assert_eq!(drops[0].load(Ordering::SeqCst), 0);
187        unsafe {
188            let delivered =
189                Box::from_raw(core::ptr::slice_from_raw_parts_mut(pointer.cast_mut(), 1));
190            release(&delivered[0]);
191        }
192        assert_eq!(drops[0].load(Ordering::SeqCst), 1);
193    }
194    #[test]
195    fn empty_and_overaligned_buffers_have_the_correct_layout() {
196        #[repr(align(128))]
197        #[derive(Clone, Copy)]
198        struct Aligned(u8);
199        unsafe fn no_release(row: &Aligned) {
200            assert_eq!(row.0, 42);
201        }
202        let empty = unsafe { OutputRows::<Aligned>::new(0, no_release) };
203        assert!(empty.is_empty());
204        drop(empty);
205        let mut rows = unsafe { OutputRows::new(1, no_release) };
206        assert!(rows.as_ptr().is_aligned());
207        unsafe {
208            rows.write_next(Aligned(42));
209            rows.commit_row();
210        }
211        drop(rows);
212    }
213    #[test]
214    fn cleanup_continues_when_a_destructor_panics_during_conversion_unwinding() {
215        let drops = counts();
216        let result = std::panic::catch_unwind({
217            let drops = drops.clone();
218            move || {
219                let mut rows = unsafe { OutputRows::new(2, release) };
220                unsafe {
221                    rows.write_next(Row {
222                        first: Box::into_raw(resource(&drops, 0, true)),
223                        second: Box::into_raw(resource(&drops, 1, false)),
224                    });
225                    rows.commit_row();
226                    rows.write_next(Row {
227                        first: Box::into_raw(resource(&drops, 2, false)),
228                        second: core::ptr::null_mut(),
229                    });
230                    rows.commit_row();
231                }
232                panic!("injected conversion panic");
233            }
234        });
235        assert!(result.is_err());
236        assert!(drops[..3].iter().all(|n| n.load(Ordering::SeqCst) == 1));
237    }
238    #[test]
239    fn nested_completed_rows_and_a_later_partial_child_are_reclaimed_once() {
240        #[derive(Clone, Copy)]
241        struct Parent {
242            data: *const Row,
243            len: usize,
244        }
245        unsafe fn release_parent(parent: &Parent) {
246            if parent.len != 0 {
247                let rows = Box::from_raw(core::ptr::slice_from_raw_parts_mut(
248                    parent.data.cast_mut(),
249                    parent.len,
250                ));
251                for row in &rows {
252                    release(row);
253                }
254            }
255        }
256        let drops = counts();
257        let mut parents = unsafe { OutputRows::new(2, release_parent) };
258        let mut first = unsafe { OutputRows::new(1, release) };
259        unsafe {
260            first.write_next(Row {
261                first: Box::into_raw(resource(&drops, 0, false)),
262                second: core::ptr::null_mut(),
263            });
264            first.commit_row();
265            parents.write_next(Parent {
266                data: first.as_ptr(),
267                len: first.len(),
268            });
269            first.commit();
270            parents.commit_row();
271        }
272        let mut second = unsafe { OutputRows::new(2, release) };
273        unsafe {
274            second.write_next(Row {
275                first: Box::into_raw(resource(&drops, 1, false)),
276                second: core::ptr::null_mut(),
277            });
278            second.commit_row();
279        }
280        let mut pending = resource(&drops, 2, false);
281        unsafe {
282            second.write_next(Row {
283                first: &mut *pending,
284                second: core::ptr::null_mut(),
285            });
286        }
287        drop(second);
288        drop(parents);
289        drop(pending);
290        assert!(drops[..3].iter().all(|n| n.load(Ordering::SeqCst) == 1));
291    }
