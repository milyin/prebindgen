6/// Selected delivery of a logical sequence.
7#[derive(Clone, Copy, Debug, PartialEq, Eq)]
8pub enum JniSequence {
9    /// Array of legacy `Object[]` rows, also used for inputs and custom schemas.
10    Buffered,
11    /// JVM array of generated immutable typed rows, with cached constructors.
12    /// Custom writers must create this exact schema's row objects through the
13    /// registered `prebindgen_jni_runtime::OutputRowType`, never `Object[]` rows.
14    TypedBuffered,
15    /// Native row ranges inside an enclosing bulk frame.
16    Bulk,
17    /// Each row is delivered through the function's sequence callback; the
18    /// collection's scalar status distinguishes present (zero) from absent.
19    Callback,
20}
21
22pub(crate) fn input(
23    wire: syn::Ident,
24    length: syn::Ident,
25    index: syn::Ident,
26    element: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
27    active: &Condition,
28) -> Result<SequenceInput<JniWire, crate::JniChain>> {
29    let env = prebindgen_conversions::template_parameter("environment");
30    let row = format_ident!("{wire}_row");
31    let count = crate::group::row_fields(element).len();
32    let capacity = i32::try_from(count + 8).map_err(|_| "too many collection element fields")?;
33    let frame = format_ident!("{wire}_frame");
34    let mut bind = quote! {let #frame=unsafe {::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;let #row=::prebindgen_jni_runtime::row(&mut #env,&#wire,#index,#count)?;};
35    let mut validation_values = Vec::new();
36    for (position, field) in crate::group::row_fields(element).enumerate() {
37        let name = &field.name;
38        let read = field.kind.read_array_element(&quote!(#row), position);
39        bind.extend(quote! {
40            let #name=#read;
41            if #name.is_err() && #env.exception_check().map_err(|e|e.to_string())? {return Err("JNI collection field decoding failed".into());}
42        });
43        let borrow = quote!(*);
44        validation_values.push((
45            field.name.clone(),
46            quote!(#borrow #name.as_ref().map_err(|e|e.clone())?),
47        ));
48    }
49    Ok(SequenceInput {
50        controls: vec![prebindgen_flat::Wire {
51            name: wire.clone(),
52            kind: JniWire::Rows {
53                condition: kotlin_condition(active),
54                null_niche_available: true,
55            },
56        }],
57        metadata: JniChain::Sequence(JniSequence::Buffered),
58        open: quote! {if #wire.is_null(){return Err("null collection".into());}let #length=#env.get_array_length(&#wire).map_err(|e|e.to_string())? as usize;},
59        bind,
60        validation_values,
61        release: quote!(drop(#frame);),
62    })
63}
64pub(crate) fn output(
65    wire: syn::Ident,
66    length: syn::Ident,
67    index: syn::Ident,
68    element: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
69    values: &[(syn::Ident, TokenStream)],
70) -> Result<SequenceOutput<JniWire, crate::JniChain>> {
71    let env = prebindgen_conversions::template_parameter("environment");
72    let row = format_ident!("{wire}_row");
73    let count = crate::group::row_fields(element).len();
74    let capacity = i32::try_from(count + 8).map_err(|_| "too many collection fields")?;
75    let typed = typed_rows::supported(element);
76    let needs_frame = !typed || group::row_fields(element).any(|w| w.kind.is_reference());
77    let constructor = format_ident!("{wire}_row_constructor");
78    let (prepare, encode) = if typed {
79        let cache = typed_rows::cache(element)?;
80        let mut statements = TokenStream::new();
81        let mut arguments = Vec::new();
82        for (i, (field, value)) in crate::group::row_values(element, values)?
83            .into_iter()
84            .enumerate()
85        {
86            let argument = field
87                .kind
88                .output_argument(value, &format_ident!("{wire}_row_argument_{i}"));
89            statements.extend(argument.statements);
90            arguments.push(argument.value);
91        }
92        (
93            quote!(let #constructor=#cache.get()?;),
94            quote!(#statements let #row=unsafe {#constructor.construct(&mut #env,&[#(#arguments),*])}?;),
95        )
96    } else {
97        (
98            quote!(),
99            quote!(let #row=::prebindgen_jni_runtime::array(&mut #env,#count)?;),
100        )
101    };
102    let encode = if typed {
103        encode
104    } else {
105        let fields = encode_array(crate::group::row_values(element, values)?, &row, 0);
106        quote!(#encode #fields)
107    };
108    let frame = format_ident!("{wire}_frame");
109    let (enter, leave) = if needs_frame {
110        (
111            quote!(let #frame=unsafe {::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;),
112            quote!(drop(#frame);),
113        )
114    } else {
115        (quote!(), quote!())
116    };
117    let retain_row = if needs_frame {
118        quote!()
119    } else {
120        quote!(let #row=#env.auto_local(#row);)
121    };
122    let owner = format_ident!("{wire}_owner");
123    let release = crate::cleanup::helper(
124        element,
125        if typed {
126            crate::cleanup::Carrier::TypedObjects
127        } else {
128            crate::cleanup::Carrier::Objects
129        },
130    )?;
131    let (own, complete, commit) = if let Some(release) = release {
132        (
133            if typed {
134                quote!(let mut #owner=unsafe {::prebindgen_jni_runtime::OutputRows::new_objects(&mut #env, &#wire, #release)};)
135            } else {
136                quote!(let mut #owner=unsafe {::prebindgen_jni_runtime::OutputRows::new(&mut #env, &#wire, #release)}?;)
137            },
138            quote!(unsafe {#owner.complete_row();}),
139            quote!(#owner.commit();),
140        )
141    } else {
142        (quote!(), quote!(), quote!())
143    };
144    let commit_element = prebindgen_conversions::template_parameter("commit_element");
145    Ok(SequenceOutput {
146        controls: vec![prebindgen_flat::Wire {
147            name: wire.clone(),
148            kind: JniWire::Rows {
149                condition: ("true").into(),
150                null_niche_available: true,
151            },
152        }],
153        commit_each: true,
154        fallible: true,
155        metadata: JniChain::Sequence(if typed {
156            JniSequence::TypedBuffered
157        } else {
158            JniSequence::Buffered
159        }),
160        begin: quote!(#prepare let #wire=::prebindgen_jni_runtime::array(&mut #env,#length)?; #own),
161        enter,
162        store: quote! {
163            #encode
164            #retain_row
165            #env.set_object_array_element(&#wire,#index as i32,&#row).map_err(|e|e.to_string())?;
166            #commit_element
167            #complete
168            #leave
169        },
170        finish: quote!(let #wire=Some(#wire);),
171        values: vec![(wire.clone(), quote!(#wire))],
172        commit,
173    })
174}

176pub(crate) fn input_list(
177    carrier: &prebindgen_flat::Wire<JniWire>,
178    sequence: &JniSequence,
179    element: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
180    context: &KotlinContext<'_>,
181    active: &str,
182) -> Result<KotlinInput> {
183    if *sequence == JniSequence::Bulk {
184        return crate::bulk::input_list(carrier, element, context, active);
185    }
186    if *sequence != JniSequence::Buffered {
187        return Err("collection input requires a buffered array or bulk sequence".into());
188    }
189    let mut child = input_kotlin(element, context)?;
190    child.prepare_arguments(crate::kotlin::input_handles(element, false))?;
191    if child.parameters.len() != 1 {
192        return Err("Kotlin collection elements must expose exactly one value; declare a data class for expanded fields".into());
193    }
194    let (parameter, ty) = child.parameters[0]
195        .split_once(": ")
196        .ok_or("invalid element parameter")?;
197    let name = carrier.name.to_string();
198    let state = format!("{name}_elements");
199    let JniWire::Rows { condition, .. } = &carrier.kind else {
200        return Err("sequence requires an object-array control".into());
201    };
202    let resources = crate::kotlin::input_handles(element, false);
203    let owned = crate::kotlin::input_handles(element, true);
204    let values = child
205        .prepared_arguments
206        .iter()
207        .map(|value| format!("{}.InputValidation.value {{ {value} }}", context.package))
208        .collect::<Vec<_>>();
209    if values.len() != child.arguments.len() {
210        return Err("collection input converter must provide prepared wire values".into());
211    }
212    let mut setup = String::new();
213    for binding in &child.setup {
214        setup.push_str("val ");
215        setup.push_str(&binding.name);
216        setup.push_str(" = ");
217        setup.push_str(&binding.expression);
218        setup.push('\n');
219    }
220    let snapshot = format!("{name}_snapshot");
221    let index = format!("{name}_index");
222    let rows = format!("kotlin.Array<kotlin.Any?>({snapshot}.size) {{ {index} -> val {parameter} = {snapshot}[{index}]\n{setup}kotlin.arrayOf<kotlin.Any?>({}) }}",values.join(", "));
223    let columns =
224        if crate::group::row_fields(element).any(|w| matches!(w.kind, JniWire::Handle { .. })) {
225            format!(
226                "intArrayOf({})",
227                crate::group::row_fields(element)
228                    .map(|w| match &w.kind {
229                        JniWire::Handle { handle, .. } if handle.access == Access::Move => "2",
230                        JniWire::Handle { .. } => "1",
231                        _ => "0",
232                    })
233                    .collect::<Vec<_>>()
234                    .join(", ")
235            )
236        } else {
237            "null".into()
238        };
239    let expression = if resources {
240        let count = crate::handles::kinds(element).len();
241        let groups = crate::handles::column_groups(element)?;
242        format!("if ({condition}) {name}?.toList()?.let {{ {snapshot} -> {}.InputRows({rows}, {columns}, {count}, {groups}) }} else null",context.package)
243    } else {
244        format!("if ({condition}) {name}?.toList()?.let {{ {snapshot} -> {rows} }} else null")
245    };
246    Ok(KotlinInput {
247        handle_inputs: Some(crate::handles::carrier_inputs(element, &state)),
248        parameter_paths: vec![prebindgen_declarations::ComponentPath::default()],
249        parameters: vec![format!(
250            "{name}: kotlin.collections.List<{ty}>{}",
251            if active == "true" { "" } else { "?" }
252        )],
253        setup: vec![KotlinBinding {
254            name: state.clone(),
255            expression,
256            absent: "null".into(),
257        }],
258        arguments: vec![if resources {
259            format!("{state}?.encode()")
260        } else {
261            state.clone()
262        }],
263        prepared_arguments: vec![state.clone()],
264        locks: vec![],
265        lock_groups: if resources {
266            vec![format!("{state}?.handles.orEmpty()")]
267        } else {
268            vec![]
269        },
270        consumed: if owned {
271            vec![format!("{state}?.consume(__pb_consume)")]
272        } else {
273            vec![]
274        },
275    })
276}

278pub(crate) fn output_list(
279    element: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
280    context: &KotlinContext<'_>,
281    owners: Option<&str>,
282    typed: bool,
283) -> Result<KotlinOutput> {
284    if owners.is_none() && crate::kotlin::input_handles(element, false) {
285        return Err("handle collection output requires an ownership scope".into());
286    }
287    let row = format!("{}_row", context.output_prefix);
288    let failure = format!("{}failure", context.output_prefix);
289    let ctx = KotlinContext {
290        failure: &failure,
291        cleanup_failure: Some(&failure),
292        result: &row,
293        output_prefix: &format!("{}item_", context.output_prefix),
294        ..*context
295    };
296    let mut child = crate::kotlin::output_kotlin_scoped(element, &ctx, owners)?;
297    if typed {
298        let fields = group::row_fields(element).collect::<Vec<_>>();
299        let values = (0..fields.len())
300            .map(|i| format!("{row}.c{i}"))
301            .collect::<Vec<_>>();
302        returns::remap(&mut child, &row, &fields, 0, &values);
303    }
304    if child.delivered.len() != 1 {
305        return Err("Kotlin collection elements must deliver exactly one value; declare a data class for expanded outputs".into());
306    }
307    let ty = &child.result_types[0];
308    let value = &child.delivered[0];
309    let raw = format!("({}[0] as kotlin.Array<*>)", context.result);
310    let row_type = if typed {
311        let (name, _) = sequence_callback::capture(element, &mut names::Names::default())?;
312        format!("{}.JNINative.{name}", context.package)
313    } else {
314        "kotlin.Array<*>".into()
315    };
316    let output = format!("{}list", context.output_prefix);
317    let index = format!("{}index", context.output_prefix);
318    let cleanup = format!("{}cleanupError", context.output_prefix);
319    let record_row_failure = kotlin::record_cleanup_failure(&ctx, &cleanup);
320    let record_outer_failure = kotlin::record_cleanup_failure(context, &cleanup);
321    let current_failure = context.cleanup_failure.unwrap_or(context.failure);
322    let finish_row = context
323        .cleanup_failure
324        .map(|target| format!("{target} = {failure}"))
325        .unwrap_or_default();
326    Ok(KotlinOutput {
327        result_names: vec![None],
328        delivered: vec![format!("requireNotNull({output})")],
329        result_types: vec![format!("kotlin.collections.List<{ty}>")],
330        output_locals: format!("var {output}: kotlin.collections.MutableList<{ty}>? = null\n"),
331        materialize: format!(
332            "{output} = kotlin.collections.ArrayList({raw}.size)\nfor ({index} in {raw}.indices) {{\nval {row} = {raw}[{index}] as {row_type}\n{}\n{}\n{output}.add({value})\n}}\n",
333            child.output_locals, child.materialize
334        ),
335        cleanup: if child.cleanup.is_empty() { String::new() } else { format!(
336            "try {{\nfor ({index} in {raw}.indices) {{\nvar {failure}: kotlin.Throwable? = {current_failure}\ntry {{ val {row} = {raw}[{index}] as {row_type}\n{}\n{}\n}} catch ({cleanup}: Throwable) {{ {record_row_failure} }}\n{finish_row}\n}}\n}} catch ({cleanup}: Throwable) {{ {record_outer_failure} }}\n",
337            child.output_locals, child.cleanup
338        ) },
339    })
340}

342/// The adapter selects its native scope before looking for a sequence.
343#[allow(clippy::type_complexity)]
344pub(crate) fn return_sequence(
345    layout: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
346) -> Option<(
347    &prebindgen_flat::Wire<JniWire>,
348    &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
349    &JniSequence,
350)> {
351    outer_sequences(layout)
352        .into_iter()
353        .next()
354        .and_then(|(element, controls, metadata)| match controls {
355            [carrier] => Some((carrier, element, metadata)),
356            _ => None,
357        })
358}
359pub(crate) fn callback_target(
360    layout: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
361) -> Result<syn::Ident> {
362    if outer_sequences(layout).len() != 1 || crate::group::result_packet_fields(layout).len() > 2 {
363        return Err(
364            "callback_return permits only one sequence and its optional presence flag; cannot discard additional return values".into(),
365        );
366    }
367    return_sequence(layout)
368        .map(|(carrier, _, _)| carrier.name.clone())
369        .ok_or_else(|| "callback_return requires a sequence boundary".into())
370}

372/// Whether the successful return uses the function's sequence callback.
373pub(crate) fn callback_delivery(
374    node: &prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
375) -> Result<bool> {
376    let sequences = outer_sequences(node);
377    if !sequences
378        .iter()
379        .any(|(_, _, metadata)| **metadata == JniSequence::Callback)
380    {
381        return Ok(false);
382    }
383    callback_optional(node)?;
384    callback_target(node)?;
385    if crate::group::result_packet_fields(node).len() != 1 {
386        return Err("callback_return cannot discard additional return values".into());
387    }
388    Ok(true)
389}

391/// A fold's nullable accumulator can represent one absent collection, not
392/// distinguish None from Some(None). Integer niches must not silently enable
393/// that previously unsupported shape. Options inside rows are independent.
394pub(crate) fn callback_optional(
395    node: &prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
396) -> Result<bool> {
397    use prebindgen_flat::WireNode;
398    match node {
399        WireNode::Chain {
400            meta: JniChain::Optional(_),
401            child,
402            ..
403        } => {
404            if callback_optional(child)? {
405                return Err("callback_return cannot discard additional return values from nested optional collections".into());
406            }
407            Ok(true)
408        }
409        WireNode::Branch {
410            meta: JniBranch::Group(_),
411            children,
412            ..
413        } => {
414            let mut optional = false;
415            for child in children {
416                optional |= callback_optional(child)?;
417            }
418            Ok(optional)
419        }
420        _ => Ok(false),
421    }
422}

424/// Function sequence callbacks are not available in parameters, error channels or rows.
425pub(crate) fn validate_buffered(
426    node: &prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
427) -> Result<()> {
428    match node {
429        prebindgen_flat::WireNode::Leaf(_) => {}
430        prebindgen_flat::WireNode::Chain {
431            child: element,
432            meta: JniChain::Sequence(metadata),
433            ..
434        } => {
435            if *metadata == JniSequence::Callback {
436                return Err(
437                    "callback sequence is only supported as the function's successful return"
438                        .into(),
439                );
440            }
441            validate_buffered(element)?;
442        }
443        prebindgen_flat::WireNode::Chain {
444            meta: JniChain::Optional(_) | JniChain::Bulk(_),
445            child: element,
446            ..
447        } => validate_buffered(element)?,
448        prebindgen_flat::WireNode::Branch {
449            meta: JniBranch::Group(_),
450            children: elements,
451            ..
452        }
453        | prebindgen_flat::WireNode::Branch {
454            meta: JniBranch::Alternatives | JniBranch::Enum(_),
455            children: elements,
456            ..
457        } => {
458            for element in elements {
459                validate_buffered(element)?;
460            }
461        }
462    }
463    Ok(())
464}

466// A sequence owns a separate row scope: find the outer occurrences only.
467#[allow(clippy::type_complexity)] // Borrow the element schema, controls and delivery policy together.
468fn outer_sequences(
469    node: &prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
470) -> Vec<(
471    &prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
472    &[prebindgen_flat::Wire<JniWire>],
473    &JniSequence,
474)> {
475    fn visit<'a>(
476        node: &'a prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
477        out: &mut Vec<(
478            &'a prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
479            &'a [prebindgen_flat::Wire<JniWire>],
480            &'a JniSequence,
481        )>,
482    ) {
483        use prebindgen_flat::WireNode;
484        match node {
485            WireNode::Chain {
486                meta: JniChain::Sequence(meta),
487                child,
488                controls,
489            } => out.push((child, controls, meta)),
490            WireNode::Chain { child, .. } => visit(child, out),
491            WireNode::Branch { children, .. } => {
492                for child in children {
493                    visit(child, out)
494                }
495            }
496            WireNode::Leaf(_) => {}
497        }
498    }
499    let mut out = vec![];
500    visit(node, &mut out);
501    out
502}

504#[cfg(test)]
505mod delivery_tests {
506    use super::*;
507    use prebindgen_flat::WireNode;
508    fn leaf() -> prebindgen_flat::WireNode<JniWire, JniChain, JniBranch> {
509        WireNode::Leaf(vec![prebindgen_flat::Wire {
510            name: syn::parse_quote!(element),
511            kind: JniWire::scalar(Primitive::I32),
512        }])
513    }
514    fn sequence(
515        delivery: JniSequence,
516        name: &str,
517        element: prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
518    ) -> prebindgen_flat::WireNode<JniWire, JniChain, JniBranch> {
519        let kind = match delivery {
520            JniSequence::Bulk => JniWire::NativeRows {
521                condition: "true".into(),
522            },
523            JniSequence::Callback => JniWire::selector(),
524            JniSequence::Buffered | JniSequence::TypedBuffered => JniWire::Rows {
525                condition: "true".into(),
526                null_niche_available: true,
527            },
528        };
529        WireNode::Chain {
530            controls: vec![prebindgen_flat::Wire {
531                name: syn::parse_str(name).unwrap(),
532                kind,
533            }],
534            child: Box::new(element),
535            meta: JniChain::Sequence(delivery),
536        }
537    }
538    #[test]
539    fn delivery_is_explicit_and_callback_scope_is_checked() {
540        let visitor = sequence(JniSequence::Callback, "visitor", leaf());
541        assert!(callback_delivery(&visitor).unwrap());
542        assert!(validate_buffered(&visitor)
543            .unwrap_err()
544            .contains("successful return"));
545        let packed = sequence(JniSequence::Buffered, "rows", leaf());
546        assert!(!callback_delivery(&packed).unwrap());
547        validate_buffered(&packed).unwrap();
548        let nested = sequence(JniSequence::Buffered, "rows", visitor.clone());
549        assert!(crate::group::validate_layout(&nested)
550            .unwrap_err()
551            .contains("successful return"));
552        let extra = WireNode::Branch {
553            controls: vec![],
554            children: vec![visitor, leaf()],
555            meta: JniBranch::Group(JniGroup::Values),
556        };
557        assert!(callback_delivery(&extra)
558            .unwrap_err()
559            .contains("additional return"));
560    }
