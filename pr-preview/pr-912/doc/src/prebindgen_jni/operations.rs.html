27/// Resolved incoming conversion. Configuration cannot enter callback domains:
28/// ```compile_fail
29/// use prebindgen_jni::{JniInputArrow,JniInputConfigArrow};
30/// fn final_arrow(config:JniInputConfigArrow)->JniInputArrow { config }
31/// ```
32/// Final leaves and enum branches cannot contain configuration payloads:
33/// ```compile_fail
34/// use prebindgen_jni::{JniInput,JniInputLeafOp};
35/// let _:JniInput=JniInput::Leaf(JniInputLeafOp::ConfigDefault(()));
36/// ```
37/// ```compile_fail
38/// use prebindgen_jni::{JniInput,JniInputBranch};
39/// let _:JniInput=JniInput::Branch {
40///     operation:JniInputBranch::ConfigEnumFields(()),children:vec![],
41/// };
42/// ```
43pub type JniInput = InputNode<JniInputOps, JniOutputOps>;
44/// Resolved outgoing conversion. Partial named fields have no final payload:
45/// ```compile_fail
46/// use prebindgen_jni::{JniOutput,JniOutputBranch,JniNamedFields,ProjectionMode};
47/// let _:JniOutput=JniOutput::Branch {
48///     operation:JniOutputBranch::ConfigNamedFields(JniNamedFields {fields:vec![],mode:ProjectionMode::Owned}),
49///     children:vec![],
50/// };
51/// ```
52/// None is a configuration payload, not an inhabitant of the final payload type:
53/// ```compile_fail
54/// use prebindgen_jni::{JniOutput,JniOutputLeafOp};
55/// let _:JniOutput=JniOutput::Leaf(JniOutputLeafOp::ConfigValue(None));
56/// ```
57pub type JniOutput = OutputNode<JniInputOps, JniOutputOps>;
58/// Incoming conversion configuration.
59/// ```
60/// use prebindgen_jni::{JniInputConfig,JniOutputConfig,prelude::*};
61/// let _:JniInputConfig=JniInputConfig::auto();
62/// let _:JniOutputConfig=JniOutputConfig::auto();
63/// ```
64pub type JniInputConfig = InputNode<JniInputConfigOps, JniOutputConfigOps>;
65/// Outgoing conversion configuration.
66pub type JniOutputConfig = OutputNode<JniInputConfigOps, JniOutputConfigOps>;

305pub(crate) fn lift_input_bulk(tree: JniInput) -> JniInput {
306    match tree {
307        JniInput::Branch {
308            operation: JniInputBranch::Enum { value },
309            children: variants,
310        } => JniInput::Branch {
311            operation: JniInputBranch::Enum { value },
312            children: variants.into_iter().map(lift_input_bulk).collect(),
313        },
314        JniInput::Unary {
315            operation: JniInputUnary::Convert { conversion },
316            child: input,
317        } => JniInput::Unary {
318            operation: JniInputUnary::Convert { conversion },
319            child: Box::new(lift_input_bulk(*input)),
320        },
321        JniInput::Unary {
322            operation: JniInputUnary::Bulk,
323            child: c,
324        } => JniInput::Unary {
325            operation: JniInputUnary::Bulk,
326            child: Box::new(lift_input_bulk(*c)),
327        },
328        JniInput::Unary {
329            operation: JniInputUnary::Temporary,
330            child: c,
331        } => JniInput::Unary {
332            operation: JniInputUnary::Temporary,
333            child: Box::new(lift_input_bulk(*c)),
334        },
335        JniInput::Unary {
336            operation: JniInputUnary::Sequence,
337            child: c,
338        } => {
339            let child = lift_input_bulk(*c);
340            if input_has_bulk(&child) {
341                JniInput::Unary {
342                    operation: JniInputUnary::Bulk,
343                    child: Box::new(JniInput::Unary {
344                        operation: JniInputUnary::Sequence,
345                        child: Box::new(child),
346                    }),
347                }
348            } else {
349                JniInput::Unary {
350                    operation: JniInputUnary::Sequence,
351                    child: Box::new(child),
352                }
353            }
354        }
355        JniInput::Unary {
356            operation: JniInputUnary::Optional,
357            child: c,
358        } => match lift_input_bulk(*c) {
359            JniInput::Unary {
360                operation: JniInputUnary::Bulk,
361                child: c,
362            } => JniInput::Unary {
363                operation: JniInputUnary::Bulk,
364                child: Box::new(JniInput::Unary {
365                    operation: JniInputUnary::Optional,
366                    child: c,
367                }),
368            },
369            c => JniInput::Unary {
370                operation: JniInputUnary::Optional,
371                child: Box::new(c),
372            },
373        },
374        JniInput::Branch {
375            operation: JniInputBranch::Fields,
376            children: c,
377        } => JniInput::Branch {
378            operation: JniInputBranch::Fields,
379            children: c.into_iter().map(lift_input_bulk).collect(),
380        },
381        JniInput::Branch {
382            operation: JniInputBranch::Record,
383            children: c,
384        } => JniInput::Branch {
385            operation: JniInputBranch::Record,
386            children: c.into_iter().map(lift_input_bulk).collect(),
387        },
388        JniInput::Branch {
389            operation: JniInputBranch::Selector,
390            children: c,
391        } => JniInput::Branch {
392            operation: JniInputBranch::Selector,
393            children: c.into_iter().map(lift_input_bulk).collect(),
394        },
395        JniInput::Branch {
396            operation: JniInputBranch::Constructor { function },
397            children: args,
398        } => JniInput::Branch {
399            operation: JniInputBranch::Constructor { function },
400            children: args.into_iter().map(lift_input_bulk).collect(),
401        },
402        JniInput::Branch {
403            operation,
404            children,
405        } => JniInput::Branch {
406            operation,
407            children: children.into_iter().map(lift_input_bulk).collect(),
408        },
409        c => c,
410    }
411}
412pub(crate) fn lift_output_bulk(tree: JniOutput) -> JniOutput {
413    match tree {
414        JniOutput::Branch {
415            operation: JniOutputBranch::Enum { value, mode },
416            children: variants,
417        } => JniOutput::Branch {
418            operation: JniOutputBranch::Enum { value, mode },
419            children: variants.into_iter().map(lift_output_bulk).collect(),
420        },
421        JniOutput::Unary {
422            operation: JniOutputUnary::Convert { conversion },
423            child: output,
424        } => JniOutput::Unary {
425            operation: JniOutputUnary::Convert { conversion },
426            child: Box::new(lift_output_bulk(*output)),
427        },
428        JniOutput::Unary {
429            operation: JniOutputUnary::Bulk,
430            child: c,
431        } => JniOutput::Unary {
432            operation: JniOutputUnary::Bulk,
433            child: Box::new(lift_output_bulk(*c)),
434        },
435        JniOutput::Unary {
436            operation: JniOutputUnary::Sequence { delivery, mode },
437            child: element,
438        } => {
439            let child = lift_output_bulk(*element);
440            if delivery != SequenceDelivery::Callback && output_has_bulk(&child) {
441                JniOutput::Unary {
442                    operation: JniOutputUnary::Bulk,
443                    child: Box::new(JniOutput::Unary {
444                        operation: JniOutputUnary::Sequence { delivery, mode },
445                        child: Box::new(child),
446                    }),
447                }
448            } else {
449                JniOutput::Unary {
450                    operation: JniOutputUnary::Sequence { delivery, mode },
451                    child: Box::new(child),
452                }
453            }
454        }
455        JniOutput::Unary {
456            operation: JniOutputUnary::Optional,
457            child: c,
458        } => match lift_output_bulk(*c) {
459            JniOutput::Unary {
460                operation: JniOutputUnary::Bulk,
461                child: c,
462            } => JniOutput::Unary {
463                operation: JniOutputUnary::Bulk,
464                child: Box::new(JniOutput::Unary {
465                    operation: JniOutputUnary::Optional,
466                    child: c,
467                }),
468            },
469            c => JniOutput::Unary {
470                operation: JniOutputUnary::Optional,
471                child: Box::new(c),
472            },
473        },
474        JniOutput::Branch {
475            operation: JniOutputBranch::Fields,
476            children: c,
477        } => JniOutput::Branch {
478            operation: JniOutputBranch::Fields,
479            children: c.into_iter().map(lift_output_bulk).collect(),
480        },
481        JniOutput::Branch {
482            operation: JniOutputBranch::OwnedFields,
483            children: c,
484        } => JniOutput::Branch {
485            operation: JniOutputBranch::OwnedFields,
486            children: c.into_iter().map(lift_output_bulk).collect(),
487        },
488        JniOutput::Branch {
489            operation: JniOutputBranch::Record { mode },
490            children: fields,
491        } => JniOutput::Branch {
492            operation: JniOutputBranch::Record { mode },
493            children: fields.into_iter().map(lift_output_bulk).collect(),
494        },
495        JniOutput::Branch {
496            operation: JniOutputBranch::Combine,
497            children: c,
498        } => JniOutput::Branch {
499            operation: JniOutputBranch::Combine,
500            children: c.into_iter().map(lift_output_bulk).collect(),
501        },
502        JniOutput::Unary {
503            operation: JniOutputUnary::Result { error },
504            child: ok,
505        } => JniOutput::Unary {
506            operation: JniOutputUnary::Result { error },
507            child: Box::new(lift_output_bulk(*ok)),
508        },
509        JniOutput::Unary {
510            operation: JniOutputUnary::Getter { function },
511            child: output,
512        } => JniOutput::Unary {
513            operation: JniOutputUnary::Getter { function },
514            child: Box::new(lift_output_bulk(*output)),
515        },
516        JniOutput::Unary {
517            operation: JniOutputUnary::Into { function },
518            child: output,
519        } => JniOutput::Unary {
520            operation: JniOutputUnary::Into { function },
521            child: Box::new(lift_output_bulk(*output)),
522        },
523        JniOutput::Unary {
524            operation: JniOutputUnary::Named { name },
525            child: output,
526        } => JniOutput::Unary {
527            operation: JniOutputUnary::Named { name },
528            child: Box::new(lift_output_bulk(*output)),
529        },
530        JniOutput::Branch {
531            operation,
532            children,
533        } => JniOutput::Branch {
534            operation,
535            children: children.into_iter().map(lift_output_bulk).collect(),
536        },
537        c => c,
538    }
539}

541fn input_has_bulk(tree: &JniInput) -> bool {
542    match tree {
543        JniInput::Branch {
544            operation: JniInputBranch::Enum { .. },
545            children: variants,
546            ..
547        } => variants.iter().any(input_has_bulk),
548        JniInput::Unary {
549            operation: JniInputUnary::Convert { .. },
550            child: input,
551            ..
552        } => input_has_bulk(input),
553        JniInput::Unary {
554            operation: JniInputUnary::Bulk,
555            child: _,
556        } => true,
557        JniInput::Unary {
558            operation: JniInputUnary::Optional,
559            child: c,
560        }
561        | JniInput::Unary {
562            operation: JniInputUnary::Sequence,
563            child: c,
564        } => input_has_bulk(c),
565        JniInput::Branch {
566            operation: JniInputBranch::Fields,
567            children: c,
568        }
569        | JniInput::Branch {
570            operation: JniInputBranch::Record,
571            children: c,
572        }
573        | JniInput::Branch {
574            operation: JniInputBranch::Selector,
575            children: c,
576        } => c.iter().any(input_has_bulk),
577        JniInput::Branch {
578            operation: JniInputBranch::Constructor { .. },
579            children: args,
580            ..
581        } => args.iter().any(input_has_bulk),
582        JniInput::Branch { children, .. } => children.iter().any(input_has_bulk),
583        _ => false,
584    }
585}
586fn output_has_bulk(tree: &JniOutput) -> bool {
587    match tree {
588        JniOutput::Branch {
589            operation: JniOutputBranch::Enum { .. },
590            children: variants,
591            ..
592        } => variants.iter().any(output_has_bulk),
593        // Frames inside callback elements belong to those elements, including
594        // when the callback sequence is nested in another structural node.
595        JniOutput::Unary {
596            operation:
597                JniOutputUnary::Sequence {
598                    delivery: SequenceDelivery::Callback,
599                    ..
600                },
601            ..
602        } => false,
603        JniOutput::Unary {
604            operation: JniOutputUnary::Convert { .. },
605            child: output,
606            ..
607        } => output_has_bulk(output),
608        JniOutput::Unary {
609            operation: JniOutputUnary::Bulk,
610            child: _,
611        } => true,
612        JniOutput::Unary {
613            operation: JniOutputUnary::Optional,
614            child: c,
615        }
616        | JniOutput::Unary {
617            operation: JniOutputUnary::Named { .. },
618            child: c,
619            ..
620        }
621        | JniOutput::Unary {
622            operation: JniOutputUnary::Getter { .. },
623            child: c,
624            ..
625        }
626        | JniOutput::Unary {
627            operation: JniOutputUnary::Into { .. },
628            child: c,
629            ..
630        }
631        | JniOutput::Unary {
632            operation: JniOutputUnary::Sequence { .. },
633            child: c,
634            ..
635        } => output_has_bulk(c),
636        JniOutput::Branch {
637            operation: JniOutputBranch::Fields,
638            children: c,
639        }
640        | JniOutput::Branch {
641            operation: JniOutputBranch::OwnedFields,
642            children: c,
643        }
644        | JniOutput::Branch {
645            operation: JniOutputBranch::Combine,
646            children: c,
647        }
648        | JniOutput::Branch {
649            operation: JniOutputBranch::Record { .. },
650            children: c,
651            ..
652        } => c.iter().any(output_has_bulk),
653        JniOutput::Branch { children, .. } => children.iter().any(output_has_bulk),
654        _ => false,
655    }
656}

658pub(crate) fn sequence_callback(
659    cx: &mut crate::JniResolve<'_>,
660    ty: &syn::Type,
661    mut output: JniOutput,
662    structural_default: bool,
663) -> Result<JniOutput> {
664    fn select(output: &mut JniOutput) -> usize {
665        match output {
666            JniOutput::Unary {
667                operation: JniOutputUnary::Sequence { delivery, .. },
668                ..
669            } => {
670                *delivery = SequenceDelivery::Callback;
671                1
672            }
673            JniOutput::Unary {
674                operation: JniOutputUnary::Optional,
675                child,
676            }
677            | JniOutput::Unary {
678                operation: JniOutputUnary::Bulk,
679                child,
680            }
681            | JniOutput::Unary {
682                operation: JniOutputUnary::Named { .. },
683                child,
684                ..
685            }
686            | JniOutput::Unary {
687                operation: JniOutputUnary::Getter { .. },
688                child,
689                ..
690            }
691            | JniOutput::Unary {
692                operation: JniOutputUnary::Into { .. },
693                child,
694                ..
695            }
696            | JniOutput::Unary {
697                operation: JniOutputUnary::Convert { .. },
698                child,
699                ..
700            } => select(child),
701            JniOutput::Branch {
702                operation: JniOutputBranch::Combine,
703                children,
704            }
705            | JniOutput::Branch {
706                operation: JniOutputBranch::Fields,
707                children,
708            }
709            | JniOutput::Branch {
710                operation: JniOutputBranch::OwnedFields,
711                children,
712            }
713            | JniOutput::Branch {
714                operation: JniOutputBranch::Record { .. },
715                children,
716                ..
717            } => children.iter_mut().map(select).sum(),
718            _ => 0,
719        }
720    }
721    if select(&mut output) != 1 {
722        fn structural(ty: &syn::Type, output: &JniOutput) -> Option<crate::JniOutputConfig> {
723            match output {
724                JniOutput::Leaf(JniOutputLeafOp::Value)
725                    if prebindgen_flat::sequence_parts(ty).is_some() =>
726                {
727                    Some(crate::JniOutputConfig::Unary {
728                        operation: crate::JniOutputUnary::Sequence {
729                            delivery: SequenceDelivery::Callback,
730                            mode: None,
731                        },
732                        child: Box::new(crate::JniOutputConfig::auto()),
733                    })
734                }
735                JniOutput::Unary {
736                    operation: JniOutputUnary::Optional,
737                    child,
738                } => Some(crate::JniOutputConfig::Unary {
739                    operation: crate::JniOutputUnary::Optional,
740                    child: Box::new(structural(prebindgen_flat::option_inner(ty)?, child)?),
741                }),
742                _ => None,
743            }
744        }
745        if structural_default {
746            if let Some(tree) = structural(ty, &output) {
747                output = cx.output(ty, &tree)?;
748            } else {
749                return Err("callback_return requires exactly one outer sequence".into());
750            }
751        } else {
752            return Err("callback_return requires exactly one outer sequence".into());
753        }
754    }
755    Ok(output)
756}
