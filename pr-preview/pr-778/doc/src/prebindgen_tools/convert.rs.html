1//! Custom conversions: a source type that crosses as another type.
2//!
3//! `convert!(Millis).input(fun!(millis_from_raw)).output(fun!(millis_to_raw))`
4//! says that `Millis` never crosses as itself: it crosses as `u64` — the
5//! **representation** — and these two functions convert. The adapter lowers
6//! the representation like any other type and wraps the resulting wires in
7//! the conversion's stages ([`Stage::apply`]).
8//!
9//! A stage is a function (a flat item, or a binding-local path with a
10//! stated signature) or a standard trait impl (`From`, `Into`, `TryFrom`,
11//! `TryInto`). A function returning `Result` makes its stage fallible; the
12//! error is reported through its `Display`.
13

16    Flat,
17};

19use quote::{quote, ToTokens};

23/// A function a declaration refers to: a flat item by name, or a
24/// binding-local path with its signature stated.
25#[derive(Clone, Debug)]
26pub struct FnRef {
27    pub path: syn::Path,
28    pub sig: Option<syn::Signature>,
29}
30
31impl FnRef {
32    pub fn new(path: syn::Path) -> Self {
33        Self { path, sig: None }
34    }
35
36    /// State the signature of a function the flat model does not know.
37    pub fn sig(mut self, sig: syn::Signature) -> Self {
38        self.sig = Some(sig);
39        self
40    }
41
42    /// The last path segment: the flat name, or the local function's name.
43    pub fn name(&self) -> &syn::Ident {
44        &self.path.segments.last().expect("a non-empty path").ident
45    }
46
47    /// Whether this names a flat item (a bare name, no stated signature).
48    pub fn is_flat(&self) -> bool {
49        self.sig.is_none() && self.path.segments.len() == 1 && self.path.leading_colon.is_none()
50    }
51
52    /// The function as the model reads it: the flat item, or the stated
53    /// signature lowered against the flat namespace.
54    pub fn resolve(&self, flat: &Flat) -> Result<Function, String> {
55        match &self.sig {
56            Some(sig) => {
57                let mut sig = sig.clone();
58                sig.ident = self.name().clone();
59                let item: syn::ItemFn = syn::parse_quote!(pub #sig { unimplemented!() });
60                flat.lower_signature(&item)
61                    .map_err(|e| format!("`{}`: {e}", path_string(&self.path)))
62            }
63            None if !self.is_flat() => Err(format!(
64                "`{}` is not a #[prebindgen] function: state its signature with `.sig(sig!(..))`",
65                path_string(&self.path)
66            )),
67            None => flat
68                .function(self.name())
69                .cloned()
70                .ok_or_else(|| format!("`{}` is not a #[prebindgen] function", self.name())),
71        }
72    }
73
74    /// The path generated code calls.
75    pub fn callee(&self, q: &Qualifier<'_>) -> TokenStream {
76        if self.is_flat() {
77            q.path(self.name())
78        } else {
79            self.path.to_token_stream()
80        }
81    }
82}
83
84fn path_string(p: &syn::Path) -> String {
85    p.to_token_stream().to_string().replace(' ', "")
86}

88/// How one direction of a conversion is carried out.
89#[derive(Clone, Debug)]
90pub enum Via {
91    Fn(FnRef),
92    From(syn::Type),
93    Into(syn::Type),
94    TryFrom(syn::Type),
95    TryInto(syn::Type),
96}
97
98impl From<FnRef> for Via {
99    fn from(f: FnRef) -> Self {
100        Via::Fn(f)
101    }
102}

104/// A declared conversion for one source type.
105#[derive(Clone, Debug)]
106pub struct Conversion {
107    pub ty: syn::Type,
108    pub input: Option<Via>,
109    pub output: Option<Via>,
110}
111
112impl Conversion {
113    pub fn new(ty: syn::Type) -> Self {
114        Self {
115            ty,
116            input: None,
117            output: None,
118        }
119    }
120
121    /// How a representation value becomes the source type.
122    pub fn input(mut self, via: impl Into<Via>) -> Self {
123        self.input = Some(via.into());
124        self
125    }
126
127    /// How the source type becomes its representation.
128    pub fn output(mut self, via: impl Into<Via>) -> Self {
129        self.output = Some(via.into());
130        self
131    }
132
133    /// Resolve both directions against the model.
134    pub fn resolve(&self, flat: &Flat) -> Result<ResolvedConversion, String> {
135        let target = flat
136            .classify(&self.ty)
137            .map_err(|e| format!("convert!({}): {e}", self.ty.to_token_stream()))?;
138        let input = self
139            .input
140            .as_ref()
141            .map(|v| Stage::resolve(flat, v, &target, Direction::In))
142            .transpose()?;
143        let output = self
144            .output
145            .as_ref()
146            .map(|v| Stage::resolve(flat, v, &target, Direction::Out))
147            .transpose()?;
148        Ok(ResolvedConversion {
149            target,
150            input,
151            output,
152        })
153    }
154}

156/// A conversion resolved against the model.
157#[derive(Clone, Debug)]
158pub struct ResolvedConversion {
159    /// The source type being converted.
160    pub target: TypeRef,
161    /// Representation → source.
162    pub input: Option<Stage>,
163    /// Source → representation.
164    pub output: Option<Stage>,
165}
166
167#[derive(Clone, Copy, PartialEq, Eq, Debug)]
168enum Direction {
169    In,
170    Out,
171}
172
173/// One direction of a conversion.
174#[derive(Clone, Debug)]
175pub struct Stage {
176    /// The representation type on the boundary side.
177    pub repr: TypeRef,
178    /// Whether the stage can fail.
179    pub fallible: bool,
180    how: How,
181}
182
183#[derive(Clone, Debug)]
184enum How {
185    /// Call a function; `by_ref` when it takes its argument by reference.
186    Call {
187        fun: FnRef,
188        by_ref: bool,
189    },
190    From(TypeRef),
191    Into(TypeRef),
192    TryFrom(TypeRef),
193    TryInto(TypeRef),
194}
195
196impl Stage {
197    fn resolve(flat: &Flat, via: &Via, target: &TypeRef, dir: Direction) -> Result<Self, String> {
198        let classify = |t: &syn::Type| {
199            flat.classify(t)
200                .map_err(|e| format!("conversion type `{}`: {e}", t.to_token_stream()))
201        };
202        Ok(match via {
203            Via::Fn(fun) => {
204                let f = fun.resolve(flat)?;
205                let name = path_string(&fun.path);
206                let [param] = f.params.as_slice() else {
207                    return Err(format!(
208                        "conversion fn `{name}` must take exactly one argument, takes {}",
209                        f.params.len()
210                    ));
211                };
212                let (ret, fallible) = match f.ret.kind() {
213                    TypeKind::Fallible { ok, .. } => ((**ok).clone(), true),
214                    _ => (f.ret.clone(), false),
215                };
216                // The side facing the converted type must be that type: an
217                // input stage returns it, an output stage takes it (or a
218                // borrow of it).
219                let facing = match dir {
220                    Direction::In => &ret,
221                    Direction::Out => match param.ty.kind() {
222                        TypeKind::Ref { inner, .. } => &**inner,
223                        _ => &param.ty,
224                    },
225                };
226                if facing.key() != target.key() {
227                    let (what, side) = match dir {
228                        Direction::In => ("input", "return"),
229                        Direction::Out => ("output", "take"),
230                    };
231                    return Err(format!(
232                        "{what} conversion fn `{name}` must {side} `{target}`, not `{facing}`"
233                    ));
234                }
235                match dir {
236                    Direction::In => Stage {
237                        repr: param.ty.clone(),
238                        fallible,
239                        how: How::Call {
240                            fun: fun.clone(),
241                            by_ref: false,
242                        },
243                    },
244                    Direction::Out => {
245                        let by_ref = matches!(param.ty.kind(), TypeKind::Ref { .. });
246                        Stage {
247                            repr: ret,
248                            fallible,
249                            how: How::Call {
250                                fun: fun.clone(),
251                                by_ref,
252                            },
253                        }
254                    }
255                }
256            }
257            Via::From(t) => {
258                let r = classify(t)?;
259                Stage {
260                    repr: r.clone(),
261                    fallible: false,
262                    how: How::From(r),
263                }
264            }
265            Via::Into(t) => {
266                let r = classify(t)?;
267                Stage {
268                    repr: r.clone(),
269                    fallible: false,
270                    how: How::Into(r),
271                }
272            }
273            Via::TryFrom(t) => {
274                let r = classify(t)?;
275                Stage {
276                    repr: r.clone(),
277                    fallible: true,
278                    how: How::TryFrom(r),
279                }
280            }
281            Via::TryInto(t) => {
282                let r = classify(t)?;
283                Stage {
284                    repr: r.clone(),
285                    fallible: true,
286                    how: How::TryInto(r),
287                }
288            }
289        })
290    }
291
292    /// Apply the stage to `value`. For an input stage `value` is the
293    /// representation and the result the source type; for an output stage
294    /// the other way round. A fallible stage produces an expression using
295    /// `?` on a `Result<_, String>`.
296    pub fn apply(&self, q: &Qualifier<'_>, target: &TypeRef, value: &TokenStream) -> TokenStream {
297        let t = q.ty(target);
298        let call = match &self.how {
299            How::Call { fun, by_ref } => {
300                let callee = fun.callee(q);
301                if *by_ref {
302                    quote!(#callee(&#value))
303                } else {
304                    quote!(#callee(#value))
305                }
306            }
307            How::From(r) => {
308                let r = q.ty(r);
309                quote!(<#t as ::core::convert::From<#r>>::from(#value))
310            }
311            How::Into(r) => {
312                let r = q.ty(r);
313                quote!(<#t as ::core::convert::Into<#r>>::into(#value))
314            }
315            How::TryFrom(r) => {
316                let r = q.ty(r);
317                quote!(<#t as ::core::convert::TryFrom<#r>>::try_from(#value))
318            }
319            How::TryInto(r) => {
320                let r = q.ty(r);
321                quote!(<#t as ::core::convert::TryInto<#r>>::try_into(#value))
322            }
323        };
324        if self.fallible {
325            quote!(#call.map_err(|__e| ::std::string::ToString::to_string(&__e))?)
326        } else {
327            call
328        }
329    }
330}

332impl Stage {
333    /// The function this stage calls, when it calls one.
334    pub fn function(&self) -> Option<&FnRef> {
335        match &self.how {
336            How::Call { fun, .. } => Some(fun),
337            _ => None,
338        }
339    }
340
341    /// This input stage over its representation's input: the wires `repr`
342    /// reads become the converted value. The representation is bound to `r`
343    /// and `check` (statements, possibly using `?`) runs on it first.
344    pub fn decode(
345        &self,
346        q: &Qualifier<'_>,
347        target: &TypeRef,
348        repr: Input,
349        r: &syn::Ident,
350        check: TokenStream,
351    ) -> Input {
352        let applied = self.apply(q, target, &r.to_token_stream());
353        let fallible = repr.fallible || self.fallible || !check.is_empty();
354        let e = repr.expr;
355        Input {
356            wires: repr.wires,
357            expr: quote!({ let #r = #e; #check #applied }),
358            fallible,
359            pass: None,
360        }
361    }
362
363    /// This output stage ahead of its representation's output: `value`
364    /// becomes the representation, bound to `r`, which `repr` reads. `check`
365    /// runs on it in between.
366    pub fn encode(
367        &self,
368        q: &Qualifier<'_>,
369        target: &TypeRef,
370        value: &TokenStream,
371        r: &syn::Ident,
372        check: TokenStream,
373        repr: Output,
374    ) -> Output {
375        let applied = self.apply(q, target, value);
376        let fallible = repr.fallible || self.fallible || !check.is_empty();
377        let e = &repr.expr;
378        Output {
379            wires: repr.wires.clone(),
380            expr: quote!({ let #r = #applied; #check #e }),
381            fallible,
382        }
383    }
384}
