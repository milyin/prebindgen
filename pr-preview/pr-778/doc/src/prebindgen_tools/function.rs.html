1//! The function wrapper writer.
2//!
3//! A wrapper is one exported function that receives wires, converts them to
4//! the source function's arguments, calls it, and hands the result back as
5//! wires:
6//!
7//! ```text
8//! <attrs>
9//! pub unsafe extern "C" fn <name><generics>(<leading>, <param wires>, <return wires>, <trailing>) -> <return type> {
10//!     <prologue>
11//!     let <param> = <param input>;          // on failure: <fail>
12//!     ...
13//!     let __result = <callee>(<params>);
14//!     <return body>
15//! }
16//! ```
17//!
18//! The writer owns the shape; the adapter owns every decision, through
19//! [`FunctionCallbacks`]: which wires a parameter becomes and how they turn
20//! back into the argument, which wires the result leaves on, and what the
21//! wrapper does when a conversion fails. The builder methods customize the
22//! header and the fixed parts of the body.
23
24use prebindgen_flat::flat::{Function, TypeRef};
25use proc_macro2::TokenStream;
26use quote::{quote, ToTokens};
27
28use crate::wire::{Input, Wire};
29
30/// How a wrapper delivers the source function's result.
31#[derive(Clone, Debug, Default)]
32pub struct Return {
33    /// The wrapper's return type; `None` for `()`.
34    pub ty: Option<TokenStream>,
35    /// Extra parameters the result needs — out-parameters, a builder object,
36    /// an error slot — placed after the parameters' wires.
37    pub wires: Vec<Wire>,
38    /// Statements that consume [`RESULT`] and end in the wrapper's return
39    /// value (a tail expression, or nothing for `()`).
40    pub body: TokenStream,
41}
42
43/// The local the source function's result is bound to; [`Return::body`]
44/// reads it.
45pub const RESULT: &str = "__result";
46
47/// The local a failed conversion's message (`String`) is bound to;
48/// [`FunctionCallbacks::fail`] reads it.
49pub const ERROR: &str = "__err";
50
51/// The result local, as an ident.
52pub fn result_ident() -> syn::Ident {
53    syn::Ident::new(RESULT, proc_macro2::Span::call_site())
54}
55
56/// The error local, as an ident.
57pub fn error_ident() -> syn::Ident {
58    syn::Ident::new(ERROR, proc_macro2::Span::call_site())
59}
60
61/// The adapter's decisions for one wrapper.
62pub trait FunctionCallbacks {
63    type Error;
64
65    /// How the result of type `ret` leaves the wrapper. Called first.
66    fn ret(&mut self, ret: &TypeRef) -> Result<Return, Self::Error>;
67
68    /// The wires parameter `name` of type `ty` arrives on, and how they
69    /// become the argument. Called after [`Self::ret`], in parameter order.
70    fn param(&mut self, name: &syn::Ident, ty: &TypeRef) -> Result<Input, Self::Error>;
71
72    /// Statements run when a parameter's conversion fails, with the message
73    /// bound to [`ERROR`]. They must diverge — return from the wrapper or
74    /// panic. `ret` is what [`Self::ret`] returned.
75    fn fail(&mut self, ret: &Return) -> TokenStream;
76}
77
78/// Builds the call expression from the callee and the arguments.
79type CallFn<'f> = Box<dyn Fn(&TokenStream, &[TokenStream]) -> TokenStream + 'f>;
80
81/// Writes one wrapper around a flat [`Function`].
82pub struct FunctionWriter<'f> {
83    func: &'f Function,
84    name: syn::Ident,
85    callee: TokenStream,
86    attrs: Vec<TokenStream>,
87    vis: TokenStream,
88    abi: Option<String>,
89    unsafety: bool,
90    generics: TokenStream,
91    leading: Vec<Wire>,
92    trailing: Vec<Wire>,
93    prologue: TokenStream,
94    call: Option<CallFn<'f>>,
95}
96
97impl<'f> FunctionWriter<'f> {
98    /// A wrapper around `func`, called through `callee` (usually
99    /// [`Qualifier::path`](crate::Qualifier::path) of the function's name),
100    /// exported under the function's own name until renamed.
101    pub fn new(func: &'f Function, callee: impl ToTokens) -> Self {
102        Self {
103            func,
104            name: func.name.clone(),
105            callee: callee.to_token_stream(),
106            attrs: Vec::new(),
107            vis: quote!(pub),
108            abi: Some("C".to_string()),
109            unsafety: true,
110            generics: TokenStream::new(),
111            leading: Vec::new(),
112            trailing: Vec::new(),
113            prologue: TokenStream::new(),
114            call: None,
115        }
116    }
117
118    /// The exported name.
119    pub fn name(mut self, name: syn::Ident) -> Self {
120        self.name = name;
121        self
122    }
123
124    /// An attribute on the wrapper (`#[no_mangle]`, `#[allow(..)]`).
125    pub fn attr(mut self, attr: impl ToTokens) -> Self {
126        self.attrs.push(attr.to_token_stream());
127        self
128    }
129
130    /// The visibility (default `pub`).
131    pub fn vis(mut self, vis: impl ToTokens) -> Self {
132        self.vis = vis.to_token_stream();
133        self
134    }
135
136    /// The ABI (default `"C"`); `None` writes a plain Rust function.
137    pub fn abi(mut self, abi: Option<&str>) -> Self {
138        self.abi = abi.map(str::to_string);
139        self
140    }
141
142    /// Whether the wrapper is an `unsafe fn` (default yes: it dereferences
143    /// what the foreign side hands it).
144    pub fn unsafety(mut self, unsafety: bool) -> Self {
145        self.unsafety = unsafety;
146        self
147    }
148
149    /// Generic parameters, brackets included: `<'a>`.
150    pub fn generics(mut self, generics: impl ToTokens) -> Self {
151        self.generics = generics.to_token_stream();
152        self
153    }
154
155    /// A parameter before the parameters' wires (`env`, `class`).
156    pub fn leading(mut self, wire: Wire) -> Self {
157        self.leading.push(wire);
158        self
159    }
160
161    /// A parameter after everything else (an error sink).
162    pub fn trailing(mut self, wire: Wire) -> Self {
163        self.trailing.push(wire);
164        self
165    }
166
167    /// Statements at the top of the body, before any conversion.
168    pub fn prologue(mut self, stmts: impl ToTokens) -> Self {
169        self.prologue.extend(stmts.to_token_stream());
170        self
171    }
172
173    /// Replace the call expression. `f` gets the callee and the arguments as
174    /// they would be passed.
175    pub fn call_with(
176        mut self,
177        f: impl Fn(&TokenStream, &[TokenStream]) -> TokenStream + 'f,
178    ) -> Self {
179        self.call = Some(Box::new(f));
180        self
181    }
182
183    /// Write the wrapper.
184    pub fn write<C: FunctionCallbacks>(self, cb: &mut C) -> Result<TokenStream, C::Error> {
185        let func = self.func;
186        // The result first: how a failed input is reported depends on it.
187        let ret = cb.ret(&func.ret)?;
188        let mut inputs = Vec::with_capacity(func.params.len());
189        for p in &func.params {
190            inputs.push(cb.param(&p.name, &p.ty)?);
191        }
192        let fail = cb.fail(&ret);
193
194        let mut params: Vec<TokenStream> = self.leading.iter().map(Wire::decl).collect();
195        let mut stmts = Vec::new();
196        let mut args = Vec::new();
197        let err = error_ident();
198        for (p, input) in func.params.iter().zip(&inputs) {
199            params.extend(input.wires.iter().map(Wire::decl));
200            let name = &p.name;
201            let e = &input.expr;
202            if input.fallible {
203                let r = input.result();
204                stmts.push(quote! {
205                    let #name = match #r {
206                        ::core::result::Result::Ok(__v) => __v,
207                        ::core::result::Result::Err(#err) => { #fail }
208                    };
209                });
210            } else {
211                stmts.push(quote!(let #name = #e;));
212            }
213            args.push(input.pass.clone().unwrap_or_else(|| quote!(#name)));
214        }
215        params.extend(ret.wires.iter().map(Wire::decl));
216        params.extend(self.trailing.iter().map(Wire::decl));
217
218        let callee = &self.callee;
219        let call = match &self.call {
220            Some(f) => f(callee, &args),
221            None => quote!(#callee(#(#args),*)),
222        };
223        let result = result_ident();
224        let body = &ret.body;
225        let ret_ty = ret.ty.as_ref().map(|t| quote!(-> #t));
226        let attrs = &self.attrs;
227        let vis = &self.vis;
228        let unsafety = self.unsafety.then(|| quote!(unsafe));
229        let abi = self.abi.as_ref().map(|a| {
230            let lit = syn::LitStr::new(a, proc_macro2::Span::call_site());
231            quote!(extern #lit)
232        });
233        let name = &self.name;
234        let generics = &self.generics;
235        let prologue = &self.prologue;
236        Ok(quote! {
237            #(#attrs)*
238            #vis #unsafety #abi fn #name #generics(#(#params),*) #ret_ty {
239                #prologue
240                #(#stmts)*
241                let #result = #call;
242                #body
243            }
244        })
245    }
246}
