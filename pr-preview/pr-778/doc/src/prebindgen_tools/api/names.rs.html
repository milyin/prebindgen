4/// An identifier from a name, accepting raw spellings (`r#type`) and
5/// escaping a Rust keyword into its raw form, so a name taken from any
6/// source can always be turned into an ident.
7pub fn ident(name: &str) -> syn::Ident {
8    if let Some(raw) = name.strip_prefix("r#") {
9        return syn::Ident::new_raw(raw, Span::call_site());
10    }
11    syn::parse_str::<syn::Ident>(name)
12        .unwrap_or_else(|_| syn::Ident::new_raw(name, Span::call_site()))
13}

15/// The bare name of an identifier: `r#type` → `type`.
16pub fn bare(id: &syn::Ident) -> String {
17    let s = id.to_string();
18    s.strip_prefix("r#").map(str::to_string).unwrap_or(s)
19}

21/// `base` joined to `suffix` with an underscore: `p` + `id` → `p_id`.
22pub fn join(base: &syn::Ident, suffix: &str) -> syn::Ident {
23    ident(&format!("{}_{suffix}", bare(base)))
24}

26/// `PascalCase` or `camelCase` → `snake_case`.
27pub fn snake(s: &str) -> String {
28    let mut out = String::with_capacity(s.len() + 4);
29    let chars: Vec<char> = s.chars().collect();
30    for (i, &c) in chars.iter().enumerate() {
31        if c.is_uppercase() {
32            let prev_lower =
33                i > 0 && (chars[i - 1].is_lowercase() || chars[i - 1].is_ascii_digit());
34            let next_lower = chars.get(i + 1).is_some_and(|n| n.is_lowercase());
35            let prev_upper = i > 0 && chars[i - 1].is_uppercase();
36            if i > 0 && (prev_lower || (prev_upper && next_lower)) && !out.ends_with('_') {
37                out.push('_');
38            }
39            out.extend(c.to_lowercase());
40        } else {
41            out.push(c);
42        }
43    }
44    out
45}

47/// `snake_case` → `lowerCamelCase`. A name without underscores is returned
48/// with its first letter lowered.
49pub fn camel(s: &str) -> String {
50    let p = pascal(s);
51    let mut chars = p.chars();
52    match chars.next() {
53        Some(first) => first.to_lowercase().chain(chars).collect(),
54        None => String::new(),
55    }
56}

58/// `snake_case` → `PascalCase`. Leading/trailing/double underscores are
59/// dropped; an already-Pascal name is returned unchanged.
60pub fn pascal(s: &str) -> String {
61    let mut out = String::with_capacity(s.len());
62    for part in s.split('_').filter(|p| !p.is_empty()) {
63        let mut chars = part.chars();
64        if let Some(first) = chars.next() {
65            out.extend(first.to_uppercase());
66            out.extend(chars);
67        }
68    }
69    out
70}

72/// A readable, identifier-safe name for a type, for naming generated helpers:
73/// `Option<Vec<Payload>>` → `Option_Vec_Payload`, `&mut Calculator` →
74/// `mut_Calculator`, `&str` → `ref_str`, `[u8; 4]` → `array_u8_4`.
75///
76/// Distinct types give distinct names; lifetimes do not participate, so two
77/// spellings that differ only by a lifetime share a helper.
78pub fn mangle(ty: &TypeRef) -> String {
79    let mut out = String::new();
80    mangle_into(ty, &mut out);
81    out
82}
83
84fn mangle_into(ty: &TypeRef, out: &mut String) {
85    let push = |out: &mut String, s: &str| {
86        if !out.is_empty() && !out.ends_with('_') {
87            out.push('_');
88        }
89        out.push_str(s);
90    };
91    match ty.kind() {
92        TypeKind::Scalar(k) => push(out, k.as_str()),
93        TypeKind::Str => push(out, "str"),
94        TypeKind::String => push(out, "String"),
95        TypeKind::Unit => push(out, "unit"),
96        TypeKind::Optional(t) => {
97            push(out, "Option");
98            mangle_into(t, out)
99        }
100        TypeKind::Vec(t) => {
101            push(out, "Vec");
102            mangle_into(t, out)
103        }
104        TypeKind::Slice(t) => {
105            push(out, "slice");
106            mangle_into(t, out)
107        }
108        TypeKind::Boxed(t) => {
109            push(out, "Box");
110            mangle_into(t, out)
111        }
112        TypeKind::Cow { inner, .. } => {
113            push(out, "Cow");
114            mangle_into(inner, out)
115        }
116        TypeKind::Uninit(t) => {
117            push(out, "Uninit");
118            mangle_into(t, out)
119        }
120        TypeKind::Fallible { ok, err } => {
121            push(out, "Result");
122            mangle_into(ok, out);
123            mangle_into(err, out)
124        }
125        TypeKind::Ref { mutable, inner, .. } => {
126            push(out, if *mutable { "mut" } else { "ref" });
127            mangle_into(inner, out)
128        }
129        TypeKind::Array { elem, extent } => {
130            push(out, "array");
131            mangle_into(elem, out);
132            push(out, &extent.value.to_string())
133        }
134        TypeKind::Named { id, args } => {
135            push(out, &id.name.replace("::", "_"));
136            for a in args {
137                if let GenericArg::Type(t) = a {
138                    mangle_into(t, out)
139                }
140            }
141        }
142        TypeKind::Callback { args } => {
143            push(out, "Fn");
144            for a in args {
145                mangle_into(a, out)
146            }
147        }
148    }
149}
