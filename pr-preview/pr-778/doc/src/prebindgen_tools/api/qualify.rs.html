10/// Spells source items and types as Rust paths usable from a binding crate.
11///
12/// The qualifier borrows a [`Flat`] model and reads the declaring crate from
13/// each item's source location. [`path`](Self::path) qualifies one item;
14/// [`ty`](Self::ty) recursively qualifies a type expression. Both return
15/// tokens for generated Rust without changing the model or choosing an ABI.
16///
17/// See the [qualification module](index.html) for path selection rules,
18/// including nested containers and binding-local names.
19///
20/// # Example: call a source function from generated Rust
21///
22/// Build a model with the source crate recorded on its items, then qualify
23/// both the function to call and its return type. The resulting tokens can
24/// be placed in a generated wrapper:
25///
26/// ```
27/// use prebindgen::SourceLocation;
28/// use prebindgen_flat::Flat;
29/// use prebindgen_tools::Qualifier;
30/// use quote::quote;
31///
32/// let source = syn::parse_file(
33///     "pub struct Payload; pub fn make() -> Payload { Payload }",
34/// ).unwrap();
35/// let location = SourceLocation {
36///     crate_name: Some("source-crate".into()),
37///     ..Default::default()
38/// };
39/// let flat = Flat::builder()
40///     .items(source.items.into_iter().map(|item| (item, location.clone())))
41///     .build().unwrap();
42/// let qualifier = Qualifier::new(&flat);
43/// let function = flat.function("make").unwrap();
44/// let callee = qualifier.path(&function.name);
45/// let return_type = qualifier.ty(&function.ret);
46/// let wrapper = quote! {
47///     pub fn make_wrapper() -> #return_type { #callee() }
48/// };
49/// assert_eq!(wrapper.to_string(), quote! {
50///     pub fn make_wrapper() -> source_crate::Payload { source_crate::make() }
51/// }.to_string());
52/// ```
53#[derive(Clone)]
54pub struct Qualifier<'a> {
55    flat: &'a Flat,
56    default_module: Option<syn::Path>,
57    crate_paths: HashMap<String, syn::Path>,
58}
59
60impl<'a> Qualifier<'a> {
61    /// A qualifier over `flat`. Items are qualified with the crate that
62    /// captured them, as stamped by [`prebindgen::Source`].
63    pub fn new(flat: &'a Flat) -> Self {
64        Self {
65            flat,
66            default_module: None,
67            crate_paths: HashMap::new(),
68        }
69    }
70
71    /// The module used for an item whose capture carries no crate name — a
72    /// synthetic stream fed to [`Flat::builder`] without a stamp.
73    ///
74    /// A recorded crate name or its configured override takes precedence.
75    /// Unknown names never use this fallback. Pass `None` to remove a previously configured fallback.
76    pub fn with_default_module(mut self, module: Option<syn::Path>) -> Self {
77        self.default_module = module;
78        self
79    }
80
81    /// Name items captured from `crate_name` through `path` in generated Rust.
82    ///
83    /// Use for a renamed dependency or a module re-exporting a source API.
84    /// This overrides the recorded crate's path without changing source
85    /// locations. Hyphens and underscores in `crate_name` are equivalent;
86    /// a later override for the same crate replaces the earlier one.
87    /// Items from other crates and unknown names are unaffected.
88    pub fn with_crate_path(mut self, crate_name: impl Into<String>, path: syn::Path) -> Self {
89        self.crate_paths
90            .insert(crate_name.into().replace('-', "_"), path);
91        self
92    }
93
94    /// The source prefix for a known item: a configured override, its recorded
95    /// crate name (with hyphens replaced by underscores), or the fallback if no
96    /// crate name was recorded.
97    ///
98    /// Returns `None` for an unknown name, a known item with no prefix, or
99    /// a recorded crate name that cannot be parsed as a Rust path.
100    fn module_of(&self, name: &str) -> Option<syn::Path> {
101        let element = self.flat.element(name)?;
102        match &element.location().crate_name {
103            Some(c) => {
104                let name = c.replace('-', "_");
105                self.crate_paths
106                    .get(&name)
107                    .cloned()
108                    .or_else(|| syn::parse_str(&name).ok())
109            }
110            None => self.default_module.clone(),
111        }
112    }
113
114    /// The path generated code calls or names the flat item `name` by:
115    /// `perftest_flat::storage_new`. A name the model does not know is
116    /// returned unchanged.
117    pub fn path(&self, name: &syn::Ident) -> TokenStream {
118        match self.module_of(&crate::names::bare(name)) {
119            Some(m) => quote!(#m::#name),
120            None => quote!(#name),
121        }
122    }
123
124    /// Spell a model type with named items qualified by their source crate.
125    ///
126    /// Recurses into containers, borrows, generic type arguments, and callback
127    /// arguments. Named array lengths are qualified too. Standard types use
128    /// absolute `::core` or `::std` paths, and source lifetimes are retained.
129    /// The result still describes the source type, before ABI conversion.
130    pub fn ty(&self, ty: &TypeRef) -> TokenStream {
131        self.render_type(ty, false)
132    }
133
134    /// Qualify a type while eliding every explicit type lifetime recursively.
135    ///
136    /// Use for a `let` annotation or closure argument that cannot name the
137    /// source's lifetime parameters. References lose their explicit lifetime
138    /// (`&'a T` → `&T`); `Cow` and named generic lifetime arguments use `'_`.
139    /// This applies inside every supported shape, including arrays, `Result`,
140    /// generic type arguments and callback arguments, and includes `'static`.
141    /// A callback's required `+ 'static` bound is retained.
142    /// Use [`ty`](Self::ty) when explicit lifetime constraints must be retained.
143    pub fn ty_elided(&self, ty: &TypeRef) -> TokenStream {
144        self.render_type(ty, true)
145    }
146
147    fn render_type(&self, ty: &TypeRef, elide: bool) -> TokenStream {
148        let child = |ty: &TypeRef| self.render_type(ty, elide);
149        let render_lifetime = |l: &syn::Lifetime| {
150            if elide {
151                quote!('_)
152            } else {
153                quote!(#l)
154            }
155        };
156        match ty.kind() {
157            TypeKind::Scalar(k) => {
158                let id = crate::names::ident(k.as_str());
159                quote!(#id)
160            }
161            TypeKind::Str => quote!(str),
162            TypeKind::String => quote!(::std::string::String),
163            TypeKind::Unit => quote!(()),
164            TypeKind::Optional(t) => {
165                let t = child(t);
166                quote!(::core::option::Option<#t>)
167            }
168            TypeKind::Vec(t) => {
169                let t = child(t);
170                quote!(::std::vec::Vec<#t>)
171            }
172            TypeKind::Slice(t) => {
173                let t = child(t);
174                quote!([#t])
175            }
176            TypeKind::Boxed(t) => {
177                let t = child(t);
178                quote!(::std::boxed::Box<#t>)
179            }
180            TypeKind::Cow { lifetime, inner } => {
181                let t = child(inner);
182                let lifetime = render_lifetime(lifetime);
183                quote!(::std::borrow::Cow<#lifetime, #t>)
184            }
185            TypeKind::Uninit(t) => {
186                let t = child(t);
187                quote!(::core::mem::MaybeUninit<#t>)
188            }
189            TypeKind::Fallible { ok, err } => {
190                let (ok, err) = (child(ok), child(err));
191                quote!(::core::result::Result<#ok, #err>)
192            }
193            TypeKind::Ref {
194                lifetime,
195                mutable,
196                inner,
197            } => {
198                let t = child(inner);
199                let m = mutable.then(|| quote!(mut));
200                let lifetime = lifetime.as_ref().filter(|_| !elide);
201                quote!(& #lifetime #m #t)
202            }
203            TypeKind::Array { elem, extent } => {
204                let e = child(elem);
205                let len = match &extent.source {
206                    ExtentSource::Literal => {
207                        let n = proc_macro2::Literal::usize_unsuffixed(extent.value);
208                        quote!(#n)
209                    }
210                    ExtentSource::Const(id) => {
211                        let name = crate::names::ident(&id.name);
212                        let path = self.path(&name);
213                        quote!(#path)
214                    }
215                };
216                quote!([#e; #len])
217            }
218            TypeKind::Named { id, args } => {
219                let head: TokenStream = match id.ident() {
220                    Some(name) => self.path(&name),
221                    None => id.name.parse().expect("a type path the model built"),
222                };
223                if args.is_empty() {
224                    head
225                } else {
226                    let args = args.iter().map(|a| match a {
227                        GenericArg::Lifetime(l) => render_lifetime(l),
228                        GenericArg::Type(t) => child(t),
229                    });
230                    quote!(#head<#(#args),*>)
231                }
232            }
233            TypeKind::Callback { args } => {
234                let args = args.iter().map(child);
235                quote!(impl Fn(#(#args),*) + Send + Sync + 'static)
236            }
237        }
238    }
