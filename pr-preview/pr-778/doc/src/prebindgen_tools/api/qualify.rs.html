1use prebindgen_flat::{
2    flat::{ExtentSource, GenericArg, TypeKind, TypeRef},
3    Flat,
4};
5use proc_macro2::TokenStream;
6use quote::quote;
7
8/// Rewrites flat-namespace spellings into paths the generated crate can name.
9#[derive(Clone)]
10pub struct Qualifier<'a> {
11    flat: &'a Flat,
12    default_module: Option<syn::Path>,
13}
14
15impl<'a> Qualifier<'a> {
16    /// A qualifier over `flat`. Items are qualified with the crate that
17    /// captured them, as stamped by [`prebindgen::Source`].
18    pub fn new(flat: &'a Flat) -> Self {
19        Self {
20            flat,
21            default_module: None,
22        }
23    }
24
25    /// The module used for an item whose capture carries no crate name — a
26    /// synthetic stream fed to [`Flat::builder`] without a stamp.
27    pub fn with_default_module(mut self, module: Option<syn::Path>) -> Self {
28        self.default_module = module;
29        self
30    }
31
32    /// The flat model this qualifier reads.
33    pub fn flat(&self) -> &'a Flat {
34        self.flat
35    }
36
37    /// The module that declares `name`, if `name` is a flat item.
38    pub fn module_of(&self, name: &str) -> Option<syn::Path> {
39        let element = self.flat.element(name)?;
40        match &element.location().crate_name {
41            Some(c) => syn::parse_str(&c.replace('-', "_")).ok(),
42            None => self.default_module.clone(),
43        }
44    }
45
46    /// The path generated code calls or names the flat item `name` by:
47    /// `perftest_flat::storage_new`. A name the model does not know is
48    /// returned unchanged.
49    pub fn path(&self, name: &syn::Ident) -> TokenStream {
50        match self.module_of(&crate::names::bare(name)) {
51            Some(m) => quote!(#m::#name),
52            None => quote!(#name),
53        }
54    }
55
56    /// `ty` spelled with every flat item qualified by its declaring crate.
57    pub fn ty(&self, ty: &TypeRef) -> TokenStream {
58        match ty.kind() {
59            TypeKind::Scalar(k) => {
60                let id = crate::names::ident(k.as_str());
61                quote!(#id)
62            }
63            TypeKind::Str => quote!(str),
64            TypeKind::String => quote!(::std::string::String),
65            TypeKind::Unit => quote!(()),
66            TypeKind::Optional(t) => {
67                let t = self.ty(t);
68                quote!(::core::option::Option<#t>)
69            }
70            TypeKind::Vec(t) => {
71                let t = self.ty(t);
72                quote!(::std::vec::Vec<#t>)
73            }
74            TypeKind::Slice(t) => {
75                let t = self.ty(t);
76                quote!([#t])
77            }
78            TypeKind::Boxed(t) => {
79                let t = self.ty(t);
80                quote!(::std::boxed::Box<#t>)
81            }
82            TypeKind::Cow { lifetime, inner } => {
83                let t = self.ty(inner);
84                quote!(::std::borrow::Cow<#lifetime, #t>)
85            }
86            TypeKind::Uninit(t) => {
87                let t = self.ty(t);
88                quote!(::core::mem::MaybeUninit<#t>)
89            }
90            TypeKind::Fallible { ok, err } => {
91                let (ok, err) = (self.ty(ok), self.ty(err));
92                quote!(::core::result::Result<#ok, #err>)
93            }
94            TypeKind::Ref {
95                lifetime,
96                mutable,
97                inner,
98            } => {
99                let t = self.ty(inner);
100                let m = mutable.then(|| quote!(mut));
101                quote!(& #lifetime #m #t)
102            }
103            TypeKind::Array { elem, extent } => {
104                let e = self.ty(elem);
105                let len = match &extent.source {
106                    ExtentSource::Literal => {
107                        let n = proc_macro2::Literal::usize_unsuffixed(extent.value);
108                        quote!(#n)
109                    }
110                    ExtentSource::Const(id) => {
111                        let name = crate::names::ident(&id.name);
112                        let path = self.path(&name);
113                        quote!(#path)
114                    }
115                };
116                quote!([#e; #len])
117            }
118            TypeKind::Named { id, args } => {
119                let head: TokenStream = match id.ident() {
120                    Some(name) => self.path(&name),
121                    None => id.name.parse().expect("a type path the model built"),
122                };
123                if args.is_empty() {
124                    head
125                } else {
126                    let args = args.iter().map(|a| match a {
127                        GenericArg::Lifetime(l) => quote!(#l),
128                        GenericArg::Type(t) => self.ty(t),
129                    });
130                    quote!(#head<#(#args),*>)
131                }
132            }
133            TypeKind::Callback { args } => {
134                let args = args.iter().map(|a| self.ty(a));
135                quote!(impl Fn(#(#args),*) + Send + Sync + 'static)
136            }
137        }
138    }
139
140    /// `ty` with its lifetimes erased (`&'a T` → `&T`), for a position that
141    /// cannot name the source's lifetime parameters: a `let` annotation or a
142    /// closure argument in generated code.
143    pub fn ty_elided(&self, ty: &TypeRef) -> TokenStream {
144        match ty.kind() {
145            TypeKind::Ref { mutable, inner, .. } => {
146                let t = self.ty_elided(inner);
147                let m = mutable.then(|| quote!(mut));
148                quote!(& #m #t)
149            }
150            TypeKind::Cow { inner, .. } => {
151                let t = self.ty_elided(inner);
152                quote!(::std::borrow::Cow<'_, #t>)
153            }
154            TypeKind::Optional(t) => {
155                let t = self.ty_elided(t);
156                quote!(::core::option::Option<#t>)
157            }
158            TypeKind::Vec(t) => {
159                let t = self.ty_elided(t);
160                quote!(::std::vec::Vec<#t>)
161            }
162            TypeKind::Boxed(t) => {
163                let t = self.ty_elided(t);
164                quote!(::std::boxed::Box<#t>)
165            }
166            TypeKind::Slice(t) => {
167                let t = self.ty_elided(t);
168                quote!([#t])
169            }
170            _ => self.ty(ty),
171        }
172    }
173}
