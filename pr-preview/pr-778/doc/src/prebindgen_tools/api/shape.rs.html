3/// How a value is held where it appears.
4#[derive(Clone, Copy, Debug, PartialEq, Eq)]
5pub enum Access {
6    /// By value. This describes access, not whether the type is sized or
7    /// supported at a binding boundary.
8    Owned,
9    /// `&T`.
10    Shared,
11    /// `&mut T`.
12    Exclusive,
13}
14
15impl Access {
16    /// Whether `ty` is owned, shared, or exclusively borrowed at its outer layer.
17    pub fn of(ty: &TypeRef) -> Self {
18        match ty.kind() {
19            TypeKind::Ref { mutable: true, .. } => Self::Exclusive,
20            TypeKind::Ref { .. } => Self::Shared,
21            _ => Self::Owned,
22        }
23    }
24}

26/// The source text container, independently of how it is accessed.
27///
28/// [`Shape::Str`] pairs this kind with [`Access`]. For example, `&String`
29/// has [`TextKind::String`] + [`Access::Shared`], while `&str` has
30/// [`TextKind::Str`] + [`Access::Shared`]. A by-value `String` keeps
31/// [`TextKind::String`] and uses [`Access::Owned`]; `&mut String` uses
32/// [`Access::Exclusive`]. Borrowing changes access, not the container kind.
33#[derive(Clone, Copy, Debug, PartialEq, Eq)]
34pub enum TextKind {
35    /// The growable, owned string container.
36    String,
37    /// The unsized string slice.
38    Str,
39}
40
41/// The source sequence container, independently of how it is accessed.
42///
43/// [`Shape::Seq`] pairs this kind with [`Access`] and an element type. For
44/// example, `&Vec<T>` has [`SequenceKind::Vec`] + [`Access::Shared`], while
45/// `&[T]` has [`SequenceKind::Slice`] + [`Access::Shared`]. A by-value
46/// `Vec<T>` keeps [`SequenceKind::Vec`] and uses [`Access::Owned`];
47/// `&mut Vec<T>` uses [`Access::Exclusive`]. Borrowing changes access, not
48/// the container kind.
49#[derive(Clone, Copy, Debug, PartialEq, Eq)]
50pub enum SequenceKind {
51    /// The growable vector container.
52    Vec,
53    /// The unsized slice.
54    Slice,
55}

57/// The outer layer of a model type, with any declaration the adapter made for it.
58///
59/// Returned by [`shape()`]. Child types borrow the input model; they have not
60/// been classified recursively. The adapter matches this enum to choose a
61/// representation and calls `shape()` on whichever children it needs.
62/// `D` is the adapter's declaration type, often borrowed from its registry.
63#[derive(Debug)]
64pub enum Shape<'t, D> {
65    /// `()`.
66    Unit,
67    /// A primitive.
68    Scalar(ScalarKind),
69    /// A string container or string slice, by value or borrowed.
70    Str {
71        /// Whether the source names `String` or `str`.
72        kind: TextKind,
73        /// By value, shared borrow, or exclusive borrow.
74        access: Access,
75    },
76    /// A vector or slice, by value or borrowed.
77    Seq {
78        /// The sequence element to lower recursively.
79        elem: &'t TypeRef,
80        /// Whether the source names `Vec<T>` or `[T]`.
81        kind: SequenceKind,
82        /// By value, shared borrow, or exclusive borrow.
83        access: Access,
84    },
85    /// `[T; N]`.
86    Array {
87        /// The array element to lower recursively.
88        elem: &'t TypeRef,
89        /// The fixed number of elements.
90        len: usize,
91    },
92    /// `Option<T>`.
93    Option(&'t TypeRef),
94    /// `Box<T>`.
95    Boxed(&'t TypeRef),
96    /// `Cow<T>`, including `Cow<str>` and `Cow<[T]>`. The child is `T`.
97    Cow(&'t TypeRef),
98    /// A borrow of anything [`Shape::Str`], [`Shape::Seq`] and
99    /// [`Shape::Declared`] do not cover.
100    Ref {
101        /// The referenced type to lower recursively.
102        inner: &'t TypeRef,
103        /// Shared or exclusive access.
104        access: Access,
105    },
106    /// A type the adapter declared, with its original ownership or borrow.
107    Declared {
108        /// The original type, including generic arguments and any borrow.
109        ty: &'t TypeRef,
110        /// The adapter's declaration returned by `shape_declared`.
111        declaration: D,
112    },
113    /// A named type for which the adapter returned no declaration.
114    Undeclared(&'t str),
115    /// `impl Fn(A, B)`.
116    Callback(&'t [TypeRef]),
117    /// `Result<T, E>`.
118    Result {
119        /// Successful value to lower recursively.
120        ok: &'t TypeRef,
121        /// Error value to lower recursively.
122        err: &'t TypeRef,
123    },
124    /// `&mut MaybeUninit<T>`: a slot the callee fills.
125    Out(&'t TypeRef),
126}

128/// Describe one outer layer of a source type for a language adapter.
129///
130/// Pass the [`TypeRef`] being converted and `shape_declared`, a callback that
131/// returns the adapter's declaration for a type or `None`. The callback sees
132/// the complete type, so an adapter can distinguish `Vec<u8>` from
133/// `Vec<Payload>`. A returned declaration makes the result
134/// [`Shape::Declared`], which retains the original `ty`.
135///
136/// `shape` asks about the exact `ty` first. For a borrow with no exact
137/// declaration, it also asks about the borrowed inner type, so a declaration
138/// for `Payload` still covers `&Payload`. An exact declaration for
139/// `&Payload` takes precedence. Bare `MaybeUninit<T>` and the
140/// `&mut MaybeUninit<T>` output slot are handled before the callback.
141///
142/// # Walking into child types
143///
144/// An adapter can get `ty` from a function parameter in the flat model, then
145/// call `shape(ty, shape_declared)` and match the returned [`Shape`]. Some
146/// variants hold a child [`TypeRef`]; call `shape` again with that child and
147/// the same closure. This example shows each call for `Option<Vec<Payload>>`:
148///
149/// ```
150/// use std::collections::HashMap;
151/// use prebindgen::SourceLocation;
152/// use prebindgen_flat::Flat;
153/// use prebindgen_tools::{shape, Access, SequenceKind, Shape};
154///
155/// let source = syn::parse_file(
156///     "pub struct Payload; pub fn send(value: Option<Vec<Payload>>, bytes: Vec<u8>) {}"
157/// ).unwrap();
158/// let flat = Flat::builder()
159///     .items(source.items.into_iter().map(|item| (item, SourceLocation::default())))
160///     .build().unwrap();
161/// let ty = &flat.function("send").unwrap().params[0].ty;
162/// let declarations = HashMap::from([("Payload".to_owned(), "opaque handle")]);
163/// let shape_declared = |candidate: &prebindgen_flat::flat::TypeRef| {
164///     match candidate.kind() {
165///         prebindgen_flat::flat::TypeKind::Named { id, .. } =>
166///             declarations.get(&id.name).copied(),
167///         _ => None,
168///     }
169/// };
170///
171/// let Shape::Option(inner) = shape(ty, shape_declared).unwrap() else { panic!() };
172/// let Shape::Seq { elem, kind: SequenceKind::Vec, access: Access::Owned } = shape(inner, shape_declared).unwrap()
173///     else { panic!() };
174/// assert!(matches!(
175///     shape(elem, shape_declared).unwrap(),
176///     Shape::Declared {
177///         declaration: "opaque handle", ty: declared_ty
178///     } if std::ptr::eq(declared_ty, elem)
179/// ));
180///
181/// // A declaration may target a complete generic type, too.
182/// let bytes = &flat.function("send").unwrap().params[1].ty;
183/// let bytes_key = bytes.key();
184/// assert!(matches!(
185///     shape(bytes, |candidate| (candidate.key() == bytes_key).then_some("blob"))
186///         .unwrap(),
187///     Shape::Declared { ty, declaration: "blob" } if std::ptr::eq(ty, bytes)
188/// ));
189/// ```
190///
191/// The first two calls return built-in shapes because the callback returns
192/// `None` for `Option<Vec<Payload>>` and `Vec<Payload>`. The third returns
193/// [`Shape::Declared`] for `Payload`. The adapter decides how to represent
194/// each layer on its boundary and which children to visit.
195/// A successful call describes only the current layer; it does not establish
196/// that the adapter can convert the complete type.
197///
198/// # Container kind and access
199///
200/// Without an adapter declaration, container kinds and access remain explicit:
201///
202/// | Source type | Shape |
203/// |---|---|
204/// | `String` | `Str { kind: String, access: Owned }` |
205/// | `&String` | `Str { kind: String, access: Shared }` |
206/// | `&str` | `Str { kind: Str, access: Shared }` |
207/// | `Vec<T>` | `Seq { elem: T, kind: Vec, access: Owned }` |
208/// | `&Vec<T>` | `Seq { elem: T, kind: Vec, access: Shared }` |
209/// | `&[T]` | `Seq { elem: T, kind: Slice, access: Shared }` |
210/// | `&mut [T]` | `Seq { elem: T, kind: Slice, access: Exclusive }` |
211/// | `Cow<str>` / `Cow<[T]>` | `Cow(str)` / `Cow([T])` |
212/// | `[T; N]` | `Array { elem: T, len: N }`, using the resolved length |
213/// | `&mut MaybeUninit<T>` | `Out(T)`, a slot the callee fills |
214///
215/// Mutable borrows of `String`, `str`, `Vec<T>`, and `[T]` have the same
216/// container kind as shared borrows, with [`Access::Exclusive`]. `Cow`
217/// remains a wrapper: its runtime ownership is not an access mode.
218/// `Option`, `Box`, `Cow`, `Result`, and callbacks expose their children
219/// without visiting them. Lifetimes remain on the original [`TypeRef`].
220///
221/// Bare `str` and `[T]` classify as unsized containers with [`Access::Owned`].
222/// This lets an adapter inspect a `Cow` child without inventing another type.
223/// It does not make those types valid by-value parameters: the adapter must
224/// reject unsupported uses, including mutable containers it cannot implement.
225///
226/// # Errors
227///
228/// Bare `MaybeUninit<T>` is refused; only `&mut MaybeUninit<T>` is recognized
229/// as an output slot. A different borrow initially returns `Ref`; visiting
230/// its child then encounters the bare-`MaybeUninit` error.
231///
232/// See the [`shape` module](mod@crate::shape) for a runnable recursive example.
233pub fn shape<D>(
234    ty: &TypeRef,
235    shape_declared: impl Fn(&TypeRef) -> Option<D>,
236) -> Result<Shape<'_, D>, String> {
237    match ty.kind() {
238        TypeKind::Uninit(_) => {
239            return Err("`MaybeUninit<T>` crosses only as `&mut MaybeUninit<T>`".into())
240        }
241        TypeKind::Ref {
242            mutable: true,
243            inner,
244            ..
245        } => {
246            if let TypeKind::Uninit(t) = inner.kind() {
247                return Ok(Shape::Out(t));
248            }
249        }
250        _ => {}
251    }
252    if let Some(declaration) = shape_declared(ty) {
253        return Ok(Shape::Declared { ty, declaration });
254    }
255    if let TypeKind::Ref { inner, .. } = ty.kind() {
256        if !matches!(inner.kind(), TypeKind::Uninit(_)) {
257            if let Some(declaration) = shape_declared(inner) {
258                return Ok(Shape::Declared { ty, declaration });
259            }
260        }
261    }
262    Ok(match ty.kind() {
263        TypeKind::Unit => Shape::Unit,
264        TypeKind::Scalar(k) => Shape::Scalar(*k),
265        TypeKind::String => Shape::Str {
266            kind: TextKind::String,
267            access: Access::Owned,
268        },
269        TypeKind::Str => Shape::Str {
270            kind: TextKind::Str,
271            access: Access::Owned,
272        },
273        TypeKind::Slice(elem) => Shape::Seq {
274            elem,
275            kind: SequenceKind::Slice,
276            access: Access::Owned,
277        },
278        TypeKind::Optional(t) => Shape::Option(t),
279        TypeKind::Vec(t) => Shape::Seq {
280            elem: t,
281            kind: SequenceKind::Vec,
282            access: Access::Owned,
283        },
284        TypeKind::Array { elem, extent } => Shape::Array {
285            elem,
286            len: extent.value,
287        },
288        TypeKind::Boxed(t) => Shape::Boxed(t),
289        TypeKind::Cow { inner, .. } => Shape::Cow(inner),
290        TypeKind::Ref { mutable, inner, .. } => {
291            let access = Access::of(ty);
292            match inner.kind() {
293                TypeKind::Str => Shape::Str {
294                    kind: TextKind::Str,
295                    access,
296                },
297                TypeKind::Slice(elem) => Shape::Seq {
298                    elem,
299                    kind: SequenceKind::Slice,
300                    access,
301                },
302                TypeKind::Vec(elem) => Shape::Seq {
303                    elem,
304                    kind: SequenceKind::Vec,
305                    access,
306                },
307                TypeKind::Uninit(t) if *mutable => Shape::Out(t),
308                TypeKind::Named { id, .. } => Shape::Undeclared(&id.name),
309                TypeKind::String => Shape::Str {
310                    kind: TextKind::String,
311                    access,
312                },
313                _ => Shape::Ref { inner, access },
314            }
315        }
316        TypeKind::Named { id, .. } => Shape::Undeclared(&id.name),
317        TypeKind::Callback { args } => Shape::Callback(args),
318        TypeKind::Fallible { ok, err } => Shape::Result { ok, err },
319        TypeKind::Uninit(_) => {
320            return Err("`MaybeUninit<T>` crosses only as `&mut MaybeUninit<T>`".into())
321        }
322    })
323}
