3/// How a value is held where it appears.
4#[derive(Clone, Copy, Debug, PartialEq, Eq)]
5pub enum Access {
6    /// By value: the callee owns it.
7    Owned,
8    /// `&T`.
9    Shared,
10    /// `&mut T`.
11    Exclusive,
12}
13
14impl Access {
15    /// Whether `ty` is owned, shared, or exclusively borrowed at its outer layer.
16    pub fn of(ty: &TypeRef) -> Self {
17        match ty.kind() {
18            TypeKind::Ref { mutable: true, .. } => Self::Exclusive,
19            TypeKind::Ref { .. } => Self::Shared,
20            _ => Self::Owned,
21        }
22    }
23}

25/// How a run of values (text or a sequence) is held.
26#[derive(Clone, Copy, Debug, PartialEq, Eq)]
27pub enum Holding {
28    /// `String`, `Vec<T>`.
29    Owned,
30    /// `&str`, `&[T]`.
31    Borrowed,
32    /// `&Vec<T>`.
33    BorrowedVec,
34    /// `Cow<str>`, `Cow<[T]>`.
35    Cow,
36}

38/// The outer layer of a model type, with any declaration the adapter made for it.
39///
40/// Returned by [`shape()`]. Child types borrow the input model; they have not
41/// been classified recursively. The adapter matches this enum to choose a
42/// representation and calls `shape()` on whichever children it needs.
43/// `D` is the adapter's declaration type, often borrowed from its registry.
44#[derive(Debug)]
45pub enum Shape<'t, D> {
46    /// `()`.
47    Unit,
48    /// A primitive.
49    Scalar(ScalarKind),
50    /// `String`, `&str`, `Cow<str>`.
51    Str(Holding),
52    /// `Vec<T>`, `&[T]`, `&Vec<T>`, `Cow<[T]>`.
53    Seq {
54        /// The sequence element to lower recursively.
55        elem: &'t TypeRef,
56        /// Whether the sequence is owned, borrowed, or held in a `Cow`.
57        holding: Holding,
58    },
59    /// `[T; N]`.
60    Array {
61        /// The array element to lower recursively.
62        elem: &'t TypeRef,
63        /// The fixed number of elements.
64        len: usize,
65    },
66    /// `Option<T>`.
67    Option(&'t TypeRef),
68    /// `Box<T>`.
69    Boxed(&'t TypeRef),
70    /// `Cow<T>` over anything but `str` or a slice.
71    Cow(&'t TypeRef),
72    /// A borrow of anything [`Shape::Str`], [`Shape::Seq`] and
73    /// [`Shape::Declared`] do not cover.
74    Ref {
75        /// The referenced type to lower recursively.
76        inner: &'t TypeRef,
77        /// Shared or exclusive access.
78        access: Access,
79    },
80    /// A type the adapter declared, with its original ownership or borrow.
81    Declared {
82        /// The original type, including generic arguments and any borrow.
83        ty: &'t TypeRef,
84        /// The adapter's declaration returned by `shape_declared`.
85        declaration: D,
86    },
87    /// A named type for which the adapter returned no declaration.
88    Undeclared(&'t str),
89    /// `impl Fn(A, B)`.
90    Callback(&'t [TypeRef]),
91    /// `Result<T, E>`.
92    Result {
93        /// Successful value to lower recursively.
94        ok: &'t TypeRef,
95        /// Error value to lower recursively.
96        err: &'t TypeRef,
97    },
98    /// `&mut MaybeUninit<T>`: a slot the callee fills.
99    Out(&'t TypeRef),
100}

102/// Describe one outer layer of a source type for a language adapter.
103///
104/// Pass the [`TypeRef`] being converted and `shape_declared`, a callback that
105/// returns the adapter's declaration for a type or `None`. The callback sees
106/// the complete type, so an adapter can distinguish `Vec<u8>` from
107/// `Vec<Payload>`. A returned declaration makes the result
108/// [`Shape::Declared`], which retains the original `ty`.
109///
110/// `shape` asks about the exact `ty` first. For a borrow with no exact
111/// declaration, it also asks about the borrowed inner type, so a declaration
112/// for `Payload` still covers `&Payload`. An exact declaration for
113/// `&Payload` takes precedence. Unsupported bare `str`, slices, and
114/// `MaybeUninit<T>`, and the `&mut MaybeUninit<T>` output slot, are handled
115/// before the callback.
116///
117/// # Walking into child types
118///
119/// An adapter can get `ty` from a function parameter in the flat model, then
120/// call `shape(ty, shape_declared)` and match the returned [`Shape`]. Some
121/// variants hold a child [`TypeRef`]; call `shape` again with that child and
122/// the same closure. This example shows each call for `Option<Vec<Payload>>`:
123///
124/// ```
125/// use std::collections::HashMap;
126/// use prebindgen::SourceLocation;
127/// use prebindgen_flat::Flat;
128/// use prebindgen_tools::{shape, Holding, Shape};
129///
130/// let source = syn::parse_file(
131///     "pub struct Payload; pub fn send(value: Option<Vec<Payload>>, bytes: Vec<u8>) {}"
132/// ).unwrap();
133/// let flat = Flat::builder()
134///     .items(source.items.into_iter().map(|item| (item, SourceLocation::default())))
135///     .build().unwrap();
136/// let ty = &flat.function("send").unwrap().params[0].ty;
137/// let declarations = HashMap::from([("Payload".to_owned(), "opaque handle")]);
138/// let shape_declared = |candidate: &prebindgen_flat::flat::TypeRef| {
139///     match candidate.kind() {
140///         prebindgen_flat::flat::TypeKind::Named { id, .. } =>
141///             declarations.get(&id.name).copied(),
142///         _ => None,
143///     }
144/// };
145///
146/// let Shape::Option(inner) = shape(ty, shape_declared).unwrap() else { panic!() };
147/// let Shape::Seq { elem, holding: Holding::Owned } = shape(inner, shape_declared).unwrap()
148///     else { panic!() };
149/// assert!(matches!(
150///     shape(elem, shape_declared).unwrap(),
151///     Shape::Declared {
152///         declaration: "opaque handle", ty: declared_ty
153///     } if std::ptr::eq(declared_ty, elem)
154/// ));
155///
156/// // A declaration may target a complete generic type, too.
157/// let bytes = &flat.function("send").unwrap().params[1].ty;
158/// let bytes_key = bytes.key();
159/// assert!(matches!(
160///     shape(bytes, |candidate| (candidate.key() == bytes_key).then_some("blob"))
161///         .unwrap(),
162///     Shape::Declared { ty, declaration: "blob" } if std::ptr::eq(ty, bytes)
163/// ));
164/// ```
165///
166/// The first two calls return built-in shapes because the callback returns
167/// `None` for `Option<Vec<Payload>>` and `Vec<Payload>`. The third returns
168/// [`Shape::Declared`] for `Payload`. The adapter decides how to represent
169/// each layer on its boundary and which children to visit.
170/// A successful call describes only the current layer; it does not establish
171/// that the adapter can convert the complete type.
172///
173/// # Fallbacks
174///
175/// Without an adapter declaration, a named `T`, `&T`, or `&mut T` returns
176/// [`Shape::Undeclared`] with its name. This means the adapter has no
177/// declaration for the source type; the adapter decides whether to reject it.
178/// Without a declaration, `String` is [`Shape::Str`] with [`Holding::Owned`],
179/// whereas `&String` and `&mut String` are [`Shape::Ref`].
180///
181/// # Borrows and containers
182///
183/// Several common combinations are returned as a single shape:
184///
185/// | Source type | Shape |
186/// |---|---|
187/// | `&str` | `Str(Borrowed)` |
188/// | `Vec<T>` | `Seq { elem: T, holding: Owned }` |
189/// | `&[T]` | `Seq { elem: T, holding: Borrowed }` |
190/// | `&Vec<T>` | `Seq { elem: T, holding: BorrowedVec }` |
191/// | `Cow<str>` / `Cow<[T]>` | `Str(Cow)` / `Seq { elem: T, holding: Cow }` |
192/// | `[T; N]` | `Array { elem: T, len: N }`, using the resolved length |
193/// | `&mut MaybeUninit<T>` | `Out(T)`, a slot the callee fills |
194///
195/// Other borrows, including `&mut str`, `&mut [T]`, and `&mut Vec<T>`,
196/// remain [`Shape::Ref`] with their inner type and access. `Option`, `Box`,
197/// other `Cow` types, `Result`, and callbacks expose their children without
198/// visiting them. Lifetimes are not represented in `Shape`; the original
199/// [`TypeRef`] remains available to the caller.
200///
201/// # Errors
202///
203/// A bare `str`, bare slice `[T]`, or bare `MaybeUninit<T>` returns an error
204/// explaining the supported form. The `&mut MaybeUninit<T>` case above is
205/// recognized before its inner type is visited. A different borrow such as
206/// `&MaybeUninit<T>` initially returns `Ref`; classifying its child then
207/// encounters the bare-`MaybeUninit` error.
208///
209/// See the [`shape` module](mod@crate::shape) for a runnable recursive example.
210pub fn shape<D>(
211    ty: &TypeRef,
212    shape_declared: impl Fn(&TypeRef) -> Option<D>,
213) -> Result<Shape<'_, D>, String> {
214    match ty.kind() {
215        TypeKind::Str => return Err("a bare `str` cannot cross; use `&str` or `String`".into()),
216        TypeKind::Slice(_) => {
217            return Err("a bare slice cannot cross; use `&[T]` or `Vec<T>`".into())
218        }
219        TypeKind::Uninit(_) => {
220            return Err("`MaybeUninit<T>` crosses only as `&mut MaybeUninit<T>`".into())
221        }
222        TypeKind::Ref {
223            mutable: true,
224            inner,
225            ..
226        } => {
227            if let TypeKind::Uninit(t) = inner.kind() {
228                return Ok(Shape::Out(t));
229            }
230        }
231        _ => {}
232    }
233    if let Some(declaration) = shape_declared(ty) {
234        return Ok(Shape::Declared { ty, declaration });
235    }
236    if let TypeKind::Ref { inner, .. } = ty.kind() {
237        if !matches!(
238            inner.kind(),
239            TypeKind::Str | TypeKind::Slice(_) | TypeKind::Uninit(_)
240        ) {
241            if let Some(declaration) = shape_declared(inner) {
242                return Ok(Shape::Declared { ty, declaration });
243            }
244        }
245    }
246    Ok(match ty.kind() {
247        TypeKind::Unit => Shape::Unit,
248        TypeKind::Scalar(k) => Shape::Scalar(*k),
249        TypeKind::String => Shape::Str(Holding::Owned),
250        TypeKind::Optional(t) => Shape::Option(t),
251        TypeKind::Vec(t) => Shape::Seq {
252            elem: t,
253            holding: Holding::Owned,
254        },
255        TypeKind::Array { elem, extent } => Shape::Array {
256            elem,
257            len: extent.value,
258        },
259        TypeKind::Boxed(t) => Shape::Boxed(t),
260        TypeKind::Cow { inner, .. } => match inner.kind() {
261            TypeKind::Str => Shape::Str(Holding::Cow),
262            TypeKind::Slice(e) => Shape::Seq {
263                elem: e,
264                holding: Holding::Cow,
265            },
266            _ => Shape::Cow(inner),
267        },
268        TypeKind::Ref { mutable, inner, .. } => {
269            let access = Access::of(ty);
270            match inner.kind() {
271                TypeKind::Str if !mutable => Shape::Str(Holding::Borrowed),
272                TypeKind::Slice(e) if !mutable => Shape::Seq {
273                    elem: e,
274                    holding: Holding::Borrowed,
275                },
276                TypeKind::Vec(e) if !mutable => Shape::Seq {
277                    elem: e,
278                    holding: Holding::BorrowedVec,
279                },
280                TypeKind::Named { id, .. } => Shape::Undeclared(id.name.as_str()),
281                _ => Shape::Ref { inner, access },
282            }
283        }
284        TypeKind::Named { id, .. } => Shape::Undeclared(id.name.as_str()),
285        TypeKind::Callback { args } => Shape::Callback(args),
286        TypeKind::Fallible { ok, err } => Shape::Result { ok, err },
287        TypeKind::Str | TypeKind::Slice(_) | TypeKind::Uninit(_) => unreachable!(),
288    })
