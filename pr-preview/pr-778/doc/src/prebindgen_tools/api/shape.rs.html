3/// How a value is held where it appears.
4#[derive(Clone, Copy, Debug, PartialEq, Eq)]
5pub enum Access {
6    /// By value: the callee owns it.
7    Owned,
8    /// `&T`.
9    Shared,
10    /// `&mut T`.
11    Exclusive,
12}
13
14/// How a run of values (text or a sequence) is held.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum Holding {
17    /// `String`, `Vec<T>`.
18    Owned,
19    /// `&str`, `&[T]`.
20    Borrowed,
21    /// `&Vec<T>`.
22    BorrowedVec,
23    /// `Cow<str>`, `Cow<[T]>`.
24    Cow,
25}
26
27/// A type, one layer deep.
28#[derive(Debug)]
29pub enum Shape<'t, S> {
30    /// `()`.
31    Unit,
32    /// A primitive.
33    Scalar(ScalarKind),
34    /// `String`, `&str`, `Cow<str>`.
35    Str(Holding),
36    /// `Vec<T>`, `&[T]`, `&Vec<T>`, `Cow<[T]>`.
37    Seq {
38        /// The sequence element to lower recursively.
39        elem: &'t TypeRef,
40        /// Whether the sequence is owned, borrowed, or held in a `Cow`.
41        holding: Holding,
42    },
43    /// `[T; N]`.
44    Array {
45        /// The array element to lower recursively.
46        elem: &'t TypeRef,
47        /// The fixed number of elements.
48        len: usize,
49    },
50    /// `Option<T>`.
51    Option(&'t TypeRef),
52    /// `Box<T>`.
53    Boxed(&'t TypeRef),
54    /// `Cow<T>` over anything but `str` or a slice.
55    Cow(&'t TypeRef),
56    /// A borrow of anything [`Shape::Str`], [`Shape::Seq`] and
57    /// [`Shape::Declared`] do not cover.
58    Ref {
59        /// The referenced type to lower recursively.
60        inner: &'t TypeRef,
61        /// Shared or exclusive access.
62        access: Access,
63    },
64    /// A named type the adapter has a setting for, held by value or
65    /// borrowed.
66    Declared {
67        /// Name in the flat namespace.
68        name: &'t str,
69        /// The adapter's setting returned by the `lookup` callback to [`shape`].
70        setting: S,
71        /// Whether this use owns or borrows the named type.
72        access: Access,
73    },
74    /// A named type the adapter has no setting for.
75    Undeclared(&'t str),
76    /// `impl Fn(A, B)`.
77    Callback(&'t [TypeRef]),
78    /// `Result<T, E>`.
79    Result {
80        /// Successful value to lower recursively.
81        ok: &'t TypeRef,
82        /// Error value to lower recursively.
83        err: &'t TypeRef,
84    },
85    /// `&mut MaybeUninit<T>`: a slot the callee fills.
86    Out(&'t TypeRef),
87}
88
89/// The outermost layer of `ty`. `lookup` gives the adapter's setting for a
90/// named type (`String` included, so an adapter may declare it); a borrow of
91/// a named type is that type with its [`Access`].
92///
93/// A bare `str` or slice, and a `MaybeUninit` outside `&mut`, are refused:
94/// no value of them can cross.
95pub fn shape<'t, S>(
96    ty: &'t TypeRef,
97    lookup: impl Fn(&str) -> Option<S>,
98) -> Result<Shape<'t, S>, String> {
99    let named = |name: &'t str, access: Access| match lookup(name) {
100        Some(setting) => Shape::Declared {
101            name,
102            setting,
103            access,
104        },
105        None => Shape::Undeclared(name),
106    };
107    Ok(match ty.kind() {
108        TypeKind::Unit => Shape::Unit,
109        TypeKind::Scalar(k) => Shape::Scalar(*k),
110        TypeKind::String => match lookup("String") {
111            Some(setting) => Shape::Declared {
112                name: "String",
113                setting,
114                access: Access::Owned,
115            },
116            None => Shape::Str(Holding::Owned),
117        },
118        TypeKind::Str => return Err("a bare `str` cannot cross; use `&str` or `String`".into()),
119        TypeKind::Slice(_) => {
120            return Err("a bare slice cannot cross; use `&[T]` or `Vec<T>`".into())
121        }
122        TypeKind::Optional(t) => Shape::Option(t),
123        TypeKind::Vec(t) => Shape::Seq {
124            elem: t,
125            holding: Holding::Owned,
126        },
127        TypeKind::Array { elem, extent } => Shape::Array {
128            elem,
129            len: extent.value,
130        },
131        TypeKind::Boxed(t) => Shape::Boxed(t),
132        TypeKind::Cow { inner, .. } => match inner.kind() {
133            TypeKind::Str => Shape::Str(Holding::Cow),
134            TypeKind::Slice(e) => Shape::Seq {
135                elem: e,
136                holding: Holding::Cow,
137            },
138            _ => Shape::Cow(inner),
139        },
140        TypeKind::Ref { mutable, inner, .. } => {
141            let access = if *mutable {
142                Access::Exclusive
143            } else {
144                Access::Shared
145            };
146            match inner.kind() {
147                TypeKind::Str if !mutable => Shape::Str(Holding::Borrowed),
148                TypeKind::Slice(e) if !mutable => Shape::Seq {
149                    elem: e,
150                    holding: Holding::Borrowed,
151                },
152                TypeKind::Vec(e) if !mutable => Shape::Seq {
153                    elem: e,
154                    holding: Holding::BorrowedVec,
155                },
156                TypeKind::Uninit(t) if *mutable => Shape::Out(t),
157                TypeKind::Named { id, .. } => named(&id.name, access),
158                TypeKind::String => match lookup("String") {
159                    Some(setting) => Shape::Declared {
160                        name: "String",
161                        setting,
162                        access,
163                    },
164                    None => Shape::Ref { inner, access },
165                },
166                _ => Shape::Ref { inner, access },
167            }
168        }
169        TypeKind::Named { id, .. } => named(&id.name, Access::Owned),
170        TypeKind::Callback { args } => Shape::Callback(args),
171        TypeKind::Fallible { ok, err } => Shape::Result { ok, err },
172        TypeKind::Uninit(_) => {
173            return Err("`MaybeUninit<T>` crosses only as `&mut MaybeUninit<T>`".into())
174        }
175    })
176}
