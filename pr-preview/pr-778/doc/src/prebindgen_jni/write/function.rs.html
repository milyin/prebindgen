33/// Write one binding: the extern into `out`, the wrapper returned.
34pub(crate) fn function(plan: &Plan, out: &mut Out, b: &Binding) -> Res<Wrapper> {
35    out.rust.push(extern_rust(plan, b)?);
36    out.externs.push(extern_kotlin(plan, b)?);
37    wrapper(plan, b)
38}

42fn extern_rust(plan: &Plan, b: &Binding) -> Res<TokenStream> {
43    let symbol = jni_symbol(&plan.harness_fqn(), &b.ext_name);
44    let jobject = quote!(::prebindgen_jni_runtime::jni::objects::JObject<'a>);
45    let callee = match &b.callee {
46        Callee::Call(c) | Callee::Value(c) => c.clone(),
47    };
48    let mut w = FunctionWriter::new(&b.func, callee)
49        .name(format_ident!("{}", symbol))
50        .abi(Some("system"))
51        .generics(quote!(<'a>))
52        .attr(quote!(#[no_mangle]))
53        .attr(quote!(#[allow(non_snake_case, unused_mut, unused_variables, unused_braces, unused_parens, unused_unsafe, dead_code, clippy::all)]))
54        .leading(Wire::new(
55            format_ident!("__env"),
56            quote!(::prebindgen_jni_runtime::jni::JNIEnv<'a>),
57        ))
58        .leading(Wire::new(
59            format_ident!("_class"),
60            quote!(::prebindgen_jni_runtime::jni::objects::JClass<'a>),
61        ))
62        .prologue(quote!(let mut __env = __env; let env = &mut __env;))
63        .trailing(Wire::new(format_ident!("__error_sink"), jobject.clone()));
64    if let Callee::Value(_) = b.callee {
65        w = w.call_with(|value, _| value.clone());
66    }
67    if let EPlan::Domain { .. } = b.err {
68        w = w.trailing(Wire::new(format_ident!("__domain_sink"), jobject));
69    }
70    w.write(&mut RustCb { plan, b })
71}

79impl FunctionCallbacks for RustCb<'_, '_> {
80    type Error = crate::Error;
81
82    fn ret(&mut self, _ret: &TypeRef) -> Res<Return> {
83        let (plan, b) = (self.plan, self.b);
84        let r = result_ident();
85        let whole = match &b.err {
86            EPlan::None => rust_ret_value(plan, &b.ret, r.to_token_stream())?,
87            e => {
88                let ok = rust_ret_value(plan, &b.ret, quote!(__ok))?;
89                let on_err = rust_error(plan, e)?;
90                quote!(match #r {
91                    ::core::result::Result::Ok(__ok) => #ok,
92                    ::core::result::Result::Err(__e) => #on_err,
93                })
94            }
95        };
96        let (ty, default) = ret_wire(&b.ret);
97        let wires = if uses_sink(&b.ret) {
98            vec![Wire::new(
99                format_ident!("__sink"),
100                quote!(::prebindgen_jni_runtime::jni::objects::JObject<'a>),
101            )]
102        } else {
103            Vec::new()
104        };
105        let rty = ty.clone().unwrap_or(quote!(()));
106        let e = error_ident();
107        Ok(Return {
108            ty,
109            wires,
110            body: quote! {
111                let __r: ::core::result::Result<#rty, ::std::string::String> = (|| #whole)();
112                match __r {
113                    ::core::result::Result::Ok(__v) => __v,
114                    ::core::result::Result::Err(#e) => {
115                        if !#e.is_empty() {
116                            __jni_signal(env, &__error_sink, &#e);
117                        }
118                        #default
119                    }
120                }
121            },
122        })
123    }
124
125    fn param(&mut self, name: &syn::Ident, _ty: &TypeRef) -> Res<Input> {
126        let plan = self.plan;
127        let p = self
128            .b
129            .params
130            .iter()
131            .find(|p| match p {
132                PPlan::Receiver(fp) | PPlan::Value(fp, _) => fp.name == *name,
133                PPlan::Selector(s, _) => s.param.name == *name,
134            })
135            .expect("a planned parameter");
136        let (fp, packed) = match p {
137            PPlan::Selector(s, _) => return plan.selector_input(s),
138            PPlan::Receiver(fp) => (fp, false),
139            PPlan::Value(fp, packed) => (fp, *packed),
140        };
141        let root = names::bare(&fp.name);
142        let mut input = plan.rs_decode(&fp.ty, &root, 0)?;
143        if packed {
144            let leaves: Vec<Leaf> = plan
145                .leaves(&fp.ty, Dir::In)?
146                .into_iter()
147                .map(|l| l.under(&root))
148                .collect();
149            input = pack::pack_input(&root, &leaves, input);
150        }
151        // A borrowed parameter lends the decoded value.
152        Ok(match plan.shape(&fp.ty)? {
153            Shape::Str(Holding::Borrowed)
154            | Shape::Seq {
155                holding: Holding::Borrowed | Holding::BorrowedVec,
156                ..
157            }
158            | Shape::Ref {
159                access: prebindgen_tools::Access::Shared,
160                ..
161            } => input.with_pass(quote!(&#name)),
162            Shape::Ref { .. } => {
163                return err(format!(
164                    "`{name}`: a `&mut` value parameter cannot cross from Kotlin"
165                ))
166            }
167            _ => input,
168        })
169    }
170
171    fn fail(&mut self, _ret: &Return) -> TokenStream {
172        let e = error_ident();
173        let (_, default) = ret_wire(&self.b.ret);
174        quote!({
175            __jni_signal(env, &__error_sink, &#e);
176            return #default;
177        })
178    }
179}

181fn uses_sink(ret: &RPlan) -> bool {
182    matches!(
183        ret,
184        RPlan::Sink { .. } | RPlan::Builder { .. } | RPlan::Fold { .. }
185    )

188/// The extern's return type, and the value it returns on failure.
189fn ret_wire(ret: &RPlan) -> (Option<TokenStream>, TokenStream) {
190    let object = (
191        Some(quote!(::prebindgen_jni_runtime::jni::sys::jobject)),
192        quote!(::core::ptr::null_mut()),
193    );
194    match ret {
195        RPlan::Unit => (None, quote!(())),
196        RPlan::Direct { leaf, .. } => match leaf.prim() {
197            Some(p) => (Some(p.rs()), p.rs_default()),
198            None => object,
199        },
200        _ => object,
201    }
202}

204/// The encoding of a successful value per the return plan, as statements
205/// ending in `Ok(<return wire>)`.
206fn rust_ret_value(plan: &Plan, ret: &RPlan, value: TokenStream) -> Res<TokenStream> {
207    Ok(match ret {
208        RPlan::Unit => quote!({ let _ = #value; ::core::result::Result::Ok(()) }),
209        RPlan::Direct { ty, leaf } => {
210            let out = plan.rs_encode(ty, value, "r", 1)?;
211            let bind = out.bind();
212            let n = &out.wires[0].name;
213            let v = if leaf.is_obj() {
214                quote!(#n.into_raw())
215            } else {
216                quote!(#n)
217            };
218            quote!({ #bind ::core::result::Result::Ok(#v) })
219        }
220        RPlan::Sink {
221            ty, leaves, iface, ..
222        } => call_sink(&plan.rs_encode(ty, value, "r", 1)?, leaves, iface),
223        RPlan::Builder {
224            whole,
225            exp,
226            iface,
227            leaves,
228            ..
229        } => {
230            let iface = format!("{iface}Raw");
231            layered(whole, value, &mut |t, v| {
232                let x = format_ident!("__x");
233                let d = plan.deliver(t, x.to_token_stream(), "r", "", Some(exp), &[], false, 1)?;
234                let call = call_sink(&d.output, leaves, &iface);
235                Ok(quote!({ let #x = #v; #call }))
236            })?
237        }
238        RPlan::Fold {
239            whole,
240            exp,
241            columns_iface,
242            leaves,
243            ..
244        } => layered(whole, value, &mut |t, v| {
245            let TypeKind::Vec(elem) = t.kind() else {
246                return err(format!("`{t}`: a folded result must be a `Vec`"));
247            };
248            let e = format_ident!("__x");
249            let d = plan.deliver(elem, e.to_token_stream(), "r", "", Some(exp), &[], false, 1)?;
250            let cols = columns(&d.output, leaves, &e, quote!(__items));
251            let mut col_leaves = vec![Leaf::new(LeafTy::Prim(Prim::I))];
252            col_leaves.extend(leaves.iter().map(Leaf::column));
253            let call = call_sink(&cols, &col_leaves, columns_iface);
254            Ok(
255                quote!({ let __items: ::std::vec::Vec<_> = ::core::iter::IntoIterator::into_iter(#v).collect(); #call }),
256            )
257        })?,
258    })
259}

261/// Walk the layers a builder or folder absorbs — `Box`, the first `Option`
262/// (absent: a null result) — down to the value `core` delivers (the `Vec`
263/// itself, for a fold).
264fn layered(
265    ty: &TypeRef,
266    value: TokenStream,
267    core: &mut dyn FnMut(&TypeRef, TokenStream) -> Res<TokenStream>,
268) -> Res<TokenStream> {
269    match ty.kind() {
270        TypeKind::Boxed(t) => layered(t, quote!((*#value)), core),
271        TypeKind::Optional(t) => {
272            let inner = unbox(t, quote!(__some), core)?;
273            Ok(quote!(match #value {
274                ::core::option::Option::Some(__some) => #inner,
275                ::core::option::Option::None => ::core::result::Result::Ok(::core::ptr::null_mut()),
276            }))
277        }
278        _ => core(ty, value),
279    }
280}

282/// Below the `Option`: only `Box` layers remain.
283fn unbox(
284    ty: &TypeRef,
285    value: TokenStream,
286    core: &mut dyn FnMut(&TypeRef, TokenStream) -> Res<TokenStream>,
287) -> Res<TokenStream> {
288    match ty.kind() {
289        TypeKind::Boxed(t) => unbox(t, quote!((*#value)), core),
290        _ => core(ty, value),
291    }
292}

294/// Columns of a sequence of deliveries: a count and one array per leaf.
295fn columns(elem: &Output, leaves: &[Leaf], x: &syn::Ident, items: TokenStream) -> Output {
296    let rt = quote!(::prebindgen_jni_runtime);
297    let mut setup = Vec::new();
298    let mut pushes = Vec::new();
299    let mut finals = vec![quote!(__n as i32)];
300    let mut wires = vec![Wire::new(format_ident!("__cn"), Prim::I.rs())];
301    for (k, (l, w)) in leaves.iter().zip(&elem.wires).enumerate() {
302        let col = format_ident!("__col{}", k);
303        let local = &w.name;
304        wires.push(Wire::new(format_ident!("__colw{}", k), l.column().rs()));
305        match l.prim() {
306            Some(p) => {
307                let write = format_ident!("{}", p.array_helpers().1);
308                setup.push(quote!(let mut #col = ::std::vec::Vec::with_capacity(__n);));
309                pushes.push(quote!(#col.push(#local);));
310                finals.push(quote!(#rt::#write(env, &#col)?));
311            }
312            None => {
313                setup.push(quote!(let #col = #rt::new_object_array(env, __n)?;));
314                pushes.push(quote!(#rt::object_array_set(env, &#col, __i, #local)?;));
315                finals.push(quote!(#col));
316            }
317        }
318    }
319    let bind = elem.bind();
320    Output::fallible(
321        wires,
322        quote!({
323            let __n = #items.len();
324            #(#setup)*
325            for (__i, #x) in #items.into_iter().enumerate() {
326                #bind
327                #(#pushes)*
328            }
329            (#(#finals),*)
330        }),
331    )
332}

334/// What an `Err` does: a message for the binding-error channel, or a
335/// delivery to the typed handler (the empty message then says it was
336/// delivered).
337fn rust_error(plan: &Plan, e: &EPlan) -> Res<TokenStream> {
338    Ok(match e {
339        EPlan::None => quote!(unreachable!()),
340        EPlan::Binding => quote!(::core::result::Result::Err(
341            ::std::string::ToString::to_string(&__e)
342        )),
343        EPlan::Domain {
344            ty,
345            raw_iface,
346            leaves,
347            ..
348        } => {
349            let d = plan.deliver(ty, quote!(__e), "e", "", None, &[], false, 1)?;
350            let bind = d.output.bind();
351            let values = d
352                .output
353                .wires
354                .iter()
355                .zip(leaves)
356                .map(|(w, l)| l.jvalue(&w.name.to_token_stream()));
357            let desc = method_desc(leaves, "V");
358            let fqn = raw_iface.replace('.', "/");
359            quote!({
360                #bind
361                static __D: ::prebindgen_jni_runtime::CachedIfaceMethod = ::prebindgen_jni_runtime::CachedIfaceMethod::new();
362                __D.call_void(env, #fqn, "run", #desc, &__domain_sink, &[#(#values),*])?;
363                ::core::result::Result::Err(::std::string::String::new())
364            })
365        }
366    })
367}

369/// Encode `out`'s wires as `jvalue`s and call the sink interface's `run`,
370/// returning its result object.
371fn call_sink(out: &Output, leaves: &[Leaf], iface: &str) -> TokenStream {
372    let bind = out.bind();
373    let values = out
374        .wires
375        .iter()
376        .zip(leaves)
377        .map(|(w, l)| l.jvalue(&w.name.to_token_stream()));
378    let desc = method_desc(leaves, "Ljava/lang/Object;");
379    let fqn = iface.replace('.', "/");
380    quote!({
381        #bind
382        static __S: ::prebindgen_jni_runtime::CachedIfaceMethod = ::prebindgen_jni_runtime::CachedIfaceMethod::new();
383        let __o = __S.call_object(env, #fqn, "run", #desc, &__sink, &[#(#values),*])?;
384        ::core::result::Result::Ok(__o.into_raw())
385    })
386}

390/// The extern's trailing object parameters: sink, error sink, domain sink.
391fn trailing(b: &Binding) -> Vec<&'static str> {
392    let mut v = Vec::new();
393    if uses_sink(&b.ret) {
394        v.push("sink");
395    }
396    v.push("errorSink");
397    if let EPlan::Domain { .. } = b.err {
398        v.push("domainSink");
399    }
400    v
401}

403fn extern_kotlin(plan: &Plan, b: &Binding) -> Res<String> {
404    let mut params = Vec::new();
405    for p in &b.params {
406        for l in plan.param_leaves(p)? {
407            params.push(format!("{}: {}", raw_name(&l), l.kt_raw()));
408        }
409    }
410    params.extend(trailing(b).iter().map(|n| format!("{n}: Any")));
411    let ret = match &b.ret {
412        RPlan::Unit => String::new(),
413        RPlan::Direct { leaf, .. } => format!(": {}", leaf.kt_raw()),
414        _ => ": Any?".to_string(),
415    };
416    Ok(format!(
417        "    external fun {}({}){ret}",
418        b.ext_name,
419        params.join(", ")
420    ))
421}

425/// The Kotlin type the wrapper returns, and its type parameters.
426fn kt_ret(plan: &Plan, ret: &RPlan) -> Res<(String, String)> {
427    let generic = |optional: bool, t: &str| {
428        let r = if optional {
429            format!("{t}?")
430        } else {
431            t.to_string()
432        };
433        (r, format!("<{t}> "))
434    };
435    Ok(match ret {
436        RPlan::Unit => ("Unit".to_string(), String::new()),
437        RPlan::Direct { ty, .. } | RPlan::Sink { ty, .. } => (plan.kt_type(ty)?, String::new()),
438        RPlan::Builder { optional, .. } => generic(*optional, "R"),
439        RPlan::Fold { optional, .. } => generic(*optional, "A"),
440    })
441}

443fn wrapper(plan: &Plan, b: &Binding) -> Res<Wrapper> {
444    let base = &plan.base_pkg;
445    let (ret, generics) = kt_ret(plan, &b.ret)?;
446    let mut cx = KtEnc::default();
447    let mut sig: Vec<SigParam> = Vec::new();
448    let mut args: Vec<String> = Vec::new();
449    for p in &b.params {
450        match p {
451            PPlan::Receiver(fp) => {
452                args.extend(plan.kt_encode(&fp.ty, "this", false, &mut cx, true)?);
453            }
454            PPlan::Value(fp, packed) => {
455                let n = kt_ident(&names::camel(&names::bare(&fp.name)));
456                sig.push(SigParam::Plain(n.clone(), plan.kt_type(&fp.ty)?));
457                let exprs = plan.kt_encode(&fp.ty, &n, false, &mut cx, true)?;
458                if *packed {
459                    args.extend(pack::pack_kotlin(&plan.leaves(&fp.ty, Dir::In)?, &exprs));
460                } else {
461                    args.extend(exprs);
462                }
463            }
464            PPlan::Selector(s, split) => {
465                args.extend(plan.selector_encode(s, &mut cx)?);
466                sig.push(if *split {
467                    SigParam::Split(s.clone())
468                } else {
469                    SigParam::Selector(s.clone())
470                });
471            }
472        }
473    }
474    // Tail parameters: accumulator, error handlers, builder / folder.
475    let mut tail: Vec<(String, String)> = Vec::new();
476    if let RPlan::Fold { .. } = b.ret {
477        tail.push(("acc".to_string(), "A".to_string()));
478    }
479    let je = format!("{base}.JniErrorHandler");
480    match &b.err {
481        EPlan::Domain { handler, .. } => {
482            tail.push(("onBindingError".to_string(), format!("{je}<{ret}>")));
483            tail.push(("onError".to_string(), format!("{handler}<{ret}>")));
484        }
485        _ => tail.push(("onError".to_string(), format!("{je}<{ret}>"))),
486    }
487    match &b.ret {
488        RPlan::Builder { iface, .. } => tail.push(("build".to_string(), format!("{iface}<R>"))),
489        RPlan::Fold { iface, .. } => tail.push(("fold".to_string(), format!("{iface}<A>"))),
490        _ => {}
491    }
492    let binding_handler = match b.err {
493        EPlan::Domain { .. } => "onBindingError",
494        _ => "onError",
495    };
496
497    // The call.
498    let mut call_args = args;
499    match &b.ret {
500        RPlan::Sink { kt_sink, .. } => call_args.push(kt_sink.clone()),
501        RPlan::Builder { .. } => call_args.push("build.asRaw()".to_string()),
502        RPlan::Fold {
503            columns_iface,
504            params,
505            leaves,
506            ..
507        } => {
508            let raws: Vec<String> = leaves.iter().map(raw_name).collect();
509            // The parameters decode over the raw leaf names: bind each to the
510            // element of its column.
511            let rebinds: Vec<String> = leaves
512                .iter()
513                .zip(&raws)
514                .map(|(l, r)| match l.prim() {
515                    Some(_) => format!("val {r} = __c_{r}[__i]"),
516                    None => format!("val {r} = __c_{r}[__i] as {}", l.kt_raw()),
517                })
518                .collect();
519            let decodes: Vec<String> = params.iter().map(|p| p.decode.clone()).collect();
520            let mut col_params = vec!["n".to_string()];
521            col_params.extend(raws.iter().map(|r| format!("__c_{r}")));
522            call_args.push(format!(
523                "{columns_iface} {{ {} ->\n            var __a = acc\n            for (__i in 0 until n) {{\n                {}\n                __a = fold.run(__a{}{})\n            }}\n            __a\n        }}",
524                col_params.join(", "),
525                rebinds.join("\n                "),
526                if decodes.is_empty() { "" } else { ", " },
527                decodes.join(", ")
528            ));
529        }
530        _ => {}
531    }
532    call_args.push("__bcap".to_string());
533    if let EPlan::Domain { .. } = b.err {
534        call_args.push("__dcap".to_string());
535    }
536    let call = format!(
537        "{}.{}({})",
538        plan.harness_fqn(),
539        b.ext_name,
540        call_args.join(", ")
541    );
542    let (pre, call) = guard_handles(plan, &cx.handles, call, binding_handler);
543
544    // Body.
545    let mut body = pre;
546    body.extend(cx.prelude);
547    body.push(format!(
548        "val __bcap = {base}.JniErrorHandlerCapture.acquire()"
549    ));
550    if let EPlan::Domain { capture, .. } = &b.err {
551        body.push(format!("val __dcap = {capture}.acquire()"));
552    }
553    body.push(format!("val __ret = {call}"));
554    body.push(format!(
555        "if (__bcap.failed) return {binding_handler}.run(__bcap.ze0)"
556    ));
557    if let EPlan::Domain { params, leaves, .. } = &b.err {
558        // The handler's parameters decode over the raw leaf names: read them
559        // off the capture.
560        let rebinds: Vec<String> = leaves
561            .iter()
562            .map(|l| {
563                let r = raw_name(l);
564                if l.is_obj() && l.nullable {
565                    format!("val {r} = __dcap.{r}")
566                } else {
567                    format!("val {r} = __dcap.{r}!!")
568                }
569            })
570            .collect();
571        let decodes: Vec<String> = params.iter().map(|p| p.decode.clone()).collect();
572        body.push(format!(
573            "if (__dcap.failed) {{\n        {}\n        return onError.run({})\n    }}",
574            rebinds.join("\n        "),
575            decodes.join(", ")
576        ));
577    }
578    let ret_expr = match &b.ret {
579        RPlan::Unit => "Unit".to_string(),
580        RPlan::Direct { ty, .. } => plan.kt_decode(ty, &["__ret".to_string()], false, 0)?,
581        _ => format!("__ret as {ret}"),
582    };
583    body.push(format!("return {ret_expr}"));
584
585    // Signature.
586    let vis = match &b.placement {
587        Placement::Method(c) if c.iface.is_some() => "public override",
588        _ => "public",
589    };
590    let suppress = if uses_sink(&b.ret) {
591        "@Suppress(\"UNCHECKED_CAST\")\n"
592    } else {
593        ""
594    };
595    let mut sig_strs: Vec<String> = Vec::new();
596    for p in &sig {
597        match p {
598            SigParam::Plain(n, t) => sig_strs.push(format!("{n}: {t}")),
599            SigParam::Selector(s) | SigParam::Split(s) => {
600                for (n, t) in plan.selector_params(s)? {
601                    sig_strs.push(format!("{n}: {t}"));
602                }
603            }
604        }
605    }
606    sig_strs.extend(tail.iter().map(|(n, t)| format!("{n}: {t}")));
607    let mut text = format!(
608        "{suppress}{vis} fun {generics}{}({}): {ret} {{\n    {}\n}}\n",
609        b.kt_name,
610        sig_strs.join(", "),
611        body.join("\n    ")
612    );
613    // Split overloads.
614    let splits = sig
615        .iter()
616        .filter(|p| matches!(p, SigParam::Split(_)))
617        .count();
618    if splits > 0 {
619        let head = format!("public fun {generics}");
620        for o in plan.split_overloads(&head, &b.kt_name, &sig, &tail, &ret, splits > 1)? {
621            text.push('\n');
622            text.push_str(&o);
623        }
624    }
625    let header = format!(
626        "fun {generics}{}({}): {ret}",
627        b.kt_name,
628        sig_strs.join(", ")
629    );
630    Ok(Wrapper { text, header })
631}

633/// The checks and locks around a call that passes handles: each must be
634/// open, a consumed one may not appear twice, every one is locked (in
635/// address order) for the call, and a consumed one is marked afterwards.
636/// Returns the statements to run first and the guarded call.
637fn guard_handles(
638    plan: &Plan,
639    handles: &[HandleSite],
640    mut call: String,
641    binding_handler: &str,
642) -> (Vec<String>, String) {
643    let base = &plan.base_pkg;
644    let fail = |msg: &str| format!("return {binding_handler}.run(\"{msg}\")");
645    let ptr = |h: &HandleSite| {
646        if h.nullable {
647            format!("({}?.ptr ?: 0L)", h.expr)
648        } else {
649            format!("{}.ptr", h.expr)
650        }
651    };
652    let mut pre = Vec::new();
653    let consumed: Vec<&HandleSite> = handles.iter().filter(|h| h.consumed).collect();
654    for (i, a) in consumed.iter().enumerate() {
655        for c in &consumed[i + 1..] {
656            if a.class.rust != c.class.rust {
657                continue;
658            }
659            let (pa, pc) = (ptr(a), ptr(c));
660            pre.push(format!(
661                "if (({pa} and -2L) != 0L && ({pa} and -2L) == ({pc} and -2L)) return {binding_handler}.run(\"Aliasing arguments: '{}' and '{}' are the same native resource; a consumed handle may not be passed twice in one call.\")",
662                a.expr, c.expr
663            ));
664        }
665    }
666    for h in handles {
667        let closed = if h.nullable {
668            format!("{}?.isClosed() == true", h.expr)
669        } else {
670            format!("{}.isClosed()", h.expr)
671        };
672        pre.push(format!(
673            "if ({closed}) {}",
674            fail("Operation on a closed native handle.")
675        ));
676    }
677    if !consumed.is_empty() {
678        let marks = consumed
679            .iter()
680            .map(|h| {
681                if h.nullable {
682                    format!("{}?.markConsumed()", h.expr)
683                } else {
684                    format!("{}.markConsumed()", h.expr)
685                }
686            })
687            .collect::<Vec<_>>()
688            .join("; ");
689        call = format!(
690            "try {{\n            {call}\n        }} finally {{\n            {marks}\n        }}"
691        );
692    }
693    if plan.handle_locks && !handles.is_empty() {
694        if handles.iter().all(|h| !h.nullable) && handles.len() <= 3 {
695            let hs = handles
696                .iter()
697                .map(|h| h.expr.clone())
698                .collect::<Vec<_>>()
699                .join(", ");
700            call = format!("{base}.withSortedHandleLocks({hs}) {{\n        {call}\n    }}");
701        } else {
702            let adds = handles
703                .iter()
704                .map(|h| {
705                    if h.nullable {
706                        format!("{}?.let {{ __locks.add(it) }}", h.expr)
707                    } else {
708                        format!("__locks.add({})", h.expr)
709                    }
710                })
711                .collect::<Vec<_>>()
712                .join("; ");
713            call = format!(
714                "run {{\n        val __locks = ArrayList<{base}.NativeHandle>(); {adds}\n        {base}.withSortedHandleLocks(__locks) {{\n            {call}\n        }}\n    }}"
715            );
716        }
717    }
718    (pre, call)
719}
