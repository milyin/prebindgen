1//! Optional expression protocol for stored plans used by the C and JNI adapters.
2//! Scoped requests keep prepared temporaries alive while evaluating a continuation.
3//! Other generators can define entirely different request enums.
4use crate::{Access, InputPlan, OutputPlan, TemplateContext};
5use prebindgen_flat::{Conversion, Function, Result, WireKind};
6use proc_macro2::TokenStream;
7use quote::quote;

9/// Slot requests understood by the stored input plans.
10pub enum InputFragment<'a, C> {
11    /// Validate and prepare storage, then evaluate `next` in that storage's scope.
12    Validate {
13        /// Bindings used to render prepared templates.
14        context: &'a C,
15        /// Expression evaluated inside the scope introduced by this request.
16        next: TokenStream,
17    },
18    /// Acquire ownership, then evaluate `next` while acquired owners remain alive.
19    Acquire {
20        /// Bindings used to render prepared templates.
21        context: &'a C,
22        /// Expression evaluated inside the scope introduced by this request.
23        next: TokenStream,
24    },
25    /// Construct the argument as `target`, then evaluate `next` in its scope.
26    Construct {
27        /// Bindings used to render prepared templates.
28        context: &'a C,
29        /// Caller-selected argument binding.
30        target: &'a syn::Ident,
31        /// Expression evaluated inside the scope introduced by this request.
32        next: TokenStream,
33    },
34    /// Unit expression acknowledging ownership, or exactly `()` if unnecessary.
35    Acknowledge {
36        /// Bindings used to render prepared templates.
37        context: &'a C,
38    },
39    /// Check addresses against `seen: Vec<(usize, bool)>`, then record them.
40    /// The boolean means exclusive access (including ownership transfer).
41    CheckAliases {
42        /// Bindings used to render prepared templates.
43        context: &'a C,
44        /// Caller-owned collection of previously checked allocation accesses.
45        seen: &'a syn::Ident,
46    },
47    /// Check aliases within this parameter when no other parameter has resources.
48    /// A single address, or shared-only accesses, cannot conflict and needs no
49    /// runtime bookkeeping. Dynamic exclusive sets still check internal aliases.
50    CheckInternalAliases {
51        /// Bindings used to render prepared templates.
52        context: &'a C,
53    },
54}
55/// Slot requests understood by stored output plans.
56pub enum OutputFragment<'a, C> {
57    /// Prepare output temporaries, then evaluate `next` while they remain alive.
58    Prepare {
59        /// Bindings used to render prepared templates.
60        context: &'a C,
61        /// Caller-selected source-result binding.
62        source: &'a syn::Ident,
63        /// Expression evaluated inside the scope introduced by this request.
64        next: TokenStream,
65    },
66    /// Deliver output, then evaluate `next` with delivery bindings still in scope.
67    Deliver {
68        /// Bindings used to render prepared templates.
69        context: &'a C,
70        /// Caller-selected source-result binding.
71        source: &'a syn::Ident,
72        /// Expression evaluated inside the scope introduced by this request.
73        next: TokenStream,
74    },
75    /// Unit expression committing ownership after successful delivery.
76    Commit {
77        /// Bindings used to render prepared templates.
78        context: &'a C,
79        /// Caller-selected source-result binding.
80        source: &'a syn::Ident,
81    },
82    /// Wrapper's final result expression.
83    Result {
84        /// Bindings used to render prepared templates.
85        context: &'a C,
86        /// Caller-selected source-result binding.
87        source: &'a syn::Ident,
88    },
89}
90fn expression(code: TokenStream) -> Result<TokenStream> {
91    syn::parse2::<syn::Expr>(code.clone())
92        .map_err(|e| format!("conversion did not emit an expression: {e}"))?;
93    Ok(code)
94}

96impl<W: WireKind, C: TemplateContext + 'static, N, B> Conversion for InputPlan<W, C, N, B> {
97    type Wire = W;
98    type Chain = N;
99    type Branch = B;
100    type SlotFragment<'a> = InputFragment<'a, C>;
101
102    fn code_fragment_slot(
103        &self,
104        mode: InputFragment<'_, C>,
105        _expected_type: &syn::Type,
106    ) -> Result<TokenStream> {
107        expression(match mode {
108            InputFragment::Validate { context, next } => {
109                let code = context.render(&self.validation)?;
110                scope(code, next)?
111            }
112            InputFragment::Acquire { context, next } => {
113                let code = context.render(&self.acquire)?;
114                scope(code, next)?
115            }
116            InputFragment::Construct {
117                context,
118                target,
119                next,
120            } => {
121                let code = context.render(&self.statements)?;
122                let value = context.render(&self.argument)?;
123                scope(quote!(#code let #target = #value;), next)?
124            }
125            InputFragment::Acknowledge { context } => match &self.acknowledgement {
126                Some(code) => {
127                    let code = context.render(code)?;
128                    quote!({ #code })
129                }
130                None => quote!(()),
131            },
132            InputFragment::CheckAliases { context, seen } => {
133                if self.resources.is_empty() {
134                    return Ok(quote!(()));
135                }
136                let mut checks = TokenStream::new();
137                for resource in &self.resources {
138                    let active = context.render(&resource.active)?;
139                    let exclusive = resource.access != Access::Shared;
140                    let identities = match &resource.identity {
141                        crate::ResourceIdentity::Single(code) => {
142                            let code = context.render(code)?;
143                            quote!(::core::iter::once((#code) as usize))
144                        }
145                        crate::ResourceIdentity::Many(code) => context.render(code)?,
146                    };
147                    checks.extend(quote! {
148                        if #active { for identity in #identities { #seen.push((identity, #exclusive)); } }
149                    });
150                }
151                if !self.resources.is_empty() {
152                    checks.extend(quote! {
153                        #seen.sort_unstable_by_key(|entry| entry.0);
154                        if #seen.windows(2).any(|pair| pair[0].0 == pair[1].0 && (pair[0].1 || pair[1].1)) {
155                            return Err("conflicting input aliases".into());
156                        }
157                    });
158                }
159                quote!({ #checks })
160            }
161            InputFragment::CheckInternalAliases { context } => {
162                if self.resources.iter().all(|r| r.access == Access::Shared)
163                    || matches!(self.resources.as_slice(), [r] if matches!(r.identity, crate::ResourceIdentity::Single(_)))
164                {
165                    return Ok(quote!(()));
166                }
167                // Resource expressions can refer to validation locals. Keep the
168                // private scratch binding from shadowing any of those locals.
169                let mut used = std::collections::BTreeSet::new();
170                for resource in &self.resources {
171                    reserve(&context.render(&resource.active)?, &mut used);
172                    let identity = match &resource.identity {
173                        crate::ResourceIdentity::Single(code)
174                        | crate::ResourceIdentity::Many(code) => code,
175                    };
176                    reserve(&context.render(identity)?, &mut used);
177                }
178                let mut name = "__prebindgen_parameter_resources".to_string();
179                while used.contains(&name) {
180                    name.push('_');
181                }
182                let seen = syn::Ident::new(&name, proc_macro2::Span::call_site());
183                let check = self.code_fragment_slot(
184                    InputFragment::CheckAliases {
185                        context,
186                        seen: &seen,
187                    },
188                    &syn::parse_quote!(()),
189                )?;
190                quote!({ let mut #seen: Vec<(usize, bool)> = Vec::new(); let _: () = #check; })
191            }
192        })
193    }
194}
195impl<W: WireKind, C: TemplateContext + 'static, N, B> Conversion for OutputPlan<W, C, N, B> {
196    type Wire = W;
197    type Chain = N;
198    type Branch = B;
199    type SlotFragment<'a> = OutputFragment<'a, C>;
200
201    fn code_fragment_slot(
202        &self,
203        mode: OutputFragment<'_, C>,
204        _expected_type: &syn::Type,
205    ) -> Result<TokenStream> {
206        expression(match mode {
207            OutputFragment::Prepare {
208                context,
209                source,
210                next,
211            } => {
212                let code = crate::emission::render_output(context, &self.statements, source)?;
213                scope(code, next)?
214            }
215            OutputFragment::Deliver {
216                context,
217                source,
218                next,
219            } => {
220                let code =
221                    crate::emission::render_output(context, &self.delivery.statements, source)?;
222                scope(code, next)?
223            }
224            OutputFragment::Commit { context, source } => {
225                let code = crate::emission::render_output(context, &self.commit, source)?;
226                quote!({ #code })
227            }
228            OutputFragment::Result { context, source } => {
229                crate::emission::render_output(context, &self.delivery.result, source)?
230            }
231        })
232    }
233}
234/// Compose the opt-in staged protocol as a fallible expression.
235/// `result_type` is the wrapper result type inside `Result<_, String>`.
236///
237/// Generators choosing this helper use address alias checks and one shared
238/// acknowledgement. Other generators can compose their own request protocols.
239/// This is deliberately outside Flat: none of these phases are required by its traits.
240pub fn emit_call<I, O, C: 'static, M>(
241    function: &Function<I, O, M>,
242    context: &C,
243    source: &syn::Ident,
244    result_type: &syn::Type,
245    parameter_names: &[syn::Ident],
246) -> Result<TokenStream>
247where
248    for<'a> I: Conversion<SlotFragment<'a> = InputFragment<'a, C>>,
249    for<'a> O: Conversion<SlotFragment<'a> = OutputFragment<'a, C>>,
250{
251    let inputs = function.input_conversions()?;
252    let output = function.output_conversion()?;
253    let unit = syn::parse_quote!(());
254    let expected = syn::parse_quote!(::core::result::Result<#result_type, ::std::string::String>);
255    let continuation = crate::template_parameter("continuation");
256    let path = &function.path;
257    let mut acknowledge: Option<TokenStream> = None;
258    for input in &inputs {
259        let code = input.code_fragment_slot(InputFragment::Acknowledge { context }, &unit)?;
260        if code.to_string() == quote!(()).to_string() {
261            continue;
262        }
263        if acknowledge
264            .as_ref()
265            .is_some_and(|previous| previous.to_string() != code.to_string())
266        {
267            return Err(format!(
268                "`{}`: conflicting input acknowledgements",
269                function.name
270            ));
271        }
272        acknowledge = Some(code);
273    }
274    let validation = inputs
275        .iter()
276        .map(|input| {
277            input.code_fragment_slot(
278                InputFragment::Validate {
279                    context,
280                    next: continuation.clone(),
281                },
282                &expected,
283            )
284        })
285        .collect::<Result<Vec<_>>>()?;
286    let acquire = inputs
287        .iter()
288        .map(|input| {
289            input.code_fragment_slot(
290                InputFragment::Acquire {
291                    context,
292                    next: continuation.clone(),
293                },
294                &expected,
295            )
296        })
297        .collect::<Result<Vec<_>>>()?;
298    let prepare = output.code_fragment_slot(
299        OutputFragment::Prepare {
300            context,
301            source,
302            next: continuation.clone(),
303        },
304        &expected,
305    )?;
306    let deliver = output.code_fragment_slot(
307        OutputFragment::Deliver {
308            context,
309            source,
310            next: continuation.clone(),
311        },
312        &expected,
313    )?;
314    let commit = output.code_fragment_slot(OutputFragment::Commit { context, source }, &unit)?;
315    let result =
316        output.code_fragment_slot(OutputFragment::Result { context, source }, result_type)?;
317    let mut used = parameter_names
318        .iter()
319        .map(ToString::to_string)
320        .collect::<std::collections::BTreeSet<_>>();
321    reserve(
322        &quote!(#path #source #(#validation)* #(#acquire)* #prepare #deliver #commit #result #acknowledge),
323        &mut used,
324    );
325    let mut fresh = |base: String| {
326        let mut name = base;
327        while !used.insert(name.clone()) {
328            name.push('_');
329        }
330        syn::Ident::new(&name, proc_macro2::Span::call_site())
331    };
332    let seen = fresh("__prebindgen_resources".into());
333    let args: Vec<_> = (0..inputs.len())
334        .map(|i| fresh(format!("__prebindgen_argument_{i}")))
335        .collect();
336    let mut checks = inputs
337        .iter()
338        .enumerate()
339        .map(|(index, input)| {
340            let code = input.code_fragment_slot(
341                InputFragment::CheckAliases {
342                    context,
343                    seen: &seen,
344                },
345                &unit,
346            )?;
347            Ok((index, code))
348        })
349        .collect::<Result<Vec<_>>>()?;
350    checks.retain(|(_, code)| code.to_string() != quote!(()).to_string());
351    let alias_checks = if let [(index, _)] = checks.as_slice() {
352        let check = inputs[*index]
353            .code_fragment_slot(InputFragment::CheckInternalAliases { context }, &unit)?;
354        quote!(let _: () = #check;)
355    } else if checks.is_empty() {
356        quote!()
357    } else {
358        let checks = checks.into_iter().map(|(_, code)| code);
359        quote! {
360            let mut #seen: Vec<(usize, bool)> = Vec::new();
361            #(let _: () = #checks;)*
362        }
363    };
364    let delivered = plug(&deliver, quote!({ let _: () = #commit; Ok(#result) }))?;
365    let mut body = plug(&prepare, delivered)?;
366    body = scope(quote!(let #source = #path(#(#args),*);), body)?;
367    for (input, target) in inputs.iter().zip(&args).rev() {
368        body = input.code_fragment_slot(
369            InputFragment::Construct {
370                context,
371                target,
372                next: body,
373            },
374            &expected,
375        )?;
376    }
377    for scope in acquire.iter().rev() {
378        body = plug(scope, body)?;
379    }
380    body = if let Some(ack) = acknowledge {
381        scope(quote!(let _: () = #ack;), body)?
382    } else {
383        body
384    };
385    body = scope(alias_checks, body)?;
386    for scope in validation.iter().rev() {
387        body = plug(scope, body)?;
388    }
389    Ok(body)
390}
391fn plug(scope: &TokenStream, next: TokenStream) -> Result<TokenStream> {
392    let code = crate::render_template(scope, &|name| {
393        if name == "continuation" {
394            Ok(next.clone())
395        } else {
396            Err(format!("unknown composition parameter `{name}`"))
397        }
398    })?;
399    flatten_tail(code)
400}

402// Scoped requests use a tail continuation. Merge unlabelled tail blocks to keep
403// generated syntax shallow for functions with hundreds of wire parameters.
404// Statements retain their order, and locals still surround the continuation.
405fn scope(prefix: TokenStream, next: TokenStream) -> Result<TokenStream> {
406    flatten_tail(quote!({ #prefix #next }))
407}
408fn flatten_tail(code: TokenStream) -> Result<TokenStream> {
409    let mut expr: syn::Expr = syn::parse2(code).map_err(|e| e.to_string())?;
410    if let syn::Expr::Block(outer) = &mut expr {
411        while let Some(syn::Stmt::Expr(syn::Expr::Block(inner), None)) = outer.block.stmts.last() {
412            if inner.label.is_some() || !inner.attrs.is_empty() {
413                break;
414            }
415            let Some(syn::Stmt::Expr(syn::Expr::Block(inner), None)) = outer.block.stmts.pop()
416            else {
417                unreachable!()
418            };
419            outer.block.stmts.extend(inner.block.stmts);
420        }
421    }
422    Ok(quote!(#expr))
423}
424fn reserve(code: &TokenStream, used: &mut std::collections::BTreeSet<String>) {
425    for token in code.clone() {
426        match token {
427            proc_macro2::TokenTree::Ident(name) => {
428                used.insert(name.to_string());
429            }
430            proc_macro2::TokenTree::Group(group) => reserve(&group.stream(), used),
431            _ => {}
432        }
433    }
434}
