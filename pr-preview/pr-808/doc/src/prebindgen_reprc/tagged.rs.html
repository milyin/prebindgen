1//! Target-compiled tagged-union carriers, independent of source enum layouts.
2use crate::{Record, Result};
3use proc_macro2::TokenStream;
4use quote::{format_ident, quote};
5use syn::{Ident, Type};
6
7/// A C-representation tag followed by a union of converted variant records.
8/// Empty variants have no union member. Ownership stays with the adapter.
9#[derive(Clone, Debug)]
10pub struct TaggedUnion {
11    name: Ident,
12    arms: Vec<Option<Record>>,
13}
14impl TaggedUnion {
15    /// Build a signed-i32-tagged carrier. Arm indices are zero-based wire tags.
16    /// Records must have distinct valid names and C-compatible fields.
17    pub fn new(name: Ident, arms: Vec<Option<Record>>) -> Result<Self> {
18        if arms.is_empty() || i32::try_from(arms.len()).is_err() {
19            return Err("tagged union needs 1..=i32::MAX variants".into());
20        }
21        let mut names = std::collections::BTreeSet::new();
22        for arm in arms.iter().flatten() {
23            if arm.definition().ident == name
24                || arm.definition().ident == format_ident!("{name}_payload")
25                || !names.insert(arm.definition().ident.to_string())
26            {
27                return Err("duplicate tagged-union arm record".into());
28            }
29        }
30        // Reuse the record validator for the public carrier identifier.
31        Record::new(
32            name.clone(),
33            [(syn::parse_quote!(tag), syn::parse_quote!(i32))],
34        )?;
35        Ok(Self { name, arms })
36    }
37    /// The carrier's Rust identifier.
38    pub fn name(&self) -> &Ident {
39        &self.name
40    }
41    /// Generated union identifier, when at least one variant needs storage.
42    pub fn payload_name(&self) -> Option<Ident> {
43        self.arms
44            .iter()
45            .any(Option::is_some)
46            .then(|| format_ident!("{}_payload", self.name))
47    }
48    /// The source-order arm records; None denotes a zero-storage arm.
49    pub fn arms(&self) -> &[Option<Record>] {
50        &self.arms
51    }
52    /// Emit all arm records, the payload union and the carrier structure.
53    /// ManuallyDrop permits owning/non-Copy carrier fields without implicit drop.
54    pub fn definition(&self) -> TokenStream {
55        let name = &self.name;
56        if let Some(payload) = self.payload_name() {
57            let definitions = self.arms.iter().flatten().map(Record::definition);
58            let members = self.arms.iter().enumerate().filter_map(|(index, record)| {
59                record.as_ref().map(|r| {
60                    let field = format_ident!("a{index}");
61                    let ty = r.definition().ident;
62                    quote!(pub #field: ::core::mem::ManuallyDrop<#ty>)
63                })
64            });
65            quote!(#(#definitions)* #[repr(C)] pub union #payload {pub __empty: [u8;0],#(#members,)*} #[repr(C)] pub struct #name {pub tag:i32,pub payload:#payload})
66        } else {
67            quote!(#[repr(C)] pub struct #name {pub tag:i32})
68        }
69    }
70    /// Target-side byte offset of the discriminator.
71    pub fn tag_offset(&self) -> TokenStream {
72        let n = &self.name;
73        quote!(::core::mem::offset_of!(#n,tag))
74    }
75    /// Target-side byte offset of an active variant member. Union arms start at zero.
76    /// Compute component offsets separately to preserve Rust 1.85 compatibility.
77    pub fn field_offset(&self, arm: usize, field: &Ident) -> Result<TokenStream> {
78        let record = self
79            .arms
80            .get(arm)
81            .and_then(Option::as_ref)
82            .ok_or("variant has no payload storage")?;
83        if !record.fields().any(|(name, _)| name == field) {
84            return Err("unknown tagged-union field".into());
85        }
86        let root = &self.name;
87        let arm = record.definition().ident;
88        Ok(quote!(
89            ::core::mem::offset_of!(#root,payload) + ::core::mem::offset_of!(#arm,#field)
90        ))
91    }
92    /// Carrier type syntax, for embedding in another generated record.
93    pub fn ty(&self) -> Type {
94        let name = &self.name;
95        syn::parse_quote!(#name)
96    }
97}

99#[cfg(test)]
100mod tests {
101    use super::*;
102    use std::{fs, process::Command};
103    use syn::parse_quote;
104    #[test]
105    fn empty_arms_have_no_payload_record() {
106        let tag = TaggedUnion::new(parse_quote!(Empty), vec![None, None]).unwrap();
107        assert!(tag.payload_name().is_none());
108        assert!(!tag.definition().to_string().contains("union"));
109        assert!(tag.field_offset(0, &parse_quote!(missing)).is_err());
110    }
111    #[test]
112    fn target_union_layout_matches_c() {
113        let arm = Record::new(
114            parse_quote!(Numeric),
115            [
116                (parse_quote!(small), parse_quote!(u8)),
117                (parse_quote!(large), parse_quote!(u64)),
118            ],
119        )
120        .unwrap();
121        let pointer = Record::new(
122            parse_quote!(Pointer),
123            [
124                (parse_quote!(data), parse_quote!(*const u8)),
125                (parse_quote!(length), parse_quote!(usize)),
126            ],
127        )
128        .unwrap();
129        let union =
130            TaggedUnion::new(parse_quote!(Message), vec![None, Some(arm), Some(pointer)]).unwrap();
131        let definition = union.definition();
132        let tag = union.tag_offset();
133        let small = union.field_offset(1, &parse_quote!(small)).unwrap();
134        let large = union.field_offset(1, &parse_quote!(large)).unwrap();
135        let data = union.field_offset(2, &parse_quote!(data)).unwrap();
136        let source = quote!(#definition fn main(){for n in [::core::mem::size_of::<Message>(),::core::mem::align_of::<Message>(),#tag,#small,#large,#data]{println!("{}",n);}});
137        let dir = tempfile::tempdir().unwrap();
138        let rust = dir.path().join("layout.rs");
139        let rust_exe = dir.path().join("rust-layout");
140        fs::write(&rust, source.to_string()).unwrap();
141        let output = Command::new(std::env::var_os("RUSTC").unwrap_or_else(|| "rustc".into()))
142            .args(["--edition=2021", "-Dwarnings"])
143            .arg(&rust)
144            .arg("-o")
145            .arg(&rust_exe)
146            .output()
147            .unwrap();
148        assert!(
149            output.status.success(),
150            "{}",
151            String::from_utf8_lossy(&output.stderr)
152        );
153        let c = dir.path().join("layout.c");
154        let c_exe = dir.path().join("c-layout");
155        fs::write(&c,r#"#include <stddef.h>
156#include <stdint.h>
157#include <stdio.h>
158typedef struct { uint8_t small; uint64_t large; } Numeric;
159typedef struct { const uint8_t *data; size_t length; } Pointer;
160typedef struct { int32_t tag; union { Numeric numeric; Pointer pointer; } payload; } Message;
161int main(void) { size_t layout[]={sizeof(Message),_Alignof(Message),offsetof(Message,tag),offsetof(Message,payload)+offsetof(Numeric,small),offsetof(Message,payload)+offsetof(Numeric,large),offsetof(Message,payload)+offsetof(Pointer,data)}; for(size_t i=0;i<sizeof(layout)/sizeof(layout[0]);++i) printf("%zu\n",layout[i]); return 0; }
162"#).unwrap();
163        let output = Command::new(std::env::var_os("CC").unwrap_or_else(|| "cc".into()))
164            .args(["-std=c11", "-Wall", "-Wextra", "-Werror"])
165            .arg(&c)
166            .arg("-o")
167            .arg(&c_exe)
168            .output()
169            .unwrap();
170        assert!(
171            output.status.success(),
172            "{}",
173            String::from_utf8_lossy(&output.stderr)
174        );
175        assert_eq!(
176            Command::new(rust_exe).output().unwrap().stdout,
177            Command::new(c_exe).output().unwrap().stdout
178        );
179    }
