1//! Preparation-only representation declarations and defaults.
2//! Resolve choices and copy required metadata into slots before generation.
3use crate::ComponentPath;
4use prebindgen_flat::{type_key, Conversion, Flat, Result};
5use std::collections::BTreeMap;
6
7/// Identity used only while resolving preparation declarations.
8#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
9pub struct RepresentationId(pub String);
10
11/// A logical component in a representation, not an executable conversion.
12/// Every component must have at least one wire when a conversion selects this
13/// representation. Partial projections must select their own representation.
14#[derive(Clone, Debug)]
15pub struct Component {
16    /// Stable path used by conversion wires.
17    pub path: ComponentPath,
18    /// Public presentation name, separate from generated Rust local names.
19    pub name: String,
20    /// Optional source record field path. Derived views need no field association.
21    pub source_fields: Option<Vec<String>>,
22    /// Representation of the component, when it has a separately declared shape.
23    pub representation: Option<RepresentationId>,
24}
25
26/// Shape declaration used during preparation, never referenced by a conversion slot.
27#[derive(Clone, Debug)]
28pub struct Representation<R> {
29    /// Qualified Rust type represented by this shape.
30    pub source_type: syn::Type,
31    /// Logical components; declaration order is retained.
32    pub components: Vec<Component>,
33    /// Adapter-specific type declaration and materialization information.
34    pub data: R,
35}
36
37/// Representation registry and optional canonical defaults by Rust type.
38#[derive(Clone, Debug)]
39pub struct Representations<R> {
40    entries: BTreeMap<RepresentationId, Representation<R>>,
41    canonical: BTreeMap<String, RepresentationId>,
42}
43impl<R> Default for Representations<R> {
44    fn default() -> Self {
45        Self {
46            entries: BTreeMap::new(),
47            canonical: BTreeMap::new(),
48        }
49    }
50}
51impl<R> Representations<R> {
52    /// Insert or replace a shape. Reject duplicate component identities atomically.
53    pub fn insert(&mut self, id: RepresentationId, shape: Representation<R>) -> Result<()> {
54        let mut paths = std::collections::BTreeSet::new();
55        for component in &shape.components {
56            if !paths.insert(&component.path) {
57                return Err(format!("duplicate component in representation `{}`", id.0));
58            }
59        }
60        self.entries.insert(id, shape);
61        Ok(())
62    }
63    /// Resolve a representation without interpreting its adapter data.
64    pub fn get(&self, id: &RepresentationId) -> Result<&Representation<R>> {
65        self.entries
66            .get(id)
67            .ok_or_else(|| format!("unknown representation `{}`", id.0))
68    }
69    /// Iterate declarations in identity order during preparation.
70    pub fn iter(&self) -> impl Iterator<Item = (&RepresentationId, &Representation<R>)> {
71        self.entries.iter()
72    }
73    /// Designate an existing representation as its source type's default.
74    /// This does not populate or change any occurrence conversion.
75    pub fn set_canonical(&mut self, id: &RepresentationId) -> Result<()> {
76        let key = type_key(&self.get(id)?.source_type);
77        self.canonical.insert(key, id.clone());
78        Ok(())
79    }
80    /// Look up a preparation default. Copy the selected data into prepared slots.
81    pub fn canonical(&self, ty: &syn::Type) -> Option<&RepresentationId> {
82        self.canonical.get(&type_key(ty))
83    }
84    /// Validate nested references after all declarations have been registered.
85    fn validate_references(&self) -> Result<()> {
86        for shape in self.entries.values() {
87            for component in &shape.components {
88                if let Some(id) = &component.representation {
89                    self.get(id)?;
90                }
91            }
92        }
93        for (ty, id) in &self.canonical {
94            if type_key(&self.get(id)?.source_type) != *ty {
95                return Err(format!(
96                    "canonical representation `{}` changed source type",
97                    id.0
98                ));
99            }
100        }
101        Ok(())
102    }
103}
104
105impl<R> Representations<R> {
106    /// Validate all registered type shapes, including schemas not used by a function.
107    pub fn validate<I: Conversion, O: Conversion>(&self, flat: &Flat<I, O>) -> Result<()> {
108        self.validate_references()?;
109        for (id, shape) in self.iter() {
110            flat.resolve_type(&shape.source_type)?;
111            for component in &shape.components {
112                if let Some(fields) = &component.source_fields {
113                    let mut ty = &shape.source_type;
114                    for field in fields {
115                        ty = &flat
116                            .record(ty)?
117                            .fields
118                            .iter()
119                            .find(|f| f.name == field.as_str())
120                            .ok_or_else(|| {
121                                format!(
122                                    "unknown source field `{field}` in representation `{}`",
123                                    id.0
124                                )
125                            })?
126                            .ty;
127                    }
128                    if let Some(child) = &component.representation {
129                        if !prebindgen_flat::same_type(ty, &self.get(child)?.source_type) {
130                            return Err(format!(
131                                "component `{}` has an incompatible source type",
132                                component.name
133                            ));
134                        }
135                    }
136                }
137            }
138        }
139        Ok(())
140    }
141    /// Validate an occurrence mapping during preparation, before installing its slot.
142    /// Partial projections require a matching declaration; several wires may share a component.
143    pub fn validate_mapping<'a>(
144        &self,
145        ty: &syn::Type,
146        id: &RepresentationId,
147        mapping: impl IntoIterator<Item = (&'a syn::Ident, &'a ComponentPath)>,
148    ) -> Result<()> {
149        let wires: Vec<_> = mapping.into_iter().collect();
150        let shape = self.get(id)?;
151        if !prebindgen_flat::same_type(ty, &shape.source_type) {
152            return Err(format!(
153                "representation `{}` does not match occurrence type",
154                id.0
155            ));
156        }
157        for (name, path) in &wires {
158            if !shape.components.iter().any(|c| c.path == **path) {
159                return Err(format!(
160                    "wire `{}` has no component in representation `{}`",
161                    name, id.0
162                ));
163            }
164        }
165        for component in &shape.components {
166            if !wires.iter().any(|(_, path)| **path == component.path) {
167                return Err(format!(
168                    "component `{}` has no wire in representation `{}`",
169                    component.name, id.0
170                ));
171            }
172        }
173        Ok(())
174    }
175}
