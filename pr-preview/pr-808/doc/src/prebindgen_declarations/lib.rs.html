1//! Declaration language and composition for preparing conversion slots.
2//!
3//! [`Declarations::fill`], [`bind_input`] and [`bind_output`] build conversions
4//! directly in a [`Bindings`] model. Recipes and codec input data are used only
5//! during construction; slots retain wires and code needed by wrapper writers.
6//! Generation uses [`prebindgen_flat::emit_call`] and the call emitter in `prebindgen-flat`;
7//! this crate owns only declaration resolution and slot preparation.
8//! Each [`Wire`] carries the adapter's own description in [`Wire::kind`].
9//! Reusable value and boxed-allocation leaves live in `prebindgen-conversions`.
10#![warn(missing_docs)]
11use prebindgen_flat::{
12    same_type, type_key, ComponentSegment, Condition, Flat, InputPlan, OutputPlan, Result, Wire,
13};

18/// A recipe for converting boundary inputs into one source argument.
19///
20/// Binding expands the recipe into wires and Rust conversion code.
21#[derive(Clone, Debug)]
22pub enum Input {
23    /// Convert the occurrence type itself with [`Codec::input_leaf`].
24    Leaf,
25    /// Construct a named record from children in field order; an empty list builds unit.
26    Fields(Vec<Input>),
27    /// Call a source function after preparing its arguments with the child recipes.
28    Constructor {
29        /// Unqualified source function name; its result must match the occurrence type.
30        function: String,
31        /// Recipes in the constructor's parameter order.
32        args: Vec<Input>,
33    },
34    /// Choose an owned input using an `i32` arm index followed by all arms' wires.
35    /// Inactive arms are neither validated nor acquired; indices are zero-based.
36    Selector(Vec<Input>),
37}
38impl Input {
39    /// Construct a record from field recipes, or unit from an empty list.
40    ///
41    /// `fields` holds one recipe per field of the source record, in declaration
42    /// order. Binding verifies the count against the record.
43    pub fn fields(fields: impl IntoIterator<Item = Input>) -> Self {
44        Self::Fields(fields.into_iter().collect())
45    }
46    /// Call a named source constructor with the supplied argument recipes.
47    ///
48    /// Binding checks its result type and argument count. Argument validation and
49    /// ownership acquisition precede the source call.
50    ///
51    /// # Arguments
52    ///
53    /// * `name` — unqualified name of a source function whose result type is the
54    ///   input's type.
55    /// * `args` — one recipe per parameter of that function, in order.
56    pub fn constructor(name: impl Into<String>, args: impl IntoIterator<Item = Input>) -> Self {
57        Self::Constructor {
58            function: name.into(),
59            args: args.into_iter().collect(),
60        }
61    }
62    /// Choose between owned representations of the same input type.
63    ///
64    /// `arms` holds one recipe per alternative; each must produce the input's
65    /// type. The caller passes the zero-based index of the chosen alternative in an
66    /// extra wire ahead of all arms' wires.
67    ///
68    /// Requires at least one arm. Each arm retains its own wires; this does not
69    /// merge equivalent wire parameters or generate destination-language overloads.
70    pub fn selector(arms: impl IntoIterator<Item = Input>) -> Self {
71        Self::Selector(arms.into_iter().collect())
72    }
73}
74/// A result conversion recipe, interpreted together with its source type and codec input data.
75///
76/// Field projections are observations rather than moves. Adapters decide how to
77/// copy or otherwise expose a foreign value observed through a shared reference.
78#[derive(Clone, Debug)]
79pub enum Output {
80    /// Convert the occurrence type itself with [`Codec::output_leaf`].
81    Leaf,
82    /// Observe named fields in declaration order; an empty list represents unit.
83    Fields(Vec<Output>),
84    /// A shared-borrow accessor, followed by the output conversion of its result.
85    Getter {
86        /// Source getter taking exactly one shared reference to the current type.
87        function: String,
88        /// Conversion of the getter's owned result; borrowed results are unsupported.
89        output: Box<Output>,
90    },
91    /// Produce several views of one value. At most one child may consume it.
92    /// Observations run before consumption, while wire order follows this list.
93    Combine(Vec<Output>),
94}
95impl Output {
96    /// Observe record fields in declaration order; use an empty list for unit.
97    ///
98    /// `fields` holds one recipe per field of the source record, in declaration order.
99    pub fn fields(fields: impl IntoIterator<Item = Output>) -> Self {
100        Self::Fields(fields.into_iter().collect())
101    }
102    /// Observe a value through a source getter and convert the getter's result.
103    ///
104    /// The getter must take exactly `&T` for the current type `T`, and return an
105    /// owned value. Binding rejects mismatched signatures.
106    ///
107    /// # Arguments
108    ///
109    /// * `name` — unqualified name of the source getter.
110    /// * `output` — recipe for the getter's result.
111    pub fn getter(name: impl Into<String>, output: Output) -> Self {
112        Self::Getter {
113            function: name.into(),
114            output: Box::new(output),
115        }
116    }
117    /// Produce multiple outputs from one value.
118    ///
119    /// `parts` holds one recipe per view of the value; their wires appear in this
120    /// order. Binding rejects an empty list or more than one consuming part. The
121    /// framework uses [`Codec::consumes`] to order observations before a move.
122    pub fn combine(parts: impl IntoIterator<Item = Output>) -> Self {
123        Self::Combine(parts.into_iter().collect())
124    }
125    fn consumes<C: Codec>(&self, ty: &Type, extra: &C::Extra, codec: &C) -> bool {
126        match self {
127            Self::Leaf => codec.consumes(ty, extra),
128            Self::Combine(p) => p.iter().any(|p| p.consumes(ty, extra, codec)),
129            _ => false,
130        }
131    }
132}

134/// Language-specific leaf conversions. Recursive composition stays in this declaration compiler.
135///
136/// A leaf is a source type the adapter converts itself: a value copied across
137/// the boundary or an object kept in Rust. `prebindgen-conversions` provides helpers
138/// for both representations. The
139/// adapter chooses by type; a recipe only says where the leaves are.
140///
141/// Input leaves must validate without consuming in `validation`, then
142/// acquire ownership without failure or source calls in `acquire`. Each acquired
143/// value needs an ordinary Rust owner so later failures drop it. `statements`
144/// can construct arguments from those owners. Every input wire must retain
145/// the supplied activation condition. Output leaves keep ownership in Rust
146/// locals until their infallible `commit` runs after successful delivery.
147///
148/// # Implementing a codec
149///
150/// This minimal codec uses Rust values as its carriers. It illustrates converter
151/// construction without a foreign ABI or object storage. A language adapter
152/// additionally writes wrappers and chooses native and destination-language
153/// carrier types for its wire kinds.
154///
155/// ```
156/// use prebindgen_flat::{Result, Condition, InputPlan, OutputPlan, validate_bindings};
157/// use prebindgen_conversions::value;
158/// use prebindgen_declarations::{
159///     Codec, Declarations,
160///     Bindings, fun, expand_param, expand_return, bind_input, Input,
161/// };
162/// use proc_macro2::TokenStream;
163/// use quote::quote;
164/// use syn::{Ident, Type};
165///
166/// struct RustCarriers;
167/// impl Codec for RustCarriers {
168///     type Extra = ();
169///     type Wire = Option<value::Value>; // `None` marks a selector.
170///
171///     fn selector(&self, _: &()) -> Self::Wire { None }
172///     fn consumes(&self, _: &Type, _: &()) -> bool { false }
173///     fn input_leaf(
174///         &self, ty: &Type, _: &(), wire: Ident, var: Ident, active: &Condition,
175///     ) -> Result<InputPlan<Self::Wire>> {
176///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
177///         let decode = quote!(#wire.clone());
178///         value::input(ty, wire, var, active, decode, Some(kind))
179///     }
180///     fn output_leaf(
181///         &self, ty: &Type, _: &(), wire: Ident, v: TokenStream, borrowed: bool,
182///     ) -> Result<OutputPlan<Self::Wire>> {
183///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
184///         value::output(ty, wire, v, borrowed, |v| v, Some(kind))
185///     }
186/// }
187///
188/// let source = syn::parse_file(r#"
189///     pub fn from_i32(value: i32) -> i64 { value as i64 }
190///     pub fn to_i32(value: &i64) -> i32 { *value as i32 }
191///     pub fn echo(value: i64) -> i64 { value }
192/// "#)?;
193/// let mut flat = Bindings::<Option<value::Value>>::parse(
194///     source.items.into_iter().map(|item| (item, Default::default())),
195/// )?;
196/// let defaults = Declarations::new(&flat, vec![
197///     expand_param!(i64).variant(fun!(from_i32)).variant_self().into(),
198///     expand_return!(i64).field(fun!(to_i32)).field_self().into(),
199/// ])?;
200/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
201/// let function = flat.function("echo")?;
202/// // Input: selector, constructor argument, direct value. Output: two values.
203/// assert_eq!(function.params[0].ty.slot.as_ref().unwrap().wires.len(), 3);
204/// assert_eq!(function.ret.slot.as_ref().unwrap().wires.len(), 2);
205/// // Replace just this parameter's converter; later default filling preserves it.
206/// bind_input(&mut flat, &RustCarriers, "echo", "value", &Input::Leaf, &())?;
207/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
208/// validate_bindings(&flat)?;
209/// let input = flat.function("echo")?.params[0].ty.slot.as_ref().unwrap();
210/// assert_eq!(input.wires.len(), 1);
211/// assert!(matches!(input.wires[0].kind, Some(value::Value::Scalar(_))));
212/// assert!(input.resources.is_empty()); // No foreign ownership in this codec.
213/// # Ok::<(), Box<dyn std::error::Error>>(())
214/// ```
215pub trait Codec {
216    /// Input data borrowed while building one occurrence's conversion; use `()` if unused.
217    /// The same value is passed through nested conversions. It is not stored in Flat;
218    /// hooks retain only the data needed for emission in their code and wires.
219    type Extra;
220    /// The adapter's description of one wire, stored in [`Wire::kind`], such as
221    /// its carrier type or a destination type name.
222    type Wire: Clone + std::fmt::Debug;
223    /// Describe the `i32` arm index wire that the declaration compiler creates for [`Input::Selector`].
224    ///
225    /// `extra` is the occurrence data of the conversion that contains the selector.
226    fn selector(&self, extra: &Self::Extra) -> Self::Wire;
227    /// Whether a leaf output consumes the current source value.
228    ///
229    /// This must agree with [`Self::output_leaf`]. It governs move ordering and
230    /// double-consumption rejection in [`Output::Combine`], not input ownership.
231    /// A non-consuming leaf receives a shared reference with `borrowed` set.
232    ///
233    /// # Arguments
234    ///
235    /// * `ty` — owned source type of the value being delivered.
236    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`].
237    fn consumes(&self, ty: &Type, extra: &Self::Extra) -> bool;
238    /// Prepare a leaf input of source type `ty` using occurrence data `extra`.
239    ///
240    /// All fallible checks belong in [`InputPlan::validation`]; do not
241    /// consume values there. Return a generation error when this type or access
242    /// mode is unsupported.
243    ///
244    /// # Arguments
245    ///
246    /// * `ty` — the source parameter type at this position, as written: `T`,
247    ///   `&T` or `&mut T`.
248    /// * `extra` — occurrence data of the conversion, as passed to [`bind_input`]
249    ///   or [`Declarations::fill`].
250    /// * `wire` — fresh identifier for the boundary parameter this leaf adds.
251    ///   A leaf that needs more wires or locals derives their names from it.
252    /// * `var` — fresh identifier for the Rust local that holds the converted value.
253    ///   [`InputPlan::argument`] normally refers to it.
254    /// * `active` — the selector arms under which this leaf is used. Copy it to every
255    ///   returned wire and resource, and guard validation and acquisition with it.
256    fn input_leaf(
257        &self,
258        ty: &Type,
259        extra: &Self::Extra,
260        wire: syn::Ident,
261        var: syn::Ident,
262        active: &Condition,
263    ) -> Result<InputPlan<Self::Wire>>;
264    /// Complete a composed parameter plan before it is stored in a slot.
265    /// Called once per bound parameter, including records and selectors.
266    fn finish_input(&self, _plan: &mut InputPlan<Self::Wire>, _extra: &Self::Extra) -> Result<()> {
267        Ok(())
268    }
269    /// Complete a composed result plan, including unit and multi-wire results.
270    /// Install call-level delivery here, after all output expressions are known.
271    fn finish_output(
272        &self,
273        _plan: &mut OutputPlan<Self::Wire>,
274        _extra: &Self::Extra,
275    ) -> Result<()> {
276        Ok(())
277    }
278    /// Prepare delivery of the Rust expression `value` as one or more wires.
279    ///
280    /// Keep unpublished allocations under Rust ownership and transfer them only in
281    /// [`OutputPlan::commit`]. Return a generation error for unsupported
282    /// conversions. Install final delivery in [`Codec::finish_output`], rather
283    /// than on nested leaf plans; recursive composition rejects leaf delivery.
284    ///
285    /// # Arguments
286    ///
287    /// * `ty` — the owned source type of the value.
288    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`]
289    ///   or [`Declarations::fill`].
290    /// * `wire` — fresh identifier for the delivered value. A leaf that needs more
291    ///   wires or locals derives their names from it.
292    /// * `value` — Rust expression producing the value to deliver.
293    /// * `borrowed` — `true` when `value` is a shared reference to `ty`, such as a
294    ///   record field observed in place; do not move out of it. `false` when
295    ///   `value` is owned.
296    fn output_leaf(
297        &self,
298        ty: &Type,
299        extra: &Self::Extra,
300        wire: syn::Ident,
301        value: TokenStream,
302        borrowed: bool,
303    ) -> Result<OutputPlan<Self::Wire>>;
304}
305/// A flat model whose slots contain only conversion data for wrapper emission.
306///
307/// Unbound source helpers have `None` slots. Bound functions need `Some` for their
308/// result and every parameter; [`prebindgen_flat::validate_bindings`] checks this before emission.
309/// Record fields retain only names and source types. Recipes and [`Codec::Extra`]
310/// are construction inputs, not stored in this model.
311///
312/// Output conversion data cannot be attached to an input position:
313/// ```compile_fail
314/// use prebindgen_declarations::Bindings;
315/// fn swap(flat: &mut Bindings<()>) {
316///     let function = flat.function_mut("foo").unwrap();
317///     function.params[0].ty.slot = function.ret.slot.clone();
318/// }
319/// ```
320/// Nor can a recipe be stored in a slot:
321/// ```compile_fail
322/// use prebindgen_declarations::{Bindings, Input};
323/// fn configure(flat: &mut Bindings<()>) {
324///     flat.function_mut("foo").unwrap().params[0].ty.slot = Some(Input::Leaf);
325/// }
326/// ```
327pub type Bindings<W> = Flat<InputPlan<W>, OutputPlan<W>>;
328
329/// Build and attach a converter to a named function parameter.
330///
331/// Resolves the occurrence's source type from `flat`, checks recipe compatibility,
332/// and recursively composes wires and code. Nested conversions use the supplied
333/// `extra`, not data from referenced constructors or getters. Recipes and `extra`
334/// are borrowed only for this call. A failure leaves the existing slot unchanged.
335/// Rebinding an occurrence replaces its converter without changing other slots.
336/// Generated names are scoped by parameter position, so separate calls cannot
337/// collide within the wrapper. Keep the function signature fixed after binding.
338/// Check completeness with [`prebindgen_flat::validate_bindings`].
339///
340/// # Arguments
341///
342/// * `flat` — the model holding the function; the parameter's slot is replaced.
343/// * `codec` — the adapter's leaf conversions.
344/// * `function` — unqualified name of the source function.
345/// * `parameter` — name of the parameter, as written in the source.
346/// * `recipe` — how to build the argument from boundary inputs.
347/// * `extra` — occurrence data passed to every codec call for this conversion.
348///
349/// Input binding rejects output recipes:
350/// ```compile_fail
351/// use prebindgen_declarations::{bind_input, Bindings, Codec, Output};
352/// fn wrong<C: Codec>(flat: &mut Bindings<C::Wire>, codec: &C, extra: &C::Extra) {
353///     bind_input(flat, codec, "foo", "x", &Output::Leaf, extra).unwrap();
354/// }
355/// ```
356pub fn bind_input<C: Codec>(
357    flat: &mut Bindings<C::Wire>,
358    codec: &C,
359    function: &str,
360    parameter: &str,
361    recipe: &Input,
362    extra: &C::Extra,
363) -> Result<()> {
364    let f = flat.function(function)?;
365    let index = f
366        .params
367        .iter()
368        .position(|p| p.name == parameter)
369        .ok_or_else(|| format!("`{function}`: unknown parameter `{parameter}`"))?;
370    let mut conversion = Compiler::new(flat, codec, format!("p{index}")).input(
371        &f.params[index].ty.ty,
372        recipe,
373        extra,
374        &Condition::default(),
375    )?;
376    codec.finish_input(&mut conversion, extra)?;
377    flat.function_mut(function)?.params[index].ty.slot = Some(conversion);
378    Ok(())
379}

381/// Build and attach a converter to a function result, replacing any existing one.
382///
383/// Uses the same construction and failure rules as [`bind_input`]. Output code
384/// uses [`prebindgen_flat::source_result_expr`] for the source call's result local.
385/// The recipe is checked for compatible types and signatures, and for repeated
386/// consumption of the same value. Unit results use `Output::fields([])`.
387///
388/// # Arguments
389///
390/// * `flat` — the model holding the function; the result's slot is replaced.
391/// * `codec` — the adapter's leaf conversions.
392/// * `function` — unqualified name of the source function.
393/// * `recipe` — how to deliver the result as boundary outputs.
394/// * `extra` — occurrence data passed to every codec call for this conversion.
395///
396/// Output binding rejects input recipes:
397/// ```compile_fail
398/// use prebindgen_declarations::{bind_output, Bindings, Codec, Input};
399/// fn wrong<C: Codec>(flat: &mut Bindings<C::Wire>, codec: &C, extra: &C::Extra) {
400///     bind_output(flat, codec, "foo", &Input::Leaf, extra).unwrap();
401/// }
402/// ```
403pub fn bind_output<C: Codec>(
404    flat: &mut Bindings<C::Wire>,
405    codec: &C,
406    function: &str,
407    recipe: &Output,
408    extra: &C::Extra,
409) -> Result<()> {
410    let mut conversion = Compiler::new(flat, codec, "ret".into()).output(
411        &flat.function(function)?.ret.ty,
412        recipe,
413        extra,
414        prebindgen_flat::source_result_expr(),
415        false,
416    )?;
417    codec.finish_output(&mut conversion, extra)?;
418    flat.function_mut(function)?.ret.slot = Some(conversion);
419    Ok(())
420}

422struct Compiler<'a, C: Codec> {
423    flat: &'a Bindings<C::Wire>,
424    codec: &'a C,
425    prefix: String,
426    next: std::cell::Cell<usize>,
427}
428impl<'a, C: Codec> Compiler<'a, C> {
429    fn new(flat: &'a Bindings<C::Wire>, codec: &'a C, prefix: String) -> Self {
430        Self {
431            flat,
432            codec,
433            prefix,
434            next: std::cell::Cell::new(0),
435        }
436    }
437    fn name(&self) -> syn::Ident {
438        let next = self.next.get();
439        let n = format_ident!("{}_w{next}", self.prefix);
440        self.next.set(next + 1);
441        n
442    }
443    fn input(
444        &self,
445        ty: &Type,
446        recipe: &Input,
447        extra: &C::Extra,
448        active: &Condition,
449    ) -> Result<InputPlan<C::Wire>> {
450        let var = self.name();
451        let mut wires = Vec::new();
452        let mut validation = TokenStream::new();
453        let mut acquire = TokenStream::new();
454        let mut resources = Vec::new();
455        let mut acknowledgement = None;
456        let mut statements = TokenStream::new();
457        let argument = quote!(#var);
458        match recipe {
459            Input::Leaf => {
460                let w = self.name();
461                return self.codec.input_leaf(ty, extra, w, var, active);
462            }
463            Input::Fields(children) => {
464                if type_key(ty) == "()" && children.is_empty() {
465                    return Ok(InputPlan {
466                        acknowledgement,
467                        wires,
468                        validation,
469                        acquire,
470                        resources,
471                        statements,
472                        argument: quote!(()),
473                    });
474                }
475                let r = self.flat.record(ty)?;
476                if r.fields.len() != children.len() {
477                    return Err(format!(
478                        "`{}`: expected {} field converters, got {}",
479                        type_key(ty),
480                        r.fields.len(),
481                        children.len()
482                    ));
483                }
484                let mut fields = Vec::new();
485                for (field, child) in r.fields.iter().zip(children) {
486                    let mut p = self.input(&field.ty, child, extra, active)?;
487                    prefix(
488                        &mut p.wires,
489                        ComponentSegment::Field(field.name.to_string()),
490                    );
491                    validation.extend(p.validation);
492                    acquire.extend(p.acquire);
493                    resources.extend(p.resources);
494                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
495                    statements.extend(p.statements);
496                    wires.extend(p.wires);
497                    let name = &field.name;
498                    let arg = p.argument;
499                    fields.push(quote!(#name:#arg));
500                }
501                statements.extend(quote!(let #var:#ty=#ty{#(#fields),*};));
502            }
503            Input::Constructor { function, args } => {
504                let f = self.flat.function(function)?;
505                if !same_type(&f.ret.ty, ty) {
506                    return Err(format!(
507                        "constructor `{function}` returns `{}`, expected `{}`",
508                        type_key(&f.ret.ty),
509                        type_key(ty)
510                    ));
511                }
512                if f.params.len() != args.len() {
513                    return Err(format!("constructor `{function}`: wrong argument count"));
514                }
515                let mut call_args = Vec::new();
516                for (param, child) in f.params.iter().zip(args) {
517                    let mut p = self.input(&param.ty.ty, child, extra, active)?;
518                    prefix(
519                        &mut p.wires,
520                        ComponentSegment::Argument(param.name.to_string()),
521                    );
522                    validation.extend(p.validation);
523                    acquire.extend(p.acquire);
524                    resources.extend(p.resources);
525                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
526                    statements.extend(p.statements);
527                    wires.extend(p.wires);
528                    call_args.push(p.argument);
529                }
530                let path = &f.path;
531                statements.extend(quote!(let #var:#ty=#path(#(#call_args),*);));
532            }
533            Input::Selector(arms) => {
534                if arms.is_empty() {
535                    return Err("an input selector needs at least one alternative".into());
536                }
537                if matches!(ty, Type::Reference(_)) {
538                    return Err("selector results must be owned in this subset".into());
539                }
540                let tag = self.name();
541                wires.push(Wire {
542                    component: prebindgen_flat::ComponentPath(vec![ComponentSegment::Selector]),
543                    name: tag.clone(),
544                    kind: self.codec.selector(extra),
545                    active: active.clone(),
546                });
547                let count = arms.len() as i32;
548                let invalid = active.and(quote!(!(0..#count).contains(&#tag)));
549                validation.extend(quote!(if #invalid {return Err("invalid selector".into());}));
550                let mut cases = Vec::new();
551                for (i, arm) in arms.iter().enumerate() {
552                    let mut p = self.input(ty, arm, extra, &active.arm(tag.clone(), i as i32))?;
553                    prefix(&mut p.wires, ComponentSegment::Alternative(i));
554                    validation.extend(p.validation);
555                    acquire.extend(p.acquire);
556                    resources.extend(p.resources);
557                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
558                    let body = p.statements;
559                    let value = p.argument;
560                    wires.extend(p.wires);
561                    let i = i as i32;
562                    cases.push(quote!(#i=>{#body #value}));
563                }
564                statements = quote!(let #var:#ty=match #tag{#(#cases,)* _=>unreachable!("validated selector")};);
565            }
566        }
567        Ok(InputPlan {
568            acknowledgement,
569            wires,
570            validation,
571            acquire,
572            resources,
573            statements,
574            argument,
575        })
576    }
577    fn output(
578        &self,
579        ty: &Type,
580        recipe: &Output,
581        extra: &C::Extra,
582        value: TokenStream,
583        borrowed: bool,
584    ) -> Result<OutputPlan<C::Wire>> {
585        let mut out = OutputPlan {
586            delivery: Default::default(),
587            commit: TokenStream::new(),
588            wires: Vec::new(),
589            statements: TokenStream::new(),
590            values: Vec::new(),
591        };
592        match recipe {
593            Output::Leaf => {
594                let w = self.name();
595                return self.codec.output_leaf(ty, extra, w, value, borrowed);
596            }
597            Output::Fields(children) => {
598                if type_key(ty) == "()" && children.is_empty() {
599                    return Ok(out);
600                }
601                let r = self.flat.record(ty)?;
602                if r.fields.len() != children.len() {
603                    return Err("wrong output field count".into());
604                }
605                for (field, child) in r.fields.iter().zip(children) {
606                    let name = &field.name;
607                    let mut p =
608                        self.output(&field.ty, child, extra, quote!(&(#value).#name), true)?;
609                    prefix(&mut p.wires, ComponentSegment::Field(name.to_string()));
610                    append(&mut out, p)?;
611                }
612            }
613            Output::Getter { function, output } => {
614                let f = self.flat.function(function)?;
615                let expected: Type = syn::parse_quote!(&#ty);
616                if f.params.len() != 1 || !same_type(&f.params[0].ty.ty, &expected) {
617                    return Err(format!(
618                        "getter `{function}` must take exactly `&{}`",
619                        type_key(ty)
620                    ));
621                }
622                if matches!(f.ret.ty, Type::Reference(_)) {
623                    return Err("borrowed getter results are outside this subset".into());
624                }
625                let v = self.name();
626                let path = &f.path;
627                let arg = if borrowed { value } else { quote!(&#value) };
628                out.statements = quote!(let #v=#path(#arg););
629                let mut child = self.output(&f.ret.ty, output, extra, quote!(#v), false)?;
630                prefix(&mut child.wires, ComponentSegment::Getter(function.clone()));
631                append(&mut out, child)?;
632            }
633            Output::Combine(parts) => {
634                if parts.is_empty() {
635                    return Err("an output combination needs at least one part".into());
636                }
637                if parts
638                    .iter()
639                    .filter(|p| p.consumes(ty, extra, self.codec))
640                    .count()
641                    > 1
642                {
643                    return Err("a combination cannot consume the same value twice".into());
644                }
645                let mut bodies = Vec::new();
646                for (index, child) in parts.iter().enumerate() {
647                    let consuming = child.consumes(ty, extra, self.codec);
648                    let arg = if borrowed || consuming {
649                        value.clone()
650                    } else {
651                        quote!(&#value)
652                    };
653                    let mut p = self.output(ty, child, extra, arg, borrowed || !consuming)?;
654                    prefix(&mut p.wires, ComponentSegment::View(index));
655                    bodies.push((consuming, std::mem::take(&mut p.statements)));
656                    append(&mut out, p)?;
657                }
658                // Wire order follows the declaration. Observations execute before the move.
659                bodies.sort_by_key(|(consumes, _)| *consumes);
660                for (_, body) in bodies {
661                    out.statements.extend(body);
662                }
663            }
664        }
665        Ok(out)
666    }
667}
668fn append<W>(to: &mut OutputPlan<W>, from: OutputPlan<W>) -> Result<()> {
669    if !from.delivery.statements.is_empty()
670        || from.delivery.result.to_string() != quote!(()).to_string()
671    {
672        return Err("composed output delivery must be installed by Codec::finish_output".into());
673    }
674    to.commit.extend(from.commit);
675    to.wires.extend(from.wires);
676    to.statements.extend(from.statements);
677    to.values.extend(from.values);
678    Ok(())
679}

705fn merge_acknowledgement(
706    target: &mut Option<TokenStream>,
707    incoming: Option<TokenStream>,
708) -> Result<()> {
709    if let Some(code) = incoming {
710        if let Some(previous) = target {
711            if previous.to_string() != code.to_string() {
712                return Err("conflicting input acknowledgements".into());
713            }
714        } else {
715            *target = Some(code);
716        }
717    }
718    Ok(())
