1//! Declaration language and composition for preparing conversion slots.
2//!
3//! [`Declarations::fill`], [`bind_input`] and [`bind_output`] build conversions
4//! directly in a [`Bindings`] model. Recipes and codec input data are used only
5//! during construction; slots retain wires and code needed by wrapper writers.
6//! Generation is owned by the language generator;
7//! this crate owns only declaration resolution and slot preparation.
8//! Each [`Wire`] carries the adapter's own description in [`Wire::kind`].
9//! Reusable value and boxed-allocation leaves live in `prebindgen-conversions`.
10#![warn(missing_docs)]

21/// A recipe for converting boundary inputs into one source argument.
22///
23/// Binding expands the recipe into wires and Rust conversion code.
24#[derive(Clone, Debug)]
25pub enum Input {
26    /// Convert the occurrence type itself with [`Codec::input_leaf`].
27    Leaf,
28    /// Construct a named record from children in field order; an empty list builds unit.
29    Fields(Vec<Input>),
30    /// Call a source function after preparing its arguments with the child recipes.
31    Constructor {
32        /// Unqualified source function name; its result must match the occurrence type.
33        function: String,
34        /// Recipes in the constructor's parameter order.
35        args: Vec<Input>,
36    },
37    /// Choose an owned input using an `i32` arm index followed by all arms' wires.
38    /// Inactive arms are neither validated nor acquired; indices are zero-based.
39    Selector(Vec<Input>),
40}
41impl Input {
42    /// Construct a record from field recipes, or unit from an empty list.
43    ///
44    /// `fields` holds one recipe per field of the source record, in declaration
45    /// order. Binding verifies the count against the record.
46    pub fn fields(fields: impl IntoIterator<Item = Input>) -> Self {
47        Self::Fields(fields.into_iter().collect())
48    }
49    /// Call a named source constructor with the supplied argument recipes.
50    ///
51    /// Binding checks its result type and argument count. Argument validation and
52    /// ownership acquisition precede the source call.
53    ///
54    /// # Arguments
55    ///
56    /// * `name` — unqualified name of a source function whose result type is the
57    ///   input's type.
58    /// * `args` — one recipe per parameter of that function, in order.
59    pub fn constructor(name: impl Into<String>, args: impl IntoIterator<Item = Input>) -> Self {
60        Self::Constructor {
61            function: name.into(),
62            args: args.into_iter().collect(),
63        }
64    }
65    /// Choose between owned representations of the same input type.
66    ///
67    /// `arms` holds one recipe per alternative; each must produce the input's
68    /// type. The caller passes the zero-based index of the chosen alternative in an
69    /// extra wire ahead of all arms' wires.
70    ///
71    /// Requires at least one arm. Each arm retains its own wires; this does not
72    /// merge equivalent wire parameters or generate destination-language overloads.
73    pub fn selector(arms: impl IntoIterator<Item = Input>) -> Self {
74        Self::Selector(arms.into_iter().collect())
75    }
76}
77/// A result conversion recipe, interpreted together with its source type and codec input data.
78///
79/// Field projections are observations rather than moves. Adapters decide how to
80/// copy or otherwise expose a foreign value observed through a shared reference.
81#[derive(Clone, Debug)]
82pub enum Output {
83    /// Convert the occurrence type itself with [`Codec::output_leaf`].
84    Leaf,
85    /// Observe named fields in declaration order; an empty list represents unit.
86    Fields(Vec<Output>),
87    /// A shared-borrow accessor, followed by the output conversion of its result.
88    Getter {
89        /// Source getter taking exactly one shared reference to the current type.
90        function: String,
91        /// Conversion of the getter's owned result; borrowed results are unsupported.
92        output: Box<Output>,
93    },
94    /// Produce several views of one value. At most one child may consume it.
95    /// Observations run before consumption, while wire order follows this list.
96    Combine(Vec<Output>),
97}
98impl Output {
99    /// Observe record fields in declaration order; use an empty list for unit.
100    ///
101    /// `fields` holds one recipe per field of the source record, in declaration order.
102    pub fn fields(fields: impl IntoIterator<Item = Output>) -> Self {
103        Self::Fields(fields.into_iter().collect())
104    }
105    /// Observe a value through a source getter and convert the getter's result.
106    ///
107    /// The getter must take exactly `&T` for the current type `T`, and return an
108    /// owned value. Binding rejects mismatched signatures.
109    ///
110    /// # Arguments
111    ///
112    /// * `name` — unqualified name of the source getter.
113    /// * `output` — recipe for the getter's result.
114    pub fn getter(name: impl Into<String>, output: Output) -> Self {
115        Self::Getter {
116            function: name.into(),
117            output: Box::new(output),
118        }
119    }
120    /// Produce multiple outputs from one value.
121    ///
122    /// `parts` holds one recipe per view of the value; their wires appear in this
123    /// order. Binding rejects an empty list or more than one consuming part. The
124    /// framework uses [`Codec::consumes`] to order observations before a move.
125    pub fn combine(parts: impl IntoIterator<Item = Output>) -> Self {
126        Self::Combine(parts.into_iter().collect())
127    }
128    fn consumes<C: Codec>(&self, ty: &Type, extra: &C::Extra, codec: &C) -> bool {
129        match self {
130            Self::Leaf => codec.consumes(ty, extra),
131            Self::Combine(p) => p.iter().any(|p| p.consumes(ty, extra, codec)),
132            _ => false,
133        }
134    }
135}

137/// Language-specific leaf conversions. Recursive composition stays in this declaration compiler.
138///
139/// A leaf is a source type the adapter converts itself: a value copied across
140/// the boundary or an object kept in Rust. `prebindgen-conversions` provides helpers
141/// for both representations. The
142/// adapter chooses by type; a recipe only says where the leaves are.
143///
144/// Input leaves must validate without consuming in `validation`, then
145/// acquire ownership without failure or source calls in `acquire`. Each acquired
146/// value needs an ordinary Rust owner so later failures drop it. `statements`
147/// can construct arguments from those owners. Every input wire must retain
148/// the supplied activation condition. Output leaves keep ownership in Rust
149/// locals until their infallible `commit` runs after successful delivery.
150///
151/// # Implementing a codec
152///
153/// This minimal codec uses Rust values as its carriers. It illustrates converter
154/// construction without a foreign ABI or object storage. A language adapter
155/// additionally writes wrappers and chooses native and destination-language
156/// carrier types for its wire kinds.
157///
158/// ```
159/// use prebindgen_flat::{Result, validate_bindings};
160/// use prebindgen_conversions::{InputPlan, OutputPlan};
161/// use prebindgen_conversions::value;
162/// use prebindgen_declarations::{
163///     Codec, Declarations,
164///     Bindings, fun, expand_param, expand_return, bind_input, Input,
165/// };
166/// use proc_macro2::TokenStream;
167/// use quote::quote;
168/// use syn::{Ident, Type};
169///
170/// use prebindgen_declarations::Condition;
171/// fn carrier(kind: value::Value) -> Type {
172///     match kind {
173///         value::Value::Scalar(p) => p.rust_type(),
174///         value::Value::Text => syn::parse_quote!(String),
175///         value::Value::Bytes => syn::parse_quote!(Vec<u8>),
176///     }
177/// }
178/// struct RustCarriers;
179/// impl Codec for RustCarriers {
180///     type Extra = ();
181///     type Context = ();
182///     type Input = InputPlan<Self::Wire>;
183///     type Output = OutputPlan<Self::Wire>;
184///     type Wire = Type; // The Rust carrier type is the complete wire description.
185///
186///     fn selector(&self, _: &()) -> Self::Wire { syn::parse_quote!(i32) }
187///     fn consumes(&self, _: &Type, _: &()) -> bool { false }
188///     fn input_leaf(
189///         &self, ty: &Type, _: &(), wire: Ident, var: Ident, active: &Condition,
190///     ) -> Result<InputPlan<Self::Wire, Self::Context>> {
191///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
192///         let decode = quote!(#wire.clone());
193///         value::input(ty, wire, var, &active.rust(), decode, carrier(kind))
194///     }
195///     fn output_leaf(
196///         &self, ty: &Type, _: &(), wire: Ident, v: TokenStream, borrowed: bool,
197///     ) -> Result<OutputPlan<Self::Wire, Self::Context>> {
198///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
199///         value::output(ty, wire, v, borrowed, |v| v, carrier(kind))
200///     }
201/// }
202///
203/// let source = syn::parse_file(r#"
204///     pub fn from_i32(value: i32) -> i64 { value as i64 }
205///     pub fn to_i32(value: &i64) -> i32 { *value as i32 }
206///     pub fn echo(value: i64) -> i64 { value }
207/// "#)?;
208/// let mut flat = Bindings::<Type>::parse(
209///     source.items.into_iter().map(|item| (item, Default::default())),
210/// )?;
211/// let defaults = Declarations::new(&flat, vec![
212///     expand_param!(i64).variant(fun!(from_i32)).variant_self().into(),
213///     expand_return!(i64).field(fun!(to_i32)).field_self().into(),
214/// ])?;
215/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
216/// let function = flat.function("echo")?;
217/// // Input: selector, constructor argument, direct value. Output: two values.
218/// assert_eq!(function.params[0].ty.slot.as_ref().unwrap().wires.len(), 3);
219/// assert_eq!(function.ret.slot.as_ref().unwrap().wires.len(), 2);
220/// // Replace just this parameter's converter; later default filling preserves it.
221/// bind_input(&mut flat, &RustCarriers, "echo", "value", &Input::Leaf, &())?;
222/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
223/// validate_bindings(&flat)?;
224/// let input = flat.function("echo")?.params[0].ty.slot.as_ref().unwrap();
225/// assert_eq!(input.wires.len(), 1);
226/// assert_eq!(prebindgen_flat::type_key(&input.wires[0].kind), "i64");
227/// assert!(input.resources.is_empty()); // No foreign ownership in this codec.
228/// # Ok::<(), Box<dyn std::error::Error>>(())
229/// ```
230pub trait Codec {
231    /// Input data borrowed while building one occurrence's conversion; use `()` if unused.
232    /// The same value is passed through nested conversions. It is not stored in Flat;
233    /// hooks retain only the data needed for emission in their code and wires.
234    type Extra;
235    /// Context supplied by the language generator when emitting stored templates.
236    type Context: prebindgen_conversions::TemplateContext + 'static;
237    /// Final self-contained input converter installed after preparation.
238    type Input: prebindgen_flat::InputConversion + From<InputPlan<Self::Wire, Self::Context>>;
239    /// Final self-contained output converter installed after preparation.
240    type Output: prebindgen_flat::OutputConversion + From<OutputPlan<Self::Wire, Self::Context>>;
241    /// Preserve a logical path in private preparation metadata when the codec needs it.
242    /// The final converter decides how to use this mapping; Flat never interprets it.
243    /// Called from the innermost component outward; prepend each segment to its path.
244    fn prefix_component(&self, _wire: &mut Self::Wire, _segment: ComponentSegment) {}
245
246    /// Private preparation metadata for a leaf wire. Final slot conversion may
247    /// replace it with a transport-only description before exposing wires to a writer.
248    type Wire: prebindgen_flat::WireKind + Clone + std::fmt::Debug;
249    /// Describe the `i32` arm index wire that the declaration compiler creates for [`Input::Selector`].
250    ///
251    /// `extra` is the occurrence data of the conversion that contains the selector.
252    fn selector(&self, extra: &Self::Extra) -> Self::Wire;
253    /// Whether a leaf output consumes the current source value.
254    ///
255    /// This must agree with [`Self::output_leaf`]. It governs move ordering and
256    /// double-consumption rejection in [`Output::Combine`], not input ownership.
257    /// A non-consuming leaf receives a shared reference with `borrowed` set.
258    ///
259    /// # Arguments
260    ///
261    /// * `ty` — owned source type of the value being delivered.
262    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`].
263    fn consumes(&self, ty: &Type, extra: &Self::Extra) -> bool;
264    /// Prepare a leaf input of source type `ty` using occurrence data `extra`.
265    ///
266    /// All fallible checks belong in [`InputPlan::validation`]; do not
267    /// consume values there. Return a generation error when this type or access
268    /// mode is unsupported.
269    ///
270    /// # Arguments
271    ///
272    /// * `ty` — the source parameter type at this position, as written: `T`,
273    ///   `&T` or `&mut T`.
274    /// * `extra` — occurrence data of the conversion, as passed to [`bind_input`]
275    ///   or [`Declarations::fill`].
276    /// * `wire` — fresh identifier for the boundary parameter this leaf adds.
277    ///   A leaf that needs more wires or locals derives their names from it.
278    /// * `var` — fresh identifier for the Rust local that holds the converted value.
279    ///   [`InputPlan::argument`] normally refers to it.
280    /// * `active` — the selector arms under which this leaf is used. Copy it to every
281    ///   returned wire and resource, and guard validation and acquisition with it.
282    fn input_leaf(
283        &self,
284        ty: &Type,
285        extra: &Self::Extra,
286        wire: syn::Ident,
287        var: syn::Ident,
288        active: &Condition,
289    ) -> Result<InputPlan<Self::Wire, Self::Context>>;
290    /// Complete a composed parameter plan before it is stored in a slot.
291    /// Called once per bound parameter, including records and selectors.
292    fn finish_input(
293        &self,
294        _plan: &mut InputPlan<Self::Wire, Self::Context>,
295        _extra: &Self::Extra,
296    ) -> Result<()> {
297        Ok(())
298    }
299    /// Complete a composed result plan, including unit and multi-wire results.
300    /// Install call-level delivery here, after all output expressions are known.
301    fn finish_output(
302        &self,
303        _plan: &mut OutputPlan<Self::Wire, Self::Context>,
304        _extra: &Self::Extra,
305    ) -> Result<()> {
306        Ok(())
307    }
308    /// Prepare delivery of the Rust expression `value` as one or more wires.
309    ///
310    /// Keep unpublished allocations under Rust ownership and transfer them only in
311    /// [`OutputPlan::commit`]. Return a generation error for unsupported
312    /// conversions. Install final delivery in [`Codec::finish_output`], rather
313    /// than on nested leaf plans; recursive composition rejects leaf delivery.
314    ///
315    /// # Arguments
316    ///
317    /// * `ty` — the owned source type of the value.
318    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`]
319    ///   or [`Declarations::fill`].
320    /// * `wire` — fresh identifier for the delivered value. A leaf that needs more
321    ///   wires or locals derives their names from it.
322    /// * `value` — Rust expression producing the value to deliver.
323    /// * `borrowed` — `true` when `value` is a shared reference to `ty`, such as a
324    ///   record field observed in place; do not move out of it. `false` when
325    ///   `value` is owned.
326    fn output_leaf(
327        &self,
328        ty: &Type,
329        extra: &Self::Extra,
330        wire: syn::Ident,
331        value: TokenStream,
332        borrowed: bool,
333    ) -> Result<OutputPlan<Self::Wire, Self::Context>>;
334}
335/// A flat model whose slots contain only conversion data for wrapper emission.
336///
337/// Unbound source helpers have `None` slots. Bound functions need `Some` for their
338/// result and every parameter; [`prebindgen_flat::validate_bindings`] checks this before emission.
339/// Record fields retain only names and source types. Recipes and [`Codec::Extra`]
340/// are construction inputs, not stored in this model.
341///
342/// Output conversion data cannot be attached to an input position:
343/// ```compile_fail
344/// use prebindgen_declarations::Bindings;
345/// fn swap(flat: &mut Bindings<syn::Type>) {
346///     let function = flat.function_mut("foo").unwrap();
347///     function.params[0].ty.slot = function.ret.slot.clone();
348/// }
349/// ```
350/// Nor can a recipe be stored in a slot:
351/// ```compile_fail
352/// use prebindgen_declarations::{Bindings, Input};
353/// fn configure(flat: &mut Bindings<syn::Type>) {
354///     flat.function_mut("foo").unwrap().params[0].ty.slot = Some(Input::Leaf);
355/// }
356/// ```
357pub type Bindings<W, C = ()> = Flat<InputPlan<W, C>, OutputPlan<W, C>>;

359/// Build and attach a converter to a named function parameter.
360///
361/// Resolves the occurrence's source type from `flat`, checks recipe compatibility,
362/// and recursively composes wires and code. Nested conversions use the supplied
363/// `extra`, not data from referenced constructors or getters. Recipes and `extra`
364/// are borrowed only for this call. A failure leaves the existing slot unchanged.
365/// Rebinding an occurrence replaces its converter without changing other slots.
366/// Generated names are scoped by parameter position, so separate calls cannot
367/// collide within the wrapper. Keep the function signature fixed after binding.
368/// Check completeness with [`prebindgen_flat::validate_bindings`].
369///
370/// # Arguments
371///
372/// * `flat` — the model holding the function; the parameter's slot is replaced.
373/// * `codec` — the adapter's leaf conversions.
374/// * `function` — unqualified name of the source function.
375/// * `parameter` — name of the parameter, as written in the source.
376/// * `recipe` — how to build the argument from boundary inputs.
377/// * `extra` — occurrence data passed to every codec call for this conversion.
378///
379/// Input binding rejects output recipes:
380/// ```compile_fail
381/// use prebindgen_declarations::{bind_input, Codec, Output};
382/// use prebindgen_flat::Flat;
383/// fn wrong<C: Codec>(flat: &mut Flat<C::Input, C::Output>, codec: &C, extra: &C::Extra) {
384///     bind_input(flat, codec, "foo", "x", &Output::Leaf, extra).unwrap();
385/// }
386/// ```
387pub fn bind_input<C: Codec>(
388    flat: &mut Flat<C::Input, C::Output>,
389    codec: &C,
390    function: &str,
391    parameter: &str,
392    recipe: &Input,
393    extra: &C::Extra,
394) -> Result<()> {
395    let f = flat.function(function)?;
396    let index = f
397        .params
398        .iter()
399        .position(|p| p.name == parameter)
400        .ok_or_else(|| format!("`{function}`: unknown parameter `{parameter}`"))?;
401    let mut conversion = Compiler::new(flat, codec, format!("p{index}")).input(
402        &f.params[index].ty.ty,
403        recipe,
404        extra,
405        &Condition::default(),
406    )?;
407    codec.finish_input(&mut conversion, extra)?;
408    flat.function_mut(function)?.params[index].ty.slot = Some(conversion.into());
409    Ok(())
410}

412/// Build and attach a converter to a function result, replacing any existing one.
413///
414/// Uses the same construction and failure rules as [`bind_input`]. Output code
415/// uses [`prebindgen_conversions::source_result_expr`] for the source call's result local.
416/// The recipe is checked for compatible types and signatures, and for repeated
417/// consumption of the same value. Unit results use `Output::fields([])`.
418///
419/// # Arguments
420///
421/// * `flat` — the model holding the function; the result's slot is replaced.
422/// * `codec` — the adapter's leaf conversions.
423/// * `function` — unqualified name of the source function.
424/// * `recipe` — how to deliver the result as boundary outputs.
425/// * `extra` — occurrence data passed to every codec call for this conversion.
426///
427/// Output binding rejects input recipes:
428/// ```compile_fail
429/// use prebindgen_declarations::{bind_output, Codec, Input};
430/// use prebindgen_flat::Flat;
431/// fn wrong<C: Codec>(flat: &mut Flat<C::Input, C::Output>, codec: &C, extra: &C::Extra) {
432///     bind_output(flat, codec, "foo", &Input::Leaf, extra).unwrap();
433/// }
434/// ```
435pub fn bind_output<C: Codec>(
436    flat: &mut Flat<C::Input, C::Output>,
437    codec: &C,
438    function: &str,
439    recipe: &Output,
440    extra: &C::Extra,
441) -> Result<()> {
442    let mut conversion = Compiler::new(flat, codec, "ret".into()).output(
443        &flat.function(function)?.ret.ty,
444        recipe,
445        extra,
446        prebindgen_conversions::source_result_expr(),
447        false,
448    )?;
449    codec.finish_output(&mut conversion, extra)?;
450    flat.function_mut(function)?.ret.slot = Some(conversion.into());
451    Ok(())
452}

454struct Compiler<'a, C: Codec> {
455    flat: &'a Flat<C::Input, C::Output>,
456    codec: &'a C,
457    prefix: String,
458    next: std::cell::Cell<usize>,
459}
460impl<'a, C: Codec> Compiler<'a, C> {
461    fn new(flat: &'a Flat<C::Input, C::Output>, codec: &'a C, prefix: String) -> Self {
462        Self {
463            flat,
464            codec,
465            prefix,
466            next: std::cell::Cell::new(0),
467        }
468    }
469    fn name(&self) -> syn::Ident {
470        let next = self.next.get();
471        let n = format_ident!("{}_w{next}", self.prefix);
472        self.next.set(next + 1);
473        n
474    }
475    fn prefix(&self, wires: &mut [Wire<C::Wire>], segment: ComponentSegment) {
476        for wire in wires {
477            self.codec.prefix_component(&mut wire.kind, segment.clone());
478        }
479    }
480    fn input(
481        &self,
482        ty: &Type,
483        recipe: &Input,
484        extra: &C::Extra,
485        active: &Condition,
486    ) -> Result<InputPlan<C::Wire, C::Context>> {
487        let var = self.name();
488        let mut wires = Vec::new();
489        let mut validation = TokenStream::new();
490        let mut acquire = TokenStream::new();
491        let mut resources = Vec::new();
492        let mut acknowledgement = None;
493        let mut statements = TokenStream::new();
494        let argument = quote!(#var);
495        match recipe {
496            Input::Leaf => {
497                let w = self.name();
498                return self.codec.input_leaf(ty, extra, w, var, active);
499            }
500            Input::Fields(children) => {
501                if type_key(ty) == "()" && children.is_empty() {
502                    return Ok(InputPlan {
503                        context: Default::default(),
504                        acknowledgement,
505                        wires,
506                        validation,
507                        acquire,
508                        resources,
509                        statements,
510                        argument: quote!(()),
511                    });
512                }
513                let r = self.flat.record(ty)?;
514                if r.fields.len() != children.len() {
515                    return Err(format!(
516                        "`{}`: expected {} field converters, got {}",
517                        type_key(ty),
518                        r.fields.len(),
519                        children.len()
520                    ));
521                }
522                let mut fields = Vec::new();
523                for (field, child) in r.fields.iter().zip(children) {
524                    let mut p = self.input(&field.ty, child, extra, active)?;
525                    self.prefix(
526                        &mut p.wires,
527                        ComponentSegment::Field(field.name.to_string()),
528                    );
529                    validation.extend(p.validation);
530                    acquire.extend(p.acquire);
531                    resources.extend(p.resources);
532                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
533                    statements.extend(p.statements);
534                    wires.extend(p.wires);
535                    let name = &field.name;
536                    let arg = p.argument;
537                    fields.push(quote!(#name:#arg));
538                }
539                statements.extend(quote!(let #var:#ty=#ty{#(#fields),*};));
540            }
541            Input::Constructor { function, args } => {
542                let f = self.flat.function(function)?;
543                if !same_type(&f.ret.ty, ty) {
544                    return Err(format!(
545                        "constructor `{function}` returns `{}`, expected `{}`",
546                        type_key(&f.ret.ty),
547                        type_key(ty)
548                    ));
549                }
550                if f.params.len() != args.len() {
551                    return Err(format!("constructor `{function}`: wrong argument count"));
552                }
553                let mut call_args = Vec::new();
554                for (param, child) in f.params.iter().zip(args) {
555                    let mut p = self.input(&param.ty.ty, child, extra, active)?;
556                    self.prefix(
557                        &mut p.wires,
558                        ComponentSegment::Argument(param.name.to_string()),
559                    );
560                    validation.extend(p.validation);
561                    acquire.extend(p.acquire);
562                    resources.extend(p.resources);
563                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
564                    statements.extend(p.statements);
565                    wires.extend(p.wires);
566                    call_args.push(p.argument);
567                }
568                let path = &f.path;
569                statements.extend(quote!(let #var:#ty=#path(#(#call_args),*);));
570            }
571            Input::Selector(arms) => {
572                if arms.is_empty() {
573                    return Err("an input selector needs at least one alternative".into());
574                }
575                if matches!(ty, Type::Reference(_)) {
576                    return Err("selector results must be owned in this subset".into());
577                }
578                let tag = self.name();
579                wires.push(Wire {
580                    name: tag.clone(),
581                    kind: {
582                        let mut kind = self.codec.selector(extra);
583                        self.codec
584                            .prefix_component(&mut kind, ComponentSegment::Selector);
585                        kind
586                    },
587                });
588                let count = arms.len() as i32;
589                let invalid = active.and(quote!(!(0..#count).contains(&#tag)));
590                validation.extend(quote!(if #invalid {return Err("invalid selector".into());}));
591                let mut cases = Vec::new();
592                for (i, arm) in arms.iter().enumerate() {
593                    let mut p = self.input(ty, arm, extra, &active.arm(tag.clone(), i as i32))?;
594                    self.prefix(&mut p.wires, ComponentSegment::Alternative(i));
595                    validation.extend(p.validation);
596                    acquire.extend(p.acquire);
597                    resources.extend(p.resources);
598                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
599                    let body = p.statements;
600                    let value = p.argument;
601                    wires.extend(p.wires);
602                    let i = i as i32;
603                    cases.push(quote!(#i=>{#body #value}));
604                }
605                statements = quote!(let #var:#ty=match #tag{#(#cases,)* _=>unreachable!("validated selector")};);
606            }
607        }
608        Ok(InputPlan {
609            context: Default::default(),
610            acknowledgement,
611            wires,
612            validation,
613            acquire,
614            resources,
615            statements,
616            argument,
617        })
618    }
619    fn output(
620        &self,
621        ty: &Type,
622        recipe: &Output,
623        extra: &C::Extra,
624        value: TokenStream,
625        borrowed: bool,
626    ) -> Result<OutputPlan<C::Wire, C::Context>> {
627        let mut out = OutputPlan {
628            context: Default::default(),
629            delivery: Default::default(),
630            commit: TokenStream::new(),
631            wires: Vec::new(),
632            statements: TokenStream::new(),
633            values: Vec::new(),
634        };
635        match recipe {
636            Output::Leaf => {
637                let w = self.name();
638                return self.codec.output_leaf(ty, extra, w, value, borrowed);
639            }
640            Output::Fields(children) => {
641                if type_key(ty) == "()" && children.is_empty() {
642                    return Ok(out);
643                }
644                let r = self.flat.record(ty)?;
645                if r.fields.len() != children.len() {
646                    return Err("wrong output field count".into());
647                }
648                for (field, child) in r.fields.iter().zip(children) {
649                    let name = &field.name;
650                    let mut p =
651                        self.output(&field.ty, child, extra, quote!(&(#value).#name), true)?;
652                    self.prefix(&mut p.wires, ComponentSegment::Field(name.to_string()));
653                    append(&mut out, p)?;
654                }
655            }
656            Output::Getter { function, output } => {
657                let f = self.flat.function(function)?;
658                let expected: Type = syn::parse_quote!(&#ty);
659                if f.params.len() != 1 || !same_type(&f.params[0].ty.ty, &expected) {
660                    return Err(format!(
661                        "getter `{function}` must take exactly `&{}`",
662                        type_key(ty)
663                    ));
664                }
665                if matches!(f.ret.ty, Type::Reference(_)) {
666                    return Err("borrowed getter results are outside this subset".into());
667                }
668                let v = self.name();
669                let path = &f.path;
670                let arg = if borrowed { value } else { quote!(&#value) };
671                out.statements = quote!(let #v=#path(#arg););
672                let mut child = self.output(&f.ret.ty, output, extra, quote!(#v), false)?;
673                self.prefix(&mut child.wires, ComponentSegment::Getter(function.clone()));
674                append(&mut out, child)?;
675            }
676            Output::Combine(parts) => {
677                if parts.is_empty() {
678                    return Err("an output combination needs at least one part".into());
679                }
680                if parts
681                    .iter()
682                    .filter(|p| p.consumes(ty, extra, self.codec))
683                    .count()
684                    > 1
685                {
686                    return Err("a combination cannot consume the same value twice".into());
687                }
688                let mut bodies = Vec::new();
689                for (index, child) in parts.iter().enumerate() {
690                    let consuming = child.consumes(ty, extra, self.codec);
691                    let arg = if borrowed || consuming {
692                        value.clone()
693                    } else {
694                        quote!(&#value)
695                    };
696                    let mut p = self.output(ty, child, extra, arg, borrowed || !consuming)?;
697                    self.prefix(&mut p.wires, ComponentSegment::View(index));
698                    bodies.push((consuming, std::mem::take(&mut p.statements)));
699                    append(&mut out, p)?;
700                }
701                // Wire order follows the declaration. Observations execute before the move.
702                bodies.sort_by_key(|(consumes, _)| *consumes);
703                for (_, body) in bodies {
704                    out.statements.extend(body);
705                }
706            }
707        }
708        Ok(out)
709    }
710}
711fn append<W: prebindgen_flat::WireKind, C>(
712    to: &mut OutputPlan<W, C>,
713    from: OutputPlan<W, C>,
714) -> Result<()> {
715    if !from.delivery.statements.is_empty()
716        || from.delivery.result.to_string() != quote!(()).to_string()
717    {
718        return Err("composed output delivery must be installed by Codec::finish_output".into());
719    }
720    to.commit.extend(from.commit);
721    to.wires.extend(from.wires);
722    to.statements.extend(from.statements);
723    to.values.extend(from.values);
724    Ok(())
725}

745fn merge_acknowledgement(
746    target: &mut Option<TokenStream>,
747    incoming: Option<TokenStream>,
748) -> Result<()> {
749    if let Some(code) = incoming {
750        if let Some(previous) = target {
751            if previous.to_string() != code.to_string() {
752                return Err("conflicting input acknowledgements".into());
753            }
754        } else {
755            *target = Some(code);
756        }
757    }
758    Ok(())
