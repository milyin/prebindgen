4/// Construct a named Rust record from field conversion plans in declaration order.
5///
6/// Adapters can use this when a single language-level leaf represents a record.
7/// The caller supplies every field exactly once, with the correct source type,
8/// and chooses a fresh `var` identifier. Child plans must use the same context
9/// and activation condition and must not have been finalized with `finish_input`.
10/// Validation, acquisition, resources, acknowledgements and construction remain
11/// in their respective phases. Field paths are prepended through the codec.
12/// Finalize only the complete occurrence with [`Codec::finish_input`].
13pub fn record_input<C: Codec>(
14    codec: &C,
15    ty: &Type,
16    var: syn::Ident,
17    fields: impl IntoIterator<Item = Result<(syn::Ident, InputPlan<C::Wire, C::Context>)>>,
18) -> Result<InputPlan<C::Wire, C::Context>> {
19    let mut out = InputPlan {
20        context: Default::default(),
21        acknowledgement: None,
22        wires: Vec::new(),
23        validation: TokenStream::new(),
24        acquire: TokenStream::new(),
25        resources: Vec::new(),
26        statements: TokenStream::new(),
27        argument: quote!(#var),
28    };
29    let mut values = Vec::new();
30    for field in fields {
31        let (name, mut child) = field?;
32        for wire in &mut child.wires {
33            codec.prefix_component(&mut wire.kind, ComponentSegment::Field(name.to_string()));
34        }
35        out.validation.extend(child.validation);
36        out.acquire.extend(child.acquire);
37        out.resources.extend(child.resources);
38        merge_acknowledgement(&mut out.acknowledgement, child.acknowledgement)?;
39        out.statements.extend(child.statements);
40        out.wires.extend(child.wires);
41        let argument = child.argument;
42        values.push(quote!(#name: #argument));
43    }
44    out.statements
45        .extend(quote!(let #var: #ty = #ty { #(#values),* };));
46    Ok(out)
47}

49/// Observe a named Rust record through shared field references, evaluating it once.
50///
51/// The caller validates the field count and types against the source record.
52/// `fields` supplies source field names in declaration order. `convert` receives
53/// each field's index and a shared-reference expression; it must convert that
54/// borrowed field without consuming it. Choose a fresh `var` identifier and use
55/// the same context for all children. Field paths are prepended through the codec.
56/// Child delivery must remain unset; finalize only the complete occurrence with
57/// [`Codec::finish_output`]. Commit phases are preserved until successful delivery.
58pub fn record_output<C: Codec>(
59    codec: &C,
60    var: syn::Ident,
61    value: TokenStream,
62    fields: impl IntoIterator<Item = syn::Ident>,
63    mut convert: impl FnMut(usize, TokenStream) -> Result<OutputPlan<C::Wire, C::Context>>,
64) -> Result<OutputPlan<C::Wire, C::Context>> {
65    let mut fields = fields.into_iter().peekable();
66    let binding = if fields.peek().is_some() {
67        quote!(let #var = &(#value);)
68    } else {
69        quote!(let _ = &(#value);)
70    };
71    let mut out = OutputPlan {
72        context: Default::default(),
73        delivery: Default::default(),
74        commit: TokenStream::new(),
75        wires: Vec::new(),
76        statements: binding,
77        values: Vec::new(),
78    };
79    for (index, name) in fields.enumerate() {
80        let mut child = convert(index, quote!(&#var.#name))?;
81        for wire in &mut child.wires {
82            codec.prefix_component(&mut wire.kind, ComponentSegment::Field(name.to_string()));
83        }
84        append(&mut out, child)?;
85    }
86    Ok(out)
87}

89/// Decompose an owned record into field conversions in declaration order.
90///
91/// Unlike [`record_output`], each child receives an owned field expression and
92/// may move it. The caller must provide each field exactly once, ensure Rust
93/// permits moving its fields (in particular, the record must not implement
94/// `Drop`), and mark the containing conversion as consuming. Choose a fresh
95/// `var`. Unmoved fields and prepared child allocations retain Rust ownership
96/// on failure. Children must leave delivery unset; commit follows final delivery.
97pub fn record_output_owned<C: Codec>(
98    codec: &C,
99    var: syn::Ident,
100    value: TokenStream,
101    fields: impl IntoIterator<Item = syn::Ident>,
102    mut convert: impl FnMut(usize, TokenStream) -> Result<OutputPlan<C::Wire, C::Context>>,
103) -> Result<OutputPlan<C::Wire, C::Context>> {
104    let mut fields = fields.into_iter().peekable();
105    let binding = if fields.peek().is_some() {
106        quote!(let #var = #value;)
107    } else {
108        quote!(let _ = #value;)
109    };
110    let mut out = OutputPlan {
111        context: Default::default(),
112        delivery: Default::default(),
113        commit: TokenStream::new(),
114        wires: Vec::new(),
115        statements: binding,
116        values: Vec::new(),
117    };
118    for (index, name) in fields.enumerate() {
119        let mut child = convert(index, quote!(#var.#name))?;
120        for wire in &mut child.wires {
121            codec.prefix_component(&mut wire.kind, ComponentSegment::Field(name.to_string()));
122        }
123        append(&mut out, child)?;
124    }
125    Ok(out)
126}
