4/// Interpret adapter-specific trees into the fixed schema’s parameter/result slots.
5/// Adapters select their operation vocabularies and dispatch to concrete helpers.
6/// Optional features reject unsupported work or decline a specialization.
7/// All operations run during generation; the returned fragments run in the library.
8#[allow(clippy::type_complexity)]
9pub trait Codec {
10    /// Input vocabulary interpreted by this adapter.
11    type InOps: InputOps;
12    /// Output vocabulary interpreted by this adapter.
13    type OutOps: OutputOps;
14    /// Adapter-owned state scoped to one conversion compilation.
15    type CompileState: Default;

17    /// Interpret one resolved input operation.
18    #[allow(clippy::too_many_arguments)]
19    fn compile_input_tree(
20        &self,

24        extra: &Self::Extra,
25        active: &Condition,

28    where
29        Self: Sized;
30    /// Interpret one resolved output operation.
31    #[allow(clippy::too_many_arguments)]
32    fn compile_output_tree(
33        &self,
34        cx: &Compiler<'_, Self>,
35        ty: &Type,
36        tree: &ResolvedOutput<Self>,
37        extra: &Self::Extra,
38        value: TokenStream,
39        borrowed: bool,
40        wire: Option<&syn::Ident>,
41    ) -> Result<CodecOutputPlan<Self>>
42    where
43        Self: Sized;
44    /// Interpret a function or callback result, including its error channel.
45    #[allow(clippy::too_many_arguments)]
46    fn compile_output_root(
47        &self,
48        cx: &Compiler<'_, Self>,
49        ty: &Type,
50        tree: &ResolvedOutput<Self>,
51        extra: &Self::Extra,
52        value: TokenStream,
53        borrowed: bool,

55    ) -> Result<CodecOutputPlan<Self>>
56    where
57        Self: Sized,
58    {
59        if reference_return {
60            self.compile_output_return(cx, ty, tree, extra, value, borrowed)

63        }
64    }

72        value: TokenStream,

75    where
76        Self: Sized,

79    }

423    /// Explicit projection; the default preserves field-policy and reference
424    /// rejection. An adapter override keeps its own existing type-specific rules.
425    fn output_projected(
426        &self,
427        mut request: OutputLeaf<'_, Self>,
428        projection: &Projection,
429    ) -> Result<CodecOutputPlan<Self>> {
430        if !projection.fields.is_empty() {
431            return Err("this output converter does not support field projection overrides".into());
432        }
433        let mut payload = request.ty;
434        while let Some(inner) = option_inner(payload) {
435            payload = inner;
436        }
437        if projection.mode == ProjectionMode::Owned && matches!(payload, Type::Reference(_)) {
438            return Err("owned projection cannot consume a reference payload".into());
439        }
440        request.access = if projection.mode == ProjectionMode::Borrowed {
441            ValueAccess::Shared
442        } else {
443            ValueAccess::Owned
444        };
445        self.output_leaf(request)
446    }

449/// Prepare the ordinary mapped-String error path, also usable by a codec that
450/// overrides typed source errors and needs the default for undeclared types.
451pub fn prepare_mapped_source_error<C: Codec + ?Sized>(
452    codec: &C,
453    request: SourceError<'_, C>,
454) -> Result<SourceErrorPlan<C>> {
455    let SourceError {
456        flat,
457        ty,
458        extra,
459        value: _,
460        message,
461    } = request;
462    let message = message.ok_or_else(|| {
463        format!(
464            "Result error `{}` requires an explicit error_mapper or on_error mapper",
465            type_key(ty)
466        )
467    })?;
468    let propagate = codec.mapped_error(flat, ty, message, extra)?;
469    let text = syn::parse_quote!(String);
470    let output = codec.output_leaf(OutputLeaf::new(
471        flat,
472        &text,
473        extra,
474        format_ident!("source_error_message"),
475        quote!(source_error_message),
476        ValueAccess::Owned,
477    ))?;
478    Ok(SourceErrorPlan {
479        payload: output.outcomes.success,
480        propagate,
481    })
482}
