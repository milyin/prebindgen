1//! Recursive construction and matching of source enum payloads.
2use super::*;
3use std::collections::BTreeMap;
4
5/// Collect enum variant/member overrides, rejecting duplicate keys.
6pub fn overrides<'a, T, Fields, Name>(
7    entries: impl IntoIterator<Item = (&'a str, Fields)>,
8) -> BTreeMap<String, BTreeMap<String, T>>
9where
10    Fields: IntoIterator<Item = (Name, T)>,
11    Name: AsRef<str>,
12{
13    let mut variants = BTreeMap::new();
14    for (name, fields) in entries {
15        let mut members = BTreeMap::new();
16        for (field, tree) in fields {
17            assert!(
18                members.insert(field.as_ref().to_owned(), tree).is_none(),
19                "duplicate enum field override"
20            );
21        }
22        assert!(
23            variants.insert(name.into(), members).is_none(),
24            "duplicate enum variant override"
25        );
26    }
27    variants
28}
29
30/// Check override names against the source enum's variants and fields.
31pub fn validate_overrides<V, F, T>(
32    source: &prebindgen_flat::PayloadEnum<V, F>,
33    entries: &BTreeMap<String, BTreeMap<String, T>>,
34) -> Result<()> {
35    for (variant, fields) in entries {
36        let arm = source
37            .variants
38            .iter()
39            .find(|v| v.name.to_string().trim_start_matches("r#") == variant)
40            .ok_or_else(|| format!("unknown enum variant `{variant}`"))?;
41        for field in fields.keys() {
42            if !arm.fields.iter().any(|f| f.key() == *field) {
43                return Err(format!("unknown enum field `{variant}.{field}`"));
44            }
45        }
46    }
47    Ok(())
48}
