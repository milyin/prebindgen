1//! Recursive construction and matching of source enum payloads.
2use super::*;
3use std::collections::BTreeMap;
4
5pub(crate) fn overrides<'a, T>(
6    entries: impl IntoIterator<Item = (&'a str, Vec<(&'a str, T)>)>,
7) -> BTreeMap<String, BTreeMap<String, T>> {
8    let mut variants = BTreeMap::new();
9    for (name, fields) in entries {
10        let mut members = BTreeMap::new();
11        for (field, tree) in fields {
12            assert!(
13                members.insert(field.into(), tree).is_none(),
14                "duplicate enum field override"
15            );
16        }
17        assert!(
18            variants.insert(name.into(), members).is_none(),
19            "duplicate enum variant override"
20        );
21    }
22    variants
23}
24
25fn validate_overrides<V, F, T>(
26    source: &prebindgen_flat::PayloadEnum<V, F>,
27    entries: &BTreeMap<String, BTreeMap<String, T>>,
28) -> Result<()> {
29    for (variant, fields) in entries {
30        let arm = source
31            .variants
32            .iter()
33            .find(|v| v.name.to_string().trim_start_matches("r#") == variant)
34            .ok_or_else(|| format!("unknown enum variant `{variant}`"))?;
35        for field in fields.keys() {
36            if !arm.fields.iter().any(|f| f.key() == *field) {
37                return Err(format!("unknown enum field `{variant}.{field}`"));
38            }
39        }
40    }
41    Ok(())
42}

44impl DeclDefaults {
45    #[allow(clippy::too_many_arguments)]
46    pub(crate) fn enum_input_recipe<C: DeclCodec>(
47        &self,
48        flat: &Flat<C::Schema>,
49        codec: &C,
50        ty: &Type,
51        extra: &C::Extra,
52        entries: &BTreeMap<
53            String,
54            BTreeMap<String, DeclInput<DeclInputRequest, DeclOutputRequest>>,
55        >,
56        value: bool,
57        stack: &mut Vec<String>,
58    ) -> Result<crate::ResolvedInput<C>> {
59        let source = flat.payload_enum(ty)?;
60        validate_overrides(source, entries)?;
61        let default = if value {
62            DeclInput::value()
63        } else {
64            DeclInput::auto()
65        };
66        let variants = source
67            .variants
68            .iter()
69            .map(|arm| {
70                let fields = entries.get(arm.name.to_string().trim_start_matches("r#"));
71                arm.fields
72                    .iter()
73                    .map(|field| {
74                        let recipe = fields.and_then(|f| f.get(&field.key())).unwrap_or(&default);
75                        let previous = self.input_role.get();
76                        if value {
77                            self.input_role.set(InputRole::EnumField);
78                        }
79                        let result = self.input(flat, codec, &field.ty, recipe, extra, stack);
80                        self.input_role.set(previous);
81                        result.map_err(|e| {
82                            format!("{}::{}.{}: {e}", type_key(ty), arm.name, field.key())
83                        })
84                    })
85                    .collect::<Result<Vec<_>>>()
86            })
87            .collect::<Result<Vec<_>>>()?;
88        crate::resolution::resolve_input_enum(codec, variants, value)
89    }
90    #[allow(clippy::too_many_arguments)]
91    pub(crate) fn enum_output_recipe<C: DeclCodec>(
92        &self,
93        flat: &Flat<C::Schema>,
94        codec: &C,
95        ty: &Type,
96        extra: &C::Extra,
97        entries: &BTreeMap<
98            String,
99            BTreeMap<String, DeclOutput<DeclInputRequest, DeclOutputRequest>>,
100        >,
101        value: bool,
102        stack: &mut Vec<String>,
103        getter: bool,
104    ) -> Result<crate::ResolvedOutput<C>> {
105        let source = flat.payload_enum(ty)?;
106        validate_overrides(source, entries)?;
107        let default = if value {
108            DeclOutput::value()
109        } else {
110            DeclOutput::auto()
111        };
112        let variants = source
113            .variants
114            .iter()
115            .map(|arm| {
116                let fields = entries.get(arm.name.to_string().trim_start_matches("r#"));
117                arm.fields
118                    .iter()
119                    .map(|field| {
120                        let recipe = fields.and_then(|f| f.get(&field.key())).unwrap_or(&default);
121                        self.output(flat, codec, &field.ty, recipe, extra, stack, getter)
122                            .map_err(|e| {
123                                format!("{}::{}.{}: {e}", type_key(ty), arm.name, field.key())
124                            })
125                    })
126                    .collect::<Result<Vec<_>>>()
127            })
128            .collect::<Result<Vec<_>>>()?;
129        crate::resolution::resolve_output_enum(codec, variants, value, None)
130    }
131}
