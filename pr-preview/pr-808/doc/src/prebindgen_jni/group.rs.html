6/// Meaning and encoding of a parent with one child description.
7#[derive(Clone, Debug)]
8pub enum JniChain {
9    /// Zero or one payload occurrence.
10    Optional(JniOptional),
11    /// Zero or more element occurrences.
12    Sequence(JniSequence),
13    /// Expanded fields grouped into one C-representation bulk frame.
14    Bulk(std::rc::Rc<crate::bulk::BulkRecord>),
15}
16/// Meaning of an ordered collection of child descriptions.
17#[derive(Clone, Debug)]
18pub enum JniBranch {
19    /// Related values or source fields, all belonging to the parent.
20    Group(JniGroup),
21    /// Exactly one selected arm; at least one arm is required.
22    Alternatives,
23    /// Source enum tag and selected payload, grouped or explicitly expanded.
24    Enum(std::rc::Rc<crate::sealed::JniEnumGroup>),
25}
26/// Relationships between the child values of a group.
27#[derive(Clone, Debug)]
28pub enum JniGroup {
29    /// Ordered conversion components, such as constructor arguments or output views.
30    Values,
31    /// A source constructor argument, retained for public Kotlin parameter naming.
32    Argument(String),
33    /// An explicitly named output view.
34    Named(String),
35    /// Source record fields. JNI may associate the record with a Kotlin data class.
36    Fields(Option<DataGroup>),
37    /// One named source field.
38    Field(syn::Ident),
39}
40pub(crate) fn validate_selector(wire: &Wire<JniWire>) -> Result<()> {
41    if !matches!(wire.kind, JniWire::Selector { .. }) {
42        return Err("selector requires the adapter's integer tag representation".into());
43    }
44    wire.kind.validate()
45}
46/// How this optional occurrence distinguishes absence from a present payload.
47#[derive(Clone, Debug)]
48pub enum JniOptional {
49    /// A dedicated 0/1 tag. The binding is also the normalized presence value.
50    Flag(syn::Ident),
51    /// An excluded representation in one payload value.
52    Niche {
53        /// Normalized presence local used by conversion conditions.
54        presence: syn::Ident,
55        /// Payload binding and reserved absent representation.
56        reservation: prebindgen_declarations::NicheReservation<JniNiche>,
57    },
58}
59impl JniOptional {
60    /// Binding of the normalized presence condition.
61    pub fn presence(&self) -> &syn::Ident {
62        match self {
63            Self::Flag(flag) => flag,
64            Self::Niche { presence, .. } => presence,
65        }
66    }
67    /// Binding whose value distinguishes absent and present states.
68    pub fn discriminator(&self) -> &syn::Ident {
69        match self {
70            Self::Flag(flag) => flag,
71            Self::Niche { reservation, .. } => &reservation.discriminator,
72        }
73    }
74    /// Reserved payload representation; a flag encoding has no reservation.
75    pub fn niche(&self) -> Option<&JniNiche> {
76        match self {
77            Self::Flag(_) => None,
78            Self::Niche { reservation, .. } => Some(&reservation.niche),
79        }
80    }
81    pub(crate) fn from_parts(
82        presence: syn::Ident,
83        discriminator: syn::Ident,
84        niche: Option<JniNiche>,
85        controls: &[Wire<JniWire>],
86    ) -> Result<Self> {
87        match niche {
88            Some(niche) if controls.is_empty() => Ok(Self::Niche {
89                presence,
90                reservation: prebindgen_declarations::NicheReservation {
91                    discriminator,
92                    niche,
93                },
94            }),
95            None if controls.len() == 1
96                && presence == discriminator
97                && controls[0].name == discriminator =>
98            {
99                validate_selector(&controls[0])?;
100                Ok(Self::Flag(presence))
101            }
102            _ => {
103                Err("optional encoding requires either one matching flag or a payload niche".into())
104            }
105        }
106    }
107}

109// Definitions in one native scope; sequence element definitions belong to row scopes.
110pub(crate) fn values(
111    node: &WireNode<JniWire, JniChain, JniBranch>,
112) -> std::vec::IntoIter<&Wire<JniWire>> {
113    fn visit<'a>(
114        node: &'a WireNode<JniWire, JniChain, JniBranch>,
115        out: &mut Vec<&'a Wire<JniWire>>,
116    ) {
117        match node {
118            WireNode::Leaf(wires) => out.extend(wires),
119            WireNode::Chain {
120                meta: JniChain::Sequence(_) | JniChain::Bulk(_),
121                controls,
122                ..
123            } => out.extend(controls),
124            WireNode::Chain {
125                child, controls, ..
126            } => {
127                out.extend(controls);
128                visit(child, out);
129            }
130            WireNode::Branch {
131                children, controls, ..
132            } => {
133                out.extend(controls);
134                for child in children {
135                    visit(child, out);
136                }
137            }
138        }
139    }
140    let mut out = vec![];
141    visit(node, &mut out);
142    out.into_iter()
143}

178pub(crate) fn validate_layout(
179    node: &WireNode<JniWire, crate::JniChain, crate::JniBranch>,
180) -> Result<()> {
181    prebindgen_conversions::validate_wire_names(prebindgen_flat::wire_definitions(
182        std::iter::once(node),
183    ))?;
184    for wire in values(node) {
185        wire.kind.validate()?;
186    }
187    fn visit(
188        node: &WireNode<JniWire, crate::JniChain, crate::JniBranch>,
189        bindings: &mut std::collections::BTreeSet<String>,
190        in_fields: bool,
191        reserved: &mut std::collections::BTreeMap<String, Vec<JniNiche>>,
192    ) -> Result<()> {
193        match node {
194            WireNode::Chain {
195                meta: JniChain::Bulk(_),
196                child,
197                controls,
198            } => {
199                if controls.is_empty() || !matches!(controls[0].kind, JniWire::BulkBuffer { .. }) {
200                    return Err("bulk scope requires a byte-array carrier".into());
201                }
202                visit(child, bindings, in_fields, reserved)?;
203            }
204            WireNode::Leaf(wires) => {
205                for w in wires {
206                    if matches!(w.kind, JniWire::Rows { .. }) {
207                        return Err("array carrier requires a sequence node".into());
208                    }
209                }
210            }
211            WireNode::Branch {
212                meta: JniBranch::Alternatives | JniBranch::Enum(_),
213                children: alternatives,
214                controls,
215                ..
216            } => {
217                if alternatives.is_empty() {
218                    return Err("selection has no alternatives".into());
219                }
220                if let [selector] = controls.as_slice() {
221                    validate_selector(selector)?;
222                } else {
223                    return Err("alternatives require one selector control".into());
224                }
225                for arm in alternatives {
226                    visit(arm, bindings, false, reserved)?;
227                }
228            }
229            WireNode::Chain {
230                child: element,
231                controls,
232                meta: JniChain::Sequence(metadata),
233                ..
234            } => {
235                let [carrier] = controls.as_slice() else {
236                    return Err("sequence requires one carrier control".into());
237                };
238                let valid = match metadata {
239                    JniSequence::Buffered | JniSequence::TypedBuffered => {
240                        matches!(carrier.kind, JniWire::Rows { .. })
241                    }
242                    JniSequence::Bulk => matches!(carrier.kind, JniWire::NativeRows { .. }),
243                    JniSequence::Callback => {
244                        matches!(&carrier.kind, JniWire::Selector { .. })
245                    }
246                };
247                if !valid {
248                    return Err("sequence carrier control is incompatible with its delivery".into());
249                }
250                if *metadata == JniSequence::TypedBuffered && !typed_rows::supported(element) {
251                    return Err(
252                        "typed output row constructor exceeds 254 JVM parameter slots".into(),
253                    );
254                }
255                if values(element).len() == 0 && *metadata != JniSequence::Bulk {
256                    return Err("sequence requires a nonempty element layout".into());
257                }
258                crate::sequence::validate_buffered(element)?;
259                validate_layout(element)?;
260            }
261            WireNode::Chain {
262                child: element,
263                meta: JniChain::Optional(o),
264                controls,
265                ..
266            } => {
267                match o {
268                    JniOptional::Flag(presence) => {
269                        let [flag] = controls.as_slice() else {
270                            return Err("flag optional requires one control".into());
271                        };
272                        validate_selector(flag)?;
273                        if &flag.name != presence {
274                            return Err("optional flag reference does not match its control".into());
275                        }
276                    }
277                    JniOptional::Niche { .. } if !controls.is_empty() => {
278                        return Err("niche optional must not have controls".into())
279                    }
280                    _ => {}
281                }
282                if !bindings.insert(o.presence().to_string()) {
283                    return Err(format!(
284                        "duplicate optional presence binding `{}`",
285                        o.presence()
286                    ));
287                }
288                if let Some(niche) = o.niche() {
289                    let wire = crate::group::values(element)
290                        .find(|w| w.name == *o.discriminator())
291                        .ok_or("optional discriminator outside payload")?;
292                    wire.kind.validate_niche(niche)?;
293                    if wire.kind.available_niche().as_ref() == Some(niche) {
294                        return Err(
295                            "available JNI niche is already reserved by an optional node".into(),
296                        );
297                    }
298                    let values = reserved.entry(o.discriminator().to_string()).or_default();
299                    if values.contains(niche) {
300                        return Err("duplicate JNI niche reservation on one wire".into());
301                    }
302                    values.push(niche.clone());
303                }
304                visit(element, bindings, false, reserved)?;
305            }
306            WireNode::Branch {
307                children,
308                meta: JniBranch::Group(metadata),
309                controls,
310                ..
311            } => {
312                if !controls.is_empty() {
313                    return Err("group metadata does not support control wires".into());
314                }
315                if let JniGroup::Fields(Some(record)) = metadata {
316                    if !bindings.insert(record.root.clone()) {
317                        return Err(format!("duplicate Kotlin group binding `{}`", record.root));
318                    }
319                }
320                match metadata {
321                    JniGroup::Named(name) => {
322                        validate_name(name)?;
323                    }
324                    JniGroup::Field(_) if !in_fields => {
325                        return Err(
326                            "named field must be a direct child of a field collection".into()
327                        )
328                    }
329                    JniGroup::Fields(_) => {
330                        let mut fields = std::collections::BTreeSet::new();
331                        for child in children.as_slice() {
332                            let WireNode::Branch {
333                                meta: JniBranch::Group(JniGroup::Field(name)),
334                                ..
335                            } = child
336                            else {
337                                return Err("field collection requires named field children".into());
338                            };
339                            if !fields.insert(name.unraw().to_string()) {
340                                return Err(format!("duplicate field `{name}`"));
341                            }
342                        }
343                    }
344                    _ => {}
345                }
346
347                for child in children {
348                    visit(
349                        child,
350                        bindings,
351                        matches!(metadata, JniGroup::Fields(_)),
352                        reserved,
353                    )?;
354                }
355            }
356        }
357        Ok(())
358    }
359    visit(
360        node,
361        &mut Default::default(),
362        false,
363        &mut Default::default(),
364    )
365}

413pub(crate) fn prepare_optional<'cfg>(
414    request: Optional<'_, JniCodec<'cfg>>,
415) -> Result<OptionalEncoding<JniCodec<'cfg>>> {
416    let prepared = prebindgen_declarations::prepare_optional_payload(
417        values_mut(request.payload),
418        request.direction,
419        JniWire::available_niche,
420        JniWire::set_niche,
421        |wire, niche, value| Ok(niche.present(wire, value)),
422        optional::remaining,
423        |wire, niche| optional::absent(wire, niche, false),
424    )?;
425    let discriminator = prepared
426        .reservation
427        .as_ref()
428        .map_or_else(|| request.presence.clone(), |r| r.discriminator.clone());
429    let (presence, controls) = if prepared.reservation.is_some() {
430        (
431            Presence::Niche {
432                input_test: prepared.input_test,
433            },
434            Vec::new(),
435        )
436    } else {
437        let flag = Wire {
438            name: request.presence.clone(),
439            kind: JniWire::selector(),
440        };
441        (Presence::Flag(flag.clone()), vec![flag])
442    };
443    let metadata = JniOptional::from_parts(
444        request.presence,
445        discriminator,
446        prepared.reservation.map(|r| r.niche),
447        &controls,
448    )
449    .map(JniChain::Optional)?;
450    Ok(OptionalEncoding {
451        presence,
452        metadata,
453        absent_values: prepared.absent_values,
454    })
455}

1008/// Physical/error decoding shape, ignoring wire names and Kotlin presentation
1009/// groups. Nested carriers and optional encodings must agree as well as the
1010/// top-level JVM signature, because the latter cannot describe rows in a buffer.
1011pub(crate) fn error_wire_shape(node: &kotlin::JniNode) -> Vec<String> {
1012    fn wires(wires: &[Wire<JniWire>], out: &mut Vec<String>) {
1013        for wire in wires {
1014            let detail = match &wire.kind {
1015                JniWire::Scalar { primitive, .. } | JniWire::Native128 { primitive, .. } => {
1016                    format!("{primitive:?}")
1017                }
1018                JniWire::Handle { handle, .. } => {
1019                    format!("{}:{:?}", type_key(&handle.ty), handle.access)
1020                }
1021                JniWire::Enum { enum_type, .. } => type_key(&enum_type.mapping.ty),
1022                JniWire::NativeBuffer { text } => text.to_string(),
1023                JniWire::Callback { callback } => callback.fqn(),
1024                _ => String::new(),
1025            };
1026            out.push(format!(
1027                "{:?}:{}:{detail}",
1028                std::mem::discriminant(&wire.kind),
1029                type_key(&wire.kind.rust_type())
1030            ));
1031        }
1032    }
1033    fn visit(node: &kotlin::JniNode, out: &mut Vec<String>) {
1034        match node {
1035            WireNode::Leaf(fields) => wires(fields, out),
1036            WireNode::Branch {
1037                meta: JniBranch::Group(_),
1038                controls,
1039                children,
1040            } => {
1041                wires(controls, out);
1042                for child in children {
1043                    visit(child, out);
1044                }
1045            }
1046            WireNode::Branch {
1047                controls, children, ..
1048            } => {
1049                out.push("alternatives[".into());
1050                wires(controls, out);
1051                for child in children {
1052                    out.push("arm[".into());
1053                    visit(child, out);
1054                    out.push("]".into());
1055                }
1056                out.push("]".into());
1057            }
1058            WireNode::Chain {
1059                meta,
1060                controls,
1061                child,
1062            } => {
1063                let label = match meta {
1064                    JniChain::Optional(JniOptional::Flag(_)) => "optional:flag".into(),
1065                    JniChain::Optional(JniOptional::Niche { reservation, .. }) => {
1066                        let index =
1067                            prebindgen_flat::wire_definitions(std::iter::once(child.as_ref()))
1068                                .position(|wire| wire.name == reservation.discriminator);
1069                        format!("optional:niche:{:?}:{index:?}", reservation.niche)
1070                    }
1071                    JniChain::Sequence(delivery) => format!("sequence:{delivery:?}"),
1072                    JniChain::Bulk(_) => {
1073                        let mut fields = group::row_fields(child)
1074                            .map(|w| {
1075                                type_key(
1076                                    &bulk::field_type(&w.kind).expect("prepared bulk storage type"),
1077                                )
1078                            })
1079                            .collect::<Vec<_>>();
1080                        if fields.is_empty() {
1081                            fields.push("u8".into());
1082                        }
1083                        format!("bulk:{fields:?}")
1084                    }
1085                };
1086                out.push(label);
1087                out.push("[".into());
1088                wires(controls, out);
1089                visit(child, out);
1090                out.push("]".into());
1091            }
1092        }
1093    }
1094    let mut out = Vec::new();
1095    visit(node, &mut out);
1096    out
1097}
