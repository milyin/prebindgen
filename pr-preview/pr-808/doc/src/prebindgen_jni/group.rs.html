6/// Meaning and encoding of a parent with one child description.
7#[derive(Clone, Debug)]
8pub enum JniChain {
9    /// Zero or one payload occurrence.
10    Optional(JniOptional),
11    /// Zero or more element occurrences.
12    Sequence(JniSequence),
13}
14/// Meaning of an ordered collection of child descriptions.
15#[derive(Clone, Debug)]
16pub enum JniBranch {
17    /// Related values or source fields, all belonging to the parent.
18    Group(JniGroup),
19    /// Exactly one selected arm; at least one arm is required.
20    Alternatives,
21}
22/// Relationships between the child values of a group.
23#[derive(Clone, Debug)]
24pub enum JniGroup {
25    /// Ordered conversion components, such as constructor arguments or output views.
26    Values,
27    /// An explicitly named output view.
28    Named(String),
29    /// Source record fields. JNI may associate the record with a Kotlin data class.
30    Fields(Option<DataGroup>),
31    /// One named source field.
32    Field(syn::Ident),
33}
34pub(crate) fn validate_selector(wire: &Wire<JniWire>) -> Result<()> {
35    if !matches!(wire.kind, JniWire::Selector { .. }) {
36        return Err("selector requires the adapter's integer tag representation".into());
37    }
38    wire.kind.validate()
39}
40/// How this optional occurrence distinguishes absence from a present payload.
41#[derive(Clone, Debug)]
42pub enum JniOptional {
43    /// A dedicated 0/1 tag. The binding is also the normalized presence value.
44    Flag(syn::Ident),
45    /// An excluded representation in one payload value.
46    Niche {
47        /// Normalized presence local used by conversion conditions.
48        presence: syn::Ident,
49        /// Payload binding and reserved absent representation.
50        reservation: prebindgen_declarations::NicheReservation<JniNiche>,
51    },
52}
53impl JniOptional {
54    /// Binding of the normalized presence condition.
55    pub fn presence(&self) -> &syn::Ident {
56        match self {
57            Self::Flag(flag) => flag,
58            Self::Niche { presence, .. } => presence,
59        }
60    }
61    /// Binding whose value distinguishes absent and present states.
62    pub fn discriminator(&self) -> &syn::Ident {
63        match self {
64            Self::Flag(flag) => flag,
65            Self::Niche { reservation, .. } => &reservation.discriminator,
66        }
67    }
68    /// Reserved payload representation; a flag encoding has no reservation.
69    pub fn niche(&self) -> Option<&JniNiche> {
70        match self {
71            Self::Flag(_) => None,
72            Self::Niche { reservation, .. } => Some(&reservation.niche),
73        }
74    }
75    pub(crate) fn from_parts(
76        presence: syn::Ident,
77        discriminator: syn::Ident,
78        niche: Option<JniNiche>,
79        controls: &[Wire<JniWire>],
80    ) -> Result<Self> {
81        match niche {
82            Some(niche) if controls.is_empty() => Ok(Self::Niche {
83                presence,
84                reservation: prebindgen_declarations::NicheReservation {
85                    discriminator,
86                    niche,
87                },
88            }),
89            None if controls.len() == 1
90                && presence == discriminator
91                && controls[0].name == discriminator =>
92            {
93                validate_selector(&controls[0])?;
94                Ok(Self::Flag(presence))
95            }
96            _ => {
97                Err("optional encoding requires either one matching flag or a payload niche".into())
98            }
99        }
100    }
101}

103// Definitions in one native scope; sequence element definitions belong to row scopes.
104fn values(node: &WireNode<JniWire, JniChain, JniBranch>) -> std::vec::IntoIter<&Wire<JniWire>> {
105    fn visit<'a>(
106        node: &'a WireNode<JniWire, JniChain, JniBranch>,
107        out: &mut Vec<&'a Wire<JniWire>>,
108    ) {
109        match node {
110            WireNode::Leaf(wires) => out.extend(wires),
111            WireNode::Chain {
112                meta: JniChain::Sequence(_),
113                controls,
114                ..
115            } => out.extend(controls),
116            WireNode::Chain {
117                child, controls, ..
118            } => {
119                out.extend(controls);
120                visit(child, out);
121            }
122            WireNode::Branch {
123                children, controls, ..
124            } => {
125                out.extend(controls);
126                for child in children {
127                    visit(child, out);
128                }
129            }
130        }
131    }
132    let mut out = vec![];
133    visit(node, &mut out);
134    out.into_iter()
135}

170pub(crate) fn validate_layout(
171    node: &WireNode<JniWire, crate::JniChain, crate::JniBranch>,
172) -> Result<()> {
173    prebindgen_conversions::validate_wire_names(prebindgen_flat::wire_definitions(
174        std::iter::once(node),
175    ))?;
176    for wire in values(node) {
177        wire.kind.validate()?;
178    }
179    fn visit(
180        node: &WireNode<JniWire, crate::JniChain, crate::JniBranch>,
181        bindings: &mut std::collections::BTreeSet<String>,
182        in_fields: bool,
183        reserved: &mut std::collections::BTreeMap<String, Vec<JniNiche>>,
184    ) -> Result<()> {
185        match node {
186            WireNode::Leaf(wires) => {
187                for w in wires {
188                    if matches!(w.kind, JniWire::Rows { .. }) {
189                        return Err("array carrier requires a sequence node".into());
190                    }
191                }
192            }
193            WireNode::Branch {
194                meta: JniBranch::Alternatives,
195                children: alternatives,
196                controls,
197                ..
198            } => {
199                if alternatives.is_empty() {
200                    return Err("selection has no alternatives".into());
201                }
202                if let [selector] = controls.as_slice() {
203                    validate_selector(selector)?;
204                } else {
205                    return Err("alternatives require one selector control".into());
206                }
207                for arm in alternatives {
208                    visit(arm, bindings, false, reserved)?;
209                }
210            }
211            WireNode::Chain {
212                child: element,
213                controls,
214                meta: JniChain::Sequence(metadata),
215                ..
216            } => {
217                let [carrier] = controls.as_slice() else {
218                    return Err("sequence requires one carrier control".into());
219                };
220                let valid = match metadata {
221                    JniSequence::Buffered => matches!(carrier.kind, JniWire::Rows { .. }),
222                    JniSequence::Callback => {
223                        matches!(&carrier.kind, JniWire::Rows { condition,.. } if condition=="true")
224                    }
225                };
226                if !valid {
227                    return Err("sequence carrier control is incompatible with its delivery".into());
228                }
229                if values(element).len() == 0 {
230                    return Err("sequence requires a nonempty element layout".into());
231                }
232                crate::sequence::validate_buffered(element)?;
233                validate_layout(element)?;
234            }
235            WireNode::Chain {
236                child: element,
237                meta: JniChain::Optional(o),
238                controls,
239                ..
240            } => {
241                match o {
242                    JniOptional::Flag(presence) => {
243                        let [flag] = controls.as_slice() else {
244                            return Err("flag optional requires one control".into());
245                        };
246                        validate_selector(flag)?;
247                        if &flag.name != presence {
248                            return Err("optional flag reference does not match its control".into());
249                        }
250                    }
251                    JniOptional::Niche { .. } if !controls.is_empty() => {
252                        return Err("niche optional must not have controls".into())
253                    }
254                    _ => {}
255                }
256                if !bindings.insert(o.presence().to_string()) {
257                    return Err(format!(
258                        "duplicate optional presence binding `{}`",
259                        o.presence()
260                    ));
261                }
262                if let Some(niche) = o.niche() {
263                    let wire = crate::group::values(element)
264                        .find(|w| w.name == *o.discriminator())
265                        .ok_or("optional discriminator outside payload")?;
266                    wire.kind.validate_niche(niche)?;
267                    if wire.kind.available_niche().as_ref() == Some(niche) {
268                        return Err(
269                            "available JNI niche is already reserved by an optional node".into(),
270                        );
271                    }
272                    let values = reserved.entry(o.discriminator().to_string()).or_default();
273                    if values.contains(niche) {
274                        return Err("duplicate JNI niche reservation on one wire".into());
275                    }
276                    values.push(niche.clone());
277                }
278                visit(element, bindings, false, reserved)?;
279            }
280            WireNode::Branch {
281                children,
282                meta: JniBranch::Group(metadata),
283                controls,
284                ..
285            } => {
286                if !controls.is_empty() {
287                    return Err("group metadata does not support control wires".into());
288                }
289                if let JniGroup::Fields(Some(record)) = metadata {
290                    if !bindings.insert(record.root.clone()) {
291                        return Err(format!("duplicate Kotlin group binding `{}`", record.root));
292                    }
293                }
294                match metadata {
295                    JniGroup::Named(name) => {
296                        validate_name(name)?;
297                    }
298                    JniGroup::Field(_) if !in_fields => {
299                        return Err(
300                            "named field must be a direct child of a field collection".into()
301                        )
302                    }
303                    JniGroup::Fields(_) => {
304                        let mut fields = std::collections::BTreeSet::new();
305                        for child in children.as_slice() {
306                            let WireNode::Branch {
307                                meta: JniBranch::Group(JniGroup::Field(name)),
308                                ..
309                            } = child
310                            else {
311                                return Err("field collection requires named field children".into());
312                            };
313                            if !fields.insert(name.unraw().to_string()) {
314                                return Err(format!("duplicate field `{name}`"));
315                            }
316                        }
317                    }
318                    _ => {}
319                }
320
321                for child in children {
322                    visit(
323                        child,
324                        bindings,
325                        matches!(metadata, JniGroup::Fields(_)),
326                        reserved,
327                    )?;
328                }
329            }
330        }
331        Ok(())
332    }
333    visit(
334        node,
335        &mut Default::default(),
336        false,
337        &mut Default::default(),
338    )
339}
