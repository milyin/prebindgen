56/// Resolve and bind one input with no generator-wide defaults.
57pub fn bind_input(
58    flat: &mut JniBindings,
59    codec: &JniCodec,
60    function: &str,
61    parameter: &str,
62    tree: &JniInputConfig,
63    extra: &(),
64) -> Result<()> {
65    let defaults = DeclJniDefaults::new(flat, vec![], vec![], vec![])?;
66    let ty = &flat
67        .function(function)?
68        .params
69        .iter()
70        .find(|p| p.name == parameter)
71        .ok_or("unknown parameter")?
72        .ty
73        .ty;
74    let tree = defaults.resolve_input(flat, codec, ty, tree, extra)?;
75    prebindgen_declarations::bind_input(flat, codec, function, parameter, &tree, extra)
76}
77/// Resolve and bind one output with no generator-wide defaults.
78pub fn bind_output(
79    flat: &mut JniBindings,
80    codec: &JniCodec,
81    function: &str,
82    tree: &JniOutputConfig,
83    extra: &(),
84) -> Result<()> {
85    let defaults = DeclJniDefaults::new(flat, vec![], vec![], vec![])?;
86    let tree =
87        defaults.resolve_output(flat, codec, &flat.function(function)?.ret.ty, tree, extra)?;
88    prebindgen_declarations::bind_output(flat, codec, function, &tree, extra)
89}
