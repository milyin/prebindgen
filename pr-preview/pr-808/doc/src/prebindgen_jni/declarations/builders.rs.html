13/// Adapter-specific options applied after the common expansion is built.
14#[derive(Clone, Debug, Default)]
15#[non_exhaustive]
16pub struct JniExpansionOptions {
17    /// Pack this expansion using the JNI bulk carrier.
18    pub bulk: bool,
19}
20/// Select the JNI bulk carrier for an input or output expansion.
21pub trait JniBulkExpansion: Sized {
22    /// Group the expansion in a bulk C-representation carrier.
23    /// Constructor, getter, ownership and local conversion rules are retained.
24    /// Ordinary output collections combine expansions into one frame. Callback
25    /// delivery keeps packing inside each element: bulk `Row` gives one frame
26    /// per row of `Vec<Row>`; bulk `Vec<Row>` gives one frame per inner collection
27    /// of `Vec<Vec<Row>>`.
28    fn bulk(self) -> Self;
29}
30impl JniBulkExpansion for DeclJniExpandParam {
31    fn bulk(mut self) -> Self {
32        self.extension.get_or_insert_with(Default::default).bulk = true;
33        self
34    }
35}
36impl JniBulkExpansion for DeclJniExpandReturn {
37    fn bulk(mut self) -> Self {
38        self.extension.get_or_insert_with(Default::default).bulk = true;
39        self
40    }
41}
42/// Start input expansion using this adapter's node constructors.
43pub fn expand_param(ty: Type) -> DeclJniExpandParam {
44    prebindgen_declarations::DeclExpandParam::new(
45        ty,
46        |function: DeclJniFunction| JniInputConfig::call(function.reference_name(false)),
47        || JniInputConfig::Leaf(JniInputLeafOp::Value),
48        |ty, alternatives, extension| {
49            let mut input =
50                alternatives.build(JniInputConfig::selector, JniInputConfig::auto, |child| {
51                    JniInputConfig::Unary {
52                        operation: JniInputUnary::Temporary,
53                        child: Box::new(child),
54                    }
55                });
56            if extension.is_some_and(|options| options.bulk) {
57                input = JniInputConfig::Unary {
58                    operation: JniInputUnary::Bulk,
59                    child: Box::new(input),
60                };
61            }
62            JniInputConfig::for_type(ty, input)
63        },
64    )
65}
66/// Start return expansion using this adapter's node constructors.
67pub fn expand_return(ty: Type) -> DeclJniExpandReturn {
68    prebindgen_declarations::DeclExpandReturn::new(
69        ty,
70        |function| {
71            prebindgen_declarations::getter_view(
72                function,
73                |name| JniOutputConfig::getter(name, JniOutputConfig::auto()),
74                |name, child| JniOutputConfig::Unary {
75                    operation: JniOutputUnary::Named { name },
76                    child: Box::new(child),
77                },
78            )
79        },
80        || JniOutputConfig::Leaf(JniOutputLeafOp::Value),
81        |policy| JniOutputConfig::Leaf(JniOutputLeafOp::Projected(policy)),
82        super::configuration::owned_field_view,
83        |ty, views, extension| {
84            let mut output = JniOutputConfig::combine(views);
85            if extension.is_some_and(|options| options.bulk) {
86                output = JniOutputConfig::Unary {
87                    operation: JniOutputUnary::Bulk,
88                    child: Box::new(output),
89                };
90            }
91            JniOutputConfig::for_type(ty, output)
92        },
93    )
94}
95/// Decompose a source accessor's returned record into named field views.
96pub fn fields(function: syn::Ident) -> DeclJniFields {
97    prebindgen_declarations::DeclFields::new(
98        function,
99        |output| match output {
100            JniOutputConfig::Unary {
101                operation: JniOutputUnary::ConfigForType { ty },
102                child: output,
103            } if matches!(output.as_ref(),JniOutputConfig::Branch {operation:JniOutputBranch::Combine,children} if children.is_empty()) => {
104                JniOutputConfig::for_type(*ty, JniOutputConfig::Leaf(JniOutputLeafOp::Discard))
105            }
106            output => output,
107        },
108        |function, overrides, _extension| {
109            let (fields, children) = overrides.into_iter().unzip();
110            JniOutputConfig::getter(
111                function,
112                JniOutputConfig::Branch {
113                    operation: JniOutputBranch::ConfigNamedFields(JniNamedFields {
114                        fields,
115                        mode: prebindgen_declarations::ProjectionMode::Borrowed,
116                    }),
117                    children,
118                },
119            )
120        },
121    )
122}

128/// Callback declaration containing this language's argument and result trees.
129/// ```
130/// use prebindgen_jni::{callback,expand_param,expand_return,fun};
131/// let handler = callback!(impl Fn(Packet) -> Decision)
132///     .name("PacketHandler")

135/// let _ = fun!(register).expand_param("callback", handler);
136/// ```

138impl prebindgen_declarations::CallbackInput<JniInputConfig, JniOutputConfig> for JniInputConfigOps {
139    fn callback_input(declaration: DeclJniCallback) -> JniInputConfig {
140        prebindgen_declarations::callback_input(
141            declaration,
142            JniInputConfig::auto,
143            |children| JniOutputConfig::Branch {
144                operation: JniOutputBranch::Arguments,
145                children,
146            },
147            |metadata, domain, codomain| JniInputConfig::Arrow {
148                operation: JniInputArrow::ConfigCallback(Box::new(metadata)),
149                domain: Box::new(domain),
150                codomain: Box::new(codomain),
151            },
152            JniInputConfig::for_type,
153        )
154    }
