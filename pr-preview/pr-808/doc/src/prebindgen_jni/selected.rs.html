1//! Active physical values in one native row. The tree supplies all predicates;
2//! no runtime ownership registry or row-by-row plan is required.
3use super::*;
4use crate::kotlin::JniNode;
5use prebindgen_flat::WireNode;

7#[derive(Clone, Debug)]
8pub(crate) enum Test {
9    Arm(usize, i32),
10    Present(usize, Option<JniNiche>),
11}
12impl Test {
13    pub(crate) fn index(&self) -> usize {
14        match self {
15            Self::Arm(i, _) | Self::Present(i, _) => *i,
16        }
17    }
18    pub(crate) fn rust(&self, wire: &JniWire, value: TokenStream) -> TokenStream {
19        match self {
20            Self::Arm(_, arm) => quote!((#value)==#arm),
21            Self::Present(_, Some(niche)) => niche.present(wire, value),
22            Self::Present(_, None) => quote!((#value)==1),
23        }
24    }
25    pub(crate) fn kotlin(&self, value: &str) -> String {
26        match self {
27            Self::Arm(_, arm) => format!("({value}) == {arm}"),
28            Self::Present(_, Some(JniNiche::Null)) => format!("({value}) != null"),
29            Self::Present(_, Some(JniNiche::Integer(n))) => format!(
30                "({value}).toLong() != {}",
31                crate::wire::jvm::long_literal(*n)
32            ),
33            Self::Present(_, None) => format!("({value}) == 1"),
34        }
35    }
36}
37#[derive(Clone, Debug)]
38pub(crate) struct Selection {
39    pub(crate) tests: Vec<Vec<Test>>,
40    kinds: Vec<JniWire>,
41}
42impl Selection {
43    pub(crate) fn new(node: &JniNode) -> Result<Self> {
44        let wires = group::row_fields(node).collect::<Vec<_>>();
45        let indices = wires
46            .iter()
47            .enumerate()
48            .map(|(i, w)| (w.name.to_string(), i))
49            .collect::<BTreeMap<_, _>>();
50        let mut tests = vec![vec![]; wires.len()];
51        fn visit(
52            node: &JniNode,
53            path: &[Test],
54            indices: &BTreeMap<String, usize>,
55            out: &mut [Vec<Test>],
56        ) -> Result<()> {
57            let set = |wires: &[prebindgen_flat::Wire<JniWire>], out: &mut [Vec<Test>]| {
58                for w in wires {
59                    out[indices[&w.name.to_string()]] = path.to_vec();
60                }
61            };
62            match node {
63                WireNode::Leaf(wires) => set(wires, out),
64                WireNode::Chain {
65                    controls,
66                    meta: JniChain::Sequence(_) | JniChain::Bulk(_),
67                    ..
68                } => set(controls, out),
69                WireNode::Chain {
70                    controls,
71                    child,
72                    meta: JniChain::Optional(optional),
73                } => {
74                    set(controls, out);
75                    let mut next = path.to_vec();
76                    let donor = indices[&optional.discriminator().to_string()];
77                    next.push(Test::Present(donor, optional.niche().cloned()));
78                    visit(child, &next, indices, out)?;
79                    // The discriminator must be readable before testing presence.
80                    out[donor].retain(|t| !matches!(t,Test::Present(i,_) if *i==donor));
81                }
82                WireNode::Branch {
83                    controls,
84                    children,
85                    meta,
86                } => {
87                    set(controls, out);
88                    for (i, child) in children.iter().enumerate() {
89                        let mut next = path.to_vec();
90                        if matches!(meta, JniBranch::Enum(_) | JniBranch::Alternatives) {
91                            next.push(Test::Arm(indices[&controls[0].name.to_string()], i as i32));
92                        }
93                        visit(child, &next, indices, out)?;
94                    }
95                }
96            }
97            Ok(())
98        }
99        visit(node, &[], &indices, &mut tests)?;
100        Ok(Self {
101            tests,
102            kinds: wires.iter().map(|w| w.kind.clone()).collect(),
103        })
104    }
105    pub(crate) fn rust(
106        &self,
107        index: usize,
108        mut value: impl FnMut(usize) -> TokenStream,
109    ) -> TokenStream {
110        self.tests[index]
111            .iter()
112            .map(|t| t.rust(&self.kinds[t.index()], value(t.index())))
113            .reduce(|a, b| quote!((#a)&&(#b)))
114            .unwrap_or_else(|| quote!(true))
115    }
116    pub(crate) fn kotlin(&self, index: usize, mut value: impl FnMut(usize) -> String) -> String {
117        let tests = self.tests[index]
118            .iter()
119            .map(|t| t.kotlin(&value(t.index())))
120            .collect::<Vec<_>>();
121        if tests.is_empty() {
122            "true".into()
123        } else {
124            tests.join(" && ")
125        }
126    }
127    pub(crate) fn order(&self) -> Result<Vec<usize>> {
128        fn add(i: usize, s: &Selection, states: &mut [u8], out: &mut Vec<usize>) -> Result<()> {
129            if states[i] == 2 {
130                return Ok(());
131            }
132            if states[i] == 1 {
133                return Err("cyclic selected-value discriminator".into());
134            }
135            states[i] = 1;
136            for t in &s.tests[i] {
137                add(t.index(), s, states, out)?;
138            }
139            states[i] = 2;
140            out.push(i);
141            Ok(())
142        }
143        let mut states = vec![0; self.tests.len()];
144        let mut order = vec![];
145        for i in 0..states.len() {
146            add(i, self, &mut states, &mut order)?;
147        }
148        Ok(order)
149    }
150}
