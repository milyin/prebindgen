6/// Borrowed roots at function and callback boundaries. Each root keeps its direction.
7#[derive(Clone, Copy)]
8pub(crate) enum BoundaryNode<'a> {
9    Input(&'a JniInputWireNode),
10    Output(&'a JniOutputWireNode),
11}
12impl<'a> BoundaryNode<'a> {
13    pub(crate) fn wires(self) -> Vec<&'a prebindgen_conversions::Wire<JniWire>> {
14        match self {
15            Self::Input(node) => {
16                prebindgen_conversions::wire_definitions(std::iter::once(node)).collect()
17            }
18            Self::Output(node) => {
19                prebindgen_conversions::wire_definitions(std::iter::once(node)).collect()
20            }
21        }
22    }
23}

25/// Prepared boundary roots, including typed error converters and declarations
26/// whose callback interfaces are emitted without a direct exported occurrence.
27pub(crate) fn nodes<I: KotlinInputConversion, O: KotlinOutputConversion>(
28    flat: &Flat<JniSchema<I, O>>,
29) -> Result<Vec<BoundaryNode<'_>>> {
30    let mut nodes = Vec::new();
31    for source in flat.functions.values() {
32        for binding in &source.metadata {
33            let call = binding.call()?;
34            for plan in &call.inputs {
35                nodes.push(BoundaryNode::Input(plan.input_node()));
36            }
37            let outcomes = call.output.output_outcomes();
38            nodes.extend(
39                std::iter::once(&outcomes.success)
40                    .chain(outcomes.source_error.iter())
41                    .chain(outcomes.binding_failure.iter())
42                    .map(BoundaryNode::Output),
43            );
44            if let Some(output) = binding
45                .metadata
46                .error
47                .as_deref()
48                .and_then(|e| e.output.as_ref())
49            {
50                nodes.push(BoundaryNode::Output(&output.output_outcomes().success));
51            }
52        }
53    }
54    for input in crate::callback::declarations(flat)? {
55        nodes.push(BoundaryNode::Input(input.input_node()));
56    }
57    Ok(nodes)
58}

60/// Metadata schema for a Kotlin/JNI binding model.
61/// `InConverter` and `OutConverter` select the converters stored in each export.
62/// Source parameter/return attachments are unit; type metadata keeps the JNI
63/// representation. Default containers also accept custom converters through their
64/// owned containers. No schema value is needed to construct Flat.
65pub struct JniSchema<InConverter = JniInputConverter, OutConverter = JniOutputConverter>(
66    std::marker::PhantomData<fn() -> (InConverter, OutConverter)>,
67);
68impl<InConverter, OutConverter> prebindgen_flat::Schema for JniSchema<InConverter, OutConverter> {
69    type Parameter = ();
70    type Return = ();
71    type Module = JniModule;
72    type Function = Vec<JniBinding<InConverter, OutConverter>>;
73    type Type = JniType;
74    type Field = data::JniField;
75    type Variant = ();
76    type Record = ();
77    type Enum = ();
78    type PayloadEnum = ();
79}

103/// Kotlin/JNI declaration information shared by occurrences of a Rust type.

116    pub(crate) errors: BTreeMap<bool, Rc<errors::ErrorBinding>>,
117    pub(crate) error_class: Option<String>,

141// Borrow the actual Flat type table, independent of its export converter types.
142pub(crate) type JniTypes = BTreeMap<String, prebindgen_flat::TypeInfo<JniType, data::JniField>>;
143
144pub(crate) fn handle_classes(types: &JniTypes) -> impl Iterator<Item = (&syn::Type, &Class)> {
145    let mut classes = types
146        .values()
147        .filter_map(|source| {
148            source
149                .metadata
150                .class
151                .as_deref()
152                .filter(|class| matches!(class.representation, ClassRepresentation::Handle { .. }))
153                .map(|class| (&source.ty, class))
154        })
155        .collect::<Vec<_>>();
156    // Preserve emitted declaration order across captured and external types.
157    classes.sort_by_key(|(ty, _)| type_key(ty));
158    classes.into_iter()
159}

161pub(crate) fn source_handle<'a>(
162    types: &'a JniTypes,
163    ty: &syn::Type,
164) -> Result<(&'a syn::Type, &'a Class)> {
165    let source = types
166        .values()
167        .find(|entry| prebindgen_flat::same_type(&entry.ty, ty))
168        .ok_or_else(|| format!("unregistered JNI source type `{}`", type_key(ty)))?;
169    let class = source
170        .metadata
171        .class
172        .as_deref()
173        .filter(|class| matches!(class.representation, ClassRepresentation::Handle { .. }))
174        .ok_or_else(|| format!("no handle class for `{}`", type_key(ty)))?;
175    Ok((&source.ty, class))
176}

178pub(crate) fn source_type<'a>(
179    types: &'a JniTypes,
180    reference: &prebindgen_flat::TypeRef,
181) -> Result<&'a prebindgen_flat::TypeInfo<JniType, data::JniField>> {
182    types
183        .get(reference.key())
184        .ok_or_else(|| format!("missing JNI source type entry `{}`", reference.key()))
185}
186pub(crate) fn source_data_class<'a>(
187    types: &'a JniTypes,
188    reference: &prebindgen_flat::TypeRef,
189) -> Result<(&'a prebindgen_flat::Record<data::JniField>, &'a Class)> {
190    let source = source_type(types, reference)?;
191    let class = source
192        .metadata
193        .class
194        .as_deref()
195        .filter(|class| matches!(class.representation, ClassRepresentation::Data { .. }))
196        .ok_or("record conversion requires a declared data class")?;
197    match &source.definition {
198        prebindgen_flat::TypeDefinition::Record(record) => Ok((record, class)),
199        _ => Err(format!("`{}` is not a source record", reference.key())),
200    }
201}
202pub(crate) fn source_enum<'a>(
203    types: &'a JniTypes,
204    reference: &prebindgen_flat::TypeRef,
205) -> Result<&'a prebindgen_flat::PayloadEnum<(), data::JniField>> {
206    match &source_type(types, reference)?.definition {
207        prebindgen_flat::TypeDefinition::PayloadEnum(enumeration) => Ok(enumeration),
208        _ => Err(format!(
209            "`{}` is not a source payload enum",
210            reference.key()
211        )),
212    }
