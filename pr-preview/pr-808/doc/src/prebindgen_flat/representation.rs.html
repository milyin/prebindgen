1//! Named representation schemas shared by type declarations and conversions.
2use crate::{type_key, Result};
3use std::collections::BTreeMap;
4
5/// Stable identity of a resolved representation; independent of generated names.
6#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
7pub struct RepresentationId(pub String);
8
9/// Typed step identifying a logical component independently of generated names.
10#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
11pub enum ComponentSegment {
12    /// A source record field.
13    Field(String),
14    /// A constructor argument.
15    Argument(String),
16    /// An input selector arm.
17    Alternative(usize),
18    /// One simultaneous output view.
19    View(usize),
20    /// A source getter's result.
21    Getter(String),
22    /// An input selector's discriminator.
23    Selector,
24    /// An adapter-defined logical component with no implied source projection.
25    Named(String),
26}
27impl ComponentSegment {
28    /// Default presentation name; applications may assign a different name in the schema.
29    pub fn name(&self) -> String {
30        match self {
31            Self::Field(name) | Self::Argument(name) | Self::Getter(name) | Self::Named(name) => {
32                name.clone()
33            }
34            Self::Alternative(index) | Self::View(index) => index.to_string(),
35            Self::Selector => "selector".into(),
36        }
37    }
38}

40/// Logical component identity, including nesting. Several wires may implement it.
41#[derive(Clone, Debug, Default, PartialEq, Eq, PartialOrd, Ord)]
42pub struct ComponentPath(pub Vec<ComponentSegment>);
43impl ComponentPath {
44    /// Prefix a component with a containing field, view, or alternative.
45    pub fn prepend(&mut self, segment: ComponentSegment) {
46        self.0.insert(0, segment);
47    }
48}

50/// A logical component in a representation, not an executable conversion.
51/// Every component must have at least one wire when a conversion selects this
52/// representation. Partial projections must select their own representation.
53#[derive(Clone, Debug)]
54pub struct Component {
55    /// Stable path used by conversion wires.
56    pub path: ComponentPath,
57    /// Public presentation name, separate from generated Rust local names.
58    pub name: String,
59    /// Optional source record field path. Derived views need no field association.
60    pub source_fields: Option<Vec<String>>,
61    /// Representation of the component, when it has a separately declared shape.
62    pub representation: Option<RepresentationId>,
63}

65/// Resolved shape used to generate a destination type or describe an occurrence.
66#[derive(Clone, Debug)]
67pub struct Representation<R> {
68    /// Qualified Rust type represented by this shape.
69    pub source_type: syn::Type,
70    /// Logical components; declaration order is retained.
71    pub components: Vec<Component>,
72    /// Adapter-specific type declaration and materialization information.
73    pub data: R,
74}
75
76/// Representation registry and optional canonical defaults by Rust type.
77#[derive(Clone, Debug)]
78pub struct Representations<R> {
79    entries: BTreeMap<RepresentationId, Representation<R>>,
80    canonical: BTreeMap<String, RepresentationId>,
81}
82impl<R> Default for Representations<R> {
83    fn default() -> Self {
84        Self {
85            entries: BTreeMap::new(),
86            canonical: BTreeMap::new(),
87        }
88    }
89}
90impl<R> Representations<R> {
91    /// Insert or replace a shape. Reject duplicate component identities atomically.
92    pub fn insert(&mut self, id: RepresentationId, shape: Representation<R>) -> Result<()> {
93        let mut paths = std::collections::BTreeSet::new();
94        for component in &shape.components {
95            if !paths.insert(&component.path) {
96                return Err(format!("duplicate component in representation `{}`", id.0));
97            }
98        }
99        self.entries.insert(id, shape);
100        Ok(())
101    }
102    /// Resolve a representation without interpreting its adapter data.
103    pub fn get(&self, id: &RepresentationId) -> Result<&Representation<R>> {
104        self.entries
105            .get(id)
106            .ok_or_else(|| format!("unknown representation `{}`", id.0))
107    }
108    /// Iterate in identity order, for generation of selected type declarations.
109    pub fn iter(&self) -> impl Iterator<Item = (&RepresentationId, &Representation<R>)> {
110        self.entries.iter()
111    }
112    /// Designate an existing representation as its source type's default.
113    /// This does not populate or change any occurrence conversion.
114    pub fn set_canonical(&mut self, id: &RepresentationId) -> Result<()> {
115        let key = type_key(&self.get(id)?.source_type);
116        self.canonical.insert(key, id.clone());
117        Ok(())
118    }
119    /// Look up a preparation default; generation uses explicit occurrence references.
120    pub fn canonical(&self, ty: &syn::Type) -> Option<&RepresentationId> {
121        self.canonical.get(&type_key(ty))
122    }
123    /// Validate nested references after all declarations have been registered.
124    pub fn validate(&self) -> Result<()> {
125        for shape in self.entries.values() {
126            for component in &shape.components {
127                if let Some(id) = &component.representation {
128                    self.get(id)?;
129                }
130            }
131        }
132        for (ty, id) in &self.canonical {
133            if type_key(&self.get(id)?.source_type) != *ty {
134                return Err(format!(
135                    "canonical representation `{}` changed source type",
136                    id.0
137                ));
138            }
139        }
140        Ok(())
141    }
142}
