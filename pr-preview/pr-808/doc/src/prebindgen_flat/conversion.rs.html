6/// Transport metadata that supplies the Rust carrier type of a wire.
7/// The returned type describes the boundary value, not the source value being converted.
8/// For example, a boolean conversion may use `u8` as its Rust carrier type.
9///
10/// ```
11/// use prebindgen_flat::WireKind;
12/// struct BoolCarrier;
13/// impl WireKind for BoolCarrier {
14///     fn rust_type(&self) -> syn::Type { syn::parse_quote!(u8) }
15/// }
16/// let ty = BoolCarrier.rust_type();
17/// assert_eq!(quote::quote!(#ty).to_string(), "u8");
18/// ```
19pub trait WireKind {
20    /// Rust type used to declare this wire in a wrapper signature or result structure.
21    fn rust_type(&self) -> syn::Type;
22}
23impl WireKind for syn::Type {
24    fn rust_type(&self) -> syn::Type {
25        self.clone()
26    }
27}
28
29/// One boundary parameter or result component used by wrapper emission.
30///
31/// `W` must implement [`WireKind`]. Call [`Wire::rust_type`]
32/// to obtain the Rust carrier type supplied by the kind.
33///
34/// ```
35/// use prebindgen_flat::Wire;
36/// let wire: Wire<syn::Type> = Wire {
37///     name: syn::parse_quote!(value),
38///     kind: syn::parse_quote!(i64),
39/// };
40/// let ty = wire.rust_type();
41/// assert_eq!(quote::quote!(#ty).to_string(), "i64");
42/// ```
43/// A wire kind without a Rust carrier type is rejected:
44///
45/// ```compile_fail
46/// use prebindgen_flat::Wire;
47/// struct Untyped;
48/// let wire = Wire { name: syn::parse_quote!(value), kind: Untyped };
49/// ```
50#[derive(Clone, Debug)]
51pub struct Wire<W: WireKind> {
52    /// Generated identifier used by conversion code and wrapper signatures.
53    pub name: syn::Ident,
54    /// Transport description implementing [`WireKind`].
55    /// Logical fields, selection rules and ownership behavior belong inside the converter.
56    pub kind: W,
57}
58
59impl<W: WireKind> Wire<W> {
60    /// Rust carrier type supplied by this wire's [`WireKind`].
61    /// Used to declare a wrapper parameter or result field.
62    pub fn rust_type(&self) -> syn::Type {
63        self.kind.rust_type()
64    }
65}

67/// Check that every bound function has an input converter for each parameter
68/// and an output converter, including unit results.
69///
70/// Functions with entirely empty slots remain source-only helpers. This check
71/// neither builds conversions nor changes the model. Adapters call it before
72/// writing wrappers. It does not validate code fragments supplied directly by callers.
73pub fn validate_bindings<I: InputConversion, O: OutputConversion>(
74    flat: &crate::Flat<I, O>,
75) -> Result<()> {
76    for f in flat.functions.values() {
77        if f.ret.slot.is_some() {
78            for p in &f.params {
79                if p.ty.slot.is_none() {
80                    return Err(format!(
81                        "`{}`: parameter `{}` has no input converter",
82                        f.name, p.name
83                    ));
84                }
85            }
86        } else if f.params.iter().any(|p| p.ty.slot.is_some()) {
87            return Err(format!(
88                "`{}`: bound inputs have no output converter",
89                f.name
90            ));
91        }
92    }
93    Ok(())
94}
95/// Format generated tokens as a Rust source file.
96///
97/// `tokens` must be a sequence of Rust items. Returns a syntax error if they do not form a file. This does not
98/// resolve names or type-check generated code; the binding crate must compile it.
99pub fn rust_source(tokens: TokenStream) -> Result<String> {
100    let file = syn::parse2(tokens).map_err(|e| format!("generated Rust: {e}"))?;
101    Ok(prettyplease::unparse(&file))
102}

104/// Prepared input conversion with a generator-defined request protocol.
105///
106/// Every successful request returns one Rust expression, including `()` for
107/// a no-op. Statements may appear inside a block expression. The caller decides
108/// how expressions are combined and binds any values needed by later requests.
109/// `expected_type` is the caller's expectation, not a type-checking guarantee:
110/// the implementation may use it to configure emission or reject a request.
111/// Rust compilation performs the final type check. No implicit casts are required.
112///
113/// Request enums belong to the generator. Each variant can carry only the names,
114/// settings or other data it needs; no common context or phase order is imposed.
115/// Their lifetime parameter allows requests to borrow that data.
116///
117/// The wire list is stable and includes conditional wires in signature order.
118/// Wire request positions index this exact list; invalid positions must return
119/// an error. Activity, access modes and other policies may be requested through
120/// generator-defined variants rather than being interpreted by Flat.
121///
122/// A reusable owned-box converter can define its own small request vocabulary.
123/// The pointee is preparation data, not a hardcoded application type. The slot
124/// request borrows an error message only for validation; acquisition needs no
125/// context. The caller supplies `expected_type` to every request and binds the
126/// resulting value itself. This example chooses syntactic type matching as its
127/// policy; other implementations may use the expectation to configure emission.
128///
129/// ```
130/// use prebindgen_flat::{bind_fragment, same_type, InputConversion, Result, Wire, WireKind};
131/// use proc_macro2::TokenStream;
132/// use quote::quote;
133/// use syn::{parse_quote, Type};
134///
135/// enum Carrier { OpaquePointer }
136/// impl WireKind for Carrier {
137///     fn rust_type(&self) -> Type { parse_quote!(*mut ::std::ffi::c_void) }
138/// }
139/// enum SlotRequest<'a> {
140///     Validate { null_message: &'a str },
141///     TakeOwnership,
142/// }
143/// enum WireRequest { Value, Active }
144/// struct OwnedBoxInput {
145///     pointee: Type,
146///     wires: [Wire<Carrier>; 1],
147/// }
148/// fn expect(actual: &Type, expected: &Type) -> Result<()> {
149///     if same_type(actual, expected) { Ok(()) }
150///     else { Err("unsupported expected type".into()) }
151/// }
152/// impl InputConversion for OwnedBoxInput {
153///     type Wire = Carrier;
154///     type SlotFragment<'a> = SlotRequest<'a>;
155///     type WireFragment<'a> = WireRequest;
156///     fn wires(&self) -> &[Wire<Carrier>] { &self.wires }
157///     fn code_fragment_slot(&self, mode: SlotRequest<'_>, expected: &Type)
158///         -> Result<TokenStream>
159///     {
160///         let name = &self.wires[0].name;
161///         let pointee = &self.pointee;
162///         match mode {
163///             SlotRequest::Validate { null_message } => {
164///                 expect(&parse_quote!(()), expected)?;
165///                 Ok(quote!({
166///                     if #name.is_null() { return Err(#null_message.into()); }
167///                 }))
168///             }
169///             SlotRequest::TakeOwnership => {
170///                 expect(&parse_quote!(::std::boxed::Box<#pointee>), expected)?;
171///                 // The caller guarantees a live, uniquely owned allocation
172///                 // created by compatible Box::into_raw. Validation alone cannot.
173///                 Ok(quote!(unsafe {
174///                     ::std::boxed::Box::<#pointee>::from_raw(#name.cast::<#pointee>())
175///                 }))
176///             }
177///         }
178///     }
179///     fn code_fragment_wire(&self, pos: usize, mode: WireRequest, expected: &Type)
180///         -> Result<TokenStream>
181///     {
182///         let wire = self.wires.get(pos).ok_or("invalid wire position")?;
183///         match mode {
184///             WireRequest::Value => {
185///                 expect(&wire.rust_type(), expected)?;
186///                 let name = &wire.name;
187///                 Ok(quote!(#name))
188///             }
189///             WireRequest::Active => {
190///                 expect(&parse_quote!(Option<bool>), expected)?;
191///                 Ok(quote!(None)) // An expression meaning unconditional activity.
192///             }
193///         }
194///     }
195/// }
196/// let slot = OwnedBoxInput {
197///     pointee: parse_quote!(u64), // Supplied by preparation from the source type.
198///     wires: [Wire { name: parse_quote!(raw_value), kind: Carrier::OpaquePointer }],
199/// };
200/// let validation = slot.code_fragment_slot(
201///     SlotRequest::Validate { null_message: "null owned handle" }, &parse_quote!(()))?;
202/// let expected = parse_quote!(::std::boxed::Box<u64>);
203/// let value = slot.code_fragment_slot(SlotRequest::TakeOwnership, &expected)?;
204/// let binding = bind_fragment(&parse_quote!(argument), &expected, value)?;
205/// let _: syn::Block = syn::parse2(quote!({ #validation; #binding }))?;
206/// let active = slot.code_fragment_wire(0, WireRequest::Active, &parse_quote!(Option<bool>))?;
207/// assert_eq!(active.to_string(), "None");
208/// let wire_value = slot.code_fragment_wire(0, WireRequest::Value, &slot.wires()[0].rust_type())?;
209/// assert_eq!(wire_value.to_string(), "raw_value");
210/// # Ok::<(), Box<dyn std::error::Error>>(())
211/// ```
212pub trait InputConversion {
213    /// Generator-defined boundary carrier.
214    type Wire: WireKind;
215    /// Requests concerning the whole input slot.
216    type SlotFragment<'a>;
217    /// Requests concerning one input wire.
218    type WireFragment<'a>;
219    /// Boundary inputs in signature order.
220    fn wires(&self) -> &[Wire<Self::Wire>];
221    /// Emit an expression for the whole slot.
222    fn code_fragment_slot(
223        &self,
224        mode: Self::SlotFragment<'_>,
225        expected_type: &syn::Type,
226    ) -> Result<TokenStream>;
227    /// Emit an expression for wire `pos`, or an error for an invalid position.
228    fn code_fragment_wire(
229        &self,
230        pos: usize,
231        mode: Self::WireFragment<'_>,
232        expected_type: &syn::Type,
233    ) -> Result<TokenStream>;
234}

236/// Prepared output conversion with a generator-defined request protocol.
237///
238/// Has the same expression and expected-type contract as [`InputConversion`],
239/// but its request enums may be entirely different. In particular, a request can
240/// carry the source-result identifier without imposing a shared context type.
241pub trait OutputConversion {
242    /// Generator-defined boundary carrier.
243    type Wire: WireKind;
244    /// Requests concerning the whole output slot.
245    type SlotFragment<'a>;
246    /// Requests concerning one output wire.
247    type WireFragment<'a>;
248    /// Boundary outputs in delivery order.
249    fn wires(&self) -> &[Wire<Self::Wire>];
250    /// Emit an expression for the whole slot.
251    fn code_fragment_slot(
252        &self,
253        mode: Self::SlotFragment<'_>,
254        expected_type: &syn::Type,
255    ) -> Result<TokenStream>;
256    /// Emit an expression for wire `pos`, or an error for an invalid position.
257    fn code_fragment_wire(
258        &self,
259        pos: usize,
260        mode: Self::WireFragment<'_>,
261        expected_type: &syn::Type,
262    ) -> Result<TokenStream>;
263}

265/// Bind an emitted expression to a caller-selected name and expected Rust type.
266///
267/// Rejects empty or statement-only fragments. The generated type annotation
268/// asks the Rust compiler to check the expectation in the wrapper's environment.
269pub fn bind_fragment(
270    target: &syn::Ident,
271    expected_type: &syn::Type,
272    fragment: TokenStream,
273) -> Result<TokenStream> {
274    let expression: syn::Expr =
275        syn::parse2(fragment).map_err(|error| format!("expected a Rust expression: {error}"))?;
276    Ok(quote!(let #target: #expected_type = #expression;))
277}
278impl<I: InputConversion, O: OutputConversion> crate::Function<I, O> {
279    /// Filled input slots, preserving source parameter order and grouping.
280    pub fn input_conversions(&self) -> Result<Vec<&I>> {
281        self.params
282            .iter()
283            .map(|p| {
284                p.ty.slot.as_ref().ok_or_else(|| {
285                    format!("`{}`: parameter `{}` has no conversion", self.name, p.name)
286                })
287            })
288            .collect()
289    }
290    /// Filled result slot, borrowed without requesting any generated code.
291    pub fn output_conversion(&self) -> Result<&O> {
292        let output = self
293            .ret
294            .slot
295            .as_ref()
296            .ok_or_else(|| format!("`{}`: result has no conversion", self.name))?;
297        Ok(output)
298    }
299    /// Boundary input wires read directly from the slots in parameter order.
300    pub fn inputs(&self) -> Result<impl Iterator<Item = &Wire<I::Wire>>> {
301        Ok(self
302            .input_conversions()?
303            .into_iter()
304            .flat_map(|i| i.wires()))
305    }
