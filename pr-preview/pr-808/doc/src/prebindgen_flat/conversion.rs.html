6/// One boundary parameter or result component used by wrapper emission.
7///
8/// ```
9/// use prebindgen_flat::{Condition, Wire};
10/// let wire = Wire {
11///     component: Default::default(),
12///     name: syn::parse_quote!(value),
13///     kind: "i64",
14///     active: Condition::default(),
15/// };
16/// assert_eq!(wire.kind, "i64");
17/// ```
18#[derive(Clone, Debug)]
19pub struct Wire<W> {
20    /// Logical component supplied by this wire; several wires may share a path.
21    pub component: crate::ComponentPath,
22    /// Generated identifier used by conversion code and wrapper signatures.
23    pub name: syn::Ident,
24    /// Adapter description of the carrier, supplied by a conversion.
25    /// The framework carries it into the conversion without interpreting it.
26    pub kind: W,
27    /// Input selector path; outputs normally use the always-active default.
28    pub active: Condition,
29}

31/// An input conversion split into call-wide validation, acquisition and construction.
32///
33/// Optional concrete storage implementing [`InputConversion`]. [`emit_call`] orders
34/// them across all parameters; wrapper writers do not assemble them themselves.
35#[derive(Clone, Debug)]
36pub struct InputPlan<W> {
37    /// Flattened boundary parameters in signature order.
38    pub wires: Vec<Wire<W>>,
39    /// Shared call acknowledgement, if this input needs one.
40    pub acknowledgement: Option<TokenStream>,
41    /// Validate and prepare every active leaf before any ownership transfer.
42    pub validation: TokenStream,
43    /// Infallible acquisition of all owned inputs, before constructors can run.
44    pub acquire: TokenStream,
45    /// Active foreign resources whose identities must be checked across all arguments.
46    pub resources: Vec<Resource>,
47    /// Argument construction after all acquisitions, including source constructor calls.
48    pub statements: TokenStream,
49    /// Rust expression passed to the source function after `statements`.
50    pub argument: TokenStream,
51}
52/// Output conversion whose owned temporaries survive until successful delivery.
53///
54/// Optional concrete storage implementing [`OutputConversion`].
55/// [`emit_call`] commits only after adapter delivery succeeds.
56#[derive(Clone, Debug)]
57pub struct OutputPlan<W> {
58    /// Delivery and return code prepared with this result slot.
59    pub delivery: Delivery,
60    /// Boundary results in delivery order; one entry per expression in `values`.
61    pub wires: Vec<Wire<W>>,
62    /// Transfer ownership only after all foreign delivery operations succeed.
63    pub commit: TokenStream,
64    /// Code producing owned output temporaries and evaluating observations before moves.
65    pub statements: TokenStream,
66    /// Delivery expressions in wire order; they must remain valid until commit.
67    pub values: Vec<TokenStream>,
68}
69/// A leaf is active only on its selected path. Adapters use the same condition
70/// for foreign-side resource locking and native conversion code.
71#[derive(Clone, Debug, Default)]
72pub struct Condition(
73    /// Conjunction of `(selector identifier, required arm index)` terms.
74    /// An empty vector is always active.
75    pub Vec<(syn::Ident, i32)>,
76);
77impl Condition {
78    /// Return a new condition additionally requiring one selector arm.
79    ///
80    /// Does not modify the original condition; no additional runtime flag is generated.
81    ///
82    /// # Arguments
83    ///
84    /// * `tag` — identifier of the selector's index wire.
85    /// * `index` — zero-based index of the required arm.
86    pub fn arm(&self, tag: syn::Ident, index: i32) -> Self {
87        let mut result = self.clone();
88        result.0.push((tag, index));
89        result
90    }
91    /// Emit this condition AND a Rust predicate, short-circuiting the predicate
92    /// when the selector path is inactive. `predicate` is a Rust `bool` expression.
93    pub fn and(&self, predicate: TokenStream) -> TokenStream {
94        if self.0.is_empty() {
95            predicate
96        } else {
97            let condition = self.rust();
98            quote!((#condition) && (#predicate))
99        }
100    }
101    /// Emit the conjunction as a Rust boolean expression; an empty condition emits `true`.
102    pub fn rust(&self) -> TokenStream {
103        self.0
104            .iter()
105            .map(|(tag, i)| quote!(#tag == #i))
106            .reduce(|a, b| quote!((#a) && (#b)))
107            .unwrap_or_else(|| quote!(true))
108    }
109}
110/// Access to one foreign resource during an exported call.
111#[derive(Clone, Copy, Debug, PartialEq, Eq)]
112pub enum Access {
113    /// Read-only borrow; multiple active shared accesses may alias.
114    Shared,
115    /// Mutable borrow; conflicts with every other active use of the resource.
116    Exclusive,
117    /// Ownership transfer; conflicts with every other active use of the resource.
118    Move,
119}
120/// The adapter supplies identity; flat enforces call-wide access conflicts.
121#[derive(Clone, Debug)]
122pub struct Resource {
123    /// Side-effect-free Rust expression identifying a live resource, such as an address.
124    /// All identities compared within a call must share an equality-compatible domain.
125    pub identity: TokenStream,
126    /// Borrow or ownership requirement imposed by this input conversion.
127    pub access: Access,
128    /// Path on which this resource is accessed; inactive resources do not conflict.
129    pub active: Condition,
130}
131/// Emit pairwise alias checks across all active input resources.
132///
133/// `resources` holds the [`InputPlan::resources`] of all of one exported
134/// function's parameters together, so that aliases across parameters are found.
135///
136/// Run this code after leaf validation and before **any** ownership acquisition.
137/// Shared/shared pairs and mutually exclusive selector arms do not conflict.
138/// Conflicts return `Err("conflicting input aliases".into())` from the enclosing
139/// wrapper. This does not validate pointer liveness, synchronize threads, or
140/// acquire resources; those remain adapter/caller responsibilities.
141pub fn validate_resources(resources: &[Resource]) -> TokenStream {
142    let mut result = TokenStream::new();
143    for (i, a) in resources.iter().enumerate() {
144        for b in &resources[i + 1..] {
145            if a.access == Access::Shared && b.access == Access::Shared {
146                continue;
147            }
148            let x = &a.identity;
149            let y = &b.identity;
150            // Different alternatives of the same selector cannot conflict.
151            if a.active
152                .0
153                .iter()
154                .any(|(tag, arm)| b.active.0.iter().any(|(t, i)| t == tag && i != arm))
155            {
156                continue;
157            }
158            let mut active = a.active.clone();
159            for term in &b.active.0 {
160                if !active.0.contains(term) {
161                    active.0.push(term.clone());
162                }
163            }
164            let conflict = active.and(quote!((#x) == (#y)));
165            result.extend(quote!(if #conflict {return Err("conflicting input aliases".into());}));
166        }
167    }
168    result
169}

171/// Check that every bound function has an input converter for each parameter
172/// and an output converter, including unit results.
173///
174/// Functions with entirely empty slots remain source-only helpers. This check
175/// neither builds conversions nor changes the model. Adapters call it before
176/// writing wrappers. It does not validate code fragments supplied directly by callers.
177pub fn validate_bindings<I: InputConversion, O: OutputConversion>(
178    flat: &crate::Flat<I, O>,
179) -> Result<()> {
180    for f in flat.functions.values() {
181        if f.ret.slot.is_some() {
182            for p in &f.params {
183                if p.ty.slot.is_none() {
184                    return Err(format!(
185                        "`{}`: parameter `{}` has no input converter",
186                        f.name, p.name
187                    ));
188                }
189            }
190        } else if f.params.iter().any(|p| p.ty.slot.is_some()) {
191            return Err(format!(
192                "`{}`: bound inputs have no output converter",
193                f.name
194            ));
195        }
196    }
197    Ok(())
198}
199/// Format generated tokens as a Rust source file.
200///
201/// `tokens` must be a sequence of Rust items. Returns a syntax error if they do not form a file. This does not
202/// resolve names or type-check generated code; the binding crate must compile it.
203pub fn rust_source(tokens: TokenStream) -> Result<String> {
204    let file = syn::parse2(tokens).map_err(|e| format!("generated Rust: {e}"))?;
205    Ok(prettyplease::unparse(&file))
206}

214/// A fully prepared input conversion stored in a slot.
215/// Metadata and generated names must stay consistent across method calls.
216/// Code runs in the environment supplied by the language wrapper writer.
217/// Preparation resolves errors before installing the slot; [`emit_call`]
218/// reads this interface directly, without an intermediate plan.
219pub trait InputConversion {
220    /// Adapter-specific boundary description.
221    type Wire;
222    /// Boundary parameters in signature order.
223    fn wires(&self) -> &[Wire<Self::Wire>];
224    /// Shared call acknowledgement, after validation and before acquisition.
225    /// All contributing inputs must return the same token spelling; assembly emits
226    /// it once. Return `None` when this input needs no acknowledgement. Slot-local
227    /// fallible checks belong in `validation`, not in this shared operation.
228    fn acknowledgement(&self, _context: &EmissionContext) -> Result<Option<TokenStream>> {
229        Ok(None)
230    }
231    /// Resources used for call-wide alias checking.
232    fn resources(&self) -> &[Resource];
233    /// Fallible validation before any ownership transfer.
234    fn validation(&self, context: &EmissionContext) -> Result<TokenStream>;
235    /// Infallible ownership acquisition after acknowledgement.
236    fn acquisition(&self, context: &EmissionContext) -> Result<TokenStream>;
237    /// Construct the argument after all inputs have been acquired.
238    fn construction(&self, context: &EmissionContext) -> Result<TokenStream>;
239    /// Expression passed to the source function.
240    fn argument(&self, context: &EmissionContext) -> Result<TokenStream>;
241}
242impl<W> InputConversion for InputPlan<W> {
243    type Wire = W;
244    fn wires(&self) -> &[Wire<Self::Wire>] {
245        &self.wires
246    }
247    fn acknowledgement(&self, context: &EmissionContext) -> Result<Option<TokenStream>> {
248        self.acknowledgement
249            .as_ref()
250            .map(|code| context.render(code))
251            .transpose()
252    }
253    fn resources(&self) -> &[Resource] {
254        &self.resources
255    }
256    fn validation(&self, context: &EmissionContext) -> Result<TokenStream> {
257        context.render(&self.validation)
258    }
259    fn acquisition(&self, context: &EmissionContext) -> Result<TokenStream> {
260        context.render(&self.acquire)
261    }
262    fn construction(&self, context: &EmissionContext) -> Result<TokenStream> {
263        context.render(&self.statements)
264    }
265    fn argument(&self, context: &EmissionContext) -> Result<TokenStream> {
266        context.render(&self.argument)
267    }
268}
269/// A fully prepared output conversion stored in a slot.
270/// Metadata and generated names must stay consistent across method calls.
271/// Code runs in the environment supplied by the language wrapper writer.
272/// Preparation resolves errors before installing the slot; [`emit_call`]
273/// reads this interface directly, without an intermediate plan.
274/// Source-value templates use [`source_result_expr`]; custom implementations can
275/// use the supplied context directly.
276pub trait OutputConversion {
277    /// Adapter-specific boundary description.
278    type Wire;
279    /// Boundary results in delivery order.
280    fn wires(&self) -> &[Wire<Self::Wire>];
281    /// Output expressions in wire order, valid until commit.
282    fn values(&self, context: &EmissionContext) -> Result<Vec<TokenStream>>;
283    /// Prepare owned output temporaries before delivery.
284    fn preparation(&self, context: &EmissionContext) -> Result<TokenStream>;
285    /// Fallible delivery code; ownership remains uncommitted on failure.
286    fn delivery(&self, context: &EmissionContext) -> Result<TokenStream>;
287    /// Infallibly transfer ownership after successful delivery.
288    fn commit(&self, context: &EmissionContext) -> Result<TokenStream>;
289    /// Infallible wrapper result expression evaluated after commit.
290    fn result(&self, context: &EmissionContext) -> Result<TokenStream>;
291}
292impl<W> OutputConversion for OutputPlan<W> {
293    type Wire = W;
294    fn wires(&self) -> &[Wire<Self::Wire>] {
295        &self.wires
296    }
297    fn values(&self, context: &EmissionContext) -> Result<Vec<TokenStream>> {
298        self.values
299            .iter()
300            .map(|code| context.render(code))
301            .collect()
302    }
303    fn preparation(&self, context: &EmissionContext) -> Result<TokenStream> {
304        context.render(&self.statements)
305    }
306    fn delivery(&self, context: &EmissionContext) -> Result<TokenStream> {
307        context.render(&self.delivery.statements)
308    }
309    fn commit(&self, context: &EmissionContext) -> Result<TokenStream> {
310        context.render(&self.commit)
311    }
312    fn result(&self, context: &EmissionContext) -> Result<TokenStream> {
313        context.render(&self.delivery.result)
314    }
315}

325/// Result-slot glue executed after output preparation and before ownership commit.
326#[derive(Clone, Debug)]
327pub struct Delivery {
328    /// Deliver every output; failure must leave owned temporaries uncommitted.
329    pub statements: TokenStream,
330    /// Infallible wrapper result expression evaluated after commit.
331    pub result: TokenStream,

342impl<I: InputConversion, O: OutputConversion> crate::Function<I, O> {
343    /// Filled input slots, preserving source parameter order and grouping.
344    pub fn input_conversions(&self) -> Result<Vec<&I>> {
345        self.params
346            .iter()
347            .map(|p| {
348                p.ty.slot.as_ref().ok_or_else(|| {
349                    format!("`{}`: parameter `{}` has no conversion", self.name, p.name)
350                })
351            })
352            .collect()
353    }
354    /// Filled result slot. Emission checks the rendered wire and value counts.
355    pub fn output_conversion(&self) -> Result<&O> {
356        let output = self
357            .ret
358            .slot
359            .as_ref()
360            .ok_or_else(|| format!("`{}`: result has no conversion", self.name))?;
361        Ok(output)
362    }
363    /// Boundary input wires read directly from the slots in parameter order.
364    pub fn inputs(&self) -> Result<impl Iterator<Item = &Wire<I::Wire>>> {
365        Ok(self
366            .input_conversions()?
367            .into_iter()
368            .flat_map(|i| i.wires()))
369    }
370    /// Output expressions paired with their wire descriptions, directly from the slot.
371    pub fn outputs(
372        &self,
373        context: &EmissionContext,
374    ) -> Result<impl ExactSizeIterator<Item = OutputValue<'_, O::Wire>>> {
375        let output = self.output_conversion()?;
376        let values = output.values(context)?;
377        if output.wires().len() != values.len() {
378            return Err(format!("`{}`: output wire/value count differs", self.name));
379        }
380        Ok(output
381            .wires()
382            .iter()
383            .zip(values)
384            .map(|(wire, value)| OutputValue { wire, value }))
385    }
386}

388/// Assemble a function call directly from filled slots, without a generator callback.
389/// Validates slot completeness, output arity and agreement on shared acknowledgement.
390/// The returned code runs inside the wrapper's fallible boundary, in this order:
391/// all validation, alias checks, one acknowledgement, all acquisition, argument
392/// construction, source invocation, output preparation, delivery and commit.
393pub fn emit_call<I: InputConversion, O: OutputConversion>(
394    function: &crate::Function<I, O>,
395    context: &EmissionContext,
396) -> Result<TokenStream> {
397    let inputs = function.input_conversions()?;
398    let output = function.output_conversion()?;
399    let _ = function.outputs(context)?;
400    let mut acknowledgement: Option<TokenStream> = None;
401    for input in &inputs {
402        let Some(fragment) = input.acknowledgement(context)? else {
403            continue;
404        };
405        if let Some(previous) = &acknowledgement {
406            if previous.to_string() != fragment.to_string() {
407                return Err(format!(
408                    "`{}`: conflicting input acknowledgements",
409                    function.name
410                ));
411            }
412        } else {
413            acknowledgement = Some(fragment);
414        }
415    }
416    let validation = inputs
417        .iter()
418        .map(|i| i.validation(context))
419        .collect::<Result<Vec<_>>>()?;
420    let resources: Vec<_> = inputs
421        .iter()
422        .flat_map(|i| i.resources().iter().cloned())
423        .collect();
424    let aliases = context.render(&validate_resources(&resources))?;
425    let acquire = inputs
426        .iter()
427        .map(|i| i.acquisition(context))
428        .collect::<Result<Vec<_>>>()?;
429    let construct = inputs
430        .iter()
431        .map(|i| i.construction(context))
432        .collect::<Result<Vec<_>>>()?;
433    let args = inputs
434        .iter()
435        .map(|i| i.argument(context))
436        .collect::<Result<Vec<_>>>()?;
437    let path = &function.path;
438    let prepare = output.preparation(context)?;
439    let deliver = output.delivery(context)?;
440    let commit = output.commit(context)?;
441    let result = output.result(context)?;
442    let source = &context.source_result;
443    Ok(quote! {
444        #(#validation)* #aliases #acknowledgement #(#acquire)* #(#construct)*
445        let #source = #path(#(#args),*);
446        #prepare #deliver #commit Ok(#result)
447    })
