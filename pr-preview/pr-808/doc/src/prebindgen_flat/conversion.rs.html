6/// One boundary parameter or result component used by wrapper emission.
7///
8/// ```
9/// use prebindgen_flat::{Condition, Wire};
10/// let wire = Wire {
11///     component: Default::default(),
12///     name: syn::parse_quote!(value),
13///     kind: "i64",
14///     active: Condition::default(),
15/// };
16/// assert_eq!(wire.kind, "i64");
17/// ```
18#[derive(Clone, Debug)]
19pub struct Wire<W> {
20    /// Logical component supplied by this wire; several wires may share a path.
21    pub component: crate::ComponentPath,
22    /// Generated identifier used by conversion code and wrapper signatures.
23    pub name: syn::Ident,
24    /// Adapter description of the carrier, supplied by a conversion.
25    /// The framework carries it into the conversion without interpreting it.
26    pub kind: W,
27    /// Input selector path; outputs normally use the always-active default.
28    pub active: Condition,
29}

31/// An input conversion split into call-wide validation, acquisition and construction.
32///
33/// Conversion implementers provide these phases. [`PreparedCall::emit`] orders
34/// them across all parameters; wrapper writers do not assemble them themselves.
35#[derive(Clone, Debug)]
36pub struct InputPlan<W> {
37    /// Explicit resolved representation, if this conversion exposes a named shape.
38    pub representation: Option<crate::RepresentationId>,
39    /// Flattened boundary parameters in signature order.
40    pub wires: Vec<Wire<W>>,
41    /// Shared wrapper requirements contributed by this slot.
42    pub requirements: Vec<CallRequirement>,
43    /// Validate and prepare every active leaf before any ownership transfer.
44    pub validation: TokenStream,
45    /// Infallible acquisition of all owned inputs, before constructors can run.
46    pub acquire: TokenStream,
47    /// Active foreign resources whose identities must be checked across all arguments.
48    pub resources: Vec<Resource>,
49    /// Argument construction after all acquisitions, including source constructor calls.
50    pub statements: TokenStream,
51    /// Rust expression passed to the source function after `statements`.
52    pub argument: TokenStream,
53}
54/// Output conversion whose owned temporaries survive until successful delivery.
55///
56/// Conversion implementers provide preparation and deferred commit code.
57/// [`PreparedCall::emit`] commits only after adapter delivery succeeds.
58#[derive(Clone, Debug)]
59pub struct OutputPlan<W> {
60    /// Delivery and return code prepared with this result slot.
61    pub delivery: Delivery,
62    /// Explicit resolved representation, if this conversion exposes a named shape.
63    pub representation: Option<crate::RepresentationId>,
64    /// Boundary results in delivery order; one entry per expression in `values`.
65    pub wires: Vec<Wire<W>>,
66    /// Shared wrapper requirements contributed by this slot.
67    pub requirements: Vec<CallRequirement>,
68    /// Transfer ownership only after all foreign delivery operations succeed.
69    pub commit: TokenStream,
70    /// Code producing owned output temporaries and evaluating observations before moves.
71    pub statements: TokenStream,
72    /// Delivery expressions in wire order; they must remain valid until commit.
73    pub values: Vec<TokenStream>,
74}
75/// A leaf is active only on its selected path. Adapters use the same condition
76/// for foreign-side resource locking and native conversion code.
77#[derive(Clone, Debug, Default)]
78pub struct Condition(
79    /// Conjunction of `(selector identifier, required arm index)` terms.
80    /// An empty vector is always active.
81    pub Vec<(syn::Ident, i32)>,
82);
83impl Condition {
84    /// Return a new condition additionally requiring one selector arm.
85    ///
86    /// Does not modify the original condition; no additional runtime flag is generated.
87    ///
88    /// # Arguments
89    ///
90    /// * `tag` — identifier of the selector's index wire.
91    /// * `index` — zero-based index of the required arm.
92    pub fn arm(&self, tag: syn::Ident, index: i32) -> Self {
93        let mut result = self.clone();
94        result.0.push((tag, index));
95        result
96    }
97    /// Emit this condition AND a Rust predicate, short-circuiting the predicate
98    /// when the selector path is inactive. `predicate` is a Rust `bool` expression.
99    pub fn and(&self, predicate: TokenStream) -> TokenStream {
100        if self.0.is_empty() {
101            predicate
102        } else {
103            let condition = self.rust();
104            quote!((#condition) && (#predicate))
105        }
106    }
107    /// Emit the conjunction as a Rust boolean expression; an empty condition emits `true`.
108    pub fn rust(&self) -> TokenStream {
109        self.0
110            .iter()
111            .map(|(tag, i)| quote!(#tag == #i))
112            .reduce(|a, b| quote!((#a) && (#b)))
113            .unwrap_or_else(|| quote!(true))
114    }
115}
116/// Access to one foreign resource during an exported call.
117#[derive(Clone, Copy, Debug, PartialEq, Eq)]
118pub enum Access {
119    /// Read-only borrow; multiple active shared accesses may alias.
120    Shared,
121    /// Mutable borrow; conflicts with every other active use of the resource.
122    Exclusive,
123    /// Ownership transfer; conflicts with every other active use of the resource.
124    Move,
125}
126/// The adapter supplies identity; flat enforces call-wide access conflicts.
127#[derive(Clone, Debug)]
128pub struct Resource {
129    /// Side-effect-free Rust expression identifying a live resource, such as an address.
130    /// All identities compared within a call must share an equality-compatible domain.
131    pub identity: TokenStream,
132    /// Borrow or ownership requirement imposed by this input conversion.
133    pub access: Access,
134    /// Path on which this resource is accessed; inactive resources do not conflict.
135    pub active: Condition,
136}
137/// Emit pairwise alias checks across all active input resources.
138///
139/// `resources` holds the [`InputPlan::resources`] of all of one exported
140/// function's parameters together, so that aliases across parameters are found.
141///
142/// Run this code after leaf validation and before **any** ownership acquisition.
143/// Shared/shared pairs and mutually exclusive selector arms do not conflict.
144/// Conflicts return `Err("conflicting input aliases".into())` from the enclosing
145/// wrapper. This does not validate pointer liveness, synchronize threads, or
146/// acquire resources; those remain adapter/caller responsibilities.
147pub fn validate_resources(resources: &[Resource]) -> TokenStream {
148    let mut result = TokenStream::new();
149    for (i, a) in resources.iter().enumerate() {
150        for b in &resources[i + 1..] {
151            if a.access == Access::Shared && b.access == Access::Shared {
152                continue;
153            }
154            let x = &a.identity;
155            let y = &b.identity;
156            // Different alternatives of the same selector cannot conflict.
157            if a.active
158                .0
159                .iter()
160                .any(|(tag, arm)| b.active.0.iter().any(|(t, i)| t == tag && i != arm))
161            {
162                continue;
163            }
164            let mut active = a.active.clone();
165            for term in &b.active.0 {
166                if !active.0.contains(term) {
167                    active.0.push(term.clone());
168                }
169            }
170            let conflict = active.and(quote!((#x) == (#y)));
171            result.extend(quote!(if #conflict {return Err("conflicting input aliases".into());}));
172        }
173    }
174    result
175}

177/// Check that every bound function has an input converter for each parameter
178/// and an output converter, including unit results.
179///
180/// Functions with entirely empty slots remain source-only helpers. This check
181/// neither builds conversions nor changes the model. Adapters call it before
182/// writing wrappers. It does not validate code fragments supplied directly by callers.
183pub fn validate_bindings<I: InputConversion, O: OutputConversion, R>(
184    flat: &crate::Flat<I, O, R>,
185) -> Result<()> {
186    for f in flat.functions.values() {
187        if f.ret.slot.is_some() {
188            for p in &f.params {
189                if p.ty.slot.is_none() {
190                    return Err(format!(
191                        "`{}`: parameter `{}` has no input converter",
192                        f.name, p.name
193                    ));
194                }
195            }
196        } else if f.params.iter().any(|p| p.ty.slot.is_some()) {
197            return Err(format!(
198                "`{}`: bound inputs have no output converter",
199                f.name
200            ));
201        }
202    }
203    Ok(())
204}
205/// Format generated tokens as a Rust source file.
206///
207/// `tokens` must be a sequence of Rust items. Returns a syntax error if they do not form a file. This does not
208/// resolve names or type-check generated code; the binding crate must compile it.
209pub fn rust_source(tokens: TokenStream) -> Result<String> {
210    let file = syn::parse2(tokens).map_err(|e| format!("generated Rust: {e}"))?;
211    Ok(prettyplease::unparse(&file))
212}

214/// Identifier bound to the source function result by [`PreparedCall::emit`].
215/// Output conversion builders must use this identifier when constructing result
216/// expressions. The name is reserved within the generated call body.
217pub fn source_result() -> syn::Ident {
218    syn::parse_quote!(__result)
219}
220
221/// Expression referring to the source result bound by [`PreparedCall::emit`].
222/// Use [`source_result`] instead when an identifier is needed for interpolation.
223pub fn source_result_expr() -> TokenStream {
224    let result = source_result();
225    quote!(#result)
226}

228/// Opaque input conversion stored in a parameter slot.
229/// Implementations describe phases; wrapper writers use [`prepare_call`].
230pub trait InputConversion {
231    /// Adapter-specific boundary description.
232    type Wire;
233    /// Resolve code and boundary inputs for this occurrence.
234    fn emit(&self) -> Result<InputPlan<Self::Wire>>;
235}
236
237/// Opaque output conversion stored in a result slot.
238/// Build source-value expressions with [`source_result`], the same identifier
239/// bound by the call emitter.
240pub trait OutputConversion {
241    /// Adapter-specific boundary description.
242    type Wire;
243    /// Resolve code, boundary outputs and delivery glue for this occurrence.
244    /// The glue must match the environment supplied by the wrapper writer.
245    fn emit(&self) -> Result<OutputPlan<Self::Wire>>;
246}
247impl<W: Clone> InputConversion for InputPlan<W> {
248    type Wire = W;
249    fn emit(&self) -> Result<InputPlan<W>> {
250        Ok(self.clone())
251    }
252}
253impl<W: Clone> OutputConversion for OutputPlan<W> {
254    type Wire = W;
255    fn emit(&self) -> Result<OutputPlan<W>> {
256        Ok(self.clone())
257    }
258}

276/// A shared wrapper requirement supplied by a slot during preparation.
277/// Requirements with the same key must have identical token spellings. Assembly preserves
278/// first occurrence order and emits each requirement once, even when several
279/// parameters need it. Keys are adapter-owned, preferably namespaced.
280#[derive(Clone, Debug)]
281pub struct CallRequirement {
282    /// Stable identity within one call, such as `jni.ownership`.
283    pub key: String,
284    /// Additional Rust parameter declarations, without a trailing comma.
285    /// The wrapper writer places these after all wire parameters.
286    pub parameters: TokenStream,
287    /// Fallible code after all validation and alias checks, before acquisition.
288    /// It must not consume inputs. Shared acknowledgement belongs here.
289    pub acknowledge: TokenStream,
290}

292/// Result-slot glue executed after output preparation and before ownership commit.
293#[derive(Clone, Debug)]
294pub struct Delivery {
295    /// Deliver every output; failure must leave owned temporaries uncommitted.
296    pub statements: TokenStream,
297    /// Infallible wrapper result expression evaluated after commit.
298    pub result: TokenStream,
299}
300impl Default for Delivery {
301    fn default() -> Self {
302        Self {
303            statements: TokenStream::new(),
304            result: quote!(()),
305        }

309/// Resolved call with opaque conversion phases and inspectable boundary mappings.
310/// Created by [`prepare_call`].
311pub struct PreparedCall<I, O> {
312    path: syn::Path,
313    inputs: Vec<InputPlan<I>>,
314    output: OutputPlan<O>,
315    requirements: Vec<CallRequirement>,
316}
317impl<I, O> PreparedCall<I, O> {
318    /// Combined wrapper requirements, in first occurrence order, without duplicates.
319    pub fn requirements(&self) -> &[CallRequirement] {
320        &self.requirements
321    }
322
323    /// Preserve parameter grouping when mapping components to representation fields.
324    pub fn parameters(&self) -> impl ExactSizeIterator<Item = InputBoundary<'_, I>> {
325        self.inputs.iter().map(|input| InputBoundary {
326            representation: input.representation.as_ref(),
327            wires: &input.wires,
328        })
329    }
330    /// Boundary inputs in function and component order.
331    pub fn inputs(&self) -> impl Iterator<Item = &Wire<I>> {
332        self.inputs.iter().flat_map(|input| &input.wires)
333    }
334    /// Boundary outputs paired with their delivery expressions.
335    pub fn outputs(&self) -> impl ExactSizeIterator<Item = OutputValue<'_, O>> {
336        self.output
337            .wires
338            .iter()
339            .zip(&self.output.values)
340            .map(|(wire, value)| OutputValue { wire, value })
341    }
342    /// Assemble validation, alias checks, acquisition, construction, source call,
343    /// output preparation, delivery and commit in their required order.
344    /// The returned statements belong inside the adapter's fallible boundary closure.
345    pub fn emit(&self) -> TokenStream {
346        let validation = self.inputs.iter().map(|i| &i.validation);
347        let resources: Vec<_> = self
348            .inputs
349            .iter()
350            .flat_map(|i| i.resources.iter().cloned())
351            .collect();
352        let aliases = validate_resources(&resources);
353        let acquire = self.inputs.iter().map(|i| &i.acquire);
354        let construct = self.inputs.iter().map(|i| &i.statements);
355        let args = self.inputs.iter().map(|i| &i.argument);
356        let path = &self.path;
357        let prepare = &self.output.statements;
358        let commit = &self.output.commit;
359        let source = source_result();
360        let acknowledge = self.requirements.iter().map(|r| &r.acknowledge);
361        let Delivery {
362            statements: deliver,
363            result,
364        } = &self.output.delivery;
365        quote! {
366            #(#validation)* #aliases #(#acknowledge)* #(#acquire)* #(#construct)*
367            let #source = #path(#(#args),*);
368            #prepare #deliver #commit Ok(#result)
369        }
370    }
371}
372impl<I: InputConversion, O: OutputConversion> crate::Function<I, O> {
373    /// Resolve all slots once. Missing slots and inconsistent output arity fail
374    /// before a writer can emit a partial signature or call body.
375    fn resolve_call(&self) -> Result<PreparedCall<I::Wire, O::Wire>> {
376        let inputs = self
377            .params
378            .iter()
379            .map(|p| {
380                p.ty.slot
381                    .as_ref()
382                    .ok_or_else(|| {
383                        format!("`{}`: parameter `{}` has no conversion", self.name, p.name)
384                    })?
385                    .emit()
386            })
387            .collect::<Result<Vec<_>>>()?;
388        let output = self
389            .ret
390            .slot
391            .as_ref()
392            .ok_or_else(|| format!("`{}`: result has no conversion", self.name))?
393            .emit()?;
394        if output.wires.len() != output.values.len() {
395            return Err(format!("`{}`: output wire/value count differs", self.name));
396        }
397        let mut requirements: Vec<CallRequirement> = Vec::new();
398        for requirement in inputs
399            .iter()
400            .flat_map(|i| &i.requirements)
401            .chain(&output.requirements)
402        {
403            if let Some(previous) = requirements.iter().find(|r| r.key == requirement.key) {
404                if previous.parameters.to_string() != requirement.parameters.to_string()
405                    || previous.acknowledge.to_string() != requirement.acknowledge.to_string()
406                {
407                    return Err(format!(
408                        "`{}`: conflicting call requirement `{}`",
409                        self.name, requirement.key
410                    ));
411                }
412            } else {
413                requirements.push(requirement.clone());
414            }
415        }
416        Ok(PreparedCall {
417            requirements,
418            path: self.path.clone(),
419            inputs,
420            output,
421        })
422    }
423}

425/// Prepare a selected function and validate its explicit representation mappings.
426/// Call [`crate::Flat::validate_representations`] once before a generation pass.
427/// This method checks only the selected call's mappings. Canonical defaults
428/// are never substituted during generation.
429pub fn prepare_call<I: InputConversion, O: OutputConversion, R>(
430    flat: &crate::Flat<I, O, R>,
431    name: &str,
432) -> Result<PreparedCall<I::Wire, O::Wire>> {
433    let function = flat.function(name)?;
434    let call = function.resolve_call()?;
435    for (parameter, input) in function.params.iter().zip(&call.inputs) {
436        flat.check_mapping(
437            &parameter.ty.ty,
438            input.representation.as_ref(),
439            &input.wires,
440        )?;
441    }
442    flat.check_mapping(
443        &function.ret.ty,
444        call.output.representation.as_ref(),
445        &call.output.wires,
446    )?;
447    Ok(call)
448}

450impl<I: InputConversion, O: OutputConversion, R> crate::Flat<I, O, R> {
451    /// Validate all registered type shapes, including schemas not used by a function.
452    pub fn validate_representations(&self) -> Result<()> {
453        self.representations.validate()?;
454        for (id, shape) in self.representations.iter() {
455            self.resolve_type(&shape.source_type)?;
456            for component in &shape.components {
457                if let Some(fields) = &component.source_fields {
458                    let mut ty = &shape.source_type;
459                    for field in fields {
460                        ty = &self
461                            .record(ty)?
462                            .fields
463                            .iter()
464                            .find(|f| f.name == field.as_str())
465                            .ok_or_else(|| {
466                                format!(
467                                    "unknown source field `{field}` in representation `{}`",
468                                    id.0
469                                )
470                            })?
471                            .ty;
472                    }
473                    if let Some(child) = &component.representation {
474                        if !crate::same_type(ty, &self.representations.get(child)?.source_type) {
475                            return Err(format!(
476                                "component `{}` has an incompatible source type",
477                                component.name
478                            ));
479                        }
480                    }
481                }
482            }
483        }
484        Ok(())
485    }
486    fn check_mapping<W>(
487        &self,
488        ty: &syn::Type,
489        id: Option<&crate::RepresentationId>,
490        wires: &[Wire<W>],
491    ) -> Result<()> {
492        let Some(id) = id else {
493            return Ok(());
494        };
495        let shape = self.representations.get(id)?;
496        if !crate::same_type(ty, &shape.source_type) {
497            return Err(format!(
498                "representation `{}` does not match occurrence type",
499                id.0
500            ));
501        }
502        for wire in wires {
503            if !shape.components.iter().any(|c| c.path == wire.component) {
504                return Err(format!(
505                    "wire `{}` has no component in representation `{}`",
506                    wire.name, id.0
507                ));
508            }
509        }
510        for component in &shape.components {
511            if !wires.iter().any(|wire| wire.component == component.path) {
512                return Err(format!(
513                    "component `{}` has no wire in representation `{}`",
514                    component.name, id.0
515                ));
516            }
517        }
518        Ok(())
519    }
520}
