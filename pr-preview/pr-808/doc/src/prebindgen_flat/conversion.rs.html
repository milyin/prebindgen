6/// One boundary parameter or result component used by wrapper emission.
7///
8/// ```
9/// use prebindgen_flat::Wire;
10/// let wire = Wire {
11///     name: syn::parse_quote!(value),
12///     kind: "i64",
13/// };
14/// assert_eq!(wire.kind, "i64");
15/// ```
16#[derive(Clone, Debug)]
17pub struct Wire<W> {
18    /// Generated identifier used by conversion code and wrapper signatures.
19    pub name: syn::Ident,
20    /// Transport description used to declare the boundary type.
21    /// Logical fields, selection rules and ownership behavior belong inside the converter.
22    pub kind: W,
23}

25/// An input conversion split into call-wide validation, acquisition and construction.
26///
27/// Optional concrete storage implementing [`InputConversion`]. [`emit_call`] orders
28/// them across all parameters; wrapper writers do not assemble them themselves.
29#[derive(Clone, Debug)]
30pub struct InputPlan<W, C = syn::Ident> {
31    /// Context type supplied by the language generator at emission.
32    pub context: std::marker::PhantomData<fn(&C)>,
33    /// Flattened boundary parameters in signature order.
34    pub wires: Vec<Wire<W>>,
35    /// Shared call acknowledgement, if this input needs one.
36    pub acknowledgement: Option<TokenStream>,
37    /// Validate and prepare every active leaf before any ownership transfer.
38    pub validation: TokenStream,
39    /// Infallible acquisition of all owned inputs, before constructors can run.
40    pub acquire: TokenStream,
41    /// Active foreign resources whose identities must be checked across all arguments.
42    pub resources: Vec<Resource>,
43    /// Argument construction after all acquisitions, including source constructor calls.
44    pub statements: TokenStream,
45    /// Rust expression passed to the source function after `statements`.
46    pub argument: TokenStream,
47}
48/// Output conversion whose owned temporaries survive until successful delivery.
49///
50/// Optional concrete storage implementing [`OutputConversion`].
51/// [`emit_call`] commits only after adapter delivery succeeds.
52#[derive(Clone, Debug)]
53pub struct OutputPlan<W, C = syn::Ident> {
54    /// Context type supplied by the language generator at emission.
55    pub context: std::marker::PhantomData<fn(&C)>,
56    /// Delivery and return code prepared with this result slot.
57    pub delivery: Delivery,
58    /// Boundary results in delivery order; one entry per expression in `values`.
59    pub wires: Vec<Wire<W>>,
60    /// Transfer ownership only after all foreign delivery operations succeed.
61    pub commit: TokenStream,
62    /// Code producing owned output temporaries and evaluating observations before moves.
63    pub statements: TokenStream,
64    /// Delivery expressions in wire order; they must remain valid until commit.
65    pub values: Vec<TokenStream>,
66}
67/// Access to one foreign resource during an exported call.
68#[derive(Clone, Copy, Debug, PartialEq, Eq)]
69pub enum Access {
70    /// Read-only borrow; multiple active shared accesses may alias.
71    Shared,
72    /// Mutable borrow; conflicts with every other active use of the resource.
73    Exclusive,
74    /// Ownership transfer; conflicts with every other active use of the resource.
75    Move,
76}
77/// The adapter supplies identity; flat enforces call-wide access conflicts.
78#[derive(Clone, Debug)]
79pub struct Resource {
80    /// Side-effect-free Rust expression identifying a live resource, such as an address.
81    /// All identities compared within a call must share an equality-compatible domain.
82    pub identity: TokenStream,
83    /// Borrow or ownership requirement imposed by this input conversion.
84    pub access: Access,
85    /// Side-effect-free Rust boolean expression; false resources do not conflict.
86    pub active: TokenStream,
87}
88/// Emit pairwise alias checks across all active input resources.
89///
90/// `resources` holds the [`InputPlan::resources`] of all of one exported
91/// function's parameters together, so that aliases across parameters are found.
92///
93/// Run this code after leaf validation and before **any** ownership acquisition.
94/// Shared/shared pairs and resources whose predicates are not both true do not conflict.
95/// Conflicts return `Err("conflicting input aliases".into())` from the enclosing
96/// wrapper. This does not validate pointer liveness, synchronize threads, or
97/// acquire resources; those remain adapter/caller responsibilities.
98pub fn validate_resources(resources: &[Resource]) -> TokenStream {
99    let mut result = TokenStream::new();
100    for (i, a) in resources.iter().enumerate() {
101        for b in &resources[i + 1..] {
102            if a.access == Access::Shared && b.access == Access::Shared {
103                continue;
104            }
105            let x = &a.identity;
106            let y = &b.identity;
107            let active_a = &a.active;
108            let active_b = &b.active;
109            let conflict = quote!((#active_a) && (#active_b) && ((#x) == (#y)));
110            result.extend(quote!(if #conflict {return Err("conflicting input aliases".into());}));
111        }
112    }
113    result
114}

116/// Check that every bound function has an input converter for each parameter
117/// and an output converter, including unit results.
118///
119/// Functions with entirely empty slots remain source-only helpers. This check
120/// neither builds conversions nor changes the model. Adapters call it before
121/// writing wrappers. It does not validate code fragments supplied directly by callers.
122pub fn validate_bindings<I: InputConversion, O: OutputConversion>(
123    flat: &crate::Flat<I, O>,
124) -> Result<()> {
125    for f in flat.functions.values() {
126        if f.ret.slot.is_some() {
127            for p in &f.params {
128                if p.ty.slot.is_none() {
129                    return Err(format!(
130                        "`{}`: parameter `{}` has no input converter",
131                        f.name, p.name
132                    ));
133                }
134            }
135        } else if f.params.iter().any(|p| p.ty.slot.is_some()) {
136            return Err(format!(
137                "`{}`: bound inputs have no output converter",
138                f.name
139            ));
140        }
141    }
142    Ok(())
143}
144/// Format generated tokens as a Rust source file.
145///
146/// `tokens` must be a sequence of Rust items. Returns a syntax error if they do not form a file. This does not
147/// resolve names or type-check generated code; the binding crate must compile it.
148pub fn rust_source(tokens: TokenStream) -> Result<String> {
149    let file = syn::parse2(tokens).map_err(|e| format!("generated Rust: {e}"))?;
150    Ok(prettyplease::unparse(&file))
151}

159/// A fully prepared input conversion stored in a slot.
160/// Metadata and generated names must stay consistent across method calls.
161/// Code runs in the environment supplied by the language wrapper writer.
162/// Preparation resolves errors before installing the slot; [`emit_call`]
163/// reads this interface directly, without an intermediate plan.
164///
165/// This conversion accepts a `u8` wire parameter and builds a Rust `bool` argument.
166/// It validates the wire value before constructing the argument. Since a scalar
167/// has no foreign resource to acquire, it needs no resources or acknowledgement.
168///
169/// ```
170/// use prebindgen_flat::{EmissionContext, InputConversion, Resource, Result, Wire};
171/// use proc_macro2::TokenStream;
172/// use quote::quote;
173/// use syn::{parse_quote, Ident, Type};
174///
175/// // The language generator defines the context and supplies it during emission.
176/// struct Context {
177///     source_result: Ident,
178/// }
179/// impl EmissionContext for Context {
180///     fn source_result(&self) -> &Ident { &self.source_result }
181/// }
182///
183/// // Preparation stores the wire description and argument variable name in the slot.
184/// struct BoolInput {
185///     wires: [Wire<Type>; 1],
186///     argument_name: Ident,
187/// }
188/// impl InputConversion for BoolInput {
189///     type Wire = Type;
190///     type Context = Context;
191///
192///     fn wires(&self) -> &[Wire<Type>] {
193///         &self.wires
194///     }
195///
196///     fn resources(&self, _: &Context) -> Result<Vec<Resource>> {
197///         Ok(vec![]) // A scalar has no foreign resource identity.
198///     }
199///
200///     fn validation(&self, _: &Context) -> Result<TokenStream> {
201///         let wire = &self.wires[0].name;
202///         Ok(quote! {
203///             if #wire > 1 {
204///                 return Err("expected 0 or 1 for bool".into());
205///             }
206///         })
207///     }
208///
209///     // The default acknowledgement() returns None: no ownership is transferred.
210///     fn acquisition(&self, _: &Context) -> Result<TokenStream> {
211///         Ok(quote!())
212///     }
213///
214///     fn construction(&self, _: &Context) -> Result<TokenStream> {
215///         let wire = &self.wires[0].name;
216///         let argument = &self.argument_name;
217///         Ok(quote!(let #argument = #wire != 0;))
218///     }
219///
220///     fn argument(&self, _: &Context) -> Result<TokenStream> {
221///         let argument = &self.argument_name;
222///         Ok(quote!(#argument))
223///     }
224/// }
225///
226/// let slot = BoolInput {
227///     wires: [Wire {
228///         name: parse_quote!(raw_enabled),
229///         kind: parse_quote!(u8),
230///     }],
231///     argument_name: parse_quote!(enabled),
232/// };
233/// let context = Context { source_result: parse_quote!(returned) };
234///
235/// // Individual phases can be emitted directly. emit_call orders these phases
236/// // across all input slots and passes enabled to the wrapped function.
237/// assert_eq!(slot.construction(&context)?.to_string(),
238///            quote!(let enabled = raw_enabled != 0;).to_string());
239/// assert_eq!(slot.argument(&context)?.to_string(), "enabled");
240/// # Ok::<(), String>(())
241/// ```
242///
243/// The converter owns any conditional behavior. Wires describe transport only.
244pub trait InputConversion {
245    /// Adapter-specific boundary description.
246    type Wire;
247    /// Emission data defined and supplied by the language generator.
248    type Context: EmissionContext;
249    /// Boundary parameters in signature order.
250    fn wires(&self) -> &[Wire<Self::Wire>];
251    /// Shared call acknowledgement, after validation and before acquisition.
252    /// All contributing inputs must return the same token spelling; assembly emits
253    /// it once. Return `None` when this input needs no acknowledgement. Slot-local
254    /// fallible checks belong in `validation`, not in this shared operation.
255    fn acknowledgement(&self, _context: &Self::Context) -> Result<Option<TokenStream>> {
256        Ok(None)
257    }
258    /// Resources used for call-wide alias checking, with identities emitted for this context.
259    fn resources(&self, context: &Self::Context) -> Result<Vec<Resource>>;
260    /// Fallible validation before any ownership transfer.
261    fn validation(&self, context: &Self::Context) -> Result<TokenStream>;
262    /// Infallible ownership acquisition after acknowledgement.
263    fn acquisition(&self, context: &Self::Context) -> Result<TokenStream>;
264    /// Construct the argument after all inputs have been acquired.
265    fn construction(&self, context: &Self::Context) -> Result<TokenStream>;
266    /// Expression passed to the source function.
267    fn argument(&self, context: &Self::Context) -> Result<TokenStream>;
268}
269impl<W, C: TemplateContext> InputConversion for InputPlan<W, C> {
270    type Wire = W;
271    type Context = C;
272    fn wires(&self) -> &[Wire<Self::Wire>] {
273        &self.wires
274    }
275    fn acknowledgement(&self, context: &Self::Context) -> Result<Option<TokenStream>> {
276        self.acknowledgement
277            .as_ref()
278            .map(|code| context.render(code))
279            .transpose()
280    }
281    fn resources(&self, context: &Self::Context) -> Result<Vec<Resource>> {
282        self.resources
283            .iter()
284            .map(|resource| {
285                Ok(Resource {
286                    identity: context.render(&resource.identity)?,
287                    access: resource.access,
288                    active: context.render(&resource.active)?,
289                })
290            })
291            .collect()
292    }
293    fn validation(&self, context: &Self::Context) -> Result<TokenStream> {
294        context.render(&self.validation)
295    }
296    fn acquisition(&self, context: &Self::Context) -> Result<TokenStream> {
297        context.render(&self.acquire)
298    }
299    fn construction(&self, context: &Self::Context) -> Result<TokenStream> {
300        context.render(&self.statements)
301    }
302    fn argument(&self, context: &Self::Context) -> Result<TokenStream> {
303        context.render(&self.argument)
304    }
305}
306/// A fully prepared output conversion stored in a slot.
307/// Metadata and generated names must stay consistent across method calls.
308/// Code runs in the environment supplied by the language wrapper writer.
309/// Preparation resolves errors before installing the slot; [`emit_call`]
310/// reads this interface directly, without an intermediate plan.
311/// Source-value templates use [`source_result_expr`]; custom implementations can
312/// use the supplied context directly.
313pub trait OutputConversion {
314    /// Adapter-specific boundary description.
315    type Wire;
316    /// Emission data defined and supplied by the language generator.
317    type Context: EmissionContext;
318    /// Boundary results in delivery order.
319    fn wires(&self) -> &[Wire<Self::Wire>];
320    /// Output expressions in wire order, valid until commit.
321    fn values(&self, context: &Self::Context) -> Result<Vec<TokenStream>>;
322    /// Prepare owned output temporaries before delivery.
323    fn preparation(&self, context: &Self::Context) -> Result<TokenStream>;
324    /// Fallible delivery code; ownership remains uncommitted on failure.
325    fn delivery(&self, context: &Self::Context) -> Result<TokenStream>;
326    /// Infallibly transfer ownership after successful delivery.
327    fn commit(&self, context: &Self::Context) -> Result<TokenStream>;
328    /// Infallible wrapper result expression evaluated after commit.
329    fn result(&self, context: &Self::Context) -> Result<TokenStream>;
330}
331impl<W, C: TemplateContext> OutputConversion for OutputPlan<W, C> {
332    type Wire = W;
333    type Context = C;
334    fn wires(&self) -> &[Wire<Self::Wire>] {
335        &self.wires
336    }
337    fn values(&self, context: &Self::Context) -> Result<Vec<TokenStream>> {
338        self.values
339            .iter()
340            .map(|code| context.render(code))
341            .collect()
342    }
343    fn preparation(&self, context: &Self::Context) -> Result<TokenStream> {
344        context.render(&self.statements)
345    }
346    fn delivery(&self, context: &Self::Context) -> Result<TokenStream> {
347        context.render(&self.delivery.statements)
348    }
349    fn commit(&self, context: &Self::Context) -> Result<TokenStream> {
350        context.render(&self.commit)
351    }
352    fn result(&self, context: &Self::Context) -> Result<TokenStream> {
353        context.render(&self.delivery.result)
354    }
355}

365/// Result-slot glue executed after output preparation and before ownership commit.
366#[derive(Clone, Debug)]
367pub struct Delivery {
368    /// Deliver every output; failure must leave owned temporaries uncommitted.
369    pub statements: TokenStream,

382impl<I: InputConversion, O: OutputConversion> crate::Function<I, O> {
383    /// Filled input slots, preserving source parameter order and grouping.
384    pub fn input_conversions(&self) -> Result<Vec<&I>> {
385        self.params
386            .iter()
387            .map(|p| {
388                p.ty.slot.as_ref().ok_or_else(|| {
389                    format!("`{}`: parameter `{}` has no conversion", self.name, p.name)
390                })
391            })
392            .collect()
393    }
394    /// Filled result slot. Emission checks the rendered wire and value counts.
395    pub fn output_conversion(&self) -> Result<&O> {
396        let output = self
397            .ret
398            .slot
399            .as_ref()
400            .ok_or_else(|| format!("`{}`: result has no conversion", self.name))?;
401        Ok(output)
402    }
403    /// Boundary input wires read directly from the slots in parameter order.
404    pub fn inputs(&self) -> Result<impl Iterator<Item = &Wire<I::Wire>>> {
405        Ok(self
406            .input_conversions()?
407            .into_iter()
408            .flat_map(|i| i.wires()))
409    }
410    /// Output expressions paired with their wire descriptions, directly from the slot.
411    pub fn outputs(
412        &self,
413        context: &O::Context,
414    ) -> Result<impl ExactSizeIterator<Item = OutputValue<'_, O::Wire>>> {
415        let output = self.output_conversion()?;
416        let values = output.values(context)?;
417        if output.wires().len() != values.len() {
418            return Err(format!("`{}`: output wire/value count differs", self.name));
419        }
420        Ok(output
421            .wires()
422            .iter()
423            .zip(values)
424            .map(|(wire, value)| OutputValue { wire, value }))
425    }
426}

428/// Assemble a function call directly from filled slots, without a generator callback.
429/// Validates slot completeness, output arity and agreement on shared acknowledgement.
430/// The returned code runs inside the wrapper's fallible boundary, in this order:
431/// all validation, alias checks, one acknowledgement, all acquisition, argument
432/// construction, source invocation, output preparation, delivery and commit.
433/// Input and output contexts must match, even when the function has no parameters.
434///
435/// ```compile_fail
436/// use prebindgen_flat::{emit_call, Function, InputConversion, OutputConversion, EmissionContext};
437/// struct OtherContext(syn::Ident);
438/// impl EmissionContext for OtherContext {
439///     fn source_result(&self) -> &syn::Ident { &self.0 }
440/// }
441/// fn incompatible<I, O>(function: &Function<I, O>, context: &syn::Ident)
442/// where I: InputConversion<Context = syn::Ident>,
443///       O: OutputConversion<Context = OtherContext>,
444/// {
445///     emit_call(function, context).unwrap();
446/// }
447/// ```
448pub fn emit_call<I: InputConversion, O: OutputConversion<Context = I::Context>>(
449    function: &crate::Function<I, O>,
450    context: &I::Context,
451) -> Result<TokenStream> {
452    let inputs = function.input_conversions()?;
453    let output = function.output_conversion()?;
454    let _ = function.outputs(context)?;
455    let mut acknowledgement: Option<TokenStream> = None;
456    for input in &inputs {
457        let Some(fragment) = input.acknowledgement(context)? else {
458            continue;
459        };
460        if let Some(previous) = &acknowledgement {
461            if previous.to_string() != fragment.to_string() {
462                return Err(format!(
463                    "`{}`: conflicting input acknowledgements",
464                    function.name
465                ));
466            }
467        } else {
468            acknowledgement = Some(fragment);
469        }
470    }
471    let validation = inputs
472        .iter()
473        .map(|i| i.validation(context))
474        .collect::<Result<Vec<_>>>()?;
475    let resources: Vec<_> = inputs
476        .iter()
477        .map(|i| i.resources(context))
478        .collect::<Result<Vec<_>>>()?
479        .into_iter()
480        .flatten()
481        .collect();
482    let aliases = validate_resources(&resources);
483    let acquire = inputs
484        .iter()
485        .map(|i| i.acquisition(context))
486        .collect::<Result<Vec<_>>>()?;
487    let construct = inputs
488        .iter()
489        .map(|i| i.construction(context))
490        .collect::<Result<Vec<_>>>()?;
491    let args = inputs
492        .iter()
493        .map(|i| i.argument(context))
494        .collect::<Result<Vec<_>>>()?;
495    let path = &function.path;
496    let prepare = output.preparation(context)?;
497    let deliver = output.delivery(context)?;
498    let commit = output.commit(context)?;
499    let result = output.result(context)?;
500    let source = context.source_result();
501    Ok(quote! {
502        #(#validation)* #aliases #acknowledgement #(#acquire)* #(#construct)*
503        let #source = #path(#(#args),*);
504        #prepare #deliver #commit Ok(#result)
505    })
