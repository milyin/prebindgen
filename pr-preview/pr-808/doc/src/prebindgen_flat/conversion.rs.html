6/// Transport metadata that supplies the Rust carrier type of a wire.
7/// The returned type describes the boundary value, not the source value being converted.
8/// For example, a boolean conversion may use `u8` as its Rust carrier type.
9///
10/// ```
11/// use prebindgen_flat::WireKind;
12/// struct BoolCarrier;
13/// impl WireKind for BoolCarrier {
14///     fn rust_type(&self) -> syn::Type { syn::parse_quote!(u8) }
15/// }
16/// let ty = BoolCarrier.rust_type();
17/// assert_eq!(quote::quote!(#ty).to_string(), "u8");
18/// ```
19pub trait WireKind {
20    /// Rust type used to declare this wire in a wrapper signature or result structure.
21    fn rust_type(&self) -> syn::Type;
22}
23impl WireKind for syn::Type {
24    fn rust_type(&self) -> syn::Type {
25        self.clone()
26    }
27}

29/// One boundary parameter or result component used by wrapper emission.
30///
31/// In conversion slots, `W` implements [`WireKind`]. Call [`Wire::rust_type`]
32/// to obtain the Rust carrier type supplied by the kind.
33///
34/// ```
35/// use prebindgen_flat::Wire;
36/// let wire: Wire<syn::Type> = Wire {
37///     name: syn::parse_quote!(value),
38///     kind: syn::parse_quote!(i64),
39/// };
40/// let ty = wire.rust_type();
41/// assert_eq!(quote::quote!(#ty).to_string(), "i64");
42/// ```
43#[derive(Clone, Debug)]
44pub struct Wire<W> {
45    /// Generated identifier used by conversion code and wrapper signatures.
46    pub name: syn::Ident,
47    /// Transport description implementing [`WireKind`] in conversion slots.
48    /// Logical fields, selection rules and ownership behavior belong inside the converter.
49    pub kind: W,
50}
51
52impl<W: WireKind> Wire<W> {
53    /// Rust carrier type supplied by this wire's [`WireKind`].
54    /// Used to declare a wrapper parameter or result field.
55    pub fn rust_type(&self) -> syn::Type {
56        self.kind.rust_type()
57    }
58}

60/// An input conversion split into call-wide validation, acquisition and construction.
61///
62/// Optional concrete storage implementing [`InputConversion`]. [`emit_call`] orders
63/// them across all parameters; wrapper writers do not assemble them themselves.
64#[derive(Clone, Debug)]
65pub struct InputPlan<W, C = syn::Ident> {
66    /// Context type supplied by the language generator at emission.
67    pub context: std::marker::PhantomData<fn(&C)>,
68    /// Flattened boundary parameters in signature order.
69    pub wires: Vec<Wire<W>>,
70    /// Shared call acknowledgement, if this input needs one.
71    pub acknowledgement: Option<TokenStream>,
72    /// Validate and prepare every active leaf before any ownership transfer.
73    pub validation: TokenStream,
74    /// Infallible acquisition of all owned inputs, before constructors can run.
75    pub acquire: TokenStream,
76    /// Active foreign resources whose identities must be checked across all arguments.
77    pub resources: Vec<Resource>,
78    /// Argument construction after all acquisitions, including source constructor calls.
79    pub statements: TokenStream,
80    /// Rust expression passed to the source function after `statements`.
81    pub argument: TokenStream,
82}
83/// Output conversion whose owned temporaries survive until successful delivery.
84///
85/// Optional concrete storage implementing [`OutputConversion`].
86/// [`emit_call`] commits only after adapter delivery succeeds.
87#[derive(Clone, Debug)]
88pub struct OutputPlan<W, C = syn::Ident> {
89    /// Context type supplied by the language generator at emission.
90    pub context: std::marker::PhantomData<fn(&C)>,
91    /// Delivery and return code prepared with this result slot.
92    pub delivery: Delivery,
93    /// Boundary results in delivery order; one entry per expression in `values`.
94    pub wires: Vec<Wire<W>>,
95    /// Transfer ownership only after all foreign delivery operations succeed.
96    pub commit: TokenStream,
97    /// Code producing owned output temporaries and evaluating observations before moves.
98    pub statements: TokenStream,
99    /// Delivery expressions in wire order; they must remain valid until commit.
100    pub values: Vec<TokenStream>,
101}
102/// Access to one foreign resource during an exported call.
103#[derive(Clone, Copy, Debug, PartialEq, Eq)]
104pub enum Access {
105    /// Read-only borrow; multiple active shared accesses may alias.
106    Shared,
107    /// Mutable borrow; conflicts with every other active use of the resource.
108    Exclusive,
109    /// Ownership transfer; conflicts with every other active use of the resource.
110    Move,
111}
112/// The adapter supplies identity; flat enforces call-wide access conflicts.
113#[derive(Clone, Debug)]
114pub struct Resource {
115    /// Side-effect-free Rust expression identifying a live resource, such as an address.
116    /// All identities compared within a call must share an equality-compatible domain.
117    pub identity: TokenStream,
118    /// Borrow or ownership requirement imposed by this input conversion.
119    pub access: Access,
120    /// Side-effect-free Rust boolean expression; false resources do not conflict.
121    pub active: TokenStream,
122}
123/// Emit pairwise alias checks across all active input resources.
124///
125/// `resources` holds the [`InputPlan::resources`] of all of one exported
126/// function's parameters together, so that aliases across parameters are found.
127///
128/// Run this code after leaf validation and before **any** ownership acquisition.
129/// Shared/shared pairs and resources whose predicates are not both true do not conflict.
130/// Conflicts return `Err("conflicting input aliases".into())` from the enclosing
131/// wrapper. This does not validate pointer liveness, synchronize threads, or
132/// acquire resources; those remain adapter/caller responsibilities.
133pub fn validate_resources(resources: &[Resource]) -> TokenStream {
134    let mut result = TokenStream::new();
135    for (i, a) in resources.iter().enumerate() {
136        for b in &resources[i + 1..] {
137            if a.access == Access::Shared && b.access == Access::Shared {
138                continue;
139            }
140            let x = &a.identity;
141            let y = &b.identity;
142            let active_a = &a.active;
143            let active_b = &b.active;
144            let conflict = quote!((#active_a) && (#active_b) && ((#x) == (#y)));
145            result.extend(quote!(if #conflict {return Err("conflicting input aliases".into());}));
146        }
147    }
148    result
149}

151/// Check that every bound function has an input converter for each parameter
152/// and an output converter, including unit results.
153///
154/// Functions with entirely empty slots remain source-only helpers. This check
155/// neither builds conversions nor changes the model. Adapters call it before
156/// writing wrappers. It does not validate code fragments supplied directly by callers.
157pub fn validate_bindings<I: InputConversion, O: OutputConversion>(
158    flat: &crate::Flat<I, O>,
159) -> Result<()> {
160    for f in flat.functions.values() {
161        if f.ret.slot.is_some() {
162            for p in &f.params {
163                if p.ty.slot.is_none() {
164                    return Err(format!(
165                        "`{}`: parameter `{}` has no input converter",
166                        f.name, p.name
167                    ));
168                }
169            }
170        } else if f.params.iter().any(|p| p.ty.slot.is_some()) {
171            return Err(format!(
172                "`{}`: bound inputs have no output converter",
173                f.name
174            ));
175        }
176    }
177    Ok(())
178}
179/// Format generated tokens as a Rust source file.
180///
181/// `tokens` must be a sequence of Rust items. Returns a syntax error if they do not form a file. This does not
182/// resolve names or type-check generated code; the binding crate must compile it.
183pub fn rust_source(tokens: TokenStream) -> Result<String> {
184    let file = syn::parse2(tokens).map_err(|e| format!("generated Rust: {e}"))?;
185    Ok(prettyplease::unparse(&file))
186}

194/// A fully prepared input conversion stored in a slot.
195/// Metadata and generated names must stay consistent across method calls.
196/// Code runs in the environment supplied by the language wrapper writer.
197/// Preparation resolves errors before installing the slot; [`emit_call`]
198/// reads this interface directly, without an intermediate plan.
199///
200/// This conversion accepts a `u8` wire parameter and builds a Rust `bool` argument.
201/// It validates the wire value before constructing the argument. Since a scalar
202/// has no foreign resource to acquire, it needs no resources or acknowledgement.
203///
204/// ```
205/// use prebindgen_flat::{EmissionContext, InputConversion, Resource, Result, Wire};
206/// use proc_macro2::TokenStream;
207/// use quote::quote;
208/// use syn::{parse_quote, Ident, Type};
209///
210/// // The language generator defines the context and supplies it during emission.
211/// struct Context {
212///     source_result: Ident,
213/// }
214/// impl EmissionContext for Context {
215///     fn source_result(&self) -> &Ident { &self.source_result }
216/// }
217///
218/// // Preparation stores the wire description and argument variable name in the slot.
219/// struct BoolInput {
220///     wires: [Wire<Type>; 1],
221///     argument_name: Ident,
222/// }
223/// impl InputConversion for BoolInput {
224///     type Wire = Type;
225///     type Context = Context;
226///
227///     fn wires(&self) -> &[Wire<Type>] {
228///         &self.wires
229///     }
230///
231///     fn resources(&self, _: &Context) -> Result<Vec<Resource>> {
232///         Ok(vec![]) // A scalar has no foreign resource identity.
233///     }
234///
235///     fn validation(&self, _: &Context) -> Result<TokenStream> {
236///         let wire = &self.wires[0].name;
237///         Ok(quote! {
238///             if #wire > 1 {
239///                 return Err("expected 0 or 1 for bool".into());
240///             }
241///         })
242///     }
243///
244///     // The default acknowledgement() returns None: no ownership is transferred.
245///     fn acquisition(&self, _: &Context) -> Result<TokenStream> {
246///         Ok(quote!())
247///     }
248///
249///     fn construction(&self, _: &Context) -> Result<TokenStream> {
250///         let wire = &self.wires[0].name;
251///         let argument = &self.argument_name;
252///         Ok(quote!(let #argument = #wire != 0;))
253///     }
254///
255///     fn argument(&self, _: &Context) -> Result<TokenStream> {
256///         let argument = &self.argument_name;
257///         Ok(quote!(#argument))
258///     }
259/// }
260///
261/// let slot = BoolInput {
262///     wires: [Wire {
263///         name: parse_quote!(raw_enabled),
264///         kind: parse_quote!(u8),
265///     }],
266///     argument_name: parse_quote!(enabled),
267/// };
268/// let context = Context { source_result: parse_quote!(returned) };
269///
270/// // Individual phases can be emitted directly. emit_call orders these phases
271/// // across all input slots and passes enabled to the wrapped function.
272/// assert_eq!(slot.construction(&context)?.to_string(),
273///            quote!(let enabled = raw_enabled != 0;).to_string());
274/// assert_eq!(slot.argument(&context)?.to_string(), "enabled");
275/// # Ok::<(), String>(())
276/// ```
277///
278/// The converter owns any conditional behavior. Wires describe transport only.
279pub trait InputConversion {
280    /// Adapter-specific boundary description.
281    type Wire: WireKind;
282    /// Emission data defined and supplied by the language generator.
283    type Context: EmissionContext;
284    /// Boundary parameters in signature order.
285    fn wires(&self) -> &[Wire<Self::Wire>];
286    /// Shared call acknowledgement, after validation and before acquisition.
287    /// All contributing inputs must return the same token spelling; assembly emits
288    /// it once. Return `None` when this input needs no acknowledgement. Slot-local
289    /// fallible checks belong in `validation`, not in this shared operation.
290    fn acknowledgement(&self, _context: &Self::Context) -> Result<Option<TokenStream>> {
291        Ok(None)
292    }
293    /// Resources used for call-wide alias checking, with identities emitted for this context.
294    fn resources(&self, context: &Self::Context) -> Result<Vec<Resource>>;
295    /// Fallible validation before any ownership transfer.
296    fn validation(&self, context: &Self::Context) -> Result<TokenStream>;
297    /// Infallible ownership acquisition after acknowledgement.
298    fn acquisition(&self, context: &Self::Context) -> Result<TokenStream>;
299    /// Construct the argument after all inputs have been acquired.
300    fn construction(&self, context: &Self::Context) -> Result<TokenStream>;
301    /// Expression passed to the source function.
302    fn argument(&self, context: &Self::Context) -> Result<TokenStream>;
303}
304impl<W: WireKind, C: TemplateContext> InputConversion for InputPlan<W, C> {
305    type Wire = W;
306    type Context = C;
307    fn wires(&self) -> &[Wire<Self::Wire>] {
308        &self.wires
309    }
310    fn acknowledgement(&self, context: &Self::Context) -> Result<Option<TokenStream>> {
311        self.acknowledgement
312            .as_ref()
313            .map(|code| context.render(code))
314            .transpose()
315    }
316    fn resources(&self, context: &Self::Context) -> Result<Vec<Resource>> {
317        self.resources
318            .iter()
319            .map(|resource| {
320                Ok(Resource {
321                    identity: context.render(&resource.identity)?,
322                    access: resource.access,
323                    active: context.render(&resource.active)?,
324                })
325            })
326            .collect()
327    }
328    fn validation(&self, context: &Self::Context) -> Result<TokenStream> {
329        context.render(&self.validation)
330    }
331    fn acquisition(&self, context: &Self::Context) -> Result<TokenStream> {
332        context.render(&self.acquire)
333    }
334    fn construction(&self, context: &Self::Context) -> Result<TokenStream> {
335        context.render(&self.statements)
336    }
337    fn argument(&self, context: &Self::Context) -> Result<TokenStream> {
338        context.render(&self.argument)
339    }
340}
341/// A fully prepared output conversion stored in a slot.
342/// Metadata and generated names must stay consistent across method calls.
343/// Code runs in the environment supplied by the language wrapper writer.
344/// Preparation resolves errors before installing the slot; [`emit_call`]
345/// reads this interface directly, without an intermediate plan.
346/// Source-value templates use [`source_result_expr`]; custom implementations can
347/// use the supplied context directly.
348pub trait OutputConversion {
349    /// Adapter-specific boundary description.
350    type Wire: WireKind;
351    /// Emission data defined and supplied by the language generator.
352    type Context: EmissionContext;
353    /// Boundary results in delivery order.
354    fn wires(&self) -> &[Wire<Self::Wire>];
355    /// Output expressions in wire order, valid until commit.
356    fn values(&self, context: &Self::Context) -> Result<Vec<TokenStream>>;
357    /// Prepare owned output temporaries before delivery.
358    fn preparation(&self, context: &Self::Context) -> Result<TokenStream>;
359    /// Fallible delivery code; ownership remains uncommitted on failure.
360    fn delivery(&self, context: &Self::Context) -> Result<TokenStream>;
361    /// Infallibly transfer ownership after successful delivery.
362    fn commit(&self, context: &Self::Context) -> Result<TokenStream>;
363    /// Infallible wrapper result expression evaluated after commit.
364    fn result(&self, context: &Self::Context) -> Result<TokenStream>;
365}
366impl<W: WireKind, C: TemplateContext> OutputConversion for OutputPlan<W, C> {
367    type Wire = W;
368    type Context = C;
369    fn wires(&self) -> &[Wire<Self::Wire>] {
370        &self.wires
371    }
372    fn values(&self, context: &Self::Context) -> Result<Vec<TokenStream>> {
373        self.values
374            .iter()
375            .map(|code| context.render(code))
376            .collect()
377    }
378    fn preparation(&self, context: &Self::Context) -> Result<TokenStream> {
379        context.render(&self.statements)
380    }
381    fn delivery(&self, context: &Self::Context) -> Result<TokenStream> {
382        context.render(&self.delivery.statements)
383    }
384    fn commit(&self, context: &Self::Context) -> Result<TokenStream> {
385        context.render(&self.commit)
386    }
387    fn result(&self, context: &Self::Context) -> Result<TokenStream> {
388        context.render(&self.delivery.result)
389    }
390}

417impl<I: InputConversion, O: OutputConversion> crate::Function<I, O> {
418    /// Filled input slots, preserving source parameter order and grouping.
419    pub fn input_conversions(&self) -> Result<Vec<&I>> {
420        self.params
421            .iter()
422            .map(|p| {
423                p.ty.slot.as_ref().ok_or_else(|| {
424                    format!("`{}`: parameter `{}` has no conversion", self.name, p.name)
425                })
426            })
427            .collect()
428    }
429    /// Filled result slot. Emission checks the rendered wire and value counts.
430    pub fn output_conversion(&self) -> Result<&O> {
431        let output = self
432            .ret
433            .slot
434            .as_ref()
435            .ok_or_else(|| format!("`{}`: result has no conversion", self.name))?;
436        Ok(output)
437    }
438    /// Boundary input wires read directly from the slots in parameter order.
439    pub fn inputs(&self) -> Result<impl Iterator<Item = &Wire<I::Wire>>> {
440        Ok(self
441            .input_conversions()?
442            .into_iter()
443            .flat_map(|i| i.wires()))
444    }
445    /// Output expressions paired with their wire descriptions, directly from the slot.
446    pub fn outputs(
447        &self,
448        context: &O::Context,
449    ) -> Result<impl ExactSizeIterator<Item = OutputValue<'_, O::Wire>>> {
450        let output = self.output_conversion()?;
451        let values = output.values(context)?;
452        if output.wires().len() != values.len() {
453            return Err(format!("`{}`: output wire/value count differs", self.name));
454        }
455        Ok(output
456            .wires()
457            .iter()
458            .zip(values)
459            .map(|(wire, value)| OutputValue { wire, value }))
460    }
461}

463/// Assemble a function call directly from filled slots, without a generator callback.
464/// Validates slot completeness, output arity and agreement on shared acknowledgement.
465/// The returned code runs inside the wrapper's fallible boundary, in this order:
466/// all validation, alias checks, one acknowledgement, all acquisition, argument
467/// construction, source invocation, output preparation, delivery and commit.
468/// Input and output contexts must match, even when the function has no parameters.
469///
470/// ```compile_fail
471/// use prebindgen_flat::{emit_call, Function, InputConversion, OutputConversion, EmissionContext};
472/// struct OtherContext(syn::Ident);
473/// impl EmissionContext for OtherContext {
474///     fn source_result(&self) -> &syn::Ident { &self.0 }
475/// }
476/// fn incompatible<I, O>(function: &Function<I, O>, context: &syn::Ident)
477/// where I: InputConversion<Context = syn::Ident>,
478///       O: OutputConversion<Context = OtherContext>,
479/// {
480///     emit_call(function, context).unwrap();
481/// }
482/// ```
483pub fn emit_call<I: InputConversion, O: OutputConversion<Context = I::Context>>(
484    function: &crate::Function<I, O>,
485    context: &I::Context,
486) -> Result<TokenStream> {
487    let inputs = function.input_conversions()?;
488    let output = function.output_conversion()?;
489    let _ = function.outputs(context)?;
490    let mut acknowledgement: Option<TokenStream> = None;
491    for input in &inputs {
492        let Some(fragment) = input.acknowledgement(context)? else {
493            continue;
494        };
495        if let Some(previous) = &acknowledgement {
496            if previous.to_string() != fragment.to_string() {
497                return Err(format!(
498                    "`{}`: conflicting input acknowledgements",
499                    function.name
500                ));
501            }
502        } else {
503            acknowledgement = Some(fragment);
504        }
505    }
506    let validation = inputs
507        .iter()
508        .map(|i| i.validation(context))
509        .collect::<Result<Vec<_>>>()?;
510    let resources: Vec<_> = inputs
511        .iter()
512        .map(|i| i.resources(context))
513        .collect::<Result<Vec<_>>>()?
514        .into_iter()
515        .flatten()
516        .collect();
517    let aliases = validate_resources(&resources);
518    let acquire = inputs
519        .iter()
520        .map(|i| i.acquisition(context))
521        .collect::<Result<Vec<_>>>()?;
522    let construct = inputs
523        .iter()
524        .map(|i| i.construction(context))
525        .collect::<Result<Vec<_>>>()?;
526    let args = inputs
527        .iter()
528        .map(|i| i.argument(context))
529        .collect::<Result<Vec<_>>>()?;
530    let path = &function.path;
531    let prepare = output.preparation(context)?;
532    let deliver = output.delivery(context)?;
533    let commit = output.commit(context)?;
534    let result = output.result(context)?;
535    let source = context.source_result();
536    Ok(quote! {
537        #(#validation)* #aliases #acknowledgement #(#acquire)* #(#construct)*
538        let #source = #path(#(#args),*);
539        #prepare #deliver #commit Ok(#result)
540    })
