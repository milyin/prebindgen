6/// Transport metadata that supplies the Rust carrier type of a wire.
7/// The returned type describes the boundary value, not the source value being converted.
8/// For example, a boolean conversion may use `u8` as its Rust carrier type.
9///
10/// ```
11/// use prebindgen_flat::WireKind;
12/// struct BoolCarrier;
13/// impl WireKind for BoolCarrier {
14///     fn rust_type(&self) -> syn::Type { syn::parse_quote!(u8) }
15/// }
16/// let ty = BoolCarrier.rust_type();
17/// assert_eq!(quote::quote!(#ty).to_string(), "u8");
18/// ```
19pub trait WireKind {
20    /// Rust type used to declare this wire in a wrapper signature or result structure.
21    fn rust_type(&self) -> syn::Type;
22}
23impl WireKind for syn::Type {
24    fn rust_type(&self) -> syn::Type {
25        self.clone()
26    }
27}

29/// One boundary parameter or result component used by wrapper emission.
30///
31/// `W` must implement [`WireKind`]. Call [`Wire::rust_type`]
32/// to obtain the Rust carrier type supplied by the kind.
33///
34/// ```
35/// use prebindgen_flat::Wire;
36/// let wire: Wire<syn::Type> = Wire {
37///     name: syn::parse_quote!(value),
38///     kind: syn::parse_quote!(i64),
39/// };
40/// let ty = wire.rust_type();
41/// assert_eq!(quote::quote!(#ty).to_string(), "i64");
42/// ```
43/// A wire kind without a Rust carrier type is rejected:
44///
45/// ```compile_fail
46/// use prebindgen_flat::Wire;
47/// struct Untyped;
48/// let wire = Wire { name: syn::parse_quote!(value), kind: Untyped };
49/// ```
50#[derive(Clone, Debug)]
51pub struct Wire<W: WireKind> {
52    /// Generated identifier used by conversion code and wrapper signatures.
53    pub name: syn::Ident,
54    /// Transport description implementing [`WireKind`].
55    /// Logical fields, selection rules and ownership behavior belong inside the converter.
56    pub kind: W,
57}
58
59impl<W: WireKind> Wire<W> {
60    /// Rust carrier type supplied by this wire's [`WireKind`].
61    /// Used to declare a wrapper parameter or result field.
62    pub fn rust_type(&self) -> syn::Type {
63        self.kind.rust_type()
64    }
65}

67/// An input conversion split into call-wide validation, acquisition and construction.
68///
69/// Optional concrete storage implementing [`InputConversion`]. [`emit_call`] orders
70/// them across all parameters; wrapper writers do not assemble them themselves.
71#[derive(Clone, Debug)]
72pub struct InputPlan<W: WireKind, C = ()> {
73    /// Context type supplied by the language generator at emission.
74    pub context: std::marker::PhantomData<fn(&C)>,
75    /// Flattened boundary parameters in signature order.
76    pub wires: Vec<Wire<W>>,
77    /// Shared call acknowledgement, if this input needs one.
78    pub acknowledgement: Option<TokenStream>,
79    /// Validate and prepare every active leaf before any ownership transfer.
80    pub validation: TokenStream,
81    /// Infallible acquisition of all owned inputs, before constructors can run.
82    pub acquire: TokenStream,
83    /// Active foreign resources whose identities must be checked across all arguments.
84    pub resources: Vec<Resource>,
85    /// Argument construction after all acquisitions, including source constructor calls.
86    pub statements: TokenStream,
87    /// Prepared value expression assigned to the caller-supplied target after `statements`.
88    pub argument: TokenStream,
89}
90/// Output conversion whose owned temporaries survive until successful delivery.
91///
92/// Optional concrete storage implementing [`OutputConversion`].
93/// [`emit_call`] commits only after adapter delivery succeeds.
94#[derive(Clone, Debug)]
95pub struct OutputPlan<W: WireKind, C = ()> {
96    /// Context type supplied by the language generator at emission.
97    pub context: std::marker::PhantomData<fn(&C)>,
98    /// Delivery and return code prepared with this result slot.
99    pub delivery: Delivery,
100    /// Boundary results in delivery order; one entry per expression in `values`.
101    pub wires: Vec<Wire<W>>,
102    /// Transfer ownership only after all foreign delivery operations succeed.
103    pub commit: TokenStream,
104    /// Code producing owned output temporaries and evaluating observations before moves.
105    pub statements: TokenStream,
106    /// Delivery expressions in wire order; they must remain valid until commit.
107    pub values: Vec<TokenStream>,
108}
109/// Access to one foreign resource during an exported call.
110#[derive(Clone, Copy, Debug, PartialEq, Eq)]
111pub enum Access {
112    /// Read-only borrow; multiple active shared accesses may alias.
113    Shared,
114    /// Mutable borrow; conflicts with every other active use of the resource.
115    Exclusive,
116    /// Ownership transfer; conflicts with every other active use of the resource.
117    Move,
118}
119/// The adapter supplies identity; flat enforces call-wide access conflicts.
120#[derive(Clone, Debug)]
121pub struct Resource {
122    /// Side-effect-free Rust expression identifying a live resource, such as an address.
123    /// All identities compared within a call must share an equality-compatible domain.
124    pub identity: TokenStream,
125    /// Borrow or ownership requirement imposed by this input conversion.
126    pub access: Access,
127    /// Side-effect-free Rust boolean expression; false resources do not conflict.
128    pub active: TokenStream,
129}
130/// Emit pairwise alias checks across all active input resources.
131///
132/// `resources` holds the [`InputPlan::resources`] of all of one exported
133/// function's parameters together, so that aliases across parameters are found.
134///
135/// Run this code after leaf validation and before **any** ownership acquisition.
136/// Shared/shared pairs and resources whose predicates are not both true do not conflict.
137/// Conflicts return `Err("conflicting input aliases".into())` from the enclosing
138/// wrapper. This does not validate pointer liveness, synchronize threads, or
139/// acquire resources; those remain adapter/caller responsibilities.
140pub fn validate_resources(resources: &[Resource]) -> TokenStream {
141    let mut result = TokenStream::new();
142    for (i, a) in resources.iter().enumerate() {
143        for b in &resources[i + 1..] {
144            if a.access == Access::Shared && b.access == Access::Shared {
145                continue;
146            }
147            let x = &a.identity;
148            let y = &b.identity;
149            let active_a = &a.active;
150            let active_b = &b.active;
151            let conflict = quote!((#active_a) && (#active_b) && ((#x) == (#y)));
152            result.extend(quote!(if #conflict {return Err("conflicting input aliases".into());}));
153        }
154    }
155    result
156}

158/// Check that every bound function has an input converter for each parameter
159/// and an output converter, including unit results.
160///
161/// Functions with entirely empty slots remain source-only helpers. This check
162/// neither builds conversions nor changes the model. Adapters call it before
163/// writing wrappers. It does not validate code fragments supplied directly by callers.
164pub fn validate_bindings<I: InputConversion, O: OutputConversion>(
165    flat: &crate::Flat<I, O>,
166) -> Result<()> {
167    for f in flat.functions.values() {
168        if f.ret.slot.is_some() {
169            for p in &f.params {
170                if p.ty.slot.is_none() {
171                    return Err(format!(
172                        "`{}`: parameter `{}` has no input converter",
173                        f.name, p.name
174                    ));
175                }
176            }
177        } else if f.params.iter().any(|p| p.ty.slot.is_some()) {
178            return Err(format!(
179                "`{}`: bound inputs have no output converter",
180                f.name
181            ));
182        }
183    }
184    Ok(())
185}
186/// Format generated tokens as a Rust source file.
187///
188/// `tokens` must be a sequence of Rust items. Returns a syntax error if they do not form a file. This does not
189/// resolve names or type-check generated code; the binding crate must compile it.
190pub fn rust_source(tokens: TokenStream) -> Result<String> {
191    let file = syn::parse2(tokens).map_err(|e| format!("generated Rust: {e}"))?;
192    Ok(prettyplease::unparse(&file))
193}

202/// A fully prepared input conversion stored in a slot.
203/// Metadata and generated names must stay consistent across method calls.
204/// Code runs in the environment supplied by the language wrapper writer.
205/// Preparation resolves errors before installing the slot; [`emit_call`]
206/// reads this interface directly, without an intermediate plan.
207///
208/// This conversion accepts one opaque pointer wire and constructs a
209/// `Box<Document>` argument for a function such as `publish(document: Box<Document>)`.
210/// It demonstrates every method, including ownership transfer and alias checking.
211///
212/// The wrapper supplies an `&mut bool` acknowledgement parameter, initially false.
213/// Validation or alias-check failure leaves it false and ownership with the caller.
214/// After acknowledgement, the wrapper owns the document even if later code fails.
215/// The pointer must come from a compatible `Box<Document>::into_raw`, remain live,
216/// and be used exclusively for this transfer. A null check cannot establish those
217/// conditions.
218///
219/// ```
220/// use prebindgen_flat::{Access, InputConversion, Resource, Result, Wire, WireKind};
221/// use proc_macro2::TokenStream;
222/// use quote::quote;
223/// use syn::{parse_quote, Ident, Type};
224///
225/// // The generator defines a closed set of supported boundary carriers.
226/// enum WireType {
227///     DocumentHandle,
228/// }
229/// impl WireKind for WireType {
230///     fn rust_type(&self) -> Type {
231///         match self {
232///             Self::DocumentHandle => parse_quote!(*mut ::std::ffi::c_void),
233///         }
234///     }
235/// }
236///
237/// // Name of the wrapper's &mut bool parameter, supplied during generation.
238/// struct Context {
239///     acknowledged: Ident,
240/// }
241///
242/// struct DocumentInput {
243///     wires: [Wire<WireType>; 1],
244///     // Preparation chooses a private temporary unique within the wrapper.
245///     owned: Ident,
246/// }
247/// impl InputConversion for DocumentInput {
248///     type Wire = WireType;
249///     type Context = Context;
250///
251///     fn wires(&self) -> &[Wire<WireType>] {
252///         &self.wires
253///     }
254///
255///     fn resources(&self, _: &Context) -> Result<Vec<Resource>> {
256///         let wire = &self.wires[0].name;
257///         Ok(vec![Resource {
258///             identity: quote!(#wire as usize),
259///             access: Access::Move,
260///             active: quote!(true),
261///         }])
262///     }
263///
264///     fn validation(&self, _: &Context) -> Result<TokenStream> {
265///         let wire = &self.wires[0].name;
266///         Ok(quote! {
267///             if #wire.is_null() {
268///                 return Err("null document handle".into());
269///             }
270///         })
271///     }
272///
273///     fn acknowledgement(&self, context: &Context) -> Result<Option<TokenStream>> {
274///         let acknowledged = &context.acknowledged;
275///         // Shared across consuming slots; emit_call emits it once, after checks.
276///         Ok(Some(quote!(*#acknowledged = true;)))
277///     }
278///
279///     fn acquisition(&self, _: &Context) -> Result<TokenStream> {
280///         let wire = &self.wires[0].name;
281///         let owned = &self.owned;
282///         // Document is the source Rust type in the generated wrapper's scope.
283///         // Safety relies on the caller's pointer and ownership contract above.
284///         Ok(quote! {
285///             let #owned = unsafe { Box::<Document>::from_raw(#wire.cast::<Document>()) };
286///         })
287///     }
288///
289///     fn construction(&self, _: &Context, target: &Ident) -> Result<TokenStream> {
290///         let owned = &self.owned;
291///         Ok(quote!(let #target = #owned;))
292///     }
293/// }
294///
295/// let slot = DocumentInput {
296///     wires: [Wire {
297///         name: parse_quote!(raw_document),
298///         kind: WireType::DocumentHandle,
299///     }],
300///     owned: parse_quote!(__owned_document),
301/// };
302/// let context = Context { acknowledged: parse_quote!(ownership_taken) };
303///
304/// // Individual phases are available directly. emit_call orders them across
305/// // all input slots, checks resource aliases, and invokes the source function.
306/// assert_eq!(slot.wires().len(), 1);
307/// assert_eq!(slot.resources(&context)?[0].access, Access::Move);
308/// let validation = slot.validation(&context)?;
309/// let acknowledgement = slot.acknowledgement(&context)?;
310/// let acquisition = slot.acquisition(&context)?;
311/// let target = parse_quote!(document);
312/// let construction = slot.construction(&context, &target)?;
313/// assert_eq!(construction.to_string(),
314///            quote!(let document = __owned_document;).to_string());
315///
316/// // Inspect the emitted phases without compiling or executing a foreign pointer.
317/// let phases = quote! {
318///     #validation
319///     #acknowledgement
320///     #acquisition
321///     #construction
322/// };
323/// let _: syn::Block = syn::parse2(quote!({ #phases }))?;
324/// # Ok::<(), Box<dyn std::error::Error>>(())
325/// ```
326///
327/// The converter owns any conditional behavior. Wires describe transport only.
328pub trait InputConversion {
329    /// Adapter-specific boundary description.
330    type Wire: WireKind;
331    /// Emission data defined and supplied by the language generator.
332    type Context;
333    /// Boundary parameters in signature order.
334    fn wires(&self) -> &[Wire<Self::Wire>];
335    /// Shared call acknowledgement, after validation and before acquisition.
336    /// All contributing inputs must return the same token spelling; assembly emits
337    /// it once. Return `None` when this input needs no acknowledgement. Slot-local
338    /// fallible checks belong in `validation`, not in this shared operation.
339    fn acknowledgement(&self, _context: &Self::Context) -> Result<Option<TokenStream>> {
340        Ok(None)
341    }
342    /// Resources used for call-wide alias checking, with identities emitted for this context.
343    fn resources(&self, context: &Self::Context) -> Result<Vec<Resource>>;
344    /// Fallible validation before any ownership transfer.
345    fn validation(&self, context: &Self::Context) -> Result<TokenStream>;
346    /// Infallible ownership acquisition after acknowledgement.
347    fn acquisition(&self, context: &Self::Context) -> Result<TokenStream>;
348    /// Construct the argument after all inputs have been acquired, binding it to `target`.
349    /// The caller owns this identifier and passes it to the source function. Emit a
350    /// `let` binding in the surrounding scope, so borrowed argument storage remains alive.
351    /// Converter-private temporaries must not shadow other caller-supplied bindings.
352    fn construction(&self, context: &Self::Context, target: &syn::Ident) -> Result<TokenStream>;
353}
354impl<W: WireKind, C: TemplateContext> InputConversion for InputPlan<W, C> {
355    type Wire = W;
356    type Context = C;
357    fn wires(&self) -> &[Wire<Self::Wire>] {
358        &self.wires
359    }
360    fn acknowledgement(&self, context: &Self::Context) -> Result<Option<TokenStream>> {
361        self.acknowledgement
362            .as_ref()
363            .map(|code| context.render(code))
364            .transpose()
365    }
366    fn resources(&self, context: &Self::Context) -> Result<Vec<Resource>> {
367        self.resources
368            .iter()
369            .map(|resource| {
370                Ok(Resource {
371                    identity: context.render(&resource.identity)?,
372                    access: resource.access,
373                    active: context.render(&resource.active)?,
374                })
375            })
376            .collect()
377    }
378    fn validation(&self, context: &Self::Context) -> Result<TokenStream> {
379        context.render(&self.validation)
380    }
381    fn acquisition(&self, context: &Self::Context) -> Result<TokenStream> {
382        context.render(&self.acquire)
383    }
384    fn construction(&self, context: &Self::Context, target: &syn::Ident) -> Result<TokenStream> {
385        let statements = context.render(&self.statements)?;
386        let value = context.render(&self.argument)?;
387        Ok(quote!(#statements let #target = #value;))
388    }
389}
390/// A fully prepared output conversion stored in a slot.
391/// Metadata and generated names must stay consistent across method calls.
392/// Code runs in the environment supplied by the language wrapper writer.
393/// Preparation resolves errors before installing the slot; [`emit_call`]
394/// reads this interface directly, without an intermediate plan.
395/// Source-value templates use [`source_result_expr`]; custom implementations can
396/// use the explicitly supplied source binding directly. All output phases receive
397/// the same binding, chosen by the caller, for the source function's return value.
398pub trait OutputConversion {
399    /// Adapter-specific boundary description.
400    type Wire: WireKind;
401    /// Emission data defined and supplied by the language generator.
402    type Context;
403    /// Boundary results in delivery order.
404    fn wires(&self) -> &[Wire<Self::Wire>];
405    /// Output expressions in wire order, valid until commit.
406    fn values(&self, context: &Self::Context, source: &syn::Ident) -> Result<Vec<TokenStream>>;
407    /// Prepare owned output temporaries before delivery.
408    fn preparation(&self, context: &Self::Context, source: &syn::Ident) -> Result<TokenStream>;
409    /// Fallible delivery code; ownership remains uncommitted on failure.
410    fn delivery(&self, context: &Self::Context, source: &syn::Ident) -> Result<TokenStream>;
411    /// Infallibly transfer ownership after successful delivery.
412    fn commit(&self, context: &Self::Context, source: &syn::Ident) -> Result<TokenStream>;
413    /// Infallible wrapper result expression evaluated after commit.
414    fn result(&self, context: &Self::Context, source: &syn::Ident) -> Result<TokenStream>;
415}
416impl<W: WireKind, C: TemplateContext> OutputConversion for OutputPlan<W, C> {
417    type Wire = W;
418    type Context = C;
419    fn wires(&self) -> &[Wire<Self::Wire>] {
420        &self.wires
421    }
422    fn values(&self, context: &Self::Context, source: &syn::Ident) -> Result<Vec<TokenStream>> {
423        self.values
424            .iter()
425            .map(|code| crate::emission::render_output(context, code, source))
426            .collect()
427    }
428    fn preparation(&self, context: &Self::Context, source: &syn::Ident) -> Result<TokenStream> {
429        crate::emission::render_output(context, &self.statements, source)
430    }
431    fn delivery(&self, context: &Self::Context, source: &syn::Ident) -> Result<TokenStream> {
432        crate::emission::render_output(context, &self.delivery.statements, source)
433    }
434    fn commit(&self, context: &Self::Context, source: &syn::Ident) -> Result<TokenStream> {
435        crate::emission::render_output(context, &self.commit, source)
436    }
437    fn result(&self, context: &Self::Context, source: &syn::Ident) -> Result<TokenStream> {
438        crate::emission::render_output(context, &self.delivery.result, source)
439    }
440}

467impl<I: InputConversion, O: OutputConversion> crate::Function<I, O> {
468    /// Filled input slots, preserving source parameter order and grouping.
469    pub fn input_conversions(&self) -> Result<Vec<&I>> {
470        self.params
471            .iter()
472            .map(|p| {
473                p.ty.slot.as_ref().ok_or_else(|| {
474                    format!("`{}`: parameter `{}` has no conversion", self.name, p.name)
475                })
476            })
477            .collect()
478    }
479    /// Filled result slot. Emission checks the rendered wire and value counts.
480    pub fn output_conversion(&self) -> Result<&O> {
481        let output = self
482            .ret
483            .slot
484            .as_ref()
485            .ok_or_else(|| format!("`{}`: result has no conversion", self.name))?;
486        Ok(output)
487    }
488    /// Boundary input wires read directly from the slots in parameter order.
489    pub fn inputs(&self) -> Result<impl Iterator<Item = &Wire<I::Wire>>> {
490        Ok(self
491            .input_conversions()?
492            .into_iter()
493            .flat_map(|i| i.wires()))
494    }
495    /// Output expressions paired with their wire descriptions, directly from the slot.
496    pub fn outputs(
497        &self,
498        context: &O::Context,
499        source: &syn::Ident,
500    ) -> Result<impl ExactSizeIterator<Item = OutputValue<'_, O::Wire>>> {
501        let output = self.output_conversion()?;
502        let values = output.values(context, source)?;
503        if output.wires().len() != values.len() {
504            return Err(format!("`{}`: output wire/value count differs", self.name));
505        }
506        Ok(output
507            .wires()
508            .iter()
509            .zip(values)
510            .map(|(wire, value)| OutputValue { wire, value }))
511    }
512}

514/// Assemble a function call directly from filled slots, without a generator callback.
515/// Validates slot completeness, output arity and agreement on shared acknowledgement.
516/// The returned code runs inside the wrapper's fallible boundary, in this order:
517/// all validation, alias checks, one acknowledgement, all acquisition, argument
518/// construction, source invocation, output preparation, delivery and commit.
519/// `source` names the variable that receives the source function's return value.
520/// Input and output contexts must match, even when the function has no parameters.
521///
522/// ```compile_fail
523/// use prebindgen_flat::{emit_call, Function, InputConversion, OutputConversion};
524/// struct OtherContext(syn::Ident);
525/// fn incompatible<I, O>(function: &Function<I, O>, context: &syn::Ident)
526/// where I: InputConversion<Context = syn::Ident>,
527///       O: OutputConversion<Context = OtherContext>,
528/// {
529///     emit_call(function, context, &syn::parse_quote!(returned)).unwrap();
530/// }
531/// ```
532pub fn emit_call<I: InputConversion, O: OutputConversion<Context = I::Context>>(
533    function: &crate::Function<I, O>,
534    context: &I::Context,
535    source: &syn::Ident,
536) -> Result<TokenStream> {
537    let inputs = function.input_conversions()?;
538    let output = function.output_conversion()?;
539    let _ = function.outputs(context, source)?;
540    let mut acknowledgement: Option<TokenStream> = None;
541    for input in &inputs {
542        let Some(fragment) = input.acknowledgement(context)? else {
543            continue;
544        };
545        if let Some(previous) = &acknowledgement {
546            if previous.to_string() != fragment.to_string() {
547                return Err(format!(
548                    "`{}`: conflicting input acknowledgements",
549                    function.name
550                ));
551            }
552        } else {
553            acknowledgement = Some(fragment);
554        }
555    }
556    let validation = inputs
557        .iter()
558        .map(|i| i.validation(context))
559        .collect::<Result<Vec<_>>>()?;
560    let resources: Vec<_> = inputs
561        .iter()
562        .map(|i| i.resources(context))
563        .collect::<Result<Vec<_>>>()?
564        .into_iter()
565        .flatten()
566        .collect();
567    let aliases = validate_resources(&resources);
568    let acquire = inputs
569        .iter()
570        .map(|i| i.acquisition(context))
571        .collect::<Result<Vec<_>>>()?;
572    let path = &function.path;
573    let prepare = output.preparation(context, source)?;
574    let deliver = output.delivery(context, source)?;
575    let commit = output.commit(context, source)?;
576    let result = output.result(context, source)?;
577    let mut used: std::collections::BTreeSet<String> = inputs
578        .iter()
579        .flat_map(|input| input.wires().iter().map(|wire| wire.name.to_string()))
580        .chain(std::iter::once(source.to_string()))
581        .collect();
582    // Reserve names already used by surrounding phases and the source path.
583    // These tokens are emitted once; their code and evaluation order stay unchanged.
584    reserve_identifiers(
585        &quote!(#path #(#validation)* #aliases #acknowledgement #(#acquire)*
586        #prepare #deliver #commit #result),
587        &mut used,
588    );
589    let args: Vec<_> = (0..inputs.len())
590        .map(|index| {
591            let mut name = format!("__prebindgen_argument_{index}");
592            while !used.insert(name.clone()) {
593                name.push('_');
594            }
595            syn::Ident::new(&name, proc_macro2::Span::call_site())
596        })
597        .collect();
598    let construct = inputs
599        .iter()
600        .zip(&args)
601        .map(|(input, target)| input.construction(context, target))
602        .collect::<Result<Vec<_>>>()?;
603    Ok(quote! {
604        #(#validation)* #aliases #acknowledgement #(#acquire)* #(#construct)*
605        let #source = #path(#(#args),*);
606        #prepare #deliver #commit Ok(#result)
607    })
608}

610fn reserve_identifiers(code: &TokenStream, used: &mut std::collections::BTreeSet<String>) {
611    for token in code.clone() {
612        match token {
613            proc_macro2::TokenTree::Ident(name) => {
614                used.insert(name.to_string());
615            }
616            proc_macro2::TokenTree::Group(group) => reserve_identifiers(&group.stream(), used),
617            _ => {}
618        }
619    }
