7pub(crate) fn compatible(element: Primitive) -> bool {
8    !matches!(
9        element,
10        Primitive::Bool | Primitive::Char | Primitive::I128 | Primitive::U128
11    )
12}
13/// A primitive whose configured C representation is still the native scalar.
14pub(crate) fn primitive(
15    ty: &syn::Type,
16    types: &BTreeMap<String, prebindgen_flat::TypeInfo<CType>>,
17) -> Option<Primitive> {
18    let primitive = Primitive::from_type(ty).filter(|p| compatible(*p))?;
19    if types.values().any(|info| {
20        Primitive::from_type(&info.ty) == Some(primitive)
21            && (info.metadata.pointer.is_some()
22                || info.metadata.owned.is_some()
23                || info.metadata.enumeration.is_some())
24    }) {
25        return None;
26    }
27    Some(primitive)
28}
29pub(crate) fn native_element_type(element: Primitive) -> syn::Type {
30    let name = format_ident!("{}", prebindgen_flat::type_key(&element.rust_type()));
31    syn::parse_quote!(::core::primitive::#name)
32}
33pub(crate) fn ident(element: Primitive) -> syn::Ident {
34    format_ident!(
35        "__PbSliceView_{}",
36        prebindgen_flat::type_key(&element.rust_type())
37    )
38}
39pub(crate) fn c_name(element: Primitive, prefix: &str) -> String {
40    format!(
41        "{prefix}_{}_slice_view",
42        prebindgen_flat::type_key(&element.rust_type())
43    )
44}
45#[allow(clippy::too_many_arguments)]
46pub(crate) fn sequence(
47    flat: &CBindings,
48    ty: &syn::Type,
49    wire: syn::Ident,
50    value: TokenStream,
51    recipe: &COutput,
52    delivery: SequenceDelivery,
53    temporary: bool,
54) -> Result<Option<COutputPlan<CWire>>> {
55    // An explicitly requested visitor does not allocate row storage. Its leaves
56    // still enforce their own ownership/copy policies.
57    if delivery == SequenceDelivery::Callback {
58        return Ok(None);
59    }
60    if temporary
61        && !matches!(ty, syn::Type::Reference(r) if r.lifetime.as_ref().is_some_and(|l|l.ident=="static"))
62    {
63        return Err("borrowed C collection view cannot escape a temporary owner; return its owner alongside a generated heap view getter or explicitly permit a collection copy".into());
64    }
65    let (element_type, view) =
66        prebindgen_flat::sequence_parts(ty).ok_or("borrowed sequence requires a collection")?;
67    if view == prebindgen_flat::SequenceView::Mutable {
68        return Err("mutable C collection views are unsupported".into());
69    }
70    let primitive = primitive(element_type, &flat.types);
71    let Some(element) = primitive.filter(|_| matches!(recipe, COutput::Leaf(COutputLeafOp::Value)))
72    else {
73        return Err(format!("borrowed C collection `{}` cannot allocate converted rows implicitly; use an address-preserving numeric element representation or explicitly permit a collection-level copy with expand_return!(&[T]).field_self_ref() (not an element-only expansion)", prebindgen_flat::type_key(ty)));
74    };
75    let element_type = native_element_type(element);
76    let carrier = ident(element);
77    let source = format_ident!("{wire}_source");
78    Ok(Some(COutputPlan {
79        fallible: false,
80        context: Default::default(),
81        delivery: Default::default(),
82        commit: TokenStream::new(),
83        outcomes: prebindgen_flat::Outcomes {
84            success: prebindgen_flat::WireNode::Leaf(vec![Wire {
85                name: wire.clone(),
86                kind: CWire::SliceView { element },
87            }]),
88            source_error: None,
89            binding_failure: None,
90        },
91        statements: quote!(
92            let #source = &(#value);
93            let #wire: &[#element_type] = ::core::convert::AsRef::<[#element_type]>::as_ref(#source);
94            let #wire = #carrier { data: <[#element_type]>::as_ptr(#wire), len: <[#element_type]>::len(#wire) };
95        ),
96        values: vec![(wire.clone(), quote!(#wire))],
97    }))
98}
99/// A directly borrowed foreign slice, without reconstructing a Rust Vec owner.
100pub(crate) fn input_sequence(
101    flat: &CBindings,
102    ty: &syn::Type,
103    wire: syn::Ident,
104    var: syn::Ident,
105    recipe: &crate::operations::CInput,
106    active: &prebindgen_declarations::Condition,
107) -> Result<CInputPlan<CWire>> {
108    let syn::Type::Reference(reference) = ty else {
109        return Err("borrowed C input requires a reference".into());
110    };
111    if reference.mutability.is_some() {
112        return Err("mutable C collection inputs are unsupported".into());
113    }
114    if reference
115        .lifetime
116        .as_ref()
117        .is_some_and(|l| l.ident == "static")
118    {
119        return Err("shared C collection inputs cannot promise a 'static lifetime".into());
120    }
121    let target = &reference.elem;
122    let (source_element, array_length) = match target.as_ref() {
123        syn::Type::Slice(slice) => (slice.elem.as_ref(), None),
124        syn::Type::Array(array) => (array.elem.as_ref(), Some(&array.len)),
125        _ => return Err(format!("shared C input `{}` cannot reconstruct an owning collection implicitly; declare a native opaque owner input or explicitly select expand_param!({}).reconstruct()",prebindgen_flat::type_key(ty),prebindgen_flat::type_key(ty))),
126    };
127    let Some(element) = primitive(source_element, &flat.types)
128        .filter(|_| matches!(recipe, crate::operations::CInput::Leaf(CInputLeafOp::Value)))
129    else {
130        return Err(format!("shared C input `{}` cannot allocate converted elements implicitly; use a compatible numeric slice, a native opaque owner, or explicitly select expand_param!({}).reconstruct()",prebindgen_flat::type_key(ty),prebindgen_flat::type_key(ty)));
131    };
132    let native = native_element_type(element);
133    let condition = active.rust();
134    let pointer = format_ident!("{var}_pointer");
135    let slice = format_ident!("{var}_slice");
136    let argument = if array_length.is_some() {
137        quote!(unsafe {&*(#slice.as_ptr() as *const [#native;#array_length])})
138    } else {
139        quote!(#slice)
140    };
141    Ok(CInputPlan::new(
142        prebindgen_flat::WireNode::Leaf(vec![Wire {
143            name: wire.clone(),
144            kind: CWire::SliceView { element },
145        }]),
146        quote!(let #var=if #condition {
147        let #pointer=if #wire.len==0 && #wire.data.is_null(){::core::ptr::NonNull::<#native>::dangling().as_ptr() as *const #native}else{#wire.data};
148        let #slice=unsafe {::core::slice::from_raw_parts(#pointer,#wire.len)};Some(#argument)
149    } else {None};),
150        quote!(#var.expect("active slice view")),
151    ))
152}
153/// A loan anywhere in a packet, including fields inside converted rows.
154pub(crate) fn has_loans(node: &prebindgen_flat::WireNode<CWire, CChain, CBranch>) -> bool {
155    prebindgen_flat::wire_definitions(std::iter::once(node)).any(|wire| {
156        matches!(
157            wire.kind,
158            CWire::View
159                | CWire::MutableView
160                | CWire::SliceView { .. }
161                | CWire::Owned {
162                    access: Access::Shared | Access::Exclusive,
163                    ..
164                }
165        ) || matches!(&wire.kind, CWire::Pointer {pointer,..} if pointer.access != Access::Move)
166    })
167}
168pub(crate) fn collect(
169    node: &prebindgen_flat::WireNode<CWire, CChain, CBranch>,
170    out: &mut BTreeMap<String, Primitive>,
171) {
172    for wire in prebindgen_flat::wire_definitions(std::iter::once(node)) {
173        match &wire.kind {
174            CWire::SliceView { element } => {
175                out.insert(prebindgen_flat::type_key(&element.rust_type()), *element);
176            }
177            CWire::Callback { callback } => callback.slice_views(out),
178            _ => {}
179        }
180    }
181}
182pub(crate) fn emit(
183    element: Primitive,
184    prefix: &str,
185    symbols: &mut BTreeSet<String>,
186    tokens: &mut TokenStream,
187    header: &mut String,
188) -> Result<()> {
189    let name = c_name(element, prefix);
190    if !symbols.insert(name.clone()) {
191        return Err("generated C slice-view type collides with another C symbol".into());
192    }
193    let rust = ident(element);
194    let ty = native_element_type(element);
195    tokens.extend(quote!(#[allow(non_camel_case_types)] #[repr(C)] #[derive(Clone,Copy)] pub struct #rust {pub data: *const #ty,pub len: usize}));
196    let c = c_scalar(element, prefix);
197    header.push_str(&format!("/* Shared borrowed slice; never free or consume. Keep source storage alive; do not mutate it or obtain overlapping exclusive/mutable access until the view's last use. Empty views may have a non-null dangling data pointer; never dereference when len is zero. Callback argument views last only for that callback invocation. */\ntypedef struct {{ const {c} *data; size_t len; }} {name};\n"));
198    Ok(())
199}
