22/// Start input expansion using this adapter's node constructors.
23pub fn expand_param(ty: Type) -> DeclCExpandParam {
24    prebindgen_declarations::DeclExpandParam::new(
25        ty,
26        |function: DeclCFunction| {
27            prebindgen_declarations::constructor_input(
28                function,
29                CInputConfig::call,
30                |function, parameters| {
31                    let (parameters, children) = parameters.into_iter().unzip();
32                    CInputConfig::Branch {
33                        operation: CInputBranch::ConfigConstructorArguments(
34                            CConstructorArguments {
35                                function,
36                                parameters,
37                            },
38                        ),
39                        children,
40                    }
41                },
42            )
43        },
44        || CInputConfig::Leaf(CInputLeafOp::Value),
45        |ty, alternatives, extension| {
46            let input = alternatives.build(CInputConfig::selector, CInputConfig::auto, |child| {
47                CInputConfig::Unary {
48                    operation: CInputUnary::Temporary,
49                    child: Box::new(child),
50                }
51            });
52            let _ = extension;
53            CInputConfig::for_type(ty, input)
54        },
55    )
56}
57/// Start return expansion using this adapter's node constructors.
58pub fn expand_return(ty: Type) -> DeclCExpandReturn {
59    prebindgen_declarations::DeclExpandReturn::new(
60        ty,
61        |function| {
62            prebindgen_declarations::getter_view(
63                function,
64                |name, result| {
65                    COutputConfig::getter(name, result.unwrap_or_else(COutputConfig::auto))
66                },
67                |name, child| COutputConfig::Unary {
68                    operation: COutputUnary::Named { name },
69                    child: Box::new(child),
70                },
71            )
72        },
73        || COutputConfig::Leaf(COutputLeafOp::Value),
74        |policy| COutputConfig::Leaf(COutputLeafOp::Projected(policy)),
75        super::configuration::owned_field_view,
76        |ty, views, extension| {
77            let output = COutputConfig::combine(views);
78            let _ = extension;
79            COutputConfig::for_type(ty, output)
80        },
81    )
82}
83/// Observe named source-record fields, optionally through an accessor.
84pub fn fields(function: impl Into<Option<syn::Ident>>) -> DeclCFields {
85    prebindgen_declarations::DeclFields::new(
86        function,
87        |output| match output {
88            COutputConfig::Unary {
89                operation: COutputUnary::ConfigForType { ty },
90                child: output,
91            } if matches!(output.as_ref(),COutputConfig::Branch {operation:COutputBranch::Combine,children} if children.is_empty()) => {
92                COutputConfig::for_type(*ty, COutputConfig::Leaf(COutputLeafOp::Discard))
93            }
94            output => output,
95        },
96        |function, overrides, _extension| {
97            let (fields, children) = overrides.into_iter().unzip();
98            let output = COutputConfig::Branch {
99                operation: COutputBranch::ConfigNamedFields(CNamedFields {
100                    fields,
101                    mode: prebindgen_declarations::ProjectionMode::Borrowed,
102                }),
103                children,
104            };
105            match function {
106                Some(function) => COutputConfig::getter(function, output),
107                None => output,
108            }
109        },
110    )
111}

117/// Callback declaration containing this language's argument and result trees.
118pub type DeclCCallback = prebindgen_declarations::DeclCallback<CInputConfig, COutputConfig>;
119impl prebindgen_declarations::CallbackInput<CInputConfig, COutputConfig> for CInputConfigOps {
120    fn callback_input(declaration: DeclCCallback) -> CInputConfig {
121        prebindgen_declarations::callback_input(
122            declaration,
123            CInputConfig::auto,
124            |children| COutputConfig::Branch {
125                operation: COutputBranch::Arguments,
126                children,
127            },
128            |metadata, domain, codomain| CInputConfig::Arrow {
129                operation: CInputArrow::ConfigCallback(Box::new(metadata)),
130                domain: Box::new(domain),
131                codomain: Box::new(codomain),
132            },
133            CInputConfig::for_type,
134        )
135    }
136}

141pub fn param_fields() -> DeclCParamFields {
142    prebindgen_declarations::DeclFields::new(
143        None,
144        |input| input,
145        |_function, overrides, _extension| {
146            let (fields, children) = overrides.into_iter().unzip();
147            CInputConfig::Branch {
148                operation: CInputBranch::ConfigNamedFields(CInputFields { fields }),
149                children,
150            }
151        },
152    )
