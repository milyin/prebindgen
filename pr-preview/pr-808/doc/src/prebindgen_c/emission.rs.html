1//! Bindings supplied by this language generator during slot emission.
2use prebindgen_conversions::{render_template, TemplateContext};
3use prebindgen_flat::Result;
4use proc_macro2::TokenStream;
5use quote::ToTokens;
6
7/// Concrete wrapper bindings used by this language generator.
8#[derive(Clone, Debug)]
9pub struct CContext {
10    /// Generated ABI row types keyed by sequence carrier name in this function.
11    pub row_types: std::collections::BTreeMap<String, syn::Ident>,
12    /// Wrapper-local Cell counting acquired owned wire values.
13    pub acknowledgement: syn::Ident,
14    /// Wrapper parameter receiving the result structure.
15    pub output_pointer: syn::Ident,
16    /// Concrete result structure constructor.
17    pub output_type: syn::Path,
18    /// Optional explicit Result error output parameter.
19    pub error_pointer: syn::Ident,
20    /// Records that a source error has already been delivered.
21    pub error_reported: syn::Ident,
22    /// Infallible statements appended after output ownership commits.
23    pub output_committed: TokenStream,
24    /// Byte buffer carrier constructor.
25    pub buffer_type: syn::Path,
26    /// Wide integer carrier constructor.
27    pub wide_integer_type: syn::Path,
28    /// Helper that reads an input byte buffer.
29    pub read_buffer: syn::Path,
30}
31impl TemplateContext for CContext {
32    fn render(&self, code: &TokenStream) -> Result<TokenStream> {
33        render_template(code, &|name| match name {
34            "input_consumed" => {
35                let counter = &self.acknowledgement;
36                Ok(quote::quote!(&#counter))
37            }
38            "output_committed" => Ok(self.output_committed.clone()),
39            "error_pointer" => Ok(self.error_pointer.to_token_stream()),
40            "error_reported" => Ok(self.error_reported.to_token_stream()),
41            "acknowledgement" => Ok(self.acknowledgement.to_token_stream()),
42            "output_pointer" => Ok(self.output_pointer.to_token_stream()),
43            "output_type" => Ok(self.output_type.to_token_stream()),
44            "buffer_type" => Ok(self.buffer_type.to_token_stream()),
45            "wide_integer_type" => Ok(self.wide_integer_type.to_token_stream()),
46            "read_buffer" => Ok(self.read_buffer.to_token_stream()),
47            _ if name.starts_with("row_") => self
48                .row_types
49                .get(&name[4..])
50                .map(ToTokens::to_token_stream)
51                .ok_or_else(|| format!("unknown collection row `{name}`")),
52            _ => Err(format!("unknown emission parameter `{name}`")),
53        })
54    }
55}
