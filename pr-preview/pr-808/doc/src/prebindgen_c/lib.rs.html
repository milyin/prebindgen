6/// The carrier of one C wire.
7///
8/// This adapter-specific type keeps C and JNI converters from being mixed in
9/// one model: their codecs use different carriers and allocation layouts.
10#[derive(Clone, Debug)]
11pub enum CWire {
12    /// A primitive in its C scalar type.
13    Scalar(Primitive),
14    /// Text or bytes as a pointer/length buffer.
15    Buffer,
16    /// A typed opaque pointer to a boxed source value.
17    Pointer(Box<BoxedWire>),
18    /// An `int32_t` selector arm index.
19    Selector,
20}

34/// Conversions and slot glue used by [`Cbindgen`].
35/// Prepared code uses explicit emission parameters for the output pointer and
36/// result type. The writer supplies these bindings and the error boundary.
37/// Use with [`prebindgen_declarations::bind_input`] and [`prebindgen_declarations::bind_output`]
38/// to fill [`Bindings`] manually.
39///
40/// ```
41/// use prebindgen_c::{Bindings, Cbindgen, CCodec};
42/// use prebindgen_declarations::{bind_input, bind_output, Input, Output};
43/// let source = syn::parse_file("pub fn echo(x: i64) -> i64 { x }")?;
44/// let mut flat = Bindings::parse(
45///     source.items.into_iter().map(|item| (item, Default::default())),
46/// )?;
47/// bind_input(&mut flat, &CCodec, "echo", "x", &Input::Leaf, &())?;
48/// bind_output(&mut flat, &CCodec, "echo", &Output::Leaf, &())?;
49/// let generated = Cbindgen::new(flat).build()?;
50/// assert!(generated.header.contains("pb_echo("));
51/// syn::parse_file(&generated.rust)?;
52/// # Ok::<(), Box<dyn std::error::Error>>(())
53/// ```
54pub struct CCodec;
55impl Codec for CCodec {
56    type Extra = ();
57    type Wire = CWire;
58    type Context = CContext;
59    fn finish_output(&self, plan: &mut OutputPlan<CWire>, _: &()) -> Result<()> {
60        let mut values: Vec<_> = plan
61            .values
62            .iter()
63            .enumerate()
64            .map(|(i, value)| {
65                let field = format_ident!("v{i}");
66                quote!(#field: #value)
67            })
68            .collect();
69        if values.is_empty() {
70            values.push(quote!(unused: 0));
71        }
72        let result_type = prebindgen_flat::template_parameter("output_type");
73        let out = prebindgen_flat::template_parameter("output_pointer");
74        plan.delivery = prebindgen_flat::Delivery {
75            statements: quote!(let result = #result_type { #(#values),* }; std::ptr::write(#out, result);),
76            result: quote!(()),
77        };
78        Ok(())
79    }
80    fn selector(&self, _: &()) -> CWire {
81        CWire::Selector
82    }
83    fn consumes(&self, ty: &syn::Type, _: &()) -> bool {
84        Value::of(ty).is_none()
85    }
86    fn input_leaf(
87        &self,
88        ty: &syn::Type,
89        _: &(),
90        w: syn::Ident,
91        v: syn::Ident,
92        active: &prebindgen_flat::Condition,
93    ) -> Result<InputPlan<CWire>> {
94        let read_buffer = prebindgen_flat::template_parameter("read_buffer");
95        let (decode, kind) = match Value::of(ty) {
96            Some(Value::Scalar(p)) => (decode_scalar(p, &w), CWire::Scalar(p)),
97            // Text and bytes borrow the caller's buffer; owned targets copy it.
98            Some(Value::Text) => (
99                quote!(std::str::from_utf8(unsafe {#read_buffer(#w)?}).map_err(|_|"invalid UTF-8")?),
100                CWire::Buffer,
101            ),
102            Some(Value::Bytes) => (quote!(unsafe {#read_buffer(#w)?}), CWire::Buffer),
103            None => {
104                return prebindgen_conversions::boxed::input(
105                    &prebindgen_conversions::boxed::Storage::Direct,
106                    ty,
107                    w,
108                    v,
109                    active,
110                    |b| CWire::Pointer(Box::new(b)),
111                )
112            }
113        };
114        prebindgen_conversions::value::input(ty, w, v, active, decode, kind)
115    }
116    fn output_leaf(
117        &self,
118        ty: &syn::Type,
119        _: &(),
120        w: syn::Ident,
121        v: TokenStream,
122        borrowed: bool,
123    ) -> Result<OutputPlan<CWire>> {
124        match Value::of(ty) {
125            Some(Value::Scalar(p)) => prebindgen_conversions::value::output(
126                ty,
127                w,
128                v,
129                borrowed,
130                |v| encode_scalar(p, v),
131                CWire::Scalar(p),
132            ),
133            Some(_) => {
134                // The buffer stays owned by Rust until every output is delivered.
135                let mut out = prebindgen_conversions::value::output(
136                    ty,
137                    w.clone(),
138                    v,
139                    borrowed,
140                    |v| quote!(::core::convert::Into::<Vec<u8>>::into(#v).into_boxed_slice()),
141                    CWire::Buffer,
142                )?;
143                let buffer_type = prebindgen_flat::template_parameter("buffer_type");
144                out.values = vec![quote!(#buffer_type {data: #w.as_ptr(), len: #w.len()})];
145                out.commit = quote!(let _ = Box::into_raw(#w););
146                Ok(out)
147            }
148            None => prebindgen_conversions::boxed::output(
149                &prebindgen_conversions::boxed::Storage::Direct,
150                ty,
151                w,
152                v,
153                borrowed,
154                |b| CWire::Pointer(Box::new(b)),
155            ),
156        }
157    }
158}
159fn decode_scalar(p: Primitive, w: &syn::Ident) -> TokenStream {
160    let ty = p.rust_type();
161    match p {
162        Primitive::Bool => quote!(#w != 0),
163        Primitive::Char => quote!(char::from_u32(#w).ok_or("invalid Unicode scalar")?),
164        Primitive::I128 | Primitive::U128 => {
165            quote!(((u128::from(#w.hi) << 64) | u128::from(#w.lo)) as #ty)
166        }
167        _ => quote!(#w),
168    }
169}
170fn encode_scalar(p: Primitive, v: TokenStream) -> TokenStream {
171    let wide_integer_type = prebindgen_flat::template_parameter("wide_integer_type");
172    match p {
173        Primitive::Bool => quote!(u8::from(#v)),
174        Primitive::Char => quote!((#v) as u32),
175        Primitive::I128 | Primitive::U128 => {
176            quote!({ let bits = (#v) as u128; #wide_integer_type { hi: (bits >> 64) as u64, lo: bits as u64 } })
177        }
178        _ => v,
179    }
180}
181/// Configure C bindings, using typed opaque pointers for foreign values.
182///
183/// Start with captured source via [`Self::builder`] and [`Self::source`], or a
184/// flat model via [`Self::new`]. With captured source, declare each
185/// exported function. In either case, declare each opaque pointee type.
186/// Conversion declarations use [`prebindgen_declarations::Declarations`]. When reading
187/// captured source, add the source crate as both a build dependency and a normal
188/// dependency: generation reads its captures, and generated code calls its functions.
189///
190/// ```
191/// use prebindgen_c::{Bindings, Cbindgen};
192/// use prebindgen_declarations::fun;
193/// let source = syn::parse_file("pub fn add(a: i64, b: i64) -> i64 { a + b }")?;
194/// let flat = Bindings::parse(
195///     source.items.into_iter().map(|item| (item, Default::default())),
196/// )?;
197/// let generated = Cbindgen::new(flat)
198///     .prefix("demo")
199///     .function(fun!(add))
200///     .build()?;
201/// assert!(generated.header.contains("demo_add("));
202/// syn::parse_file(&generated.rust)?;
203/// # Ok::<(), Box<dyn std::error::Error>>(())
204/// ```
205///
206/// Generated calls return a status and write results through an output pointer;
207/// `0` means success, `-1` means failure, with a thread-local last-error message.
208/// Owned pointer inputs transfer ownership after all active inputs validate;
209/// later source failures do not restore ownership. References borrow for the
210/// call and require the caller to obey Rust aliasing and lifetime rules.
211/// Returned pointers and buffers must be freed with their generated destructors.
212pub struct Cbindgen {
213    flat: Option<Bindings>,
214    source: Option<std::path::PathBuf>,
215    expansions: Vec<prebindgen_declarations::ExpandDecl>,
216    functions: Vec<prebindgen_declarations::FunctionDecl>,
217    prefix: String,
218    pointers: Vec<(String, syn::Type)>,
219}
220impl Cbindgen {
221    /// Create an unconfigured builder with the symbol prefix `pb`.
222    pub fn builder() -> Self {
223        Self {
224            flat: None,
225            source: None,
226            expansions: Vec::new(),
227            functions: Vec::new(),
228            prefix: "pb".into(),
229            pointers: Vec::new(),
230        }
231    }
232    /// Read captures from a source crate's `PREBINDGEN_OUT_DIR`, passed as `path`.
233    ///
234    /// Only declared functions, referenced constructors/getters and required types
235    /// are selected. Automatic type discovery follows unqualified source names
236    /// (see [`Flat::parse_selected`]). Types referenced only by qualified paths
237    /// can be selected through [`Self::opaque_ptr`] using their unqualified names.
238    /// Supply exactly one source directory or flat model.
239    pub fn source(mut self, path: impl AsRef<std::path::Path>) -> Self {
240        self.source = Some(path.as_ref().to_owned());
241        self
242    }
243    /// Register a type-wide input or output expansion, built with `expand_param!`
244    /// or `expand_return!`.
245    ///
246    /// Per-function overrides replace this default; duplicate defaults are rejected
247    /// by [`Self::build`]. Registration order does not affect type lookup.
248    pub fn expand(mut self, decl: impl Into<prebindgen_declarations::ExpandDecl>) -> Self {
249        self.expansions.push(decl.into());
250        self
251    }
252    /// Export a function and build its occurrence converters from declarations.
253    ///
254    /// `decl` names the function, normally with `fun!`, and carries its overrides.
255    ///
256    /// Function renaming is currently unsupported; exports use the symbol prefix
257    /// and original source name.
258    pub fn function(mut self, decl: prebindgen_declarations::FunctionDecl) -> Self {
259        self.functions.push(decl);
260        self
261    }
262
263    /// Start from a flat model with input and output conversion slots and prefix `pb`.
264    ///
265    /// `flat` is the parsed source, with any converters already attached.
266    ///
267    /// Attach converters with [`prebindgen_declarations::bind_input`] and
268    /// [`prebindgen_declarations::bind_output`] with [`CCodec`], or fill defaults with
269    /// [`Self::function`].
270    /// Every fully bound function in the supplied model is exported.
271    pub fn new(flat: Bindings) -> Self {
272        Self {
273            flat: Some(flat),
274            source: None,
275            expansions: Vec::new(),
276            functions: Vec::new(),
277            prefix: "pb".into(),
278            pointers: Vec::new(),
279        }
280    }
281    /// Set the prefix for exported functions, C types and destructors.
282    ///
283    /// Defaults to `pb`. The prefix is validated as an identifier during build.
284    pub fn prefix(mut self, prefix: &str) -> Self {
285        self.prefix = prefix.into();
286        self
287    }
288    /// Declare an opaque C pointee and its typed destructor for a source type.
289    ///
290    /// Every leaf type other than a primitive, text or bytes needs a matching
291    /// declaration. Owned values use `Box<T>` directly; they have no `Send`
292    /// requirement. This declaration alone does not export constructors.
293    ///
294    /// # Arguments
295    ///
296    /// * `name` — the C name, without the prefix: the header declares the type
297    ///   `{prefix}_{name}` and the destructor `{prefix}_drop_{name}`.
298    /// * `ty` — the source type behind the pointer, as written in the source crate.
299    pub fn opaque_ptr(mut self, name: &str, ty: syn::Type) -> Self {
300        self.pointers.push((name.into(), ty));
301        self
302    }
303    /// Resolve declarations, build converters and generate both output texts.
304    ///
305    /// Does not write files. Returns an error for conflicting sources, invalid or
306    /// duplicate declarations, unsupported conversions, or missing opaque pointees.
307    /// Rust layout/trait requirements are checked when the generated Rust compiles.
308    pub fn build(self) -> Result<Generation> {
309        prebindgen_declarations::ident(&self.prefix)?;
310        let mut flat = match (self.flat, self.source) {
311            (Some(flat), None) => flat,
312            (None, Some(path)) => Flat::parse_selected(
313                prebindgen::Source::new(path).items_all(),
314                &prebindgen_declarations::declarations::source_functions(
315                    &self.expansions,
316                    &self.functions,
317                ),
318                &self
319                    .pointers
320                    .iter()
321                    .map(|(_, ty)| ty.clone())
322                    .collect::<Vec<_>>(),
323            )?,
324            _ => return Err("supply exactly one source or flat model".into()),
325        };
326        let declarations = prebindgen_declarations::Declarations::new(&flat, self.expansions)?;
327        let mut names = std::collections::BTreeSet::new();
328        for decl in self.functions {
329            if !names.insert(decl.name.clone()) {
330                return Err("duplicate C function declaration".into());
331            }
332            if decl.foreign.is_some() {
333                return Err("C function renaming is outside this subset".into());
334            }
335            declarations.fill(&mut flat, &CCodec, &decl, &())?;
336        }
337        let pointers = self
338            .pointers
339            .into_iter()
340            .map(|(name, ty)| {
341                prebindgen_declarations::ident(&name)?;
342                Ok((name, flat.resolve_type(&ty)?))
343            })
344            .collect::<Result<Vec<_>>>()?;
345        prebindgen_flat::validate_bindings(&flat)?;
346        generate(flat, &self.prefix, pointers)
347    }
348}

370    }

372fn rust_ty(wire: &Wire) -> TokenStream {
373    match &wire.kind {
374        CWire::Scalar(p) => match p {
375            Primitive::Bool => quote!(u8),
376            Primitive::Char => quote!(u32),
377            Primitive::I128 | Primitive::U128 => quote!(__Pb128),
378            p => {
379                let ty = p.rust_type();
380                quote!(#ty)
381            }
382        },
383        CWire::Pointer(p) => {
384            let ty = &p.ty;
385            if p.access == prebindgen_flat::Access::Shared {
386                quote!(*const #ty)
387            } else {
388                quote!(*mut #ty)
389            }
390        }
391        CWire::Selector => quote!(i32),
392        CWire::Buffer => quote!(__PbBuffer),
393    }
394}
395fn c_ty(wire: &Wire, prefix: &str, pointers: &[(String, syn::Type)]) -> String {
396    match &wire.kind {
397        CWire::Scalar(p) => match p {
398            Primitive::Bool | Primitive::U8 => "uint8_t".into(),
399            Primitive::Char | Primitive::U32 => "uint32_t".into(),
400            Primitive::I8 => "int8_t".into(),
401            Primitive::I16 => "int16_t".into(),
402            Primitive::I32 => "int32_t".into(),
403            Primitive::I64 => "int64_t".into(),
404            Primitive::U16 => "uint16_t".into(),
405            Primitive::U64 => "uint64_t".into(),
406            Primitive::Isize => "intptr_t".into(),
407            Primitive::Usize => "uintptr_t".into(),
408            Primitive::F32 => "float".into(),
409            Primitive::F64 => "double".into(),
410            Primitive::I128 | Primitive::U128 => format!("{prefix}_int128"),
411        },
412        CWire::Pointer(p) => {
413            let (name, _) = pointers
414                .iter()
415                .find(|(_, ty)| prebindgen_flat::same_type(ty, &p.ty))
416                .expect("validated pointer declaration");
417            let qualifier = if p.access == prebindgen_flat::Access::Shared {
418                "const "
419            } else {
420                ""
421            };
422            format!("{qualifier}{prefix}_{name} *")
423        }
424        CWire::Selector => "int32_t".into(),
425        CWire::Buffer => format!("{prefix}_buffer"),
426    }
427}
428fn generate<
429    I: prebindgen_flat::InputConversion<Wire = CWire, Context = CContext>,
430    O: prebindgen_flat::OutputConversion<Wire = CWire, Context = CContext>,
431>(
432    flat: Flat<I, O>,
433    prefix: &str,
434    pointers: Vec<(String, syn::Type)>,
435) -> Result<Generation> {
436    let mut calls = Vec::new();
437    let error = format_ident!("{prefix}_last_error");
438    let free = format_ident!("{prefix}_free_buffer");
439    let mut symbols = std::collections::BTreeSet::from([error.to_string(), free.to_string()]);
440    for (name, _) in &pointers {
441        if !symbols.insert(format!("{prefix}_drop_{name}")) {
442            return Err("duplicate C destructor".into());
443        }
444    }
445    for f in flat.functions.values() {
446        if f.ret.slot.is_none() {
447            continue;
448        }
449        let result_type = format_ident!("{prefix}_{}_result", f.name);
450        let context = CContext {
451            source_result: format_ident!("source_value"),
452            output_pointer: format_ident!("output"),
453            output_type: syn::parse_quote!(#result_type),
454            buffer_type: syn::parse_quote!(__PbBuffer),
455            wide_integer_type: syn::parse_quote!(__Pb128),
456            read_buffer: syn::parse_quote!(__pb_read),
457        };
458        let body = prebindgen_flat::emit_call(f, &context)?;
459        if !symbols.insert(format!("{prefix}_{}", f.name)) {
460            return Err("duplicate C export".into());
461        }
462        for wire in f
463            .inputs()?
464            .chain(f.outputs(&context)?.map(|output| output.wire))
465        {
466            if let CWire::Pointer(pointer) = &wire.kind {
467                if !pointers
468                    .iter()
469                    .any(|(_, declared)| prebindgen_flat::same_type(declared, &pointer.ty))
470                {
471                    return Err(format!(
472                        "declare a C opaque pointer for `{}`",
473                        prebindgen_flat::type_key(&pointer.ty)
474                    ));
475                }
476            }
477        }
478        calls.push((f, body, context));
479    }
480    let mut tokens = quote! {
481       #[repr(C)] #[derive(Clone,Copy)] pub struct __Pb128 {pub hi:u64,pub lo:u64}
482       #[repr(C)] #[derive(Clone,Copy)] pub struct __PbBuffer {pub data:*const u8,pub len:usize}
483       std::thread_local! {static __PB_ERROR:std::cell::RefCell<Option<std::ffi::CString>>=const {std::cell::RefCell::new(None)};}
484       fn __pb_error(e:&str){__PB_ERROR.with(|s|*s.borrow_mut()=if e.is_empty(){None}else{Some(std::ffi::CString::new(e.replace('\0',"?")).unwrap())});}
485       #[no_mangle] pub extern "C" fn #error()->*const std::ffi::c_char {__PB_ERROR.with(|s|s.borrow().as_ref().map_or(c"".as_ptr(),|e| e.as_ptr()))}
486       // Input pointers are borrowed only during the call; output buffers are owned.
487       unsafe fn __pb_read<'a>(b:__PbBuffer)->Result<&'a [u8],String>{
488        if b.len==0{return Ok(&[]);}
489        if b.data.is_null()||b.len>isize::MAX as usize{return Err("invalid buffer".into());}
490        Ok(std::slice::from_raw_parts(b.data,b.len))
491       }
492       #[doc = "# Safety
493    The buffer must have been returned by this library and not already freed."]
494    #[no_mangle] pub unsafe extern "C" fn #free(b:__PbBuffer){if !b.data.is_null(){drop(Box::from_raw(std::ptr::slice_from_raw_parts_mut(b.data as *mut u8,b.len)));}}
495      };
496    let mut header=format!("#ifndef {prefix}_H\n#define {prefix}_H\n#include <stdint.h>\n#include <stddef.h>\n#ifdef __cplusplus\nextern \"C\" {{\n#endif\ntypedef struct {{ uint64_t hi; uint64_t lo; }} {prefix}_int128;\n typedef struct {{ const uint8_t *data; size_t len; }} {prefix}_buffer;\nconst char *{error}(void);\nvoid {free}({prefix}_buffer b);\n");
497    for (name, ty) in &pointers {
498        let drop = format_ident!("{prefix}_drop_{name}");
499        tokens.extend(quote!{
500            #[doc = "# Safety
501The pointer must be null or a live allocation of the declared type returned by this library. It must not be used again after this call."]
502            #[no_mangle] pub unsafe extern "C" fn #drop(ptr:*mut #ty)->i32{
503                match ::prebindgen_runtime::boundary(||{if !ptr.is_null(){drop(Box::from_raw(ptr));}Ok(())}){Ok(())=>{__pb_error("");0},Err(e)=>{__pb_error(&e);-1}}
504            }
505        });
506        header.push_str(&format!("typedef struct {prefix}_{name} {prefix}_{name};\nint32_t {drop}({prefix}_{name} *ptr);\n"));
507    }
508    for (f, body, context) in calls {
509        let out = &context.output_pointer;
510        let name = format_ident!("{prefix}_{}", f.name);
511        let result = format_ident!("{name}_result");
512        let mut params = Vec::new();
513        let mut cparams = Vec::new();
514        for w in f.inputs()? {
515            let n = &w.name;
516            let ty = rust_ty(w);
517            params.push(quote!(#n:#ty));
518            cparams.push(format!("{} {n}", c_ty(w, prefix, &pointers)));
519        }
520        let mut fields = Vec::new();
521        header.push_str("typedef struct {\n");
522        for (i, output) in f.outputs(&context)?.enumerate() {
523            let w = output.wire;
524            let field = format_ident!("v{i}");
525            let ty = rust_ty(w);
526            fields.push(quote!(pub #field:#ty));
527            header.push_str(&format!("    {} v{i};\n", c_ty(w, prefix, &pointers)));
528        }
529        if fields.is_empty() {
530            fields.push(quote!(pub unused:u8));
531            header.push_str("    uint8_t unused;\n");
532        }
533        header.push_str(&format!("}} {result};\n"));
534        cparams.push(format!("{result} *{out}"));
535        header.push_str(&format!("int32_t {name}({});\n", cparams.join(", ")));
536        tokens.extend(quote! {
537         #[repr(C)] pub struct #result {#(#fields),*}
538         #[doc = "# Safety
539The output pointer must be writable. Active input buffers must be readable for their stated lengths. Opaque pointers must be live allocations of the declared type from this library. Consuming inputs become invalid at ownership transfer; borrows remain valid and must obey Rust aliasing rules for the duration of the call."]
540   #[allow(clippy::needless_borrow, clippy::let_unit_value, clippy::unnecessary_cast, clippy::useless_conversion)]
541   #[no_mangle] pub unsafe extern "C" fn #name(#(#params,)* #out:*mut #result)->i32{
542          if #out.is_null(){__pb_error("null result pointer");return -1;}
543          match ::prebindgen_runtime::boundary(||{
544           #body
545          }){Ok(())=>{__pb_error("");0},Err(e)=>{__pb_error(&e);-1}}
546         }
547        });
548    }
549    let guards = &flat.guards;
550    tokens.extend(quote!(#(#guards)*));
551    header.push_str("#ifdef __cplusplus\n}\n#endif\n#endif\n");
552    Ok(Generation {
553        rust: prebindgen_flat::rust_source(tokens)?,
554        header,
555    })
556}
