1//! Leaf conversions for values the adapter copies across the boundary.
2//! [`Value::of`] classifies a source type;
3//! [`input`] and [`output`] build
4//! the conversion around the adapter's own decode and encode expressions.
5use prebindgen_flat::{type_key, Primitive, Result};
6
7use prebindgen_flat::{Wire};
8use crate::{InputPlan, OutputPlan};
9use proc_macro2::TokenStream;
10use quote::quote;
11use syn::Type;

13/// A source type that has a by-value carrier: owned, or behind a shared reference.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub enum Value {
16    /// A Rust primitive.
17    Scalar(Primitive),
18    /// `String`, `&String` or `&str`.
19    Text,
20    /// `Vec<u8>`, `&Vec<u8>` or `&[u8]`.
21    Bytes,
22}
23impl Value {
24    /// Classify `ty`; `None` means it is not a value type, including `&mut` views.
25    ///
26    /// ```
27    /// use prebindgen_conversions::value::Value;
28    /// assert_eq!(Value::of(&syn::parse_quote!(&str)), Some(Value::Text));
29    /// assert_eq!(Value::of(&syn::parse_quote!(&mut String)), None);
30    /// ```
31    pub fn of(ty: &Type) -> Option<Self> {
32        let (core, borrowed) = match ty {
33            Type::Reference(r) if r.mutability.is_none() => (&*r.elem, true),
34            Type::Reference(_) => return None,
35            _ => (ty, false),
36        };
37        if let Some(p) = Primitive::from_type(core) {
38            return Some(Self::Scalar(p));
39        }
40        match type_key(core).as_str() {
41            "String" => Some(Self::Text),
42            "str" if borrowed => Some(Self::Text),
43            "Vec < u8 >" => Some(Self::Bytes),
44            "[u8]" if borrowed => Some(Self::Bytes),
45            _ => None,
46        }
47    }
48}

50/// Decode one wire into the value type `ty` during validation.
51///
52/// The generated code converts the decoded value with `Into` for owned targets and
53/// `AsRef` for views, so a view target borrows without copying.
54///
55/// # Arguments
56///
57/// * `ty` — the source parameter type; it must satisfy [`Value::of`].
58/// * `wire` — identifier of the boundary parameter `decode` reads.
59/// * `var` — identifier of the Rust local that holds the argument.
60/// * `active` — an emitted Rust boolean expression. `decode` runs
61///   only when they are selected.
62/// * `decode` — Rust expression that reads `wire` and yields the owned value or,
63///   for text and bytes, a view (`&str`, `&[u8]`). It runs during validation and
64///   may return a `String` error with `?`.
65/// * `kind` — the adapter's [`Wire::kind`] for the wire.
66///
67/// ```
68/// use prebindgen_conversions::value;
69/// use quote::quote;
70/// let input = value::input::<syn::Type, ()>(
71///     &syn::parse_quote!(&str), syn::parse_quote!(raw), syn::parse_quote!(text),
72///     &quote::quote!(true), quote::quote!(read(raw)?), syn::parse_quote!(&str),
73/// )?;
74/// assert_eq!(prebindgen_flat::type_key(&input.wires[0].rust_type()), "& str");
75/// assert!(input.acquire.is_empty() && input.resources.is_empty());
76/// # Ok::<(), Box<dyn std::error::Error>>(())
77/// ```
78pub fn input<W: prebindgen_flat::WireKind, C>(
79    ty: &Type,
80    wire: syn::Ident,
81    var: syn::Ident,
82    active: &TokenStream,
83    decode: TokenStream,
84    kind: W,
85) -> Result<InputPlan<W, C>> {
86    if Value::of(ty).is_none() {
87        return Err(format!("`{}` is not a value type", type_key(ty)));
88    }
89    let condition = active.clone();
90    let taken = quote!(#var.expect("active value"));
91    let (statements, argument) = match ty {
92        Type::Reference(r) if matches!(type_key(&r.elem).as_str(), "str" | "[u8]") => {
93            let target = &r.elem;
94            (
95                quote!(let #var = #taken;),
96                quote!(::core::convert::AsRef::<#target>::as_ref(&#var)),
97            )
98        }
99        Type::Reference(r) => {
100            let owned = &r.elem;
101            (
102                quote!(let #var: #owned = ::core::convert::Into::into(#taken);),
103                quote!(&#var),
104            )
105        }
106        _ => (
107            quote!(let #var: #ty = ::core::convert::Into::into(#taken);),
108            quote!(#var),
109        ),
110    };
111    Ok(InputPlan {
112        context: Default::default(),
113        acknowledgement: None,
114        wires: vec![Wire { name: wire, kind }],
115        validation: quote!(let #var = if #condition {Some(#decode)} else {None};),
116        acquire: quote!(),
117        resources: vec![],
118        statements,
119        argument,
120    })
121}

123/// Deliver an owned value type. A borrowed `value` is copied first: scalars by
124/// dereference, text and bytes by `Clone`.
125///
126/// The adapter may adjust the result, for example to defer an allocation's
127/// transfer to `commit`.
128///
129/// # Arguments
130///
131/// * `ty` — the owned source type; it must satisfy [`Value::of`].
132/// * `wire` — identifier of the delivered wire, also used for the local holding it.
133/// * `value` — Rust expression producing the value.
134/// * `borrowed` — `true` when `value` is a shared reference to `ty`, `false` when
135///   it is owned.
136/// * `encode` — maps a Rust expression of the owned value to the expression
137///   delivered on the wire.
138/// * `kind` — the adapter's [`Wire::kind`] for the wire.
139pub fn output<W: prebindgen_flat::WireKind, C>(
140    ty: &Type,
141    wire: syn::Ident,
142    value: TokenStream,
143    borrowed: bool,
144    encode: impl FnOnce(TokenStream) -> TokenStream,
145    kind: W,
146) -> Result<OutputPlan<W, C>> {
147    let Some(class) = Value::of(ty).filter(|_| !matches!(ty, Type::Reference(_))) else {
148        return Err(format!("`{}` is not an owned value type", type_key(ty)));
149    };
150    let mut statements = TokenStream::new();
151    let value = match (borrowed, class) {
152        (false, _) => value,
153        // Bound first: dereferencing `&place` in place is `clippy::deref_addrof`.
154        (true, Value::Scalar(_)) => {
155            statements.extend(quote!(let #wire: &#ty = #value;));
156            quote!(*#wire)
157        }
158        (true, _) => quote!(::core::clone::Clone::clone(#value)),
159    };
160    let expr = encode(value);
161    statements.extend(quote!(let #wire = #expr;));
162    Ok(OutputPlan {
163        context: Default::default(),
164        delivery: Default::default(),
165        wires: vec![Wire {
166            name: wire.clone(),
167            kind,
168        }],
169        statements,
170        values: vec![quote!(#wire)],
171        commit: quote!(),
172    })
173}
