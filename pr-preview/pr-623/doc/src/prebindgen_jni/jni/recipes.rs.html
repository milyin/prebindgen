1//! JniGen's Kotlin class declarations, lowered into the registry's recipe
2//! vocabulary.
3//!
4//! A build script writes `ptr_class!`, `data_class!`, `enum_class!` or
5//! `sealed_class!`, each naming the Kotlin type one Rust type becomes, and
6//! `convert!` for a conversion the binding supplies itself. This turns those
7//! into recipes in a [`Recipes`] table: the shared statement of **which parts** a
8//! value gets across in, with nothing about the JNI wire in it.
9//!
10//! One recipe per declared type and direction, and every crossing nobody declared
11//! takes
12//! the recipe the registry derives from its kind. The chain of guesses this
13//! replaced — try a terminal, then a `Result`, then an optional, then a run,
14//! then a borrow, then a transparent bridge — is a lookup for the declared
15//! half, and the derived half is the registry's.
16

29/// Every type one captured item names, so a reading nested inside it can be
30/// found without the caller knowing which kind of item it came from.
31fn element_types(element: &prebindgen_registry::Element) -> Vec<&TypeRef> {
32    use prebindgen_registry::flat::Type;
33    match element {
34        prebindgen_registry::Element::Function(f) => f
35            .params
36            .iter()
37            .map(|p| &p.ty)
38            .chain(std::iter::once(&f.ret))
39            .collect(),
40        prebindgen_registry::Element::Type(Type::Struct(s)) => {
41            s.fields.iter().map(|f| &f.ty).collect()
42        }
43        prebindgen_registry::Element::Type(Type::Variant(v)) => v
44            .alternatives
45            .iter()
46            .flat_map(|a| a.fields.iter().map(|f| &f.ty))
47            .collect(),
48        prebindgen_registry::Element::Constant(c) => vec![&c.ty],
49        _ => Vec::new(),
50    }
51}

53/// One arm per alternative of a declared sum, each assembled from its own
54/// payload fields.
55///
56/// Empty for anything the model does not hold as a data-carrying enum, which
57/// includes a `sealed_class!` declared over a type the scan dropped — the scan
58/// reports that, and a recipe over no arms would report it a second time.
59fn alternatives(model: &Flat, ty: &TypeRef) -> Vec<Arm<Construct>> {
60    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
61        return Vec::new();
62    };
63    let Some(prebindgen_registry::flat::Type::Variant(v)) =
64        id.ident().and_then(|ident| model.declared_type(&ident))
65    else {
66        return Vec::new();
67    };
68    v.alternatives
69        .iter()
70        .map(|alt| Arm {
71            alternative: alt.index,
72            op: Construct::Fields,
73        })
74        .collect()
75}
76
77/// One arm per alternative of a declared sum, each taken apart into its own
78/// payload fields.
79///
80/// The deconstructing twin of [`alternatives`]. Separate because the two directions
81/// name different operations — building an alternative is `Construct::Fields`,
82/// reading one is `Deconstruct::Fields` over the reaches that get there — and
83/// a shared helper would have to be generic over an operation neither side
84/// chooses.
85fn out_alternatives(model: &Flat, ty: &TypeRef) -> Vec<Arm<Deconstruct>> {
86    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
87        return Vec::new();
88    };
89    let Some(prebindgen_registry::flat::Type::Variant(v)) =
90        id.ident().and_then(|ident| model.declared_type(&ident))
91    else {
92        return Vec::new();
93    };
94    v.alternatives
95        .iter()
96        .map(|alt| Arm {
97            alternative: alt.index,
98            op: Deconstruct::Fields((0..alt.fields.len()).map(Reach::Field).collect()),
99        })
100        .collect()
101}

103impl Declarations {
104    /// How many values this type hands out, or 0 if it hands out none.
105    ///
106    /// Model and declaration only, like the compositions it asks — so a
107    /// declaration-time caller may ask it, which is what binding a return site
108    /// needs.
109    fn out_wire_count(
110        &self,
111        registry: &impl prebindgen_registry::Conversions,
112        ty: &TypeRef,
113    ) -> usize {
114        let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
115            return 0;
116        };
117        let Some(ident) = id.ident() else { return 0 };
118        match registry.flat().declared_type(&ident) {
119            Some(prebindgen_registry::flat::Type::Variant(_)) => self
120                .sum_out_wires(registry, &ident, ty)
121                .map_or(0, |w| w.len()),
122            Some(prebindgen_registry::flat::Type::Struct(_)) => {
123                self.struct_out_wires(registry, ty).map_or(0, |w| w.len())
124            }
125            _ => 0,
126        }
127    }
128
129    /// The accessor a type's value form names, and one reach per field of the
130    /// struct it returns.
131    ///
132    /// `None` unless the declaration is exactly one `.fields(fields!(f))` whose
133    /// every field is a plain leaf. A record that states its own decomposition
134    /// — a per-field `expand_return!` override, an inlined nested class, a sum
135    /// laid out as a selector and its groups — is a shape this recipe does not
136    /// state yet, and stating half of one would describe a boundary nothing
137    /// emits.
138    fn value_form_of(
139        &self,
140        model: &Flat,
141        registry: &impl prebindgen_registry::Conversions,
142        ty: &TypeRef,
143    ) -> Option<(syn::Ident, Vec<Reach>)> {
144        use prebindgen_registry::unfold::{FieldDecon, FieldRecord};
145        let decl = self
146            .return_expand_decls
147            .iter()
148            .find(|d| *d.key() == ty.stripped_key())?;
149        let [crate::jni::LocalField::Fields(fields)] = decl.field_list() else {
150            return None;
151        };
152        let records: Vec<FieldRecord> = self.lower_value_form(registry, decl.key(), fields);
153        let ret = model.function(fields.func())?.ret.clone();
154        let ret = ret.borrow_target().unwrap_or(&ret);
155        let prebindgen_registry::flat::TypeKind::Named { id, .. } = ret.unwrapped().kind() else {
156            return None;
157        };
158        let prebindgen_registry::flat::Type::Struct(st) =
159            id.ident().and_then(|i| model.declared_type(&i))?
160        else {
161            return None;
162        };
163        let mut reaches = Vec::new();
164        for record in &records {
165            if !matches!(record.decon, FieldDecon::Default) {
166                return None;
167            }
168            let [member] = record.members.as_slice() else {
169                return None;
170            };
171            let index = st
172                .fields
173                .iter()
174                .position(|f| f.name.as_ref() == Some(member))?;
175            // A field that reaches the type being taken apart. The leaf
176            // synthesis treats `Box<T>` as a spelling of its own and stops
177            // there; a recipe keys a crossing by the value that crosses, so
178            // `Box<ZSample>` inside `ZSample`'s own value form is that recipe
179            // reaching itself. Refused here rather than by `Recipes::build`,
180            // which would refuse the whole binding over a shape the leaf
181            // synthesis handles.
182            if st.fields[index].ty.stripped_key() == ty.stripped_key() {
183                return None;
184            }
185            reaches.push(Reach::Field(index));
186        }
187        Some((fields.func().clone(), reaches))
188    }
189}

191impl Declarations {
192    /// What a value form calls each of its fields, by the Rust field ident.
193    ///
194    /// The declaration's answer, so a `.name(..)` rename carries through to the
195    /// builder parameter. `None` for a type with no value form, or one whose
196    /// records this recipe does not state — the same test [`Self::value_form_of`]
197    /// makes when it declares the recipe.
198    pub(crate) fn value_form_names(
199        &self,
200        registry: &impl prebindgen_registry::Conversions,
201        ty: &TypeRef,
202    ) -> Option<std::collections::HashMap<String, String>> {
203        use prebindgen_registry::unfold::FieldDecon;
204        let decl = self
205            .return_expand_decls
206            .iter()
207            .find(|d| *d.key() == ty.stripped_key())?;
208        let [crate::jni::LocalField::Fields(fields)] = decl.field_list() else {
209            return None;
210        };
211        self.lower_value_form(registry, decl.key(), fields)
212            .into_iter()
213            .map(|r| match (r.members.as_slice(), &r.decon) {
214                ([member], FieldDecon::Default) => Some((member.to_string(), r.name)),
215                _ => None,
216            })
217            .collect()
218    }
219}

221/// One reach per field of a declared struct, in the model's order.
222///
223/// Empty for anything the model does not hold as a struct, which includes a
224/// `data_class!` over a type the scan dropped — the scan reports that, and a
225/// recipe over no fields would report it a second time.
226fn out_fields(model: &Flat, ty: &TypeRef) -> Vec<Reach> {
227    (0..fields_of(model, ty).len()).map(Reach::Field).collect()
228}

230/// The fields of a declared struct, or none for anything the model does not
231/// hold as one.
232fn fields_of<'a>(model: &'a Flat, ty: &TypeRef) -> &'a [prebindgen_registry::flat::Field] {
233    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
234        return &[];
235    };
236    match id.ident().and_then(|ident| model.declared_type(&ident)) {
237        Some(prebindgen_registry::flat::Type::Struct(s)) => &s.fields,
238        _ => &[],
239    }
240}

263impl Declarations {
264    /// Every recipe this binding's declarations state.
265    ///
266    /// A type declared but absent from the model is skipped rather than
267    /// refused: the scan already reports it, and reporting it twice in
268    /// different words helps nobody.
269    pub(crate) fn recipes(
270        &self,
271        model: &Flat,
272        expansion_leaves: &[TypeRef],
273        registry: &impl prebindgen_registry::Conversions,
274    ) -> Result<Recipes, Vec<RecipeError>> {
275        let mut recipes = Recipes::builder();
276        // Which crossings already state a deconstructing `parts` row, so the
277        // decomposition block below adds one only where none was declared.
278        let mut parts_out: std::collections::HashSet<TypeKey> = std::collections::HashSet::new();
279        // Every Kotlin class declaration, and every `convert!`-declared
280        // conversion. The second matters as much as the first: a conversion may
281        // be declared on a type the registry would otherwise read as an arity
282        // layer, and `convert!(Option<Duration> => ..)` means the adapter emits
283        // that optional's conversion itself rather than wrapping `Duration`'s.
284        let mut declared: Vec<(TypeKey, Origin<syn::Type>)> = self
285            .declared_types()
286            .into_iter()
287            .chain(
288                self.convert_decls
289                    .iter()
290                    .map(|d| (d.key().clone(), d.rust_type().clone())),
291            )
292            .collect();
293        declared.sort_by(|a, b| a.0.as_str().cmp(b.0.as_str()));
294        declared.dedup_by(|a, b| a.0 == b.0);
295
296        for (_key, origin) in declared {
297            // The declarator's own tokens, re-parsed: a declared type may be an
298            // alias the model holds as an `Extern`, which carries no reading of
299            // its own to borrow.
300            let Ok(spelled) = syn::parse2::<syn::Type>(origin.declared_spelling()) else {
301                continue;
302            };
303            let Ok(ty) = model.classify(&spelled) else {
304                continue;
305            };
306            // Every declared Kotlin shape is one recipe with no parts today,
307            // including the two that plainly have some: a `data_class` arrives
308            // as separate JNI parameters and is rebuilt inside one generated
309            // function, and a `sealed_class` is a selector plus the live arm's
310            // slots inside another. `Atomic` is what a recipe says about that —
311            // the adapter emits the conversion itself, and how many wire values
312            // that costs is its own business — so it describes the adapter as it
313            // stands rather than as it will be.
314            // A `sealed_class` hands its value out as a tag plus every
315            // alternative's payloads, laid side by side — the deconstructing
316            // twin of the constructing recipe below, and the direction that
317            // actually has no alternative: a sum has no whole-value output
318            // conversion, because a tag plus groups is not one wire.
319            match self.types.get(&ty.key()).map(|c| &c.kind) {
320                Some(DeclaredKind::Sealed(_)) => {
321                    let arms = out_alternatives(model, &ty);
322                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
323                    if !arms.is_empty() {
324                        recipes.declare(ty.clone(), parts(), Deconstructing::Choice { arms });
325                        parts_out.insert(ty.stripped_key());
326                    }
327                }
328                // A `data_class` hands its value out as its fields too, so the
329                // foreign side reassembles the object and nothing builds one on
330                // the Rust side.
331                Some(DeclaredKind::Data) => {
332                    let reaches = out_fields(model, &ty);
333                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
334                    if !reaches.is_empty() {
335                        recipes.declare(
336                            ty.clone(),
337                            parts(),
338                            Deconstructing::Product(Deconstruct::Fields(reaches)),
339                        );
340                        parts_out.insert(ty.stripped_key());
341                    }
342                }
343                _ => {
344                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
345                }
346            }
347            // A value form: `expand_return!(T).fields(fields!(f))` calls `f`
348            // once and hands out the fields of the struct it returns. The one
349            // declaration shape that names its own accessor, and the reason
350            // `Deconstruct::ValueForm` exists.
351            if let Some((func, reaches)) = self.value_form_of(model, registry, &ty) {
352                recipes.declare(
353                    ty.clone(),
354                    parts(),
355                    Deconstructing::Product(Deconstruct::ValueForm {
356                        func,
357                        parts: reaches,
358                    }),
359                );
360                parts_out.insert(ty.stripped_key());
361            }
362            // A `data_class` also has a recipe that says what it is made of, so
363            // its constructing side names which of the two a site takes by
364            // default. See [`parts`] for why that is still `whole`.
365            match self.types.get(&ty.key()).map(|c| &c.kind) {
366                // A struct with no fields is nothing to be made of, so it
367                // states no `parts` recipe — the same condition the deconstructing
368                // side applies, and the reason `recipe_of` may be asked for the
369                // recipe by name.
370                Some(DeclaredKind::Data) if !fields_of(model, &ty).is_empty() => {
371                    recipes
372                        .declare_default(ty.clone(), whole(), Constructing::Atomic)
373                        .declare(ty, parts(), Constructing::Product(Construct::Fields));
374                }
375                // A `sealed_class` has one too, and it is a choice rather than
376                // a product: exactly one alternative is live, every one of them
377                // still crosses, and the tag says which. The arms are the
378                // model's — a recipe states which parts, never what they are.
379                Some(DeclaredKind::Sealed(_)) => {
380                    let arms = alternatives(model, &ty);
381                    recipes.declare_default(ty.clone(), whole(), Constructing::Atomic);
382                    if !arms.is_empty() {
383                        recipes.declare(ty, parts(), Constructing::Choice { arms });
384                    }
385                }
386                _ => {
387                    recipes.declare(ty, whole(), Constructing::Atomic);
388                }
389            }
390        }
391
392        // A type a callback delivers by taking it apart states that as a row,
393        // so the argument's site has a recipe to name. Without one the site
394        // could only be fabricated — a `Bound` no binding answered — which is
395        // what #622's first two attempts produced and why they were withdrawn.
396        //
397        // `Atomic` for the reason the declared-type loop above gives for the
398        // two shapes that plainly have parts: the adapter emits this
399        // conversion itself, and how many wire values that costs is its own
400        // business. The parts are stated by the deconstructor declaration the
401        // registry already resolved into an `UnfoldPlan`, which this row does
402        // not restate — it names the crossing that plan decomposes. Deleting
403        // that second statement is #613 step 5b's, not this row's.
404        //
405        // The source is the plan's own owned core, so a plan keyed under `&T`
406        // and one keyed under `T` state one row; a type that already declared
407        // `parts` — a `data_class`, a `sealed_class`, a value form — keeps it.
408        let mut decomposed: BTreeMap<TypeKey, TypeRef> = BTreeMap::new();
409        for plan in registry.callback_arg_plans().values() {
410            let key = plan.source.stripped_key();
411            if parts_out.contains(&key) {
412                continue;
413            }
414            decomposed.entry(key).or_insert_with(|| plan.source.clone());
415        }
416        for ty in decomposed.into_values() {
417            recipes.declare(ty, parts(), Deconstructing::Atomic);
418        }
419
420        // Every implicit Optional keeps its established whole/default input
421        // row and also offers the allocation-free primitive `pair` row. A
422        // parameter site selects `pair` only after the compiled payload proves
423        // that it is one JNI primitive with no niche, projection, or stage.
424        //
425        // Product-shaped payloads additionally offer `parts`; that row is
426        // selected by the existing flattening bindings below.
427        let mut implicit: BTreeMap<TypeKey, (TypeRef, TypeRef)> = self
428            .implicit_optionals(model)
429            .into_iter()
430            .map(|(key, (outer, inner))| (key, (outer.clone(), inner.clone())))
431            .collect();
432        // A combined constructor expansion represents an inactive arm by
433        // Option-wrapping each required leaf. Those readings are synthesized
434        // after the Flat model was built, so a model walk cannot discover
435        // them. File them in the same table as source-written implicit
436        // Optionals: otherwise a nullable primitive leaf falls back to a boxed
437        // JObject and Rust has to call `intValue()`/`longValue()` across JNI.
438        for leaf in expansion_leaves {
439            let Some(inner) = leaf.optional_inner() else {
440                continue;
441            };
442            if !self.explicitly_declares(leaf) {
443                implicit
444                    .entry(leaf.stripped_key())
445                    .or_insert_with(|| (leaf.clone(), inner.clone()));
446            }
447        }
448        for (outer, _) in implicit.values() {
449            recipes
450                .declare_default(outer.clone(), whole(), Constructing::Optional)
451                .declare(outer.clone(), pair(), Constructing::Optional)
452                .declare(outer.clone(), parts(), Constructing::Optional);
453        }
454        for (outer, _) in implicit.into_values() {
455            recipes
456                .declare_default(outer.clone(), whole(), Deconstructing::Optional)
457                .declare(outer.clone(), parts(), Deconstructing::Optional);
458        }
459
460        // A fixed-size array of JNI primitives is one Kotlin `ByteArray` or
461        // `LongArray`, bulk-copied with nothing boxed — one wire value, not a
462        // run this adapter walks. The registry reads `[T; N]` as a run unless
463        // something says otherwise, and this is JniGen saying otherwise for
464        // every array the model holds. Nothing is enumerated twice: a recipe for
465        // a crossing nobody uses is inert.
466        // Keyed by identity in a `BTreeMap`, so each array type is keyed once
467        // and the declaration order is the key order — no sort, no second pass
468        // rebuilding a key it already has, and nothing cloned before the entry
469        // turns out to be new.
470        let mut arrays: BTreeMap<TypeKey, &TypeRef> = BTreeMap::new();
471        for ty in model
472            .elements()
473            .flat_map(element_types)
474            .flat_map(|ty| ty.walk())
475            .filter(|ty| matches!(ty.kind(), prebindgen_registry::flat::TypeKind::Array { .. }))
476        {
477            arrays.entry(ty.key()).or_insert(ty);
478        }
479        for ty in arrays.into_values() {
480            recipes
481                .declare(ty.clone(), whole(), Deconstructing::Atomic)
482                .declare(ty.clone(), whole(), Constructing::Atomic);
483        }
484
485        recipes.build(model)
486    }
487}

489impl Declarations {
490    /// Which recipe each part of a `data_class` takes.
491    ///
492    /// A field that is itself a `data_class` crosses as **its** fields too, so
493    /// the part site asks for [`parts`] rather than letting the field take its
494    /// own default. That is the one thing a site can say that a recipe cannot: the
495    /// same crossing is read one way on its own and another inside a product.
496    ///
497    /// Without it a nested class contributes a single wire and the flattening
498    /// stops one layer down.
499    /// Whether a `data_class` field is itself flattened, or stays one value.
500    ///
501    /// A `data_class` declared `.jobject_input()` crosses Kotlin → Rust as a
502    /// single `JObject` — that is the whole point of the opt-in — so binding
503    /// its parts would flatten a boundary the declaration drew.
504    fn field_crosses_as_its_fields(&self, ty: &TypeRef) -> bool {
505        // The field's own key, not the stripped one: a `Box<Leaf>` field
506        // crosses as one value through `Box<Leaf>`'s conversion, and only a
507        // parameter spelled that way is peeled back to the class. Binding the
508        // stripped key here would flatten a field the rebuild has no node for.
509        self.types.get(&ty.key()).is_some_and(|c| match c.kind {
510            DeclaredKind::Data => !c.jobject_input,
511            // A `sealed_class` field crosses as a tag plus every alternative's
512            // slots. Whether it *can* is the adapter's answer at compile time,
513            // not a declaration: a payload with no slot form leaves the sum
514            // object-shaped, which is the fragment `Compile::choice` hands
515            // back. So the site asks, and the recipe answers.
516            DeclaredKind::Sealed(_) => true,
517            _ => false,
518        })
519    }
520
521    /// Optional crossings whose payload can use the shared `parts` recipe.
522    ///
523    /// Explicit declarations keep full control of the outer crossing: a
524    /// `convert!(Option<T> => ..)` is an atomic recipe, not an implicit request
525    /// to flatten `T`.
526    fn implicit_optionals<'a>(
527        &self,
528        model: &'a Flat,
529    ) -> BTreeMap<TypeKey, (&'a TypeRef, &'a TypeRef)> {
530        let mut optionals = BTreeMap::new();
531        for ty in model
532            .elements()
533            .flat_map(element_types)
534            .flat_map(|ty| ty.walk())
535        {
536            let Some(inner) = ty.optional_inner() else {
537                continue;
538            };
539            if !self.explicitly_declares(ty) {
540                optionals.entry(ty.stripped_key()).or_insert((ty, inner));
541            }
542        }
543        optionals
544    }
545
546    fn explicitly_declares(&self, ty: &TypeRef) -> bool {
547        self.types.contains_key(&ty.key())
548            || self
549                .convert_decls
550                .iter()
551                .any(|decl| decl.key() == &ty.key() || decl.key() == &ty.stripped_key())
552    }

554    pub(crate) fn bindings(
555        &self,
556        model: &Flat,
557        registry: &impl prebindgen_registry::Conversions,
558        recipes: &prebindgen_registry::recipe::Recipes,
559    ) -> Result<prebindgen_registry::recipe::Bindings, Vec<RecipeError>> {
560        use prebindgen_registry::recipe::{
561            Ask, Bindings, Crossing, Direction, Origin, RecipeName, Site,
562        };
563
564        let mut bound = Bindings::builder();
565        let mut declared: Vec<TypeKey> = self
566            .types
567            .iter()
568            .filter(|(_, c)| matches!(c.kind, DeclaredKind::Data))
569            .map(|(k, _)| k.clone())
570            .collect();
571        declared.sort_by(|a, b| a.as_str().cmp(b.as_str()));
572
573        // The named Optional row is the representation suitable inside another
574        // composed shape. Where its child has a Product row it delegates to that
575        // row; a scalar child stays on its default row and lets the JNI compiler
576        // choose a niche or a `(present, value)` intermediate from the compiled
577        // child facts. The whole/default Optional row remains untouched.
578        //
579        // Bind both directions: function inputs construct the Rust optional,
580        // while returns and callback arguments deconstruct it.
581        for (outer, inner) in self.implicit_optionals(model).into_values() {
582            for direction in [Direction::Construct, Direction::Deconstruct] {
583                let outer = Crossing::new(outer.clone(), direction);
584                let row = outer.row(parts());
585                let inner = Crossing::new(inner.clone(), direction);
586                bound.bind(
587                    Site::part(&row, 0),
588                    inner.clone(),
589                    if recipes.key_of(&inner.key(), &parts()).is_some() {
590                        Ask::Recipe(parts())
591                    } else {
592                        Ask::Default
593                    },
594                    Origin::Part,
595                );
596            }
597        }
598
599        for key in declared {
600            let Some(ident) = key.ident() else { continue };
601            let Some(prebindgen_registry::flat::Type::Struct(s)) = model.declared_type(&ident)
602            else {
603                continue;
604            };
605            let Ok(ty) = model.classify(&syn::parse_quote!(#ident)) else {
606                continue;
607            };
608            let building = Crossing::new(ty.clone(), Direction::Construct);
609            let handing_out = Crossing::new(ty, Direction::Deconstruct);
610            for (index, field) in s.fields.iter().enumerate() {
611                // A field selects `parts` exactly when its crossing declares
612                // that row AND the declaration says this occurrence decomposes.
613                // Direct classes retain the existing `.jobject_input()` and
614                // erased-wrapper boundaries; every implicit Optional selects its
615                // part representation, which decides whether the child is
616                // another composition or one scalar intermediate.
617                if field.ty.optional_inner().is_none()
618                    && !self.field_crosses_as_its_fields(&field.ty)
619                {
620                    continue;
621                }
622                for (of, direction) in [
623                    (&building, Direction::Construct),
624                    (&handing_out, Direction::Deconstruct),
625                ] {
626                    let row = of.row(parts());
627                    let field_crossing = Crossing::new(field.ty.clone(), direction);
628                    if recipes.key_of(&field_crossing.key(), &parts()).is_none() {
629                        continue;
630                    }
631                    bound.bind(
632                        Site::part(&row, index),
633                        field_crossing,
634                        Ask::Recipe(parts()),
635                        Origin::Part,
636                    );
637                }
638            }
639        }
640        // Every function whose return the binding takes apart, bound to the
641        // `parts` recipe of its return type.
642        //
643        // This is what a site naming several values is: `Ask::Recipe` already
644        // lets a site pick a recipe, and a `parts` fragment already occupies
645        // several wires — the two facts stages 3 and 4 established, meeting.
646        // The registry needs nothing new to express a decomposed return.
647        for f in model.functions() {
648            // The value that crosses, through the layers a decomposition looks
649            // through: a `&T` return decomposes T, and so does an `Option<T>`
650            // or a `Vec<T>` — the shape rides the delivery, not the recipe.
651            let ret = f.ret.borrow_target().unwrap_or(&f.ret);
652            let ret = ret.optional_inner().unwrap_or(ret);
653            let ret = ret.sequence_elem().unwrap_or(ret);
654            let ret = ret.borrow_target().unwrap_or(ret);
655            let crossing = Crossing::new(ret.clone(), Direction::Deconstruct);
656            // Only where the type states one. A `sealed_class` always does; a
657            // `data_class` states one unless a field of it declines, and a
658            // return whose type states none crosses whole.
659            if recipes.key_of(&crossing.key(), &parts()).is_none() {
660                continue;
661            }
662            // And only where it is genuinely several. A decomposition that
663            // yields ONE value takes `Delivery::Return`: the wrapper hands that
664            // value back through its own conversion rather than through a
665            // builder, so the return site's crossing is the value's, not this
666            // type's. Asked of the composition, which is model and declaration
667            // only and so answerable here — the same property that let one
668            // composition serve both sides of `resolve`.
669            if self.out_wire_count(registry, ret) < 2 {
670                continue;
671            }
672            bound.bind(
673                Site {
674                    owner: f.name.clone(),
675                    role: prebindgen_registry::recipe::Role::Return,
676                },
677                crossing,
678                Ask::Recipe(parts()),
679                Origin::Adapter,
680            );
681        }
682
683        // A callback's `Invoke` recipe owns its argument conversions. Bind a
684        // model-derived data-class argument to `parts` so the Product fragment
685        // reaches `Compile::callback` before the trampoline is rendered.
686        for f in model.functions() {
687            for param in &f.params {
688                let prebindgen_registry::flat::TypeKind::Callback { args } =
689                    param.ty.unwrapped().kind()
690                else {
691                    continue;
692                };
693                let callback = Crossing::new(param.ty.clone(), Direction::Construct);
694                let row = callback.row(RecipeName::derived());
695                for (index, arg) in args.iter().enumerate() {
696                    let crossed = arg.borrow_target().unwrap_or(arg);
697                    let core = crossed.optional_inner().unwrap_or(crossed);
698                    if let Some(element) = core.sequence_elem() {
699                        let element = element.borrow_target().unwrap_or(element);
700                        if !self.field_crosses_as_its_fields(element) {
701                            continue;
702                        }
703                        let sequence = Crossing::new(arg.clone(), Direction::Deconstruct);
704                        let element_crossing =
705                            Crossing::new(element.borrowed(), Direction::Deconstruct);
706                        if recipes.key_of(&element_crossing.key(), &parts()).is_none() {
707                            continue;
708                        }
709                        bound.bind(
710                            Site::part(&sequence.row(RecipeName::derived()), 0),
711                            element_crossing,
712                            Ask::Recipe(parts()),
713                            Origin::Adapter,
714                        );
715                        continue;
716                    }
717                    // A model-derived data class, or a type whose deconstructor
718                    // declaration earned the `parts` row above. Both take that
719                    // row here, so the fragment the trampoline delivers and the
720                    // one the argument's own `Role::CallbackArg` site names are
721                    // one fragment rather than two rows over one crossing.
722                    if !self.field_crosses_as_its_fields(core)
723                        && registry.callback_arg_plan(&arg.key()).is_none()
724                    {
725                        continue;
726                    }
727                    let crossing = Crossing::new(arg.clone(), Direction::Deconstruct);
728                    if recipes.key_of(&crossing.key(), &parts()).is_none() {
729                        continue;
730                    }
731                    bound.bind(
732                        Site::part(&row, index),
733                        crossing,
734                        Ask::Recipe(parts()),
735                        Origin::Adapter,
736                    );
737                }
738            }
739        }
740
741        // Each value a callback delivers is a root site of its own — the
742        // function-unique `Role::CallbackArg` the registry names, not the
743        // `Role::Part` above, which is keyed by the callback recipe every
744        // function with that signature shares.
745        //
746        // Bound to `parts` exactly where the argument's crossing states that
747        // row, which is now every argument the trampoline takes apart: a
748        // `data_class`, a `sealed_class`, an implicit Optional over one of
749        // those, or a type whose deconstructor declaration earned the row
750        // declared above. An argument that crosses whole names nothing here and
751        // takes its crossing's default, attributed to the adapter.
752        for f in model.functions() {
753            for (param, p) in f.params.iter().enumerate() {
754                let prebindgen_registry::flat::TypeKind::Callback { args } =
755                    p.ty.unwrapped().kind()
756                else {
757                    continue;
758                };
759                for (arg, ty) in args.iter().enumerate() {
760                    let crossing = Crossing::new(ty.clone(), Direction::Deconstruct);
761                    if recipes.key_of(&crossing.key(), &parts()).is_none() {
762                        continue;
763                    }
764                    bound.bind(
765                        Site {
766                            owner: f.name.clone(),
767                            role: prebindgen_registry::recipe::Role::CallbackArg { param, arg },
768                        },
769                        crossing,
770                        Ask::Recipe(parts()),
771                        Origin::Adapter,
772                    );
773                }
774            }
775        }
776
777        bound.build(recipes)
778    }
