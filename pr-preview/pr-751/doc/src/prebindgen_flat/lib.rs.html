1//! # prebindgen-flat
2//!
3//! The independent flat parser and model that collectors build on.
4//!
5//! [`flat::Flat::builder`] parses `(syn::Item, [`SourceLocation`](prebindgen::SourceLocation))`
6//! records — read through [`prebindgen::Source`] — into one flat
7//! namespace — the language-agnostic index of everything a `#[prebindgen]`
8//! source crate declared. The Registry-based pipeline that resolves a binding
9//! over this model — type conversion, boundary expansion, Rust emission —
10//! ships in the separate
11//! [`prebindgen-registry`](https://docs.rs/prebindgen-registry) crate, which
12//! re-exports this crate's model modules at its own
13//! root so a language adapter names one crate for the whole pipeline.
14//!
15//! Secondary artifacts such as C headers or Kotlin sources are produced by the
16//! language adapter after the Rust registry is resolved; see the separate
17//! `prebindgen-c` and `prebindgen-jni` crates.

33/// A spelling with the spaces a reader does not want: `Option < Grade >` becomes
34/// `Option<Grade>`, `# [cfg (unix)]` becomes `#[cfg(unix)]`, and the space in
35/// `dyn Error` — the only kind that separates two words — stays.
36///
37/// Here because both the model's own output and the keys derived from it arrive
38/// spaced, and everything that shows either to a person has the same problem: a
39/// canonical type key spells a generic with spaces around its brackets, and a
40/// `TokenStream` prints a space between every pair of tokens. Neither is wrong,
41/// and neither is what a report, a build warning or a doc comment should show.
42pub fn close_up(spaced: &str) -> String {
43    let word = |c: char| c.is_alphanumeric() || c == '_';
44    let characters: Vec<char> = spaced.chars().collect();
45    let mut out = String::with_capacity(spaced.len());
46    for (index, &character) in characters.iter().enumerate() {
47        if character == ' ' {
48            let before = index.checked_sub(1).map(|i| characters[i]);
49            let after = characters.get(index + 1).copied();
50            let separates_words = before.is_some_and(word) && after.is_some_and(word);
51            if !separates_words {
52                continue;
53            }
54        }
55        out.push(character);
56    }
57    out
58}

63    fn punctuation_closes_up_and_words_stay_apart() {
64        for (key, expected) in [
65            ("Option < Grade >", "Option<Grade>"),
66            ("& [Payload]", "&[Payload]"),
67            ("dyn Error", "dyn Error"),

74                "# [cfg (target_os = \"linux\")]",
75                "#[cfg(target_os=\"linux\")]",
