1//! Build a registry: say what the binding contains, then close it.
2//!
3//! Every declaring method here records; none derives. The builder is a passive
4//! recorder precisely so it never has to call back into a generator to find out
5//! what it is meant to produce — and it is a *separate type* from [`Registry`]
6//! so that "still being described" and "finished, and answerable" cannot be
7//! confused for one another.
8
9use std::collections::{HashMap, HashSet};

11use super::*;

13/// A registry under construction.
14///
15/// Chain the declarations, hand over the conversions, then [`build`](Self::build):
16///
17/// ```ignore
18/// let registry = Registry::builder(flat)?
19///     .export(&name)
20///     .decompose(decompositions)
21///     .convert_with(|crossing, built| my_gen.convert(crossing, built))?
22///     .build()?;
23/// ```
24///
25/// The result is read-only. Nothing can add a crossing to a `Registry`, which
26/// is what makes "every crossing has a conversion" a fact about the type rather
27/// than a phase you have to be careful about.
28pub struct RegistryBuilder {
29    registry: Registry,
30    /// What has been answered so far, applied at [`Self::build`]. Not the
31    /// conversions — those stay with the adapter; see [`Answer`].
32    built: HashMap<Crossing, Answer>,
33    /// The scan runs once, on demand: it needs every declaration, and
34    /// [`Self::convert_with`] and [`Self::build`] each need
35    /// it to have run. `Some` holds the derived demand, in order.
36    order: Option<Vec<Crossing>>,
37}
38
39impl Registry {
40    /// Start describing a binding over this model.
41    ///
42    /// A `Flat` is what a registry projects, and reading captured prebindgen
43    /// output into one is [`FlatBuilder`](prebindgen_flat::flat::FlatBuilder)'s job
44    /// — so a build script says where items come from at the layer that owns
45    /// the question, and there is one such layer rather than two:
46    ///
47    /// ```
48    /// # prebindgen::Source::init_doctest_simulate();
49    /// use prebindgen_registry::{Flat, Registry};
50    ///
51    /// let flat = Flat::builder().source("source_ffi").build()?;
52    /// // Annotated only because nothing here resolves: in a build script `M` is
53    /// // fixed by the adapter passed to `resolve`, so no call site names it.
54    /// let registry: Registry = Registry::builder(flat)?.build()?;
55    /// assert!(registry.flat().function("test_function").is_some());
56    /// # Ok::<_, Box<dyn std::error::Error>>(())
57    /// ```
58    ///
59    /// Several sources compose there too, including one this crate renames:
60    ///
61    /// ```ignore
62    /// let flat = Flat::builder()
63    ///     .source(flat_crate::PREBINDGEN_OUT_DIR)
64    ///     .source_named(helpers::PREBINDGEN_OUT_DIR, "helpers")
65    ///     .build()?;
66    /// ```
67    ///
68    /// **Fails on anything the language cannot express** — a `self` receiver, an
69    /// `async fn`, a generic binder, a type form outside the grammar, or a
70    /// reference to a type the flat API does not declare. All of them at once, so
71    /// a source crate that needs migrating sees one list instead of one rebuild
72    /// per item. This is independent of what any binding declares: an
73    /// inexpressible item is a hard error whether or not it is ever named.
74    pub fn builder(flat: prebindgen_flat::flat::Flat) -> Result<RegistryBuilder, ScanError> {
75        let entries: Vec<NotExpressibleEntry> = flat
76            .unsupported()
77            .map(|u| NotExpressibleEntry {
78                name: u.name.clone(),
79                reason: u.error.to_string(),
80                location: (*u.origin.location).clone(),
81            })
82            .collect();
83        if !entries.is_empty() {
84            return Err(ScanError::NotExpressible { entries });
85        }
86
87        let mut registry = Registry::empty();
88        registry.flat = flat;
89        Ok(RegistryBuilder {
90            registry,
91            built: HashMap::new(),
92            order: None,
93        })
94    }
95}

97impl RegistryBuilder {
98    // ── configure: what this binding builds ───────────────────────────
99    //
100    // Pushed in by the generator before `resolve`. The registry never asks —
101    // it records, then derives the crossing set from what it was given.
102
103    /// An element this binding **exports**.
104    ///
105    /// The model says how to derive its crossings, so the caller does not: a
106    /// function's signature gives its parameters (in) and its return (out); a
107    /// const gives its value type (out). A name matching no element is an
108    /// error, reported with every other missing name at once by `resolve`
109    /// rather than here — a build script with three typos should learn all
110    /// three in one build.
111    pub fn export(mut self, name: &syn::Ident) -> Self {
112        self.registry.declared.functions.insert(name.clone());
113        self
114    }
115
116    /// A const this binding exports.
117    ///
118    /// Separate from [`Self::export`] only because *having a const mechanism at
119    /// all* is itself a fact: a binding that never calls this applies its
120    /// `on_const` policy to every captured const, while one that calls it emits
121    /// exactly what it names. See [`Self::declares_consts`].
122    pub fn export_const(mut self, name: &syn::Ident) -> Self {
123        self.registry
124            .declared
125            .consts
126            .get_or_insert_with(HashSet::new)
127            .insert(name.clone());
128        self
129    }
130
131    /// Declare that this binding has a const mechanism, even if it exports no
132    /// consts. Without it every captured const reaches the adapter's mandatory
133    /// `on_const` policy.
134    pub fn declares_consts(mut self) -> Self {
135        self.registry
136            .declared
137            .consts
138            .get_or_insert_with(HashSet::new);
139        self
140    }
141
142    /// A type this binding **exports**: it crosses in both directions, and its
143    /// body — a struct's fields, an enum's payloads — is scanned too.
144    ///
145    /// Takes the **type the declaration was written with**, like its sibling
146    /// [`Self::cross`], and derives the key here. It used to take the key alone,
147    /// which meant the scan had to recover tokens *from* the key to intern the
148    /// type and to diagnose its spelling — reasoning backwards from an identity
149    /// to a thing that already existed. A build script wrote `ptr_class!(Foo)`;
150    /// this is that `Foo` (#291).
151    ///
152    /// Declaring the same type twice keeps the **first** spelling. That is what
153    /// the `HashSet` this replaced did with the identity, and what
154    /// `register_class` does with a reopened declarator: the two spellings agree
155    /// on identity by construction, so the tie-break only decides which
156    /// equivalent rendering the scan reads, and it should not depend on
157    /// declaration order.
158    pub fn export_type(mut self, ty: Origin<syn::Type>) -> Self {
159        self.registry.declared.types.entry(ty.key()).or_insert(ty);
160        self
161    }
162
163    /// A type that **crosses** in one direction without being exported.
164    ///
165    /// The escape hatch for a crossing no signature can yield: a re-exported
166    /// foreign type named by a class declaration, or the value type of a
167    /// constant the binding synthesizes. Direction is explicit because these
168    /// are genuinely one-sided — which is what stops an output-only crossing
169    /// from silently lacking its input twin, the asymmetry the old
170    /// `required_output_types` had.
171    pub fn cross(mut self, dir: Direction, ty: &syn::Type) -> Self {
172        self.registry.declared.crossings.push((dir, ty.clone()));
173        self
174    }
175
176    /// `from`'s conversion needs `on`'s to exist first.
177    ///
178    /// [`Self::convert_with`] derives its order from the type structure, which
179    /// covers almost everything: an `Option<T>` visibly contains a `T`. It
180    /// cannot see a dependency the *declaration* creates — a `convert!` whose
181    /// body chains through a helper function's parameter type, say, where
182    /// nothing about the target type mentions the other side.
183    ///
184    /// State those here, and the order accounts for them. Getting it wrong is
185    /// not silent: the conversion that needed the missing one simply cannot be
186    /// built, and [`Self::build`] names it.
187    pub fn depends(mut self, from: Crossing, on: Crossing) -> Self {
188        self.registry.declared.edges.push((from, on));
189        self
190    }
191
192    /// A function this binding **references but never emits** — a helper whose
193    /// name appears in a declaration. Its absence is an error; its presence
194    /// emits nothing.
195    pub fn reference(mut self, name: &syn::Ident) -> Self {
196        self.registry.declared.helper_functions.insert(name.clone());
197        self
198    }
199
200    /// A function the **binding crate itself** defines, with the module path
201    /// generated calls should qualify it by.
202    ///
203    /// There is no `#[prebindgen]` item behind it, so this is the one input
204    /// that adds to the model rather than selecting from it: only the
205    /// signature is read, never the body. A name colliding with a captured
206    /// item is an error — the generated call would resolve the wrong function.
207    pub fn local_function(
208        mut self,
209        item_fn: syn::ItemFn,
210        origin: String,
211    ) -> Result<Self, ScanError> {
212        let ident = item_fn.sig.ident.clone();
213        // Written by hand in a build script, so the grammar is checked here or
214        // nowhere: a dropped `self` receiver would surface as an arity mismatch
215        // out of rustc on generated code, which is the wrong end of the pipeline
216        // to learn about a build.rs typo.
217        let lowered = self
218            .registry
219            .flat
220            .lower_signature(&item_fn)
221            .map_err(|error| ScanError::AdapterInvariant {
222                message: format!("binding-local fn `{ident}`: {error}"),
223            })?;
224        if self.registry.flat.element(&ident).is_some() {
225            return Err(ScanError::AdapterInvariant {
226                message: format!(
227                    "binding-local fn `{ident}` collides with a `#[prebindgen]` item — \
228                     the generated call would resolve the wrong fn; rename the \
229                     binding-local fn"
230                ),
231            });
232        }
233        self.registry.flat.add_local_function(lowered, origin);
234        Ok(self)
235    }

237    /// A function a decomposition reaches through rather than emits — excluded
238    /// from constructor composition, and the only functions a decomposer record
239    /// may name.
240    ///
241    /// Rides here until decompositions carry their own shape (step 2 of #251);
242    /// it is a property of the decomposition, not of the binding.
243    pub fn accessor(mut self, name: &syn::Ident) -> Self {
244        self.registry.declared.accessors.insert(name.clone());
245        self
246    }
247
248    /// The receiver type of a function emitted as a method. Same temporary
249    /// home as [`Self::accessor`].
250    pub fn method_receiver(mut self, name: &syn::Ident, receiver: TypeKey) -> Self {
251        self.registry
252            .declared
253            .method_receivers
254            .insert(name.clone(), receiver);
255        self
256    }

258    /// How this binding's composites cross **in pieces** instead of whole.
259    ///
260    /// Stated once, before [`Self::build`]. Replaces five separate callbacks
261    /// the registry used to make into the generator; see [`Decompositions`].
262    pub fn decompose(mut self, d: Decompositions) -> Self {
263        self.registry.declared.decompositions = d;
264        self
265    }

301    /// Readings introduced as leaves of resolved parameter expansions.
302    ///
303    /// Derives the registry first when necessary, so callers cannot silently
304    /// observe an empty pre-validation plan store. The iterator exposes only
305    /// the readings recipe declaration needs, not the expansion map's storage
306    /// or key representation.
307    pub fn expansion_leaf_readings(
308        &mut self,
309    ) -> Result<impl Iterator<Item = &prebindgen_flat::flat::TypeRef>, WriteRustError> {
310        self.derive()?;
311        Ok(self
312            .registry
313            .expansion_plans
314            .values()
315            .flat_map(|plan| plan.leaves.iter().map(|leaf| &leaf.ty)))
316    }

325    /// Run the scan and apply the decompositions, once.
326    ///
327    /// Private and idempotent: three entry points need it to have happened, and
328    /// none of them should care whether it already did.
329    fn derive(&mut self) -> Result<&[Crossing], WriteRustError> {
330        if self.order.is_none() {
331            let mut declared = std::mem::take(&mut self.registry.declared);
332            let out = (|| {
333                self.registry.scan_declared_items(&declared)?;
334                self.registry.apply_adapter_plans(&mut declared)
335            })();
336            self.registry.declared = declared;
337            out?;
338            self.order = Some(self.registry.crossings());
339        }
340        Ok(self.order.as_deref().unwrap_or_default())
341    }
342
343    /// What a conversion — or a validation — is written against right now: the
344    /// model and the full crossing population.
345    fn view(&self) -> Building<'_> {
346        Building::new(&self.registry, self.order.as_deref().unwrap_or_default())
347    }
348
349    /// Check this binding against a generator's own invariants, now that the
350    /// scan has read every declared signature.
351    ///
352    /// Earliest it can run: a missing declaration has already hard-errored, so
353    /// a check here sees only items that exist.
354    pub fn validate_with<E: Prebindgen>(mut self, adapter: &E) -> Result<Self, WriteRustError> {
355        self.derive()?;
356        adapter
357            .validate(&self.view())
358            .map_err(|message| ScanError::AdapterInvariant { message })?;
359        Ok(self)
360    }
361
362    /// Build a conversion for each crossing, in dependency order.
363    ///
364    /// `f` is called once per crossing, and answers with an [`Answer`] rather
365    /// than the conversion itself: the executable conversion belongs to the
366    /// adapter's frozen plan, and what the registry needs back is which other
367    /// crossings this one is built out of. Returning `None` records a gap
368    /// — whether that gap matters is decided by [`Self::build`], not here.
369    ///
370    /// The walk is inner types first, so by the time `f` sees `Option<Handle>`
371    /// it has already answered `Handle`.
372    ///
373    /// The one way to supply them. Nothing about it lets the registry choose
374    /// when to call back — the closure is yours, and the walk is finished
375    /// before this returns.
376    pub fn convert_with<F>(mut self, mut f: F) -> Result<Self, WriteRustError>
377    where
378        F: FnMut(&Crossing, &Building<'_>) -> Option<Answer>,
379    {
380        let order = self.derive()?.to_vec();
381        for crossing in &order {
382            if let Some(answer) = f(crossing, &self.view()) {
383                self.built.insert(crossing.clone(), answer);
384            }
385        }
386        Ok(self)
387    }

389    /// The scanned registry, with no conversions applied and no completeness
390    /// check.
391    ///
392    /// Test-only, and deliberately so: it is the state between "described" and
393    /// "answerable", which is exactly what the split exists to keep out of
394    /// everyone else's hands.
395    #[cfg(test)]
396    pub(crate) fn scanned(mut self) -> Result<Registry, ScanError> {
397        // Narrower than `build`'s error on purpose: the scan is the only phase
398        // this runs, so a test matching on `ScanError` says what it means.
399        match self.derive() {
400            Ok(_) => Ok(self.registry),
401            Err(WriteRustError::Scan(e)) => Err(e),
402            Err(other) => panic!("scanned(): unexpected non-scan failure: {other}"),
403        }
404    }
405
406    /// Close the binding: apply every conversion, check the set is complete,
407    /// and hand back a registry that can only be read.
408    ///
409    /// A crossing with no conversion is not itself a failure — the scan
410    /// over-approximates on purpose. What fails is a crossing *reachable from
411    /// an export* with none, and the error names every one at once.
412    pub fn build(mut self) -> Result<Registry, WriteRustError> {
413        self.derive()?;
414        for ((dir, key), entry) in self.built {
415            if let Some(cell) = self.registry.type_table_mut(dir).get_mut(&key) {
416                cell.entry = Some(entry);
417            }
418        }
419        crate::resolve::check_complete(&self.registry)?;
420        Ok(self.registry)
421    }
422}

424/// A builder answers the same questions a finished registry does. It used to
425/// answer one fewer — it lent out conversions handed over *so far*, so a
426/// generator writing one could see its inners and nothing else. An adapter
427/// keeps its own conversions now, so that read is gone and the rest — the
428/// model, the decompositions — is complete from the moment it is declared.
429impl Conversions for RegistryBuilder {
430    fn reading(&self, key: &TypeKey) -> Option<prebindgen_flat::flat::TypeRef> {
431        self.registry.reading(key)
432    }
433    fn flat(&self) -> &prebindgen_flat::flat::Flat {
434        &self.registry.flat
435    }
436    fn crossing_keys(&self, dir: Direction) -> Vec<TypeKey> {
437        self.order
438            .as_deref()
439            .unwrap_or_default()
440            .iter()
441            .filter(|(d, _)| *d == dir)
442            .map(|(_, k)| k.clone())
443            .collect()
444    }
445    fn callback_arg_plan(&self, key: &TypeKey) -> Option<&crate::unfold::UnfoldPlan> {
446        self.registry.callback_arg_plans.get(key)
447    }
448    fn callback_arg_plans(&self) -> &HashMap<TypeKey, crate::unfold::UnfoldPlan> {
449        &self.registry.callback_arg_plans
450    }
451    fn unfold_plans(&self) -> &HashMap<syn::Ident, crate::unfold::UnfoldPlan> {
452        &self.registry.unfold_plans
453    }
454    fn error_plans(&self) -> &HashMap<syn::Ident, crate::unfold::UnfoldPlan> {
455        &self.registry.error_plans
456    }
457    fn decon_plans(&self) -> &HashMap<crate::unfold::DeconId, crate::unfold::DeconSpec> {
458        &self.registry.decon_plans
459    }
