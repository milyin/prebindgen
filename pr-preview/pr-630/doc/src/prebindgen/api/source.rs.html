28/// Provides access to prebindgen data generated by the `#[prebindgen]` macro
29///
30/// The `Source` struct reads prebindgen data files from a directory that was
31/// initialized by [`init_prebindgen_out_dir`](crate::init_prebindgen_out_dir) in
32/// the source project's `build.rs`. This allows destination projects to access
33/// and process the collected FFI interface definitions.
34///
35/// # Workflow
36///
37/// 1. **Source project**: Calls `init_prebindgen_out_dir()` in `build.rs` and uses `#[prebindgen]` macro
38/// 2. **Destination project**: Uses `Source::new()` to read the collected data
39///
40/// # Groups
41///
42/// Items are organized into groups based on the first parameter of the `#[prebindgen]`
43/// macro. For example, `#[prebindgen("structs")]` creates items in the "structs" group.
44/// Items without an explicit group are placed in the "default" group.
45///
46/// # Example
47///
48/// The `PREBINDGEN_OUT_DIR` constant is defined in the source FFI crate using the
49/// `prebindgen_out_dir!()` macro:
50///
51/// ```rust
52/// // In source_ffi/src/lib.rs
53/// use prebindgen_proc_macro::{features, prebindgen, prebindgen_out_dir};
54///
55/// pub const PREBINDGEN_OUT_DIR: &str = prebindgen_out_dir!();
56/// pub const FEATURES: &str = features!();
57///
58/// #[prebindgen]
59/// pub fn my_function() -> i32 { 42 }
60/// ```
61///
62/// Then in the destination project's `build.rs`:
63///
64/// ```
65/// # prebindgen::Source::init_doctest_simulate();
66/// let source = prebindgen::Source::new("source_ffi");
67///
68/// // Process all items
69/// for (item, location) in source.items_all() {
70///     // Process each syn::Item...
71/// }
72/// ```
73#[derive(Clone)]
74pub struct Source {
75    crate_name: String,
76    items: HashMap<String, Vec<(syn::Item, SourceLocation)>>,
77    // Configuration needed to build a CfgFilter at iteration time
78    features_constant: Option<String>,
79    target_triple: Option<String>,
80    features_list: Vec<String>, // enabled features, from the description file
81}

100impl Source {
101    fn build_internal(
102        input_dir: &Path,
103        features_constant: Option<String>,
104        target_triple: Option<String>,
105        crate_name_override: Option<String>,
106    ) -> Self {
107        if let Some(source) = DOCTEST_SOURCE.with(|source| (*source.borrow()).clone()) {
108            return source;
109        }
110        if !input_dir.is_dir() {
111            panic!(
112                "Input directory {} does not exist or is not a directory",
113                input_dir.display()
114            );
115        }
116        // The description also carries the format these captures are in, and
117        // reading it is what verifies that this prebindgen wrote them — so it
118        // is read even when the name it holds is overridden. The override
119        // wins: it is the name THIS crate references the source crate by (a
120        // Cargo.toml dependency rename).
121        let output = Output::read(input_dir);
122        let crate_name = crate_name_override.unwrap_or(output.package.name);
123
124        let groups = Self::discover_groups(input_dir);
125        let mut items = HashMap::new();
126        for group in groups {
127            let records = Self::read_group(input_dir, &group);
128            let group_items = records
129                .iter()
130                .map(|r| {
131                    // Stamp the origin crate into every item's location: the
132                    // captured JSONL doesn't carry it (the proc-macro runs
133                    // inside the crate), but from here on the item streams
134                    // are self-describing — streams from several sources can
135                    // be chained into one `Flat::builder().items(..)` call
136                    // without losing per-item origins.
137                    let (item, mut loc) = r.parse();
138                    loc.crate_name = Some(crate_name.clone());
139                    (item, loc)
140                })
141                .collect::<Vec<_>>();
142            items.insert(group, group_items);
143        }
144
145        let features_list = output.package.features;
146
147        Self {
148            crate_name,
149            items,
150            features_constant,
151            features_list,
152            target_triple,
153        }
154    }

213    /// Returns an iterator over items from specific groups
214    ///
215    /// # Parameters
216    ///
217    /// * `groups` - Array of group names to include
218    ///
219    /// # Example
220    ///
221    /// ```
222    /// # prebindgen::Source::init_doctest_simulate();
223    /// let source = prebindgen::Source::new("source_ffi");
224    /// // Process only items from "structs" and "functions" groups
225    /// let items = source.items_in_groups(&["structs"]).collect::<Vec<_>>();
226    /// assert_eq!(items.len(), 1); // only TestStruct should be present
227    /// ```
228    pub fn items_in_groups<'a>(
229        &'a self,
230        groups: &'a [&'a str],
231    ) -> impl Iterator<Item = (syn::Item, SourceLocation)> + 'a {
232        // Build a cfg filter and apply it lazily with itertools::batching
233        let mut filter = self.build_cfg_filter();
234        groups
235            .iter()
236            .filter_map(|group| self.items.get(*group))
237            .flat_map(|records| records.iter().cloned())
238            .batching(move |iter| filter.call(iter))
239    }

241    /// Returns an iterator over items excluding specific groups
242    ///
243    /// # Parameters
244    ///
245    /// * `groups` - Array of group names to exclude
246    ///
247    /// # Example
248    ///
249    /// ```
250    /// # prebindgen::Source::init_doctest_simulate();
251    /// let source = prebindgen::Source::new("source_ffi");
252    /// let items = source.items_except_groups(&["structs"]).collect::<Vec<_>>();
253    /// assert_eq!(items.len(), 1); // only test_function should be present
254    /// ```
255    pub fn items_except_groups<'a>(
256        &'a self,
257        groups: &'a [&'a str],
258    ) -> impl Iterator<Item = (syn::Item, SourceLocation)> + 'a {
259        // Build a cfg filter and apply it lazily with itertools::batching
260        let mut filter = self.build_cfg_filter();
261        self.items
262            .iter()
263            .filter(|(group, _)| !groups.contains(&group.as_str()))
264            .flat_map(|(_, records)| records.iter().cloned())
265            .batching(move |iter| filter.call(iter))
266    }

268    /// Returns an iterator over all items from all groups
269    ///
270    /// This is the most commonly used method for processing all prebindgen items.
271    ///
272    /// # Example
273    ///
274    /// ```
275    /// # prebindgen::Source::init_doctest_simulate();
276    /// let source = prebindgen::Source::new("source_ffi");
277    /// let items: Vec<_> = source.items_all().collect();
278    /// assert_eq!(items.len(), 2); // should contain TestStruct and test_function
279    /// ```
280    pub fn items_all(&self) -> impl Iterator<Item = (syn::Item, SourceLocation)> + '_ {
281        // Build a cfg filter and apply it lazily with itertools::batching
282        let mut filter = self.build_cfg_filter();
283        self.items
284            .values()
285            .flat_map(|records| records.iter().cloned())
286            .batching(move |iter| filter.call(iter))
287    }

289    /// Internal: construct a CfgFilter from the stored configuration and features file
290    fn build_cfg_filter(&self) -> cfg_filter::CfgFilter {
291        let mut builder = cfg_filter::CfgFilter::builder();
292        if let Some(const_name) = &self.features_constant {
293            // If the provided constant isn't fully qualified, qualify it with the crate name
294            let qualified_const = if const_name.contains("::") {
295                const_name.clone()
296            } else {
297                format!("{}::{}", self.crate_name.replace('-', "_"), const_name)
298            };
299            // Prepend each feature with "crate_name/" and join with spaces
300            let features_list = self
301                .features_list
302                .iter()
303                .map(|f| format!("{}/{}", self.crate_name, f))
304                .join(" ");
305            builder = builder.predefined_features(qualified_const, features_list);
306        }
307        if let Some(target) = &self.target_triple {
308            let target_triple = TargetTriple::parse(target)
309                .unwrap_or_else(|e| panic!("Failed to parse target triple '{}': {}", target, e));
310            if let Some(arch) = target_triple.arch() {
311                builder = builder.enable_target_arch(arch);
312            }
313            if let Some(vendor) = target_triple.vendor() {
314                builder = builder.enable_target_vendor(vendor);
315            }
316            if let Some(os) = target_triple.os() {
317                builder = builder.enable_target_os(os);
318            }
319            if let Some(env) = target_triple.env() {
320                builder = builder.enable_target_env(env);
321            }
322        }
323        builder.build()
324    }

326    /// Read every capture file of one group: `<input_dir>/<group>/*.jsonl`.
327    ///
328    /// The proc-macro names each file after the record it holds, so a file that
329    /// two compilations both produced exists once. Records that only *some*
330    /// compilations see — a `#[cfg(test)]` item, say — arrive as extra files and
331    /// are merged here, which is why this deduplicates by name (plus `cfg`)
332    /// rather than trusting the file set.
333    fn read_group<P: AsRef<Path>>(input_dir: P, group: &str) -> Vec<Record> {
334        let mut record_map = HashMap::new();
335
336        for path in Self::group_files(input_dir.as_ref(), group) {
337            #[cfg(feature = "debug")]
338            println!("Reading exported file: {}", path.display());
339
340            match read_jsonl_file(&path) {
341                Ok(records) => {
342                    for record in records {
343                        // Use a HashMap to deduplicate records by name and cfg
344                        let key = if let Some(cfg) = &record.cfg {
345                            format!("{}#{}", record.name, cfg)
346                        } else {
347                            record.name.clone()
348                        };
349                        record_map.insert(key, record);
350                    }
351                }
352                Err(e) => {
353                    panic!("Failed to read {}: {}", path.display(), e);
354                }
355            }
356        }
357
358        // Return deduplicated records for this group
359        record_map.into_values().collect::<Vec<_>>()
360    }

362    /// The capture files belonging to `group`.
363    fn group_files(input_dir: &Path, group: &str) -> Vec<PathBuf> {
364        let mut files = Vec::new();
365
366        if let Ok(entries) = fs::read_dir(input_dir.join(group_dir_name(group))) {
367            for entry in entries.flatten() {
368                let path = entry.path();
369                if path
370                    .file_name()
371                    .and_then(|name| name.to_str())
372                    .is_some_and(|name| name.ends_with(JSONL_EXTENSION))
373                {
374                    files.push(path);
375                }
376            }
377        }
378
379        files
380    }

382    /// Internal method to discover all available groups from the directory
383    ///
384    /// A group is a subdirectory whose name encodes the group reversibly (so
385    /// that two groups differing only in case, or named after a Windows device,
386    /// still get one directory each), and the exact name is decoded back out of
387    /// it — see [`layout`](crate::layout).
388    ///
389    /// A capture that is *not* under such a directory is refused rather than
390    /// skipped. The description file says which format this directory is in,
391    /// but the macro that wrote the captures is a different package from the
392    /// build script that wrote the description
393    /// ([`output`](crate::api::output)), so a capture outside the layout is how
394    /// a macro that never learned the format announces itself — prebindgen ≤
395    /// 0.5.0 leaves `{group}_{pid}_{thread}.jsonl` here. Skipping those is the
396    /// silent empty binding this whole file exists to prevent.
397    fn discover_groups<P: AsRef<Path>>(input_dir: P) -> HashSet<String> {
398        let input_dir = input_dir.as_ref();
399        let mut groups = HashSet::new();
400
401        if let Ok(entries) = fs::read_dir(input_dir) {
402            for entry in entries.flatten() {
403                let path = entry.path();
404                // A group directory is ASCII, so a name that is not UTF-8 is
405                // not one — but it still has to face the damage check below
406                // rather than being skipped with its captures inside.
407                let file_name = path.file_name().and_then(|name| name.to_str());
408                if path.is_dir() {
409                    match file_name.and_then(decode_group_dir_name) {
410                        Some(group) => {
411                            groups.insert(group);
412                        }
413                        None => assert!(
414                            !holds_captures(&path),
415                            "{} holds prebindgen captures but is not a group directory; \
416                             the capture directory is damaged, or was written by a \
417                             prebindgen that lays captures out differently — rebuild the \
418                             source crate",
419                            path.display()
420                        ),
421                    }
422                } else {
423                    assert!(
424                        !file_name.is_some_and(|name| name.ends_with(JSONL_EXTENSION)),
425                        "{} is a prebindgen capture outside any group directory. The \
426                         `#[prebindgen]` macro that wrote it does not write the layout \
427                         this prebindgen reads: `prebindgen-proc-macro` and `prebindgen` \
428                         are separate dependencies of the source crate, and Cargo does \
429                         not check that they agree. Name one prebindgen in both.",
430                        path.display()
431                    );
432                }
433            }
434        }
435
436        groups
437    }

440/// Whether a directory holds capture files.
441///
442/// Used to tell a directory that is not a group's — Cargo puts other things in
443/// `OUT_DIR` — from one whose name this build cannot decode, which would mean
444/// dropping records silently.
445fn holds_captures(dir: &Path) -> bool {
446    fs::read_dir(dir).is_ok_and(|entries| {
447        entries.flatten().any(|entry| {
448            entry
449                .file_name()
450                .to_str()
451                .is_some_and(|name| name.ends_with(JSONL_EXTENSION))
452        })
453    })

456/// Builder for constructing a `Source` with custom options
457pub struct Builder {
458    input_dir: PathBuf,
459    features_constant: Option<String>,
460    target_triple: Option<String>,
461    crate_name: Option<String>,
462}
463
464impl Builder {
465    fn new<P: AsRef<Path>>(input_dir: P) -> Self {
466        let target = std::env::var("TARGET").unwrap_or_default();
467        let host = std::env::var("HOST").unwrap_or_default();
468        let target_triple = if !target.is_empty() {
469            Some(target)
470        } else if !host.is_empty() {
471            Some(host)
472        } else {
473            None
474        };
475        Self {
476            input_dir: input_dir.as_ref().to_path_buf(),
477            features_constant: Some("FEATURES".to_string()),
478            target_triple,
479            crate_name: None,
480        }
481    }
482
483    /// Override the crate name recorded by the source crate's build script
484    /// (`CARGO_PKG_NAME` at capture time) with the name **this** crate
485    /// references the source crate by. Needed when the dependency is
486    /// renamed in Cargo.toml:
487    ///
488    /// ```toml
489    /// [dependencies]
490    /// myflat = { package = "some-flat-crate", version = "..." }
491    /// ```
492    ///
493    /// The name is stamped into every yielded item's `SourceLocation`, so
494    /// it is what generated code qualifies references with
495    /// (`myflat::…`) and what qualifies the features-guard constant. Being
496    /// per-`Source`, it composes with multi-source registries — each
497    /// chained stream carries its own (possibly overridden) origin.
498    pub fn crate_name(mut self, name: impl Into<String>) -> Self {
499        self.crate_name = Some(name.into());
500        self
501    }
502
503    /// Enables or disables filtering by features when extracting collected data.
504    /// Accepts name of the constant with the list of features in the source crate.
505    ///
506    /// Pass `None` to disable feature filtering.
507    ///
508    /// Pass, for example, `Some("source_crate::FEATURES")` or `Some("FEATURES")`
509    /// to enable filtering. The value is the name of the features constant
510    /// from the source crate.
511    ///
512    /// It's important to note that the set of features to filter is determined
513    /// not by this constant, but by the `package.features` list in
514    /// `prebindgen_output.toml` in the prebindgen output directory. These are features which the source crate was
515    /// built with *as build.rs dependency*. The constant contains the features which
516    /// the source crate was built *as library dependency*.
517    /// The purpose of the constant is to use it in the assert in the
518    /// generated code to ensure that both feature lists are the same.
519    /// If it's not the case, compilation fails and it's developer's
520    /// job to ensure that the source crate is configured in the same way for
521    /// both `[dependencies]` and `[build-dependencies]`.
522    ///
523    /// Filtering is enabled by default; the default constant name is `FEATURES`.
524    ///
525    /// The features constant should be defined in the source crate as follows:
526    /// ```rust
527    /// const FEATURES: &str = prebindgen_proc_macro::features!();
528    /// ```
529    #[roxygen]
530    pub fn enable_feature_filtering(
531        mut self,
532        /// Full name of the constant with features in the source crate
533        name: Option<impl Into<String>>,
534    ) -> Self {
535        self.features_constant = name.map(|n| n.into());
536        self
537    }
538
539    /// Enables filtering source code by target-triple. This is
540    /// useful if some binding generators or custom tools over them
541    /// encounter problems with build conditions in the source code.
542    ///
543    /// Accepts the target triple string.
544    ///
545    /// Default value is the value of the environment variable
546    /// `TARGET` or `HOST` if `TARGET` is not set.
547    ///
548    /// Pass `None` to disable target filtering.
549    ///
550    /// When enabled, the code will be filtered by the target triple specified
551    /// (architecture, vendor, os, env).
552    ///
553    /// It's useful to mention that it's hard to imagine a scenario
554    /// when it's necessary to change the target triple.
555    /// The `Source` object is used in the final crate, the one which builds the
556    /// no-mangle library. On this stage the `TARGET` environment variable contains
557    /// real target triple for cross-compilation scenarios.
558    ///
559    /// There is the case when build.rs is not aware of the cross-compilation
560    /// target: the build.rs for library in [build-dependencies] section.
561    /// This is the situation which the "ffi" crate found itself in. So sometimes
562    /// for correct code generation it's necessary to inform the underlying "ffi"
563    /// crate about the real target in a way like this:
564    ///
565    /// ```sh
566    /// CROSS_TARGET=x86_64-pc-windows-gnu cargo build --target x86_64-pc-windows-gnu
567    /// ```
568    ///
569    /// But in typical situation this doesn't affect the `Source` object which
570    /// reads the data prepared by "ffi" crate on the following stage.
571    ///
572    /// So this `Some(<target-triple>)` is added mostly for API completeness,
573    /// typical usage if this function will be with `None` argument to disable
574    /// filtering by target triple if it's necessary to see the original
575    /// collected code.
576    #[roxygen]
577    pub fn enable_target_filtering(
578        mut self,
579        /// The target triple to filter collected sources
580        name: Option<impl Into<String>>,
581    ) -> Self {
582        self.target_triple = name.map(|n| n.into());
583        self
584    }
585
586    /// Build the `Source` instance
587    pub fn build(self) -> Source {
588        Source::build_internal(
589            &self.input_dir,
590            self.features_constant,
591            self.target_triple,
592            self.crate_name,
593        )
594    }

604    /// Publish a record into `group`'s directory the way the macro does.
605    fn write(dir: &Path, group: &str, name: &str) {
606        let group_dir = dir.join(group_dir_name(group));
607        fs::create_dir_all(&group_dir).unwrap();
608        let record = Record::new(
609            RecordKind::Struct,
610            name.to_string(),
611            format!("pub struct {name};"),
612            Default::default(),
613            None,
614        );
615        let serialized = record.to_jsonl_string().unwrap();
616        fs::write(
617            group_dir.join(crate::api::layout::capture_file_name(name, &serialized)),
618            format!("{serialized}\n"),
619        )
620        .unwrap();
621    }

633        let mut groups = Source::discover_groups(dir).into_iter().collect::<Vec<_>>();

650    #[test]
651    fn groups_differing_only_in_case_stay_apart() {
652        // `.join(group)` would merge these on macOS and Windows, so the
653        // directory name leads with a digest of the group.
654        let dir = tempfile::tempdir().unwrap();
655        write(dir.path(), "Foo", "Upper");
656        write(dir.path(), "foo", "Lower");
