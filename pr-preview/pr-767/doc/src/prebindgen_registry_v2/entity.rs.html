1//! Which elements of the model are the API.
2//!
3//! The model is structural: it holds everything the source wrote and the
4//! binding stated, a guard and an unsupported item included, and it does not
5//! rank them. Ranking is this crate's judgment — a declaration can name a
6//! type, a function or a constant, and nothing else — and it is stated here,
7//! once, as the view a declaration resolves to.
8
9use prebindgen_flat::flat::{Constant, Element, Flat, Function, Name, Type};
10
11/// One real item of the API: a type, a function or a constant, with its whole
12/// description.
13///
14/// The three kinds a binding can name, and the only three. Where an entity
15/// came from is not a kind: a `#[prebindgen]` item and one the binding
16/// defines itself — a helper with a stated signature, a type the source never
17/// exported that the binding represents as a handle — are the same entity,
18/// and differ in their origin alone. The location says which crate, and for
19/// the binding's own, the module generated code reaches the item through.
20#[derive(Clone, Copy, Debug)]
21pub enum Entity<'a> {
22    Type(&'a Type),
23    Function(&'a Function),
24    Constant(&'a Constant),
25}
26
27/// Which of the three kinds, on its own.
28#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
29pub enum EntityKind {
30    Type,
31    Function,
32    Constant,
33}
34
35impl<'a> Entity<'a> {
36    /// The element as an entity — `None` for a guard or an unsupported item,
37    /// which are in the model and not in the API.
38    pub fn of(element: &'a Element) -> Option<Self> {
39        match element {
40            Element::Type(t) => Some(Entity::Type(t)),
41            Element::Function(f) => Some(Entity::Function(f)),
42            Element::Constant(c) => Some(Entity::Constant(c)),
43            Element::Guard(_) | Element::Unsupported(_) => None,
44        }
45    }
46
47    /// The entity with this name, if the model holds one: captured, or the
48    /// binding's own.
49    pub fn named<N: Name + ?Sized>(flat: &'a Flat, name: &N) -> Option<Self> {
50        Entity::of(flat.element(name)?)
51    }
52
53    pub fn kind(&self) -> EntityKind {
54        match self {
55            Entity::Type(_) => EntityKind::Type,
56            Entity::Function(_) => EntityKind::Function,
57            Entity::Constant(_) => EntityKind::Constant,
58        }
59    }
60
61    pub fn name(&self) -> &'a syn::Ident {
62        match self {
63            Entity::Type(t) => t.name(),
64            Entity::Function(f) => &f.name,
65            Entity::Constant(c) => &c.name,
66        }
67    }
68
69    /// The crate stamp on the entity's location: the crate a captured item
70    /// came from, and for the binding's own, the module generated code
71    /// reaches it through.
72    pub fn crate_name(&self) -> Option<&'a str> {
73        let location = match self {
74            Entity::Type(t) => t.location(),
75            Entity::Function(f) => &f.origin.location,
76            Entity::Constant(c) => &c.origin.location,
77        };
78        location.crate_name.as_deref()
79    }
80}
