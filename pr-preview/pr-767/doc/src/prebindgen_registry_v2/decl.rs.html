1//! What a binding declared, in terms neither language owns.
2//!
3//! A frontend turns its own declaration storage into these — one [`Declaration`]
4//! per thing the user asked for, inside the
5//! [`BindingRequests`](crate::BindingRequests) it hands the engine. The
6//! declaration is what the report accounts for, and what the target looks its
7//! own configuration up by — the request carries none: the foreign name and
8//! the declarator word the report prints come from that lookup.
9
10use prebindgen_flat::flat::{Element, Flat, TypeKey};
11use serde::Serialize;
12
13/// One thing a binding asked for: what the target gets, named by what the
14/// Rust source calls it, and what the engine plans it from.
15///
16/// A name alone says neither: `Foo` may be a type given a representation or a
17/// wire mapping declared for it. The variant says which, and it says it once
18/// — what the target gets and which entity, if any, it is planned from both
19/// follow from the variant instead of being stated beside it, so they cannot
20/// disagree and a pair that means nothing (a callback backed by a captured
21/// constant) cannot be written down. What a declaration does *not* say is how
22/// the target shows the entity: a function exposed as a Kotlin `val` read
23/// through it is still [`Declaration::Function`], and the `val` is the
24/// target's choice, recorded under that declaration.
25///
26/// An **entity** is one real item of the API — a type, a function or a
27/// constant — and the three variants that name one are the registry's whole
28/// judgment of which elements those are: the model holds a guard and an
29/// unsupported item beside them and ranks none of it. Where an entity came
30/// from is not part of this either. A `#[prebindgen]` function and one the
31/// binding defines itself are both a [`Declaration::Function`]; the model
32/// holds both, and only its [`Origin`](prebindgen_flat::flat::Origin) tells
33/// them apart. Three declarations name no entity: a callback signature and a
34/// constant the binding computes on the foreign side, because nothing in Rust
35/// backs them; and a conversion, which is the binding's own wire mapping
36/// *about* a type — `convert!(Option<Payload>)` — and requires no item of that
37/// name in the model.
38///
39/// [`generate`](crate::generate) routes on this: each planner is reached by its
40/// own variants and is handed the entity they name, rather than a kind to
41/// decode and a name to look up.
42///
43/// It is also the declaration's identity — what an outcome is keyed by, what a
44/// [`Position`](crate::target::Position) is rooted at, what a target's
45/// [`Requirement`](crate::target::Requirement) resolves to. Stable across runs
46/// and across pipelines, so a report, a build script and a capability-selected
47/// test section can all name the same declaration: it is what the *source*
48/// calls the thing, not what the target does, and a rename on the foreign side
49/// must not silently retire a test's requirement. `<kind>:<name>` —
50/// `type:Stamp`, `fn:stamp_sum` — is how it prints and how the report writes
51/// it, and that spelling is a rendering: nothing reads a declaration back out
52/// of it. Declarations order as they print, so a report sorted by declaration
53/// reads in id order.
54#[derive(Clone, Debug, PartialEq, Eq, Hash)]
55pub enum Declaration {
56    /// A function, exported through a wrapper that calls it — as a foreign
57    /// function, or as whatever else the target chooses to show a call as: a
58    /// Kotlin `val` over `constant!(X).fun(fun!(f))` is this, with the `val`
59    /// the target's business.
60    Function(syn::Ident),
61    /// A constant, exposed as a foreign constant.
62    Const(syn::Ident),
63    /// A type, given a foreign representation.
64    Type(TypeKey),
65    /// A declared conversion between a type and its wire form, defined by the
66    /// binding.
67    Conversion(TypeKey),
68    /// A callback signature the binding exports as a foreign callable. No
69    /// entity backs one, so the signature is the name it goes by.
70    Callback(String),
71    /// A foreign constant the binding computes rather than reads —
72    /// `constant!(X).expr(..)`. No entity holds its value.
73    ComputedConst(String),
74}
75
76impl Declaration {
77    /// Whether this declares a type.
78    pub fn is_type(&self) -> bool {
79        matches!(self, Declaration::Type(_))
80    }
81
82    /// The name the model indexes the entity this declaration names under —
83    /// `None` for a callback, a computed constant and a conversion.
84    ///
85    /// A type key may carry arguments the item does not — `ptr_class!(
86    /// Publisher<'static>)` means the item `Publisher` — so the item's name is
87    /// the key's last segment without them. A key that is not a path at all
88    /// names no item, and is looked up as it is spelled, to be found missing.
89    pub fn entity_name(&self) -> Option<String> {
90        match self {
91            Declaration::Function(ident) | Declaration::Const(ident) => Some(ident.to_string()),
92            Declaration::Type(key) => {
93                Some(key.short_name().unwrap_or_else(|| key.as_str().to_string()))
94            }
95            Declaration::Conversion(_)
96            | Declaration::Callback(_)
97            | Declaration::ComputedConst(_) => None,
98        }
99    }
100
101    /// The element this declaration names, if the model holds it.
102    ///
103    /// Entities live in one flat namespace holding functions, types and
104    /// constants, so the name alone finds any element; the variant is what
105    /// says whether the element found is the one the declaration meant. A
106    /// declaration naming no entity finds none — which is not the same as a
107    /// missing one, and [`Self::missing_from`] is the question to ask about
108    /// presence.
109    pub(crate) fn captured<'f>(&self, flat: &'f Flat) -> Option<&'f Element> {
110        let element = flat.element(&self.entity_name()?)?;
111        matches!(
112            (self, element),
113            (Declaration::Function(_), Element::Function(_))
114                | (Declaration::Const(_), Element::Constant(_))
115                | (Declaration::Type(_), Element::Type(_))
116        )
117        .then_some(element)
118    }
119
120    /// Whether the model lacks what this declaration must name.
121    ///
122    /// Naming the wrong kind — `.fun(fun!(x))` where the source captured
123    /// `const x` — counts as missing: it is an error in the binding, and the
124    /// engine fails the run over it instead of reporting a skip. False for a
125    /// declaration that names no entity.
126    pub(crate) fn missing_from(&self, flat: &Flat) -> bool {
127        self.entity_name().is_some() && self.captured(flat).is_none()
128    }
129
130    /// The word a refusal uses for what this declaration must name.
131    pub(crate) fn describe_captured(&self) -> &'static str {
132        match self {
133            Declaration::Function(_) => "function",
134            Declaration::Const(_) => "constant",
135            Declaration::Type(_) => "type",
136            Declaration::Conversion(_)
137            | Declaration::Callback(_)
138            | Declaration::ComputedConst(_) => "item",
139        }
140    }
141
142    /// The name it goes by — the part of the printed form after the prefix,
143    /// and what an entity is looked up by.
144    pub(crate) fn name(&self) -> String {
145        match self {
146            Declaration::Function(ident) | Declaration::Const(ident) => ident.to_string(),
147            Declaration::Type(key) | Declaration::Conversion(key) => key.as_str().to_string(),
148            Declaration::Callback(name) | Declaration::ComputedConst(name) => name.clone(),
149        }
150    }
151}
152
153impl Ord for Declaration {
154    /// As they print. Two declarations of one name can print alike and still
155    /// be distinct — a captured `const X` and a computed `X` are both
156    /// `const:X` — so the variant breaks that tie, and `Ord` agrees with `Eq`.
157    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
158        fn variant(declaration: &Declaration) -> u8 {
159            match declaration {
160                Declaration::Function(_) => 0,
161                Declaration::Const(_) => 1,
162                Declaration::Type(_) => 2,
163                Declaration::Conversion(_) => 3,
164                Declaration::Callback(_) => 4,
165                Declaration::ComputedConst(_) => 5,
166            }
167        }
168        self.to_string()
169            .cmp(&other.to_string())
170            .then_with(|| variant(self).cmp(&variant(other)))
171    }
172}
173
174impl PartialOrd for Declaration {
175    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
176        Some(self.cmp(other))
177    }
178}
179
180impl std::fmt::Display for Declaration {
181    /// `<what the target gets>:<name>`. The prefix is the foreign side's word,
182    /// and it is what keeps the declarations' several naming spaces apart:
183    /// `type:Foo` and `conversion:Foo` are two declarations about one Rust
184    /// type.
185    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
186        let prefix = match self {
187            Declaration::Function(_) => "fn",
188            Declaration::Const(_) | Declaration::ComputedConst(_) => "const",
189            Declaration::Type(_) => "type",
190            Declaration::Conversion(_) => "conversion",
191            Declaration::Callback(_) => "callback",
192        };
193        write!(f, "{prefix}:{}", self.name())
194    }
195}

197impl Serialize for Declaration {
198    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
199        serializer.collect_str(self)
200    }
