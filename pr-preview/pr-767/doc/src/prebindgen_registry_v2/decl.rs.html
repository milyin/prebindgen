1//! What a binding declared, in terms neither language owns.
2//!
3//! A frontend turns its own declaration storage into these — one [`Declaration`]
4//! per thing the user asked for, inside the
5//! [`BindingRequests`](crate::BindingRequests) it hands the engine. The
6//! declaration is what the report accounts for, and what the target looks its
7//! own configuration up by — the request carries none: the foreign name and
8//! the declarator word the report prints come from that lookup.
9
10use prebindgen_flat::flat::{Element, Flat, TypeKey};
11use serde::Serialize;
12
13/// One thing a binding asked for: what the target gets, named by what the
14/// Rust source calls it, and what the engine plans it from.
15///
16/// A name alone says neither: `Sample` may be a captured type, a type key the
17/// binding coined for something the source never exported, or the name of a
18/// Kotlin constant built from an expression. The variant says which, and it
19/// says it once — what the target gets and where it comes from (a captured
20/// item, looked up in the model, or the binding itself,
21/// which [`Self::is_binding_local`] answers) both follow from the variant
22/// instead of being stated beside it, so they cannot disagree and a pair that
23/// means nothing (a callback backed by a captured constant, a function the
24/// binding both defines and selects out of the source) cannot be written
25/// down.
26///
27/// [`generate`](crate::generate) routes on this: each planner is reached by its
28/// own variants and is handed the captured item they name, rather than a kind
29/// to decode and a name to look up.
30///
31/// It is also the declaration's identity — what an outcome is keyed by, what a
32/// [`Position`](crate::target::Position) is rooted at, what a target's
33/// [`Requirement`](crate::target::Requirement) resolves to. Stable across runs
34/// and across pipelines, so a report, a build script and a capability-selected
35/// test section can all name the same declaration: it is what the *source*
36/// calls the thing, not what the target does, and a rename on the foreign side
37/// must not silently retire a test's requirement. `<kind>:<name>` —
38/// `type:Stamp`, `fn:stamp_sum` — is how it prints and how the report writes
39/// it, and that spelling is a rendering: nothing reads a declaration back out
40/// of it. Declarations order as they print, so a report sorted by declaration
41/// reads in id order.
42#[derive(Clone, Debug, PartialEq, Eq, Hash)]
43pub enum Declaration {
44    /// A captured `#[prebindgen]` function, exported as a foreign function.
45    Function(syn::Ident),
46    /// A foreign function the binding defines itself — `fun!(crate::x).sig(..)`
47    /// in the JNI frontend. Nothing in the captured source names it.
48    LocalFunction(String),
49    /// A captured `#[prebindgen]` constant, exposed as a foreign constant.
50    Const(syn::Ident),
51    /// A foreign constant read by calling a captured nullary function —
52    /// `constant!(X).fun(fun!(f))`. The target renders a constant; the engine
53    /// plans the function behind it.
54    ConstFromFunction(syn::Ident),
55    /// A foreign constant the binding computes rather than reads —
56    /// `constant!(X).expr(..)`. No captured item holds its value.
57    LocalConst(String),
58    /// A captured `#[prebindgen]` type, given a foreign representation.
59    Type(TypeKey),
60    /// A type the binding gives a representation although the source never
61    /// exported it — `String` crossing as an opaque handle.
62    LocalType(TypeKey),
63    /// A callback signature the binding exports as a foreign callable. No
64    /// captured item names one, so the signature is the name it goes by.
65    Callback(String),
66    /// A declared conversion between a Rust type and its wire form, defined by
67    /// the binding.
68    Conversion(TypeKey),
69}
70
71impl Declaration {
72    /// Whether this declares a type — captured or the binding's own.
73    pub fn is_type(&self) -> bool {
74        matches!(self, Declaration::Type(_) | Declaration::LocalType(_))
75    }
76
77    /// The captured element this declaration names, if the model holds it.
78    ///
79    /// Captured items live in one flat namespace holding functions, types and
80    /// constants, so the name alone finds any element; the variant is what says
81    /// whether the element found is the one the declaration meant. A constant
82    /// read through a function is looked up among the functions, which is what
83    /// separates [`Self::ConstFromFunction`] from [`Self::Const`]. A
84    /// binding-local declaration names no captured item and so finds none — which
85    /// is not the same as a missing one, and [`Self::missing_from`] is the
86    /// question to ask about presence.
87    pub(crate) fn captured<'f>(&self, flat: &'f Flat) -> Option<&'f Element> {
88        let element = flat.element(&self.name())?;
89        matches!(
90            (self, element),
91            (
92                Declaration::Function(_) | Declaration::ConstFromFunction(_),
93                Element::Function(_)
94            ) | (Declaration::Type(_), Element::Type(_))
95                | (Declaration::Const(_), Element::Constant(_))
96        )
97        .then_some(element)
98    }
99
100    /// Whether the binding defines this itself: a callback signature, a
101    /// binding-local conversion helper, a function or constant of its own, or a
102    /// type the target represents although the source never exported it
103    /// (`String` as an opaque handle). Such a declaration requires nothing of the
104    /// model.
105    pub fn is_binding_local(&self) -> bool {
106        matches!(
107            self,
108            Declaration::LocalFunction(_)
109                | Declaration::LocalConst(_)
110                | Declaration::LocalType(_)
111                | Declaration::Callback(_)
112                | Declaration::Conversion(_)
113        )
114    }
115
116    /// Whether the model lacks what this declaration must name.
117    ///
118    /// Naming the wrong kind — `.fun(fun!(x))` where the source captured
119    /// `const x` — counts as missing: it is an error in the binding, and the
120    /// engine fails the run over it instead of reporting a skip. False for a
121    /// binding-local declaration.
122    pub(crate) fn missing_from(&self, flat: &Flat) -> bool {
123        !self.is_binding_local() && self.captured(flat).is_none()
124    }
125
126    /// The word a refusal uses for what this declaration must name.
127    pub(crate) fn describe_captured(&self) -> &'static str {
128        match self {
129            Declaration::Function(_) | Declaration::ConstFromFunction(_) => "function",
130            Declaration::Const(_) => "constant",
131            Declaration::Type(_) => "type",
132            _ => "binding-local item",
133        }
134    }
135
136    /// The name it goes by — the part of the printed form after the prefix,
137    /// and what a captured item is looked up by.
138    pub(crate) fn name(&self) -> String {
139        match self {
140            Declaration::Function(ident)
141            | Declaration::Const(ident)
142            | Declaration::ConstFromFunction(ident) => ident.to_string(),
143            Declaration::Type(key) | Declaration::LocalType(key) | Declaration::Conversion(key) => {
144                key.as_str().to_string()
145            }
146            Declaration::LocalFunction(name)
147            | Declaration::LocalConst(name)
148            | Declaration::Callback(name) => name.clone(),
149        }
150    }
151}

153impl Ord for Declaration {
154    /// As they print. A captured and a binding-local declaration of one kind and
155    /// name print alike and are still distinct, so the variant breaks that
156    /// tie, and `Ord` agrees with `Eq`.
157    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
158        fn variant(declaration: &Declaration) -> u8 {
159            match declaration {
160                Declaration::Function(_) => 0,
161                Declaration::LocalFunction(_) => 1,
162                Declaration::Const(_) => 2,
163                Declaration::ConstFromFunction(_) => 3,
164                Declaration::LocalConst(_) => 4,
165                Declaration::Type(_) => 5,
166                Declaration::LocalType(_) => 6,
167                Declaration::Callback(_) => 7,
168                Declaration::Conversion(_) => 8,
169            }
170        }
171        self.to_string()
172            .cmp(&other.to_string())
173            .then_with(|| variant(self).cmp(&variant(other)))
174    }
175}
176
177impl PartialOrd for Declaration {
178    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
179        Some(self.cmp(other))
180    }
181}

183impl std::fmt::Display for Declaration {
184    /// `<what the target gets>:<name>`. The prefix is the foreign side's word —
185    /// a function the binding defines and a captured one are both a `fn` —
186    /// and it is what keeps the declarations' several naming spaces apart: `type:Foo`
187    /// and `conversion:Foo` are two declarations about one Rust type, and
188    /// `const:f` and `fn:f` are the `val` read through `f` and `f` itself.
189    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
190        let name = self.name();
191        match self {
192            Declaration::Function(_) | Declaration::LocalFunction(_) => write!(f, "fn:{name}"),
193            Declaration::Const(_)
194            | Declaration::ConstFromFunction(_)
195            | Declaration::LocalConst(_) => {
196                write!(f, "const:{name}")
197            }
198            Declaration::Type(_) | Declaration::LocalType(_) => write!(f, "type:{name}"),
199            Declaration::Callback(_) => write!(f, "callback:{name}"),
200            Declaration::Conversion(_) => write!(f, "conversion:{name}"),
201        }
202    }
203}

205impl Serialize for Declaration {
206    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
207        serializer.collect_str(self)
208    }
