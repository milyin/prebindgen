1//! What a binding declared, in terms neither language owns.
2//!
3//! A frontend turns its own declaration storage into these — one [`Declaration`]
4//! per thing the user asked for, inside the
5//! [`BindingRequests`](crate::BindingRequests) it hands the engine. The
6//! declaration is what the report accounts for, and what the target looks its
7//! own configuration up by — the request carries none: the foreign name and
8//! the declarator word the report prints come from that lookup.
9
10use prebindgen_flat::flat::{Element, EntityKind, Flat, TypeKey};
11use serde::Serialize;
12
13/// An entity by name: which of the three kinds, and what it is called.
14///
15/// What an ignore names, and what a declaration resolves to when it names an
16/// entity at all. Less than a [`Declaration`]: it says nothing about how the
17/// target would get the item, so it cannot be a callback or a constant the
18/// binding computes on the foreign side — those name no entity.
19#[derive(Clone, Debug, PartialEq, Eq, Hash)]
20pub enum EntityName {
21    Type(TypeKey),
22    Function(syn::Ident),
23    Constant(syn::Ident),
24}
25
26impl EntityName {
27    pub fn kind(&self) -> EntityKind {
28        match self {
29            EntityName::Type(_) => EntityKind::Type,
30            EntityName::Function(_) => EntityKind::Function,
31            EntityName::Constant(_) => EntityKind::Constant,
32        }
33    }
34
35    /// The name as the model indexes it.
36    pub fn name(&self) -> String {
37        match self {
38            EntityName::Type(key) => key.as_str().to_string(),
39            EntityName::Function(ident) | EntityName::Constant(ident) => ident.to_string(),
40        }
41    }
42}
43
44impl From<EntityName> for Declaration {
45    /// The declaration that exposes the entity as what it is — what a report
46    /// row and a duplicate check compare an ignore by.
47    fn from(name: EntityName) -> Self {
48        match name {
49            EntityName::Type(key) => Declaration::Type(key),
50            EntityName::Function(ident) => Declaration::Function(ident),
51            EntityName::Constant(ident) => Declaration::Const(ident),
52        }
53    }
54}

56/// One thing a binding asked for: what the target gets, named by what the
57/// Rust source calls it, and what the engine plans it from.
58///
59/// A name alone says neither: `f` may be a function exported as itself or a
60/// Kotlin `val` read through it, and `Foo` a type given a representation or a
61/// wire mapping declared for it. The variant says which, and it says it once
62/// — what the target gets and which entity, if any, it is planned from both
63/// follow from the variant instead of being stated beside it, so they cannot
64/// disagree and a pair that means nothing (a callback backed by a captured
65/// constant) cannot be written down.
66///
67/// Where an entity came from is not part of this. A `#[prebindgen]` function
68/// and one the binding defines itself are both a [`Declaration::Function`];
69/// the model holds both, and only its [`Origin`](prebindgen_flat::flat::Origin)
70/// tells them apart. Two declarations name no entity at all, because nothing
71/// in Rust backs them: a callback signature, and a constant the binding
72/// computes on the foreign side.
73///
74/// [`generate`](crate::generate) routes on this: each planner is reached by its
75/// own variants and is handed the entity they name, rather than a kind to
76/// decode and a name to look up.
77///
78/// It is also the declaration's identity — what an outcome is keyed by, what a
79/// [`Position`](crate::target::Position) is rooted at, what a target's
80/// [`Requirement`](crate::target::Requirement) resolves to. Stable across runs
81/// and across pipelines, so a report, a build script and a capability-selected
82/// test section can all name the same declaration: it is what the *source*
83/// calls the thing, not what the target does, and a rename on the foreign side
84/// must not silently retire a test's requirement. `<kind>:<name>` —
85/// `type:Stamp`, `fn:stamp_sum` — is how it prints and how the report writes
86/// it, and that spelling is a rendering: nothing reads a declaration back out
87/// of it. Declarations order as they print, so a report sorted by declaration
88/// reads in id order.
89#[derive(Clone, Debug, PartialEq, Eq, Hash)]
90pub enum Declaration {
91    /// A function, exported as a foreign function.
92    Function(syn::Ident),
93    /// A foreign constant read by calling a nullary function —
94    /// `constant!(X).fun(fun!(f))`. The target renders a constant; the engine
95    /// plans the function behind it.
96    ConstFromFunction(syn::Ident),
97    /// A constant, exposed as a foreign constant.
98    Const(syn::Ident),
99    /// A type, given a foreign representation.
100    Type(TypeKey),
101    /// A declared conversion between a type and its wire form, defined by the
102    /// binding.
103    Conversion(TypeKey),
104    /// A callback signature the binding exports as a foreign callable. No
105    /// entity backs one, so the signature is the name it goes by.
106    Callback(String),
107    /// A foreign constant the binding computes rather than reads —
108    /// `constant!(X).expr(..)`. No entity holds its value.
109    ComputedConst(String),
110}
111
112impl Declaration {
113    /// Whether this declares a type.
114    pub fn is_type(&self) -> bool {
115        matches!(self, Declaration::Type(_))
116    }
117
118    /// The entity this declaration names, if it names one.
119    ///
120    /// Not always the declaration's own kind: a constant read through a
121    /// function names a function, which is what separates
122    /// [`Self::ConstFromFunction`] from [`Self::Const`]. `None` for a callback
123    /// and a computed constant, which nothing in Rust backs.
124    pub fn entity(&self) -> Option<EntityName> {
125        match self {
126            Declaration::Function(ident) | Declaration::ConstFromFunction(ident) => {
127                Some(EntityName::Function(ident.clone()))
128            }
129            Declaration::Const(ident) => Some(EntityName::Constant(ident.clone())),
130            Declaration::Type(key) | Declaration::Conversion(key) => {
131                Some(EntityName::Type(key.clone()))
132            }
133            Declaration::Callback(_) | Declaration::ComputedConst(_) => None,
134        }
135    }
136
137    /// The element this declaration names, if the model holds it.
138    ///
139    /// Entities live in one flat namespace holding functions, types and
140    /// constants, so the name alone finds any element; the kind is what says
141    /// whether the element found is the one the declaration meant. A
142    /// declaration naming no entity finds none — which is not the same as a
143    /// missing one, and [`Self::missing_from`] is the question to ask about
144    /// presence.
145    pub(crate) fn captured<'f>(&self, flat: &'f Flat) -> Option<&'f Element> {
146        let wanted = self.entity()?;
147        let element = flat.element(&wanted.name())?;
148        (element.entity()?.kind() == wanted.kind()).then_some(element)
149    }
150
151    /// Whether the model lacks what this declaration must name.
152    ///
153    /// Naming the wrong kind — `.fun(fun!(x))` where the source captured
154    /// `const x` — counts as missing: it is an error in the binding, and the
155    /// engine fails the run over it instead of reporting a skip. False for a
156    /// declaration that names no entity.
157    pub(crate) fn missing_from(&self, flat: &Flat) -> bool {
158        self.entity().is_some() && self.captured(flat).is_none()
159    }
160
161    /// The word a refusal uses for what this declaration must name.
162    pub(crate) fn describe_captured(&self) -> &'static str {
163        match self.entity().map(|entity| entity.kind()) {
164            Some(EntityKind::Function) => "function",
165            Some(EntityKind::Constant) => "constant",
166            Some(EntityKind::Type) => "type",
167            None => "item",
168        }
169    }
170
171    /// The name it goes by — the part of the printed form after the prefix,
172    /// and what an entity is looked up by.
173    pub(crate) fn name(&self) -> String {
174        match self {
175            Declaration::Function(ident)
176            | Declaration::ConstFromFunction(ident)
177            | Declaration::Const(ident) => ident.to_string(),
178            Declaration::Type(key) | Declaration::Conversion(key) => key.as_str().to_string(),
179            Declaration::Callback(name) | Declaration::ComputedConst(name) => name.clone(),
180        }
181    }
182}

184impl Ord for Declaration {
185    /// As they print. Two declarations of one name print alike and are still
186    /// distinct — `const:f` read through `f` and a captured `const f` — so the
187    /// variant breaks that tie, and `Ord` agrees with `Eq`.
188    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
189        fn variant(declaration: &Declaration) -> u8 {
190            match declaration {
191                Declaration::Function(_) => 0,
192                Declaration::ConstFromFunction(_) => 1,
193                Declaration::Const(_) => 2,
194                Declaration::Type(_) => 3,
195                Declaration::Conversion(_) => 4,
196                Declaration::Callback(_) => 5,
197                Declaration::ComputedConst(_) => 6,
198            }
199        }
200        self.to_string()
201            .cmp(&other.to_string())
202            .then_with(|| variant(self).cmp(&variant(other)))
203    }
204}
205
206impl PartialOrd for Declaration {
207    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
208        Some(self.cmp(other))
209    }
210}

212impl std::fmt::Display for Declaration {
213    /// `<what the target gets>:<name>`. The prefix is the foreign side's word,
214    /// and it is what keeps the declarations' several naming spaces apart:
215    /// `type:Foo` and `conversion:Foo` are two declarations about one Rust
216    /// type, and `const:f` and `fn:f` are the `val` read through `f` and `f`
217    /// itself.
218    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
219        let prefix = match self {
220            Declaration::Function(_) => "fn",
221            Declaration::ConstFromFunction(_)
222            | Declaration::Const(_)
223            | Declaration::ComputedConst(_) => "const",
224            Declaration::Type(_) => "type",
225            Declaration::Conversion(_) => "conversion",
226            Declaration::Callback(_) => "callback",
227        };
228        write!(f, "{prefix}:{}", self.name())
229    }
230}

232impl Serialize for Declaration {
233    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
234        serializer.collect_str(self)
235    }
