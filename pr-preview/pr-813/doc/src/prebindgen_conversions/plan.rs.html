1use crate::template_parameter;
2use prebindgen_flat::{Wire, WireKind};
3use proc_macro2::TokenStream;
4use quote::quote;
5/// An input conversion split into call-wide validation, acquisition and construction.
6///
7/// Optional storage for the staged request protocol in this crate.
8/// Generators may use this protocol or implement their own request types.
9#[derive(Clone, Debug)]
10pub struct InputPlan<W: WireKind, C = (), M = ()> {
11    /// Context type supplied by the language generator at emission.
12    pub context: std::marker::PhantomData<fn(&C)>,
13    /// Flattened boundary parameters in signature order.
14    pub wires: Vec<Wire<W>>,
15    /// Occurrence-level composition metadata, separate from the flat ABI wires.
16    pub metadata: M,
17    /// Shared call acknowledgement, if this input needs one.
18    pub acknowledgement: Option<TokenStream>,
19    /// Validate and prepare every active leaf before any ownership transfer.
20    pub validation: TokenStream,
21    /// Infallible acquisition of all owned inputs, before constructors can run.
22    pub acquire: TokenStream,
23    /// Active foreign resources whose identities must be checked across all arguments.
24    pub resources: Vec<Resource>,
25    /// Argument construction after all acquisitions, including source constructor calls.
26    pub statements: TokenStream,
27    /// Prepared value expression assigned to the caller-supplied target after `statements`.
28    pub argument: TokenStream,
29}
30/// Output conversion whose owned temporaries survive until successful delivery.
31///
32/// Optional concrete storage implementing [`prebindgen_flat::Conversion`].
33/// Delivery and ownership commit are separate requests.
34#[derive(Clone, Debug)]
35pub struct OutputPlan<W: WireKind, C = (), M = ()> {
36    /// Context type supplied by the language generator at emission.
37    pub context: std::marker::PhantomData<fn(&C)>,
38    /// Delivery and return code prepared with this result slot.
39    pub delivery: Delivery,
40    /// Boundary results in delivery order; one entry per expression in `values`.
41    pub wires: Vec<Wire<W>>,
42    /// Occurrence-level composition metadata, separate from the flat ABI wires.
43    pub metadata: M,
44    /// Transfer ownership only after all foreign delivery operations succeed.
45    pub commit: TokenStream,
46    /// Code producing owned output temporaries and evaluating observations before moves.
47    pub statements: TokenStream,
48    /// Delivery expressions in wire order; they must remain valid until commit.
49    pub values: Vec<TokenStream>,
50}
51/// Access to one foreign resource during an exported call.
52#[derive(Clone, Copy, Debug, PartialEq, Eq)]
53pub enum Access {
54    /// Read-only borrow; multiple active shared accesses may alias.
55    Shared,
56    /// Mutable borrow; conflicts with every other active use of the resource.
57    Exclusive,
58    /// Ownership transfer; conflicts with every other active use of the resource.
59    Move,
60}
61/// The adapter supplies identity; flat enforces call-wide access conflicts.
62#[derive(Clone, Debug)]
63pub struct Resource {
64    /// Side-effect-free Rust expression identifying a live resource, such as an address.
65    /// All identities compared within a call must share an equality-compatible domain.
66    pub identity: TokenStream,
67    /// Borrow or ownership requirement imposed by this input conversion.
68    pub access: Access,
69    /// Side-effect-free Rust boolean expression; false resources do not conflict.
70    pub active: TokenStream,
71}
72/// Result-slot glue executed after output preparation and before ownership commit.
73#[derive(Clone, Debug)]
74pub struct Delivery {
75    /// Deliver every output; failure must leave owned temporaries uncommitted.
76    pub statements: TokenStream,
77    /// Infallible wrapper result expression evaluated after commit.
78    pub result: TokenStream,
79}
80impl Default for Delivery {
81    fn default() -> Self {
82        Self {
83            statements: TokenStream::new(),
84            result: quote!(()),
85        }
86    }
87}
