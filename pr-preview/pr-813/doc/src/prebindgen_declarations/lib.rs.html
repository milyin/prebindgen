1//! Declaration language and composition for preparing conversion slots.
2//!
3//! [`Declarations::fill`], [`bind_input`] and [`bind_output`] build conversions
4//! directly in a [`Bindings`] model. Recipes and codec input data are used only
5//! during construction; slots retain wires and code needed by wrapper writers.
6//! Generation is owned by the language generator;
7//! this crate owns only declaration resolution and slot preparation.
8//! Each [`Wire`] carries the adapter's own description in [`Wire::kind`].
9//! Reusable value and boxed-allocation leaves live in `prebindgen-conversions`.
10#![warn(missing_docs)]

30/// Input preparation plan, including the codec's composition metadata.
31pub type CodecInput<C> = InputPlan<

33    <C as Codec>::Context,

43/// A recipe for converting boundary inputs into one source argument.
44///
45/// Binding expands the recipe into wires and Rust conversion code.
46#[derive(Clone, Debug)]
47pub enum Input {
48    /// Convert an optional payload, gating every conversion phase.
49    Optional(Box<Input>),
50    /// Convert the occurrence type itself with [`Codec::input_leaf`].
51    Leaf,
52    /// Construct a named record from children in field order; an empty list builds unit.
53    Fields(Vec<Input>),
54    /// Call a source function after preparing its arguments with the child recipes.
55    Constructor {
56        /// Unqualified source function name; its result must match the occurrence type.
57        function: String,
58        /// Recipes in the constructor's parameter order.
59        args: Vec<Input>,
60    },
61    /// Choose an owned input using an `i32` arm index followed by all arms' wires.
62    /// Inactive arms are neither validated nor acquired; indices are zero-based.
63    Selector(Vec<Input>),
64}
65impl Input {
66    /// Construct a record from field recipes, or unit from an empty list.
67    ///
68    /// `fields` holds one recipe per field of the source record, in declaration
69    /// order. Binding verifies the count against the record.
70    pub fn fields(fields: impl IntoIterator<Item = Input>) -> Self {
71        Self::Fields(fields.into_iter().collect())
72    }
73    /// Call a named source constructor with the supplied argument recipes.
74    ///
75    /// Binding checks its result type and argument count. Argument validation and
76    /// ownership acquisition precede the source call.
77    ///
78    /// # Arguments
79    ///
80    /// * `name` — unqualified name of a source function whose result type is the
81    ///   input's type.
82    /// * `args` — one recipe per parameter of that function, in order.
83    pub fn constructor(name: impl Into<String>, args: impl IntoIterator<Item = Input>) -> Self {
84        Self::Constructor {
85            function: name.into(),
86            args: args.into_iter().collect(),
87        }
88    }
89    /// Choose between owned representations of the same input type.
90    ///
91    /// `arms` holds one recipe per alternative; each must produce the input's
92    /// type. The caller passes the zero-based index of the chosen alternative in an
93    /// extra wire ahead of all arms' wires.
94    ///
95    /// Requires at least one arm. Each arm retains its own wires; this does not
96    /// merge equivalent wire parameters or generate destination-language overloads.
97    pub fn selector(arms: impl IntoIterator<Item = Input>) -> Self {
98        Self::Selector(arms.into_iter().collect())
99    }
100}
101/// A result conversion recipe, interpreted together with its source type and codec input data.
102///
103/// `Fields` observes fields; `OwnedFields` moves them. Explicit `Projected`
104/// leaves select borrowed or owned conversion independently of type contents.
105#[derive(Clone, Debug)]
106pub enum Output {
107    /// Convert an optional payload without preparing absent values.
108    Optional(Box<Output>),
109    /// Convert the occurrence type itself with [`Codec::output_leaf`].
110    Leaf,
111    /// Observe named fields in declaration order; an empty list represents unit.
112    Fields(Vec<Output>),
113    /// Move named fields into their output conversions.
114    OwnedFields(Vec<Output>),
115    /// Convert a leaf with an explicit ownership policy.
116    Projected(Projection),
117    /// Call a consuming accessor, then convert its owned result.
118    Into {
119        /// Source accessor taking exactly the owned occurrence type.
120        function: String,
121        /// Conversion of its owned result.
122        output: Box<Output>,
123    },
124    /// A shared-borrow accessor, followed by the output conversion of its result.
125    Getter {
126        /// Source getter taking exactly one shared reference to the current type.
127        function: String,
128        /// Conversion of the getter's owned result; borrowed results are unsupported.
129        output: Box<Output>,
130    },
131    /// Produce several views of one value. At most one child may consume it.
132    /// Observations run before consumption, while wire order follows this list.
133    Combine(Vec<Output>),
134}
135impl Output {
136    /// Observe record fields in declaration order; use an empty list for unit.
137    ///
138    /// `fields` holds one recipe per field of the source record, in declaration order.
139    pub fn fields(fields: impl IntoIterator<Item = Output>) -> Self {
140        Self::Fields(fields.into_iter().collect())
141    }
142    /// Observe a value through a source getter and convert the getter's result.
143    ///
144    /// The getter must take exactly `&T` for the current type `T`, and return an
145    /// owned value. Binding rejects mismatched signatures.
146    ///
147    /// # Arguments
148    ///
149    /// * `name` — unqualified name of the source getter.
150    /// * `output` — recipe for the getter's result.
151    pub fn getter(name: impl Into<String>, output: Output) -> Self {
152        Self::Getter {
153            function: name.into(),
154            output: Box::new(output),
155        }
156    }
157    /// Produce multiple outputs from one value.
158    ///
159    /// `parts` holds one recipe per view of the value; their wires appear in this
160    /// order. Binding rejects an empty list or more than one consuming part. The
161    /// framework uses [`Codec::consumes`] to order observations before a move.
162    pub fn combine(parts: impl IntoIterator<Item = Output>) -> Self {
163        Self::Combine(parts.into_iter().collect())
164    }
165    fn consumes<C: Codec>(&self, ty: &Type, extra: &C::Extra, codec: &C) -> bool {
166        match self {
167            Self::Optional(child) => child.consumes(option_inner(ty).unwrap_or(ty), extra, codec),
168            Self::Leaf => codec.consumes(ty, extra),
169            Self::Projected(p) => p.mode == ProjectionMode::Owned,
170            Self::OwnedFields(_) | Self::Into { .. } => true,
171            Self::Combine(p) => p.iter().any(|p| p.consumes(ty, extra, codec)),
172            _ => false,
173        }
174    }
175}

177/// Language-specific leaf conversions. Recursive composition stays in this declaration compiler.
178///
179/// A leaf is a source type the adapter converts itself: a value copied across
180/// the boundary or an object kept in Rust. `prebindgen-conversions` provides helpers
181/// for both representations. The
182/// adapter chooses by type; a recipe only says where the leaves are.
183///
184/// Input leaves must validate without consuming in `validation`, then
185/// acquire ownership without failure or source calls in `acquire`. Each acquired
186/// value needs an ordinary Rust owner so later failures drop it. `statements`
187/// can construct arguments from those owners. Every input wire must retain
188/// the supplied activation condition. Output leaves keep ownership in Rust
189/// locals until their infallible `commit` runs after successful delivery.
190///
191/// # Implementing a codec
192///
193/// This minimal codec uses Rust values as its carriers. It illustrates converter
194/// construction without a foreign ABI or object storage. A language adapter
195/// additionally writes wrappers and chooses native and destination-language
196/// carrier types for its wire kinds.
197///
198/// ```
199/// use prebindgen_flat::{Result, validate_bindings};
200/// use prebindgen_conversions::value;
201/// use prebindgen_declarations::{
202///     Codec, CodecInput, CodecOutput, Declarations,
203///     Bindings, fun, expand_param, expand_return, bind_input, Input,
204/// };
205/// use proc_macro2::TokenStream;
206/// use quote::quote;
207/// use syn::{Ident, Type};
208///
209/// use prebindgen_declarations::Condition;
210/// fn carrier(kind: value::Value) -> Type {
211///     match kind {
212///         value::Value::Scalar(p) => p.rust_type(),
213///         value::Value::Text => syn::parse_quote!(String),
214///         value::Value::Bytes => syn::parse_quote!(Vec<u8>),
215///     }
216/// }
217/// struct RustCarriers;
218/// impl Codec for RustCarriers {
219///     type Niche = ();
220///     type Group = ();
221///     type Extra = ();
222///     type Context = ();
223///     type Input = CodecInput<Self>;
224///     type Output = CodecOutput<Self>;
225///     type Wire = Type; // The Rust carrier type is the complete wire description.
226///
227///     fn selector(&self, _: &()) -> Self::Wire { syn::parse_quote!(i32) }
228///     fn consumes(&self, _: &Type, _: &()) -> bool { false }
229///     fn input_leaf(
230///         &self, ty: &Type, _: &(), wire: Ident, var: Ident, active: &Condition,
231///     ) -> Result<CodecInput<Self>> {
232///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
233///         let decode = quote!(#wire.clone());
234///         value::input(ty, wire, var, &active.rust(), decode, carrier(kind))
235///     }
236///     fn output_leaf(
237///         &self, ty: &Type, _: &(), wire: Ident, v: TokenStream, borrowed: bool,
238///     ) -> Result<CodecOutput<Self>> {
239///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
240///         value::output(ty, wire, v, borrowed, |v| v, carrier(kind))
241///     }
242/// }
243///
244/// let source = syn::parse_file(r#"
245///     pub fn from_i32(value: i32) -> i64 { value as i64 }
246///     pub fn to_i32(value: &i64) -> i32 { *value as i32 }
247///     pub fn echo(value: i64) -> i64 { value }
248/// "#)?;
249/// let mut flat = Bindings::<Type>::parse(
250///     source.items.into_iter().map(|item| (item, Default::default())),
251/// )?;
252/// let defaults = Declarations::new(&flat, vec![
253///     expand_param!(i64).variant(fun!(from_i32)).variant_self().into(),
254///     expand_return!(i64).field(fun!(to_i32)).field_self().into(),
255/// ])?;
256/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
257/// let function = flat.function("echo")?;
258/// // Input: selector, constructor argument, direct value. Output: two values.
259/// assert_eq!(function.params[0].ty.slot.as_ref().unwrap().wires.len(), 3);
260/// assert_eq!(function.ret.slot.as_ref().unwrap().wires.len(), 2);
261/// // Replace just this parameter's converter; later default filling preserves it.
262/// bind_input(&mut flat, &RustCarriers, "echo", "value", &Input::Leaf, &())?;
263/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
264/// validate_bindings(&flat)?;
265/// let input = flat.function("echo")?.params[0].ty.slot.as_ref().unwrap();
266/// assert_eq!(input.wires.len(), 1);
267/// assert_eq!(prebindgen_flat::type_key(&input.wires[0].kind), "i64");
268/// assert!(input.resources.is_empty()); // No foreign ownership in this codec.
269/// # Ok::<(), Box<dyn std::error::Error>>(())
270/// ```
271pub trait Codec {
272    /// Adapter-owned description of an absent wire representation. Use `()` if unused.
273    /// The shared compiler never interprets this metadata.
274    type Niche: Clone;
275    /// Adapter-owned group metadata (for example a destination record shape).
276    type Group: Clone;
277
278    /// Input data borrowed while building one occurrence's conversion; use `()` if unused.
279    /// The same value is passed through nested conversions. It is not stored in Flat;
280    /// hooks retain only the data needed for emission in their code and wires.
281    type Extra;
282    /// Context supplied by the language generator when emitting stored templates.
283    type Context: prebindgen_conversions::TemplateContext + 'static;
284    /// Final self-contained input converter installed after preparation.
285    type Input: prebindgen_flat::Conversion + From<CodecInput<Self>>;
286    /// Final self-contained output converter installed after preparation.
287    type Output: prebindgen_flat::Conversion + From<CodecOutput<Self>>;
288    /// Preserve a logical path in private preparation metadata when the codec needs it.
289    /// The final converter decides how to use this mapping; Flat never interprets it.
290    /// Called from the innermost component outward; prepend each segment to its path.
291    fn prefix_component(&self, _wire: &mut Self::Wire, _segment: ComponentSegment) {}
292
293    /// Private preparation metadata for a leaf wire. Final slot conversion may
294    /// replace it with a transport-only description before exposing wires to a writer.
295    type Wire: prebindgen_flat::WireKind + Clone + std::fmt::Debug;
296    /// Describe the `i32` arm index wire that the declaration compiler creates for [`Input::Selector`].
297    ///
298    /// `extra` is the occurrence data of the conversion that contains the selector.
299    fn selector(&self, extra: &Self::Extra) -> Self::Wire;
300    /// Whether a leaf output consumes the current source value.
301    ///
302    /// This must agree with [`Self::output_leaf`]. It governs move ordering and
303    /// double-consumption rejection in [`Output::Combine`], not input ownership.
304    /// A non-consuming leaf receives a shared reference with `borrowed` set.
305    ///
306    /// # Arguments
307    ///
308    /// * `ty` — owned source type of the value being delivered.
309    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`].
310    fn consumes(&self, ty: &Type, extra: &Self::Extra) -> bool;
311    /// Prepare a leaf input of source type `ty` using occurrence data `extra`.
312    ///
313    /// All fallible checks belong in [`InputPlan::validation`]; do not
314    /// consume values there. Return a generation error when this type or access
315    /// mode is unsupported.
316    ///
317    /// # Arguments
318    ///
319    /// * `ty` — the source parameter type at this position, as written: `T`,
320    ///   `&T` or `&mut T`.
321    /// * `extra` — occurrence data of the conversion, as passed to [`bind_input`]
322    ///   or [`Declarations::fill`].
323    /// * `wire` — fresh identifier for the boundary parameter this leaf adds.
324    ///   A leaf that needs more wires or locals derives their names from it.
325    /// * `var` — fresh identifier for the Rust local that holds the converted value.
326    ///   [`InputPlan::argument`] normally refers to it.
327    /// * `active` — the selector arms under which this leaf is used. Copy it to every
328    ///   returned wire and resource, and guard validation and acquisition with it.
329    fn input_leaf(
330        &self,
331        ty: &Type,
332        extra: &Self::Extra,
333        wire: syn::Ident,
334        var: syn::Ident,
335        active: &Condition,
336    ) -> Result<CodecInput<Self>>;
337    /// Convert a leaf with explicit ownership, independently of its usual policy.
338    ///
339    /// The compiler supplies either an owned value or a shared reference according
340    /// to `projection.mode`. The default supports leaves without field overrides.
341    fn output_projected(
342        &self,
343        ty: &Type,
344        extra: &Self::Extra,
345        wire: syn::Ident,
346        value: TokenStream,
347        projection: &Projection,
348    ) -> Result<CodecOutput<Self>> {
349        if !projection.fields.is_empty() {
350            return Err("this output converter does not support field projection overrides".into());
351        }
352        let mut payload = ty;
353        while let Some(inner) = option_inner(payload) {
354            payload = inner;
355        }
356        if projection.mode == ProjectionMode::Owned && matches!(payload, Type::Reference(_)) {
357            return Err("owned projection cannot consume a reference payload".into());
358        }
359        self.output_leaf(
360            ty,
361            extra,
362            wire,
363            value,
364            projection.mode == ProjectionMode::Borrowed,
365        )
366    }
367    /// An absent representation excluded from the raw wire type's valid domain.
368    /// Never advertise representations accepted as present leaf payloads. The
369    /// metadata may describe a single value or a disjoint class of values.
370    /// Plan domain overrides take precedence: the shared compiler tracks niche
371    /// reservations by enclosing options and suppresses conditionally active wires.
372    fn niche(&self, _wire: &Self::Wire) -> Option<Self::Niche> {
373        None
374    }
375    /// Recognize presence without decoding the payload. Called on original inner metadata.
376    /// The expression must be total and side-effect-free for every initialized wire
377    /// value, including inactive placeholders; it must not dereference raw pointers.
378    /// It returns false for every representation reserved by this niche.
379    fn is_present(
380        &self,
381        _wire: &Self::Wire,
382        _niche: &Self::Niche,
383        _value: TokenStream,
384    ) -> Result<TokenStream> {
385        Err("niche presence testing is unsupported by this codec".into())
386    }
387    /// Remaining excluded representation after reserving `niche` for one optional layer.
388    /// The compiler stores the result on the plan, never on the wire type. Returning
389    /// `None` exhausts this domain; enclosing options then use another wire or a flag.
390    fn remaining_niche(&self, _wire: &Self::Wire, _niche: &Self::Niche) -> Option<Self::Niche> {
391        None
392    }
393    /// Inert wire expression used for an absent payload, optionally using a niche.
394    fn absent_value(
395        &self,
396        _wire: &Self::Wire,
397        _niche: Option<&Self::Niche>,
398        _input: bool,
399    ) -> Result<TokenStream> {
400        Err("optional values are unsupported by this codec".into())
401    }
402    /// Complete a composed parameter plan before it is stored in a slot.
403    /// Called once per bound parameter, including records and selectors.
404    fn finish_input(&self, _plan: &mut CodecInput<Self>, _extra: &Self::Extra) -> Result<()> {
405        Ok(())
406    }
407    /// Complete a composed result plan, including unit and multi-wire results.
408    /// Install call-level delivery here, after all output expressions are known.
409    fn finish_output(&self, _plan: &mut CodecOutput<Self>, _extra: &Self::Extra) -> Result<()> {
410        Ok(())
411    }
412    /// Prepare delivery of the Rust expression `value` as one or more wires.
413    ///
414    /// Keep unpublished allocations under Rust ownership and transfer them only in
415    /// [`OutputPlan::commit`]. Return a generation error for unsupported
416    /// conversions. Install final delivery in [`Codec::finish_output`], rather
417    /// than on nested leaf plans; recursive composition rejects leaf delivery.
418    ///
419    /// # Arguments
420    ///
421    /// * `ty` — the owned source type of the value.
422    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`]
423    ///   or [`Declarations::fill`].
424    /// * `wire` — fresh identifier for the delivered value. A leaf that needs more
425    ///   wires or locals derives their names from it.
426    /// * `value` — Rust expression producing the value to deliver.
427    /// * `borrowed` — `true` when `value` is a shared reference to `ty`, such as a
428    ///   record field observed in place; do not move out of it. `false` when
429    ///   `value` is owned.
430    fn output_leaf(
431        &self,
432        ty: &Type,
433        extra: &Self::Extra,
434        wire: syn::Ident,
435        value: TokenStream,
436        borrowed: bool,
437    ) -> Result<CodecOutput<Self>>;
438}
439/// A flat model whose slots contain only conversion data for wrapper emission.
440///
441/// Unbound source helpers have `None` slots. Bound functions need `Some` for their
442/// result and every parameter; [`prebindgen_flat::validate_bindings`] checks this before emission.
443/// Record fields retain only names and source types. Recipes and [`Codec::Extra`]
444/// are construction inputs, not stored in this model.
445///
446/// Output conversion data cannot be attached to an input position:
447/// ```compile_fail
448/// use prebindgen_declarations::Bindings;
449/// fn swap(flat: &mut Bindings<syn::Type>) {
450///     let function = flat.function_mut("foo").unwrap();
451///     function.params[0].ty.slot = function.ret.slot.clone();
452/// }
453/// ```
454/// Nor can a recipe be stored in a slot:
455/// ```compile_fail
456/// use prebindgen_declarations::{Bindings, Input};
457/// fn configure(flat: &mut Bindings<syn::Type>) {
458///     flat.function_mut("foo").unwrap().params[0].ty.slot = Some(Input::Leaf);
459/// }
460/// ```
461pub type Bindings<W, C = (), N = (), G = ()> =
462    Flat<InputPlan<W, C, Composition<N, G>>, OutputPlan<W, C, Composition<N, G>>>;
463
464/// Build and attach a converter to a named function parameter.
465///
466/// Resolves the occurrence's source type from `flat`, checks recipe compatibility,
467/// and recursively composes wires and code. Nested conversions use the supplied
468/// `extra`, not data from referenced constructors or getters. Recipes and `extra`
469/// are borrowed only for this call. A failure leaves the existing slot unchanged.
470/// Rebinding an occurrence replaces its converter without changing other slots.
471/// Generated names are scoped by parameter position, so separate calls cannot
472/// collide within the wrapper. Keep the function signature fixed after binding.
473/// Check completeness with [`prebindgen_flat::validate_bindings`].
474///
475/// # Arguments
476///
477/// * `flat` — the model holding the function; the parameter's slot is replaced.
478/// * `codec` — the adapter's leaf conversions.
479/// * `function` — unqualified name of the source function.
480/// * `parameter` — name of the parameter, as written in the source.
481/// * `recipe` — how to build the argument from boundary inputs.
482/// * `extra` — occurrence data passed to every codec call for this conversion.
483///
484/// Input binding rejects output recipes:
485/// ```compile_fail
486/// use prebindgen_declarations::{bind_input, Codec, Output};
487/// use prebindgen_flat::Flat;
488/// fn wrong<C: Codec>(flat: &mut Flat<C::Input, C::Output>, codec: &C, extra: &C::Extra) {
489///     bind_input(flat, codec, "foo", "x", &Output::Leaf, extra).unwrap();
490/// }
491/// ```
492pub fn bind_input<C: Codec>(
493    flat: &mut Flat<C::Input, C::Output>,
494    codec: &C,
495    function: &str,
496    parameter: &str,
497    recipe: &Input,
498    extra: &C::Extra,
499) -> Result<()> {
500    let f = flat.function(function)?;
501    let index = f
502        .params
503        .iter()
504        .position(|p| p.name == parameter)
505        .ok_or_else(|| format!("`{function}`: unknown parameter `{parameter}`"))?;
506    let mut conversion = Compiler::new(flat, codec, format!("p{index}")).input(
507        &f.params[index].ty.ty,
508        recipe,
509        extra,
510        &Condition::default(),
511    )?;
512    codec.finish_input(&mut conversion, extra)?;
513    conversion.metadata.validate(&conversion.wires)?;
514    flat.function_mut(function)?.params[index].ty.slot = Some(conversion.into());
515    Ok(())
516}

518/// Build and attach a converter to a function result, replacing any existing one.
519///
520/// Uses the same construction and failure rules as [`bind_input`]. Output code
521/// uses [`prebindgen_conversions::source_result_expr`] for the source call's result local.
522/// The recipe is checked for compatible types and signatures, and for repeated
523/// consumption of the same value. Unit results use `Output::fields([])`.
524///
525/// # Arguments
526///
527/// * `flat` — the model holding the function; the result's slot is replaced.
528/// * `codec` — the adapter's leaf conversions.
529/// * `function` — unqualified name of the source function.
530/// * `recipe` — how to deliver the result as boundary outputs.
531/// * `extra` — occurrence data passed to every codec call for this conversion.
532///
533/// Output binding rejects input recipes:
534/// ```compile_fail
535/// use prebindgen_declarations::{bind_output, Codec, Input};
536/// use prebindgen_flat::Flat;
537/// fn wrong<C: Codec>(flat: &mut Flat<C::Input, C::Output>, codec: &C, extra: &C::Extra) {
538///     bind_output(flat, codec, "foo", &Input::Leaf, extra).unwrap();
539/// }
540/// ```
541pub fn bind_output<C: Codec>(
542    flat: &mut Flat<C::Input, C::Output>,
543    codec: &C,
544    function: &str,
545    recipe: &Output,
546    extra: &C::Extra,
547) -> Result<()> {
548    let mut conversion = Compiler::new(flat, codec, "ret".into()).output(
549        &flat.function(function)?.ret.ty,
550        recipe,
551        extra,
552        prebindgen_conversions::source_result_expr(),
553        false,
554    )?;
555    codec.finish_output(&mut conversion, extra)?;
556    conversion.metadata.validate(&conversion.wires)?;
557    flat.function_mut(function)?.ret.slot = Some(conversion.into());
558    Ok(())
559}

561struct Compiler<'a, C: Codec> {
562    flat: &'a Flat<C::Input, C::Output>,
563    codec: &'a C,
564    prefix: String,
565    next: std::cell::Cell<usize>,
566}
567impl<'a, C: Codec> Compiler<'a, C> {
568    fn new(flat: &'a Flat<C::Input, C::Output>, codec: &'a C, prefix: String) -> Self {
569        Self {
570            flat,
571            codec,
572            prefix,
573            next: std::cell::Cell::new(0),
574        }
575    }
576    fn name(&self) -> syn::Ident {
577        let next = self.next.get();
578        let n = format_ident!("{}_w{next}", self.prefix);
579        self.next.set(next + 1);
580        n
581    }
582    fn prefix(&self, wires: &mut [Wire<C::Wire>], segment: ComponentSegment) {
583        for wire in wires {
584            self.codec.prefix_component(&mut wire.kind, segment.clone());
585        }
586    }
587    fn input(
588        &self,
589        ty: &Type,
590        recipe: &Input,
591        extra: &C::Extra,
592        active: &Condition,
593    ) -> Result<CodecInput<C>> {
594        let var = self.name();
595        let mut metadata = Composition::default();
596        let mut wires = Vec::new();
597        let mut validation = TokenStream::new();
598        let mut acquire = TokenStream::new();
599        let mut resources = Vec::new();
600        let mut acknowledgement = None;
601        let mut statements = TokenStream::new();
602        let argument = quote!(#var);
603        match recipe {
604            Input::Optional(inner) => {
605                let ty = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
606                let tag = self.name();
607                let child = self.input(ty, inner, extra, &active.arm(tag.clone(), 1))?;
608                return optional_input(self.codec, extra, tag, var, active, child);
609            }
610            Input::Leaf => {
611                let w = self.name();
612                return self.codec.input_leaf(ty, extra, w, var, active);
613            }
614            Input::Fields(children) => {
615                if type_key(ty) == "()" && children.is_empty() {
616                    return Ok(InputPlan {
617                        metadata,
618                        context: Default::default(),
619                        acknowledgement,
620                        wires,
621                        validation,
622                        acquire,
623                        resources,
624                        statements,
625                        argument: quote!(()),
626                    });
627                }
628                let r = self.flat.record(ty)?;
629                if r.fields.len() != children.len() {
630                    return Err(format!(
631                        "`{}`: expected {} field converters, got {}",
632                        type_key(ty),
633                        r.fields.len(),
634                        children.len()
635                    ));
636                }
637                return record_input(
638                    self.codec,
639                    ty,
640                    var,
641                    r.fields.iter().zip(children).map(|(field, child)| {
642                        Ok((
643                            field.name.clone(),
644                            self.input(&field.ty, child, extra, active)?,
645                        ))
646                    }),
647                );
648            }
649            Input::Constructor { function, args } => {
650                let f = self.flat.function(function)?;
651                if !same_type(&f.ret.ty, ty) {
652                    return Err(format!(
653                        "constructor `{function}` returns `{}`, expected `{}`",
654                        type_key(&f.ret.ty),
655                        type_key(ty)
656                    ));
657                }
658                if f.params.len() != args.len() {
659                    return Err(format!("constructor `{function}`: wrong argument count"));
660                }
661                let mut call_args = Vec::new();
662                for (param, child) in f.params.iter().zip(args) {
663                    let mut p = self.input(&param.ty.ty, child, extra, active)?;
664                    self.prefix(
665                        &mut p.wires,
666                        ComponentSegment::Argument(param.name.to_string()),
667                    );
668                    metadata.merge(p.metadata)?;
669                    validation.extend(p.validation);
670                    acquire.extend(p.acquire);
671                    resources.extend(p.resources);
672                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
673                    statements.extend(p.statements);
674                    wires.extend(p.wires);
675                    call_args.push(p.argument);
676                }
677                let path = &f.path;
678                statements.extend(quote!(let #var:#ty=#path(#(#call_args),*);));
679            }
680            Input::Selector(arms) => {
681                if arms.is_empty() {
682                    return Err("an input selector needs at least one alternative".into());
683                }
684                if matches!(ty, Type::Reference(_)) {
685                    return Err("selector results must be owned in this subset".into());
686                }
687                let tag = self.name();
688                wires.push(Wire {
689                    name: tag.clone(),
690                    kind: {
691                        let mut kind = self.codec.selector(extra);
692                        self.codec
693                            .prefix_component(&mut kind, ComponentSegment::Selector);
694                        kind
695                    },
696                });
697                let count = arms.len() as i32;
698                let invalid = active.and(quote!(!(0..#count).contains(&#tag)));
699                validation.extend(quote!(if #invalid {return Err("invalid selector".into());}));
700                let mut cases = Vec::new();
701                for (i, arm) in arms.iter().enumerate() {
702                    let mut p = self.input(ty, arm, extra, &active.arm(tag.clone(), i as i32))?;
703                    self.prefix(&mut p.wires, ComponentSegment::Alternative(i));
704                    p.metadata.suppress(&p.wires);
705                    metadata.merge(p.metadata)?;
706                    validation.extend(p.validation);
707                    acquire.extend(p.acquire);
708                    resources.extend(p.resources);
709                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
710                    let body = p.statements;
711                    let value = p.argument;
712                    wires.extend(p.wires);
713                    let i = i as i32;
714                    cases.push(quote!(#i=>{#body #value}));
715                }
716                statements = quote!(let #var:#ty=match #tag{#(#cases,)* _=>unreachable!("validated selector")};);
717            }
718        }
719        Ok(InputPlan {
720            metadata,
721            context: Default::default(),
722            acknowledgement,
723            wires,
724            validation,
725            acquire,
726            resources,
727            statements,
728            argument,
729        })
730    }
731    fn output(
732        &self,
733        ty: &Type,
734        recipe: &Output,
735        extra: &C::Extra,
736        value: TokenStream,
737        borrowed: bool,
738    ) -> Result<CodecOutput<C>> {
739        let mut out = OutputPlan {
740            metadata: Default::default(),
741            context: Default::default(),
742            delivery: Default::default(),
743            commit: TokenStream::new(),
744            wires: Vec::new(),
745            statements: TokenStream::new(),
746            values: Vec::new(),
747        };
748        match recipe {
749            Output::Optional(inner) => {
750                let ty = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
751                let tag = self.name();
752                let payload = self.name();
753                let child = self.output(ty, inner, extra, quote!(#payload), borrowed)?;
754                return optional_output(self.codec, extra, tag, payload, value, borrowed, child);
755            }
756            Output::Leaf => {
757                let w = self.name();
758                return self.codec.output_leaf(ty, extra, w, value, borrowed);
759            }
760            Output::Projected(projection) => {
761                if borrowed && projection.mode == ProjectionMode::Owned {
762                    return Err(
763                        "owned projection cannot consume a shared-reference occurrence".into(),
764                    );
765                }
766                let value = if !borrowed && projection.mode == ProjectionMode::Borrowed {
767                    quote!(&(#value))
768                } else {
769                    value
770                };
771                return self
772                    .codec
773                    .output_projected(ty, extra, self.name(), value, projection);
774            }
775            Output::Fields(children) | Output::OwnedFields(children) => {
776                let owned = matches!(recipe, Output::OwnedFields(_));
777                if owned && borrowed {
778                    return Err("owned fields cannot consume a shared-reference occurrence".into());
779                }
780                if type_key(ty) == "()" && children.is_empty() {
781                    return Ok(out);
782                }
783                let r = self.flat.record(ty)?;
784                if r.fields.len() != children.len() {
785                    return Err("wrong output field count".into());
786                }
787                let convert = |index: usize, value| {
788                    self.output(&r.fields[index].ty, &children[index], extra, value, !owned)
789                };
790                return if owned {
791                    record_output_owned(
792                        self.codec,
793                        self.name(),
794                        value,
795                        r.fields.iter().map(|f| f.name.clone()),
796                        convert,
797                    )
798                } else {
799                    record_output(
800                        self.codec,
801                        self.name(),
802                        value,
803                        r.fields.iter().map(|f| f.name.clone()),
804                        convert,
805                    )
806                };
807            }
808            Output::Getter { function, output } | Output::Into { function, output } => {
809                let consuming = matches!(recipe, Output::Into { .. });
810                if consuming && borrowed {
811                    return Err(
812                        "consuming accessor cannot consume a shared-reference occurrence".into(),
813                    );
814                }
815                let f = self.flat.function(function)?;
816                let expected: Type = if consuming {
817                    ty.clone()
818                } else {
819                    syn::parse_quote!(&#ty)
820                };
821                if f.params.len() != 1 || !same_type(&f.params[0].ty.ty, &expected) {
822                    return Err(format!(
823                        "getter `{function}` must take exactly `{}`",
824                        type_key(&expected)
825                    ));
826                }
827                if matches!(f.ret.ty, Type::Reference(_)) {
828                    return Err("borrowed getter results are outside this subset".into());
829                }
830                let v = self.name();
831                let path = &f.path;
832                let arg = if borrowed || consuming {
833                    value
834                } else {
835                    quote!(&#value)
836                };
837                out.statements = quote!(let #v=#path(#arg););
838                let mut child = self.output(&f.ret.ty, output, extra, quote!(#v), false)?;
839                self.prefix(&mut child.wires, ComponentSegment::Getter(function.clone()));
840                append(&mut out, child)?;
841            }
842            Output::Combine(parts) => {
843                if parts.is_empty() {
844                    return Err("an output combination needs at least one part".into());
845                }
846                if parts
847                    .iter()
848                    .filter(|p| p.consumes(ty, extra, self.codec))
849                    .count()
850                    > 1
851                {
852                    return Err("a combination cannot consume the same value twice".into());
853                }
854                let mut bodies = Vec::new();
855                for (index, child) in parts.iter().enumerate() {
856                    let consuming = child.consumes(ty, extra, self.codec);
857                    let arg = if borrowed || consuming {
858                        value.clone()
859                    } else {
860                        quote!(&#value)
861                    };
862                    let mut p = self.output(ty, child, extra, arg, borrowed || !consuming)?;
863                    self.prefix(&mut p.wires, ComponentSegment::View(index));
864                    bodies.push((consuming, std::mem::take(&mut p.statements)));
865                    append(&mut out, p)?;
866                }
867                // Wire order follows the declaration. Observations execute before the move.
868                bodies.sort_by_key(|(consumes, _)| *consumes);
869                for (_, body) in bodies {
870                    out.statements.extend(body);
871                }
872            }
873        }
874        Ok(out)
875    }
876}
877fn append<W: prebindgen_flat::WireKind, C, M: PlanMetadata>(
878    to: &mut OutputPlan<W, C, M>,
879    from: OutputPlan<W, C, M>,
880) -> Result<()> {
881    if !from.delivery.statements.is_empty()
882        || from.delivery.result.to_string() != quote!(()).to_string()
883    {
884        return Err("composed output delivery must be installed by Codec::finish_output".into());
885    }
886    to.metadata.merge(from.metadata)?;
887    to.commit.extend(from.commit);
888    to.wires.extend(from.wires);
889    to.statements.extend(from.statements);
890    to.values.extend(from.values);
891    Ok(())
892}

912fn merge_acknowledgement(
913    target: &mut Option<TokenStream>,
914    incoming: Option<TokenStream>,
915) -> Result<()> {
916    if let Some(code) = incoming {
917        if let Some(previous) = target {
918            if previous.to_string() != code.to_string() {
919                return Err("conflicting input acknowledgements".into());
920            }
921        } else {
922            *target = Some(code);
923        }
924    }
925    Ok(())
926}
