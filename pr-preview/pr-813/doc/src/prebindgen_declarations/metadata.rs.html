5/// Describes how a conversion's FFI values fit together.
6///
7/// A conversion plan has a list of **wires**: the individual values passed across
8/// the FFI boundary. This metadata explains how to interpret those values together.
9/// It is stored in [`InputPlan::metadata`] or [`OutputPlan::metadata`] and is used
10/// while generating bindings; it is not passed across the FFI boundary.
11///
12/// For example, C can represent an `Option<i32>` with two wires:
13///
14/// ```text
15/// present: int32_t    0 for None, 1 for Some
16/// value:   int32_t    the integer inside Some; ignored for None
17/// ```
18///
19/// A **group** is a named description of the wires that together represent a
20/// value. For this example, one optional group lists `present` and `value`, says
21/// that `present` decides between `None` and `Some`, and identifies `value` as
22/// the payload. The C adapter uses that information to document the arguments;
23/// the JNI adapter uses optional groups to generate nullable Kotlin parameters
24/// and results where possible.
25///
26/// Groups can contain other groups. For example, an optional record can have an
27/// outer group describing whether the record is present and a child group
28/// describing how its fields form the record. [`groups`](Self::groups) stores all
29/// these descriptions by name. [`roots`](Self::roots) lists the groups that have
30/// no parent. Groups refer to wires by name; the wires stay in the plan's list.
31///
32/// Some types have a **niche**: an otherwise unused representation that can stand
33/// for `None` without adding a presence flag. For example, C uses `0x110000`,
34/// which is not a valid Rust `char`, for `None` in `Option<char>`. Once that value
35/// is taken, an enclosing option needs a different unused value or a flag.
36/// [`domains`](Self::domains) records which niche, if any, each wire can still
37/// offer to an enclosing option. It does not describe the wire's full set of
38/// valid values.
39///
40/// `N` is the adapter's description of a niche. `G` holds information for groups
41/// specific to an adapter, such as the Kotlin class to construct from record
42/// fields. Use `()` for either type when that information is not needed.
43#[derive(Clone, Debug)]
44pub struct Composition<N, G> {
45    /// Every group in this conversion, including nested groups, indexed by name.
46    /// Each entry describes which wires represent a value and how to interpret them.
47    pub groups: BTreeMap<String, CompositionGroup<N, G>>,
48    /// Names of the groups that are not children of another group.
49    /// They appear in the same order as their wires in the plan. A wire that needs
50    /// no grouping can be omitted from all groups; it needs no entry here.
51    pub roots: Vec<String>,
52    /// Whether each wire may supply an unused representation for another `Option`.
53    /// Keys are wire names. The three cases mean:
54    ///
55    /// - No entry: ask [`Codec::niche`] what the wire's type offers.
56    /// - `Some(n)`: use `n` as the next unused representation for `None`.
57    /// - `None`: do not use this wire to encode another `None`.
58    ///
59    /// For example, after `Option<char>` reserves `0x110000` on a C wire named
60    /// `value`, its entry offers `0x110001` for an enclosing option. Other wires
61    /// in the same optional value get `None`: their defaults for an absent value
62    /// must not accidentally be used to distinguish another option layer.
63    pub domains: BTreeMap<String, Option<N>>,
64}
65impl<N, G> Default for Composition<N, G> {
66    fn default() -> Self {
67        Self {
68            groups: BTreeMap::new(),
69            roots: vec![],
70            domains: BTreeMap::new(),
71        }
72    }
73}
74/// Describes the wires that together represent one value, such as an option or record.
75///
76/// For `Option<i32>` encoded as a flag and an integer, `members` names both wires
77/// and `kind` explains the flag. For a record, `members` names the wires for its
78/// fields and `kind` holds the adapter's instructions for constructing the record.
79/// A nested value can have its own group, listed in `children`.
80#[derive(Clone, Debug)]
81pub struct CompositionGroup<N, G> {
82    /// Name of this group, unique within the conversion plan.
83    /// Must match the key used in [`Composition::groups`].
84    pub name: String,
85    /// Names of all wires used to represent this value, in the plan's wire order.
86    /// Includes wires used by child groups and any separate presence flag.
87    /// These wires must occupy consecutive positions in the plan.
88    pub members: Vec<syn::Ident>,
89    /// Names of groups immediately inside this one, in wire order.
90    /// For an optional record, this can name the group describing the record's
91    /// fields. Each name refers to an entry in [`Composition::groups`].
92    pub children: Vec<String>,
93    /// How to interpret the listed wires: as an option or as an adapter-defined value.
94    pub kind: GroupKind<N, G>,
95}
96/// What value a group's wires represent and how to interpret them.
97#[derive(Clone, Debug)]
98pub enum GroupKind<N, G> {
99    /// An option: records how to distinguish `None` from `Some` and find its payload.
100    Optional(OptionalGroup<N>),
101    /// A value described by the language adapter, such as a Kotlin data class
102    /// constructed from several field wires.
103    Adapter(G),
104}
105/// How a group's wires encode `None` and `Some(payload)`.
106///
107/// With a separate presence flag, `discriminator` names that flag, `niche` is
108/// `None`, and `payload` lists the other wires. With a niche, `discriminator`
109/// names one of the payload wires and `niche` describes its reserved `None`
110/// representation. Each layer of a nested option has its own description.
111#[derive(Clone, Debug)]
112pub struct OptionalGroup<N> {
113    /// Name of the wire to inspect to decide whether the option is present.
114    /// This is either a separate 0/1 flag or a payload wire with a reserved
115    /// representation for `None`.
116    pub discriminator: syn::Ident,
117    /// `Some(n)` means this option uses `n` to encode `None` on `discriminator`.
118    /// `None` means `discriminator` is a separate flag: 0 for absent, 1 for present.
119    /// Unlike [`Composition::domains`], this describes the representation already
120    /// chosen for this option, not one available to an enclosing option.
121    pub niche: Option<N>,
122    /// Names of the wires encoding the value inside `Some`, in wire order.
123    /// Excludes a separate presence flag. These values are ignored for `None`.
124    pub payload: Vec<syn::Ident>,
125}
126impl<N, G> PlanMetadata for Composition<N, G> {
127    fn merge(&mut self, other: Self) -> Result<()> {
128        if other.groups.keys().any(|key| self.groups.contains_key(key))
129            || other
130                .domains
131                .keys()
132                .any(|key| self.domains.contains_key(key))
133        {
134            return Err("duplicate composition identifier".into());
135        }
136        self.groups.extend(other.groups);
137        self.domains.extend(other.domains);
138        self.roots.extend(other.roots);
139        Ok(())
140    }
141}
142impl<N, G> Composition<N, G> {
143    /// Prevent enclosing options from using these wires to encode `None`.
144    ///
145    /// Writes `None` to their [`domains`](Self::domains) entries. For example,
146    /// a wire used only in one constructor alternative cannot tell us whether
147    /// the whole optional input is present when another alternative is selected.
148    pub fn suppress<W: prebindgen_flat::WireKind>(&mut self, wires: &[Wire<W>]) {
149        for wire in wires {
150            self.domains.insert(wire.name.to_string(), None);
151        }
152    }
153    /// Add a parent group around the groups currently listed in [`roots`](Self::roots).
154    ///
155    /// Those groups become its children, and `roots` becomes just `[name]`.
156    /// `members` must name all wires belonging to the new parent, including its
157    /// children's wires. Returns an error if `name` already identifies a group.
158    pub fn enclose(
159        &mut self,
160        name: String,
161        members: Vec<syn::Ident>,
162        kind: GroupKind<N, G>,
163    ) -> Result<()> {
164        if self.groups.contains_key(&name) {
165            return Err("duplicate composition group".into());
166        }
167        let children = std::mem::take(&mut self.roots);
168        self.groups.insert(
169            name.clone(),
170            CompositionGroup {
171                name: name.clone(),
172                members,
173                children,
174                kind,
175            },
176        );
177        self.roots.push(name);
178        Ok(())
179    }
180    /// Among the supplied group names, find the group whose first wire is `name`.
181    ///
182    /// Pass [`roots`](Self::roots) to search the outermost groups, or a group's
183    /// `children` to search immediately inside it. This does not search descendants.
184    pub fn starting_at(
185        &self,
186        roots: &[String],
187        name: &syn::Ident,
188    ) -> Option<&CompositionGroup<N, G>> {
189        roots
190            .iter()
191            .filter_map(|id| self.groups.get(id))
192            .find(|g| g.members.first() == Some(name))
193    }
194}

196impl<N, G> Composition<N, G> {
197    /// Check that these descriptions agree with the conversion plan's wire list.
198    ///
199    /// Rejects duplicate wire names, missing references, empty groups, cycles and
200    /// groups whose wires are not consecutive or in order. Also checks that each
201    /// optional group's flag or niche wire and payload belong to that group.
202    /// Wires without a group are allowed.
203    pub fn validate<W: prebindgen_flat::WireKind>(&self, wires: &[Wire<W>]) -> Result<()> {
204        use std::collections::BTreeSet;
205        let names: Vec<_> = wires.iter().map(|w| w.name.clone()).collect();
206        let unique: BTreeSet<_> = names.iter().map(ToString::to_string).collect();
207        if unique.len() != names.len() {
208            return Err("duplicate wire identifier in composition".into());
209        }
210        if self.domains.keys().any(|id| !unique.contains(id)) {
211            return Err("composition domain references missing wire".into());
212        }
213        fn visit<N, G>(
214            m: &Composition<N, G>,
215            scope: &[String],
216            names: &[syn::Ident],
217            seen: &mut BTreeSet<String>,
218        ) -> Result<()> {
219            let mut cursor = 0;
220            for id in scope {
221                if !seen.insert(id.clone()) {
222                    return Err("repeated or cyclic composition group".into());
223                }
224                let g = m
225                    .groups
226                    .get(id)
227                    .ok_or("composition references missing group")?;
228                if &g.name != id {
229                    return Err("composition group name does not match its key".into());
230                }
231                let first = g.members.first().ok_or("empty composition group")?;
232                let start = names[cursor..]
233                    .iter()
234                    .position(|n| n == first)
235                    .map(|i| i + cursor)
236                    .ok_or("overlapping or unordered composition group")?;
237                let end = start + g.members.len();
238                if names.get(start..end) != Some(g.members.as_slice()) {
239                    return Err("non-contiguous composition group".into());
240                }
241                let child_names = if let GroupKind::Optional(o) = &g.kind {
242                    if !g.members.contains(&o.discriminator) {
243                        return Err("optional discriminator outside group".into());
244                    }
245                    let payload = if o.niche.is_some() {
246                        g.members.as_slice()
247                    } else {
248                        if first != &o.discriminator {
249                            return Err("optional flag must precede payload".into());
250                        }
251                        &g.members[1..]
252                    };
253                    if payload != o.payload {
254                        return Err("optional payload does not match group".into());
255                    }
256                    payload
257                } else {
258                    g.members.as_slice()
259                };
260                visit(m, &g.children, child_names, seen)?;
261                cursor = end;
262            }
263            Ok(())
264        }
265        let mut seen = BTreeSet::new();
266        visit(self, &self.roots, &names, &mut seen)?;
267        if seen.len() != self.groups.len() {
268            return Err("unreachable composition group".into());
269        }
270        Ok(())
271    }
272}

274#[cfg(test)]
275mod tests {
276    use super::*;
277    fn wires() -> Vec<Wire<syn::Type>> {
278        ["a", "b", "c"]
279            .iter()
280            .map(|n| Wire {
281                name: format_ident!("{n}"),
282                kind: syn::parse_quote!(i32),
283            })
284            .collect()
285    }
286    fn group(members: &[&str]) -> Composition<(), ()> {
287        let mut m = Composition::default();
288        m.enclose(
289            "group".into(),
290            members.iter().map(|n| format_ident!("{n}")).collect(),
291            GroupKind::Adapter(()),
292        )
293        .unwrap();
294        m
295    }
296    #[test]
297    fn validates_flat_groups_and_rejects_invalid_references() {
298        let w = wires();
299        let m = group(&["b", "c"]);
300        m.validate(&w).unwrap();
301        assert!(m
302            .validate(&[w[0].clone(), w[0].clone()])
303            .unwrap_err()
304            .contains("duplicate wire"));
305        assert!(group(&[]).validate(&w).unwrap_err().contains("empty"));
306        assert!(group(&["a", "c"])
307            .validate(&w)
308            .unwrap_err()
309            .contains("non-contiguous"));
310        let mut m = group(&["a"]);
311        m.domains.insert("missing".into(), None);
312        assert!(m.validate(&w).unwrap_err().contains("missing wire"));
313        let mut m = group(&["a"]);
314        m.groups
315            .get_mut("group")
316            .unwrap()
317            .children
318            .push("group".into());
319        assert!(m.validate(&w).unwrap_err().contains("cyclic"));
320        let mut m = group(&["a"]);
321        m.roots.push("missing".into());
322        assert!(m.validate(&w).unwrap_err().contains("missing group"));
323        let mut m = group(&["a"]);
324        assert!(m.merge(group(&["b"])).unwrap_err().contains("duplicate"));
325        m.groups.get_mut("group").unwrap().kind = GroupKind::Optional(OptionalGroup {
326            discriminator: format_ident!("c"),
327            niche: Some(()),
328            payload: vec![format_ident!("a")],
329        });
330        assert!(m
331            .validate(&w)
332            .unwrap_err()
333            .contains("discriminator outside"));
334    }
