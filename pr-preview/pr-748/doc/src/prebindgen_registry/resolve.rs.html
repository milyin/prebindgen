1//! Completeness: which conversions a binding actually needs, and whether it has
2//! them.
3//!
4//! What survives here is the completeness check. `RegistryBuilder::generate`
5//! compiles the crossings; this decides whether what it answered covers
6//! everything reachable from an exported root.
7//!
8//! There is no loop. `generate` walks the demand inner-first, so each crossing
9//! is answered once, with everything it composes from already built —
10//! including across the `impl Fn` seam, whose args cross in the opposite
11//! direction.
12//!
13//! After the loop, [`required_set`] performs a BFS from the **root** cells — the
14//! ones the binding asked for directly — through `subs` edges. It returns the
15//! reachable set rather than storing it: needing a converter is a property of the
16//! graph, so it is derived once at the end instead of written back into every
17//! cell it was computed from. The final invariant is that every reachable-but-
18//! unresolved cell is reported as an error.
19

26/// Errors surfaced by the resolution phase.
27#[derive(Debug)]
28pub enum ResolveError {
29    /// A type that was scanned as required (or transitively reached from a
30    /// required type via `subs`) ended up with no converter.
31    Unresolved { entries: Vec<UnresolvedEntry> },
32}
33
34#[derive(Debug)]
35pub struct UnresolvedEntry {
36    pub key: TypeKey,
37    pub direction: Direction,
38    pub location: Option<SourceLocation>,
39    /// Why the type is not in the model, when this engine is what took it out —
40    /// see `Registry::dropped`. A declared type is not checked against the
41    /// model when it is declared, because a binding may name a foreign one, so
42    /// a dropped type arrives here rather than as a missing declaration.
43    pub dropped: Option<String>,
44}
45
46impl std::fmt::Display for ResolveError {
47    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
48        match self {
49            ResolveError::Unresolved { entries } => {
50                writeln!(
51                    f,
52                    "{} required type(s) could not be resolved:",
53                    entries.len()
54                )?;
55                for e in entries {
56                    let dir = match e.direction {
57                        Direction::Construct => "input",
58                        Direction::Deconstruct => "output",
59                    };
60                    if let Some(loc) = e.location.as_ref() {
61                        writeln!(
62                            f,
63                            "{}:{}:{}: error: unresolved prebindgen {} type `{}`",
64                            loc.file, loc.line, loc.column, dir, e.key
65                        )?;
66                    } else {
67                        writeln!(f, "error: unresolved prebindgen {} type `{}`", dir, e.key)?;
68                    }
69                    if let Some(why) = e.dropped.as_ref() {
70                        writeln!(f, "  `{}` is not in the model: it {why}", e.key)?;
71                    }
72                }
73                // Once, after the list: several entries can be the same dropped
74                // type in two directions, and the way out is the same for all
75                // of them.
76                if entries.iter().any(|e| e.dropped.is_some()) {
77                    writeln!(f, "{}", crate::registry::declare::DROPPED_ADVICE)?;
78                }
79                Ok(())
80            }
81        }
82    }
83}

87// ──────────────────────────────────────────────────────────────────────
88// Required-flag propagation (BFS from required entries through `subs`)
89// ──────────────────────────────────────────────────────────────────────
90
91/// The cells a converter must exist for: every root, plus everything reachable
92/// from one through a resolved converter's `subs`.
93///
94/// Derived, never stored. Needing a converter is a property of the graph, and the
95/// graph is not complete until resolution has run — so computing it once here
96/// beats maintaining a flag that every edge discovery has to write back.
97fn required_set(registry: &Registry) -> HashSet<(Direction, TypeKey)> {
98    let mut required: HashSet<(Direction, TypeKey)> = HashSet::new();
99    let mut queue: VecDeque<(Direction, TypeKey)> = VecDeque::new();
100    for dir in [Direction::Construct, Direction::Deconstruct] {
101        for (key, cell) in registry.type_table(dir) {
102            if cell.root && required.insert((dir, key.clone())) {
103                queue.push_back((dir, key.clone()));
104            }
105        }
106    }
107
108    while let Some((dir, key)) = queue.pop_front() {
109        // Subs travel in the same direction as the parent — they are the inner
110        // converters this body delegates to. An unresolved cell has none to give,
111        // which is why this cannot run before the conversions have filled them.
112        let Some(entry) = registry
113            .type_table(dir)
114            .get(&key)
115            .and_then(|c| c.entry.as_ref())
116        else {
117            continue;
118        };
119        for sub_key in entry {
120            if required.insert((dir, sub_key.clone())) {
121                queue.push_back((dir, sub_key.clone()));
122            }
123        }
124    }
125    required
126}

128/// BFS from unresolved required-roots through the type graph, surfacing
129/// further unresolved entries reachable through struct fields, enum variants,
130/// generic args, and `impl Fn(...)` args. Stops at resolved nodes — their
131/// `subs` were already walked by `required_set`, so traversing through
132/// them risks reporting dependents the resolved converter doesn't actually
133/// need.
134fn collect_unresolved_descendants(
135    registry: &Registry,
136    seeds: &[(Direction, TypeKey)],
137    seen: &mut std::collections::HashSet<(Direction, TypeKey)>,
138    out: &mut Vec<UnresolvedEntry>,
139) {
140    let mut queue: VecDeque<(Direction, TypeKey)> = VecDeque::new();
141    let enqueue_edges_from =
142        |dir: Direction,
143         key: &TypeKey,
144         queue: &mut VecDeque<(Direction, TypeKey)>,
145         seen: &mut std::collections::HashSet<(Direction, TypeKey)>| {
146            for (child_dir, sub) in registry.immediate_edges(dir, key) {
147                let dep = (child_dir, sub.key());
148                if seen.insert(dep.clone()) {
149                    queue.push_back(dep);
150                }
151            }
152        };
153
154    for (dir, key) in seeds {
155        enqueue_edges_from(*dir, key, &mut queue, seen);
156    }
157
158    while let Some((dir, key)) = queue.pop_front() {
159        match registry.type_table(dir).get(&key) {
160            Some(cell) if cell.entry.is_none() => {
161                // Registered but unresolved — report it and keep walking.
162                out.push(UnresolvedEntry {
163                    key: key.clone(),
164                    direction: dir,
165                    location: Some(cell.subject.location())
166                        .filter(|l| l.has_position())
167                        .cloned(),
168                    dropped: registry.dropped(key.as_str()).cloned(),
169                });
170                enqueue_edges_from(dir, &key, &mut queue, seen);
171            }
172            None => {
173                // Not in the registry at all — can't report (no key/location
174                // worth surfacing), but its structural children may still
175                // include registered-but-unresolved types worth flagging.
176                enqueue_edges_from(dir, &key, &mut queue, seen);
177            }
178            Some(_) => {
179                // Resolved — `required_set` already walked its `subs`. Stop here
180                // to avoid spurious reports for descendants the resolved
181                // converter doesn't need.
182            }
183        }
184    }
185}

187pub(crate) fn check_complete(registry: &Registry) -> Result<(), ResolveError> {
188    let required = required_set(registry);
189    let mut entries: Vec<UnresolvedEntry> = Vec::new();
190    let mut unresolved_required_roots: Vec<(Direction, TypeKey)> = Vec::new();
191    let mut seen_unresolved: HashSet<(Direction, TypeKey)> = HashSet::new();
192
193    for dir in [Direction::Construct, Direction::Deconstruct] {
194        // Sorted, so a build that fails reports the same list every time.
195        let mut keys: Vec<&TypeKey> = registry.type_table(dir).keys().collect();
196        keys.sort_by(|a, b| a.as_str().cmp(b.as_str()));
197        for key in keys {
198            let cell = &registry.type_table(dir)[key];
199            if cell.entry.is_some() || !required.contains(&(dir, key.clone())) {
200                continue;
201            }
202            unresolved_required_roots.push((dir, key.clone()));
203            seen_unresolved.insert((dir, key.clone()));
204            entries.push(UnresolvedEntry {
205                key: key.clone(),
206                direction: dir,
207                location: Some(cell.subject.location())
208                    .filter(|l| l.has_position())
209                    .cloned(),
210                dropped: registry.dropped(key.as_str()).cloned(),
211            });
212        }
213    }
214
215    collect_unresolved_descendants(
216        registry,
217        &unresolved_required_roots,
218        &mut seen_unresolved,
219        &mut entries,
220    );
221
222    if entries.is_empty() {
223        Ok(())
224    } else {
225        Err(ResolveError::Unresolved { entries })
226    }
227}
