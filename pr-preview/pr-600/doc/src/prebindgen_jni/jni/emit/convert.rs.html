5/// Sentinel value to return through the wrapper signature when the inner
6/// closure errors. Must compile against any wire type we emit.
7pub(crate) fn sentinel_for_wire(wire: &syn::Type) -> TokenStream {
8    // Unit wire (void-returning wrappers): the value *is* the sentinel.
9    if let syn::Type::Tuple(t) = wire {
10        if t.elems.is_empty() {
11            return quote!(());
12        }
13    }
14    if let syn::Type::Path(tp) = wire {
15        if let Some(last) = tp.path.segments.last() {
16            let name = last.ident.to_string();
17            return match name.as_str() {
18                "jboolean" | "jbyte" | "jchar" | "jshort" | "jint" | "jlong" => quote!(0 as #wire),
19                "jfloat" | "jdouble" => quote!(0.0 as #wire),
20                "JObject" | "JString" | "JByteArray" | "JClass" => {
21                    quote!(jni::objects::JObject::null().into())
22                }
23                _ => quote!(unsafe { std::mem::zeroed::<#wire>() }),
24            };
25        }
26    }
27    if matches!(wire, syn::Type::Ptr(_)) {
28        return quote!(std::ptr::null());
29    }
30    quote!(unsafe { std::mem::zeroed::<#wire>() })
31}

37pub(crate) fn scalar_input(
38    kind: prebindgen_registry::flat::ScalarKind,
39) -> Option<(syn::Type, syn::Expr)> {
40    use prebindgen_registry::flat::ScalarKind;
41    // Bodies receive `v: &<wire>`; primitives are Copy so `*v` works.
42    Some(match kind {
43        ScalarKind::Bool => (
44            syn::parse_quote!(jni::sys::jboolean),
45            syn::parse_quote!(*v != 0),
46        ),
47        ScalarKind::I32 => (syn::parse_quote!(jni::sys::jint), syn::parse_quote!(*v)),
48        ScalarKind::I64 => (syn::parse_quote!(jni::sys::jlong), syn::parse_quote!(*v)),
49        ScalarKind::U8 => (
50            syn::parse_quote!(jni::sys::jint),
51            syn::parse_quote!(::core::primitive::u8::try_from(*v).map_err(|_| {
52                <__JniErr as ::core::convert::From<String>>::from(format!(
53                    "u8 input out of range: {}",
54                    *v
55                ))
56            })?),
57        ),
58        ScalarKind::U16 => (
59            syn::parse_quote!(jni::sys::jint),
60            syn::parse_quote!(::core::primitive::u16::try_from(*v).map_err(|_| {
61                <__JniErr as ::core::convert::From<String>>::from(format!(
62                    "u16 input out of range: {}",
63                    *v
64                ))
65            })?),
66        ),
67        ScalarKind::U32 => (
68            syn::parse_quote!(jni::sys::jlong),
69            syn::parse_quote!(::core::primitive::u32::try_from(*v).map_err(|_| {
70                <__JniErr as ::core::convert::From<String>>::from(format!(
71                    "u32 input out of range: {}",
72                    *v
73                ))
74            })?),
75        ),
76        // Kotlin's public surface is `ULong`, but the JNI tier receives its
77        // underlying `Long` bit pattern. Rust's `as u64` is the inverse of
78        // Kotlin's `ULong.toLong()` for all 64 bits.
79        ScalarKind::U64 => (
80            syn::parse_quote!(jni::sys::jlong),
81            syn::parse_quote!(*v as ::core::primitive::u64),
82        ),
83        ScalarKind::F64 => (syn::parse_quote!(jni::sys::jdouble), syn::parse_quote!(*v)),
84        _ => return None,
85    })
86}

88pub(crate) fn scalar_output(
89    kind: prebindgen_registry::flat::ScalarKind,
90) -> Option<(syn::Type, syn::Expr)> {
91    use prebindgen_registry::flat::ScalarKind;
92    // Output wrappers take v by value (move). Primitives are Copy, so
93    // `v as wire` works.
94    Some(match kind {
95        ScalarKind::Bool => (
96            syn::parse_quote!(jni::sys::jboolean),
97            syn::parse_quote!(v as jni::sys::jboolean),
98        ),
99        ScalarKind::I32 => (
100            syn::parse_quote!(jni::sys::jint),
101            syn::parse_quote!(v as jni::sys::jint),
102        ),
103        ScalarKind::I64 => (
104            syn::parse_quote!(jni::sys::jlong),
105            syn::parse_quote!(v as jni::sys::jlong),
106        ),
107        ScalarKind::U8 | ScalarKind::U16 => (
108            syn::parse_quote!(jni::sys::jint),
109            syn::parse_quote!(v as jni::sys::jint),
110        ),
111        ScalarKind::U32 | ScalarKind::U64 => (
112            syn::parse_quote!(jni::sys::jlong),
113            syn::parse_quote!(v as jni::sys::jlong),
114        ),
115        ScalarKind::F64 => (
116            syn::parse_quote!(jni::sys::jdouble),
117            syn::parse_quote!(v as jni::sys::jdouble),
118        ),
119        _ => return None,
120    })
121}

155pub(crate) fn is_jobject_wire(wire: &syn::Type) -> bool {
156    if let syn::Type::Path(tp) = wire {
157        if let Some(last) = tp.path.segments.last() {
158            return last.ident == "JObject";
159        }
160    }
161    false
162}

164/// True if `wire` is a JNI handle (`JObject`, `JString`, `JByteArray`,
165/// `JClass`) that natively supports a `null` discriminator. These types
166/// all impl `is_null()` and accept `JObject::null().into()` for
167/// construction.
168pub(crate) fn is_jobject_shaped_wire(wire: &syn::Type) -> bool {
169    crate::jni::wire_access::is_jni_reference_wire(wire)
170}

172/// Default niche set for a JNI wrapper wire: every `J*` handle has a
173/// genuine `null` value that no live conversion ever produces, so wrap
174/// it as a single niche; everything else (`jlong`, `jint`, `()`, …) has
175/// no implicit niche.
176///
177/// Plugins are free to declare *additional* niches on top of this for
178/// pointer-shape primitives like `Box::into_raw`-as-`jlong`.
179pub(crate) fn default_niches_for_wire(wire: &syn::Type) -> Niches {
180    if is_jobject_shaped_wire(wire) {
181        Niches::one(
182            syn::parse_quote!(jni::objects::JObject::null().into()),
183            syn::parse_quote!(v.is_null()),
184        )
185    } else {
186        Niches::empty()
187    }
188}

197/// `OwnedObject<T>` definition emitted into the destination Rust file.
198///
199/// A non-owning borrow wrapper around a `*const T` whose backing
200/// `Box<T>` lives on the Java side. The Java side hands Rust the
201/// pointer under its `NativeHandle.withPtr` read lock; for the
202/// duration of the JNI call the heap allocation is guaranteed live,
203/// so `Deref<Target = T>` exposing `&*ptr` is sound. The wrapper has
204/// no `Drop`: nothing is freed here, the Box stays with Java.
205///
206/// By-value `T` extraction is intentionally NOT through this wrapper.
207/// Consume call sites use `*Box::from_raw(ptr)` inline, taking
208/// ownership of Java's slot; `NativeHandle.consume` (write-lock +
209/// atomic null) sequences that against any concurrent borrow.
210///
211/// Co-locating the definition with the converters keeps the generated
212/// file self-contained — no `use` statement or runtime-support module
213/// is required from the host crate.
214pub(crate) fn owned_object_prerequisite_items() -> Vec<syn::Item> {
215    vec![
216        syn::parse_quote!(
217            /// See module-level docs at [`owned_object_prerequisite_items`].
218            #[allow(dead_code)]
219            pub(crate) struct OwnedObject<T: ?Sized> {
220                ptr: *const T,
221            }
222        ),
223        syn::parse_quote!(
224            impl<T: ?Sized> std::ops::Deref for OwnedObject<T> {
225                type Target = T;
226                #[inline]
227                fn deref(&self) -> &Self::Target {
228                    unsafe { &*self.ptr }
229                }
230            }
231        ),
232        syn::parse_quote!(
233            // `&mut OwnedObject<T>` coerces to `&mut T` via this impl,
234            // letting source fns that take `&mut T` opaque-handle params
235            // be called from generated wrappers. The pointer originated
236            // from `Box::into_raw` (which produces `*mut T`); the
237            // `*const T → *mut T` cast just restores the original
238            // mutability. Sequencing against concurrent borrow / consume
239            // is upheld by `NativeHandle.withPtr` on the JVM side, same
240            // as `Deref`.
241            impl<T: ?Sized> std::ops::DerefMut for OwnedObject<T> {
242                fn deref_mut(&mut self) -> &mut Self::Target {
243                    unsafe { &mut *(self.ptr as *mut T) }
244                }
245            }
246        ),
247        syn::parse_quote!(
248            impl<T: ?Sized> OwnedObject<T> {
249                /// Borrow a `T` whose backing `Box<T>` lives on the
250                /// Java side. Stores only the pointer; the wrapper
251                /// does not own the heap allocation and never frees
252                /// it on drop.
253                ///
254                /// # Safety
255                ///
256                /// `ptr` must be the result of an earlier
257                /// `Box::into_raw(Box::new(v))` and the allocation
258                /// must still be live (Java still owns it). The Java
259                /// side is responsible for sequencing this call
260                /// against any concurrent free or consume (via
261                /// `NativeHandle.withPtr` read-lock vs `consume` /
262                /// `close` write-lock) so the borrow cannot race a
263                /// deallocation on the same pointer.
264                #[allow(dead_code)]
265                pub(crate) unsafe fn from_raw(ptr: *const T) -> Self {
266                    Self { ptr }
267                }
268            }
269        ),
270    ]
271}
