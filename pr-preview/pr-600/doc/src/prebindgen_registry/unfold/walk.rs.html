1//! The walk over a decomposed value.
2//!
3//! A decomposition — an [`UnfoldPlan`](crate::unfold::UnfoldPlan) — says which
4//! **leaves** a value is delivered as and which **hoists** (intermediate value
5//! forms) they are reached through. Performing that walk is the registry's:
6//! binding each hoist once, reaching each leaf through its own path, and
7//! deciding at every step whether what it holds is owned or borrowed.
8//!
9//! What an adapter supplies is the target-language half — how a leaf is
10//! encoded, and how the delivered values are handed over — plus the two facts
11//! this module cannot know: where a source item is qualified from
12//! (`qualify`), and what each leaf is ([`DecomposedLeaf`]).
13
14use proc_macro2::TokenStream;
15use quote::{format_ident, quote};

21/// What the walk needs to know about one delivered leaf.
22///
23/// The adapter's own leaf carries far more — its wire type, its JNI or C
24/// encoding, its projection — and none of that reaches here.
25pub trait DecomposedLeaf {
26    /// Steps from the decomposed value to this leaf.
27    fn reach(&self) -> &[PathStep];
28
29    /// The leaf's own name, used in diagnostics and in local names.
30    fn name(&self) -> &str;
31
32    /// Whether this leaf *is* the decomposed value rather than a part of it.
33    fn identity(&self) -> bool;
34
35    /// The reading the leaf delivers.
36    fn source(&self) -> &TypeRef;
37
38    /// Whether the last step reaching it is a field read rather than a call.
39    fn is_field_read(&self) -> bool;
40}

42/// Compose one [`PathStep`] onto the reference expression reached so far.
43/// A `Call` applies its accessor (origin-qualified); a `Field` reads the field
44/// and re-borrows, so the result is a reference either way and steps chain
45/// uniformly.
46pub fn compose_step(
47    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
48    step: &PathStep,
49    e: TokenStream,
50) -> TokenStream {
51    match step {
52        PathStep::Call { ident, .. } => {
53            let m = qualify(ident);
54            quote!(#m::#ident(#e))
55        }
56        PathStep::Field { ident, .. } => quote!(&(#e).#ident),
57    }
58}
59/// Fold a run of steps onto `e`, borrowing wherever ownership demands it.
60///
61/// [`compose_step`] hands a `Call` its receiver as written, and an accessor
62/// takes that receiver **by reference** — so an owned value in hand has to be
63/// borrowed before the next call composes onto it. A value is in hand whenever
64/// the previous step returned one (`f(..) -> T` rather than `-> &T`), which is
65/// what [`PathStep::yields_owned`] records; `owned` says whether `e` itself
66/// started that way.
67///
68/// A `Field` step needs no borrow either way — it composes as `&(e).f`, which
69/// reads through a value and a reference alike.
70///
71/// This is the ONE place the rule lives, so every fold — a leaf's reach, a
72/// conditional hoist's prefix, a sum's matched value — answers it the same way.
73/// Splitting it produced exactly the bug it exists to prevent: ownership was
74/// handled at the optional binding and at the value form, and lost at every
75/// ordinary call in between.
76pub fn fold_steps(
77    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
78    steps: &[PathStep],
79    mut e: TokenStream,
80    mut owned: bool,
81) -> TokenStream {
82    for step in steps {
83        if owned && matches!(step, PathStep::Call { .. }) {
84            e = quote!(&#e);
85        }
86        e = compose_step(qualify, step, e);
87        owned = step.yields_owned();
88    }
89    e
90}
91/// Compose a value form's OWN call — the one step in the whole system whose
92/// receiver may be by value.
93///
94/// Four cases, from what the fold ended up holding crossed with what the
95/// accessor takes. A CONSUMING form takes its receiver by value: hand it the
96/// value when that is ours, clone when it is not — the same cost the borrowing
97/// form of the accessor would have paid, which keeps one declaration usable by
98/// both owned and `&T` roots. A borrowing form takes a reference, so an owned
99/// value is borrowed for it.
100///
101/// The decision is made from the fold's RESULT, never from where the fold
102/// began: that is what lets a consuming form sit behind ordinary accessors,
103/// where the chain in front borrows and the form itself still moves.
104///
105/// One function because both hoist paths — the conditional binding and the
106/// ordinary one — need exactly this rule, and stating it twice is what turned
107/// each new shape into a new defect.
108fn compose_value_form_call(
109    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
110    call: &PathStep,
111    e: TokenStream,
112    e_owned: bool,
113    consuming: bool,
114) -> TokenStream {
115    match (consuming, e_owned) {
116        (true, owned) => {
117            let (m, f) = (qualify(call.ident()), call.ident());
118            // Parenthesized: the clone applies to whatever the fold holds, and
119            // `&x.clone()` would parse as `&(x.clone())`.
120            let arg = if owned { e } else { quote!((#e).clone()) };
121            quote!(#m::#f(#arg))
122        }
123        (false, true) => compose_step(qualify, call, quote!(&#e)),
124        (false, false) => compose_step(qualify, call, e),
125    }
126}
127/// Start a reach from `base`, projecting the **leading run of plain field
128/// steps** directly (`&base.a.b`) instead of through a borrow of the base
129/// (`&(&base).a.b`). Returns the expression and how many steps it consumed.
130///
131/// The two forms name the same value, but the second borrows the base **as a
132/// whole**, which the borrow checker rejects once a sibling leaf has moved a
133/// different field out of it. Projecting directly makes each leaf's borrow
134/// disjoint, so a consuming value form's field moves are order-independent
135/// rather than compiling only while the borrowing leaves happen to be declared
136/// first.
137pub fn project_leading_fields(
138    base: &TokenStream,
139    base_is_ref: bool,
140    path: &[PathStep],
141) -> (TokenStream, usize) {
142    if base_is_ref {
143        return (base.clone(), 0);
144    }
145    let n = path.iter().take_while(|s| s.is_plain_field()).count();
146    if n == 0 {
147        return (quote!(&#base), 0);
148    }
149    let segs: Vec<&syn::Ident> = path[..n].iter().map(PathStep::ident).collect();
150    (quote!(&#base #(.#segs)*), n)
151}

153/// Fold a leaf's whole `path` over `base` with no optional-step handling, then
154/// apply the terminal treatment the leaf calls for: a leaf reached by a field
155/// read is **cloned** out of the place it reached, because its converter takes
156/// the field type as written (owned); every other leaf keeps the borrow its
157/// converter expects.
158///
159/// One derivation, used by every delivery an adapter renders and by the
160/// single-leaf shortcut a decomposed return takes. They drifted once while
161/// they were two — the shortcut was missing the field clone and handed `&F` to
162/// an `F` converter — which is why the walk is the registry's.
163pub fn reach_leaf_flat<L: DecomposedLeaf>(
164    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
165    leaf: &L,
166    path: &[PathStep],
167    base: TokenStream,
168    base_is_ref: bool,
169    consuming: bool,
170) -> TokenStream {
171    // An optional step BEFORE the last one needs a `match` whose `None` arm has
172    // somewhere to go. This derivation has none — it yields a plain Rust value,
173    // not a `JObject` that could be null — so the shape is refused here rather
174    // than composed into code that cannot type-check in the consumer's crate.
175    //
176    // Asked of the leaf's OWN path, not of `path`. The caller may hand a
177    // suffix: `wrapper.rs` rebases onto a hoisted local, and `Hoisted::innermost`
178    // strips the prefix that bound it — including any optional step inside it.
179    // Checking the parameter would therefore pass exactly when the hoist is the
180    // conditional one, which is the case that cannot compose (an `Option<T>`
181    // local with a field read hung off it). The full path is what the shape
182    // question is about.
183    let own_path = leaf.reach();
184    assert!(
185        !own_path.iter().rev().skip(1).any(PathStep::is_optional),
186        "jnigen unfold: leaf `{}` reaches through an optional step but is \
187         delivered as a single return value, which has no `None` arm — this \
188         shape needs callback delivery",
189        leaf.name(),
190    );
191    // Whether what this leaf reaches is OURS, and so is moved rather than
192    // borrowed or cloned. The two leaf kinds say it differently:
193    //
194    // * an IDENTITY leaf carries the answer in its `out_ty` — the plan resolved
195    //   it to the owned type exactly when the value is the plan's to give away
196    //   (`place_is_owned`: an owned root, or a field of a CONSUMING value form),
197    //   and that is also what selected the owning converter, which boxes the
198    //   move rather than cloning a borrow;
199    // * a FIELD leaf's `out_ty` is the field type as written, owned either way,
200    //   so ownership is the enclosing form's: only a consuming one gives its
201    //   fields away.
202    //
203    // How to project that place is `steps_are_movable`'s question, and it is
204    // asked there rather than restated here. This used to spell it
205    // `all(is_plain_field)`, defending the restatement on the grounds that a
206    // trailing `Option` cannot reach return delivery anyway — true, and enforced
207    // in `single_return` (`core/unfold.rs`), which is precisely why a local
208    // restatement could disagree with the rule for as long as the invariant held
209    // somewhere else. `plan.rs` says two readings would drift and the
210    // disagreement would be a borrow handed to an owning converter; this is the
211    // second reading, removed.
212    let reached_is_ours = if leaf.identity() {
213        !matches!(
214            leaf.source().kind(),
215            prebindgen_flat::flat::TypeKind::Ref { .. }
216        )
217    } else {
218        consuming
219    };
220    if reached_is_ours && steps_are_movable(path) {
221        let segs: Vec<&syn::Ident> = path.iter().map(PathStep::ident).collect();
222        return quote!(#base #(.#segs)*);
223    }
224    let (e, lead) = project_leading_fields(&base, base_is_ref, path);
225    let e = fold_steps(qualify, &path[lead..], e, false);
226    if leaf.is_field_read() {
227        quote!((#e).clone())
228    } else {
229        e
230    }
231}
232/// Every value form on a plan, evaluated **once** and bound to a local
233/// (`__vf0`, `__vf1`, …), so a struct is built once per delivery rather than
234/// once per field. The bound prefixes come back with the statements, since
235/// reaching a leaf means starting from the innermost local it sits under.
236///
237/// Shared by both delivery paths — the multi-leaf encoder below and the
238/// single-leaf `Delivery::Return` shortcut in `emit/wrapper.rs`. The shortcut
239/// used to compose its reach straight off the raw value, which for a consuming
240/// value form emitted `f(&v)` against a by-value receiver: ill-typed Rust in
241/// the consumer's crate. One binder, so the two cannot disagree about what a
242/// hoist is or who owns it.
243pub struct Hoisted {
244    /// The `let __vfN = …;` bindings, outermost-first.
245    pub stmts: TokenStream,
246    /// Each hoist's path prefix and the local it was bound to.
247    bound: Vec<(Vec<PathStep>, syn::Ident)>,
248    /// Whether each bound hoist consumed the value it decomposed.
249    consuming: Vec<bool>,
250    /// Whether each bound local is `Option<TStruct>` rather than `TStruct` —
251    /// the hoist sits under an optional step, so the value form ran only where
252    /// the value was present. Its leaves cannot be emitted as independent
253    /// statements: they share ONE `match` on the local (see
254    /// [`encode_plan_leaves`]), taken by value, so a consuming form's fields
255    /// still move out inside the arm.
256    optional: Vec<bool>,
257}
258impl Hoisted {
259    /// Index of the innermost bound hoist `path` sits under, with that prefix
260    /// already consumed. `None` for a leaf under no value form at all — a
261    /// sibling `.field()` / `.field_self()`, which still reaches from the value
262    /// itself.
263    fn innermost(&self, path: &[PathStep]) -> Option<(usize, Vec<PathStep>)> {
264        self.bound
265            .iter()
266            .enumerate()
267            .filter(|(_, (p, _))| p.len() < path.len() && path.starts_with(p))
268            .max_by_key(|(_, (p, _))| p.len())
269            .map(|(i, (p, _))| (i, path[p.len()..].to_vec()))
270    }
271
272    /// The innermost bound local `path` sits under, with that prefix already
273    /// consumed, and whether that hoist gave its value away.
274    pub fn rebase(&self, path: &[PathStep]) -> Option<(syn::Ident, Vec<PathStep>, bool)> {
275        self.innermost(path)
276            .map(|(i, rest)| (self.bound[i].1.clone(), rest, self.consuming[i]))
277    }
278
279    /// The innermost **conditional** hoist `path` sits under: its index, the
280    /// local holding the `Option`, the name its `Some` arm binds, and the steps
281    /// left to reach the leaf from there. `None` when the leaf's innermost
282    /// hoist is unconditional (or there is none) — then [`Self::rebase`]
283    /// applies and the leaf is an ordinary standalone statement.
284    pub fn conditional(
285        &self,
286        path: &[PathStep],
287    ) -> Option<(usize, syn::Ident, syn::Ident, Vec<PathStep>)> {
288        let (i, rest) = self.innermost(path)?;
289        self.optional[i].then(|| (i, self.bound[i].1.clone(), format_ident!("__u{}", i), rest))
290    }
291
292    /// The local a hoist was bound to.
293    pub fn local(&self, i: usize) -> syn::Ident {
294        self.bound[i].1.clone()
295    }
296
297    /// Whether a hoist consumed the value it decomposed.
298    pub fn consumed(&self, i: usize) -> bool {
299        self.consuming[i]
300    }
301}
302/// Fold `path` over `base` the way [`reach_leaf`] does, but yielding an
303/// `Option<…>` rather than a `JObject`: the optional steps become a
304/// `map`/`and_then` chain, so an absent value short-circuits to `None` instead
305/// of to a null object. `body` renders the innermost reached expression as a
306/// BARE value — the chain's last link wraps it.
307///
308/// This is how a CONDITIONAL value form is bound — the accessor runs only where
309/// the value it decomposes is actually present.
310fn reach_optional(
311    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
312    path: &[PathStep],
313    base: TokenStream,
314    base_is_ref: bool,
315    depth: usize,
316    body: &dyn Fn(TokenStream) -> TokenStream,
317) -> TokenStream {
318    let (e, lead) = project_leading_fields(&base, base_is_ref, path);
319    match (lead..path.len()).find(|&i| path[i].is_optional()) {
320        None => body(fold_steps(qualify, &path[lead..], e, false)),
321        Some(k) => {
322            // Through the optional step INCLUSIVE: the same fold, so the
323            // borrow in front of it is the ordinary rule rather than a second
324            // statement of it.
325            let opt_e = fold_steps(qualify, &path[lead..=k], e, false);
326            let bind = format_ident!("__hb{}", depth);
327            // What the arm binds is the step's own value: an OWNED payload is a
328            // bare `T`, so composing the next step onto it directly would hand
329            // `T` to an accessor typed for `&T`. Say it is not a reference and
330            // let `project_leading_fields` borrow it; a borrowed payload is
331            // already one and passes through.
332            //
333            // With NO steps left the binding goes to `body` untouched — that is
334            // what lets a consuming value form MOVE an owned payload rather than
335            // borrow it straight back, so the terminal case stays "already a
336            // reference" whatever the payload is.
337            let rest = &path[k + 1..];
338            let inner = reach_optional(
339                qualify,
340                rest,
341                quote!(#bind),
342                rest.is_empty() || !path[k].yields_owned(),
343                depth + 1,
344                body,
345            );
346            // `map` when this is the LAST optional step (the body yields a bare
347            // value) and `and_then` when another follows (the recursion yields
348            // an `Option` that must not nest). The equivalent `match` reads the
349            // same but generated code runs through the consumer's lints, where
350            // `clippy::manual_map` is a denial.
351            let combinator = if rest.iter().any(PathStep::is_optional) {
352                format_ident!("and_then")
353            } else {
354                format_ident!("map")
355            };
356            quote! {
357                #opt_e.#combinator(|#bind| #inner)
358            }
359        }
360    }
361}
362pub fn bind_hoists(
363    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
364    hoists: &[crate::unfold::Hoist],
365    value: &TokenStream,
366    by_ref: bool,
367) -> Hoisted {
368    let mut out = Hoisted {
369        stmts: TokenStream::new(),
370        bound: Vec::new(),
371        consuming: Vec::new(),
372        optional: Vec::new(),
373    };
374    // Value forms COMPOSE, so each hoist is built from the longest hoist that
375    // is already a proper prefix of it (they arrive outermost-first), and from
376    // `value` otherwise.
377    for (i, h) in hoists.iter().enumerate() {
378        let local = format_ident!("__vf{}", i);
379        // A hoist under an optional step binds `Option<TStruct>`: the value
380        // form runs in the `Some` arm only. Core refuses to nest these, so the
381        // enclosing value is always the plan's own — no rebase to consider.
382        if h.prefix.iter().any(PathStep::is_optional) {
383            let (last, lead) = h
384                .prefix
385                .split_last()
386                .expect("a hoist prefix ends in its value-form call");
387            let consuming = h.consuming;
388            // The value form is handed the payload only when the optional step
389            // is the LAST thing before it; any step in between composes as a
390            // borrow, so what arrives is a reference either way.
391            let owned = lead.last().is_some_and(PathStep::yields_owned);
392            let expr = reach_optional(qualify, lead, value.clone(), by_ref, 0, &|reached| {
393                compose_value_form_call(qualify, last, reached, owned, consuming)
394            });
395            out.stmts.extend(quote! { let #local = #expr; });
396            out.bound.push((h.prefix.clone(), local));
397            out.consuming.push(h.consuming);
398            out.optional.push(true);
399            continue;
400        }
401        // Where the fold starts, and whether what it starts from is OWNED. The
402        // value form's own boundary is decided below, from what the fold ends
403        // up holding — never from where it began.
404        let (from, start, start_owned) = match out.rebase(&h.prefix) {
405            // A NESTED consuming form is handed the parent's field by MOVE: a
406            // hoisted value form is an owned struct and its fields are
407            // disjoint, so moving one out leaves every sibling leaf readable.
408            // `compose_step` borrows (`&(e).f`), so a plain field run to that
409            // field is projected here instead of going through it.
410            Some((outer, rest, _))
411                if h.consuming && rest[..rest.len() - 1].iter().all(PathStep::is_plain_field) =>
412            {
413                let lead = &rest[..rest.len() - 1];
414                let segs: Vec<&syn::Ident> = lead.iter().map(PathStep::ident).collect();
415                (h.prefix.len() - 1, quote!(#outer #(.#segs)*), true)
416            }
417            // Any other rebased hoist: project its own leading field run
418            // DIRECTLY off the parent local rather than reaching it through a
419            // borrow of the parent. A sibling hoist may already have moved a
420            // different field out — that is what a consuming value form does —
421            // and `&(&__vf0).wrapper` borrows the partially moved parent as a
422            // whole where `&__vf0.wrapper` is a disjoint borrow that survives.
423            // Same invariant `project_leading_fields` states for leaf reaches,
424            // and the same reason.
425            Some((outer, rest, _)) => {
426                let (e, lead) = project_leading_fields(&quote!(#outer), false, &rest);
427                (h.prefix.len() - rest.len() + lead, e, false)
428            }
429            None if by_ref => (0, value.clone(), false),
430            None => (0, value.clone(), true),
431        };
432        // Everything before the value form is an ordinary accessor chain.
433        let last = h.prefix.len() - 1;
434        let head = &h.prefix[from..last];
435        let e = fold_steps(qualify, head, start, start_owned);
436        let e_owned = head.last().map_or(start_owned, PathStep::yields_owned);
437        // The value form itself. A CONSUMING one takes its receiver BY VALUE —
438        // that is the move the whole declaration exists for — so it is handed
439        // what the fold holds when that is ours, and a clone when it is not:
440        // the same cost the borrowing form of the accessor would have paid,
441        // which keeps one declaration usable by both owned and `&T` returns.
442        // A borrowing one takes a reference, so an owned value is borrowed.
443        //
444        // Deciding this from the fold's RESULT rather than from its start is
445        // what lets a consuming form sit behind ordinary accessors: the chain
446        // in front borrows, the form itself still moves.
447        let expr = compose_value_form_call(qualify, &h.prefix[last], e, e_owned, h.consuming);
448        out.stmts.extend(quote! { let #local = #expr; });
449        out.bound.push((h.prefix.clone(), local));
450        out.consuming.push(h.consuming);
451        out.optional.push(false);
452    }
453    out
