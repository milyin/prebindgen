1//! The walk over a decomposed value.
2//!
3//! A decomposition — an [`UnfoldPlan`](crate::unfold::UnfoldPlan) — says which
4//! **leaves** a value is delivered as and which **hoists** (intermediate value
5//! forms) they are reached through. Performing that walk is the registry's:
6//! binding each hoist once, reaching each leaf through its own path, and
7//! deciding at every step whether what it holds is owned or borrowed.
8//!
9//! What an adapter supplies is the target-language half — how a leaf is
10//! encoded, and how the delivered values are handed over — plus the two facts
11//! this module cannot know: where a source item is qualified from
12//! (`qualify`), and what each leaf is ([`DecomposedLeaf`]).
13
14use prebindgen_flat::flat::TypeRef;
15use proc_macro2::TokenStream;
16use quote::{format_ident, quote};
17
18use crate::unfold::{steps_are_movable, PathStep};

20/// What the walk needs to know about one delivered leaf.
21///
22/// The adapter's own leaf carries far more — its wire type, its JNI or C
23/// encoding, its projection — and none of that reaches here.
24pub trait DecomposedLeaf {
25    /// Steps from the decomposed value to this leaf.
26    fn reach(&self) -> &[PathStep];
27
28    /// The leaf's own name, used in diagnostics and in local names.
29    fn name(&self) -> &str;
30
31    /// Whether this leaf *is* the decomposed value rather than a part of it.
32    fn identity(&self) -> bool;
33
34    /// The reading the leaf delivers.
35    fn source(&self) -> &TypeRef;
36
37    /// Whether the last step reaching it is a field read rather than a call.
38    ///
39    /// Derived from [`Self::reach`], so no implementation can disagree with
40    /// its own path — and the answer decides whether the reached place is
41    /// cloned, which is an ownership decision rather than a spelling.
42    fn is_field_read(&self) -> bool {
43        matches!(self.reach().last(), Some(PathStep::Field { .. }))
44    }
45}

47/// A leaf's reach, rendered **around** an encoding body: the walk hands the
48/// body the reached Rust expression and gets back the encoded value, so an
49/// absent value short-circuits to the adapter's own absence rather than to one
50/// the walk chose. `dyn` on both halves because each side is written where it
51/// is known and neither can name the other's closure.
52pub type Reach<'a> = dyn Fn(&dyn Fn(TokenStream) -> TokenStream) -> TokenStream + 'a;

54/// The adapter's half of a delivery.
55///
56/// The walk in this module owns how a decomposed value is reached — which
57/// hoist a leaf sits under, what is owned, what is borrowed. What a leaf
58/// becomes in the target language, and how the delivered values are handed to
59/// the call, is the adapter's, and this is where it says so. Stated over
60/// [`DecomposedLeaf`] rather than over an adapter's own leaf type, so the walk
61/// can call it.
62pub trait DeliveryBridge {
63    /// The adapter's leaf.
64    type Leaf: DecomposedLeaf;
65
66    /// Where the source item declaring `ident` is qualified from — the one
67    /// fact about a captured path this crate cannot answer for itself.
68    fn qualify(&self, ident: &syn::Ident) -> syn::Path;
69
70    /// The sum a selector leaf names: the qualified source path of the type,
71    /// and its Flat shape.
72    fn sum(&self, leaf: &Self::Leaf) -> (syn::Path, &prebindgen_flat::flat::Variant);
73
74    /// Encode one leaf into the value the delivery call receives, and bind
75    /// that value to `slot`.
76    ///
77    /// The adapter emits the binding rather than the walk, because the slot's
78    /// type is the adapter's: one leaf may cross as a scalar and the next as a
79    /// reference. `index` is the leaf's position in the delivery, which is
80    /// what any temporary the encode needs is named from. An encoding that can
81    /// fail routes its failure through `fail`.
82    fn encode(
83        &self,
84        leaf: &Self::Leaf,
85        index: usize,
86        slot: &syn::Ident,
87        reach: &Reach<'_>,
88        fail: &dyn Fn(TokenStream) -> TokenStream,
89        emit: &crate::RustWriter,
90    ) -> TokenStream;
91
92    /// The expression that passes `slot` as one argument of the delivery call.
93    fn argument(&self, leaf: &Self::Leaf, slot: &syn::Ident) -> TokenStream;
94
95    /// What a leaf delivers when an optional step on the way to it found
96    /// nothing. [`reach_leaf`] gates on this rather than on a value of its
97    /// own: absence is a target-language representation, and this crate has
98    /// none.
99    fn absent(&self) -> TokenStream;
100}

102/// Compose one [`PathStep`] onto the reference expression reached so far.
103/// A `Call` applies its accessor (origin-qualified); a `Field` reads the field
104/// and re-borrows, so the result is a reference either way and steps chain
105/// uniformly.
106pub fn compose_step(
107    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
108    step: &PathStep,
109    e: TokenStream,
110) -> TokenStream {
111    match step {
112        PathStep::Call { ident, .. } => {
113            let m = qualify(ident);
114            quote!(#m::#ident(#e))
115        }
116        PathStep::Field { ident, .. } => quote!(&(#e).#ident),
117    }
118}
119/// Fold a run of steps onto `e`, borrowing wherever ownership demands it.
120///
121/// [`compose_step`] hands a `Call` its receiver as written, and an accessor
122/// takes that receiver **by reference** — so an owned value in hand has to be
123/// borrowed before the next call composes onto it. A value is in hand whenever
124/// the previous step returned one (`f(..) -> T` rather than `-> &T`), which is
125/// what [`PathStep::yields_owned`] records; `owned` says whether `e` itself
126/// started that way.
127///
128/// A `Field` step needs no borrow either way — it composes as `&(e).f`, which
129/// reads through a value and a reference alike.
130///
131/// This is the ONE place the rule lives, so every fold — a leaf's reach, a
132/// conditional hoist's prefix, a sum's matched value — answers it the same way.
133/// Splitting it produced exactly the bug it exists to prevent: ownership was
134/// handled at the optional binding and at the value form, and lost at every
135/// ordinary call in between.
136pub fn fold_steps(
137    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
138    steps: &[PathStep],
139    mut e: TokenStream,
140    mut owned: bool,
141) -> TokenStream {
142    for step in steps {
143        if owned && matches!(step, PathStep::Call { .. }) {
144            e = quote!(&#e);
145        }
146        e = compose_step(qualify, step, e);
147        owned = step.yields_owned();
148    }
149    e
150}
151/// Compose a value form's OWN call — the one step in the whole system whose
152/// receiver may be by value.
153///
154/// Four cases, from what the fold ended up holding crossed with what the
155/// accessor takes. A CONSUMING form takes its receiver by value: hand it the
156/// value when that is ours, clone when it is not — the same cost the borrowing
157/// form of the accessor would have paid, which keeps one declaration usable by
158/// both owned and `&T` roots. A borrowing form takes a reference, so an owned
159/// value is borrowed for it.
160///
161/// The decision is made from the fold's RESULT, never from where the fold
162/// began: that is what lets a consuming form sit behind ordinary accessors,
163/// where the chain in front borrows and the form itself still moves.
164///
165/// One function because both hoist paths — the conditional binding and the
166/// ordinary one — need exactly this rule, and stating it twice is what turned
167/// each new shape into a new defect.
168fn compose_value_form_call(
169    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
170    call: &PathStep,
171    e: TokenStream,
172    e_owned: bool,
173    consuming: bool,
174) -> TokenStream {
175    match (consuming, e_owned) {
176        (true, owned) => {
177            let (m, f) = (qualify(call.ident()), call.ident());
178            // Parenthesized: the clone applies to whatever the fold holds, and
179            // `&x.clone()` would parse as `&(x.clone())`.
180            let arg = if owned { e } else { quote!((#e).clone()) };
181            quote!(#m::#f(#arg))
182        }
183        (false, true) => compose_step(qualify, call, quote!(&#e)),
184        (false, false) => compose_step(qualify, call, e),
185    }
186}
187/// Start a reach from `base`, projecting the **leading run of plain field
188/// steps** directly (`&base.a.b`) instead of through a borrow of the base
189/// (`&(&base).a.b`). Returns the expression and how many steps it consumed.
190///
191/// The two forms name the same value, but the second borrows the base **as a
192/// whole**, which the borrow checker rejects once a sibling leaf has moved a
193/// different field out of it. Projecting directly makes each leaf's borrow
194/// disjoint, so a consuming value form's field moves are order-independent
195/// rather than compiling only while the borrowing leaves happen to be declared
196/// first.
197pub fn project_leading_fields(
198    base: &TokenStream,
199    base_is_ref: bool,
200    path: &[PathStep],
201) -> (TokenStream, usize) {
202    if base_is_ref {
203        return (base.clone(), 0);
204    }
205    let n = path.iter().take_while(|s| s.is_plain_field()).count();
206    if n == 0 {
207        return (quote!(&#base), 0);
208    }
209    let segs: Vec<&syn::Ident> = path[..n].iter().map(PathStep::ident).collect();
210    (quote!(&#base #(.#segs)*), n)
211}

213/// One leaf, and the place a reach of it starts from.
214///
215/// A caller that has already bound a hoist hands over the local it bound and
216/// the steps that are left, which is why `path` is a suffix of the leaf's own
217/// reach rather than always equal to it.
218pub struct LeafAt<'a, L> {
219    /// The leaf being reached.
220    pub leaf: &'a L,
221    /// The steps still to walk from `base`.
222    pub path: &'a [PathStep],
223    /// What the walk starts from — the delivered value, or a local a hoist or
224    /// a gate bound.
225    pub base: TokenStream,
226    /// Whether `base` is already a reference.
227    pub base_is_ref: bool,
228    /// Whether the form that produced `base` gave its value away, which is
229    /// what lets a field of it be moved rather than cloned.
230    pub consuming: bool,
231    /// Whether a FINAL optional step is gated too. An identity leaf delivers
232    /// the reached value itself, so it is; every other leaf hands the final
233    /// step's full type to its own converter, `Option` and all.
234    pub unwrap_last: bool,
235}

237/// Reach one leaf from `base` and hand what it reached to `body`.
238///
239/// Three things happen on the way, and all three are facts about Rust values
240/// rather than about any target language:
241///
242/// * **Gating.** An optional step before the leaf becomes a `match` whose
243///   `None` arm yields `absent()` — the adapter's own absence, since this
244///   crate has none to offer. `unwrap_last` says whether a FINAL optional step
245///   is gated too: an identity leaf delivers the reached value itself, so it
246///   is, while every other leaf hands the final step's full type to its own
247///   converter, `Option` and all.
248/// * **Refusal.** `absent` is `None` at a site that cannot express absence at
249///   all — a single delivered return value, which has no arm to put one in.
250///   An optional step before the end is then refused rather than composed into
251///   code the consumer's crate cannot type-check.
252/// * **Ownership**: the reached place is moved when it is ours, cloned when
253///   it is a field read of a place that is not, and borrowed otherwise.
254///
255/// One derivation, used by every delivery an adapter renders and by the
256/// single-leaf shortcut a decomposed return takes. They drifted once while
257/// they were two — the shortcut was missing the field clone and handed `&F` to
258/// an `F` converter — and a second pair, ungated here and gated in JniGen,
259/// drifted the same way until #607 folded them together.
260pub fn reach_leaf<L: DecomposedLeaf>(
261    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
262    at: LeafAt<'_, L>,
263    absent: Option<&dyn Fn() -> TokenStream>,
264    body: &dyn Fn(TokenStream) -> TokenStream,
265) -> TokenStream {
266    reach_leaf_at(qualify, at, absent, 0, body)
267}

269fn reach_leaf_at<L: DecomposedLeaf>(
270    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
271    at: LeafAt<'_, L>,
272    absent: Option<&dyn Fn() -> TokenStream>,
273    depth: usize,
274    body: &dyn Fn(TokenStream) -> TokenStream,
275) -> TokenStream {
276    let LeafAt {
277        path,
278        ref base,
279        base_is_ref,
280        unwrap_last,
281        ..
282    } = at;
283    // Every optional step on the way to the leaf becomes a `match` whose
284    // `None` arm is the adapter's absent value. A site with no absent value to
285    // give — a single delivered return, which has no arm to put one in —
286    // passes `None`, and an optional step before the end is then refused
287    // rather than composed into code the consumer's crate cannot type-check.
288    if let Some(absent) = absent {
289        let limit = if unwrap_last {
290            path.len()
291        } else {
292            // A non-identity leaf's converter takes the final step's FULL type,
293            // `Option` included, so only the steps before it are nesting.
294            path.len().saturating_sub(1)
295        };
296        let (projected, lead) = project_leading_fields(base, base_is_ref, path);
297        if let Some(k) = (lead..limit).find(|&i| path[i].is_optional()) {
298            // Through the optional step INCLUSIVE: the same fold, so the borrow
299            // in front of it is the ordinary rule rather than a second
300            // statement of it.
301            let opt_e = fold_steps(qualify, &path[lead..=k], projected, false);
302            let nested = format_ident!("__n{}", depth);
303            let inner = reach_leaf_at(
304                qualify,
305                LeafAt {
306                    path: &path[k + 1..],
307                    base: quote!(#nested),
308                    base_is_ref: true,
309                    ..at
310                },
311                Some(absent),
312                depth + 1,
313                body,
314            );
315            let gone = absent();
316            // A FIELD read composes to a borrow (`&(e).f`), so it goes through
317            // a coercion site and the destructuring stops caring which
318            // representation the source spelled the optional as (#268).
319            //
320            // A CALL composes to the accessor's own returned value, which is
321            // owned and whose payload downstream may move. Borrowing it to
322            // coerce would change that ownership, so it keeps its direct match.
323            if path[k].is_field() {
324                let opt_bind = format_ident!("__o{}", depth);
325                return quote! {
326                    {
327                        let #opt_bind: &::core::option::Option<_> = #opt_e;
328                        match #opt_bind {
329                            ::core::option::Option::Some(#nested) => { #inner }
330                            ::core::option::Option::None => #gone,
331                        }
332                    }
333                };
334            }
335            return quote! {
336                match #opt_e {
337                    ::core::option::Option::Some(#nested) => { #inner }
338                    ::core::option::Option::None => #gone,
339                }
340            };
341        }
342        return body(reached_place(qualify, &at));
343    }
344    // An optional step BEFORE the last one needs a `match` whose `None` arm has
345    // somewhere to go. This derivation has none — it yields a plain Rust value,
346    // not a representation that can carry absence — so the shape is refused
347    // here rather than composed into code that cannot type-check in the
348    // consumer's crate.
349    //
350    // Asked of the leaf's OWN path, not of `path`. The caller may hand a
351    // suffix: `wrapper.rs` rebases onto a hoisted local, and `Hoisted::innermost`
352    // strips the prefix that bound it — including any optional step inside it.
353    // Checking the parameter would therefore pass exactly when the hoist is the
354    // conditional one, which is the case that cannot compose (an `Option<T>`
355    // local with a field read hung off it). The full path is what the shape
356    // question is about.
357    let own_path = at.leaf.reach();
358    assert!(
359        !own_path.iter().rev().skip(1).any(PathStep::is_optional),
360        "unfold: leaf `{}` reaches through an optional step but is \
361         delivered as a single return value, which has no `None` arm — this \
362         shape needs callback delivery",
363        at.leaf.name(),
364    );
365    // Whether what this leaf reaches is OURS, and so is moved rather than
366    // borrowed or cloned. The two leaf kinds say it differently:
367    //
368    // * an IDENTITY leaf carries the answer in its `out_ty` — the plan resolved
369    //   it to the owned type exactly when the value is the plan's to give away
370    //   (`place_is_owned`: an owned root, or a field of a CONSUMING value form),
371    //   and that is also what selected the owning converter, which boxes the
372    //   move rather than cloning a borrow;
373    // * a FIELD leaf's `out_ty` is the field type as written, owned either way,
374    //   so ownership is the enclosing form's: only a consuming one gives its
375    //   fields away.
376    //
377    // How to project that place is `steps_are_movable`'s question, and it is
378    // asked there rather than restated here. This used to spell it
379    // `all(is_plain_field)`, defending the restatement on the grounds that a
380    // trailing `Option` cannot reach return delivery anyway — true, and enforced
381    // in `single_return` (`core/unfold.rs`), which is precisely why a local
382    // restatement could disagree with the rule for as long as the invariant held
383    // somewhere else. `plan.rs` says two readings would drift and the
384    // disagreement would be a borrow handed to an owning converter; this is the
385    // second reading, removed.
386    body(reached_place(qualify, &at))
387}

389/// The place a leaf reaches, with the terminal treatment its ownership calls
390/// for: moved out when it is ours and the path projects a place, cloned out
391/// when it is a field read of a place that is not, and borrowed otherwise.
392fn reached_place<L: DecomposedLeaf>(
393    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
394    at: &LeafAt<'_, L>,
395) -> TokenStream {
396    let LeafAt {
397        leaf,
398        path,
399        base,
400        base_is_ref,
401        consuming,
402        ..
403    } = at;
404    let (base_is_ref, consuming) = (*base_is_ref, *consuming);
405    let reached_is_ours = if leaf.identity() {
406        !matches!(
407            leaf.source().kind(),
408            prebindgen_flat::flat::TypeKind::Ref { .. }
409        )
410    } else {
411        consuming
412    };
413    if reached_is_ours && steps_are_movable(path) {
414        let segs: Vec<&syn::Ident> = path.iter().map(PathStep::ident).collect();
415        return quote!(#base #(.#segs)*);
416    }
417    let (e, lead) = project_leading_fields(base, base_is_ref, path);
418    let e = fold_steps(qualify, &path[lead..], e, false);
419    // Cloned out of the BORROW the reach yields, not out of the place behind
420    // it. The two agree for a field of an owned type and disagree for a field
421    // that is itself a reference: `place.clone()` there resolves through the
422    // reference and deep-clones the pointee, where the converter takes the
423    // field type as written. JniGen's own reach spelled it the short way and
424    // no declared field exercised the difference — the drift this step exists
425    // to close.
426    if leaf.is_field_read() {
427        quote!((#e).clone())
428    } else {
429        e
430    }
431}
432/// Every value form on a plan, evaluated **once** and bound to a local
433/// (`__vf0`, `__vf1`, …), so a struct is built once per delivery rather than
434/// once per field. The bound prefixes come back with the statements, since
435/// reaching a leaf means starting from the innermost local it sits under.
436///
437/// Shared by both delivery paths — the multi-leaf encoder below and the
438/// single-leaf `Delivery::Return` shortcut in `emit/wrapper.rs`. The shortcut
439/// used to compose its reach straight off the raw value, which for a consuming
440/// value form emitted `f(&v)` against a by-value receiver: ill-typed Rust in
441/// the consumer's crate. One binder, so the two cannot disagree about what a
442/// hoist is or who owns it.
443pub struct Hoisted {
444    /// The `let __vfN = …;` bindings, outermost-first.
445    pub stmts: TokenStream,
446    /// Each hoist's path prefix and the local it was bound to.
447    bound: Vec<(Vec<PathStep>, syn::Ident)>,
448    /// Whether each bound hoist consumed the value it decomposed.
449    consuming: Vec<bool>,
450    /// Whether each bound local is `Option<TStruct>` rather than `TStruct` —
451    /// the hoist sits under an optional step, so the value form ran only where
452    /// the value was present. Its leaves cannot be emitted as independent
453    /// statements: they share ONE `match` on the local (see
454    /// [`encode_plan_leaves`]), taken by value, so a consuming form's fields
455    /// still move out inside the arm.
456    optional: Vec<bool>,
457}
458impl Hoisted {
459    /// Index of the innermost bound hoist `path` sits under, with that prefix
460    /// already consumed. `None` for a leaf under no value form at all — a
461    /// sibling `.field()` / `.field_self()`, which still reaches from the value
462    /// itself.
463    fn innermost(&self, path: &[PathStep]) -> Option<(usize, Vec<PathStep>)> {
464        self.bound
465            .iter()
466            .enumerate()
467            .filter(|(_, (p, _))| p.len() < path.len() && path.starts_with(p))
468            .max_by_key(|(_, (p, _))| p.len())
469            .map(|(i, (p, _))| (i, path[p.len()..].to_vec()))
470    }
471
472    /// The innermost bound local `path` sits under, with that prefix already
473    /// consumed, and whether that hoist gave its value away.
474    pub fn rebase(&self, path: &[PathStep]) -> Option<(syn::Ident, Vec<PathStep>, bool)> {
475        self.innermost(path)
476            .map(|(i, rest)| (self.bound[i].1.clone(), rest, self.consuming[i]))
477    }
478
479    /// The innermost **conditional** hoist `path` sits under: its index, the
480    /// local holding the `Option`, the name its `Some` arm binds, and the steps
481    /// left to reach the leaf from there. `None` when the leaf's innermost
482    /// hoist is unconditional (or there is none) — then [`Self::rebase`]
483    /// applies and the leaf is an ordinary standalone statement.
484    pub fn conditional(
485        &self,
486        path: &[PathStep],
487    ) -> Option<(usize, syn::Ident, syn::Ident, Vec<PathStep>)> {
488        let (i, rest) = self.innermost(path)?;
489        self.optional[i].then(|| (i, self.bound[i].1.clone(), format_ident!("__u{}", i), rest))
490    }
491
492    /// The local a hoist was bound to.
493    pub fn local(&self, i: usize) -> syn::Ident {
494        self.bound[i].1.clone()
495    }
496
497    /// Whether a hoist consumed the value it decomposed.
498    pub fn consumed(&self, i: usize) -> bool {
499        self.consuming[i]
500    }
501}
502/// Fold `path` over `base` the way an adapter's own gated reach does, but
503/// yielding an `Option<…>` rather than the adapter's absent value: the optional
504/// steps become a `map`/`and_then` chain, so an absent value short-circuits to
505/// `None` instead of to whatever that adapter uses for absence. `body` renders
506/// the innermost reached expression as a BARE value — the chain's last link
507/// wraps it.
508///
509/// The gated reach this mirrors is [`reach_leaf`], which takes the absent
510/// value from the adapter rather than choosing one. This one yields an
511/// `Option` instead, because what it binds is a hoist rather than a delivered
512/// leaf: the value form runs where the value is present, and the `None` the
513/// chain produces is the local's own.
514///
515/// This is how a CONDITIONAL value form is bound — the accessor runs only where
516/// the value it decomposes is actually present.
517fn reach_optional(
518    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
519    path: &[PathStep],
520    base: TokenStream,
521    base_is_ref: bool,
522    depth: usize,
523    body: &dyn Fn(TokenStream) -> TokenStream,
524) -> TokenStream {
525    let (e, lead) = project_leading_fields(&base, base_is_ref, path);
526    match (lead..path.len()).find(|&i| path[i].is_optional()) {
527        None => body(fold_steps(qualify, &path[lead..], e, false)),
528        Some(k) => {
529            // Through the optional step INCLUSIVE: the same fold, so the
530            // borrow in front of it is the ordinary rule rather than a second
531            // statement of it.
532            let opt_e = fold_steps(qualify, &path[lead..=k], e, false);
533            let bind = format_ident!("__hb{}", depth);
534            // What the arm binds is the step's own value: an OWNED payload is a
535            // bare `T`, so composing the next step onto it directly would hand
536            // `T` to an accessor typed for `&T`. Say it is not a reference and
537            // let `project_leading_fields` borrow it; a borrowed payload is
538            // already one and passes through.
539            //
540            // With NO steps left the binding goes to `body` untouched — that is
541            // what lets a consuming value form MOVE an owned payload rather than
542            // borrow it straight back, so the terminal case stays "already a
543            // reference" whatever the payload is.
544            let rest = &path[k + 1..];
545            let inner = reach_optional(
546                qualify,
547                rest,
548                quote!(#bind),
549                rest.is_empty() || !path[k].yields_owned(),
550                depth + 1,
551                body,
552            );
553            // `map` when this is the LAST optional step (the body yields a bare
554            // value) and `and_then` when another follows (the recursion yields
555            // an `Option` that must not nest). The equivalent `match` reads the
556            // same but generated code runs through the consumer's lints, where
557            // `clippy::manual_map` is a denial.
558            let combinator = if rest.iter().any(PathStep::is_optional) {
559                format_ident!("and_then")
560            } else {
561                format_ident!("map")
562            };
563            quote! {
564                #opt_e.#combinator(|#bind| #inner)
565            }
566        }
567    }
568}
569pub fn bind_hoists(
570    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
571    hoists: &[crate::unfold::Hoist],
572    value: &TokenStream,
573    by_ref: bool,
574) -> Hoisted {
575    let mut out = Hoisted {
576        stmts: TokenStream::new(),
577        bound: Vec::new(),
578        consuming: Vec::new(),
579        optional: Vec::new(),
580    };
581    // Value forms COMPOSE, so each hoist is built from the longest hoist that
582    // is already a proper prefix of it (they arrive outermost-first), and from
583    // `value` otherwise.
584    for (i, h) in hoists.iter().enumerate() {
585        let local = format_ident!("__vf{}", i);
586        // A hoist under an optional step binds `Option<TStruct>`: the value
587        // form runs in the `Some` arm only. Core refuses to nest these, so the
588        // enclosing value is always the plan's own — no rebase to consider.
589        if h.prefix.iter().any(PathStep::is_optional) {
590            let (last, lead) = h
591                .prefix
592                .split_last()
593                .expect("a hoist prefix ends in its value-form call");
594            let consuming = h.consuming;
595            // The value form is handed the payload only when the optional step
596            // is the LAST thing before it; any step in between composes as a
597            // borrow, so what arrives is a reference either way.
598            let owned = lead.last().is_some_and(PathStep::yields_owned);
599            let expr = reach_optional(qualify, lead, value.clone(), by_ref, 0, &|reached| {
600                compose_value_form_call(qualify, last, reached, owned, consuming)
601            });
602            out.stmts.extend(quote! { let #local = #expr; });
603            out.bound.push((h.prefix.clone(), local));
604            out.consuming.push(h.consuming);
605            out.optional.push(true);
606            continue;
607        }
608        // Where the fold starts, and whether what it starts from is OWNED. The
609        // value form's own boundary is decided below, from what the fold ends
610        // up holding — never from where it began.
611        let (from, start, start_owned) = match out.rebase(&h.prefix) {
612            // A NESTED consuming form is handed the parent's field by MOVE: a
613            // hoisted value form is an owned struct and its fields are
614            // disjoint, so moving one out leaves every sibling leaf readable.
615            // `compose_step` borrows (`&(e).f`), so a plain field run to that
616            // field is projected here instead of going through it.
617            Some((outer, rest, _))
618                if h.consuming && rest[..rest.len() - 1].iter().all(PathStep::is_plain_field) =>
619            {
620                let lead = &rest[..rest.len() - 1];
621                let segs: Vec<&syn::Ident> = lead.iter().map(PathStep::ident).collect();
622                (h.prefix.len() - 1, quote!(#outer #(.#segs)*), true)
623            }
624            // Any other rebased hoist: project its own leading field run
625            // DIRECTLY off the parent local rather than reaching it through a
626            // borrow of the parent. A sibling hoist may already have moved a
627            // different field out — that is what a consuming value form does —
628            // and `&(&__vf0).wrapper` borrows the partially moved parent as a
629            // whole where `&__vf0.wrapper` is a disjoint borrow that survives.
630            // Same invariant `project_leading_fields` states for leaf reaches,
631            // and the same reason.
632            Some((outer, rest, _)) => {
633                let (e, lead) = project_leading_fields(&quote!(#outer), false, &rest);
634                (h.prefix.len() - rest.len() + lead, e, false)
635            }
636            None if by_ref => (0, value.clone(), false),
637            None => (0, value.clone(), true),
638        };
639        // Everything before the value form is an ordinary accessor chain.
640        let last = h.prefix.len() - 1;
641        let head = &h.prefix[from..last];
642        let e = fold_steps(qualify, head, start, start_owned);
643        let e_owned = head.last().map_or(start_owned, PathStep::yields_owned);
644        // The value form itself. A CONSUMING one takes its receiver BY VALUE —
645        // that is the move the whole declaration exists for — so it is handed
646        // what the fold holds when that is ours, and a clone when it is not:
647        // the same cost the borrowing form of the accessor would have paid,
648        // which keeps one declaration usable by both owned and `&T` returns.
649        // A borrowing one takes a reference, so an owned value is borrowed.
650        //
651        // Deciding this from the fold's RESULT rather than from its start is
652        // what lets a consuming form sit behind ordinary accessors: the chain
653        // in front borrows, the form itself still moves.
654        let expr = compose_value_form_call(qualify, &h.prefix[last], e, e_owned, h.consuming);
655        out.stmts.extend(quote! { let #local = #expr; });
656        out.bound.push((h.prefix.clone(), local));
657        out.consuming.push(h.consuming);
658        out.optional.push(false);
659    }
660    out
661}
