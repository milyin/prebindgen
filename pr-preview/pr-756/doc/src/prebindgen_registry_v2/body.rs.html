1//! The instructions the registry composes and the common Rust writer renders.
2//!
3//! The chapters leave this vocabulary named but undefined (`ConversionBodyId`,
4//! `FunctionBodyId`); this is the concrete form the first increment settles on.
5//! Three instructions cover the scalar and owned-struct paths: apply a target
6//! operation, construct a source struct, call the source function. Every value
7//! is a [`ValueId`] — an identity, not a name. Names are allocated once, by the
8//! writer, from definition order, which is why two operations rendered into one
9//! wrapper cannot collide.
10
11use crate::target::PrimitiveId;
12
13/// One runtime value inside one body. Not a variable name: the writer chooses
14/// those.
15#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
16pub struct ValueId(pub(crate) u32);
17
18/// What supplies an operand.
19#[derive(Clone, Debug)]
20pub enum Operand {
21    /// A value this body already has.
22    Value(ValueId),
23    /// A runtime context the boundary supplies under this name.
24    Context(String),
25}
26
27/// One step of a conversion or a wrapper.
28#[derive(Clone, Debug)]
29pub enum Instr {
30    /// Apply a registered operation, binding its result when it has one.
31    Apply {
32        primitive: PrimitiveId,
33        operands: Vec<Operand>,
34        result: Option<ValueId>,
35    },
36    /// Build a source struct from already-converted parts, in field order.
37    Construct {
38        /// The struct's declared name; the writer takes its field shape from
39        /// the model, so a tuple struct is constructed as one.
40        name: String,
41        parts: Vec<ValueId>,
42        result: ValueId,
43    },
44    /// Call the source function once, binding its result unless it returns
45    /// nothing — a `let` over a unit call names a value no one can use.
46    Call {
47        function: String,
48        args: Vec<ValueId>,
49        result: Option<ValueId>,
50    },
51}
52
53/// One instruction, and the conditions under which it exists at all.
54///
55/// `conditions` is empty for all but a field whose `#[cfg]` the capture reader
56/// could not answer: every instruction serving such a field carries it, so the
57/// statements the writer renders appear exactly where the field does. Nothing
58/// reads them — they are the tokens the source wrote, conjoined by Rust when
59/// there is more than one.
60#[derive(Clone, Debug)]
61pub struct Step {
62    pub instr: Instr,
63    pub conditions: Vec<proc_macro2::TokenStream>,
64}
65
66impl Step {
67    fn new(instr: Instr) -> Self {
68        Step {
69            instr,
70            conditions: Vec::new(),
71        }
72    }
73}
74
75/// A body under construction, with its own value-identity allocator.
76#[derive(Debug, Default)]
77pub struct BodyBuilder {
78    next: u32,
79    instrs: Vec<Step>,
80}
81
82impl BodyBuilder {
83    pub fn new() -> Self {
84        Self::default()
85    }
86
87    /// A value identity nothing else in this body uses.
88    pub fn fresh(&mut self) -> ValueId {
89        let id = ValueId(self.next);
90        self.next += 1;
91        id
92    }
93
94    pub fn push(&mut self, instr: Instr) {
95        self.instrs.push(Step::new(instr));
96    }
97
98    pub fn instrs(&self) -> &[Step] {
99        &self.instrs
100    }
101
102    pub fn into_instrs(self) -> Vec<Step> {
103        self.instrs
104    }
105
106    /// How many instructions this body already has, so a caller can name the
107    /// range it is about to add.
108    pub fn len(&self) -> usize {
109        self.instrs.len()
110    }
111
112    pub fn is_empty(&self) -> bool {
113        self.instrs.is_empty()
114    }
115
116    /// Put `conditions` on every instruction added since `from`.
117    ///
118    /// The caller is whoever knows what a stretch of instructions is *for* — a
119    /// part's conversion, say. Conditions accumulate rather than replace, so a
120    /// conditional field of a conditional field carries both.
121    pub fn condition(&mut self, from: usize, conditions: &[proc_macro2::TokenStream]) {
122        if conditions.is_empty() {
123            return;
124        }
125        for step in &mut self.instrs[from..] {
126            step.conditions.extend(conditions.iter().cloned());
127        }
128    }
129}

131/// A planned conversion's instructions, as a template.
132///
133/// [`NodeBody::carrier`] is the value handed in when the conversion is used;
134/// [`NodeBody::result`] is what it produces. An identity conversion is the
135/// degenerate case — no instructions, and the result *is* the carrier — which
136/// is what makes a scalar child render nothing.
137#[derive(Clone, Debug)]
138pub struct NodeBody {
139    pub carrier: ValueId,
140    pub instrs: Vec<Step>,
141    pub result: ValueId,
142}
143
144impl NodeBody {
145    /// Inline this conversion into `out`, with `carrier` supplying its input.
146    ///
147    /// Value identities are local to a template, so they are remapped as they
148    /// are met: an operand must already be mapped, and a result is allocated
149    /// fresh in the destination body.
150    pub fn inline(&self, carrier: ValueId, out: &mut BodyBuilder) -> ValueId {
151        let mut map = std::collections::HashMap::new();
152        map.insert(self.carrier, carrier);
153        let value = |map: &std::collections::HashMap<ValueId, ValueId>, id: ValueId| {
154            *map.get(&id)
155                .expect("a body uses no value before defining it")
156        };
157        for step in &self.instrs {
158            let instr = match &step.instr {
159                Instr::Apply {
160                    primitive,
161                    operands,
162                    result,
163                } => {
164                    let operands = operands
165                        .iter()
166                        .map(|operand| match operand {
167                            Operand::Value(id) => Operand::Value(value(&map, *id)),
168                            Operand::Context(name) => Operand::Context(name.clone()),
169                        })
170                        .collect();
171                    let result = result.map(|id| {
172                        let fresh = out.fresh();
173                        map.insert(id, fresh);
174                        fresh
175                    });
176                    Instr::Apply {
177                        primitive: *primitive,
178                        operands,
179                        result,
180                    }
181                }
182                Instr::Construct {
183                    name,
184                    parts,
185                    result,
186                } => {
187                    let parts = parts.iter().map(|id| value(&map, *id)).collect();
188                    let fresh = out.fresh();
189                    map.insert(*result, fresh);
190                    Instr::Construct {
191                        name: name.clone(),
192                        parts,
193                        result: fresh,
194                    }
195                }
196                Instr::Call {
197                    function,
198                    args,
199                    result,
200                } => {
201                    let args = args.iter().map(|id| value(&map, *id)).collect();
202                    let result = result.map(|id| {
203                        let fresh = out.fresh();
204                        map.insert(id, fresh);
205                        fresh
206                    });
207                    Instr::Call {
208                        function: function.clone(),
209                        args,
210                        result,
211                    }
212                }
213            };
214            let from = out.len();
215            out.push(instr);
216            out.condition(from, &step.conditions);
217        }
218        value(&map, self.result)
219    }
220}
