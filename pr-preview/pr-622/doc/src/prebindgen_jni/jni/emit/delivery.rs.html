1//! Output-expansion delivery: unfold plans and leaf encoding.
2//!
3//! # What this module walks
4//!
5//! Nothing. The decomposition walk is the registry's: binding the hoists,
6//! deciding where a leaf's reach starts, gating each optional step, saying
7//! what is moved and what is cloned, recognising a sum's segment and gating
8//! it, and the `match` a conditional value form's leaves share.
9//!
10//! What is here is the target-language half — the [`DeliveryBridge`]
11//! implementation for [`FrozenDelivery`]: encoding a leaf, placing it as a
12//! call argument, what a slot holds when it is not filled, and what absence
13//! looks like — together with `jvalue` layout, local-frame sizing, cached
14//! method lookup and exception routing, which are JNI policy and stay.
15//!
16//! [`DeliveryBridge`]: prebindgen_registry::unfold::DeliveryBridge

25/// Emit the output-expansion delivery body (output phase) for a function
26/// marked `.expand_output()`. The return value (`__out`) is decomposed by the
27/// plan's accessor leaves, each encoded into a JVM `Object`, and all delivered
28/// to the foreign builder lambda (`__builder`) in a single `invoke` call whose
29/// `JObject` result becomes the wrapper's return.
30///
31/// **Borrow ordering** (the user's zero-copy rationale): the reference
32/// (non-identity) accessor leaves are encoded **first** — each leaf's converter
33/// performs the single JVM copy (`&str -> jstring`), ending its borrow into
34/// `__out` — then the identity/handle leaf is emitted **last**: an owned `T`
35/// return is **moved** into the handle (`Box::into_raw(Box::new(__out))`, no
36/// clone) once the borrows are gone; a `&T` return is **cloned** via the
37/// borrowed-opaque output converter. The builder args are assembled in declared
38/// leaf order regardless of encode order.
39///
40/// Shape handling: [`UnfoldShape::Base`] decomposes the returned value
41/// directly; [`UnfoldShape::Optional`] uses the registry's Optional-over-Product
42/// chain when available, yielding one presence wire plus the child's wires.
43/// Absence skips the builder and delivers null. Leaf wires may be object
44/// (JString/JByteArray/JObject — cast via `.into()`) or primitive (boxed to
45/// `java.lang.*` via the cached `box_helper_for_wire` runtime helpers).
46///
47/// [`UnfoldShape::Base`]: prebindgen_registry::unfold::UnfoldShape::Base
48/// [`UnfoldShape::Optional`]: prebindgen_registry::unfold::UnfoldShape::Optional
49pub(crate) fn emit_unfold_delivery(
50    plan: &prebindgen_registry::unfold::UnfoldPlan,
51    output: &crate::jni::fn_plan::UnfoldOutputPlan,
52    call_expr: &TokenStream,
53    on_err: &TokenStream,
54    emit: &prebindgen_registry::RustWriter,
55) -> TokenStream {
56    use prebindgen_registry::unfold::UnfoldShape;
57    let context = &output.delivery;
58
59    let n = plan.leaves.len();
60
61    // Builder-arg locals, one per leaf in declared order. The builder is a
62    // generated typed `<Source>Builder<out R>` fun interface — its `run`
63    // method ID is resolved once per process on the interface class
64    // ([`CachedIfaceMethod`]); primitives cross as raw typed jvalues.
65    let obj_idents: Vec<syn::Ident> = (0..n).map(|i| format_ident!("__obj{}", i)).collect();
66
67    // Return-site error path: route the message to the error sink, then return
68    // the wrapper's sentinel. Threads through the shared leaf encoder.
69    let fail = |msg: TokenStream| -> TokenStream {
70        quote! {
71            signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &#msg);
72            return #on_err;
73        }
74    };
75
76    // Encode a value's leaves (`__out`, a `Some`-bound `__inner`, or a Vec
77    // `__elem`) into `__obj0…__objN` (shared with the callback trampoline),
78    // yielding the per-leaf typed jvalue arg expressions.
79    let encode_leaves = |value: &TokenStream, optional: bool| {
80        let mut delivered = Delivered::planned(plan, output.wires.clone(), output.chain.clone());
81        delivered.optional = optional;
82        encode_plan_leaves(context, delivered, &obj_idents, value, &fail, emit)
83    };
84
85    // Cached-interface call statics for the builder / folder `run`.
86    let iface_statics = |spec: &IfaceSpec| -> TokenStream {
87        let fqn_lit = syn::LitStr::new(&spec.raw_slash_fqn(), Span::call_site());
88        let descr_lit = syn::LitStr::new(&spec.descr, Span::call_site());
89        quote! {
90            #[allow(non_upper_case_globals)]
91            static __CB_MID: ::prebindgen_jni_runtime::CachedIfaceMethod =
92                ::prebindgen_jni_runtime::CachedIfaceMethod::new();
93            const __CB_FQN: &str = #fqn_lit;
94            const __CB_DESCR: &str = #descr_lit;
95        }
96    };
97
98    // Common builder-invoke (typed `run`, `Object` return = the erased `R`).
99    // Used by `Decompose`/`Optional`; its success arm yields the result
100    // `JObject`, error arms route to the sink + return the wrapper's sentinel.
101    let builder_invoke = |arg_exprs: &[TokenStream]| -> TokenStream {
102        quote! {
103            match __CB_MID.call_object(
104                &mut env, __CB_FQN, "run", __CB_DESCR, &__builder, &[#(#arg_exprs),*],
105            ) {
106                ::core::result::Result::Ok(__o) => __o,
107                ::core::result::Result::Err(__e) => {
108                    // Clears any pending JVM exception so the sink call is safe.
109                    let _ = env.exception_describe();
110                    let __e2 = <__JniErr as ::core::convert::From<String>>::from(__e.to_string());
111                    signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &__e2.to_string());
112                    #on_err
113                }
114            }
115        }
116    };
117
118    // Decompose a value into leaves then invoke the builder once (`Decompose`/
119    // `Optional`).
120    let emit_decompose = |value: &TokenStream| -> TokenStream {
121        let (leaves, arg_exprs, _) = encode_leaves(value, false);
122        let invoke = builder_invoke(&arg_exprs);
123        quote! { #leaves #invoke }
124    };
125
126    // Iterable (fold) delivery — possibly wrapped in ONE `Optional` layer.
127    // `Vec<T>` folds the elements through the typed `<Element>Folder<A>.run(acc,
128    // …)`, threading `__acc` and returning the final accumulator; per element the
129    // fold args are either the element WHOLE (M4) or its decomposed leaves (M5),
130    // with `acc` the erased `A` (`Object`). `Option<Vec<T>>` additionally yields a
131    // null result for `None` (the fold is skipped).
132    let opt_iterable = match &plan.shape {
133        UnfoldShape::Iterable(_) => Some(false),
134        UnfoldShape::Optional((), inner) if matches!(**inner, UnfoldShape::Iterable(_)) => {
135            Some(true)
136        }
137        _ => None,
138    };
139    if let Some(optional) = opt_iterable {
140        let statics = iface_statics(
141            output
142                .iface
143                .as_deref()
144                .expect("folder interface spec derivable for a resolved plan"),
145        );
146        let fold_invoke = |arg_exprs: &[TokenStream]| -> TokenStream {
147            quote! {
148                __acc = match __CB_MID.call_object(
149                    &mut env, __CB_FQN, "run", __CB_DESCR, &__fold,
150                    &[jni::sys::jvalue { l: __acc.as_raw() }, #(#arg_exprs),*],
151                ) {
152                    ::core::result::Result::Ok(__o) => __o,
153                    ::core::result::Result::Err(__e) => {
154                        let _ = env.exception_describe();
155                        let __e2 = <__JniErr as ::core::convert::From<String>>::from(__e.to_string());
156                        signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &__e2.to_string());
157                        return #on_err;
158                    }
159                };
160            }
161        };
162
163        let loop_body = if plan.element.is_some() {
164            // Whole-element (M4): encode the element via its own converter —
165            // a raw typed jvalue for a primitive-wire element, a JObject
166            // otherwise (mirrors `leaf_is_prim`; the folder interface
167            // declares the matching typed param).
168            let pipeline = output
169                .element_pipeline
170                .as_ref()
171                .expect("whole-element fold carries its frozen output pipeline");
172            let elem_call = pipeline.invoke(quote!(__elem), emit);
173            let elem_conv = quote! {
174                match #elem_call {
175                    ::core::result::Result::Ok(__w) => __w,
176                    ::core::result::Result::Err(__e) => {
177                        signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &__e.to_string());
178                        return #on_err;
179                    }
180                }
181            };
182            let elem_wire = pipeline.wire().clone();
183            // Primitive-wire elements (including an opaque **handle**, whose wire
184            // is `jlong`) cross as a raw typed jvalue; object wires (String,
185            // arrays) cross as a `JObject`. Keyed purely on the wire shape —
186            // a handle's `Some(Handle)` projection still rides its `jlong`, and
187            // the folder interface declares the matching `Long` (raw) param.
188            let elem_is_prim = matches!(jni_field_access(&elem_wire), Some((_, _, false)));
189            let enc = format_ident!("__enc");
190            let (bind_obj, arg_expr) = if elem_is_prim {
191                let letter = jni_field_access(&elem_wire).unwrap().1;
192                (
193                    TokenStream::new(),
194                    quote!(jni::sys::jvalue { #letter: __enc }),
195                )
196            } else {
197                let cast = cast_wire_to_jobject(&enc, &elem_wire, &fail);
198                (
199                    quote! { let __obj: jni::objects::JObject = #cast; },
200                    quote!(jni::sys::jvalue { l: __obj.as_raw() }),
201                )
202            };
203            let invoke = fold_invoke(&[arg_expr]);
204            quote! {
205                let __enc = #elem_conv;
206                #bind_obj
207                #invoke
208            }
209        } else {
210            // Decomposed (M5): encode each element's leaves, fold over them.
211            let (leaves, arg_exprs, _) = encode_leaves(&quote!(__elem), false);
212            let invoke = fold_invoke(&arg_exprs);
213            quote! {
214                #leaves
215                #invoke
216            }
217        };
218        // Fold the elements of `__vec` into `__acc` (`into_iter()` yields the
219        // element type exactly as written — owned `T`, or `&T` for a borrow).
220        let fold = quote! {
221            let mut __acc = __acc;
222            for __elem in __vec.into_iter() {
223                #loop_body
224            }
225            __acc
226        };
227        // `Option<Vec<T>>`: `None` ⇒ null result; `Some(vec)` ⇒ fold. A bare
228        // `Vec<T>` folds the returned value directly.
229        return if optional {
230            quote! {
231                #statics
232                let __out = #call_expr;
233                match __out {
234                    ::core::option::Option::Some(__vec) => { #fold }
235                    ::core::option::Option::None => #on_err,
236                }
237            }
238        } else {
239            quote! {
240                #statics
241                let __vec = #call_expr;
242                #fold
243            }
244        };
245    }
246
247    match &plan.shape {
248        UnfoldShape::Base => {
249            let statics = iface_statics(
250                output
251                    .iface
252                    .as_deref()
253                    .expect("builder interface spec derivable for a registered declaration"),
254            );
255            let body = emit_decompose(&quote!(__out));
256            quote! {
257                #statics
258                let __out = #call_expr;
259                #body
260            }
261        }
262        UnfoldShape::Optional((), inner) => {
263            match **inner {
264                UnfoldShape::Base => {}
265                _ => panic!(
266                    "emit_unfold_delivery: Optional inner must be Base (scalar) or \
267                     Iterable (`Option<Vec<T>>`, handled above)"
268                ),
269            }
270            let statics = iface_statics(
271                output
272                    .iface
273                    .as_deref()
274                    .expect("builder interface spec derivable for a registered declaration"),
275            );
276            let (leaves, arg_exprs, present) = encode_leaves(&quote!(__out), true);
277            if let Some(present) = present {
278                let invoke = builder_invoke(&arg_exprs);
279                quote! {
280                    #statics
281                    let __out = #call_expr;
282                    #leaves
283                    if #present != 0 {
284                        #invoke
285                    } else {
286                        #on_err
287                    }
288                }
289            } else {
290                // A non-composed declaration keeps the established delivery
291                // path until its shape has a registry recipe.
292                let body = emit_decompose(&quote!(__inner));
293                quote! {
294                    #statics
295                    let __out = #call_expr;
296                    match __out {
297                        ::core::option::Option::Some(__inner) => { #body }
298                        ::core::option::Option::None => #on_err,
299                    }
300                }
301            }
302        }
303        UnfoldShape::Iterable(_) => {
304            unreachable!("Iterable delivery is handled by the `opt_iterable` branch above")
305        }
306    }
307}

309/// Cast an encoded wire local to a `JObject` for the erased `invoke`: object
310/// wires pass through / `.into()`; primitive wires box to `java.lang.*`.
311/// `fail(msg)` — `msg` an expression yielding `String` — produces the
312/// diverging on-error statements (sink + sentinel at a return site, `Err` in
313/// the trampoline). Returns an expression yielding `JObject`.
314pub(crate) fn cast_wire_to_jobject(
315    enc: &syn::Ident,
316    wire: &syn::Type,
317    fail: &dyn Fn(TokenStream) -> TokenStream,
318) -> TokenStream {
319    if is_jobject_wire(wire) {
320        quote!(#enc)
321    } else if matches!(jni_field_access(wire), Some((_, _, true))) {
322        quote!(#enc.into())
323    } else if let Some(helper) = box_helper_for_wire(wire) {
324        let on_fail = fail(quote!(__e));
325        quote! {
326            match ::prebindgen_jni_runtime::#helper(&mut env, #enc) {
327                ::core::result::Result::Ok(__o) => __o,
328                ::core::result::Result::Err(__e) => {
329                    #on_fail
330                }
331            }
332        }
333    } else {
334        panic!(
335            "jnigen unfold: leaf has unsupported wire `{}`",
336            wire.to_token_stream()
337        )
338    }
339}

341/// One delivery site, as the encoder sees it: what it hands out, and the value
342/// forms it evaluates once on the way.
343///
344/// Not an `UnfoldPlan`. A plan carries the delivery's shape as well — whether
345/// it folds, whether it is optional, what the builder's generic is — and none
346/// of that reaches this encoder, which writes only the statements that fill the
347/// slots. Naming exactly what it reads is what lets the caller assemble one
348/// from a recipe.
349pub(crate) struct Delivered<'a> {
350    /// The values handed out, in the order the builder receives them.
351    pub(crate) wires: std::rc::Rc<Vec<crate::jni::compile::OutWire>>,
352    /// The value forms to evaluate once and bind to a local, outermost first.
353    ///
354    /// Per **site** rather than per value: two values reached through one
355    /// accessor share its call, and that is the whole point of a hoist.
356    pub(crate) hoists: &'a [prebindgen_registry::unfold::Hoist],
357    /// Whether the delivered value is reached through a borrow.
358    pub(crate) by_ref: bool,
359    /// True when these leaves are the model-derived data-class Product itself.
360    pub(crate) fixed_product: bool,
361    /// Whether the fixed Product is wrapped in one outer Optional shape.
362    pub(crate) optional: bool,
363
364    /// Exact composed child handed down by an enclosing registry recipe.
365    pub(crate) chain: Option<crate::jni::compile::ComposedChain>,
366}

374/// Callback delivery facts frozen while the registry is available. Rust
375/// source types remain as opaque readings inside the wires/pipelines; the only
376/// syntax retained here is origin qualification and Flat alternative shape,
377/// both consumed with the writer by the final Invoke renderer.
378#[derive(Clone)]
379pub(crate) struct FrozenDelivery {
380    /// The site's own leaves, shared with the plan that froze them rather than
381    /// copied: a delivery context may reference canonical leaves, not own a
382    /// second conversion graph over them (#622 review).
383    wires: std::rc::Rc<Vec<crate::jni::compile::OutWire>>,
384    hoists: Vec<prebindgen_registry::unfold::Hoist>,
385    by_ref: bool,
386    fixed_product: bool,
387    optional: bool,
388    chain: Option<crate::jni::compile::ComposedChain>,
389    modules: std::collections::BTreeMap<String, syn::Path>,
390    sums: std::collections::BTreeMap<String, FrozenSum>,
391    /// Which call the encoded slots ride: a typed builder `run` taking
392    /// `jvalue`s, or a `fromParts` factory spelled with a signature string and
393    /// typed `JValue`s. The walk is the same either way — what differs is how a
394    /// primitive slot is held and handed over.
395    factory: bool,
396    /// Per-leaf JVM descriptors, by leaf name. Filled for the factory
397    /// convention only, from the same derivation the builder interface uses.
398    descriptors: std::collections::BTreeMap<String, String>,
399}

402    /// Every converter encoding these leaves calls: each leaf's own pipeline,
403    /// and the composed converter when the delivery has one.
404    pub(crate) fn calls(&self, out: &mut Vec<prebindgen_registry::write::ArtifactKey>) {
405        for wire in self.wires.iter() {
406            wire.calls(out);
407        }
408        if let Some(chain) = &self.chain {
409            out.push(prebindgen_registry::write::ArtifactKey::Operation(
410                chain.operation.clone(),
411            ));
412        }
413    }

434    /// The construction both conventions share: origin qualification for every
435    /// call step, and the sum model behind every selector. What a delivery
436    /// needs to render without asking the registry anything.
437    #[allow(clippy::too_many_arguments)]
438    fn build(
439        ext: &Declarations,
440        registry: &impl Conversions,
441        hoists: Vec<prebindgen_registry::unfold::Hoist>,
442        by_ref: bool,
443        fixed_product: bool,
444        optional: bool,
445        wires: std::rc::Rc<Vec<crate::jni::compile::OutWire>>,
446        chain: Option<crate::jni::compile::ComposedChain>,
447    ) -> Self {
448        let mut modules = std::collections::BTreeMap::new();
449        for step in hoists
450            .iter()
451            .flat_map(|hoist| &hoist.prefix)
452            .chain(wires.iter().flat_map(|wire| wire.reach()))
453        {
454            if let prebindgen_registry::unfold::PathStep::Call { ident, .. } = step {
455                modules
456                    .entry(ident.to_string())
457                    .or_insert_with(|| ext.fn_module(registry, ident));
458            }
459        }
460        let mut sums = std::collections::BTreeMap::new();
461        for wire in wires.iter().filter(|wire| wire.is_tag()) {
462            let prebindgen_registry::flat::TypeKind::Named { id, .. } =
463                wire.out_ty.unwrapped().kind()
464            else {
465                panic!("jnigen sum unfold: selector type is not named")
466            };
467            let ident = id.ident().unwrap_or_else(|| {
468                panic!(
469                    "jnigen sum unfold: selector type `{}` is not an identifier",
470                    id.name
471                )
472            });
473            let Some(prebindgen_registry::flat::Type::Variant(sum)) =
474                registry.flat().declared_type(&ident)
475            else {
476                panic!("jnigen sum unfold: no indexed sum `{ident}`")
477            };
478            let module = ext.fn_module(registry, &ident);
479            sums.entry(id.name.clone()).or_insert_with(|| FrozenSum {
480                source: syn::parse_quote!(#module::#ident),
481                model: sum.clone(),
482            });
483        }
484        Self {
485            wires,
486            hoists,
487            by_ref,
488            fixed_product,
489            optional,
490            chain,
491            modules,
492            sums,
493            factory: false,
494            descriptors: std::collections::BTreeMap::new(),
495        }
496    }

498    /// The delivery a **whole-object struct encode** renders through: the
499    /// struct's own frozen decomposition, handed to its Kotlin `fromParts`.
500    ///
501    /// No hoists and no chain — a data class's leaves are read straight off the
502    /// value, which arrives as `&Source`. What makes it a delivery at all is
503    /// that the leaves and the walk are the same ones a fixed-builder site
504    /// uses; only the call at the end differs.
505    pub(crate) fn for_value_struct(
506        ext: &Declarations,
507        registry: &impl Conversions,
508        wires: Vec<crate::jni::compile::OutWire>,
509    ) -> Option<Self> {
510        let descriptors = crate::jni::iface::leaf_descriptors(ext, &wires)?;
511        let mut frozen = Self::build(
512            ext,
513            registry,
514            Vec::new(),
515            true,
516            true,
517            false,
518            std::rc::Rc::new(wires),
519            None,
520        );
521        frozen.factory = true;
522        frozen.descriptors = frozen
523            .wires
524            .iter()
525            .map(|wire| wire.name.clone())
526            .zip(descriptors)
527            .collect();
528        Some(frozen)
529    }

537    /// How a converted **primitive** sits in its slot: the value itself under
538    /// the factory convention, wrapped in the `jvalue` member the descriptor
539    /// names under the builder's.
540    pub(crate) fn hold_prim(&self, letter: &syn::Ident, value: TokenStream) -> TokenStream {
541        match self.factory {
542            true => value,
543            false => quote!(jni::sys::jvalue { #letter: #value }),
544        }
545    }

547    /// A **selector**'s value in the form its slot holds — a tag's alternative
548    /// number, a presence flag's `1`.
549    ///
550    /// Asked here rather than spelled by the segment emitters, which build
551    /// these two slots themselves instead of through [`DeliveryBridge::encode`]
552    /// (a selector is assigned, not converted) and so would otherwise state the
553    /// slot convention a second time.
554    pub(crate) fn selector_value(
555        &self,
556        leaf: &crate::jni::compile::OutWire,
557        value: TokenStream,
558    ) -> TokenStream {
559        if self.factory {
560            return value;
561        }
562        let letter = match leaf.is_tag() {
563            true => format_ident!("i"),
564            false => format_ident!("z"),
565        };
566        self.hold_prim(&letter, value)
567    }

569    /// The JVM descriptor a non-primitive leaf's slot occupies, frozen with the

571    /// which resolves its method id from the interface descriptor instead.
572    pub(crate) fn object_descriptor(&self, leaf: &crate::jni::compile::OutWire) -> String {
573        match self.factory {
574            false => String::new(),
575            true => self
576                .descriptors
577                .get(&leaf.name)
578                .unwrap_or_else(|| {
579                    panic!(
580                        "frozen JNI factory has no descriptor for leaf `{}`",
581                        leaf.name
582                    )
583                })
584                .clone(),
585        }
586    }

603    /// The JVM signature of the `fromParts` factory these leaves feed —
604    /// each slot's descriptor in order, returning the class itself.
605    pub(crate) fn factory_signature(&self, java_class_name: &str) -> String {
606        let mut sig = String::from("(");
607        for wire in self.wires.iter() {
608            sig.push_str(&crate::jni::emit::sum_out::leaf_slot(self, wire).descriptor);
609        }
610        sig.push_str(&format!(")L{java_class_name};"));
611        sig
612    }

630impl prebindgen_registry::unfold::DeliveryBridge for FrozenDelivery {
631    type Leaf = crate::jni::compile::OutWire;
632
633    fn qualify(&self, ident: &syn::Ident) -> syn::Path {
634        self.modules
635            .get(&ident.to_string())
636            .unwrap_or_else(|| panic!("frozen JNI delivery has no origin for `{ident}`"))
637            .clone()
638    }
639
640    fn sum(
641        &self,
642        leaf: &crate::jni::compile::OutWire,
643    ) -> (syn::Path, &prebindgen_registry::flat::Variant) {
644        let prebindgen_registry::flat::TypeKind::Named { id, .. } = leaf.out_ty.unwrapped().kind()
645        else {
646            panic!("jnigen sum unfold: selector type is not named")
647        };
648        let sum = self
649            .sums
650            .get(&id.name)
651            .unwrap_or_else(|| panic!("frozen JNI delivery has no sum `{}`", id.name));
652        (sum.source.clone(), &sum.model)
653    }
654
655    /// A leaf's own encode: run its output converter on the reached Rust
656    /// value, then present the result the way its slot carries it. A
657    /// primitive-wire leaf is a typed `jvalue` and crosses with no JNI call at
658    /// all; every other leaf becomes a `JObject`, boxing where the typed `run`
659    /// descriptor declares an object.
660    ///
661    /// Both forms encode INSIDE `reach`, so an absent value on the way to the
662    /// leaf yields this adapter's absence — a JVM null — rather than a value
663    /// the walk invented.
664    fn encode(
665        &self,
666        leaf: &crate::jni::compile::OutWire,
667        index: usize,
668        slot: &syn::Ident,
669        reach: &prebindgen_registry::unfold::Reach<'_>,
670        fail: &dyn Fn(TokenStream) -> TokenStream,
671        emit: &prebindgen_registry::RustWriter,
672    ) -> TokenStream {
673        let pipeline = match &leaf.abi {
674            Some(crate::jni::compile::OutAbi::Value(value)) => &value.pipeline,
675            Some(crate::jni::compile::OutAbi::Tag) | Some(crate::jni::compile::OutAbi::Present) => {
676                unreachable!("a selector is encoded with its own segment")
677            }
678            None => panic!(
679                "jnigen delivery: leaf `{}` reached Rust rendering without a frozen output ABI",
680                leaf.name
681            ),
682        };
683        let wire = pipeline.wire().clone();
684        let conv_fail = fail(quote!(__e.to_string()));
685        let convert = |input: TokenStream| -> TokenStream {
686            let call = pipeline.invoke(input, emit);
687            quote! {
688                match #call {
689                    ::core::result::Result::Ok(__w) => __w,
690                    ::core::result::Result::Err(__e) => {
691                        #conv_fail
692                    }
693                }
694            }
695        };
696        let encoded = format_ident!("__enc{}", index);
697        if self.leaf_is_prim(leaf) {
698            let letter = jni_field_access(&wire)
699                .expect("leaf_is_prim guarantees a primitive wire")
700                .1;
701            let held = self.hold_prim(&letter, quote!(#encoded));
702            let expr = reach(&|reached| {
703                let converted = convert(quote!(#reached));
704                quote! {{
705                    let #encoded = #converted;
706                    #held
707                }}
708            });
709            let ty = crate::jni::emit::sum_out::leaf_slot(self, leaf).ty;
710            return quote! { let #slot: #ty = #expr; };
711        }
712        let cast = cast_wire_to_jobject(&encoded, &wire, fail);
713        let expr = reach(&|reached| {
714            let converted = convert(quote!(#reached));
715            quote! {{
716                let #encoded = #converted;
717                #cast
718            }}
719        });
720        quote! { let #slot: jni::objects::JObject = #expr; }
721    }
722
723    /// A slot is a `jvalue` for a primitive-wire leaf and a `JObject`
724    /// otherwise, and an unfilled one carries that shape's own empty value: a
725    /// zero of the right jvalue member, or a JVM null.
726    fn slot(&self, leaf: &crate::jni::compile::OutWire) -> prebindgen_registry::unfold::Slot {
727        let slot = crate::jni::emit::sum_out::leaf_slot(self, leaf);
728        prebindgen_registry::unfold::Slot {
729            ty: slot.ty,
730            default: slot.default,
731        }
732    }
733
734    /// A leaf whose value was not there delivers a JVM null. Every slot that
735    /// can be absent is an object slot for exactly that reason: a primitive
736    /// one has no null to carry.
737    fn absent(&self) -> TokenStream {
738        quote!(jni::objects::JObject::null())
739    }
740
741    /// How a filled slot rides the typed `run` call: a primitive-wire leaf IS
742    /// its jvalue, and every other leaf's `JObject` passes its raw pointer in
743    /// the `l` slot. Matches the descriptor [`crate::jni::iface`] derives for
744    /// the same leaf.
745    fn argument(&self, leaf: &crate::jni::compile::OutWire, slot: &syn::Ident) -> TokenStream {
746        match (self.factory, self.leaf_is_prim(leaf)) {
747            // A `fromParts` call is spelled with a signature string, so its
748            // arguments are typed `JValue`s: the wire value itself for a
749            // primitive, a borrow of the object otherwise.
750            (true, true) => quote!(jni::objects::JValue::from(#slot)),
751            (true, false) => quote!(jni::objects::JValue::Object(&#slot)),
752            (false, true) => quote!(#slot),
753            (false, false) => quote!(jni::sys::jvalue { l: #slot.as_raw() }),
754        }
755    }
756}

765        Self {
766            wires,
767            hoists: &plan.hoists,
768            by_ref: plan.by_ref,
769            chain,
770            fixed_product: plan.fixed_builder,
771            optional: plan.is_optional_base(),
772        }

776/// Encode a plan's leaves off `value` (`__out`, a `Some`-bound `__inner`, a Vec
777/// `__elem`, an owned callback arg, or a domain error `__de`) into the
778/// `obj_idents` locals, in declared-leaf order. Reference (non-identity)
779/// leaves are encoded first — ending their borrow into the value — and the
780/// identity leaf last (move owned / clone `&T`). Each leaf's value is reached
781/// by folding its accessor `path` over `value`; every `Option`-returning
782/// nesting step on the path wraps the rest in a `match Some/None` (`None` ⇒ a
783/// null leaf) — see [`reach_leaf`]. Error arms are produced by `fail` (see
784/// [`cast_wire_to_jobject`]). Shared by the return-delivery site
785/// ([`emit_unfold_delivery`]), the callback trampoline, and the domain-error
786/// arm of fallible externs (whose `fail` routes an encoding failure to the
787/// binding-error channel).
788pub(crate) fn encode_plan_leaves(
789    context: &FrozenDelivery,
790    site: Delivered<'_>,
791    obj_idents: &[syn::Ident],
792    value: &TokenStream,
793    fail: &dyn Fn(TokenStream) -> TokenStream,
794    emit: &prebindgen_registry::RustWriter,
795) -> (TokenStream, Vec<TokenStream>, Option<syn::Ident>) {
796    let Delivered {
797        wires,
798        hoists,
799        by_ref,
800        chain,
801        fixed_product,
802        optional,
803    } = site;
804    // Per-fn origin qualification: each accessor call is prefixed with the
805    // module of the crate that defines it (multi-source bindings).
806    let qualify = |id: &syn::Ident| -> syn::Path { context.qualify(id) };
807    let n = wires.len();
808
809    // The argument expression per leaf, in leaf order — how a filled slot
810    // rides the call, which is the bridge's answer rather than this loop's.
811    let mut arg_exprs: Vec<TokenStream> = Vec::with_capacity(n);
812    for (idx, leaf) in wires.iter().enumerate() {
813        arg_exprs.push(context.argument(leaf, &obj_idents[idx]));
814    }
815
816    // A fixed decomposition has one Product, Optional or Choice intermediate.
817    // Invoke its converter once, then adapt the intermediate leaves to the JNI
818    // call ABI. Callback and ordinary output delivery therefore share the
819    // same Rust-value walk; only the final delivery remains JNI-specific.
820    if fixed_product && hoists.is_empty() {
821        let chain = chain;
822        if let Some(chain) = chain.filter(|chain| match &chain.layout {
823            crate::jni::compile::JLayout::Optional(inner) => optional && inner.leaf_count() == n,
824            layout => !optional && layout.leaf_count() == n,
825        }) {
826            chain.activate();
827            let encoded: Vec<syn::Ident> = (0..n)
828                .map(|index| format_ident!("__chain_wire{index}"))
829                .collect();
830            let present = optional.then(|| format_ident!("__chain_present"));
831            let pattern_values: Vec<syn::Ident> = present
832                .iter()
833                .cloned()
834                .chain(encoded.iter().cloned())
835                .collect();
836            let pattern = chain.layout.pattern(&pattern_values);
837            let converter = emit.operation_ident("jni", &chain.operation);
838            let on_chain_error = fail(quote!(__chain_error.to_string()));
839            let mut stmts = quote! {
840                let #pattern = match #converter(&mut env, #value) {
841                    ::core::result::Result::Ok(__intermediate) => __intermediate,
842                    ::core::result::Result::Err(__chain_error) => {
843                        #on_chain_error
844                    }
845                };
846            };
847            for (index, leaf) in wires.iter().enumerate() {
848                let encoded = &encoded[index];
849                let object = &obj_idents[index];
850                if leaf.is_tag() {
851                    stmts.extend(quote! {
852                        let #object = jni::sys::jvalue { i: #encoded };
853                    });
854                    continue;
855                }
856                let wire = context.leaf_wire(leaf);
857                if context.leaf_is_prim(leaf) {
858                    let (_, member, _) = jni_field_access(&wire)
859                        .expect("a primitive Product leaf has a JNI jvalue member");
860                    stmts.extend(quote! {
861                        let #object = jni::sys::jvalue { #member: #encoded };
862                    });
863                } else {
864                    let cast = cast_wire_to_jobject(encoded, &wire, fail);
865                    stmts.extend(quote! {
866                        let #object: jni::objects::JObject = #cast;
867                    });
868                }
869            }
870            return (stmts, arg_exprs, present);
871        }
872    }
873    let hoisted = bind_hoists(&qualify, hoists, value, by_ref);
874    let mut stmts = hoisted.stmts.clone();
875
876    // Where each leaf's reach starts is the walk's answer, not this loop's:
877    // `Hoisted::place` decides which value form a leaf sits under, what is
878    // left of its path, and whether that form gave its value away.
879    let place = |leaf: &crate::jni::compile::OutWire| hoisted.place(leaf, value, by_ref);
880
881    // Which leaves form a segment is the plan's own answer, read by the
882    // walk: a selector plus the group leaves after it.
883    let sum_segments = prebindgen_registry::unfold::segments(&wires);
884
885    // Leaves under a conditional value form are collected per hoist and emitted
886    // below as ONE `match` on its `Option` local — the same treatment a sum's
887    // groups get, and for the same reason: their slots exist unconditionally
888    // but only one arm computes them. Built BEFORE the sum pass, because a
889    // conditional form may carry a sum field and that segment has to land in
890    // the arm too: emitted ahead of it, its `match` would reach a binding the
891    // arm has not introduced yet.
892    // Which arm a leaf belongs in is its place's answer, asked here the same
893    // way every other consumer asks it — a bucket exists exactly where some
894    // leaf reports one.
895    let mut cond_stmts: std::collections::BTreeMap<usize, TokenStream> = wires
896        .iter()
897        .filter_map(|leaf| place(leaf).conditional)
898        .map(|i| (i, TokenStream::new()))
899        .collect();
900
901    for seg in &sum_segments {
902        let at = place(&wires[seg.start]);
903        // The segment's own encode — which slot carries which alternative's
904        // value, and what the selector is set to — is this adapter's. Reaching
905        // the sum it is encoded from, and gating the whole segment when that
906        // reach passes through an optional step, is the walk's.
907        let group_args = std::cell::RefCell::new(Vec::new());
908        let group_stmts = prebindgen_registry::unfold::segment(
909            context,
910            &qualify,
911            &at,
912            seg.start,
913            &wires[seg.clone()],
914            &obj_idents[seg.clone()],
915            &|matched| {
916                // A segment is selected by a tag or by a presence flag, and
917                // the two fill their group differently: one alternative's
918                // payload, or the child's own leaves off the value the gate
919                // unwrapped.
920                let (stmts, args) = crate::jni::emit::encode_segment_group(
921                    context,
922                    &wires[seg.clone()],
923                    &obj_idents[seg.clone()],
924                    matched,
925                    seg.start,
926                    &qualify,
927                    fail,
928                    emit,
929                );
930                *group_args.borrow_mut() = args;
931                stmts
932            },
933        );
934        // The whole segment — its slot declarations and its `match` — is
935        // routed like any other leaf under the same form.
936        match at.conditional {
937            Some(i) => cond_stmts
938                .get_mut(&i)
939                .expect("a conditional leaf's hoist has a bucket")
940                .extend(group_stmts),
941            None => stmts.extend(group_stmts),
942        }
943        let group_args = group_args.into_inner();
944        for (k, e) in group_args.into_iter().enumerate() {
945            arg_exprs[seg.start + k] = e;
946        }
947    }
948
949    let in_sum = |i: usize| sum_segments.iter().any(|s| s.contains(&i));
950    let mut order: Vec<usize> = (0..n)
951        .filter(|&i| !wires[i].identity && !in_sum(i))
952        .collect();
953    order.extend((0..n).filter(|&i| wires[i].identity && !in_sum(i)));
954
955    for idx in order {
956        let leaf = &wires[idx];
957        let obj_ident = &obj_idents[idx];
958        let at = place(leaf);
959        // Route this leaf's statements: into its conditional arm, or straight
960        // out. The arm is the place's own answer. Shadows `stmts` for the rest
961        // of the body, so every `extend` below lands in the right place
962        // without knowing which case it is in.
963        let stmts: &mut TokenStream = match at.conditional {
964            Some(i) => cond_stmts.get_mut(&i).expect("collected above"),
965            None => &mut stmts,
966        };
967        let owned_place = at.owned(leaf);
968        let (value, by_ref, path, consuming) = (&at.base, at.base_is_ref, &at.path, at.consuming);
969        // Every reach below is the registry's, with this adapter's absence in
970        // its `None` arms. The terminal treatment — move, clone or borrow —
971        // comes with it, which is what let the three-way dispatch this loop
972        // used to make disappear.
973        let absent = || DeliveryBridge::absent(context);
974        let gated = |path: &[PathStep],
975                     base: TokenStream,
976                     base_is_ref: bool,
977                     unwrap_last: bool,
978                     body: &dyn Fn(TokenStream) -> TokenStream| {
979            prebindgen_registry::unfold::reach_leaf(
980                &qualify,
981                prebindgen_registry::unfold::LeafAt {
982                    leaf,
983                    path,
984                    base,
985                    base_is_ref,
986                    consuming,
987                    unwrap_last,
988                },
989                Some(&absent),
990                body,
991            )
992        };
993        let (frozen_pipeline, frozen_projection) = match &leaf.abi {
994            Some(crate::jni::compile::OutAbi::Value(value)) => {
995                (&value.pipeline, value.projection.as_ref())
996            }
997            Some(crate::jni::compile::OutAbi::Tag) | Some(crate::jni::compile::OutAbi::Present) => {
998                unreachable!("selector segments are encoded above")
999            }
1000            None => panic!(
1001                "jnigen delivery: leaf `{}` reached Rust rendering without a frozen output ABI",
1002                leaf.name
1003            ),
1004        };
1005        let projection = frozen_projection;
1006        let conv_fail = fail(quote!(__e.to_string()));
1007        let conv = |input: TokenStream| -> TokenStream {
1008            let call = frozen_pipeline.invoke(input, emit);
1009            quote! {
1010                match #call {
1011                    ::core::result::Result::Ok(__w) => __w,
1012                    ::core::result::Result::Err(__e) => {
1013                        #conv_fail
1014                    }
1015                }
1016            }
1017        };
1018
1019        // Bind `obj_ident` to a JObject-yielding `expr`.
1020        let bind_obj = |obj_ident: &syn::Ident, expr: TokenStream| -> TokenStream {
1021            quote! {
1022                let #obj_ident: jni::objects::JObject = #expr;
1023            }
1024        };
1025
1026        if leaf.identity {
1027            // Identity leaf: deliver the value itself. Its projection decides
1028            // how: a `ptr_class` Handle is cloned (`&T`, reached by the path)
1029            // or — at the root of an owned value — moved into a fresh Box,
1030            // and crosses as the RAW `jlong` (the receiver constructs the
1031            // typed class in bytecode — a native `new_object` would cost a
1032            // descriptor parse + FindClass + GetMethodID + NewObjectA per
1033            // delivery). A nullable handle (an `Option` nesting step on the
1034            // path) boxes to `java.lang.Long` / null. The whole path is `Option`-unwrapped
1035            // (`unwrap_last`): an optional nesting step makes the leaf null
1036            // when the value is absent.
1037            let proj = projection.unwrap_or_else(|| {
1038                panic!(
1039                    "jnigen unfold: identity leaf `{}` has no projection — \
1040                     `.accessor_record_id()` requires a ptr_class type",
1041                    leaf.out_ty.key()
1042                )
1043            });
1044            // `owned_place` — bound above from `LeafPlace::owned` — is the
1045            // place this handle lives when it is OURS to give away: the owned
1046            // root, or a field of a CONSUMING value form, which handed its
1047            // value over so its handle fields move out like every other field
1048            // rather than being cloned through the borrowed converter (which
1049            // would also demand a `Clone` the type need not have). Which it is
1050            // was decided in the plan and is read there, not restated here.
1051            match proj.kind {
1052                ProjectionKind::Handle => {
1053                    let handle_ident = format_ident!("__h{}", idx);
1054                    if let (Some(place), false) = (&owned_place, leaf.nullable) {
1055                        // Ours, and always present: move into a Box, raw jlong.
1056                        stmts.extend(quote! {
1057                            let #obj_ident: jni::sys::jvalue = jni::sys::jvalue {
1058                                j: std::boxed::Box::into_raw(std::boxed::Box::new(#place))
1059                                    as jni::sys::jlong,
1060                            };
1061                        });
1062                    } else if let (Some(place), false) =
1063                        (&owned_place, path.last().is_some_and(PathStep::is_optional))
1064                    {
1065                        // Ours, always present, but the SLOT is nullable — the
1066                        // leaf hangs off a conditional value form, so the absent
1067                        // case is the enclosing `match`'s other arm, not an
1068                        // `Option` here. Move into the Box and box the jlong, so
1069                        // both arms fill the slot with the same shape.
1070                        let box_fail = fail(quote!(__e.to_string()));
1071                        stmts.extend(bind_obj(
1072                            obj_ident,
1073                            quote! {{
1074                                let #handle_ident: jni::sys::jlong =
1075                                    std::boxed::Box::into_raw(std::boxed::Box::new(#place))
1076                                        as jni::sys::jlong;
1077                                match ::prebindgen_jni_runtime::box_jlong(&mut env, #handle_ident) {
1078                                    ::core::result::Result::Ok(__o) => __o,
1079                                    ::core::result::Result::Err(__e) => {
1080                                        #box_fail
1081                                    }
1082                                }
1083                            }},
1084                        ));
1085                    } else if let Some(place) = &owned_place {
1086                        // Ours, behind an `Option`: match the option BY VALUE so
1087                        // the present handle is moved into its Box, boxed
1088                        // `java.lang.Long` when present / JVM null when absent.
1089                        // Matching `&place` here is what used to clone it back
1090                        // through the borrowed converter.
1091                        let box_fail = fail(quote!(__e.to_string()));
1092                        stmts.extend(bind_obj(
1093                            obj_ident,
1094                            quote! {{
1095                                match #place {
1096                                    ::core::option::Option::Some(__n) => {
1097                                        let #handle_ident: jni::sys::jlong =
1098                                            std::boxed::Box::into_raw(std::boxed::Box::new(__n))
1099                                                as jni::sys::jlong;
1100                                        match ::prebindgen_jni_runtime::box_jlong(&mut env, #handle_ident) {
1101                                            ::core::result::Result::Ok(__o) => __o,
1102                                            ::core::result::Result::Err(__e) => {
1103                                                #box_fail
1104                                            }
1105                                        }
1106                                    }
1107                                    ::core::option::Option::None => jni::objects::JObject::null(),
1108                                }
1109                            }},
1110                        ));
1111                    } else if !leaf.nullable {
1112                        // Reached non-null handle: clone via the converter,
1113                        // raw jlong (no Option steps on the path).
1114                        let expr = gated(path, value.clone(), by_ref, true, &|reached| {
1115                            let __encoded = conv(quote!(#reached));
1116                            quote! {{
1117                                let #handle_ident: jni::sys::jlong = #__encoded;
1118                                jni::sys::jvalue { j: #handle_ident }
1119                            }}
1120                        });
1121                        stmts.extend(quote! {
1122                            let #obj_ident: jni::sys::jvalue = #expr;
1123                        });
1124                    } else {
1125                        // Nullable handle (Option nesting step): boxed
1126                        // `java.lang.Long` when present (cached valueOf),
1127                        // JVM null when absent — matching the `Long?` param.
1128                        let box_fail = fail(quote!(__e.to_string()));
1129                        let expr = gated(path, value.clone(), by_ref, true, &|reached| {
1130                            let __encoded = conv(quote!(#reached));
1131                            quote! {{
1132                                let #handle_ident: jni::sys::jlong = #__encoded;
1133                                match ::prebindgen_jni_runtime::box_jlong(&mut env, #handle_ident) {
1134                                    ::core::result::Result::Ok(__o) => __o,
1135                                    ::core::result::Result::Err(__e) => {
1136                                        #box_fail
1137                                    }
1138                                }
1139                            }}
1140                        });
1141                        stmts.extend(bind_obj(obj_ident, expr));
1142                    }
1143                }
1144                ProjectionKind::Unsigned64 => {
1145                    let enc_ident = format_ident!("__enc{}", idx);
1146                    let encode = |reached: TokenStream| {
1147                        let encoded = conv(reached);
1148                        quote! {{
1149                            let #enc_ident: jni::sys::jlong = #encoded;
1150                            jni::sys::jvalue { j: #enc_ident }
1151                        }}
1152                    };
1153                    if path.is_empty() && !by_ref {
1154                        let expr = encode(value.clone());
1155                        stmts.extend(quote! { let #obj_ident: jni::sys::jvalue = #expr; });
1156                    } else if !leaf.nullable {
1157                        let expr = gated(path, value.clone(), by_ref, true, &|reached| {
1158                            encode(quote!(*#reached))
1159                        });
1160                        stmts.extend(quote! { let #obj_ident: jni::sys::jvalue = #expr; });
1161                    } else {
1162                        let box_fail = fail(quote!(__e.to_string()));
1163                        let expr = gated(path, value.clone(), by_ref, true, &|reached| {
1164                            let __encoded = conv(quote!(*#reached));
1165                            quote! {{
1166                                let #enc_ident: jni::sys::jlong = #__encoded;
1167                                match ::prebindgen_jni_runtime::box_jlong(&mut env, #enc_ident) {
1168                                    ::core::result::Result::Ok(__o) => __o,
1169                                    ::core::result::Result::Err(__e) => {
1170                                        #box_fail
1171                                    }
1172                                }
1173                            }}
1174                        });
1175                        stmts.extend(bind_obj(obj_ident, expr));
1176                    }
1177                }
1178            }
1179            continue;
1180        }
1181
1182        // Leaf reach. An `Accessor` leaf walks its accessor-fn path — unwrapping
1183        // every `Option` nesting step (`None` ⇒ a null leaf). A `Field` leaf
1184        // (synthesized `data_class`) reaches a struct field and clones it
1185        // (`value.a.b.clone()`); the converter (`Option<Box<String>>` → nullable
1186        // String, …) carries any nullability, so there is no path `Option` to
1187        // unwrap. `reach(body)` dispatches on the source and feeds the reached
1188        // Rust expression to `body`.
1189        //
1190        // Which of the two it is, asked of the wire: a field read ends its
1191        // reach in a field step, an accessor's ends at the call. That is what
1192        // `LeafSource` said, and saying it off the reach keeps one answer.
1193        // A non-identity leaf's converter takes the final step's full type,
1194        // `Option` included, so the last step is not unwrapped here.
1195        let reach = |body: &dyn Fn(TokenStream) -> TokenStream| -> TokenStream {
1196            gated(path, value.clone(), by_ref, false, body)
1197        };
1198
1199        // The encode itself is the bridge's: this loop hands it the leaf's
1200        // reach and it says what the slot holds. Reaching is still decided
1201        // here — which is what the steps after this one move.
1202        stmts.extend(context.encode(leaf, idx, obj_ident, &reach, fail, emit));
1203    }
1204
1205    // Each conditional value form's leaves share one `match` on its `Option`
1206    // local — the walk's shape, filled with this adapter's slots.
1207    for (i, body) in cond_stmts {
1208        stmts.extend(prebindgen_registry::unfold::conditional_arm(
1209            context, &hoisted, i, &wires, obj_idents, body,
1210        ));
1211    }
1212    (stmts, arg_exprs, None)
1213}

1215fn frozen_leaf_wire(leaf: &crate::jni::compile::OutWire) -> syn::Type {
1216    match &leaf.abi {
1217        Some(crate::jni::compile::OutAbi::Tag) => syn::parse_quote!(jni::sys::jint),
1218        Some(crate::jni::compile::OutAbi::Present) => syn::parse_quote!(jni::sys::jboolean),
1219        Some(crate::jni::compile::OutAbi::Value(value)) => value.pipeline.wire().clone(),
1220        None => panic!("frozen JNI delivery leaf `{}` has no output ABI", leaf.name),
1221    }
1222}

1224/// True when a plan leaf crosses the typed `run` as a **raw primitive**
1225/// `jvalue`: non-nullable, no projection (not a handle), and a
1226/// primitive JNI wire. Must agree with the descriptor chunk
1227/// [`crate::jni::iface`] derives for the same leaf — a
1228/// nullable primitive boxes (object chunk), object wires pass as objects.
1229///
1230/// The synthesized sum selector is a `jint` by definition — it is assigned,
1231/// never converted — unless it is NULLABLE: the sum then sits under a
1232/// conditional value form, and the absent case needs a representation the
1233/// tag's own variants do not provide. A raw `jint` has none (zero is a real
1234/// variant), so the selector boxes like any other nullable leaf.
1235fn frozen_leaf_is_prim(leaf: &crate::jni::compile::OutWire) -> bool {
1236    if leaf.is_tag() {
1237        return !leaf.nullable;
1238    }
1239    // A presence flag is a `jboolean` the gate assigns — there is nothing to
1240    // box and no value behind it that could be absent.
1241    if matches!(leaf.abi, Some(crate::jni::compile::OutAbi::Present)) {
1242        return true;
1243    }
1244    if leaf.nullable {
1245        return false;
1246    }
1247    let Some(crate::jni::compile::OutAbi::Value(value)) = &leaf.abi else {
1248        panic!("frozen JNI delivery leaf `{}` has no output ABI", leaf.name)
1249    };
1250    let proj_ok = match &value.projection {
1251        None => true,
1252        Some(projection) => matches!(
1253            projection.kind,
1254            ProjectionKind::Handle | ProjectionKind::Unsigned64
1255        ),
1256    };
1257    proj_ok && matches!(jni_field_access(value.pipeline.wire()), Some((_, _, false)))
1258}

1260/// The wire half of [`leaf_is_prim`]: does a leaf of this type occupy a **raw
1261/// primitive** slot? Split out so the interface derivation can ask the question
1262/// about a leaf whose own `nullable` flag it is in the middle of computing (an
1263/// inert sum group slot).
1264pub(crate) fn leaf_ty_is_prim(
1265    ext: &Declarations,
1266    out_ty: &prebindgen_registry::flat::TypeRef,
1267) -> bool {
1268    let Some(entry) = ext.out_frag(out_ty) else {
1269        return false;
1270    };
1271    // No projection (plain primitive/enum wire) — or an opaque HANDLE, whose
1272    // converter's wire is the raw `jlong` the typed `run` declares as `Long`
1273    // (`J`): the receiver constructs the typed class in bytecode. A nullable
1274    // handle boxes to `java.lang.Long` instead (object chunk). Fixed-size
1275    // arrays and other object wires stay objects (for example, `[B`).
1276    let proj_ok = match &entry.metadata.projection {
1277        None => true,
1278        Some(p) => matches!(p.kind, ProjectionKind::Handle | ProjectionKind::Unsigned64),
1279    };
1280    proj_ok && matches!(jni_field_access(&entry.destination), Some((_, _, false)))
1281}
