15/// Stable, target-independent identifier for a canonical record schema.
16///
17/// This is a compatibility identifier, not a cryptographic authentication hash.
18pub fn schema_id(schema: &str) -> u64 {
19    schema
20        .as_bytes()
21        .iter()
22        .fold(0xcbf29ce484222325, |hash, byte| {
23            (hash ^ u64::from(*byte)).wrapping_mul(0x100000001b3)
24        })
25}
