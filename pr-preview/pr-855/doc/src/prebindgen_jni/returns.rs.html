7pub(crate) enum NativeReturn {
8    Unit,
9    Direct(JniWire),
10    Callback,
11}
12impl NativeReturn {
13    pub(crate) fn rust_type(&self) -> syn::Type {
14        match self {
15            Self::Direct(wire) => wire.native_return_type(),
16            _ => syn::parse_quote!(()),
17        }
18    }
19    pub(crate) fn kotlin_type(&self) -> &str {
20        match self {
21            Self::Direct(wire) => wire.kotlin_type(),
22            _ => "Unit",
23        }
24    }
25}
26pub(crate) fn packet_delivery(plan: &OutputPlan<JniWire>) -> Result<Delivery> {
27    let env = prebindgen_conversions::template_parameter("environment");
28    let count = crate::group::result_packet_fields(&plan.outcomes.success).len() + 1;
29    let encode = encode_array(
30        crate::group::result_packet(&plan.outcomes.success, &plan.values)?,
31        &syn::parse_quote!(result),
32        1,
33    );
34    let mark = prebindgen_conversions::template_parameter("mark_output");
35    Ok(Delivery {
36        statements: quote!(let result = ::prebindgen_jni_runtime::array(&mut #env, #count)?; #encode #mark),
37        result: quote!(result.into_raw()),
38    })
39}
40fn acknowledge_rows(plan: &OutputPlan<JniWire>) -> Result<TokenStream> {
41    if sequence::callback_delivery(&plan.outcomes.success)? {
42        let env = prebindgen_conversions::template_parameter("environment");
43        Ok(
44            quote!(#env.set_boolean_array_region(&__pb_output_accepted,0,&[1]).map_err(|e|e.to_string())?;),
45        )
46    } else {
47        Ok(TokenStream::new())
48    }
49}

51pub(crate) fn select(plan: &OutputPlan<JniWire>) -> Option<NativeReturn> {
52    let mut packet = packet_delivery(plan).ok()?;
53    packet.statements.extend(acknowledge_rows(plan).ok()?);
54    if plan.delivery.statements.to_string() != packet.statements.to_string()
55        || plan.delivery.result.to_string() != packet.result.to_string()
56    {
57        return None;
58    }
59    let fields = crate::group::result_packet_fields(&plan.outcomes.success).collect::<Vec<_>>();
60    Some(match fields.as_slice() {
61        [] => NativeReturn::Unit,
62        [wire] => NativeReturn::Direct(wire.kind.clone()),
63        _ => NativeReturn::Callback,
64    })
65}
66fn chunks(wires: &[&Wire<JniWire>]) -> Vec<std::ops::Range<usize>> {
67    let mut ranges = Vec::new();
68    let mut start = 0;
69    let mut slots = 0;
70    for (index, wire) in wires.iter().enumerate() {
71        let width = wire.kind.parameter_slots();
72        if slots + width > 254 {
73            ranges.push(start..index);
74            start = index;
75            slots = 0;
76        }
77        slots += width;
78    }
79    ranges.push(start..wires.len());
80    ranges
81}

83pub(crate) fn callback_delivery(
84    plan: &OutputPlan<JniWire>,
85    sink: &syn::Ident,
86    domain: bool,
87) -> Result<Delivery> {
88    let env = prebindgen_conversions::template_parameter("environment");
89    let fields = crate::group::result_packet(&plan.outcomes.success, &plan.values)?;
90    let wires = fields.iter().map(|(wire, _)| *wire).collect::<Vec<_>>();
91    let mut prepare = TokenStream::new();
92    let mut arguments = Vec::new();
93    for (i, (wire, value)) in fields.iter().enumerate() {
94        let argument = wire
95            .kind
96            .output_argument(value, &format_ident!("__pb_return_argument_{i}"));
97        prepare.extend(argument.statements);
98        arguments.push(argument.value);
99    }
100    let mut invoke = TokenStream::new();
101    for (index, range) in chunks(&wires).into_iter().enumerate() {
102        let descriptor = format!(
103            "({})V",
104            wires[range.clone()]
105                .iter()
106                .map(|w| w.kind.jvm_descriptor())
107                .collect::<String>()
108        );
109        let arguments = &arguments[range];
110        let method = format!("{}{index}", if domain { "domainError" } else { "success" });
111        if domain {
112            invoke.extend(quote!(unsafe { ::prebindgen_jni_runtime::invoke_error_callback(&mut #env, &#sink, #method, #descriptor, &[#(#arguments),*]) }?;));
113        } else {
114            let cache = format_ident!("__PB_RETURN_METHOD_{index}");
115            invoke.extend(quote!(static #cache: ::prebindgen_jni_runtime::ReturnCallback = ::prebindgen_jni_runtime::ReturnCallback::new();
116                unsafe { #cache.invoke(&mut #env, &#sink, #method, #descriptor, &[#(#arguments),*]) }?;));
117        }
118    }
119    Ok(Delivery {
120        statements: quote!(#prepare #invoke),
121        result: quote!(()),
122    })
123}
124pub(crate) fn delivery(
125    plan: &OutputPlan<JniWire>,
126    sink: &syn::Ident,
127    domain: bool,
128) -> Result<Delivery> {
129    if domain {
130        return callback_delivery(plan, sink, true);
131    }
132    let mut delivery = match select(plan)
133        .ok_or("custom output protocol cannot use a return capture")?
134    {
135        NativeReturn::Unit => Delivery::default(),
136        NativeReturn::Direct(wire) => {
137            let fields = crate::group::result_packet(&plan.outcomes.success, &plan.values)?;
138            let argument = wire.output_argument(fields[0].1, &format_ident!("__pb_direct_object"));
139            let prepare = argument.statements;
140            let argument = argument.value;
141            let member = wire.value_member();
142            Delivery {
143                statements: quote!(#prepare let __pb_direct_result = unsafe { (#argument).#member };),
144                result: quote!(__pb_direct_result),
145            }
146        }
147        NativeReturn::Callback => callback_delivery(plan, sink, false)?,
148    };
149    delivery.statements.extend(acknowledge_rows(plan)?);
150    Ok(delivery)
151}

153/// Replace top-level packet reads with typed native locals/capture fields.
154/// Nested collection rows keep their independent row arrays and distinct local
155/// names; only the caller-supplied top-level `result[index]` references are mapped.
156pub(crate) fn remap(
157    output: &mut KotlinOutput,
158    result: &str,
159    fields: &[&Wire<JniWire>],
160    offset: usize,
161    values: &[String],
162) {
163    for text in output.delivered.iter_mut().chain([
164        &mut output.output_locals,
165        &mut output.materialize,
166        &mut output.cleanup,
167    ]) {
168        for (i, (wire, value)) in fields.iter().zip(values).enumerate().rev() {
169            let old = format!("{result}[{}]", offset + i);
170            if !wire.kind.is_reference() {
171                *text = text.replace(&format!("{old} as {}", wire.kind.kotlin_type()), value);
172            }
173            *text = text.replace(&old, value);
174        }
175    }
176}
177pub(crate) fn remap_text(
178    text: &str,
179    result: &str,
180    fields: &[&Wire<JniWire>],
181    offset: usize,
182    values: &[String],
183) -> String {
184    let mut output = KotlinOutput {
185        materialize: text.into(),
186        ..Default::default()
187    };
188    remap(&mut output, result, fields, offset, values);
189    output.materialize
190}

192pub(crate) fn capture(name: &str, success: &[&Wire<JniWire>], domain: &[&Wire<JniWire>]) -> String {
193    let mut fields = String::new();
194    let mut reset = String::new();
195    let mut methods = String::new();
196    for (prefix, method, wires) in [("s", "success", success), ("d", "domainError", domain)] {
197        for (i, wire) in wires.iter().enumerate() {
198            let ty = wire.kind.kotlin_type();
199            let default = wire.kind.kotlin_default();
200            fields.push_str(&format!("    var {prefix}{i}: {ty} = {default}\n"));
201            if wire.kind.is_reference() {
202                reset.push_str(&format!("        {prefix}{i} = {default}\n"));
203            }
204        }
205        let ranges = chunks(wires);
206        for (chunk, range) in ranges.iter().enumerate() {
207            let params = range
208                .clone()
209                .map(|i| format!("v{i}: {}", wires[i].kind.kotlin_type()))
210                .collect::<Vec<_>>();
211            let mut body = String::new();
212            for i in range.clone() {
213                use std::fmt::Write as _;
214                write!(body, "this.{prefix}{i} = v{i}; ").expect("writing to a String");
215            }
216            if chunk + 1 == ranges.len() {
217                body.push_str(if method == "success" {
218                    "status = 1"
219                } else {
220                    "status = 3"
221                });
222            }
223            methods.push_str(&format!(
224                "    fun {method}{chunk}({}) {{ {body} }}\n",
225                params.join(", ")
226            ));
227        }
228    }
229    format!("internal class {name} private constructor(private val pool: Pool) {{\n    var status: Int = 0\n    var message: String? = null\n    var discardSuccess: Boolean = false\n{fields}{methods}    fun bindingError(message: String?, committed: Boolean) {{ this.message = message; discardSuccess = committed; status = 2 }}\n    fun release() {{\n{reset}        status = 0; message = null; discardSuccess = false\n        pool.depth--\n    }}\n    private class Pool {{\n        val frames = java.util.ArrayList<{name}>()\n        var depth = 0\n        fun acquire(): {name} {{\n            val index = depth\n            if (index == frames.size) frames.add({name}(this))\n            depth = index + 1\n            return frames[index]\n        }}\n    }}\n    companion object {{\n        private val local = ThreadLocal.withInitial {{ Pool() }}\n        fun acquire(): {name} = local.get().acquire()\n    }}\n}}\n")
230}

232pub(crate) fn binding_failure(
233    env: &syn::Ident,
234    sink: &syn::Ident,
235    kind: &NativeReturn,
236) -> TokenStream {
237    let result = if matches!(kind, NativeReturn::Direct(_)) {
238        quote!(direct_output)
239    } else {
240        TokenStream::new()
241    };
242    quote! {
243        if domain_output.is_none() && matches!(#env.exception_check(), Ok(false)) {
244            let reported = ::prebindgen_runtime::boundary(|| {
245                let message = ::prebindgen_jni_runtime::text_object(&mut #env, &e)?;
246                unsafe { ::prebindgen_jni_runtime::invoke_error_callback(&mut #env, &#sink, "bindingError", "(Ljava/lang/String;Z)V", &[
247                    ::prebindgen_jni_runtime::jni::sys::jvalue { l: message.as_raw() },
248                    ::prebindgen_jni_runtime::jni::sys::jvalue { z: u8::from(success_transferred) },
249                ]) }
250            });
251            if let Err(message) = reported { ::prebindgen_jni_runtime::fail(&mut #env, &message); }
252        }
253        #result
254    }
255}

263        JniGen::builder()

277                    .fun(fun!(owned))
278                    .fun(fun!(failed)),

281            .unwrap()
282    }
283    fn externs(generation: &JniGen) -> BTreeMap<String, syn::ItemFn> {
284        syn::parse_file(&generation.rust)
285            .unwrap()
286            .items
287            .into_iter()
288            .filter_map(|item| {
289                if let syn::Item::Fn(f) = item {
290                    if f.sig.abi.is_some() {
291                        return Some((f.sig.ident.to_string(), f));
292                    }
293                }
294                None
295            })
296            .collect()
297    }
298    #[test]
299    fn native_results_do_not_box_or_allocate_success_packets() {
300        let generation = generate(&prepared()).unwrap();
301        let functions = externs(&generation);
302        for name in ["value", "unit", "owned", "failed"] {
303            let function = functions
304                .iter()
305                .find(|(n, _)| n.contains(&format!("call_1{name}_1")))
306                .unwrap()
307                .1;
308            let code = quote!(#function).to_string();
309            assert!(
310                !code.contains("prebindgen_jni_runtime :: array"),
311                "{name}: {code}"
312            );
313            assert!(!code.contains("valueOf"), "{name}: {code}");
314            assert!(code.contains("bindingError"));
315            assert!(!quote!(#function).to_string().contains("jobjectArray {"));
316        }
317        let kotlin = &generation.kotlin["test/Functions.kt"];
318        assert!(!kotlin.contains("when (result[0])"));
319        assert!(kotlin.contains("returnCapture.release()"));
