6/// The native Rust value and target storage of one inline C owner.
7#[derive(Clone, Debug)]
8pub struct COwned {
9    /// Unprefixed C type and accessor stem.
10    pub name: String,
11    /// Original, qualified Rust type.
12    pub ty: syn::Type,
13    /// Storage layout of that Rust type on the binding target.
14    pub layout: OpaqueLayout,
15    pub(crate) empty: TokenStream,
16    pub(crate) explicit: bool,
17}
18impl COwned {
19    pub(crate) fn ident(&self) -> syn::Ident {
20        format_ident!("__PbOwned_{}", self.name)
21    }
22    pub(crate) fn guard_ident(&self) -> syn::Ident {
23        format_ident!("__PbOwnedGuard_{}", self.name)
24    }
25    pub(crate) fn helper(&self, operation: &str) -> syn::Ident {
26        helper(&self.name, operation)
27    }
28}
29pub(crate) fn helper(name: &str, operation: &str) -> syn::Ident {
30    format_ident!("__pb_owned_{name}_{operation}")
31}
32pub(crate) fn referent(ty: &syn::Type) -> &syn::Type {
33    if let syn::Type::Reference(r) = ty {
34        &r.elem
35    } else {
36        ty
37    }
38}
39pub(crate) fn declaration(flat: &Bindings, ty: &syn::Type) -> Option<Rc<COwned>> {
40    flat.type_info(referent(ty)).ok()?.metadata.owned.clone()
41}
42pub(crate) fn pointer_declared(flat: &Bindings, ty: &syn::Type) -> bool {
43    flat.type_info(referent(ty))
44        .ok()
45        .is_some_and(|t| t.metadata.pointer.is_some())
46}
47pub(crate) fn native_input(flat: &Bindings, ty: &syn::Type) -> bool {
48    pointer_declared(flat, ty) || declaration(flat, ty).is_some_and(|o| o.explicit)
49}
50pub(crate) fn native_output(flat: &Bindings, ty: &syn::Type) -> bool {
51    pointer_declared(flat, ty) || declaration(flat, ty).is_some()
52}
53fn has_borrow(ty: &syn::Type) -> bool {
54    match ty {
55        syn::Type::Reference(_) => true,
56        syn::Type::Path(p) => p.path.segments.iter().any(|s| match &s.arguments {
57            syn::PathArguments::AngleBracketed(a) => a.args.iter().any(|a| match a {
58                syn::GenericArgument::Lifetime(_) => true,
59                syn::GenericArgument::Type(t) => has_borrow(t),
60                _ => false,
61            }),
62            _ => false,
63        }),
64        syn::Type::Tuple(t) => t.elems.iter().any(has_borrow),
65        syn::Type::Slice(s) => has_borrow(&s.elem),
66        syn::Type::Array(a) => has_borrow(&a.elem),
67        _ => false,
68    }
69}
70pub(crate) fn std_owner(ty: &syn::Type) -> bool {
71    !has_borrow(ty)
72        && (matches!(
73            prebindgen_flat::type_key(ty).as_str(),
74            "String" | "Vec < u8 >"
75        ) || matches!(
76            prebindgen_flat::sequence_parts(ty),
77            Some((_, prebindgen_flat::SequenceView::Owned))
78        ))
79}
80pub(crate) fn output(owner: Rc<COwned>, wire: syn::Ident, value: TokenStream) -> OutputPlan<CWire> {
81    let ident = owner.ident();
82    OutputPlan {
83        fallible: false,
84        context: Default::default(),
85        delivery: Default::default(),
86        outcomes: prebindgen_flat::Outcomes {
87            success: prebindgen_flat::WireNode::Leaf(vec![Wire {
88                name: wire.clone(),
89                kind: CWire::Owned {
90                    owner,
91                    access: Access::Move,
92                    by_value: true,
93                },
94            }]),
95            source_error: None,
96            binding_failure: None,
97        },
98        statements: quote!(let #wire = #ident::guard(#value);),
99        values: vec![(wire.clone(), quote!(#wire.0))],
100        commit: quote!(::core::mem::forget(#wire);),
101    }
102}
103pub(crate) fn borrowed(
104    owner: Rc<COwned>,
105    wire: syn::Ident,
106    value: TokenStream,
107    access: Access,
108) -> OutputPlan<CWire> {
109    let ident = owner.ident();
110    let native = &owner.ty;
111    let pointer = if access == Access::Shared {
112        quote!(*const #native as *const #ident)
113    } else {
114        quote!(*mut #native as *mut #ident)
115    };
116    OutputPlan {
117        fallible: false,
118        context: Default::default(),
119        delivery: Default::default(),
120        outcomes: prebindgen_flat::Outcomes {
121            success: prebindgen_flat::WireNode::Leaf(vec![Wire {
122                name: wire.clone(),
123                kind: CWire::Owned {
124                    owner,
125                    access,
126                    by_value: false,
127                },
128            }]),
129            source_error: None,
130            binding_failure: None,
131        },
132        statements: quote!(let #wire = #value as #pointer;),
133        values: vec![(wire.clone(), quote!(#wire))],
134        commit: quote!(),
135    }
136}
137pub(crate) fn input(
138    owner: Rc<COwned>,
139    ty: &syn::Type,
140    wire: syn::Ident,
141    var: syn::Ident,
142    active: &Condition,
143    by_value: bool,
144) -> InputPlan<CWire> {
145    let access = match ty {
146        syn::Type::Reference(r) if r.mutability.is_none() => Access::Shared,
147        syn::Type::Reference(_) => Access::Exclusive,
148        _ => Access::Move,
149    };
150    let condition = active.rust();
151    let native = &owner.ty;
152    let pointer = format_ident!("{wire}_pointer");
153    let local = format_ident!("{wire}_storage");
154    let ptr = if by_value {
155        quote!(&mut #local as *mut _ as *mut #native)
156    } else {
157        quote!(#wire as *mut #native)
158    };
159    let storage = by_value.then(|| {
160        if access == Access::Move {
161            quote!(#[allow(unused_parens)] let #local=#wire;)
162        } else {
163            quote!(#[allow(unused_parens)] let mut #local=#wire;)
164        }
165    });
166    let empty_var = format_ident!("{wire}_empty");
167    let empty = &owner.empty;
168    let prepare_empty = (access == Access::Move && !by_value)
169        .then(|| quote!(let #empty_var = if #condition {Some(#empty)} else {None};));
170    let acquire = if access == Access::Move && by_value {
171        quote!(let #var=if #condition {let #pointer=::core::ptr::addr_of!(#local).cast::<#native>(); let value=unsafe {::core::ptr::read(#pointer)}; __prebindgen_input_consumed.set(__prebindgen_input_consumed.get()+1); Some(value)} else {None};)
172    } else if access == Access::Move {
173        quote!(let #var = if #condition {Some(unsafe {::core::ptr::replace(#pointer, #empty_var.expect("prepared empty owner"))})} else {None};)
174    } else {
175        quote!()
176    };
177    // A by-value carrier may move into a prepared row. Derive every pointer
178    // where it is used, from that row's captured storage.
179    let borrowed_pointer = if by_value {
180        ptr.clone()
181    } else {
182        quote!(#pointer)
183    };
184    let (statements, argument) = match access {
185        Access::Move => (
186            quote!(let #var: #ty = #var.expect("active inline owner");),
187            quote!(#var),
188        ),
189        Access::Shared => (
190            quote!(let #var: #ty = unsafe {&*(#borrowed_pointer)};),
191            quote!(#var),
192        ),
193        Access::Exclusive => (
194            quote!(let #var: #ty = unsafe {&mut *(#borrowed_pointer)};),
195            quote!(#var),
196        ),
197    };
198    InputPlan {
199        fallible: !by_value,
200        context: Default::default(),
201        acknowledgement: None,
202        node: prebindgen_flat::WireNode::Leaf(vec![Wire {
203            name: wire.clone(),
204            kind: CWire::Owned {
205                owner,
206                access,
207                by_value,
208            },
209        }]),
210        validation: if by_value {
211            quote!(#storage)
212        } else {
213            quote!(#storage let #pointer = #ptr; if #condition {::prebindgen_runtime::validate_pointer(#pointer)?;} #prepare_empty)
214        },
215        acquire,
216        statements,
217        argument,
218        resources: vec![prebindgen_conversions::Resource {
219            registered: by_value,
220            identity: prebindgen_conversions::ResourceIdentity::Single(quote!(#pointer.addr())),
221            active: condition,
222            access,
223        }],
224    }
225}

227pub(crate) fn auto_name(ty: &syn::Type) -> String {
228    if prebindgen_flat::type_key(ty) == "String" {
229        return "owned_string".into();
230    }
231    let key = prebindgen_flat::type_key(ty);
232    let hash = key.bytes().fold(0xcbf29ce484222325u64, |h, b| {
233        (h ^ u64::from(b)).wrapping_mul(0x100000001b3)
234    });
235    let mut readable = String::new();
236    for c in key.chars() {
237        if c.is_ascii_alphanumeric() {
238            readable.push(c.to_ascii_lowercase());
239        } else if !readable.ends_with('_') {
240            readable.push('_');
241        }
242    }
243    format!("owned_{}_{hash:016x}", readable.trim_matches('_'))
244}

246pub(crate) fn collect(ty: &syn::Type, types: &mut std::collections::BTreeMap<String, syn::Type>) {
247    if std_owner(ty) {
248        types.insert(prebindgen_flat::type_key(ty), ty.clone());
249    }
250    if let syn::Type::Reference(r) = ty {
251        collect(&r.elem, types);
252    }
253    if let Some(t) = prebindgen_flat::option_inner(ty) {
254        collect(t, types);
255    }
256    if let Some((t, e)) = prebindgen_flat::result_parts(ty) {
257        collect(t, types);
258        collect(e, types);
259    }
260    if let Some((t, _)) = prebindgen_flat::sequence_parts(ty) {
261        collect(t, types);
262    }
263    if let Ok(Some(sig)) = prebindgen_flat::callback_signature(ty) {
264        for t in sig.arguments {
265            collect(&t, types);
266        }
267        collect(&sig.return_type, types);
268    }
269}

271pub(crate) fn emit(
272    owner: &COwned,
273    layout: crate::layouts::StorageLayout,
274    prefix: &str,
275    tokens: &mut TokenStream,
276    header: &mut String,
277) {
278    let ident = owner.ident();
279    let native = &owner.ty;
280    let (definition, c) = {
281        let size = layout.size;
282        let align = layout.alignment;
283        let a = syn::LitInt::new(&align.to_string(), proc_macro2::Span::call_site());
284        (quote!(#[allow(non_camel_case_types)] #[repr(C, align(#a))] #[derive(Clone,Copy)] pub struct #ident { storage: ::core::mem::MaybeUninit<[u8;#size]> }),format!("typedef struct {{\n#ifdef __cplusplus\n    alignas({align})\n#else\n    _Alignas({align})\n#endif\n    unsigned char _storage[{size}];\n}} {prefix}_{};\n",owner.name))
285    };
286    let guard = format_ident!("__PbOwnedGuard_{}", owner.name);
287    let drop = format_ident!("{prefix}_drop_{}", owner.name);
288    let empty = &owner.empty;
289    tokens.extend(quote! {
290        #definition
291        const _: () = { assert!(::core::mem::size_of::<#ident>() == ::core::mem::size_of::<#native>(), "opaque owner size differs from its Rust type"); assert!(::core::mem::align_of::<#ident>() == ::core::mem::align_of::<#native>(), "opaque owner alignment differs from its Rust type"); };
292        #[allow(non_camel_case_types)] struct #guard(#ident);
293        impl Drop for #guard {fn drop(&mut self){unsafe {::core::ptr::drop_in_place(&mut self.0 as *mut #ident as *mut #native);}}}
294        #[allow(dead_code)] impl #ident {
295            fn from_rust(value: #native) -> Self {let mut storage=::core::mem::MaybeUninit::<Self>::uninit(); unsafe {::core::ptr::write(storage.as_mut_ptr().cast::<#native>(),value);storage.assume_init()}}
296            fn guard(value: #native) -> #guard {#guard(Self::from_rust(value))}
297            unsafe fn as_rust(&self) -> &#native { &*(self as *const Self as *const #native) }
298            unsafe fn as_rust_mut(&mut self) -> &mut #native { &mut *(self as *mut Self as *mut #native) }
299            unsafe fn into_rust(self) -> #native { ::core::ptr::read(&self as *const Self as *const #native) }
300            fn empty() -> Self { Self::from_rust(#empty) }
301        }
302        #[doc="# Safety\nThe pointer must be null or point to one live owner of this exact type. The owner is replaced by its empty value; previous views become invalid."]
303        #[no_mangle] pub unsafe extern "C" fn #drop(value:*mut #ident) {
304            if let Err(error) = ::prebindgen_runtime::boundary(|| { if !value.is_null() {drop(::core::ptr::replace(value as *mut #native, #empty));} Ok(()) }) { panic!("C destructor failed: {}",error); }
305        }
306    });
307    header.push_str(&c);
308    header.push_str(&format!("/* Owning value: move, never duplicate ownership. Drop after the last use of its views; consumption replaces it with an empty value. */\nvoid {drop}({prefix}_{} *value);\n",owner.name));
309}

311/// Register ordinary Rust helper functions so expansions can reference them.
312pub(crate) fn accessors(flat: &mut Bindings) -> Result<()> {
313    let owners: Vec<_> = flat
314        .types
315        .values()
316        .filter_map(|t| t.metadata.owned.clone())
317        .collect();
318    for owner in owners {
319        if !std_owner(&owner.ty) {
320            continue;
321        }
322        let ty = &owner.ty;
323        let len = owner.helper("len");
324        let mut functions: Vec<syn::ItemFn> = vec![
325            syn::parse_quote!(#[allow(dead_code,clippy::ptr_arg)] fn #len(value: &#ty) -> usize {value.len()}),
326        ];
327        if matches!(
328            prebindgen_flat::type_key(ty).as_str(),
329            "String" | "Vec < u8 >"
330        ) {
331            let view = owner.helper("view");
332            let expression = if prebindgen_flat::type_key(ty) == "String" {
333                quote!(value.as_bytes())
334            } else {
335                quote!(value.as_slice())
336            };
337            functions.push(syn::parse_quote!(#[allow(dead_code,clippy::ptr_arg)] fn #view(value: &#ty) -> &[u8] {#expression}));
338        }
339        for function in functions {
340            let name = function.sig.ident.clone();
341            if flat.functions.contains_key(&name.to_string()) {
342                return Err(format!(
343                    "generated owner accessor `{name}` collides with a source function"
344                ));
345            }
346            let ret = match &function.sig.output {
347                syn::ReturnType::Type(_, ty) => (**ty).clone(),
348                _ => unreachable!(),
349            };
350            flat.functions.insert(
351                name.to_string(),
352                prebindgen_flat::Function {
353                    metadata: None,
354                    name: name.clone(),
355                    path: syn::parse_quote!(#name),
356                    params: vec![prebindgen_flat::Param {
357                        name: syn::parse_quote!(value),
358                        ty: prebindgen_flat::TypeUse {
359                            ty: syn::parse_quote!(&#ty),
360                            slot: None,
361                        },
362                    }],
363                    ret: prebindgen_flat::TypeUse {
364                        ty: ret,
365                        slot: None,
366                    },
367                    location: Default::default(),
368                },
369            );
370            flat.metadata.helpers.push(function);
371        }
372    }
373    Ok(())
374}
375pub(crate) fn emit_accessors(
376    owner: &COwned,
377    prefix: &str,
378    types: &std::collections::BTreeMap<String, prebindgen_flat::TypeInfo<CType>>,
379    symbols: &mut std::collections::BTreeSet<String>,
380    tokens: &mut TokenStream,
381    header: &mut String,
382) -> Result<()> {
383    if !std_owner(&owner.ty) {
384        return Ok(());
385    }
386    let carrier = owner.ident();
387    let len = format_ident!("{prefix}_{}_len", owner.name);
388    if !symbols.insert(len.to_string()) {
389        return Err("generated owner length accessor collides with a C symbol".into());
390    }
391    tokens.extend(quote!(#[doc="# Safety\nThe owner must be live, aligned, and of this exact type."] #[no_mangle] pub unsafe extern "C" fn #len(value:*const #carrier)->usize {(*value).as_rust().len()}));
392    header.push_str(&format!(
393        "size_t {len}(const {prefix}_{} *value);\n",
394        owner.name
395    ));
396    let key = prebindgen_flat::type_key(&owner.ty);
397    if matches!(key.as_str(), "String" | "Vec < u8 >") {
398        let view = format_ident!("{prefix}_{}_view", owner.name);
399        let helper = owner.helper("view");
400        if !symbols.insert(view.to_string()) {
401            return Err("generated owner view accessor collides with a C symbol".into());
402        }
403        tokens.extend(quote!(#[doc="# Safety\nThe owner must remain live and unmodified while its returned view is used."] #[no_mangle] pub unsafe extern "C" fn #view(value:*const #carrier)->__PbBuffer {let bytes=#helper((*value).as_rust());__PbBuffer{data:bytes.as_ptr(),len:bytes.len()}}));
404        header.push_str(&format!("/* Borrowed view; never free it. The owner retains the allocation. */\n{prefix}_buffer_view {view}(const {prefix}_{} *value);\n",owner.name));
405    } else if let Some((element, _)) = prebindgen_flat::sequence_parts(&owner.ty) {
406        let get = format_ident!("{prefix}_{}_get", owner.name);
407        if !symbols.insert(get.to_string()) {
408            return Err("generated owner element accessor collides with a C symbol".into());
409        }
410        let declared = types
411            .values()
412            .find(|t| prebindgen_flat::same_type(&t.ty, element))
413            .and_then(|t| {
414                t.metadata
415                    .pointer
416                    .as_ref()
417                    .or_else(|| t.metadata.owned.as_ref().map(|o| &o.name))
418            });
419        let c_element = if let Some(name) = declared {
420            format!("{prefix}_{name}")
421        } else if let Some(Value::Scalar(primitive)) = Value::of(element)
422            .filter(|v| !matches!(v, Value::Scalar(Primitive::I128 | Primitive::U128)))
423        {
424            c_scalar(primitive, prefix)
425        } else {
426            "void".into()
427        };
428        tokens.extend(quote!(#[doc="# Safety\nThe owner must remain live and unmodified while an element pointer is used."] #[no_mangle] pub unsafe extern "C" fn #get(value:*const #carrier,index:usize)->*const #element {(*value).as_rust().get(index).map_or(::core::ptr::null(), |element|element as *const #element)}));
429        header.push_str(&format!("/* Borrowed element; null for an out-of-range index. */\nconst {c_element} *{get}(const {prefix}_{} *value,size_t index);\n",owner.name));
430    }
431    Ok(())
432}

434pub(crate) fn validate_type(ty: &syn::Type) -> Result<()> {
435    if has_borrow(ty)
436        || matches!(ty, syn::Type::Reference(_))
437        || prebindgen_flat::option_inner(ty).is_some()
438        || prebindgen_flat::result_parts(ty).is_some()
439        || prebindgen_flat::callback_signature(ty)?.is_some()
440    {
441        return Err("opaque storage requires an owned non-optional value type".into());
442    }
443    Ok(())
444}
