1//! Explicit bindings between prepared conversion code and its wrapper.
2use prebindgen_flat::Result;
3use proc_macro2::{Group, TokenStream, TokenTree};
4use quote::{quote, ToTokens};
5
6/// Optional rendering support for stored [`crate::InputPlan`] and
7/// [`crate::OutputPlan`] templates. Custom conversions need not implement this.
8pub trait TemplateContext {
9    /// Resolve local binding and transfer markers before inserting the rendered
10    /// continuation. Already-rendered Rust remains unchanged.
11    fn render(&self, code: &TokenStream) -> Result<TokenStream>;
12}
13
14/// Ownership reporting required by the generated boundary.
15#[derive(Clone, Debug, Default)]
16pub enum InputTransfers {
17    /// No ownership transfer is allowed at this boundary.
18    #[default]
19    Forbidden,
20    /// Record completed takes in this boundary's local `Cell<usize>`.
21    Report(syn::Ident),
22    /// Errors abort the call; no incoming-owner receipt is observable.
23    Abort,
24}
25impl InputTransfers {
26    /// Emit code for completed ownership takes according to the boundary contract.
27    pub fn render(&self, count: &TokenStream) -> Result<TokenStream> {
28        match self {
29            Self::Report(counter) => Ok(quote!({#counter.set(#counter.get()+(#count));})),
30            Self::Abort => Ok(quote!({})),
31            Self::Forbidden => Err(
32                "input transfer requires an ownership receipt at this recoverable boundary".into(),
33            ),
34        }
35    }
36}
37impl TemplateContext for () {
38    fn render(&self, code: &TokenStream) -> Result<TokenStream> {
39        render_template(
40            code,
41            &|name| Err(format!("unknown emission parameter `{name}`")),
42            &|count| InputTransfers::Forbidden.render(count),
43        )
44    }
45}
46/// Record successful incoming ownership takes, before later fallible/unwinding work.
47/// `count` must be a side-effect-free `usize` expression. An aborting boundary may
48/// omit its evaluation. Never report a take which has not happened.
49pub fn record_input_transfer(count: TokenStream) -> TokenStream {
50    quote!(__prebindgen_input_transfer! {#count})
51}
52/// Bind a prepared conversion's target while preserving adapter markers.
53pub(crate) fn bind_target(code: &TokenStream, target: &syn::Ident) -> Result<TokenStream> {
54    render_template(
55        code,
56        &|name| {
57            Ok(if name == "input_target" {
58                target.to_token_stream()
59            } else {
60                template_parameter(name)
61            })
62        },
63        &|count| Ok(record_input_transfer(count.clone())),
64    )
65}
66
67/// Resolve the call's source binding before generator-specific template parameters.
68pub fn render_output(
69    context: &impl TemplateContext,
70    code: &TokenStream,
71    source: &syn::Ident,
72) -> Result<TokenStream> {
73    let code = render_template(
74        code,
75        &|name| {
76            Ok(if name == "source_result" {
77                source.to_token_stream()
78            } else {
79                template_parameter(name)
80            })
81        },
82        &|count| Ok(record_input_transfer(count.clone())),
83    )?;
84    context.render(&code)
85}
86
87/// Insert an explicit template marker. The context renderer defines its meaning.
88/// Ordinary identifiers and string literals are never substituted.
89pub fn template_parameter(name: &str) -> TokenStream {
90    let name = syn::Ident::new(name, proc_macro2::Span::call_site());
91    quote!(__prebindgen_emit!(#name))
92}
93
94/// Resolve explicit markers recursively using a language generator's bindings.
95/// `__prebindgen_emit!(...)` is reserved for [`template_parameter`].
96pub fn render_template(
97    code: &TokenStream,
98    resolve: &impl Fn(&str) -> Result<TokenStream>,
99    transfer: &impl Fn(&TokenStream) -> Result<TokenStream>,
100) -> Result<TokenStream> {
101    let tokens: Vec<_> = code.clone().into_iter().collect();
102    let mut output = TokenStream::new();
103    let mut index = 0;
104    while index < tokens.len() {
105        if matches!(&tokens[index], TokenTree::Ident(id) if id == "__prebindgen_emit" || id == "__prebindgen_input_transfer")
106            && matches!(tokens.get(index + 1), Some(TokenTree::Punct(p)) if p.as_char() == '!')
107        {
108            let Some(TokenTree::Group(group)) = tokens.get(index + 2) else {
109                return Err("invalid emission parameter marker".into());
110            };
111            let binding = if matches!(&tokens[index],TokenTree::Ident(id) if id == "__prebindgen_input_transfer")
112            {
113                let count = render_template(&group.stream(), resolve, transfer)?;
114                transfer(&count)?
115            } else {
116                let name = group.stream().to_string();
117                if name == "continuation" {
118                    template_parameter(&name)
119                } else {
120                    resolve(&name)?
121                }
122            };
123            output.extend(binding);
124            index += 3;
125            continue;
126        }
127        match &tokens[index] {
128            TokenTree::Group(group) => {
129                let mut rendered = Group::new(
130                    group.delimiter(),
131                    render_template(&group.stream(), resolve, transfer)?,
132                );
133                rendered.set_span(group.span());
134                output.extend([TokenTree::Group(rendered)]);
135            }
136            other => output.extend([other.clone()]),
137        }
138        index += 1;
139    }
140    Ok(output)
141}
