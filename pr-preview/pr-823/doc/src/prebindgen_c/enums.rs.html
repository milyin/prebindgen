5/// C conversions read representation and error declarations from Flat.
6#[derive(Default)]
7pub(crate) struct EnumCodec<'a> {
8    pub(crate) return_owners:
9        Option<&'a std::cell::RefCell<std::collections::BTreeMap<String, syn::Type>>>,
10}
11impl EnumCodec<'_> {
12    fn mapping<'a>(
13        &self,
14        flat: &'a Bindings,
15        ty: &syn::Type,
16    ) -> Option<(&'a str, &'a std::rc::Rc<EnumMapping>)> {
17        let ty = if let syn::Type::Reference(r) = ty {
18            &r.elem
19        } else {
20            ty
21        };
22        flat.type_info(ty)
23            .ok()?
24            .metadata
25            .enumeration
26            .as_ref()
27            .map(|e| (e.name.as_str(), &e.mapping))
28    }
29    fn wire(name: &str, mapping: &std::rc::Rc<EnumMapping>) -> CWire {
30        CWire::enumeration(name, mapping.clone())
31    }
32}

118    fn result_error_value(
119        &self,
120        flat: &prebindgen_flat::Flat<Self::Schema>,
121        ty: &syn::Type,
122        value: TokenStream,
123        message: Option<TokenStream>,
124        extra: &(),
125    ) -> Result<TokenStream> {
126        if let Some(name) = flat
127            .type_info(ty)
128            .ok()
129            .and_then(|t| t.metadata.error.as_ref())
130            .map(|e| e.name.as_str())
131        {
132            if message.is_some() {
133                return Err("structured C errors cannot also select a message mapper".into());
134            }
135            let helper = errors::helper(name);
136            let pointer = prebindgen_conversions::template_parameter("error_pointer");
137            let reported = prebindgen_conversions::template_parameter("error_reported");
138            Ok(
139                quote!({unsafe {#helper(#value,#pointer,&mut #reported)}?;#reported=true;String::new()}),
140            )
141        } else {
142            CCodec.result_error_value(flat, ty, value, message, extra)
143        }
144    }

170        CCodec.selector(e)
171    }
172    fn consumes(&self, flat: &prebindgen_flat::Flat<Self::Schema>, ty: &syn::Type, e: &()) -> bool {
173        if let Some(inner) = prebindgen_declarations::option_inner(ty) {
174            return self.consumes(flat, inner, e);
175        }
176        self.mapping(flat, ty).is_none() && CCodec.consumes(flat, ty, e)
177    }
178    fn callback_input(
179        &self,
180        flat: &Bindings,
181        callback: &prebindgen_declarations::CallbackInput,
182        _: &(),
183        wire: syn::Ident,
184        var: syn::Ident,
185        active: &Condition,
186    ) -> Result<InputPlan<CWire>> {
187        if callback.foreign.is_some()
188            || callback.namespace.is_some()
189            || !callback.explicit_arguments.is_empty()
190            || callback.explicit_return
191        {
192            return Err("configured callbacks are unsupported by the C adapter".into());
193        }
194        callback::input(flat, callback, wire, var, active)
195    }
196    fn input_leaf(
197        &self,
198        flat: &prebindgen_flat::Flat<Self::Schema>,
199        ty: &syn::Type,
200        e: &(),
201        w: syn::Ident,
202        v: syn::Ident,
203        active: &Condition,
204    ) -> Result<InputPlan<CWire>> {
205        if prebindgen_flat::callback_signature(ty)?.is_some() {
206            return Err("resolve Callback operations before leaf compilation".into());
207        }
208        if prebindgen_declarations::option_inner(ty).is_some() {
209            return Err("resolve Optional operations before leaf compilation".into());
210        }
211        if let Some((name, m)) = self.mapping(flat, ty) {
212            return prebindgen_declarations::enum_input(m, ty, w, v, active, Self::wire(name, m));
213        }
214        let plan = CCodec.input_leaf(flat, ty, e, w.clone(), v, active)?;
215        if plan
216            .resources
217            .iter()
218            .any(|resource| resource.access == prebindgen_conversions::Access::Move)
219        {
220            if let Some(owners) = self.return_owners {
221                owners.borrow_mut().insert(w.to_string(), ty.clone());
222            }
223        }
224        Ok(plan)
225    }
226    fn output_leaf(
227        &self,
228        flat: &prebindgen_flat::Flat<Self::Schema>,
229        ty: &syn::Type,
230        e: &(),
231        w: syn::Ident,
232        v: TokenStream,
233        borrowed: bool,
234    ) -> Result<OutputPlan<CWire>> {
235        if prebindgen_declarations::option_inner(ty).is_some() {
236            return Err("resolve Optional operations before leaf compilation".into());
237        }
238        if let Some((name, m)) = self.mapping(flat, ty) {
239            if matches!(ty,syn::Type::Reference(r) if r.mutability.is_some()) {
240                return Err("mutable-reference outputs are unsupported".into());
241            }
242            return Ok(prebindgen_declarations::enum_output(
243                m,
244                w,
245                v,
246                Self::wire(name, m),
247            ));
248        }
249        CCodec.output_leaf(flat, ty, e, w, v, borrowed)
250    }
251
252    fn optional_niche(
253        &self,
254        payload: &prebindgen_flat::WireNode<Self::Wire, Self::Chain, Self::Branch>,
255    ) -> Option<prebindgen_declarations::NicheReservation<Self::Niche>> {
256        CCodec.optional_niche(payload)
257    }
258    fn optional_presence(
259        &self,
260        payload: &prebindgen_flat::WireNode<Self::Wire, Self::Chain, Self::Branch>,
261        reservation: &prebindgen_declarations::NicheReservation<Self::Niche>,
262    ) -> Result<TokenStream> {
263        CCodec.optional_presence(payload, reservation)
264    }
265    fn absent_output_values(
266        &self,
267        payload: &prebindgen_flat::WireNode<Self::Wire, Self::Chain, Self::Branch>,
268        reservation: Option<&prebindgen_declarations::NicheReservation<Self::Niche>>,
269    ) -> Result<Vec<(syn::Ident, TokenStream)>> {
270        CCodec.absent_output_values(payload, reservation)
271    }
272    fn retire_niches(
273        &self,
274        payload: &mut prebindgen_flat::WireNode<Self::Wire, Self::Chain, Self::Branch>,
275        reservation: Option<&prebindgen_declarations::NicheReservation<Self::Niche>>,
276    ) -> Result<()> {
277        CCodec.retire_niches(payload, reservation)
278    }
279    fn validate_input(&self, plan: &prebindgen_declarations::CodecInput<Self>) -> Result<()> {
280        CCodec.validate_input(plan)
281    }
282    fn validate_output(&self, plan: &prebindgen_declarations::CodecOutput<Self>) -> Result<()> {
283        CCodec.validate_output(plan)
284    }

302        let codec = EnumCodec::default();
303        let mut model = flat("pub enum E { A, B }pub fn echo(x:E)->E{}");
304        let ty = model.resolve_type(&syn::parse_quote!(E)).unwrap();
305        let mapping =
306            std::rc::Rc::new(EnumMapping::new(model.enums.values().next().unwrap()).unwrap());
307        for name in ["first", "updated"] {
308            model.type_metadata_mut(&ty).unwrap().enumeration = Some(std::rc::Rc::new(CEnum {
309                name: name.into(),
310                mapping: mapping.clone(),
311            }));
312            let output = codec
313                .output_leaf(
314                    &model,
315                    &ty,
316                    &(),
