1//! Questions about the model, answered through the registry that projects it.
2//!
3//! # The planning boundary
4//!
5//! Planning must not reach the Rust type captured behind a [`TypeRef`].
6//! Planning is everything up to final Rust emission: resolving declarations,
7//! compiling fragments, planning sites, and validating the result. It may carry
8//! a `TypeRef` opaquely and use the semantic facts the model exposes, but it
9//! may not obtain a `syn::Type` or tokens for one, and may not branch on the
10//! captured spelling to reach a decision. Formatting a `TypeRef` for a
11//! diagnostic is not covered — deciding from that text is. If a planner needs a
12//! fact available only from the syntax, Flat is incomplete: add the fact to
13//! Flat rather than opening an emission escape.
14//!
15//! Wire types an adapter authors are outside this rule — `*mut c_void` and
16//! `jlong` are the adapter's output vocabulary, not captured source syntax.
17//!
18//! In a normal build of an adapter built on this crate the phase is structural:
19//! [`RustWriter`]'s constructor is private to the registry, callbacks receive
20//! one only during final file assembly, and this crate's model re-export omits
21//! `prebindgen_flat::RustEmitter` so no receiver of the adapter's own can be
22//! supplied. Two escapes are deliberate rather than closed: the non-default
23//! `testing` feature exposes `RustWriter::for_test` and
24//! `RustWriter::for_registry_test` for out-of-crate adapter test suites, and a
25//! crate depending on `prebindgen-flat` directly can implement the unsealed
26//! `RustEmitter` itself. For those the rule is policy. `docs/model.md` states
27//! it in full.
28//!
29//! [`TypeRef`]: prebindgen_flat::flat::TypeRef
30//! [`RustWriter`]: crate::RustWriter
31
32use std::collections::HashMap;
33
34use super::{
35    view::{default_module_of, origin_module_of},
36    *,
37};
38
39impl Registry {
40    /// The parameter-side fold for each `(function, parameter)` position.
41    ///
42    /// Inherent rather than on [`Conversions`]: a fold is read when a wrapper's
43    /// parameters are emitted, never while a conversion is being built, so no
44    /// generic caller needs it.
45    pub fn expansion_plans(&self) -> &HashMap<(syn::Ident, syn::Ident), crate::expand::FoldPlan> {
46        &self.expansion_plans
47    }
48
49    /// The parsed model this registry projects.
50    pub fn flat(&self) -> &prebindgen_flat::flat::Flat {
51        &self.flat
52    }
53
54    /// Every **named** item the model holds — functions, structs, either enum
55    /// shape, consts — regardless of whether the stream carried an origin stamp.
56    ///
57    /// Lives here so an adapter that needs "anything the source crate defines"
58    /// does not enumerate element kinds itself: a new kind is taught here once
59    /// instead of drifting in each adapter. An **alias is deliberately absent**
60    /// — see the arm below — and callers are expected to pair this with
61    /// `origin_module(..).unwrap_or_else(default_module)`.
62    pub fn named_item_idents(&self) -> impl Iterator<Item = &syn::Ident> {
63        use prebindgen_flat::flat::{Element, Type};
64        self.flat.elements().filter_map(|e| match e {
65            // An `Extern` names a type without declaring a body, and is
66            // deliberately absent: its caller decides which names to qualify in
67            // generated Rust, and qualifying an alias would move that output.
68            Element::Type(Type::Extern(_)) => None,
69            Element::Function(_) | Element::Type(_) | Element::Constant(_) => e.name(),
70            Element::Guard(_) | Element::Unsupported(_) => None,
71        })
72    }
73
74    /// Whether the source declares a type under this name — **including an
75    /// alias**.
76    ///
77    /// An alias counts because `#[prebindgen] pub type Handle = ..` *is* a
78    /// declaration of that name: it can be declared bare by an adapter (landing
79    /// in the no-indexed-body branch below, which is what
80    /// `ptr_class(ZKeyExpr<'static>)` relies on), so a diagnostic that says
81    /// "no such captured item" would be false.
82    pub(super) fn declares_type(&self, ident: &syn::Ident) -> bool {
83        self.flat.declared_type(ident).is_some()
84    }
85
86    /// The origin crate's **module path** for an item, read off the element's
87    /// own [`SourceLocation`] stamp, or `None` when unknown — callers then fall
88    /// back to [`Self::default_module`].
89    pub fn origin_module(&self, ident: &syn::Ident) -> Option<syn::Path> {
90        // Off the element's own location, which covers both populations: a
91        // captured item stamped at capture time, and a binding-local fn stamped
92        // by `add_local_function`.
93        origin_module_of(&self.flat, ident)
94    }
95
96    /// The default module for references with no recorded origin: the
97    /// first-seen item origin. `None` for an origin-less item-level
98    /// registry (adapters then fall back to `crate`). To change a module
99    /// name, override it at the source — a stream's origin stamps
100    /// (`Source::builder(dir).crate_name("myflat")`) — never here: a
101    /// registry-level override could only fix ONE module, which is
102    /// incomplete with chained multi-source streams.
103    pub fn default_module(&self) -> Option<syn::Path> {
104        default_module_of(&self.flat)
105    }
106
107    /// Module paths of every ingested source, ingestion order — e.g. for a
108    /// glob import that must see all sources' items.
109    pub fn all_source_modules(&self) -> Vec<syn::Path> {
110        self.flat
111            .source_modules()
112            .iter()
113            .filter_map(|m| syn::parse_str(m).ok())
114            .collect()
115    }
116}
