20const ALLOW: &str = "#[allow(non_snake_case, non_camel_case_types, unused_variables, unused_mut, unused_unsafe, unused_parens, unused_braces, dead_code, clippy::all)]";

22fn allow() -> TokenStream {
23    ALLOW.parse().unwrap()
24}

26/// Write every element of the plan.
27pub(crate) fn write(plan: &Plan) -> Res<RustFile> {
28    let mut file = RustFile::new();
29    file.push(prelude(plan));
30    for item in &plan.items {
31        file.push(match item {
32            Item::Type(name) => type_item(plan, name)?,
33            Item::Closure { ty, name } => closure_struct(plan, ty, name)?,
34            Item::Function {
35                func,
36                exported,
37                panic,
38            } => function(plan, func, exported, *panic)?,
39        });
40    }
41    file.guards(plan.flat);
42    Ok(file)
43}

45/// The allocation helpers every binding uses, and the free function.
46fn prelude(plan: &Plan) -> TokenStream {
47    let allow = allow();
48    let free = plan.free_fn.as_ref().map(|f| {
49        quote! {
50            #[no_mangle]
51            #allow
52            pub unsafe extern "C" fn #f(p: *mut ::core::ffi::c_void) {
53                free(p);
54            }
55        }
56    });
57    quote! {
58        extern "C" {
59            fn malloc(size: usize) -> *mut ::core::ffi::c_void;
60            fn free(ptr: *mut ::core::ffi::c_void);
61        }
62        #allow
63        pub(crate) fn __cbg_alloc_cstr(s: ::std::string::String) -> *mut ::core::ffi::c_char {
64            let c = ::std::ffi::CString::new(s).unwrap_or_default();
65            let bytes = c.as_bytes_with_nul();
66            unsafe {
67                let p = malloc(bytes.len()) as *mut u8;
68                if p.is_null() {
69                    return ::core::ptr::null_mut();
70                }
71                ::core::ptr::copy_nonoverlapping(bytes.as_ptr(), p, bytes.len());
72                p as *mut ::core::ffi::c_char
73            }
74        }
75        #allow
76        pub(crate) unsafe fn __cbg_alloc_array<W>(v: ::std::vec::Vec<W>) -> (*mut W, usize) {
77            let n = v.len();
78            if n == 0 {
79                return (::core::ptr::null_mut(), 0);
80            }
81            let p = malloc(n.wrapping_mul(::core::mem::size_of::<W>())) as *mut W;
82            if p.is_null() {
83                return (::core::ptr::null_mut(), 0);
84            }
85            for (i, e) in v.into_iter().enumerate() {
86                ::core::ptr::write(p.add(i), e);
87            }
88            (p, n)
89        }
90        #free
91    }
92}

96fn type_item(plan: &Plan, name: &str) -> Res<TokenStream> {
97    let t = plan.ctype(name).expect("a planned type");
98    match &t.kind {
99        Kind::Opaque => Ok(opaque(plan, t)),
100        Kind::Error { message } => Ok(opaque_error(plan, t, message)),
101        Kind::Enum => enum_mirror(plan, t),
102        Kind::Union => union_mirror(plan, t),
103        Kind::Data => data_mirror(plan, t),
104        Kind::ReprC { assume_valid } => repr_c(plan, t, *assume_valid),
105    }
106}

108/// A handle: an opaque struct and its destructor.
109fn opaque(plan: &Plan, t: &CType) -> TokenStream {
110    let (c, drop) = (&t.c, &t.drop);
111    let src = plan.source(&t.rust.to_string());
112    let allow = allow();
113    quote! {
114        #[repr(C)]
115        #allow
116        pub struct #c { _private: [u8; 0] }
117        #[no_mangle]
118        #allow
119        pub unsafe extern "C" fn #drop(this_: *mut #c) {
120            if !this_.is_null() {
121                drop(::std::boxed::Box::from_raw(this_ as *mut #src));
122            }
123        }
124    }
125}

127/// An error type: handed to C as its message.
128fn opaque_error(plan: &Plan, t: &CType, message: &syn::Ident) -> TokenStream {
129    let src = plan.source(&t.rust.to_string());
130    let msg = plan.source(&message.to_string());
131    let f = format_ident!("__cbg_out_{}", t.rust);
132    let allow = allow();
133    quote! {
134        #allow
135        pub(crate) fn #f(v: #src) -> *mut ::core::ffi::c_char {
136            __cbg_alloc_cstr(#msg(&v))
137        }
138    }
139}

141/// A fieldless enum: a `#[repr(C)]` mirror, validated on the way in.
142fn enum_mirror(plan: &Plan, t: &CType) -> Res<TokenStream> {
143    let FlatType::Enum(e) = plan.element(&t.rust.to_string())? else {
144        return err(format!("`{}` is not a fieldless enum", t.rust));
145    };
146    let emit = Emit::new();
147    let c = &t.c;
148    let src = plan.source(&t.rust.to_string());
149    let values = e.values.iter().map(|v| {
150        let n = &v.name;
151        match emit.discriminant(v) {
152            Some(d) => quote!(#n = #d),
153            None => quote!(#n),
154        }
155    });
156    let names: Vec<&syn::Ident> = e.values.iter().map(|v| &v.name).collect();
157    let (fin, fout) = (
158        format_ident!("__cbg_in_{}", t.rust),
159        format_ident!("__cbg_out_{}", t.rust),
160    );
161    let size_msg = format!("`{c}`: a #[repr(C)] enum must have the size of a C `int`");
162    let align_msg = format!("`{c}`: a #[repr(C)] enum must have the alignment of a C `int`");
163    let bad = format!("invalid discriminant {{}} for `{c}`");
164    let allow = allow();
165    Ok(quote! {
166        #[repr(C)]
167        #[derive(Copy, Clone, Debug, Eq, PartialEq)]
168        #allow
169        pub enum #c { #(#values),* }
170        #allow
171        pub(crate) unsafe fn #fin(v: ::core::mem::MaybeUninit<#c>) -> ::core::result::Result<#src, ::std::string::String> {
172            const _: () = {
173                assert!(::core::mem::size_of::<#c>() == ::core::mem::size_of::<::core::ffi::c_int>(), #size_msg);
174                assert!(::core::mem::align_of::<#c>() == ::core::mem::align_of::<::core::ffi::c_int>(), #align_msg);
175            };
176            let __raw: ::core::ffi::c_int = ::core::ptr::read(v.as_ptr() as *const ::core::ffi::c_int);
177            #(
178                if __raw == #c::#names as ::core::ffi::c_int {
179                    return ::core::result::Result::Ok(#src::#names);
180                }
181            )*
182            ::core::result::Result::Err(::std::format!(#bad, __raw))
183        }
184        #allow
185        pub(crate) fn #fout(v: #src) -> #c {
186            match v { #(#src::#names => #c::#names),* }
187        }
188    })
189}

191/// A struct or union field: one value slot.
192struct Fields<'p, 'f> {
193    plan: &'p Plan<'f>,
194    owner: &'p syn::Ident,
195    /// The slot's name for a field.
196    wire: fn(&Field) -> syn::Ident,
197}
198
199impl Fields<'_, '_> {
200    fn field_in(&mut self, field: &Field) -> Res<Input<CWire>> {
201        let w = (self.wire)(field);
202        self.plan
203            .value_in(&field.ty, &w)
204            .map_err(|e| Error(format!("`{}`: field `{w}`: {}", self.owner, e.0)))
205    }
206
207    fn field_out(&mut self, field: &Field) -> Res<Output<CWire>> {
208        let w = (self.wire)(field);
209        self.plan
210            .value_out(&field.ty, &w)
211            .map_err(|e| Error(format!("`{}`: field `{w}`: {}", self.owner, e.0)))
212    }
213}

215fn named_wire(f: &Field) -> syn::Ident {
216    match &f.name {
217        Some(n) => n.clone(),
218        None => format_ident!("_{}", f.index),
219    }
220}

222fn positional_wire(f: &Field) -> syn::Ident {
223    format_ident!("__f{}", f.index)

226/// A data-carrying enum: a `#[repr(C)]` enum mirror, its tag validated on
227/// the way in, and a typed drop when an arm owns memory.
228fn union_mirror(plan: &Plan, t: &CType) -> Res<TokenStream> {
229    let FlatType::Variant(v) = plan.element(&t.rust.to_string())? else {
230        return err(format!("`{}` is not a data-carrying enum", t.rust));
231    };
232    let c = &t.c;
233    let src = plan.source(&t.rust.to_string());
234    let allow = allow();
235    let mirror = sum_mirror(
236        v,
237        &src,
238        c,
239        vec![quote!(#[repr(C)]), allow.clone()],
240        &mut Fields {
241            plan,
242            owner: &t.rust,
243            wire: positional_wire,
244        },
245    )?;
246    let n = v.alternatives.len() as i64;
247    let size_msg = format!(
248        "`{c}`: a #[repr(C)] enum with payload variants must be at least as large as its C `int` discriminant"
249    );
250    let bad = format!("invalid tag {{}} for `{c}` (expected 0..{n})");
251    let (fin, fout) = (
252        format_ident!("__cbg_in_{}", t.rust),
253        format_ident!("__cbg_out_{}", t.rust),
254    );
255    let def = &mirror.def;
256    let in_body = mirror.input.result();
257    let out_expr = &mirror.output.expr;
258    let check = quote! {
259        const _: () = {
260            assert!(::core::mem::size_of::<#c>() >= ::core::mem::size_of::<::core::ffi::c_int>(), #size_msg);
261        };
262    };
263    // The tag C wrote, read before the enum is: C can write any `int`.
264    let tag = |p: TokenStream| {
265        quote! {
266            let __tag: ::core::ffi::c_int = ::core::ptr::read(#p as *const ::core::ffi::c_int);
267            let __valid = (__tag as i64) >= 0 && (__tag as i64) < #n;
268        }
269    };
270    let tag_in = tag(quote!(v.as_ptr()));
271    let tag_drop = tag(quote!((*this_).as_ptr()));
272    let mut arms = Vec::new();
273    for (alt, wires) in v.alternatives.iter().zip(&mirror.alternatives) {
274        let mut releases = Vec::new();
275        for (f, w) in alt.fields.iter().zip(wires) {
276            let n = &w.name;
277            if let Some(r) = plan.release(&f.ty, &quote!((*#n)))? {
278                releases.push(r);
279            }
280        }
281        if !releases.is_empty() {
282            let aname = &alt.name;
283            let binds: Vec<syn::Ident> = wires.iter().map(|w| w.name.clone()).collect();
284            let pat = Record::Alt(alt).pattern(&quote!(#c::#aname), &binds);
285            arms.push(quote!(#pat => { #(#releases)* }));
286        }
287    }
288    let drop = (!arms.is_empty()).then(|| {
289        let d = &t.drop;
290        quote! {
291            #[no_mangle]
292            #allow
293            pub unsafe extern "C" fn #d(this_: *mut ::core::mem::MaybeUninit<#c>) {
294                if this_.is_null() {
295                    return;
296                }
297                #check
298                #tag_drop
299                if !__valid {
300                    return;
301                }
302                match (*this_).assume_init_mut() {
303                    #(#arms)*
304                    _ => {}
305                }
306            }
307        }
308    });
309    Ok(quote! {
310        #def
311        #allow
312        pub(crate) unsafe fn #fin(v: ::core::mem::MaybeUninit<#c>) -> ::core::result::Result<#src, ::std::string::String> {
313            #check
314            #tag_in
315            if !__valid {
316                return ::core::result::Result::Err(::std::format!(#bad, __tag));
317            }
318            let v = v.assume_init();
319            #in_body
320        }
321        #allow
322        pub(crate) fn #fout(v: #src) -> ::core::mem::MaybeUninit<#c> {
323            ::core::mem::MaybeUninit::new(#out_expr)
324        }
325        #drop
326    })
327}

329/// A struct crossing field by field: a mirror, both conversions, and a
330/// release of the memory its fields own.
331fn data_mirror(plan: &Plan, t: &CType) -> Res<TokenStream> {
332    let FlatType::Struct(s) = plan.element(&t.rust.to_string())? else {
333        return err(format!("`{}` is not a struct", t.rust));
334    };
335    let c = &t.c;
336    let src = plan.source(&t.rust.to_string());
337    let allow = allow();
338    let mirror = struct_mirror(
339        s,
340        &src,
341        c,
342        vec![quote!(#[repr(C)]), allow.clone()],
343        &mut Fields {
344            plan,
345            owner: &t.rust,
346            wire: named_wire,
347        },
348    )?;
349    let (fin, fout, frel) = (
350        format_ident!("__cbg_in_{}", t.rust),
351        format_ident!("__cbg_out_{}", t.rust),
352        format_ident!("__cbg_release_{}", t.rust),
353    );
354    let def = &mirror.def;
355    let (in_ret, in_body) = if mirror.input.fallible {
356        (
357            quote!(::core::result::Result<#src, ::std::string::String>),
358            mirror.input.result(),
359        )
360    } else {
361        (src.clone(), mirror.input.expr.clone())
362    };
363    let out_expr = &mirror.output.expr;
364    let mut releases = Vec::new();
365    for (f, w) in s.fields.iter().zip(&mirror.wires) {
366        let n = &w.name;
367        if let Some(r) = plan.release(&f.ty, &quote!(v.#n))? {
368            releases.push(r);
369        }
370    }
371    let release = (!releases.is_empty()).then(|| {
372        quote! {
373            #allow
374            pub(crate) unsafe fn #frel(v: &mut #c) { #(#releases)* }
375        }
376    });
377    Ok(quote! {
378        #def
379        #allow
380        pub(crate) unsafe fn #fin(v: #c) -> #in_ret { #in_body }
381        #allow
382        pub(crate) fn #fout(v: #src) -> #c { #out_expr }
383        #release
384    })
385}

387/// A `#[repr(C)]` struct: a mirror with the same layout, reinterpreted.
388fn repr_c(plan: &Plan, t: &CType, assume_valid: bool) -> Res<TokenStream> {
389    let FlatType::Struct(s) = plan.element(&t.rust.to_string())? else {
390        return err(format!("`{}` is not a struct", t.rust));
391    };
392    let c = &t.c;
393    let src = plan.source(&t.rust.to_string());
394    let mut fields = Vec::new();
395    for f in &s.fields {
396        let Some(fname) = &f.name else {
397            return err(format!("repr_c_struct `{}` must have named fields", t.rust));
398        };
399        let wire = repr_c_field(plan, &t.rust, fname, &f.ty, assume_valid)?;
400        fields.push(quote!(pub #fname: #wire));
401    }
402    let drop = &t.drop;
403    let allow = allow();
404    Ok(quote! {
405        #[repr(C)]
406        #allow
407        pub struct #c { #(#fields),* }
408        const _: () = {
409            assert!(::core::mem::size_of::<#src>() == ::core::mem::size_of::<#c>(), "repr_c_struct: Rust type and C mirror differ in size");
410            assert!(::core::mem::align_of::<#src>() == ::core::mem::align_of::<#c>(), "repr_c_struct: Rust type and C mirror differ in alignment");
411        };
412        impl ::prebindgen_c_runtime::Transmute for #c {
413            type Rust = #src;
414            #[inline]
415            fn from_rust(value: Self::Rust) -> Self {
416                let __v = ::core::mem::ManuallyDrop::new(value);
417                unsafe { ::core::ptr::read(&*__v as *const Self::Rust as *const Self) }
418            }
419            #[inline]
420            fn into_rust(self) -> Self::Rust {
421                let __v = ::core::mem::ManuallyDrop::new(self);
422                unsafe { ::core::ptr::read(&*__v as *const Self as *const Self::Rust) }
423            }
424            #[inline]
425            fn as_rust(&self) -> &Self::Rust {
426                unsafe { &*(self as *const Self as *const Self::Rust) }
427            }
428            #[inline]
429            fn as_rust_mut(&mut self) -> &mut Self::Rust {
430                unsafe { &mut *(self as *mut Self as *mut Self::Rust) }
431            }
432        }
433        #[no_mangle]
434        #allow
435        pub unsafe extern "C" fn #drop(this_: *mut #c) {
436            if !this_.is_null() {
437                ::core::ptr::drop_in_place(<#c as ::prebindgen_c_runtime::Transmute>::as_rust_mut(&mut *this_));
438            }
439        }
440    })
441}

443/// The C spelling of a `repr_c_struct` field — the same layout as the Rust
444/// field, so the struct can be reinterpreted.
445fn repr_c_field(
446    plan: &Plan,
447    owner: &syn::Ident,
448    field: &syn::Ident,
449    ty: &TypeRef,
450    assume_valid: bool,
451) -> Res<TokenStream> {
452    let refuse = |why: &str| {
453        err(format!(
454            "repr_c_struct `{owner}`: field `{field}` ({ty}) {why}"
455        ))
456    };
457    let opaque = |t: &TypeRef| match plan.shape(t) {
458        Shape::Declared {
459            ty: declared_ty,
460            declaration:
461                Setting::Type(CType {
462                    kind: Kind::Opaque,
463                    c,
464                    ..
465                }),
466            ..
467        } if Access::of(declared_ty) == Access::Owned => Some(c.clone()),
468        _ => None,
469    };
470    Ok(match ty.kind() {
471        TypeKind::Scalar(ScalarKind::Bool) if !assume_valid => {
472            return refuse(
473                "is a `bool`, whose invalid bytes a reinterpretation cannot reject; \
474                 declare `.assume_c_field_validity()` or use a data_struct",
475            )
476        }
477        TypeKind::Scalar(k) => {
478            let id = names::ident(k.as_str());
479            quote!(#id)
480        }
481        TypeKind::Boxed(inner) => match opaque(inner) {
482            Some(c) => {
483                return refuse(&format!(
484                    "is a non-null owned pointer (`*mut {c}`), so a consumed struct cannot be \
485                     left droppable; use `Option<Box<_>>`"
486                ))
487            }
488            None => return refuse("is not a representable field"),
489        },
490        TypeKind::Optional(inner) => match inner.kind() {
491            TypeKind::Boxed(b) => match opaque(b) {
492                Some(c) => quote!(*mut #c),
493                None => return refuse("is not a representable field"),
494            },
495            _ => return refuse("is not a representable field"),
496        },
497        TypeKind::Named { .. } => match plan.shape(ty) {
498            Shape::Declared {
499                ty: declared_ty,
500                declaration: Setting::Type(t),
501                ..
502            } if Access::of(declared_ty) == Access::Owned => match t.kind {
503                Kind::ReprC { .. } => t.c.to_token_stream(),
504                Kind::Enum if assume_valid => t.c.to_token_stream(),
505                _ => return refuse("is not a representable field"),
506            },
507            Shape::Undeclared(_) => return refuse("names an undeclared type"),
508            _ => return refuse("is not a representable field"),
509        },
510        _ => return refuse("is not a representable field"),
511    })
512}

514/// A callback's closure struct: `{ context, call, drop }`.
515fn closure_struct(plan: &Plan, ty: &TypeRef, name: &syn::Ident) -> Res<TokenStream> {
516    let TypeKind::Callback { args } = ty.kind() else {
517        return err(format!("`{ty}` is not an `impl Fn(..)` callback"));
518    };
519    let mut wires = Vec::new();
520    for (i, a) in args.iter().enumerate() {
521        let out = plan.callback_arg(i, a)?;
522        wires.extend(out.wires().into_iter().map(|w| w.ty.rust()));
523    }
524    let allow = allow();
525    Ok(quote! {
526        #[repr(C)]
527        #allow
528        pub struct #name {
529            pub context: *mut ::core::ffi::c_void,
530            pub call: ::core::option::Option<unsafe extern "C" fn(#(#wires,)* *mut ::core::ffi::c_void)>,
531            pub drop: ::core::option::Option<unsafe extern "C" fn(*mut ::core::ffi::c_void)>,
532        }
533    })
534}

538/// An exported function.
539fn function(plan: &Plan, func: &Function, exported: &syn::Ident, panic: bool) -> Res<TokenStream> {
540    let ret = plan.ret(&func.ret)?;
541    let fail = fail(&ret);
542    let prologue = alias_preflight(plan, func, &fail)?;
543    if !prologue.is_empty() && !panic && ret.on_error.is_none() {
544        return err(format!(
545            "function `{}` can be handed aliased resources but has no error channel: declare \
546             it `.panic()` or return `Result`",
547            func.name
548        ));
549    }
550    let callee = &func.name;
551    let mut params = Vec::new();
552    let mut stmts = Vec::new();
553    let mut args = Vec::new();
554    for p in &func.params {
555        let name = &p.name;
556        let Param { input, pass } = plan.param(name, &p.ty)?;
557        if input.fallible && ret.on_error.is_none() && !panic {
558            return err(format!(
559                "function `{}`: parameter `{name}` ({}) can fail to convert (a null pointer, an invalid value) and the function has no error channel: declare it `.panic()` or return `Result`",
560                func.name, p.ty
561            ));
562        }
563        params.extend(input.wires().into_iter().map(Wire::decl));
564        if input.fallible {
565            let r = input.result();
566            stmts.push(quote!(let #name = match #r {
567                ::core::result::Result::Ok(__v) => __v,
568                ::core::result::Result::Err(__err) => { #fail }
569            };));
570        } else {
571            let e = &input.expr;
572            stmts.push(quote!(let #name = #e;));
573        }
574        args.push(pass.unwrap_or_else(|| quote!(#name)));
575    }
576    params.extend(ret.ret.wires.iter().map(Wire::decl));
577    let body = &ret.ret.body;
578    let ret_ty = ret.ret.ty.as_ref().map(|t| {
579        let t = t.rust();
580        quote!(-> #t)
581    });
582    let allow = allow();
583    Ok(quote! {
584        #[no_mangle]
585        #allow
586        pub unsafe extern "C" fn #exported(#(#params),*) #ret_ty {
587            #prologue
588            #(#stmts)*
589            let __result = #callee(#(#args),*);
590            #body
591        }
592    })
593}

595/// What a failed input does: report through the error slot and return the
596/// failure value, or abort.
597fn fail(ret: &CRet) -> TokenStream {
598    let e = format_ident!("__err");
599    match &ret.on_error {
600        Some(default) => quote! {{
601            if !e.is_null() {
602                *e = __cbg_alloc_cstr(#e);
603            }
604            return #default;
605        }},
606        None => quote!({
607            panic!("{}", #e);
608        }),
609    }
610}

612/// Two parameters naming one resource, where at least one consumes it or
613/// borrows it exclusively, would free it twice or alias a `&mut`: compare the
614/// pointers before any conversion runs.
615fn alias_preflight(plan: &Plan, func: &Function, fail: &TokenStream) -> Res<TokenStream> {
616    let label = |a: Access| match a {
617        Access::Owned => "consumed",
618        Access::Exclusive => "exclusively borrowed",
619        Access::Shared => "borrowed",
620    };
621    let mut handles = Vec::new();
622    for p in &func.params {
623        if let Shape::Declared {
624            ty,
625            declaration:
626                Setting::Type(CType {
627                    kind: Kind::Opaque | Kind::ReprC { .. },
628                    ..
629                }),
630        } = plan.shape(&p.ty)
631        {
632            handles.push((&p.name, declared_name(ty), Access::of(ty)));
633        }
634    }
635    let e = format_ident!("__err");
636    let mut checks = Vec::new();
637    for (i, (a, ta, ua)) in handles.iter().enumerate() {
638        for (b, tb, ub) in &handles[i + 1..] {
639            if ta != tb || (*ua == Access::Shared && *ub == Access::Shared) {
640                continue;
641            }
642            let msg = format!(
643                "aliasing arguments: `{a}` ({}) and `{b}` ({}) are the same `{ta}` — a consumed \
644                 or exclusively-borrowed resource may not be named twice in one call",
645                label(*ua),
646                label(*ub)
647            );
648            checks.push(quote! {
649                if !(#a as *const ()).is_null() && (#a as *const ()) == (#b as *const ()) {
650                    let #e = ::std::string::String::from(#msg);
651                    #fail
652                }
653            });
654        }
655    }
656    Ok(checks.into_iter().collect())
657}

659fn struct_mirror(
660    source: &prebindgen_flat::flat::Struct,
661    source_path: &TokenStream,
662    name: &syn::Ident,
663    attrs: Vec<TokenStream>,
664    cb: &mut Fields<'_, '_>,
665) -> Res<StructMirror> {
666    let record = Record::Struct(source);
667    let mut ins = Vec::new();
668    let mut outs = Vec::new();
669    let binds = record.binds();
670    for f in &source.fields {
671        ins.push(cb.field_in(f)?);
672        outs.push(cb.field_out(f)?);
673    }
674
675    let wires: Vec<Wire<CWire>> = ins
676        .iter()
677        .flat_map(|i| i.wires().into_iter().cloned())
678        .collect();
679    let decls = wires.iter().map(|w| {
680        let d = w.decl();
681        quote!(pub #d)
682    });
683
684    let def = quote! {
685        #(#attrs)*
686        pub struct #name { #(#decls),* }
687    };
688    let wire_names: Vec<&syn::Ident> = wires.iter().map(|w| &w.name).collect();
689    let head = source_path;
690    let rebuilt = Input::record(source, ins);
691    let input = Code {
692        expr: {
693            let e = &rebuilt.expr;
694            quote!({ let #name { #(#wire_names),* } = v; #e })
695        },
696        fallible: rebuilt.fallible,
697    };
698    let pat = record.pattern(head, &binds);
699    let fallible = outs.iter().any(|o| o.fallible);
700    let out_binds: Vec<TokenStream> = outs.iter().zip(&binds).map(|(o, b)| o.bind(b)).collect();
701    let output = Code {
702        expr: quote!({ let #pat = v; #(#out_binds)* #name { #(#wire_names),* } }),
703        fallible,
704    };
705    Ok(StructMirror {
706        def,
707        wires,
708        input,
709        output,
710    })
711}

713fn sum_mirror(
714    source: &prebindgen_flat::flat::Variant,
715    source_path: &TokenStream,
716    name: &syn::Ident,
717    attrs: Vec<TokenStream>,
718    cb: &mut Fields<'_, '_>,
719) -> Res<SumMirror> {
720    let src = source_path;
721    let mut variants = Vec::new();
722    let mut alternatives = Vec::new();
723    let mut in_arms = Vec::new();
724    let mut out_arms = Vec::new();
725    let mut in_fallible = false;
726    let mut out_fallible = false;
727    for alt in &source.alternatives {
728        let record = Record::Alt(alt);
729        let aname = &alt.name;
730        let binds = record.binds();
731        let mut ins = Vec::new();
732        let mut outs = Vec::new();
733        for f in &alt.fields {
734            ins.push(cb.field_in(f)?);
735            outs.push(cb.field_out(f)?);
736        }
737        // A mirror alternative has one mirror field per source field; a
738        // field that needs several wires is kept whole as a tuple would
739        // lose the per-field names C sees, so it is refused here by
740        // construction: the adapter hands one wire per field.
741        let wires: Vec<Wire<CWire>> = ins
742            .iter()
743            .flat_map(|i| i.wires().into_iter().cloned())
744            .collect();
745        let mirror_head = quote!(#name::#aname);
746        let source_head = quote!(#src::#aname);
747        // Declaration: same delimiters as the source, wire types in place.
748        let tys: Vec<TokenStream> = wires.iter().map(|w| w.ty.rust()).collect();
749        let decl = record.construct(&quote!(#aname), &tys);
750        variants.push(decl);
751        // In: match the mirror alternative, binding its wires by name.
752        let wire_binds: Vec<syn::Ident> = wires.iter().map(|w| w.name.clone()).collect();
753        let mirror_pat = record.pattern(&mirror_head, &wire_binds);
754        let parts = alt.fields.iter().map(Seg::field).zip(ins).collect();
755        let rebuilt = Input::parts(source.type_ref(), parts, |values| {
756            record.construct(&source_head, &values)
757        });
758        in_fallible |= rebuilt.fallible;
759        let e = &rebuilt.expr;
760        in_arms.push(quote!(#mirror_pat => #e));
761        // Out: match the source alternative, produce the mirror one.
762        let source_pat = record.pattern(&source_head, &binds);
763        out_fallible |= outs.iter().any(|o| o.fallible);
764        let out_binds: Vec<TokenStream> = outs.iter().zip(&binds).map(|(o, b)| o.bind(b)).collect();
765        let values: Vec<TokenStream> = wires.iter().map(|w| w.name.to_token_stream()).collect();
766        let rebuilt_mirror = record.construct(&mirror_head, &values);
767        out_arms.push(quote!(#source_pat => { #(#out_binds)* #rebuilt_mirror }));
768        alternatives.push(wires);
769    }
770
771    let def = quote! {
772        #(#attrs)*
773        pub enum #name { #(#variants),* }
774    };
775    Ok(SumMirror {
776        def,
777        alternatives,
778        input: Code {
779            expr: quote!(match v { #(#in_arms),* }),
780            fallible: in_fallible,
781        },
782        output: Code {
783            expr: quote!(match v { #(#out_arms),* }),
784            fallible: out_fallible,
785        },
786    })
787}

789/// A mirror's conversion to or from its source type: an expression over
790/// the mirror value `v`, which may use `?` on `Result<_, String>`.
791struct Code {
792    expr: TokenStream,
793    fallible: bool,
794}
795
796impl Code {
797    fn result(&self) -> TokenStream {
798        result_expr(&self.expr, self.fallible)
799    }
800}
