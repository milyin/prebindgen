8/// The parameter types of the closure an adapter builds for an `impl Fn(..)`
9/// parameter with arguments `args`, lifetimes elided: `|a0: &Payload| ..`.
10///
11/// Rust needs the annotations to give the closure the borrowed signature the
12/// callee expects. Source items are qualified as their [`ItemName`]s say;
13/// standard types are spelled with absolute paths.
14///
15/// [`ItemName`]: prebindgen_flat::flat::ItemName
16pub fn callback_arg_types(flat: &Flat, args: &[TypeRef]) -> Vec<TokenStream> {
17    args.iter().map(|t| render(flat, t)).collect()
18}
19
20/// A reference to the item named `name`: its qualified path when the model
21/// declares it, else the name as written.
22fn item(flat: &Flat, name: &str) -> TokenStream {
23    match flat.element(name).and_then(|e| e.name()) {
24        Some(n) => n.to_token_stream(),
25        None => name.parse().expect("a type path the model built"),
26    }
27}

29/// A model type spelled for generated Rust, lifetimes elided: generated code
30/// never names the source's lifetime parameters.
31pub(crate) fn render(flat: &Flat, ty: &TypeRef) -> TokenStream {
32    let child = |ty: &TypeRef| render(flat, ty);
33    match ty.kind() {
34        TypeKind::Scalar(k) => {
35            let id = crate::names::ident(k.as_str());
36            quote!(#id)
37        }
38        TypeKind::Str => quote!(str),
39        TypeKind::String => quote!(::std::string::String),
40        TypeKind::Unit => quote!(()),
41        TypeKind::Optional(t) => {
42            let t = child(t);
43            quote!(::core::option::Option<#t>)
44        }
45        TypeKind::Vec(t) => {
46            let t = child(t);
47            quote!(::std::vec::Vec<#t>)
48        }
49        TypeKind::Slice(t) => {
50            let t = child(t);
51            quote!([#t])
52        }
53        TypeKind::Boxed(t) => {
54            let t = child(t);
55            quote!(::std::boxed::Box<#t>)
56        }
57        TypeKind::Cow { inner } => {
58            let t = child(inner);
59            quote!(::std::borrow::Cow<'_, #t>)
60        }
61        TypeKind::Out(t) => {
62            let t = child(t);
63            quote!(&mut ::core::mem::MaybeUninit<#t>)
64        }
65        TypeKind::Fallible { ok, err } => {
66            let (ok, err) = (child(ok), child(err));
67            quote!(::core::result::Result<#ok, #err>)
68        }
69        TypeKind::Ref { mutable, inner } => {
70            let t = child(inner);
71            let m = mutable.then(|| quote!(mut));
72            quote!(& #m #t)
73        }
74        TypeKind::Array { elem, extent } => {
75            let e = child(elem);
76            let len = match &extent.source {
77                ExtentSource::Literal => {
78                    let n = proc_macro2::Literal::usize_unsuffixed(extent.value);
79                    quote!(#n)
80                }
81                ExtentSource::Const(id) => item(flat, &id.name),
82            };
83            quote!([#e; #len])
84        }
85        TypeKind::Named { id, args } => {
86            let head = item(flat, &id.name);
87            if args.is_empty() {
88                head
89            } else {
90                let args = args.iter().map(child);
91                quote!(#head<#(#args),*>)
92            }
93        }
94        TypeKind::Callback { args } => {
95            let args = args.iter().map(child);
96            quote!(impl Fn(#(#args),*) + Send + Sync + 'static)
97        }
98    }
99}
