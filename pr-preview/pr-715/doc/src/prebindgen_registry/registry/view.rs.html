10/// One `(direction, type)` pair that crosses the boundary.
11///
12/// Direction is part of the identity, not a separate axis: `&str` inbound
13/// decodes a `jstring` and outbound allocates one, and one may be convertible
14/// while the other is not.
15pub type Crossing = (Direction, TypeKey);
16
17/// What a conversion is built against: the model, and the conversions already
18/// available.
19///
20/// Two implementors, and the reason there are two is the fill phase.
21/// [`Building`] is the partial view a generator sees while it is still
22/// producing conversions; [`Registry`] is the total one everything else sees.
23/// A helper that serves both — reading a signature off the model, say — takes
24/// `&impl Conversions` and works either side of the boundary.
25pub trait Conversions {
26    /// The model.
27    fn flat(&self) -> &Flat;
28
29    /// The reading for `ty` — what the frontend made of it.
30    ///
31    /// On the trait because it is needed on **both** sides of the fill: a
32    /// converter is chosen for a type while the registry is still being built, and
33    /// an emitter asks about the same type afterwards. Both views answer from the
34    /// cell the scan filled, so the answer does not change across that line.
35    ///
36    /// This is what lets a generator take a crossing and reason about it without
37    /// rebuilding a spelling from the key and classifying that — the round trip
38    /// `api/core` removed from itself in #263, which is the same defect one layer
39    /// out.
40    ///
41    /// **Keyed**, because that is the one thing a caller has before it has a
42    /// reading. This is the door FROM identity TO the model's answer, and the
43    /// only lookup on this trait that does not already take a `TypeRef` — the
44    /// rest take one precisely because this exists to hand them one (#284).
45    fn reading(&self, key: &TypeKey) -> Option<TypeRef>;
46
47    /// The reading for a **spelling** — identify, then look up.
48    ///
49    /// The door for a caller holding tokens it peeled or composed itself, which
50    /// is a real position: an adapter may strip a `&` or name a wire type, and
51    /// #280 sealed minting so it cannot make a reading for the result. It asks
52    /// instead, and `None` means the registry never saw that type.
53    ///
54    /// Kept separate from the entry lookups on purpose. Those take a `TypeRef`,
55    /// so they cannot be called about a type the registry does not know — which
56    /// is the guarantee, and it survives only while getting a reading from
57    /// tokens is a visible step with a `None` to handle.
58    fn reading_of(&self, ty: &syn::Type) -> Option<TypeRef> {
59        self.reading(&TypeKey::from_type(ty))
60    }
61
62    /// Every type key that crosses in `dir`.
63    ///
64    /// The niche allocator needs the whole population, not one lookup: it picks
65    /// sentinel values no sibling conversion can produce.
66    fn crossing_keys(&self, dir: Direction) -> Vec<TypeKey>;
67
68    /// How a parameter of an exported function was expanded, keyed by the two
69    /// idents that name it.
70    ///
71    /// An adapter planning one site asks whether the parameter it belongs to
72    /// expanded, and into which leaves — the same question a
73    /// [`Role::ExpansionLeaf`](crate::recipe::Role::ExpansionLeaf) answers
74    /// positionally. Reading it here is what lets a per-site hook derive its
75    /// context from the `Bound` it is handed rather than being told it.
76    fn expansion_plans(&self) -> &HashMap<(syn::Ident, syn::Ident), crate::expand::FoldPlan>;
77
78    /// The origin crate's module path for an item, or `None` when unknown.
79    fn origin_module(&self, ident: &syn::Ident) -> Option<syn::Path> {
80        origin_module_of(self.flat(), ident)
81    }
82
83    /// The default module for references with no recorded origin.
84    fn default_module(&self) -> Option<syn::Path> {
85        default_module_of(self.flat())
86    }
87}
88
89impl Conversions for Building<'_> {
90    fn flat(&self) -> &Flat {
91        &self.registry.flat
92    }
93    fn expansion_plans(&self) -> &HashMap<(syn::Ident, syn::Ident), crate::expand::FoldPlan> {
94        self.registry.expansion_plans()
95    }
96    fn reading(&self, key: &TypeKey) -> Option<TypeRef> {
97        self.registry.reading(key)
98    }
99    fn crossing_keys(&self, dir: Direction) -> Vec<TypeKey> {
100        self.all_keys
101            .iter()
102            .filter(|(d, _)| *d == dir)
103            .map(|(_, k)| k.clone())
104            .collect()
105    }
106}
107
108impl Conversions for Registry {
109    fn flat(&self) -> &Flat {
110        &self.flat
111    }
112    fn expansion_plans(&self) -> &HashMap<(syn::Ident, syn::Ident), crate::expand::FoldPlan> {
113        Registry::expansion_plans(self)
114    }
115    fn reading(&self, key: &TypeKey) -> Option<TypeRef> {
116        Registry::reading(self, key)
117    }
118    fn crossing_keys(&self, dir: Direction) -> Vec<TypeKey> {
119        self.type_table(dir).keys().cloned().collect()
120    }
121}

123/// The registry mid-fill: the model, plus the conversions supplied so far.
124///
125/// What a generator builds a conversion *against*: the model, the
126/// decompositions, and the full crossing population.
127///
128/// It exposes exactly the reads a conversion needs, which is what keeps the
129/// half-filled state from leaking anywhere else: the resolved [`Registry`] is
130/// what the emitters get, and it is total.
131///
132/// It no longer lends out conversions built so far. An adapter composes from
133/// its own fragments, which it keeps itself, so what the registry could offer
134/// here — one `ConverterImpl` per crossing — was both less than the adapter
135/// already had and less than a multi-wire crossing needs.
136pub struct Building<'a> {
137    /// The prepared registry. Its cells are still empty at this point.
138    registry: &'a Registry,
139    /// Every crossing in the binding, resolved or not — the niche allocator
140    /// reads the population, not just what is built so far.
141    all_keys: &'a [Crossing],
142}
143
144impl<'a> Building<'a> {
145    pub(crate) fn new(registry: &'a Registry, all_keys: &'a [Crossing]) -> Self {
146        Self { registry, all_keys }
147    }
148}

150/// Shared by [`Registry::origin_module`] and [`Building::origin_module`], so the
151/// two cannot answer differently.
152pub(super) fn origin_module_of(flat: &Flat, ident: &syn::Ident) -> Option<syn::Path> {
153    let crate_name = flat.element(ident)?.location().crate_name.as_ref()?;
154    syn::parse_str(&crate_name.replace('-', "_")).ok()
155}
156
157pub(super) fn default_module_of(flat: &Flat) -> Option<syn::Path> {
158    flat.source_modules()
159        .first()
160        .and_then(|m| syn::parse_str(m).ok())
161}
