13/// What the extern calls once its inputs are decoded.
14pub(crate) enum JCallee {
15    /// The `#[prebindgen]` function itself, under the module it comes from.
16    Source(syn::Path),
17    /// A binding-defined expression — a path to a constant, or a constant
18    /// expression declared with `constant_expr!`.
19    Expr(Box<syn::Expr>),
20}
21
22/// One exported JNI extern, planned while the registry is available and
23/// rendered at the final writing boundary.
24///
25/// Everything the extern needs is here: the frozen function plan, the model
26/// element it exports, and what it calls. Nothing is looked up in a live
27/// registry while the file is assembled.
28pub(crate) struct JWrapper {
29    /// This function's frozen JNI plan — symbol, inputs, output delivery,
30    /// error channels.
31    plan: std::rc::Rc<crate::jni::fn_plan::JniFunctionPlan>,
32    /// The `#[prebindgen]` function, or the synthetic nullary getter a
33    /// declared constant is exported through.
34    function: prebindgen_registry::flat::Function,
35    /// What the extern's body calls.
36    callee: JCallee,
37}
38
39impl JWrapper {
40    /// Plan the extern for one function or constant getter.
41    pub(crate) fn new(
42        ext: &Declarations,
43        registry: &Registry,
44        f: &prebindgen_registry::flat::Function,
45        callee: Option<syn::Expr>,
46    ) -> Self {
47        // Backstop only — `validate_resolved` reports every plan failure
48        // before any writer runs, so this panic is unreachable through the
49        // write paths.
50        let plan = ext
51            .fn_plan(registry, f)
52            .unwrap_or_else(|e| panic!("{}", e.message(&f.name)));
53        Self {
54            plan,
55            function: f.clone(),
56            callee: match callee {
57                Some(expr) => JCallee::Expr(Box::new(expr)),
58                None => JCallee::Source(ext.fn_module(registry, &f.name)),
59            },
60        }
61    }
62
63    /// The `JNINative` extern symbol this wrapper exports.
64    pub(crate) fn symbol(&self) -> &str {
65        &self.plan.native_symbol
66    }
67
68    /// What this extern's body calls: the converters its plan names, and the
69    /// prelude, whose error-channel functions every failure path routes
70    /// through.
71    pub(crate) fn calls(&self) -> Vec<prebindgen_registry::write::ArtifactKey> {
72        let mut calls = vec![crate::jni::generation::prelude_key()];
73        self.plan.calls(&mut calls);
74        calls
75    }
76}

78/// The synthetic nullary getter signature a declared const is emitted
79/// through: `pub fn const_get_<ident_lower>() -> <const ty>`. Both sides —
80/// the Rust extern (the constant artifact's getter, a [`JWrapper`] whose
81/// callee is a path to the const) and the Kotlin `val`
82/// initializer (`render_const_val`) — derive the extern symbol from this one
83/// ident, so they stay in sync by construction. The body is never used.
84pub(crate) fn const_getter_fn(
85    c: &prebindgen_registry::flat::Constant,
86) -> prebindgen_registry::flat::Function {
87    let ident = format_ident!("const_get_{}", c.name.to_string().to_lowercase());
88    // No lookup: a constant element carries its own `TypeRef`.
89    prebindgen_registry::flat::Function::synthetic_getter(ident, c.ty.clone())
90}

92/// A const whose (peeled) type is a declared opaque handle is rejected: a
93/// shared closeable `val` is semantically wrong (whose `close()` is it?).
94/// Expose a factory function instead — the established idiom (e.g. zenoh's
95/// `encoding_const_*` companion factories).
96pub(crate) fn reject_handle_const(ext: &Declarations, c: &prebindgen_registry::flat::Constant) {
97    // Off the element: a constant's type is a reading, so the peel is the
98    // model's (`crate::util::head_type`) and no node is fetched to reach it.
99    reject_handle_key(
100        ext,
101        &crate::util::head_type(&c.ty).key(),
102        "const",
103        &c.name.to_string(),
104    );
105}

107/// The constant-value handle check shared by both constant kinds: peel
108/// `&`/`Option`/`Vec` layers off `ty` and reject if what remains is a
109/// declared opaque handle. `what`/`ident` shape the error message
110/// (`const MAX_LEN` / `constant fn encoding_const_x_str`).
111pub(crate) fn reject_handle_constant_type(
112    ext: &Declarations,
113    ty: &prebindgen_registry::flat::TypeRef,
114    what: &str,
115    name: &str,
116) {
117    // The same three layers, off the classification. The node loop peeled a
118    // `Type::Reference` and then re-read the last path segment's ident twice
119    // (`option_inner_type`, `vec_inner_type`) to decide what it had.
120    let mut ty = ty;
121    loop {
122        if let Some(inner) = ty
123            .borrow_target()
124            .or_else(|| ty.optional_inner())
125            .or_else(|| ty.sequence_elem())
126        {
127            ty = inner;
128            continue;
129        }
130        break;
131    }
132    reject_handle_key(ext, &ty.key(), what, name);
133}
134
135/// The refusal itself, once: a declared opaque handle cannot be a shared
136/// closeable Kotlin `val`.
137///
138/// Split from the peel because the two callers peel differently — an element's
139/// type is a reading and peels off the kind, while a **build-script-supplied**
140/// expression type has no reading until something interns it and peels off the
141/// spelling. One assertion, two ways to reach it.
142fn reject_handle_key(ext: &Declarations, key: &TypeKey, what: &str, name: &str) {
143    let is_handle = ext.types.get(key).is_some_and(|cfg| cfg.is_opaque());
144    assert!(
145        !is_handle,
146        "{what} `{name}`: type `{}` is a declared opaque handle — a shared closeable Kotlin `val` is \
147         not supported. Expose a `#[prebindgen]` factory function returning the constant and \
148         declare it as a companion constructor instead.",
149        key.as_str()
150    );
151}

153/// Validates a [`ConstDecl::fun`] declaration against the real
154/// signature: the fn must be **nullary** (a constant has no inputs), must
155/// not return a `Result` (a domain-fallible value is not a constant — and
156/// the `val` initializer's throwing `JniErrorHandler` only fits the
157/// infallible wrapper shape), and its return type must not peel to a
158/// declared opaque handle (same rationale as [`reject_handle_const`]).
159pub(crate) fn validate_constant_fn(ext: &Declarations, f: &prebindgen_registry::flat::Function) {
160    // The ELEMENT: a signature is a parameter list and a return, both already
161    // classified. This walked `sig.inputs` for the arity, matched
162    // `ReturnType::Type` for the return — an elided one the element already
163    // normalizes to `Unit` — and ran `result_ok_type` over a path to re-derive
164    // the fallibility `TypeKind::Fallible` states.
165    assert!(
166        f.params.is_empty(),
167        "constant fn `{}`: takes {} parameter(s) — a function-backed constant must be nullary \
168         (declare it with `.fun(...)` instead if it is a real function)",
169        f.name,
170        f.params.len()
171    );
172    assert!(
173        f.ret.fallible_parts().is_none(),
174        "constant fn `{}`: returns a `Result` — a function-backed constant must be \
175         infallible (declare it with `.fun(...)` instead if it can fail)",
176        f.name
177    );
178    reject_handle_constant_type(ext, &f.ret, "constant fn", &f.name.to_string());
179}

181/// The synthetic nullary getter signature an **expression constant**
182/// ([`ConstExprDecl`](crate::jni::ConstExprDecl)) is emitted through:
183/// `pub fn const_get_<val_name_lower>() -> <ty>` — the same convention as
184/// const-backed getters, so both sides derive the extern symbol from the one
185/// val name. The body is never used.
186pub(crate) fn const_expr_getter_fn(
187    kotlin_name: &str,
188    ty: &syn::Type,
189    registry: &impl Conversions,
190) -> prebindgen_registry::flat::Function {
191    let ident = format_ident!("const_get_{}", kotlin_name.to_lowercase());
192    // The one lookup this path needs: the type is named by a build script, so
193    // no element carries it. A miss means the declared type never entered the
194    // pipeline, which is a binding error worth naming rather than a `None` to
195    // absorb.
196    let ret = registry.reading_of(ty).unwrap_or_else(|| {
197        panic!(
198            "constant_expr `{kotlin_name}`: type `{}` is not a type this binding crosses — \
199             declare it, or name one that is",
200            quote::ToTokens::to_token_stream(ty),
201        )
202    });
203    prebindgen_registry::flat::Function::synthetic_getter(ident, ret)
204}
205/// Validates an expression constant's declared value type (checked on both
206/// write paths): not a `Result` (a domain-fallible value is not a constant),
207/// not (peeled to) a declared opaque handle.
208pub(crate) fn validate_constant_expr(ext: &Declarations, kotlin_name: &str, ty: &syn::Type) {
209    assert!(
210        result_ok_type(ty).is_none(),
211        "constant expr `{kotlin_name}`: type is a `Result` — an expression constant must be \
212         infallible (declare a real function with `.fun(...)` instead if it can fail)"
213    );
214    // The peel, on the SPELLING. Unlike the fn path above there is no reading
215    // to ask: a `const_expr!` type is written by the BUILD SCRIPT and names no
216    // captured item, so the model never classified it (#280) — the ledger's
217    // documented adapter-owned category, and the reason this one node walk
218    // stays where its peer's could go.
219    let mut ty = ty.clone();
220    loop {
221        if let syn::Type::Reference(r) = &ty {
222            ty = (*r.elem).clone();
223            continue;
224        }
225        if let Some(inner) = option_inner_type(&ty).or_else(|| vec_inner_type(&ty)) {
226            ty = inner;
227            continue;
228        }
229        break;
230    }
231    reject_handle_key(ext, &TypeKey::from_type(&ty), "constant expr", kotlin_name);
232}

234/// [`emit_jni_function_wrapper`] with the raw callee expression overridable:
235/// `None` = the ordinary `<origin module>::<fn ident>(args)` call; `Some(e)`
236/// splices `e` verbatim as the value the output phase converts. Used by the
237/// const getter emission (the constant artifact's), whose synthetic nullary `f`
238/// carries the signature while the value comes from
239/// `<origin module>::<CONST_IDENT>` — a path, not a call.
240impl JWrapper {
241    /// Assemble this wrapper's `#[no_mangle] extern "C"` JNI function.
242    pub(crate) fn render_fn(&self, emit: &prebindgen_registry::RustWriter) -> syn::ItemFn {
243        let f = &self.function;
244        let original_ident = &f.name;
245
246        let mut wire_params: Vec<TokenStream> = Vec::new();
247        // Each entry is a per-input decode statement. Fallible decodes are
248        // `match`-arms that, on `Err`, call `signal_error(&mut env,
249        // &__error_sink, &__e)` (invoking the caller's Kotlin sink instead of
250        // throwing a JVM exception) and `return <sentinel>;`.
251        let mut prelude: Vec<TokenStream> = Vec::new();
252        let mut call_args: Vec<TokenStream> = Vec::new();
253
254        // The lowered plan classifies both sides ONCE — the same classification
255        // the Kotlin wrapper and `external fun` renderers consume; this site
256        // renders the Rust decode/encode for each kind. The output is classified
257        // first (inside `build`) so the per-input `match`-arms can splice the
258        // function's sentinel into their early-`return` path.
259        let plan = &self.plan;
260        let wrapper_ident = syn::Ident::new(&plan.native_symbol, Span::call_site());
261        // Output (data) expansion: when output expansion was declared for this
262        // function, the return value is decomposed by the deconstructor. Two
263        // deliveries:
264        //   * `Callback` (`deconstruct_output`, `FnOutputPlan::Unfold`): the
265        //     leaves are delivered to a foreign builder/fold lambda — the
266        //     wrapper's wire return is the lambda's `JObject` result (no
267        //     `output_entry`; see `emit_unfold_delivery`).
268        //   * `Return` (`convert_output`, `is_convert`): the single decomposed
269        //     value is **returned** directly through its ordinary output
270        //     converter — the wrapper behaves exactly like a normal function
271        //     whose return type is `convert_out_ty`.
272        let unfold_plan = plan.unfold.as_ref();
273        // Error-position expansion: when the fn returns `Result<T, E>` and an error
274        // plan is declared, the **`?`** is applied here — the extern peels the
275        // `Result` (Err arm decomposes `E` into the `ze` leaves and invokes the
276        // typed DOMAIN handler), and the success path uses `T`'s converter (not the
277        // `Result<T, E>` rank-2 wrapper).
278        let error_plan = plan.error.as_ref();
279        let is_convert = matches!(&plan.output, FnOutputPlan::Value(v) if v.is_convert);
280        let wire_ty = plan.output.wire_ty();
281        let wire_return = annotate_jobject_with_lifetime(&wire_ty, "a").to_token_stream();
282        let on_err = sentinel_for_wire(&wire_ty);
283
284        for param in &plan.params {
285            let (wp, pre, call_arg) = emit_input_param(param, &on_err, emit);
286            wire_params.extend(wp);
287            prelude.extend(pre);
288            call_args.push(call_arg);
289        }
290
291        let raw_call = match &self.callee {
292            JCallee::Expr(e) => quote!(#e),
293            JCallee::Source(call_module) => quote!(#call_module::#original_ident(#(#call_args),*)),
294        };
295        // For `convert_output` (Return), the value the output converter sees is the
296        // **deconstructed** single value (the converter's accessor applied to the
297        // raw return, lifted through the shape) — not the raw return. Build that
298        // block so the normal output phase converts it. `Decompose` ⇒ `acc(raw)`;
299        // `Optional` ⇒ `raw.map(|inner| acc(inner))`.
300        let call_expr: TokenStream = if is_convert {
301            use prebindgen_registry::unfold::UnfoldShape;
302            let uplan = unfold_plan.expect("is_convert ⇒ plan");
303            let leaf = &uplan.leaves[0];
304            let by_ref = uplan.by_ref;
305            // One derivation, shared with the multi-leaf encoder — the value forms
306            // are bound by the same [`bind_hoists`] and the leaf reached by the
307            // same [`reach_leaf_flat`]. Deriving either a second time here is what
308            // let the two drift apart: this shortcut used to compose its reach
309            // straight off the raw value, which for a value form declared with
310            // `.fields_self_into(..)` emitted `f(&v)` against a by-value receiver.
311            // Origins were frozen with the leaf this reach walks, so the shortcut
312            // qualifies its accessor calls without asking the registry.
313            let delivery = match &plan.output {
314                FnOutputPlan::Value(value) => value
315                    .convert_delivery
316                    .as_ref()
317                    .expect("a convert output carries its frozen origins"),
318                FnOutputPlan::Unfold(_) => unreachable!("a convert has a value output plan"),
319            };
320            let qualify = |id: &syn::Ident| -> syn::Path {
321                prebindgen_registry::unfold::DeliveryBridge::qualify(delivery, id)
322            };
323            // `None` when the reach is the IDENTITY of `base` and no value form had
324            // to be bound — the leaf IS the value, so there is nothing to compose.
325            // Reported separately from the composed form because the two want
326            // different code: wrapping identity in a block emits `{ __cvsrc }`
327            // (`unused_braces`), and mapping it over an `Option` emits
328            // `.map(|__inner| __inner)` (`clippy::map_identity`). Generated code
329            // runs through the consumer's own lints, where both are denials.
330            let compose = |base: TokenStream, base_is_ref: bool| -> Option<TokenStream> {
331                let hoisted = bind_hoists(&qualify, &uplan.hoists, &base, base_is_ref);
332                let stmts = &hoisted.stmts;
333                let reached = match hoisted.rebase(&leaf.path) {
334                    Some((local, rest, consuming)) => reach_leaf_flat(
335                        &qualify,
336                        &crate::jni::compile::OutWire::from_leaf(leaf),
337                        &rest,
338                        quote!(#local),
339                        false,
340                        consuming,
341                    ),
342                    None => reach_leaf_flat(
343                        &qualify,
344                        &crate::jni::compile::OutWire::from_leaf(leaf),
345                        &leaf.path,
346                        base.clone(),
347                        base_is_ref,
348                        false,
349                    ),
350                };
351                if stmts.is_empty() && reached.to_string() == base.to_string() {
352                    return None;
353                }
354                Some(quote!({ #stmts #reached }))
355            };
356            match &uplan.shape {
357                UnfoldShape::Optional((), _) => match compose(quote!(__inner), by_ref) {
358                    Some(inner) => quote!({
359                        let __cvsrc = #raw_call;
360                        __cvsrc.map(|__inner| #inner)
361                    }),
362                    None => raw_call.clone(),
363                },
364                _ => match compose(quote!(__cvsrc), by_ref) {
365                    Some(v) => quote!({
366                        let __cvsrc = #raw_call;
367                        #v
368                    }),
369                    None => raw_call.clone(),
370                },
371            }
372        } else if let Some(ep) = error_plan {
373            // `Result<T, E>` peel (the automatic `?`): success ⇒ `T`; on `Err(e)`,
374            // decompose `e` into the `ze` leaves — through the SAME shared leaf
375            // encoder every output/callback delivery uses (typed jvalues, handle
376            // wraps, Option-nested accessor unwrap) — and invoke the typed DOMAIN
377            // handler (no `je`, no defaults), then return the sentinel. A failure
378            // while ENCODING the error itself degrades to the BINDING channel
379            // (`signal_binding_error`). The success `T` flows into the normal
380            // output phase.
381            let eze_idents: Vec<syn::Ident> = (0..ep.leaves.len())
382                .map(|i| format_ident!("__eze{}", i))
383                .collect();
384            let ze_fail = |msg: TokenStream| -> TokenStream {
385                quote! {
386                    signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &#msg);
387                    return #on_err;
388                }
389            };
390            let (ze_stmts, ze_args, _) = encode_plan_leaves(
391                &ep.delivery,
392                crate::jni::emit::Delivered::planned(
393                    &ep.unfold,
394                    ep.wires.clone(),
395                    ep.chain.clone(),
396                ),
397                &eze_idents,
398                &quote!(__de),
399                &ze_fail,
400                emit,
401            );
402            quote! {
403                match #raw_call {
404                    ::core::result::Result::Ok(__v) => __v,
405                    ::core::result::Result::Err(__de) => {
406                        #ze_stmts
407                        signal_domain_error(
408                            &mut env, &__domain_sink,
409                            &__DSINK_MID, __DSINK_FQN, __DSINK_DESCR,
410                            &[#(#ze_args),*],
411                        );
412                        return #on_err;
413                    }
414                }
415            }
416        } else {
417            raw_call
418        };
419
420        // Output phase. Three shapes:
421        //   * `Callback` output expansion: decompose the return value and deliver the
422        //     leaves to the foreign builder/fold (`__builder` / `__acc`+`__fold`).
423        //   * `Return` output expansion (convert) and normal returns: invoke the
424        //     frozen registry-planned pipeline once and route its `Err` through
425        //     `signal_error`. (For convert, `call_expr` above already deconstructed
426        //     the value.)
427        let mut builder_param: Option<TokenStream> = None;
428        let output_phase: TokenStream = if let FnOutputPlan::Unfold(u) = &plan.output {
429            // Iterable folds: two params (`__acc` accumulator + `__fold` callback).
430            // Decompose/Optional: a single `__builder` callback.
431            let uplan = unfold_plan.expect("Unfold output ⇒ unfold plan present");
432            builder_param = Some(unfold_builder_param(u.iterable_fold));
433            // The delivery **binds** the returned value and matches it against the
434            // canonical shape its `kind` names (`Option`, then a run). Conversion
435            // follows the SYNTAX, and this position takes no converter — nothing
436            // between the source call and the match re-spells anything — so the
437            // wrappers the classification erased have to come off here, at the
438            // emitter's own binding, or the match is an `E0308` on a spelling the
439            // model deliberately reads as optional (#292).
440            //
441            // The value delivered is the `Ok` side when the error plan applied the
442            // `?`, and the return itself otherwise — the wrappers questioned are
443            // those over whatever `call_expr` actually yields.
444            let delivered = match error_plan {
445                Some(_) => f.ret.fallible_parts().map_or(&f.ret, |(ok, _)| ok),
446                None => &f.ret,
447            };
448            let call_expr = read_through_erased_wrappers(delivered, call_expr.clone())
449                .unwrap_or_else(|| {
450                    panic!(
451                    "`{original_ident}` returns `{}`, whose leaves are delivered to a builder: \
452                     the value has to be moved out of `{}` to be decomposed, and that wrapper \
453                     does not permit it (a `Cow` payload cannot be moved through `Deref`). \
454                     Reserved rather than refused for good: rebuilding through every \
455                     transparent wrapper is #292 item 3 — until then, spell the return \
456                     without it.",
457                    delivered,
458                    delivered.erased_wrappers().join("<"),
459                )
460                });
461            emit_unfold_delivery(uplan, u, &call_expr, &on_err, emit)
462        } else {
463            let output = match &plan.output {
464                FnOutputPlan::Value(value) => value,
465                FnOutputPlan::Unfold(_) => unreachable!("normal output has a value plan"),
466            };
467            let convert = output.pipeline.invoke(quote!(__out), emit);
468            let mut phase: TokenStream = quote! { let __out = #call_expr; };
469            phase.extend(quote! {
470            match #convert {
471                ::core::result::Result::Ok(__w) => __w,
472                ::core::result::Result::Err(__e) => {
473                    signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &__e.to_string());
474                    #on_err
475                }
476            }
477        });
478            phase
479        };
480
481        // Error sinks. Both channels are typed `fun interface`s whose `run` method
482        // ID is resolved once per process on the interface class (the sink instance
483        // differs per call). The BINDING channel (`__error_sink` + `__SINK_*`, the
484        // base `JniErrorHandler`) is always present — every wrapper can hit a
485        // binding/marshalling failure. The DOMAIN channel (`__domain_sink` +
486        // `__DSINK_*`, the typed `<Src>Handler`) is present only for a fallible fn
487        // with a declared error plan; its `Err(E)` decomposition delivers the real
488        // leaves (no `je`, no fabricated defaults).
489        let error_ifaces = plan.onerror_iface.as_ref().unwrap_or_else(|| {
490            panic!(
491                "jnigen: cannot derive the onError handler interface for `{}`",
492                original_ident
493            )
494        });
495        let bsink_fqn_lit =
496            syn::LitStr::new(&error_ifaces.binding.raw_slash_fqn(), Span::call_site());
497        let bsink_descr_lit = syn::LitStr::new(&error_ifaces.binding.descr, Span::call_site());
498        let (domain_setup, domain_sink_param) = match &error_ifaces.domain {
499            Some(dsink) => {
500                let dfqn = syn::LitStr::new(&dsink.raw_slash_fqn(), Span::call_site());
501                let ddescr = syn::LitStr::new(&dsink.descr, Span::call_site());
502                (
503                    quote! {
504                        #[allow(non_upper_case_globals)]
505                        static __DSINK_MID: ::prebindgen_jni_runtime::CachedIfaceMethod =
506                            ::prebindgen_jni_runtime::CachedIfaceMethod::new();
507                        const __DSINK_FQN: &str = #dfqn;
508                        const __DSINK_DESCR: &str = #ddescr;
509                    },
510                    quote!(__domain_sink: jni::objects::JObject<'a>,),
511                )
512            }
513            None => (quote!(), quote!()),
514        };
515        let sinks_setup = quote! {
516            #[allow(non_upper_case_globals)]
517            static __SINK_MID: ::prebindgen_jni_runtime::CachedIfaceMethod =
518                ::prebindgen_jni_runtime::CachedIfaceMethod::new();
519            const __SINK_FQN: &str = #bsink_fqn_lit;
520            const __SINK_DESCR: &str = #bsink_descr_lit;
521            #domain_setup
522        };
523
524        // Trailing sink params: `__error_sink` (binding) always, then
525        // `__domain_sink` (typed domain error) for a fallible fn — a capture is
526        // passed for each. Declared after the wire params + builder so the order
527        // matches the Kotlin `external fun`.
528        syn::parse_quote! {
529            #[no_mangle]
530            #[allow(non_snake_case, unused_mut, unused_variables, dead_code)]
531            pub unsafe extern "C" fn #wrapper_ident<'a>(
532                mut env: jni::JNIEnv<'a>,
533                _class: jni::objects::JClass<'a>,
534                #(#wire_params,)*
535                #builder_param
536                __error_sink: jni::objects::JObject<'a>,
537                #domain_sink_param
538            ) -> #wire_return {
539                #sinks_setup
540                #(#prelude)*
541                #output_phase
542            }
543        }
544    }
545}

547fn unfold_builder_param(iterable_fold: bool) -> TokenStream {
548    // An `Iterable` fold (incl. `Option<Vec<T>>`) takes `(acc, fold)`; every
549    // other delivery takes a single `build`.
550    if iterable_fold {
551        quote!(__acc: jni::objects::JObject<'a>, __fold: jni::objects::JObject<'a>,)
552    } else {
553        quote!(__builder: jni::objects::JObject<'a>,)
554    }
555}

557/// Render the Rust-side decode for one source-fn parameter from its lowered
558/// [`PlanParam`]: the wire params, prelude decode statements, and the call
559/// argument. The classification (which crossing form) lives in the plan; this
560/// site renders the frozen Rust decode operation over the leaf's exact native
561/// ABI.
562#[allow(clippy::type_complexity)]
563fn emit_input_param(
564    param: &PlanParam,
565    on_err: &TokenStream,
566    emit: &prebindgen_registry::RustWriter,
567) -> (Vec<TokenStream>, Vec<TokenStream>, TokenStream) {
568    // Constructor-expansion: this parameter's wire form is the fold plan's
569    // flattened leaves. Decode each leaf with its own converter, run the
570    // (pure-Rust) fold to build the value, then pass it to the call.
571    let leaf = match &param.form {
572        ParamForm::Expanded { plan: fold, leaves } => {
573            return emit_expanded_param(fold, leaves, &param.ident, on_err, emit);
574        }
575        ParamForm::Single(leaf) => &**leaf,
576    };
577    let arg_ident = &param.ident;
578    let arg_ty = &param.ty;
579
580    let wire_params: Vec<TokenStream> = leaf
581        .native
582        .iter()
583        .map(|native| {
584            let ident = &native.rust_ident;
585            let wire = &native.rust_wire;
586            quote!(#ident: #wire)
587        })
588        .collect();
589
590    match &leaf.rust {
591        // Flattenable data_class param: cross its fields as separate wire
592        // params and reconstruct the struct inline — no per-call
593        // `env.get_field(...)` reflection. The `JNINative` extern and the
594        // Kotlin call-site destructure read the same plan so the three
595        // sites can't drift.
596        RustParamOp::FlattenStruct(plan) => {
597            let (decode, call_arg) = render_flat_input_decode(plan, arg_ident, on_err, emit);
598            (wire_params, vec![decode], call_arg)
599        }
600
601        // Bare `Option<primitive>` / `Option<enum>` param: cross as a
602        // `(present: jboolean, value: <wire>)` pair instead of a boxed
603        // `java.lang.*` `JObject`. The registry's selected Optional fragment
604        // rebuilds the value from two raw scalars — no
605        // `env.call_method("intValue", …)` unbox.
606        RustParamOp::OptionalPair(sp) => {
607            let pid = &sp.present_ident;
608            let vid = &sp.value_ident;
609            let converter = emit.operation_ident("jni", &sp.chain.operation);
610            let decode = quote! {
611                let #arg_ident = match #converter(&mut env, (#pid, #vid)) {
612                    ::core::result::Result::Ok(__value) => __value,
613                    ::core::result::Result::Err(__error) => {
614                        signal_binding_error(
615                            &mut env,
616                            &__error_sink,
617                            &__SINK_MID,
618                            __SINK_FQN,
619                            __SINK_DESCR,
620                            &__error.to_string(),
621                        );
622                        return #on_err;
623                    }
624                };
625            };
626            (wire_params, vec![decode], quote!(#arg_ident))
627        }
628
629        // Handles, value projections, callbacks, Vec-build inputs, and plain
630        // values all execute the registry-planned pipeline. The plan already
631        // fixed the sole native leaf's name and type.
632        RustParamOp::Pipeline { wire_ident } => {
633            let (prelude, call_arg) =
634                emit_plain_decode(&leaf.pipeline, arg_ident, arg_ty, on_err, emit, wire_ident);
635            (wire_params, prelude, call_arg)
636        }
637    }
638}

640/// The ordinary converter decode shared by every pass-through kind emits a
641/// wire parameter, one frozen-pipeline invocation, and the call argument
642/// (`&decoded` / `.as_deref()` per the source parameter's Rust shape).
643fn emit_plain_decode(
644    pipeline: &crate::jni::chain::JPipeline,
645    arg_ident: &syn::Ident,
646    arg_ty: &prebindgen_registry::flat::TypeRef,
647    on_err: &TokenStream,
648    emit: &prebindgen_registry::RustWriter,
649    wire_ident: &syn::Ident,
650) -> (Vec<TokenStream>, TokenStream) {
651    use prebindgen_registry::flat::TypeKind;
652    /// `&mut T`, read off the kind — and off `kind()` rather than through
653    /// `is_exclusive_borrow`, which also sees through a `Box` and refuses an
654    /// out-parameter's slot. This is the borrow the source wrote.
655    fn is_mut_ref(t: &prebindgen_registry::flat::TypeRef) -> bool {
656        matches!(t.kind(), TypeKind::Ref { mutable: true, .. })
657    }
658    /// `Option<&T>` / `Option<&mut T>` → `Some(is_mut)`, the shape
659    /// `option_inner_ref_mutability` used to fish out of a path.
660    fn opt_ref_mut(t: &prebindgen_registry::flat::TypeRef) -> Option<bool> {
661        let TypeKind::Optional(inner) = t.kind() else {
662            return None;
663        };
664        match inner.kind() {
665            TypeKind::Ref { mutable, .. } => Some(*mutable),
666            _ => None,
667        }
668    }
669    let mut prelude: Vec<TokenStream> = Vec::new();
670    // Binding for the final `arg_ident` needs `mut` when the source
671    // fn takes `&mut T` — the call site below emits `&mut arg_ident`,
672    // which requires a mutable binding. Also for `Option<&mut T>`
673    // where the call site needs `.as_deref_mut()`.
674    let arg_mut: TokenStream = if is_mut_ref(arg_ty) || matches!(opt_ref_mut(arg_ty), Some(true)) {
675        quote!(mut)
676    } else {
677        quote!()
678    };
679    if let Some(value) = pipeline.invoke_infallible(quote!(#wire_ident), emit) {
680        prelude.push(quote!(let #arg_mut #arg_ident = #value;));
681    } else {
682        let decode_call = pipeline.invoke(quote!(#wire_ident), emit);
683        prelude.push(quote!(
684            let #arg_mut #arg_ident = match #decode_call {
685                ::core::result::Result::Ok(__v) => __v,
686                ::core::result::Result::Err(__e) => {
687                    signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &__e.to_string());
688                    return #on_err;
689                }
690            };
691        ));
692    }
693    let call_arg = match arg_ty.kind() {
694        TypeKind::Ref { mutable: true, .. } => quote!(&mut #arg_ident),
695        TypeKind::Ref { .. } => quote!(&#arg_ident),
696        // An Optional borrow carries either an owned value assembled by a
697        // registry chain (`Option<T>`) or an `OwnedObject<T>` pointing into a
698        // locked Kotlin handle. The frozen pipeline says which one it yields;
699        // the call site performs only the final borrow.
700        _ if matches!(opt_ref_mut(arg_ty), Some(false)) => {
701            if pipeline.borrowed_optional_value() {
702                quote!(#arg_ident.as_ref())
703            } else {
704                quote!(#arg_ident.as_deref())
705            }
706        }
707        _ if matches!(opt_ref_mut(arg_ty), Some(true)) => {
708            if pipeline.borrowed_optional_value() {
709                quote!(#arg_ident.as_mut())
710            } else {
711                quote!(#arg_ident.as_deref_mut())
712            }
713        }
714        _ => quote!(#arg_ident),
715    };
716    (prelude, call_arg)
717}

719/// Emit the wire params, decode prelude, and call argument for one
720/// constructor-expanded parameter. Each classified leaf is decoded with its
721/// own resolved input converter (reusing the by-value-handle consume fast
722/// path where the leaf is a direct owned handle); the leaves then feed
723/// [`prebindgen_registry::expand::emit_fold`], whose `Result<_, String>` is routed
724/// through the same error sink as any fallible input. The returned call
725/// argument is the built value (`&value` when the original parameter was `&T`).
726pub(crate) fn emit_expanded_param(
727    plan: &ExpandedParamPlan,
728    leaves: &[PlanLeaf],
729    orig_param: &syn::Ident,
730    on_err: &TokenStream,
731    emit: &prebindgen_registry::RustWriter,
732) -> (Vec<TokenStream>, Vec<TokenStream>, TokenStream) {
733    let mut wire_params: Vec<TokenStream> = Vec::new();
734    let fold = plan.fold();
735    let mut prelude: Vec<TokenStream> = Vec::new();
736    let mut leaf_locals: Vec<syn::Ident> = Vec::new();
737
738    debug_assert_eq!(fold.leaves.len(), leaves.len());
739    for (leaf, classified) in fold.leaves.iter().zip(leaves) {
740        let leaf_ty = &leaf.ty;
741        // The ascription generated Rust writes for this leaf's local.
742        let leaf_ty_tokens = emit.emit_source_type(leaf_ty);
743        let local = format_ident!("__exp_{}", leaf.name);
744        wire_params.extend(classified.native.iter().map(|native| {
745            let ident = &native.rust_ident;
746            let wire = &native.rust_wire;
747            quote!(#ident: #wire)
748        }));
749
750        match &classified.rust {
751            // An expansion leaf can itself be a data class. Reuse the
752            // recursive rebuild operation used by an ordinary parameter.
753            RustParamOp::FlattenStruct(flat) => {
754                let (decode, _) = render_flat_input_decode(flat, &local, on_err, emit);
755                prelude.push(decode);
756            }
757            // Selector-dispatched Optional leaf: rebuild from its already
758            // frozen `(present, value)` native pair.
759            RustParamOp::OptionalPair(sp) => {
760                let present_ident = &sp.present_ident;
761                let value_ident = &sp.value_ident;
762                let converter = emit.operation_ident("jni", &sp.chain.operation);
763                prelude.push(quote!(
764                    let #local: #leaf_ty_tokens = match #converter(
765                        &mut env,
766                        (#present_ident, #value_ident),
767                    ) {
768                        ::core::result::Result::Ok(__value) => __value,
769                        ::core::result::Result::Err(__error) => {
770                            signal_binding_error(
771                                &mut env,
772                                &__error_sink,
773                                &__SINK_MID,
774                                __SINK_FQN,
775                                __SINK_DESCR,
776                                &__error.to_string(),
777                            );
778                            return #on_err;
779                        }
780                    };
781                ));
782            }
783            RustParamOp::Pipeline { wire_ident } => {
784                let pipeline = &classified.pipeline;
785                let decode_call = pipeline.invoke(quote!(#wire_ident), emit);
786                prelude.push(quote!(
787                    let #local = match #decode_call {
788                        ::core::result::Result::Ok(__v) => __v,
789                        ::core::result::Result::Err(__e) => {
790                            signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &__e.to_string());
791                            return #on_err;
792                        }
793                    };
794                ));
795            }
796        }
797        leaf_locals.push(local);
798    }
799
800    // The fold itself (language-agnostic). Its `Err(String)` is lifted into
801    // `__JniErr` and routed through the same sink as fallible inputs.
802    let fold_expr = plan.emit(&leaf_locals);
803    let folded = format_ident!("__folded_{}", orig_param);
804    prelude.push(quote!(
805        let #folded = match #fold_expr {
806            ::core::result::Result::Ok(__v) => __v,
807            ::core::result::Result::Err(__e) => {
808                let __je = <__JniErr as ::core::convert::From<::std::string::String>>::from(__e);
809                signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &__je.to_string());
810                return #on_err;
811            }
812        };
813    ));
814
815    // `Option<&T>` ⇒ `folded.as_ref()`; `&T` ⇒ `&folded`; by-value (incl.
816    // `Option<T>`) ⇒ `folded`.
817    let call_arg = match (fold.produces_option(), fold.by_ref) {
818        (true, true) => quote!(#folded.as_ref()),
819        (false, true) => quote!(&#folded),
820        (_, false) => quote!(#folded),
821    };
822    (wire_params, prelude, call_arg)
823}
