1//! Turning recipes into whatever an adapter emits from.
2//!
3//! [`Recipes`] says how a value gets across in terms of its parts; an adapter
4//! says what that costs on the wire. Its answer for one recipe row is a **fragment**,
5//! a type the adapter defines and this module never looks inside. A fragment is
6//! not a wire value: it may occupy none, one or several, and the count is never
7//! asked for.
8//!
9//! An adapter writes [`Compile`], one hook per recipe shape. [`Compiler`] drives
10//! the recursion over the table, builds each recipe's fragment **once**, and hands
11//! every hook the fragments its parts already produced — so an adapter says
12//! what one crossing means and never writes the walk.
13//!
14//! # Per crossing, not per site
15//!
16//! A fragment is built once per crossing and reused at every site that crossing
17//! appears at. That is what makes emitted code shared without a converter
18//! table, and it rests on one assumption: **a fragment is context-free**. Where
19//! an adapter appears to need site context, the site is wrapping the fragment
20//! rather than the fragment differing, and [`Compile::plan`] — the one hook
21//! called per site — is where that wrapping belongs.
22//!
23//! A crossing here is the type **as the site spelled it**, which is finer than
24//! the [`RecipeKey`] a row is declared under. `Sample`, `&Sample` and `Box<Sample>`
25//! find one row, because the same row assembles all three — and they get
26//! three fragments, because taking a value out of a pointer, borrowing through
27//! one, and rebuilding a `Box` are three different pieces of Rust. Sharing the
28//! recipe is the declaration's business; sharing the code is not.
29

42/// What a fragment produces, which is the only thing the registry reads out of
43/// one.
44pub trait Carrier {
45    /// The Rust value this fragment yields.
46    fn yields(&self) -> Yield;
47
48    /// The crossings this fragment is built out of, which the completeness check
49    /// follows to decide what the binding must be able to make. Identities, not
50    /// spellings, so `T`, `&T` and `Box<T>` name one crossing.
51    ///
52    /// `parts` is what the registry handed the hook, which is the right answer
53    /// for most fragments. An adapter narrows where a part is no crossing of its
54    /// own — an opaque union payload, a boxed inner — and the `atomic` hook has
55    /// no parts, so a fragment that delegates from there must answer.
56    fn delegates_to(&self, parts: &[TypeKey]) -> Vec<TypeKey> {
57        parts.to_vec()
58    }
59
60    /// Take the shape the registry composed for this fragment.
61    ///
62    /// The registry knows which shape it is compiling and which child fragments
63    /// it handed to the hook, so it builds the [`ShapePlan`] rather than each
64    /// hook stating one. The fragment carries it as far as the `FragmentPlan`
65    /// the adapter freezes.
66    fn composed(&mut self, shape: ShapePlan);
67}

69/// The Rust value a fragment produces.
70#[derive(Clone, Debug)]
71pub struct Yield {
72    /// The type produced.
73    pub ty: TypeKey,
74    /// Handed over, or reached through a borrow.
75    pub mode: Mode,
76    /// How long what the fragment produces stays usable.
77    pub validity: Validity,
78}

80/// How long what a fragment produces stays usable.
81#[derive(Copy, Clone, Debug, PartialEq, Eq)]
82pub enum Validity {
83    /// The foreign side may keep it. Destroying it needs nothing else alive.
84    SelfSufficient,
85    /// Valid only while the value it was read from is alive.
86    Borrowed,
87}
88
89impl Validity {
90    /// Whether a value of this validity can be used where `needed` is required.
91    ///
92    /// Only one direction fails: something the foreign side will keep cannot be
93    /// borrowed from a value the call is about to drop.
94    pub fn satisfies(self, needed: Validity) -> bool {
95        needed == Validity::Borrowed || self == Validity::SelfSufficient
96    }
97}
98
99impl fmt::Display for Validity {
100    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
101        f.write_str(match self {
102            Validity::SelfSufficient => "self-sufficient",
103            Validity::Borrowed => "borrowed",
104        })
105    }
106}

170/// Where one part comes from.
171/// `Clone` and not `Copy`: `Path` carries its access chain, and a derived
172/// recipe is owned and dropped inside the compile, so the chain cannot be
173/// borrowed from it (#613 step 10).
174#[derive(Clone, Debug)]
175pub enum PartSource<'a> {
176    /// Parameter `index` of the constructor being called.
177    Argument {
178        /// Position in the constructor's parameter list.
179        index: usize,
180    },
181    /// A chain of field accesses, outermost first — an inlined nested class's
182    /// leaf. The one-hop case is [`Self::Field`] (#613 step 10).
183    Path {
184        /// Each hop's position in the struct it indexes. Borrowed from the
185        /// recipe row, which outlives the compile.
186        indices: Vec<usize>,
187        /// The field the chain arrives at.
188        field: &'a Field,
189    },
190    /// Field `index` of the value, read directly.
191    Field {
192        /// Position in the struct or in the arm's payload.
193        index: usize,
194        /// The field itself.
195        field: &'a Field,
196    },
197    /// The value itself, as one part — the compiled form of
198    /// [`Reach::Identity`]. Nothing is indexed and nothing is called.
199    Identity,
200    /// This accessor call.
201    Accessor {
202        /// The function to call.
203        func: &'a Function,
204    },
205}

207/// A product's parts, each already compiled.
208pub type Parts<'a, C> = &'a [(Part<'a>, &'a <C as Compile>::Fragment)];

210/// The context one hook receives, named by the adapter. [`Cx`] is generic over
211/// the **fragment** so an adapter that delegates to another can hand its own
212/// context straight through.
213pub type Ctx<'a, C> = Cx<'a, <C as Compile>::Fragment>;

215/// Why a hook has no fragment: an ordinary **gap** — nothing for this crossing,
216/// which the scan's over-approximation makes common and which matters only where
217/// an exported function reaches it — or an **error**, a wrong declaration,
218/// reported either way. Both were one answer before, which both adapters turned
219/// into a gap with `.ok()?`.
220#[derive(Debug)]
221pub enum Refusal<E> {
222    /// Nothing for this crossing, and why. A diagnostic: nothing fails on it.
223    Gap(String),
224    /// The declaration or the composition is wrong. Always reported.
225    Error(E),
226}
227
228/// So a hook can `?` its own error type: a helper that failed has said
229/// something is wrong, not that the crossing is unanswered.
230impl<E> From<E> for Refusal<E> {
231    fn from(error: E) -> Self {
232        Refusal::Error(error)
233    }
234}

237pub type Frag<C> = Result<<C as Compile>::Fragment, Refusal<<C as Compile>::Error>>;

239/// The adapter's half of code generation: what one crossing costs on the wire.
240///
241/// One hook per recipe shape, and one per operation, so an adapter never matches
242/// on [`Construct`] or [`Deconstruct`] — the registry calls only the hook that
243/// suits the recipe. Every hook is handed model elements rather than a copy of
244/// what the recipe said about them.
245pub trait Compile {
246    /// The adapter's answer for one recipe: what crosses, how it is encoded, what
247    /// it costs to clean up.
248    ///
249    /// Opaque to the registry, which only ever asks it what Rust value it
250    /// yields. Not a wire value — a fragment may occupy none, one or several.
251    type Fragment: Carrier;
252    /// The adapter's answer for one site: a root fragment plus the signature,
253    /// the call and the cleanup around it.
254    type Plan;
255    /// What the adapter reports when it cannot answer.
256    type Error;
257
258    /// No parts: the adapter emits the conversion itself.
259    fn atomic(&mut self, cx: &mut Ctx<'_, Self>, at: At<'_>) -> Frag<Self>;
260
261    /// Absent, or the inner fragment's value.
262    fn optional(
263        &mut self,
264        cx: &mut Ctx<'_, Self>,
265        at: At<'_>,
266        inner: &Self::Fragment,
267    ) -> Frag<Self>;
268
269    /// A run of the inner fragment's value.
270    ///
271    /// `elements` is derived from the collection type, not declared, so an
272    /// adapter reads it rather than checking it against anything.
273    fn sequence(
274        &mut self,
275        cx: &mut Ctx<'_, Self>,
276        at: At<'_>,
277        elements: Mode,
278        inner: &Self::Fragment,
279    ) -> Frag<Self>;
280
281    /// Call `func` to assemble the value. Its parameters are `args`.
282    fn construct(
283        &mut self,
284        cx: &mut Ctx<'_, Self>,
285        at: At<'_>,
286        func: &Function,
287        args: Parts<'_, Self>,
288    ) -> Frag<Self>;
289
290    /// Read the parts off the value where it stands.
291    fn fields(&mut self, cx: &mut Ctx<'_, Self>, at: At<'_>, parts: Parts<'_, Self>) -> Frag<Self>;
292
293    /// Call `func` once, then read the parts off what it returned.
294    fn value_form(
295        &mut self,
296        cx: &mut Ctx<'_, Self>,
297        at: At<'_>,
298        func: &Function,
299        parts: Parts<'_, Self>,
300    ) -> Frag<Self>;
301
302    /// Every arm, always.
303    ///
304    /// A choice is a run-time alternative, so the adapter is never offered one
305    /// of them to pick. Each arm arrives already composed by one of the four
306    /// hooks above, which is why this hook needs no direction of its own.
307    fn choice(
308        &mut self,
309        cx: &mut Ctx<'_, Self>,
310        at: At<'_>,
311        arms: &[(&Alternative, &Self::Fragment)],
312    ) -> Frag<Self>;
313
314    /// A callback, taken apart into the values that pass through it.
315    ///
316    /// `result` is `None` for every callback the model can describe today; it
317    /// is the position issue #216's return value lands in.
318    fn callback(
319        &mut self,
320        cx: &mut Ctx<'_, Self>,
321        at: At<'_>,
322        args: &[&Self::Fragment],
323        result: Option<&Self::Fragment>,
324    ) -> Frag<Self>;
325
326    /// The weakest validity this role accepts in **this** target.
327    ///
328    /// Not the registry's to decide, because it follows from the target's
329    /// ownership model. C hands out a `*const T` for a zero-copy accessor and
330    /// its contract says the caller neither frees nor outlives it, so a
331    /// borrowed return is correct there. The JVM keeps what it is given, so
332    /// JniGen clones instead and a borrowed return would be a use-after-free.
333    ///
334    /// The default is the strict reading: anything the foreign side may keep
335    /// past the call must be self-sufficient, and only a position that lives
336    /// for the duration of the call may borrow.
337    fn tolerates(&self, role: &Role) -> Validity {
338        match role {
339            Role::Return | Role::Error | Role::Const => Validity::SelfSufficient,
340            Role::Param { .. }
341            | Role::Receiver
342            | Role::CallbackArg { .. }
343            | Role::Part { .. }
344            // A leaf of an expanded parameter lives exactly as long as the
345            // parameter it came from, so it borrows on the same terms.
346            | Role::ExpansionLeaf { .. } => Validity::Borrowed,
347        }
348    }
349
350    /// Whether this adapter plans this site at all.
351    ///
352    /// The registry enumerates the positions a value *can* cross at, which is
353    /// the model's answer. Whether a given target has anything to do with one is
354    /// not: `prebindgen-c` plans nothing for a `()` return because C hands
355    /// nothing back there, while a JVM binding does; and JniGen answers a
356    /// callback parameter whole, so planning it as a site would freeze the same
357    /// position twice.
358    ///
359    /// The default is that every enumerated site is planned. An adapter that
360    /// says no gets no plan and no refusal — a position it has nothing to say
361    /// about is not a failure.
362    fn plans_site(&self, _site: &Site, _crossing: &Crossing) -> bool {
363        true
364    }
365
366    /// Which recipe this site takes, where the binding table cannot say.
367    ///
368    /// `None` — the default, and the answer for every site of most adapters —
369    /// means the binding decides. An adapter answers otherwise where the choice
370    /// follows from something compiled rather than from the model: JniGen picks
371    /// its `pair` row for an optional whose payload turned out to be a
372    /// niche-free primitive, which is a fact about the compiled payload and not
373    /// about the declaration.
374    ///
375    /// Asked before the root fragment is built, so the answer chooses which
376    /// fragment that is. The row must still be declared, so choosing one cannot
377    /// invent a representation outside the table.
378    fn site_recipe(&mut self, _cx: &mut Ctx<'_, Self>, _bound: &Bound) -> Option<RecipeName> {
379        None
380    }
381
382    /// Wrap the site's root fragment into a plan — the signature, the call, the
383    /// cleanup.
384    ///
385    /// The only hook called once per site; every hook above is called once per
386    /// recipe, however many sites reuse it.
387    fn plan(
388        &mut self,
389        cx: &mut Ctx<'_, Self>,
390        bound: &Bound,
391        root: &Self::Fragment,
392    ) -> Result<Self::Plan, Self::Error>;
393}

395/// What a compilation reports when it cannot finish.
396#[derive(Debug)]
397pub enum CompileError<E> {
398    /// The adapter refused a recipe or a site, possibly through a part rather
399    /// than through what was asked for. A [`Refusal::Gap`] is a failure only
400    /// where something has already reached the crossing — at a **site**, say.
401    Adapter(Refusal<E>),
402    /// The table and the model disagree with what a site asked for.
403    ///
404    /// Boxed: a `RecipeError` names a site and a crossing, which makes it much
405    /// the larger of the two variants, and this is the rare one.
406    Recipe(Box<RecipeError>),
407}
408
409impl<E: fmt::Display> fmt::Display for CompileError<E> {
410    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
411        match self {
412            CompileError::Adapter(Refusal::Gap(why)) => f.write_str(why),
413            CompileError::Adapter(Refusal::Error(e)) => write!(f, "{e}"),
414            CompileError::Recipe(e) => write!(f, "{e}"),
415        }
416    }
417}
418
419impl<E: fmt::Debug + fmt::Display> std::error::Error for CompileError<E> {}
420
421impl<E> From<RecipeError> for CompileError<E> {
422    fn from(e: RecipeError) -> Self {
423        CompileError::Recipe(Box::new(e))
424    }
425}

460/// What a compilation has produced, apart from its plans.
461///
462/// Carried between runs by an adapter whose view of the model is lent to it per
463/// callback rather than held: such an adapter is a different type on every call
464/// and so cannot keep one [`Compiler`], but the fragments it built borrow
465/// nothing and outlive any of them.
466pub struct Compiled<F> {
467    fragments: HashMap<FragmentId, Rc<F>>,
468    /// Which row answered when a crossing was compiled **as a whole**, rather
469    /// than as one part of a container. Recorded by [`Compiler::crossing`],
470    /// which is the only entry point that consults the crossing's default —
471    /// so [`Compiled::fragment`] can give back that same answer instead of
472    /// choosing between the recipes a crossing happens to have.
473    defaults: HashMap<(TypeKey, Direction), RecipeKey>,
474}
475
476impl<F> Default for Compiled<F> {
477    fn default() -> Self {
478        Self {
479            fragments: HashMap::new(),
480            defaults: HashMap::new(),
481        }
482    }
483}

485/// Cheap: a fragment sits behind an `Rc`, so a clone shares the fragments
486/// rather than copying them, and needs nothing of `F`.
487impl<F> Clone for Compiled<F> {
488    fn clone(&self) -> Self {
489        Self {
490            fragments: self.fragments.clone(),
491            defaults: self.defaults.clone(),
492        }
493    }

496impl<F> Compiled<F> {
497    /// The fragment for one crossing, compiled **as a whole**.
498    ///
499    /// What an adapter's emitters ask once they stop reading the converter
500    /// table. The answer is the one [`Compiler::crossing`] built — the
501    /// crossing's default recipe — and not one of the recipes that answer for this
502    /// type only as a part of some container. A caller that wants a particular
503    /// recipe has the recipe and asks through [`Self::recipe_fragment`].
504    ///
505    /// `None` means no site ever crossed this type in this direction.
506    ///
507    /// Handed back as the `Rc` it is stored under: an adapter that reads its
508    /// store while compilation is still filling it cannot hold a borrow across
509    /// the next write, and a fragment may hold a nontrivial semantic plan that
510    /// a recursive lookup should not copy.
511    pub fn fragment(&self, ty: &TypeKey, direction: Direction) -> Option<Rc<F>> {
512        let recipe = self.defaults.get(&(ty.clone(), direction))?;
513        self.recipe_fragment(ty, recipe)
514    }
515
516    /// The fragment for one spelled type and one row key.
517    pub fn recipe_fragment(&self, ty: &TypeKey, recipe: &RecipeKey) -> Option<Rc<F>> {
518        self.fragments
519            .get(&FragmentId::new(ty.clone(), recipe.clone()))
520            .cloned()
521    }
522
523    /// Every fragment this compilation built, in a deterministic order.
524    ///
525    /// What an adapter emits from once it stops routing its conversions back
526    /// through the converter table — handed to
527    /// [`write_rust`](crate::write::write_rust) directly. The order is by
528    /// crossing key and then by recipe name, so a file written from this is
529    /// stable across runs.
530    pub fn fragments(&self) -> Vec<&F> {
531        let mut keyed: Vec<(&FragmentId, &Rc<F>)> = self.fragments.iter().collect();
532        keyed.sort_by(|a, b| {
533            (a.0.spelling(), a.0.direction(), a.0.recipe().name()).cmp(&(
534                b.0.spelling(),
535                b.0.direction(),
536                b.0.recipe().name(),
537            ))
538        });
539        keyed.into_iter().map(|(_, f)| &**f).collect()
540    }
541}

543/// Drives an adapter over the table: one fragment per spelled crossing and
544/// globally identified row, one plan per site.
545pub struct Compiler<'a, C: Compile> {
546    conversions: &'a dyn Conversions,
547    recipes: &'a Recipes,
548    bindings: &'a Bindings,
549    compiled: Compiled<C::Fragment>,
550    /// Per fragment, the crossings it is built out of: the parts the registry
551    /// compiled for it while the row was being built, replaced by the
552    /// fragment's own [`Carrier::delegates_to`] once the hook has answered.
553    delegations: HashMap<FragmentId, Vec<TypeKey>>,
554}
555
556impl<'a, C: Compile> Compiler<'a, C> {
557    /// Drive `adapter` over this table.
558    pub fn new(
559        conversions: &'a dyn Conversions,
560        recipes: &'a Recipes,
561        bindings: &'a Bindings,
562    ) -> Self {
563        Self::resume(conversions, recipes, bindings, Compiled::default())
564    }
565
566    /// [`Self::new`], carrying on from what an earlier run built.
567    pub fn resume(
568        conversions: &'a dyn Conversions,
569        recipes: &'a Recipes,
570        bindings: &'a Bindings,
571        compiled: Compiled<C::Fragment>,
572    ) -> Self {
573        Self {
574            conversions,
575            recipes,
576            bindings,
577            compiled,
578            delegations: HashMap::new(),
579        }
580    }
581
582    /// Hand back what this run built, to carry into the next one.
583    pub fn finish(self) -> Compiled<C::Fragment> {
584        self.compiled
585    }
586
587    /// How many fragments have been built, making memoization observable.
588    pub fn compiled_fragments(&self) -> usize {
589        self.compiled.fragments.len()
590    }
591
592    /// Compile one site: pick its recipe, build the fragment, wrap it in a plan.
593    ///
594    /// `None` when a declaration bound the site to
595    /// [`Ask::Omit`](super::Ask::Omit).
596    pub fn site(
597        &mut self,
598        adapter: &mut C,
599        site: Site,
600        crossing: Crossing,
601    ) -> Result<Option<C::Plan>, CompileError<C::Error>> {
602        let Some(bound) = self.bindings.resolve(&site, &crossing, self.recipes) else {
603            return Ok(None);
604        };
605        // The adapter's own answer takes precedence, and only where it has one:
606        // a choice that follows from a compiled fragment is one the binding
607        // table cannot express.
608        let chosen = {
609            let mut cx = self.cx();
610            adapter.site_recipe(&mut cx, &bound)
611        };
612        let bound = match chosen {
613            None => bound,
614            Some(name) => {
615                let crossing_key = bound.crossing.key();
616                let Some(recipe) = self.recipes.key_of(&crossing_key, &name).cloned() else {
617                    return Err(RecipeError::NoSuchRecipe {
618                        recipe: crossing_key.row(name),
619                    }
620                    .into());
621                };
622                Bound {
623                    recipe,
624                    origin: Origin::Adapter,
625                    ..bound
626                }
627            }
628        };
629        self.plan(adapter, bound)
630    }
631
632    fn plan(
633        &mut self,
634        adapter: &mut C,
635        bound: Bound,
636    ) -> Result<Option<C::Plan>, CompileError<C::Error>> {
637        let root = self.recipe(adapter, &bound.crossing, &bound.recipe)?;
638        let needed = adapter.tolerates(&bound.site.role);
639        let got = root.yields().validity;
640        if !got.satisfies(needed) {
641            return Err(RecipeError::Validity {
642                site: bound.site,
643                needed,
644                got,
645            }
646            .into());
647        }
648        let mut cx = self.cx();
649        adapter
650            .plan(&mut cx, &bound, &root)
651            .map(Some)
652            .map_err(|e| CompileError::Adapter(e.into()))
653    }

655    /// The fragment for one crossing's default recipe.
656    ///
657    /// The per-recipe half of [`Self::site`], for an adapter that composes the
658    /// per-site wrapping itself. Built once and reused, like every other recipe.
659    /// Beside the fragment, the crossings it delegates to: what the registry's
660    /// reachability walk follows, and the reason [`Carrier::delegates_to`]
661    /// exists.
662    pub fn crossing(&mut self, adapter: &mut C, crossing: &Crossing) -> Answered<C> {
663        let recipe = self.recipes.recipe(crossing).0;
664        let fragment = self.recipe(adapter, crossing, &recipe)?;
665        let id = FragmentId::new(crossing.spelled().key(), recipe.clone());
666        // The whole-crossing answer, so the emitters can ask for it back
667        // without re-deriving which recipe a crossing defaults to.
668        self.compiled
669            .defaults
670            .insert((crossing.spelled().key(), crossing.direction()), recipe);
671        let delegates = self.delegations.get(&id).cloned().unwrap_or_default();
672        Ok((fragment, delegates))
673    }

675    /// The fragment for one crossing under a row of the table, rather than the
676    /// one the crossing defaults to.
677    ///
678    /// For a type whose table states more than one recipe: the default answers
679    /// every site, and this builds the other so it can be read through
680    /// [`Compiled::recipe_fragment`] before any site takes it. A key comes only
681    /// from [`Recipes::key_of`](super::Recipes::key_of), so this cannot reach
682    /// `recipe`'s fallback to the derived recipe — right for a name read *from*
683    /// the table, and wrong for a caller's own claim.
684    pub fn row(
685        &mut self,
686        adapter: &mut C,
687        crossing: &Crossing,
688        row: &RecipeKey,
689    ) -> Result<Rc<C::Fragment>, CompileError<C::Error>> {
690        self.recipe(adapter, crossing, row)
691    }

693    /// The fragment for one recipe, built once and reused.
694    fn recipe(&mut self, adapter: &mut C, crossing: &Crossing, recipe: &RecipeKey) -> Built<C> {
695        debug_assert_eq!(recipe.crossing(), &crossing.key());
696        // The row has global identity, while the fragment also needs the spelling:
697        // one row can serve `T`, `&T` and `Box<T>`, whose Rust differs.
698        let key = FragmentId::new(crossing.spelled().key(), recipe.clone());
699        if let Some(built) = self.compiled.fragments.get(&key) {
700            return Ok(built.clone());
701        }
702        let chosen = match self.recipes.get(recipe) {
703            Some(chosen) => chosen.clone(),
704            None => {
705                debug_assert_eq!(recipe.name(), &RecipeName::derived());
706                super::derive(crossing)
707            }
708        };
709        let at = At { crossing, recipe };
710        let (mut fragment, shape) = match &chosen {
711            Recipe::Constructing(shape) => self.constructing(adapter, at, shape)?,
712            Recipe::Deconstructing(shape) => self.deconstructing(adapter, at, shape)?,
713        };
714        fragment.composed(shape);
715        // What the hook was handed, now what the fragment says it delegates to.
716        let parts = self.delegations.remove(&key).unwrap_or_default();
717        self.delegations
718            .insert(key.clone(), fragment.delegates_to(&parts));
719        let fragment = Rc::new(fragment);
720        self.compiled.fragments.insert(key, fragment.clone());
721        Ok(fragment)
722    }

724    /// The fragment for one part of the recipe being compiled.
725    ///
726    /// Which row the part takes is the site machinery's answer, asked at
727    /// [`Role::Part`] — keyed by the recipe, because that is what compilation is
728    /// per. A root role such as [`Role::CallbackArg`] names a place in one
729    /// exported function, so it cannot answer for a recipe every function with
730    /// that signature shares; an adapter that wants a per-function answer
731    /// compiles that position as its own root site through [`Self::site`].
732    /// `direction` is the recipe's own except for a callback, whose arguments do
733    /// the other direction — Rust holds those values and pushes them out through
734    /// call. The **site** is still the recipe's either way: a part is identified
735    /// by the recipe that names it and its index, and a binding written against
736    /// that recipe has to match here, so only the part's `Crossing` carries the
737    /// swap.
738    ///
739    /// `wanted` is what the edge needs, and every edge states it: a product's
740    /// declared parameter or field mode, an optional's value, the mode a
741    /// collection lends its elements in, a callback argument's own.
742    fn part(
743        &mut self,
744        adapter: &mut C,
745        at: At<'_>,
746        direction: Direction,
747        index: usize,
748        ty: &TypeRef,
749        wanted: Mode,
750    ) -> Edge<C> {
751        self.part_of(adapter, at, direction, None, index, ty, wanted, false)
752    }

754    /// [`Self::part`] for a part inside a [`Shape::Choice`] arm, which numbers
755    /// its parts from zero like every other arm.
756    #[allow(clippy::too_many_arguments)]
757    fn part_of(
758        &mut self,
759        adapter: &mut C,
760        at: At<'_>,
761        direction: Direction,
762        arm: Option<usize>,
763        index: usize,
764        ty: &TypeRef,
765        wanted: Mode,
766        identity: bool,
767    ) -> Edge<C> {
768        let crossing = Crossing::new(ty.clone(), direction);
769        // Every edge the registry compiles for this row, which is the default
770        // answer to `Carrier::delegates_to`. An arm's parts land on the choice
771        // that owns them, because the arm has no fragment identity of its own.
772        self.delegations
773            .entry(at.fragment_id())
774            .or_default()
775            .push(ty.key());
776        let site = Site::arm_part(at.recipe, arm, index);
777        // An identity part IS its receiver, so resolving it the ordinary way
778        // finds the row being compiled and recurses without bound. It takes the
779        // crossing's DEFAULT row instead — the value's own converter, which is
780        // what a handle leaf delivers. A default that is the row being compiled
781        // is a cycle the declaration actually wrote, and says so (#613 step 10).
782        if identity {
783            let (row, _) = self.recipes.recipe(&crossing);
784            if row == *at.recipe {
785                return Err(RecipeError::Cycle {
786                    path: vec![at.crossing.key(), crossing.key()],
787                }
788                .into());
789            }
790            let fragment = self.recipe(adapter, &crossing, &row)?;
791            let edge = FragmentUse::new(
792                FragmentId::new(crossing.spelled().key(), row),
793                fragment.yields(),
794            );
795            return Ok((fragment, edge));
796        }
797        let Some(bound) = self.bindings.resolve(&site, &crossing, self.recipes) else {
798            return Err(RecipeError::UnknownRecipe {
799                site,
800                recipe: crossing.row(super::RecipeName::new("<omitted>")),
801            }
802            .into());
803        };
804        let fragment = self.recipe(adapter, &bound.crossing, &bound.recipe)?;
805        // Both contracts, here rather than at each caller: an optional's value,
806        // a run's element and a callback's argument are parts as much as a
807        // product's are, and a check written per edge is a check an edge can be
808        // added without.
809        let produced = fragment.yields();
810        let expected = part_key(ty);
811        if produced.ty != expected {
812            return Err(RecipeError::ComposedType {
813                site,
814                part: index,
815                wanted: expected,
816                got: produced.ty,
817            }
818            .into());
819        }
820        if !produced.mode.satisfies(wanted) {
821            return Err(RecipeError::Composition {
822                site,
823                part: index,
824                wanted,
825                got: produced.mode,
826            }
827            .into());
828        }
829        let edge = FragmentUse::new(
830            FragmentId::new(bound.crossing.spelled().key(), bound.recipe),
831            produced,
832        );
833        Ok((fragment, edge))
834    }

836    fn constructing(
837        &mut self,
838        adapter: &mut C,
839        at: At<'_>,
840        shape: &Shape<Construct>,
841    ) -> Composed<C> {
842        match shape {
843            Shape::Atomic => self.atomic(adapter, at),
844            Shape::Optional => self.optional(adapter, at),
845            Shape::Sequence => self.sequence(adapter, at),
846            Shape::Invoke => self.invoke(adapter, at),
847            Shape::Product(op) => {
848                let (kind, parts) = self.construct_parts(at, op, None)?;
849                let (fragment, uses) = self.product(adapter, at, None, kind, parts)?;
850                Ok((fragment, product_shape(at, uses)))
851            }
852            Shape::Choice { arms } => {
853                let mut built = Vec::new();
854                let mut arm_uses = Vec::new();
855                for arm in arms {
856                    let alternative = self.alternative(at, arm.alternative)?;
857                    let (kind, parts) =
858                        self.construct_parts(at, &arm.op, Some(&alternative.fields))?;
859                    let at_arm = Some(arm.alternative);
860                    let (fragment, uses) = self.product(adapter, at, at_arm, kind, parts)?;
861                    built.push((alternative, fragment));
862                    arm_uses.push(uses);
863                }
864                self.choice(adapter, at, built, arm_uses)
865            }
866        }
867    }

870        &mut self,
871        adapter: &mut C,
872        at: At<'_>,
873        shape: &Shape<Deconstruct>,
874    ) -> Composed<C> {
875        match shape {
876            Shape::Atomic => self.atomic(adapter, at),
877            Shape::Optional => self.optional(adapter, at),
878            Shape::Sequence => self.sequence(adapter, at),
879            Shape::Invoke => self.invoke(adapter, at),
880            Shape::Product(op) => {
881                let (kind, parts) = self.deconstruct_parts(at, op, None)?;
882                let (fragment, uses) = self.product(adapter, at, None, kind, parts)?;
883                Ok((fragment, product_shape(at, uses)))
884            }
885            Shape::Choice { arms } => {
886                let mut built = Vec::new();
887                let mut arm_uses = Vec::new();
888                for arm in arms {
889                    let alternative = self.alternative(at, arm.alternative)?;
890                    let (kind, parts) =
891                        self.deconstruct_parts(at, &arm.op, Some(&alternative.fields))?;
892                    let at_arm = Some(arm.alternative);
893                    let (fragment, uses) = self.product(adapter, at, at_arm, kind, parts)?;
894                    built.push((alternative, fragment));
895                    arm_uses.push(uses);
896                }
897                self.choice(adapter, at, built, arm_uses)
898            }
899        }

910    fn optional(&mut self, adapter: &mut C, at: At<'_>) -> Composed<C> {
911        // The payload is the crossing's own — `Option<T>` gives `T` — so the
912        // recipe states no inner type and there is none to disagree with.
913        let Some(inner) = at.crossing.value().optional_inner().cloned() else {
914            return Err(wrong_shape(at, "Optional", "an `Option`"));
915        };
916        let inner = &inner;
917        // Both layers: an `Option<&T>` holds a borrow, and a `&Option<T>` can
918        // only lend whatever it holds. Reading either alone is wrong in a
919        // different direction — `&Option<T>` would demand an owned `T` that
920        // reading through the shared optional cannot produce, and
921        // `&mut Option<T>` would demand an owned one where it lends `&mut T`.
922        let wanted = mode_of(inner).through(at.crossing.mode());
923        let (inner, value) = self.part(adapter, at, at.crossing.direction(), 0, inner, wanted)?;
924        let mut cx = self.cx();
925        let fragment = adapter
926            .optional(&mut cx, at, &inner)
927            .map_err(CompileError::Adapter)?;
928        Ok((
929            fragment,
930            ShapePlan::Optional {
931                bridge: bridge(at),
932                value,
933            },
934        ))
935    }

937    fn sequence(&mut self, adapter: &mut C, at: At<'_>) -> Composed<C> {
938        // The element is the crossing's own, for the reason `optional` gives.
939        let Some(inner) = super::sequence_elem(at.crossing.value()).cloned() else {
940            return Err(wrong_shape(at, "Sequence", "a `Vec`, slice or array"));
941        };
942        let inner = &inner;
943        // A run's element is held the way the collection lends it, which is
944        // what the adapter is told and so what its fragment must satisfy.
945        let elements = element_mode(at.crossing, inner);
946        let (inner, element) =
947            self.part(adapter, at, at.crossing.direction(), 0, inner, elements)?;
948        let mut cx = self.cx();
949        let fragment = adapter
950            .sequence(&mut cx, at, elements, &inner)
951            .map_err(CompileError::Adapter)?;
952        Ok((
953            fragment,
954            ShapePlan::Sequence {
955                bridge: bridge(at),
956                element,
957            },
958        ))
959    }

961    fn invoke(&mut self, adapter: &mut C, at: At<'_>) -> Composed<C> {
962        let direction = at.crossing.direction();
963        let Some(args) = at.crossing.value().callback_args().map(<[_]>::to_vec) else {
964            return Err(wrong_shape(at, "Invoke", "a callback type"));
965        };
966        // The one place the two directions swap: Rust holds these values and pushes
967        // them out through the call. The swap is the argument's, not the site's
968        // — the parts still belong to the callback recipe that names them.
969        let mut built = Vec::new();
970        let mut arguments = Vec::new();
971        for (index, arg) in args.iter().enumerate() {
972            let wanted = mode_of(arg);
973            let (fragment, edge) = self.part(adapter, at, direction.swap(), index, arg, wanted)?;
974            built.push(fragment);
975            arguments.push(edge);
976        }
977        let refs: Vec<&C::Fragment> = built.iter().map(|f| &**f).collect();
978        let mut cx = self.cx();
979        let fragment = adapter
980            .callback(&mut cx, at, &refs, None)
981            .map_err(CompileError::Adapter)?;
982        Ok((
983            fragment,
984            ShapePlan::Invoke {
985                bridge: FixedArity::new(arguments.len(), bridge(at)),
986                arguments,
987            },
988        ))
989    }

991    /// One product, whichever of the four hooks composes it.
992    fn product<'p>(
993        &mut self,
994        adapter: &mut C,
995        at: At<'_>,
996        arm: Option<usize>,
997        kind: ProductKind<'p>,
998        parts: Vec<Part<'p>>,
999    ) -> Assembled<C> {
1000        let mut parts = parts;
1001        if at.crossing.direction() == Direction::Deconstruct {
1002            for part in &mut parts {
1003                part.mode = part.mode.through(at.crossing.mode());
1004            }
1005        }
1006        let mut built = Vec::new();
1007        let mut uses = Vec::new();
1008        for (index, part) in parts.iter().enumerate() {
1009            // `part.mode` rather than the type's own spelling: a product edge
1010            // states what it needs — a constructor parameter, a field, an
1011            // accessor's receiver — and `part` checks against that.
1012            let (fragment, edge) = self.part_of(
1013                adapter,
1014                at,
1015                at.crossing.direction(),
1016                arm,
1017                index,
1018                &part.ty,
1019                part.mode,
1020                matches!(part.from, PartSource::Identity),
1021            )?;
1022            built.push(fragment);
1023            uses.push(edge);
1024        }
1025        let paired: Vec<(Part<'p>, &C::Fragment)> =
1026            parts.into_iter().zip(built.iter().map(|f| &**f)).collect();
1027        let mut cx = self.cx();
1028        let fragment = match kind {
1029            ProductKind::Construct(func) => adapter.construct(&mut cx, at, func, &paired),
1030            ProductKind::Fields => adapter.fields(&mut cx, at, &paired),
1031            ProductKind::ValueForm(func) => adapter.value_form(&mut cx, at, func, &paired),
1032        }
1033        .map_err(CompileError::Adapter)?;
1034        Ok((fragment, uses))
1035    }

1067    fn construct_parts(
1068        &self,
1069        at: At<'_>,
1070        op: &Construct,
1071        arm: Option<&'a [Field]>,
1072    ) -> Result<(ProductKind<'a>, Vec<Part<'a>>), CompileError<C::Error>> {
1073        match op {
1074            Construct::Fields => {
1075                let fields = match arm {
1076                    Some(fields) => fields,
1077                    None => self.fields_of(at.crossing.value()),
1078                };
1079                let parts = fields
1080                    .iter()
1081                    .enumerate()
1082                    .map(|(index, field)| Part {
1083                        from: PartSource::Field { index, field },
1084                        mode: mode_of(&field.ty),
1085                        ty: field.ty.clone(),
1086                        name: field_name(field, index),
1087                    })
1088                    .collect();
1089                Ok((ProductKind::Fields, parts))
1090            }
1091            Construct::Call(name) => {
1092                let func = self.function(at, name)?;
1093                let parts = func
1094                    .params
1095                    .iter()
1096                    .enumerate()
1097                    .map(|(index, param)| Part {
1098                        from: PartSource::Argument { index },
1099                        mode: mode_of(&param.ty),
1100                        ty: param.ty.clone(),
1101                        name: param.name.to_string(),
1102                    })
1103                    .collect();
1104                Ok((ProductKind::Construct(func), parts))
1105            }
1106        }
1107    }

1109    fn deconstruct_parts(
1110        &self,
1111        at: At<'_>,
1112        op: &Deconstruct,
1113        arm: Option<&'a [Field]>,
1114    ) -> Result<(ProductKind<'a>, Vec<Part<'a>>), CompileError<C::Error>> {
1115        match op {
1116            Deconstruct::Fields(reaches) => {
1117                let fields = match arm {
1118                    Some(fields) => fields,
1119                    None => self.fields_of(at.crossing.value()),
1120                };
1121                Ok((ProductKind::Fields, self.reaches(at, reaches, fields)?))
1122            }
1123            Deconstruct::ValueForm { func, parts } => {
1124                let func = self.function(at, func)?;
1125                let fields = self.fields_of(&func.ret);
1126                Ok((
1127                    ProductKind::ValueForm(func),
1128                    self.reaches(at, parts, fields)?,
1129                ))
1130            }
1131        }
1132    }

1134    fn reaches(
1135        &self,
1136        at: At<'_>,
1137        reaches: &[Reach],
1138        fields: &'a [Field],
1139    ) -> Result<Vec<Part<'a>>, CompileError<C::Error>> {
1140        let mut parts = Vec::new();
1141        for reach in reaches {
1142            match reach {
1143                Reach::Omit => {}
1144                // The nested shape's own parts, compiled against the field's
1145                // type. `deconstruct_parts` is the same entry the outer shape
1146                // took, one level in — so a nested `Choice` composes exactly as
1147                // a top-level one does (#613 step 10).
1148                Reach::Nested { index, shape } => {
1149                    let field = fields.get(*index).ok_or_else(|| {
1150                        CompileError::Recipe(Box::new(RecipeError::OutOfRange {
1151                            recipe: at.recipe.clone(),
1152                            index: *index,
1153                            len: fields.len(),
1154                        }))
1155                    })?;
1156                    let inner = self.fields_of(&field.ty);
1157                    let (_, mut inner_parts) = self.deconstruct_parts(at, shape, Some(inner))?;
1158                    parts.append(&mut inner_parts);
1159                }
1160                Reach::Path(indices) => {
1161                    // Walk the chain against the model, as validation did.
1162                    let mut here: &[Field] = fields;
1163                    let mut field: Option<&Field> = None;
1164                    for index in indices {
1165                        let hop = here.get(*index).ok_or_else(|| {
1166                            CompileError::Recipe(Box::new(RecipeError::OutOfRange {
1167                                recipe: at.recipe.clone(),
1168                                index: *index,
1169                                len: here.len(),
1170                            }))
1171                        })?;
1172                        field = Some(hop);
1173                        here = self.fields_of(&hop.ty);
1174                    }
1175                    let field = field.ok_or_else(|| {
1176                        CompileError::Recipe(Box::new(RecipeError::OutOfRange {
1177                            recipe: at.recipe.clone(),
1178                            index: 0,
1179                            len: 0,
1180                        }))
1181                    })?;
1182                    parts.push(Part {
1183                        from: PartSource::Path {
1184                            indices: indices.clone(),
1185                            field,
1186                        },
1187                        mode: mode_of(&field.ty),
1188                        ty: field.ty.clone(),
1189                        name: field_name(field, *indices.last().unwrap_or(&0)),
1190                    });
1191                }
1192                Reach::Field(index) => {
1193                    let field = fields.get(*index).ok_or_else(|| {
1194                        CompileError::Recipe(Box::new(RecipeError::OutOfRange {
1195                            recipe: at.recipe.clone(),
1196                            index: *index,
1197                            len: fields.len(),
1198                        }))
1199                    })?;
1200                    parts.push(Part {
1201                        from: PartSource::Field {
1202                            index: *index,
1203                            field,
1204                        },
1205                        mode: mode_of(&field.ty),
1206                        ty: field.ty.clone(),
1207                        name: field_name(field, *index),
1208                    });
1209                }
1210                Reach::Identity => parts.push(Part {
1211                    from: PartSource::Identity,
1212                    // From the SPELLED crossing, not `value()`: the latter
1213                    // strips `&`/`&mut`, which would record a borrowed
1214                    // identity row as `Owned` and lose the clone-for-borrow /
1215                    // move-for-owned distinction this form exists to carry
1216                    // (#635 review).
1217                    mode: at.crossing.mode(),
1218                    ty: at.crossing.value().clone(),
1219                    name: "self".to_string(),
1220                }),
1221                Reach::Accessor(name) => {
1222                    let func = self.function(at, name)?;
1223                    parts.push(Part {
1224                        from: PartSource::Accessor { func },
1225                        mode: mode_of(&func.ret),
1226                        ty: func.ret.clone(),
1227                        name: func.name.to_string(),
1228                    });
1229                }
1230            }
1231        }
1232        Ok(parts)
1233    }

1248    fn alternative(
1249        &self,
1250        at: At<'_>,
1251        index: usize,
1252    ) -> Result<&'a Alternative, CompileError<C::Error>> {
1253        let alternatives = match self.declared(at.crossing.value()) {
1254            Some(Type::Variant(v)) => v.alternatives.as_slice(),
1255            _ => &[],
1256        };
1257        alternatives.get(index).ok_or_else(|| {
1258            CompileError::Recipe(Box::new(RecipeError::OutOfRange {
1259                recipe: at.recipe.clone(),
1260                index,
1261                len: alternatives.len(),
1262            }))
1263        })
1264    }
1265
1266    fn fields_of(&self, ty: &TypeRef) -> &'a [Field] {
1267        match self.declared(ty) {
1268            Some(Type::Struct(s)) => s.fields.as_slice(),
1269            _ => &[],
1270        }
1271    }
1272
1273    fn declared(&self, ty: &TypeRef) -> Option<&'a Type> {
1274        let value = ty.borrow_target().unwrap_or(ty).unwrapped();
1275        match value.kind() {
1276            TypeKind::Named { id, .. } => self.conversions.flat().resolve(id),
1277            _ => None,
1278        }
1279    }
1280}

1308/// Whether a value is handed over or reached through a borrow.
1309fn mode_of(ty: &TypeRef) -> Mode {
1310    match ty.unwrapped().kind() {
1311        TypeKind::Ref { mutable: true, .. } => Mode::Exclusive,
1312        TypeKind::Ref { .. } => Mode::Shared,
1313        _ => Mode::Owned,
1314    }
1315}

1317/// How an element of a run is held once you have it.
1318///
1319/// Two facts, and both belong to the type rather than to the recipe.
1320///
1321/// How the collection hands an element over: `Vec<T>` gives its elements up,
1322/// `&[T]` and `Cow<'_, [T]>` lend them, and `&mut [T]` lends them exclusively —
1323/// a run reached through a borrow lends the same way it was reached, so
1324/// `&mut Vec<T>` yields `&mut T` and not `&T`.
1325///
1326/// And whether the element **is** a borrow: a `Vec<&T>` gives its elements up,
1327/// and what it gives up is a reference. So an element spelled `&T` is held as a
1328/// borrow however the collection hands it over, and reading only the collection
1329/// would call that element owned.
1330fn element_mode(crossing: &Crossing, elem: &TypeRef) -> Mode {
1331    // How the collection hands an element over, before the element's own
1332    // spelling is considered: a run reached through a borrow lends the same way
1333    // it was reached, and a bare `[T]` is only ever reached through one.
1334    let lent_as = match crossing.mode() {
1335        borrowed @ (Mode::Shared | Mode::Exclusive) => borrowed,
1336        Mode::Owned => match crossing.value().unwrapped().kind() {
1337            TypeKind::Slice(_) => Mode::Shared,
1338            _ => Mode::Owned,
1339        },
1340    };
1341    // Then the element's own, held through that. A `Vec<&T>` gives its elements
1342    // up and what it gives up is a borrow; a `&[&mut T]` yields `&&mut T` and
1343    // so cannot hand over the `&mut T` its element is spelled as.
1344    mode_of(elem).through(lent_as)
1345}
