23    /// The declared `opaque_ptr` under a union payload's spelling, when there is
24    /// one and the spelling does **not** already carry a `Box` —
25    /// `Option<Handle>` / `Handle` → `Some(Handle)`, `Option<Box<Handle>>` →
26    /// `None` (that shape keeps its own arm, so its emitted Rust does not move).
27    ///
28    /// Keyed on `stripped_key`, because a **declaration** is about the type: a
29    /// wrapper the model erases cannot change which declaration a payload
30    /// matches. See [`Self::payload_field_wire`] for why a union payload asks
31    /// this at all where a `repr_c_struct` mirror must not.
32    pub(super) fn declared_opaque_payload_inner(&self, fty: &TypeRef) -> Option<TypeKey> {
33        if r_boxed_inner(fty).is_some() {
34            return None;
35        }
36        let core = fty.optional_inner().unwrap_or(fty);
37        let key = core.stripped_key();
38        // The IDENTITY, not the stripped spelling: every caller turned that
39        // spelling straight back into this key (`c_type_ident`, `type_short`)
40        // or into the source path it names (`src_ty_of`).
41        self.opaque.contains_key(&key).then_some(key)
42    }

44    /// Wire type of one **tagged-union payload field**: the
45    /// [`Self::mirror_field_wire`] policy (scalar / declared `enum_type` /
46    /// opaque pointer `Option<Box<T>>` / `Box<T>`) extended with `String` →
47    /// a malloc'd `*mut c_char`, the same `char *` lowering a `data_struct`
48    /// field gets. `None` ⇒ the payload type cannot cross a tagged union.
49    ///
50    /// Unlike a `repr_c_struct` mirror — whose fields are reinterpreted
51    /// wholesale by one `Transmute` — a tagged union is rebuilt arm by arm,
52    /// so each wire here is produced by a real per-field conversion. That is
53    /// what lets `String` join the set.
54    ///
55    /// Every wire this returns is **bit-pattern-agnostic**: integers, floats and
56    /// raw pointers hold any bits legally, and the two Rust types that do *not*
57    /// — a declared `enum_type` (a discriminant no variant has) and `bool`
58    /// (anything but `0`/`1`) — are wrapped in [`::core::mem::MaybeUninit`] so
59    /// they do too. That is what makes the mirror's tag the *only* thing
60    /// [`CbindgenBuilder::in_tagged_union`] has to validate before `assume_init`.
61    ///
62    /// A `bool` reached through a nested `data_struct` payload is covered by
63    /// the same [`bool_wire`] policy, which [`c_field_wire`] and the plain
64    /// `bool` parameter now share (#170).
65    /// Why this type can **never** be a union payload, whatever converts it.
66    ///
67    /// The half of [`Self::payload_field_wire`] that is a fact about the union
68    /// rather than about the conversion: one union field carries one C wire, so
69    /// a shape needing two cannot ride in one however well it converts. The
70    /// other half — "no resolved converter" — stops being a question once the
71    /// payload is composed from its own part, which is a conversion in hand.
72    pub(super) fn payload_shape_refusal(&self, fty: &TypeRef) -> Result<(), String> {
73        if r_is_vec(fty) {
74            return Err(
75                "a `Vec` needs TWO C wires (pointer + length) and one union field carries only \
76                 one, so its length would be silently dropped — hand the sequence over through \
77                 a separate function, or wrap it in a declared `opaque_ptr` handle"
78                    .to_string(),
79            );
80        }
81        Ok(())
82    }

84    pub(super) fn payload_field_wire(&self, fty: &TypeRef) -> Result<syn::Type, String> {
85        // `String` is the one type whose two directions disagree on the wire
86        // (`*const c_char` in, `*mut c_char` out), so the union field fixes the
87        // OWNING form and the per-arm expressions convert by hand.
88        if r_is_string(fty) {
89            return Ok(syn::parse_quote!(*mut ::core::ffi::c_char));
90        }
91        if self.enums.contains_key(&fty.key()) {
92            let c = self.c_type_ident(&fty.key());
93            return Ok(syn::parse_quote!(::core::mem::MaybeUninit<#c>));
94        }
95        // `bool` is the one scalar with a restricted domain: `2` is a byte a C
96        // caller can write into the union and NOT a Rust `bool`, so holding it
97        // in the mirror is the same UB an out-of-range discriminant is. Same
98        // remedy everywhere C writes a `bool` — see `bool_wire`.
99        if r_is_bool(fty) {
100            return Ok(bool_wire());
101        }
102        // A `Vec` payload needs TWO C wires (pointer + length) and one union
103        // field can carry only one, so its length would be silently dropped.
104        // Rejected explicitly, because the converter-destination rule below
105        // would otherwise hand back the pointer alone and look like it worked.
106        if r_is_vec(fty) {
107            return Err(
108                "a `Vec` needs TWO C wires (pointer + length) and one union field carries only \
109                 one, so its length would be silently dropped — hand the sequence over through \
110                 a separate function, or wrap it in a declared `opaque_ptr` handle"
111                    .to_string(),
112            );
113        }
114        // Layout-identical shapes first (scalar, `Box<T>`/`Option<Box<T>>`
115        // opaque pointer), so what already worked keeps its exact wire.
116        if let Some(w) = self.mirror_field_wire(fty) {
117            return Ok(w);
118        }
119        // The opaque-pointer arm again, keyed on the **declaration** instead of
120        // on the spelling — which is what a *converted* position must do.
121        //
122        // `mirror_field_wire` above answers for a `repr_c_struct`, where the C
123        // type is a **layout** fact: the mirror is reinterpreted from the source
124        // struct's bytes, so `Box<T>` (a pointer) and `T` (inline) genuinely are
125        // different C types and the spelling is load-bearing. A union payload is
126        // not mirrored — it is rebuilt arm by arm through real conversions — so
127        // that reasoning does not carry over, and reusing the same
128        // `Box`-in-the-spelling test made an erased wrapper decide what C sees.
129        //
130        // Concretely: `Option<Box<Handle>>` crossed as `*mut handle_t` while
131        // `Option<Handle>` — the same optional handle to every destination
132        // language — was REFUSED, because it fell through to the
133        // converter-destination rule below where its output side is a structural
134        // marker (`()`) that cannot agree with the input's pointer. A wrapper the
135        // model erases decided whether the shape was expressible at all.
136        //
137        // So: peel the optional off the model and ask whether what is under it is
138        // a declared `opaque_ptr`. `stripped_key` rather than `key`, because a
139        // declaration is about the TYPE — see the same rule on the JNI side
140        // (#292). The two spellings now share this C type; their converter bodies
141        // differ, which is exactly the split (`kind` decides what C sees, syntax
142        // decides how the value is built).
143        if let Some(inner) = self.declared_opaque_payload_inner(fty) {
144            let c = self.c_type_ident(&inner);
145            return Ok(syn::parse_quote!(*mut #c));
146        }
147        // Otherwise the payload's wire is its **resolved converter
148        // destination** — the same source a `data_struct` field effectively
149        // uses. A union is rebuilt arm by arm through real per-field
150        // conversions, so its payloads are not constrained to the
151        // layout-preserving shapes a `repr_c_struct` mirror needs; this is
152        // what admits a nested `data_struct`, a bare `opaque_ptr` handle, and
153        // a converted leaf (`Duration` → `u64`).
154        //
155        // One field serves both directions, so they must agree on it. They can
156        // legitimately differ (a `String`'s const-ness above), which is why a
157        // disagreement is `None` — a rejection naming the payload — rather
158        // than a silent pick of one side.
159        let out_entry = self.out_frag(fty).ok_or_else(|| {
160            "no resolved OUTPUT converter — a payload crosses as its converter's destination, so \
161             it must be a scalar, a `String`, or a type this binding declares (`enum_type`, \
162             `data_struct`, `opaque_ptr`, or a `convert!` conversion)"
163                .to_string()
164        })?;
165        // Decided here rather than at the emit site, so every reason a payload
166        // can be refused is reported from ONE place, at the declaration.
167        if out_entry.function.call().fallible() {
168            return Err(
169                "its OUTPUT converter is fallible, but a union is encoded without an error \
170                 channel — the encoder always writes a live arm, so there is nowhere for the \
171                 failure to go. Use an infallible conversion for this payload"
172                    .to_string(),
173            );
174        }
175        let out = out_entry.destination.clone();
176        if let Some(inp) = self.in_frag(fty) {
177            if TypeKey::from_type(&inp.destination) != TypeKey::from_type(&out) {
178                return Err(format!(
179                    "its input and output converters disagree on the wire (`{}` in, `{}` out) \
180                     and one union field serves both directions",
181                    inp.destination.to_token_stream(),
182                    out.to_token_stream(),
183                ));
184            }
185        }
186        Ok(out)
187    }

189    /// Wire type of a `data_struct` field: the free [`c_field_wire`] policy
190    /// (`String` → `char *`, scalar → itself) plus a declared
191    /// [`CbindgenBuilder::tagged_union`] field, which crosses **by value** as its
192    /// `#[repr(C)]` mirror — the same way it crosses as a parameter or a
193    /// return. `None` ⇒ the field type is unsupported in a data struct.
194    ///
195    /// A union field with an owning payload keeps the data struct's existing
196    /// contract: the struct has no destructor, and each owning field is
197    /// released individually — here through the union's own typed drop.
198    ///
199    /// Like a union **parameter**, the field is wrapped in
200    /// [`::core::mem::MaybeUninit`]: one mirror struct serves both directions,
201    /// and on the way in its bytes are C's, so the field may not be a Rust enum
202    /// until its tag has been validated. Invisible in C either way.
203    pub(super) fn data_field_wire(&self, fty: &TypeRef) -> Option<syn::Type> {
204        if self.tagged_unions.contains_key(&fty.key()) {
205            let c = self.c_type_ident(&fty.key());
206            return Some(syn::parse_quote!(::core::mem::MaybeUninit<#c>));
207        }
208        if r_is_string(fty) {
209            return Some(syn::parse_quote!(*mut ::core::ffi::c_char));
210        }
211        // #170 instance 2: the field arrives from C by value, so it may not be
212        // a Rust `bool` until the byte has been normalised.
213        if r_is_bool(fty) {
214            return Some(bool_wire());
215        }
216        scalar_ty(fty)
217    }

219    /// True when a payload wire hands owned memory to C — a `char *` block or
220    /// an opaque pointer — and therefore has to be released by the union's
221    /// typed drop.
222    ///
223    /// A nested `data_struct` payload crosses BY VALUE, so the wire itself is
224    /// not a pointer, but its mirror's own fields may be: the union's drop
225    /// then has to reach through and release each of them through the recursive
226    /// payload-cleanup artifact plan. Without this a `String` or handle
227    /// inside a struct payload would leak, silently, for exactly the shape
228    /// zenoh-flat#30 needs.
229    pub(super) fn payload_wire_owns(
230        &self,
231        fty: &TypeRef,
232        wire: &syn::Type,
233        registry: &Registry,
234    ) -> bool {
235        if matches!(wire, syn::Type::Ptr(_)) {
236            return true;
237        }
238        !self.owning_data_struct_fields(fty, registry).is_empty()
239    }

241    /// The `(name, type)` of every field of a declared `data_struct` whose own
242    /// C wire owns memory. Empty when `fty` is not a declared data struct.
243    pub(super) fn owning_data_struct_fields<'r>(
244        &self,
245        fty: &TypeRef,
246        registry: &'r Registry,
247    ) -> Vec<(syn::Ident, &'r TypeRef)> {
248        if !self.data.contains_key(&fty.key()) {
249            return Vec::new();
250        }
251        self.struct_fields(registry, &fty.key())
252            .unwrap_or_default()
253            .into_iter()
254            .filter(|(_, fty)| self.data_field_owns(fty, registry))
255            .collect()
256    }

258    /// Whether one `data_struct` **field** hands owned memory to C: its own wire
259    /// is a pointer (`String` → `char *`), or it is a declared
260    /// [`CbindgenBuilder::tagged_union`] with an owning arm — which crosses by value,
261    /// so the pointer it owns is one level further down.
262    fn data_field_owns(&self, fty: &TypeRef, registry: &Registry) -> bool {
263        if matches!(self.data_field_wire(fty), Some(syn::Type::Ptr(_))) {
264            return true;
265        }
266        self.tagged_union_has_drop(fty, registry)
267    }

269    /// Whether a declared `tagged_union` gets a typed `<base>_drop` — i.e. it is
270    /// produced at all, and some arm's payload owns memory.
271    ///
272    /// This is the emission condition of the tagged-union artifact's drop *and*
273    /// the test for whether a
274    /// containing struct has to call it, so a union nested inside a payload
275    /// cannot be freed through a symbol that was never emitted. `false` for
276    /// anything that is not a declared tagged union.
277    pub(super) fn tagged_union_has_drop(&self, fty: &TypeRef, registry: &Registry) -> bool {
278        if !self.tagged_unions.contains_key(&fty.key()) || self.out_frag(fty).is_none() {
279            return false;
280        }
281        self.enum_alternatives(registry, &fty.key())
282            .unwrap_or_default()
283            .iter()
284            .flat_map(|a| a.fields.iter())
285            .any(|f| match self.payload_field_wire(&f.ty) {
286                // A rejected payload is reported from the emission site, which
287                // panics before any of this matters.
288                Err(_) => false,
289                Ok(wire) => self.payload_wire_owns(&f.ty, &wire, registry),
290            })
291    }

293    /// Fields (`name`, `type`) of a declared data struct, looked up from the
294    /// registry's indexed structs. `None` if the type isn't an indexed named
295    /// struct.
296    pub(super) fn struct_fields<'r>(
297        &self,
298        registry: &'r impl Conversions,
299        key: &TypeKey,
300    ) -> Option<Vec<(syn::Ident, &'r TypeRef)>> {
301        // The element, not its item. A `Struct` holds the field list the
302        // `syn::Fields::Named` match used to dig out, each field's name beside
303        // the reading of its type — so the name lookup is the key's own ident
304        // and the positional case is `f.name` being `None` rather than a
305        // `Fields` variant.
306        let st = registry.flat().struct_type(&key.ident()?)?;
307        st.fields
308            .iter()
309            .map(|f| Some((f.name.clone()?, &f.ty)))
310            .collect()
311    }

313    /// Wire type of a `repr_c_struct` field in the generated **visible** mirror: a
314    /// scalar passes through; a declared [`CbindgenBuilder::enum_type`] becomes its C enum;
315    /// an opaque pointer `Option<Box<T>>` / `Box<T>` (with `T` a declared
316    /// [`CbindgenBuilder::opaque_ptr`]) becomes `*mut t_t`. The whole-struct `Transmute`
317    /// (size/align-equal, asserted) then reinterprets each source field's bits into
318    /// this wire. `None` ⇒ the field type is unsupported in a `repr_c_struct`.
319    ///
320    /// Note what this policy cannot express: the wire *is* the source field's
321    /// type, so a field whose Rust type has restricted validity is reinterpreted
322    /// from C's bytes with no chance to check them. That gap is audited
323    /// separately by [`Self::restricted_validity_field`] — this function keeps
324    /// answering what the layout is.
325    pub(super) fn mirror_field_wire(&self, fty: &TypeRef) -> Option<syn::Type> {
326        if let Some(t) = scalar_ty(fty) {
327            return Some(t);
328        }
329        if self.enums.contains_key(&fty.key()) {
330            let c = self.c_type_ident(&fty.key());
331            return Some(syn::parse_quote!(#c));
332        }
333        // Opaque pointer: `Option<Box<T>>` (nullable, null-niche ↔ NULL) or `Box<T>`
334        // where `T` is a declared `opaque_ptr` → `*mut t_t`.
335        if let Some(inner) = r_boxed_inner(fty) {
336            if self.opaque.contains_key(&inner.key()) {
337                let c = self.c_type_ident(&inner.key());
338                return Some(syn::parse_quote!(*mut #c));
339            }
340        }
341        None
342    }

344    /// Why a `repr_c_struct` mirror field is **not** safe to reinterpret from
345    /// C-supplied bytes, or `None` when every bit pattern of its wire is a valid
346    /// value of the source field's type.
347    ///
348    /// Integers, floats and raw pointers hold any bits legally (holding a
349    /// garbage pointer is sound; dereferencing it is the caller's contract).
350    /// Two Rust types do not: `bool`, whose domain is `0`/`1` (#170 instance 3),
351    /// and a declared `enum_type`, whose domain is the declared discriminants
352    /// (#158 instance 3).
353    ///
354    /// The other positions these types cross — a parameter, a `data_struct`
355    /// field, a tagged-union payload — all go through a *per-value* converter,
356    /// so each has a hook: `bool` normalises through [`bool_wire`], an enum
357    /// validates its discriminant in `in_enum`. A `repr_c_struct` has no such
358    /// hook: one whole-struct `Transmute` reinterprets the mirror into the
359    /// source type, and by the time any generated code could look, the invalid
360    /// value already exists. Wrapping the mirror field in `MaybeUninit` moves
361    /// the problem rather than solving it — the transmute's *output* still has
362    /// the real field.
363    pub(super) fn restricted_validity_field(&self, fty: &TypeRef) -> Option<&'static str> {
364        if r_is_bool(fty) {
365            return Some("`bool` — only `0` and `1` are valid");
366        }
367        if self.enums.contains_key(&fty.key()) {
368            return Some("a declared `enum_type` — only its declared discriminants are valid");
369        }
370        None
371    }

373    /// The audit of [`Self::restricted_validity_field`] over one
374    /// `repr_c_struct`'s mirror: the offending `(field, reason)` pairs, in
375    /// declaration order. Empty ⇒ the whole mirror is safe to reinterpret.
376    pub(super) fn restricted_validity_fields(
377        &self,
378        registry: &Registry,
379        key: &TypeKey,
380    ) -> Vec<(syn::Ident, &'static str)> {
381        self.struct_fields(registry, key)
382            .unwrap_or_default()
383            .into_iter()
384            .filter_map(|(fname, fty)| {
385                self.restricted_validity_field(fty)
386                    .map(|reason| (fname, reason))
387            })
388            .collect()
389    }

391    /// Exported `#[no_mangle]` symbol for a declared function:
392    /// [`Self::mangle_function`] over the base — a `.base_name(...)` override when
393    /// set, else the Rust fn ident — or that base verbatim when no mangler is set.
394    pub(super) fn fn_symbol(&self, orig: &syn::Ident) -> syn::Ident {
395        let base = self
396            .functions
397            .get(orig)
398            .and_then(|c| c.base.clone())
399            .unwrap_or_else(|| orig.to_string());
400        match &self.mangle_function {
401            Some(f) => format_ident!("{}", f(&base)),
402            None => format_ident!("{}", base),
403        }
404    }

406    /// How one parameter *uses* the resource it names — the axis the alias rule
407    /// is stated on. `None` ⇒ the parameter names no single owned resource (a
408    /// scalar, a string, a slice block, an undeclared type), so it cannot alias
409    /// one.
410    ///
411    /// `T` and `Option<T>` share a resource domain: both arrive as the same
412    /// handle pointer, and comparing the syntactic parameter type instead would
413    /// miss `f(x: ZThing, y: Option<ZThing>)` called with the same handle
414    /// twice.
415    /// [`Self::alias_slot`] off the classification: the optional peel, the
416    /// borrow and its mutability, and the `MaybeUninit` under a `&mut` are all
417    /// what `TypeKind` states — where this walked `Option`'s type argument, a
418    /// `syn::Type::Reference` and its `mutability` field.
419    pub(super) fn alias_slot_of(&self, ty: &TypeRef) -> Option<(TypeKey, AliasAccess)> {
420        let inner = ty.optional_inner().unwrap_or(ty);
421        let declared =
422            |key: &TypeKey| self.opaque.contains_key(key) || self.value_opaque.contains_key(key);
423        if let TypeKind::Ref { mutable, inner, .. } = inner.kind() {
424            // `&mut MaybeUninit<T>` borrows T's slot exclusively just as
425            // `&mut T` does; the wire is the same pointer.
426            let elem = match inner.kind() {
427                TypeKind::Uninit(t) => t,
428                _ => inner,
429            };
430            let key = elem.key();
431            if !declared(&key) {
432                return None;
433            }
434            return Some((
435                key,
436                if *mutable {
437                    AliasAccess::Exclusive
438                } else {
439                    AliasAccess::Shared
440                },
441            ));
442        }
443        let key = inner.key();
444        declared(&key).then_some((key, AliasAccess::Consume))
445    }
