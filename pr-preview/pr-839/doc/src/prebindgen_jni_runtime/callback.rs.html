1//! Retained JVM callbacks and argument ownership transfer.
2use crate::LocalFrame;
3use jni::{
4    objects::{GlobalRef, JMethodID, JObject},
5    JNIEnv, JavaVM,
6};
7use prebindgen_runtime::{boundary, Result};
8
9/// A JVM callback retained until its owning Rust closure is dropped.
10///
11/// The object pins its class, so the method resolved at registration remains valid
12/// when a native worker thread invokes it later.
13pub struct Callback {
14    object: GlobalRef,
15    vm: JavaVM,
16    method: JMethodID,
17    name: &'static str,
18    signature: Option<&'static str>,
19}
20impl Callback {
21    /// Retain `object` and resolve its generated void `run` method once.
22    pub fn new(env: &mut JNIEnv<'_>, object: &JObject<'_>, descriptor: &str) -> Result<Self> {
23        if object.is_null() {
24            return Err("null callback".into());
25        }
26        let class = env.get_object_class(object).map_err(|e| e.to_string())?;
27        let method = env
28            .get_method_id(&class, "run", descriptor)
29            .map_err(|e| e.to_string())?;
30        env.delete_local_ref(class).map_err(|e| e.to_string())?;
31        let object = env.new_global_ref(object).map_err(|e| e.to_string())?;
32        let vm = env.get_java_vm().map_err(|e| e.to_string())?;
33        Ok(Self {
34            object,
35            vm,
36            method,
37            name: "<unnamed callback>",
38            signature: None,
39        })
40    }
41
42    /// Retain a generated callback with static diagnostic name and JVM signature.
43    ///
44    /// The context is recorded at registration; successful invocations perform
45    /// no logging, string formatting or diagnostic allocations.
46    pub fn new_named(
47        env: &mut JNIEnv<'_>,
48        object: &JObject<'_>,
49        descriptor: &'static str,
50        name: &'static str,
51    ) -> Result<Self> {
52        let mut callback = Self::new(env, object, descriptor)?;
53        callback.name = name;
54        callback.signature = Some(descriptor);
55        Ok(callback)
56    }
57
58    #[cold]
59    #[inline(never)]
60    fn report(&self, error: &str) {
61        prebindgen_runtime::report_callback_error(
62            self.name,
63            self.signature.unwrap_or("<dynamic signature>"),
64            error,
65        );
66    }
67
68    /// Run an upcall on the current native thread with a bounded local frame.
69    ///
70    /// Callback failures have no live declaring-call error handler. Report them
71    /// and clear JVM exceptions so subsequent invocations remain usable.
72    #[inline]
73    pub fn with_env(&self, capacity: i32, call: impl FnOnce(&mut JNIEnv<'_>) -> Result<()>) {
74        let result = boundary(|| {
75            let mut env = self
76                .vm
77                .attach_current_thread_as_daemon()
78                .map_err(|e| e.to_string())?;
79            let result = boundary(|| {
80                let frame = unsafe { LocalFrame::new(&env, capacity) }?;
81                let result = call(&mut env);
82                drop(frame);
83                result
84            });
85            if result.is_err() {
86                let _ = env.exception_describe();
87                let _ = env.exception_clear();
88            }
89            result
90        });
91        if let Err(error) = result {
92            self.report(&error);
93        }
94    }
95
96    /// Run a callback returning an owned Rust value inside its JNI local frame.
97    ///
98    /// Failures are returned after any pending callback exception is reported and
99    /// cleared. The source trampoline decides how to propagate the missing result.
100    #[inline]
101    pub fn with_env_result<R>(
102        &self,
103        capacity: i32,
104        call: impl FnOnce(&mut JNIEnv<'_>) -> Result<R>,
105    ) -> Result<R> {
106        let result = boundary(|| {
107            let mut env = self
108                .vm
109                .attach_current_thread_as_daemon()
110                .map_err(|e| e.to_string())?;
111            let result = boundary(|| {
112                let frame = unsafe { LocalFrame::new(&env, capacity) }?;
113                let result = call(&mut env);
114                drop(frame);
115                result
116            });
117            if result.is_err() {
118                let _ = env.exception_describe();
119                let _ = env.exception_clear();
120            }
121            result
122        });
123        if let Err(error) = &result {
124            self.report(error);
125        }
126        result
127    }
128
129    /// Invoke a cached non-unit method and keep its result in the caller's frame.
130    ///
131    /// # Safety
132    /// Arguments and `return_type` must match the descriptor passed to [`Self::new`].
133    /// References must be live in `env`, which must have no pending JVM exception.
134    #[inline]
135    pub unsafe fn invoke_return<'local>(
136        &self,
137        env: &mut JNIEnv<'local>,
138        return_type: jni::signature::ReturnType,
139        arguments: &[jni::sys::jvalue],
140    ) -> Result<jni::objects::JValueOwned<'local>> {
141        env.call_method_unchecked(&self.object, self.method, return_type, arguments)
142            .map_err(|e| e.to_string())
143    }
144
145    /// Invoke the cached method. Committed argument handles are Kotlin-owned.
146    ///
147    /// Argument ownership transfers without a per-invocation acceptance handshake.
148    /// Kotlin cleans up handles after entry; failure to enter the bridge can leak them.
149    ///
150    /// # Safety
151    /// `arguments` must match the descriptor passed to [`Self::new`], with live
152    /// references in `env`. The environment must have no pending JVM exception.
153    #[inline]
154    pub unsafe fn invoke(
155        &self,
156        env: &mut JNIEnv<'_>,
157        arguments: &[jni::sys::jvalue],
158    ) -> Result<()> {
159        env.call_method_unchecked(
160            &self.object,
161            self.method,
162            jni::signature::ReturnType::Primitive(jni::signature::Primitive::Void),
163            arguments,
164        )
165        .map(|_| ())
166        .map_err(|e| e.to_string())
167    }
168}
