1//! Structural descriptions of conversion values; adapters interpret node metadata.
2use crate::{Wire, WireKind};

4/// A value's wire definitions arranged as terminal bundles and parent nodes.
5///
6/// `C` describes a parent with exactly one child; `B` describes a parent with an
7/// ordered child list. Adapters decide what those descriptions mean: for example,
8/// optional and sequence encodings can share the chain shape, while fields and
9/// alternatives can share the branch shape. The tree imposes no delivery policy.
10/// Every wire belongs to a leaf or a parent's `controls`. Metadata may contain
11/// policies, annotations and references to those wires, but no wire descriptors.
12/// Controls belong to the parent occurrence, outside its child payloads.
13///
14/// Chain and branch metadata cannot be interchanged:
15/// ```compile_fail,E0308
16/// use prebindgen_flat::WireNode;
17/// struct ChainInfo;
18/// struct BranchInfo;
19/// let node: WireNode<syn::Type, ChainInfo, BranchInfo> = WireNode::Chain {
20///     controls: vec![], child: Box::default(), meta: BranchInfo,
21/// };
22/// ```
23/// Completion channels surround value trees and cannot be nested inside them:
24/// ```compile_fail,E0308
25/// use prebindgen_flat::{Outcomes, WireNode};
26/// let node: WireNode<syn::Type> = WireNode::Chain {
27///     controls: vec![], child: Box::new(Outcomes::<WireNode<syn::Type>>::default()), meta: (),
28/// };
29/// ```
30#[derive(Clone, Debug)]
31pub enum WireNode<W: WireKind, C = (), B = ()> {
32    /// A terminal bundle of values; an empty bundle carries no wires.
33    Leaf(Vec<Wire<W>>),
34    /// One child description, interpreted according to adapter metadata.
35    Chain {
36        /// Adapter policy or annotations for this parent.
37        meta: C,
38        /// Exactly one child description; runtime occurrence count is adapter-defined.
39        child: Box<WireNode<W, C, B>>,
40        /// Supporting wire definitions outside the child payload.
41        controls: Vec<Wire<W>>,
42    },
43    /// Ordered child descriptions, interpreted according to adapter metadata.
44    Branch {
45        /// Adapter policy or annotations for this parent.
46        meta: B,
47        /// Ordered child descriptions. Metadata determines whether an empty list is valid.
48        children: Vec<WireNode<W, C, B>>,
49        /// Supporting wire definitions outside the child payloads.
50        controls: Vec<Wire<W>>,
51    },
52}
53impl<W: WireKind, C, B> Default for WireNode<W, C, B> {
54    fn default() -> Self {
55        Self::Leaf(vec![])
56    }
57}

59/// Completion channels surrounding output value trees; never children of a value.
60#[derive(Clone, Debug)]
61pub struct Outcomes<N> {
62    /// Successful completion, including unit.
63    pub success: N,
64    /// Projected `Err(E)`; an empty value tree can represent a payload-free error.
65    pub source_error: Option<N>,
66    /// Conversion or invocation failure, separate from the source's `Err(E)`.
67    pub binding_failure: Option<N>,
68}
69impl<N: Default> Default for Outcomes<N> {
70    fn default() -> Self {
71        Self {
72            success: N::default(),
73            source_error: None,
74            binding_failure: None,
75        }
76    }
77}

79/// Visit all wire definitions, controls before children, without interpreting metadata.
80/// Each child description is visited once, regardless of runtime repetition or selection.
81pub fn wire_definitions<'a, W: WireKind + 'a, C: 'a, B: 'a>(
82    nodes: impl IntoIterator<Item = &'a WireNode<W, C, B>>,
83) -> impl std::iter::FusedIterator<Item = &'a Wire<W>> {
84    enum Frame<'a, W: WireKind, C, B> {
85        Nodes(std::slice::Iter<'a, WireNode<W, C, B>>),
86        Wires(std::slice::Iter<'a, Wire<W>>),
87    }
88    let mut roots = nodes.into_iter().fuse();
89    let mut stack = vec![];
90    std::iter::from_fn(move || loop {
91        let node = match stack.last_mut() {
92            Some(Frame::Wires(wires)) => {
93                if let Some(w) = wires.next() {
94                    return Some(w);
95                }
96                stack.pop();
97                continue;
98            }
99            Some(Frame::Nodes(nodes)) => {
100                if let Some(n) = nodes.next() {
101                    n
102                } else {
103                    stack.pop();
104                    continue;
105                }
106            }
107            None => roots.next()?,
108        };
109        match node {
110            WireNode::Leaf(w) => stack.push(Frame::Wires(w.iter())),
111            WireNode::Chain {
112                controls, child, ..
113            } => {
114                stack.push(Frame::Nodes(std::slice::from_ref(child.as_ref()).iter()));
115                stack.push(Frame::Wires(controls.iter()));
116            }
117            WireNode::Branch {
118                controls, children, ..
119            } => {
120                stack.push(Frame::Nodes(children.iter()));
121                stack.push(Frame::Wires(controls.iter()));
122            }
123        }
124    })
125    .fuse()
126}
127/// Mutably visit every definition without requiring cloneable wires or metadata.
128/// Traversal order matches [`wire_definitions`].
129pub fn wire_definitions_mut<'a, W: WireKind + 'a, C: 'a, B: 'a>(
130    nodes: impl IntoIterator<Item = &'a mut WireNode<W, C, B>>,
131) -> impl std::iter::FusedIterator<Item = &'a mut Wire<W>> {
132    enum Frame<'a, W: WireKind, C, B> {
133        Nodes(std::slice::IterMut<'a, WireNode<W, C, B>>),
134        Wires(std::slice::IterMut<'a, Wire<W>>),
135    }
136    let mut roots = nodes.into_iter().fuse();
137    let mut stack = vec![];
138    std::iter::from_fn(move || loop {
139        let node = match stack.last_mut() {
140            Some(Frame::Wires(wires)) => {
141                if let Some(w) = wires.next() {
142                    return Some(w);
143                }
144                stack.pop();
145                continue;
146            }
147            Some(Frame::Nodes(nodes)) => {
148                if let Some(n) = nodes.next() {
149                    n
150                } else {
151                    stack.pop();
152                    continue;
153                }
154            }
155            None => roots.next()?,
156        };
157        match node {
158            WireNode::Leaf(w) => stack.push(Frame::Wires(w.iter_mut())),
159            WireNode::Chain {
160                controls, child, ..
161            } => {
162                stack.push(Frame::Nodes(
163                    std::slice::from_mut(child.as_mut()).iter_mut(),
164                ));
165                stack.push(Frame::Wires(controls.iter_mut()));
166            }
167            WireNode::Branch {
168                controls, children, ..
169            } => {
170                stack.push(Frame::Nodes(children.iter_mut()));
171                stack.push(Frame::Wires(controls.iter_mut()));
172            }
173        }
174    })
175    .fuse()
176}
177/// Address relative to selected roots. The first index selects a root. At any
178/// parent, `[0, i]` selects control `i` and `[1, i]` selects child `i`; a chain
179/// has only child zero. At a leaf, one final index selects a wire in its bundle.
180/// Select the output channel before traversal. Rebuild paths after structural
181/// edits or changing the selected roots.
182#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
183pub struct WirePath(pub Vec<usize>);
184/// A wire definition borrowed together with its structural address.
185pub struct WireEntry<'a, W: WireKind> {
186    /// Address within the selected roots.
187    pub path: WirePath,
188    /// Definition in a leaf or parent controls.
189    pub wire: &'a Wire<W>,
190}
191/// Visit all wire definitions with paths, in [`wire_definitions`] order.
192pub fn wire_definitions_with_paths<'a, W: WireKind + 'a, C: 'a, B: 'a>(
193    nodes: impl IntoIterator<Item = &'a WireNode<W, C, B>>,
194) -> impl std::iter::FusedIterator<Item = WireEntry<'a, W>> {
195    fn visit<'a, W: WireKind, C, B>(
196        node: &'a WireNode<W, C, B>,
197        path: &mut Vec<usize>,
198        out: &mut Vec<WireEntry<'a, W>>,
199    ) {
200        let (controls, children) = match node {
201            WireNode::Leaf(wires) => {
202                for (i, wire) in wires.iter().enumerate() {
203                    path.push(i);
204                    out.push(WireEntry {
205                        path: WirePath(path.clone()),
206                        wire,
207                    });
208                    path.pop();
209                }
210                return;
211            }
212            WireNode::Chain {
213                controls, child, ..
214            } => (controls.as_slice(), std::slice::from_ref(child.as_ref())),
215            WireNode::Branch {
216                controls, children, ..
217            } => (controls.as_slice(), children.as_slice()),
218        };
219        for (i, wire) in controls.iter().enumerate() {
220            path.extend([0, i]);
221            out.push(WireEntry {
222                path: WirePath(path.clone()),
223                wire,
224            });
225            path.truncate(path.len() - 2);
226        }
227        for (i, child) in children.iter().enumerate() {
228            path.extend([1, i]);
229            visit(child, path, out);
230            path.truncate(path.len() - 2);
231        }
232    }
233    let mut out = vec![];
234    for (i, node) in nodes.into_iter().enumerate() {
235        visit(node, &mut vec![i], &mut out);
236    }
237    out.into_iter()
238}
239/// Resolve a path for the same roots and output channel used when producing it.
240pub fn wire_at_path<'a, W: WireKind + 'a, C: 'a, B: 'a>(
241    nodes: impl IntoIterator<Item = &'a WireNode<W, C, B>>,
242    path: &WirePath,
243) -> Option<&'a Wire<W>> {
244    fn get<'a, W: WireKind, C, B>(
245        node: &'a WireNode<W, C, B>,
246        path: &[usize],
247    ) -> Option<&'a Wire<W>> {
248        let (&index, rest) = path.split_first()?;
249        let (controls, children) = match node {
250            WireNode::Leaf(wires) => return rest.is_empty().then(|| wires.get(index)).flatten(),
251            WireNode::Chain {
252                controls, child, ..
253            } => (controls.as_slice(), std::slice::from_ref(child.as_ref())),
254            WireNode::Branch {
255                controls, children, ..
256            } => (controls.as_slice(), children.as_slice()),
257        };
258        match (index, rest) {
259            (0, [i]) => controls.get(*i),
260            (1, [i, rest @ ..]) => get(children.get(*i)?, rest),
261            _ => None,
262        }
263    }
264    let (&root, rest) = path.0.split_first()?;
265    get(nodes.into_iter().nth(root)?, rest)
266}
