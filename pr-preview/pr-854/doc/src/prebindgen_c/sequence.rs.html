7/// Selected delivery of a logical sequence.
8#[derive(Clone, Copy, Debug, PartialEq, Eq)]
9pub enum CSequence {
10    /// All rows travel in one collection carrier.
11    Buffered,
12    /// Each row is delivered through the function's sequence callback.
13    Callback,
14}

16pub(crate) fn input(
17    wire: syn::Ident,
18    length: syn::Ident,
19    index: syn::Ident,
20    element: &prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
21) -> Result<SequenceInput<CWire, crate::CChain>> {
22    let row = prebindgen_conversions::template_parameter(&format!("row_{wire}"));
23    let mut bind = TokenStream::new();
24    let mut values = Vec::new();
25    for (position, field) in crate::group::row_fields(element).enumerate() {
26        let name = &field.name;
27        let member = format_ident!("v{position}");
28        bind.extend(
29            quote!(let #name=unsafe {(*((#wire.data as *const #row).add(#index))).#member};),
30        );
31        values.push((field.name.clone(), quote!(#name)));
32    }
33    Ok(SequenceInput {
34        controls: vec![prebindgen_flat::Wire {
35            name: wire.clone(),
36            kind: CWire::Rows,
37        }],
38        metadata: CChain::Sequence(CSequence::Buffered),
39        open: quote!(let #length=#wire.len;::prebindgen_runtime::validate_slice(#wire.data as *const #row,#length)?;),
40        bind,
41        validation_values: values,
42        release: quote!(),
43    })
44}
45pub(crate) fn output(
46    wire: syn::Ident,
47    length: syn::Ident,
48    element: &prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
49    values: &[(syn::Ident, TokenStream)],
50) -> Result<SequenceOutput<CWire, crate::CChain>> {
51    let row = prebindgen_conversions::template_parameter(&format!("row_{wire}"));
52    let fields = crate::group::row_values(element, values)?
53        .into_iter()
54        .enumerate()
55        .map(|(i, (_, value))| {
56            let field = format_ident!("v{i}");
57            quote!(#field:#value)
58        });
59    Ok(SequenceOutput {
60        controls: vec![prebindgen_flat::Wire {
61            name: wire.clone(),
62            kind: CWire::Rows,
63        }],
64        commit_each: false,
65        fallible: false,
66        metadata: CChain::Sequence(CSequence::Buffered),
67        begin: quote!(let mut #wire:Vec<#row>=Vec::with_capacity(#length);),
68        enter: quote!(),
69        store: quote!(#wire.push(#row {#(#fields),*});),
70        finish: quote!(let #wire=#wire.into_boxed_slice();),
71        values: vec![(
72            wire.clone(),
73            quote!(__PbRows {data:if #wire.is_empty(){::core::ptr::null()}else{#wire.as_ptr().cast()},len:#wire.len()}),
74        )],
75        commit: quote!(let _=Box::into_raw(#wire);),
76    })
77}

79/// Visit every sequence, including callback elements that need C declaration checks.
80/// Nested schemas precede their parents so generated row types can refer to them.
81pub(crate) fn schemas<'a>(
82    node: &'a prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
83    out: &mut Vec<(
84        &'a prebindgen_flat::Wire<CWire>,
85        &'a prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
86        &'a CSequence,
87    )>,
88) {
89    match node {
90        WireNode::Leaf(_) => {}
91        WireNode::Branch {
92            meta: CBranch::Group(_),
93            children: elements,
94            ..
95        }
96        | WireNode::Branch {
97            meta: CBranch::Alternatives,
98            children: elements,
99            ..
100        } => {
101            for child in elements {
102                schemas(child, out);
103            }
104        }
105        WireNode::Chain {
106            meta: CChain::Optional(_),
107            child: element,
108            ..
109        } => schemas(element, out),
110        WireNode::Chain {
111            controls,
112            child: element,
113            meta: CChain::Sequence(metadata),
114            ..
115        } => {
116            schemas(element, out);
117            if let [carrier] = controls.as_slice() {
118                out.push((carrier, element, metadata));
119            }
120        }
121    }
122}
123#[allow(clippy::type_complexity, clippy::too_many_arguments)]
124pub(crate) fn emit_schemas(
125    schemas: Vec<(
126        &prebindgen_flat::Wire<CWire>,
127        &prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
128        &CSequence,
129    )>,
130    output_carriers: &std::collections::BTreeSet<String>,
131    context: &CContext,
132    prefix: &str,
133    pointers: &std::collections::BTreeMap<String, prebindgen_flat::TypeInfo<CType>>,
134    symbols: &mut std::collections::BTreeSet<String>,
135    tokens: &mut TokenStream,
136    header: &mut String,
137) -> Result<()> {
138    for (carrier, element, metadata) in schemas {
139        if *metadata != CSequence::Buffered {
140            continue;
141        }
142        let row = &context.row_types[&carrier.name.to_string()];
143        let base = row.to_string();
144        let base = base.trim_start_matches("__PbRow_");
145        let c_row = format!("{prefix}_{base}_row");
146        let c_rows = format!("{prefix}_{base}_rows");
147        let free_rows = format_ident!("{prefix}_{base}_rows_free");
148        let is_output = output_carriers.contains(&carrier.name.to_string());
149        let mut row_symbols = vec![c_row.clone(), c_rows.clone()];
150        if is_output {
151            row_symbols.push(free_rows.to_string());
152        }
153        for symbol in row_symbols {
154            if !symbols.insert(symbol) {
155                return Err("duplicate C collection symbol".into());
156            }
157        }
158        let fields = crate::group::row_fields(element)
159            .enumerate()
160            .map(|(i, w)| (format_ident!("v{i}"), w.rust_type()));
161        let record = prebindgen_reprc::Record::new(row.clone(), fields)?;
162        let mut definition = record.definition();
163        definition
164            .attrs
165            .push(syn::parse_quote!(#[derive(Clone,Copy)]));
166        tokens.extend(quote!(#definition));
167        header.push_str("typedef struct {\n");
168        for (i, w) in crate::group::row_fields(element).enumerate() {
169            header.push_str(&format!(
170                "    {} v{i};\n",
171                c_ty(w, prefix, pointers, context)
172            ));
173        }
174        header.push_str(&format!(
175            "}} {c_row};\ntypedef struct {{ {c_row} *data; size_t len; }} {c_rows};\n"
176        ));
177        if is_output {
178            header.push_str(&format!("/* Release only the row allocation. Free or transfer owned element fields separately. Only pass buffers returned by this library. */\nvoid {free_rows}({c_rows} rows);\n"));
179            tokens.extend(quote! {
180                    #[doc="# Safety\nThe rows must be an unchanged allocation returned with this exact element layout. This frees only row storage, not element-owned fields."]
181                    #[no_mangle]pub unsafe extern "C" fn #free_rows(rows:__PbRows){if rows.len!=0{drop(Box::from_raw(std::ptr::slice_from_raw_parts_mut(rows.data as *mut #row,rows.len)));}}
182                });
183        }
184    }
185    Ok(())
186}

188/// Reclaim a committed output that will not reach its caller, including nested rows.
189pub(crate) fn release(
190    layout: &prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
191    fields: &[(&Wire, TokenStream)],
192    context: &CContext,
193    catch_panics: bool,
194) -> TokenStream {
195    let mut schemas = Vec::new();
196    self::schemas(layout, &mut schemas);
197    let mut code = TokenStream::new();
198    for (wire, value) in fields {
199        match &wire.kind {
200            CWire::Buffer => code.extend(quote! {if !(#value).data.is_null(){drop(Box::from_raw(std::ptr::slice_from_raw_parts_mut((#value).data as *mut u8,(#value).len)));}}),
201            CWire::Pointer { pointer, .. } if pointer.access == Access::Move => {
202                let ty = &pointer.ty;
203                let drop = quote!(drop(Box::from_raw((#value) as *mut #ty)););
204                let drop = if catch_panics {
205                    quote!(let _ = ::prebindgen_runtime::boundary(|| { #drop Ok(()) });)
206                } else { drop };
207                code.extend(quote! { if !(#value).is_null() { #drop } });
208            }
209            CWire::Owned {access:Access::Move,by_value:true,..} => {
210                let drop=quote!(drop((#value).into_rust()););
211                let drop=if catch_panics {quote!(let _=::prebindgen_runtime::boundary(||{#drop Ok(())});)} else {drop};
212                code.extend(drop);
213            }
214            CWire::Rows => {
215                let row=&context.row_types[&wire.name.to_string()];
216                let element=schemas.iter().find(|(carrier,_,_)|carrier.name==wire.name).unwrap().1;
217                let fields=crate::group::row_fields(element).enumerate().map(|(i,wire)|{let field=format_ident!("v{i}");(wire,quote!(row.#field))}).collect::<Vec<_>>();
218                let child=release(element,&fields,context,catch_panics);
219                code.extend(quote! {if (#value).len != 0 {let rows=Box::from_raw(std::ptr::slice_from_raw_parts_mut((#value).data as *mut #row,(#value).len));for row in rows.iter(){#child}}});
220            }
221            _=>{}
222        }
223    }
224    code
225}

227pub(crate) fn callback_output(
228    wire: syn::Ident,
229    values: &[(syn::Ident, TokenStream)],
230    element: &prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
231    length: syn::Ident,
232) -> Result<SequenceOutput<CWire, crate::CChain>> {
233    let values = prebindgen_conversions::match_output_fields(
234        crate::group::callback_parameters(element),
235        values,
236    )?;
237    let names = (0..values.len())
238        .map(|i| format_ident!("__pb_callback_value_{i}"))
239        .collect::<Vec<_>>();
240    let prepare = names
241        .iter()
242        .zip(values)
243        .map(|(name, (_, value))| quote!(let #name=#value;));
244    let commit = prebindgen_conversions::template_parameter("commit_element");
245    Ok(SequenceOutput {
246        controls: vec![prebindgen_flat::Wire {
247            name: wire.clone(),
248            kind: CWire::scalar(Primitive::Usize),
249        }],
250        commit_each: true,
251        fallible: true, // A callback pointer must be provided; checked before source invocation.
252        metadata: CChain::Sequence(CSequence::Callback),
253        begin: quote!(),
254        enter: quote!(),
255        store: quote!(#(#prepare)* #commit unsafe { __pb_callback.expect("validated callback")(__pb_callback_context, #(#names),*); }),
256        finish: quote!(),
257        values: vec![(wire.clone(), quote!(#length))],
258        commit: quote!(),
259    })
260}

262/// A visitor may replace exactly one collection, optionally with its presence flag.
263pub(crate) fn callback_shape(
264    node: &prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
265) -> bool {
266    match node {
267        WireNode::Chain {
268            meta: CChain::Sequence(metadata),
269            ..
270        } => *metadata == CSequence::Callback,
271        WireNode::Chain {
272            child: element,
273            meta: CChain::Optional(metadata),
274            ..
275        } => matches!(metadata, COptional::Flag(_)) && callback_shape(element),
276        WireNode::Branch {
277            meta: CBranch::Group(_),
278            children: elements,
279            ..
280        } => {
281            matches!(elements.as_slice(),[only] if callback_shape(only))
282        }
283        _ => false,
284    }
285}

287/// The adapter selects its native scope before looking for a sequence.
288#[allow(clippy::type_complexity)]
289pub(crate) fn return_sequence(
290    layout: &prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
291) -> Option<(
292    &prebindgen_flat::Wire<CWire>,
293    &prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
294    &CSequence,
295)> {
296    outer_sequences(layout)
297        .into_iter()
298        .next()
299        .and_then(|(element, controls, metadata)| match controls {
300            [carrier] => Some((carrier, element, metadata)),
301            _ => None,
302        })
303}
304pub(crate) fn callback_target(
305    layout: &prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
306) -> Result<syn::Ident> {
307    if outer_sequences(layout).len() != 1 || crate::group::result_packet_fields(layout).len() > 2 {
308        return Err(
309            "callback_return permits only one sequence and its optional presence flag; cannot discard additional return values".into(),
310        );
311    }
312    return_sequence(layout)
313        .map(|(carrier, _, _)| carrier.name.clone())
314        .ok_or_else(|| "callback_return requires a sequence boundary".into())
315}

317/// Whether the successful return uses the function's sequence callback.
318pub(crate) fn callback_delivery(
319    node: &prebindgen_flat::WireNode<CWire, CChain, CBranch>,
320) -> Result<bool> {
321    let sequences = outer_sequences(node);
322    if !sequences
323        .iter()
324        .any(|(_, _, metadata)| **metadata == CSequence::Callback)
325    {
326        return Ok(false);
327    }
328    callback_target(node)?;
329    if !callback_shape(node) {
330        return Err("callback sequence cannot be combined with additional return values".into());
331    }
332    Ok(true)
333}

335/// Function sequence callbacks are not available in parameters, error channels or rows.
336pub(crate) fn validate_buffered(
337    node: &prebindgen_flat::WireNode<CWire, CChain, CBranch>,
338) -> Result<()> {
339    match node {
340        prebindgen_flat::WireNode::Leaf(_) => {}
341        prebindgen_flat::WireNode::Chain {
342            child: element,
343            meta: CChain::Sequence(metadata),
344            ..
345        } => {
346            if *metadata == CSequence::Callback {
347                return Err(
348                    "callback sequence is only supported as the function's successful return"
349                        .into(),
350                );
351            }
352            validate_buffered(element)?;
353        }
354        prebindgen_flat::WireNode::Chain {
355            meta: CChain::Optional(_),
356            child: element,
357            ..
358        } => validate_buffered(element)?,
359        prebindgen_flat::WireNode::Branch {
360            meta: CBranch::Group(_),
361            children: elements,
362            ..
363        }
364        | prebindgen_flat::WireNode::Branch {
365            meta: CBranch::Alternatives,
366            children: elements,
367            ..
368        } => {
369            for element in elements {
370                validate_buffered(element)?;
371            }
372        }
373    }
374    Ok(())
375}

377// A sequence owns a separate row scope: find the outer occurrences only.
378#[allow(clippy::type_complexity)] // Borrow the element schema, controls and delivery policy together.
379fn outer_sequences(
380    node: &prebindgen_flat::WireNode<CWire, CChain, CBranch>,
381) -> Vec<(
382    &prebindgen_flat::WireNode<CWire, CChain, CBranch>,
383    &[prebindgen_flat::Wire<CWire>],
384    &CSequence,
385)> {
386    fn visit<'a>(
387        node: &'a prebindgen_flat::WireNode<CWire, CChain, CBranch>,
388        out: &mut Vec<(
389            &'a prebindgen_flat::WireNode<CWire, CChain, CBranch>,
390            &'a [prebindgen_flat::Wire<CWire>],
391            &'a CSequence,
392        )>,
393    ) {
394        use prebindgen_flat::WireNode;
395        match node {
396            WireNode::Chain {
397                meta: CChain::Sequence(meta),
398                child,
399                controls,
400            } => out.push((child, controls, meta)),
401            WireNode::Chain { child, .. } => visit(child, out),
402            WireNode::Branch { children, .. } => {
403                for child in children {
404                    visit(child, out)
405                }
406            }
407            WireNode::Leaf(_) => {}
408        }
409    }
410    let mut out = vec![];
411    visit(node, &mut out);
412    out
413}

415#[cfg(test)]
416mod delivery_tests {
417    use super::*;
418    fn leaf() -> prebindgen_flat::WireNode<CWire, CChain, CBranch> {
419        WireNode::Leaf(vec![prebindgen_flat::Wire {
420            name: syn::parse_quote!(element),
421            kind: CWire::scalar(Primitive::I32),
422        }])
423    }
424    fn sequence(
425        delivery: CSequence,
426        name: &str,
427        element: prebindgen_flat::WireNode<CWire, CChain, CBranch>,
428    ) -> prebindgen_flat::WireNode<CWire, CChain, CBranch> {
429        let kind = match delivery {
430            CSequence::Buffered => CWire::Rows,
431            CSequence::Callback => CWire::scalar(Primitive::Usize),
432        };
433        WireNode::Chain {
434            controls: vec![prebindgen_flat::Wire {
435                name: syn::parse_str(name).unwrap(),
436                kind,
437            }],
438            child: Box::new(element),
439            meta: CChain::Sequence(delivery),
440        }
441    }
442    #[test]
443    fn delivery_is_explicit_and_callback_scope_is_checked() {
444        let visitor = sequence(CSequence::Callback, "visitor", leaf());
445        assert!(callback_delivery(&visitor).unwrap());
446        assert!(validate_buffered(&visitor)
447            .unwrap_err()
448            .contains("successful return"));
449        let packed = sequence(CSequence::Buffered, "rows", leaf());
450        assert!(!callback_delivery(&packed).unwrap());
451        validate_buffered(&packed).unwrap();
452        let nested = sequence(CSequence::Buffered, "rows", visitor.clone());
453        assert!(crate::group::validate_layout(&nested)
454            .unwrap_err()
455            .contains("successful return"));
456        let extra = WireNode::Branch {
457            controls: vec![],
458            children: vec![visitor, leaf()],
459            meta: CBranch::Group(CGroup::Values),
460        };
461        assert!(callback_delivery(&extra)
462            .unwrap_err()
463            .contains("additional return"));
464    }
