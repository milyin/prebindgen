10/// A struct or one alternative of a sum: something with fields and
11/// delimiters.
12#[derive(Clone, Copy)]
13pub enum Record<'a> {
14    /// A source struct's fields and delimiters.
15    Struct(&'a Struct),
16    /// One sum alternative's fields and delimiters.
17    Alt(&'a Alternative),
18}
19
20impl<'a> Record<'a> {
21    /// Fields in source order. [`record_in`] and [`record_out`] preserve it.
22    pub fn fields(&self) -> &'a [Field] {
23        match self {
24            Record::Struct(s) => &s.fields,
25            Record::Alt(a) => &a.fields,
26        }
27    }
28
29    /// `head { a: v0, b: v1 }` / `head(v0, v1)` / `head`, with the
30    /// delimiters the source wrote.
31    pub fn construct(&self, head: &TokenStream, values: &[TokenStream]) -> TokenStream {
32        let parts: Vec<TokenStream> = self
33            .fields()
34            .iter()
35            .zip(values)
36            .map(|(f, v)| f.bind(v))
37            .collect();
38        self.shape(head, &parts)
39    }
40
41    /// The pattern binding each field to the matching ident in `binds`.
42    pub fn pattern(&self, head: &TokenStream, binds: &[syn::Ident]) -> TokenStream {
43        self.construct(
44            head,
45            &binds
46                .iter()
47                .map(|b| b.to_token_stream())
48                .collect::<Vec<_>>(),
49        )
50    }
51
52    /// Fresh binding names for the fields: `__f0`, `__f1`, …
53    pub fn binds(&self) -> Vec<syn::Ident> {
54        (0..self.fields().len())
55            .map(|i| format_ident!("__f{}", i))
56            .collect()
57    }
58
59    fn shape(&self, head: &TokenStream, parts: &[TokenStream]) -> TokenStream {
60        let emit = Emit::new();
61        match self {
62            Record::Struct(s) => emit.shape(*s, head.clone(), parts),
63            Record::Alt(a) => emit.shape(*a, head.clone(), parts),
64        }
65    }
66}
67
68/// A record rebuilt from its fields' inputs: wires concatenated in field
69/// order, values placed into the record's constructor `head`.
70pub fn record_in(record: Record<'_>, head: &TokenStream, fields: Vec<Input>) -> Input {
71    Input::combine(fields, |values| record.construct(head, &values))
72}
73
74/// A record taken apart into its fields' outputs. `src` is the record
75/// value; `field` builds each field's output from the binding holding it.
76pub fn record_out<E>(
77    record: Record<'_>,
78    head: &TokenStream,
79    src: &TokenStream,
80    mut field: impl FnMut(&Field, &TokenStream) -> Result<Output, E>,
81) -> Result<Output, E> {
82    let binds = record.binds();
83    let mut parts = Vec::new();
84    for (f, b) in record.fields().iter().zip(&binds) {
85        parts.push(field(f, &b.to_token_stream())?);
86    }
87    let pat = record.pattern(head, &binds);
88    Ok(Output::concat(quote!(let #pat = #src;), parts))
89}
