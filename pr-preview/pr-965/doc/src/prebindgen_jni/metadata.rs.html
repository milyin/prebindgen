6/// Borrowed roots at function and callback boundaries. Each root keeps its direction.
7#[derive(Clone, Copy)]
8pub(crate) enum BoundaryNode<'a> {
9    Input(&'a JniInputWireNode),
10    Output(&'a JniOutputWireNode),
11}
12impl<'a> BoundaryNode<'a> {
13    pub(crate) fn wires(self) -> Vec<&'a prebindgen_conversions::Wire<JniWire>> {
14        match self {
15            Self::Input(node) => {
16                prebindgen_conversions::wire_definitions(std::iter::once(node)).collect()
17            }
18            Self::Output(node) => {
19                prebindgen_conversions::wire_definitions(std::iter::once(node)).collect()
20            }
21        }
22    }
23}

25/// Prepared boundary roots, including typed error converters and declarations
26/// whose callback interfaces are emitted without a direct exported occurrence.
27pub(crate) fn nodes<I: KotlinInputConversion, O: KotlinOutputConversion>(
28    flat: &Flat<JniSchema<I, O>>,
29) -> Result<Vec<BoundaryNode<'_>>> {
30    let mut nodes = Vec::new();
31    for source in flat.functions.values() {
32        for binding in &source.metadata {
33            let call = binding.call()?;
34            for plan in &call.inputs {
35                nodes.push(BoundaryNode::Input(plan.input_node()));
36            }
37            let outcomes = call.output.output_outcomes();
38            nodes.extend(
39                std::iter::once(&outcomes.success)
40                    .chain(outcomes.source_error.iter())
41                    .chain(outcomes.binding_failure.iter())
42                    .map(BoundaryNode::Output),
43            );
44        }
45    }
46    for input in crate::callback::declarations(flat)? {
47        nodes.push(BoundaryNode::Input(input.input_node()));
48    }
49    Ok(nodes)
50}

52/// Metadata schema for a Kotlin/JNI binding model.
53/// `InConverter` and `OutConverter` select the converters stored in each export.
54/// Source parameter/return attachments are unit; type metadata keeps the JNI
55/// representation. Default containers also accept custom converters through their
56/// owned containers. No schema value is needed to construct Flat.
57pub struct JniSchema<InConverter = JniInputConverter, OutConverter = JniOutputConverter>(
58    std::marker::PhantomData<fn() -> (InConverter, OutConverter)>,
59);
60impl<InConverter, OutConverter> prebindgen_flat::Schema for JniSchema<InConverter, OutConverter> {
61    type Parameter = ();
62    type Return = ();
63    type Module = JniModule;
64    type Function = Vec<JniBinding<InConverter, OutConverter>>;
65    type Type = JniType;
66    type Field = data::JniField;
67    type Variant = ();
68    type Record = ();
69    type Enum = ();
70    type PayloadEnum = ();
71}

94/// Kotlin/JNI declaration information shared by occurrences of a Rust type.

131// Borrow the actual Flat type table, independent of its export converter types.
132pub(crate) type JniTypes = BTreeMap<String, prebindgen_flat::TypeInfo<JniType, data::JniField>>;

134pub(crate) fn handle_classes(types: &JniTypes) -> impl Iterator<Item = (&syn::Type, &Class)> {
135    let mut classes = types
136        .values()
137        .filter_map(|source| {
138            source
139                .metadata
140                .class
141                .as_deref()
142                .filter(|class| matches!(class.representation, ClassRepresentation::Handle { .. }))
143                .map(|class| (&source.ty, class))
144        })
145        .collect::<Vec<_>>();
146    // Preserve emitted declaration order across captured and external types.
147    classes.sort_by_key(|(ty, _)| type_key(ty));
148    classes.into_iter()
149}

151pub(crate) fn source_handle<'a>(
152    types: &'a JniTypes,
153    ty: &syn::Type,
154) -> Result<(&'a syn::Type, &'a Class)> {
155    let source = types
156        .values()
157        .find(|entry| prebindgen_flat::same_type(&entry.ty, ty))
158        .ok_or_else(|| format!("unregistered JNI source type `{}`", type_key(ty)))?;
159    let class = source
160        .metadata
161        .class
162        .as_deref()
163        .filter(|class| matches!(class.representation, ClassRepresentation::Handle { .. }))
164        .ok_or_else(|| format!("no handle class for `{}`", type_key(ty)))?;
165    Ok((&source.ty, class))
166}

168pub(crate) fn source_type<'a>(
169    types: &'a JniTypes,
170    reference: &prebindgen_flat::TypeRef,
171) -> Result<&'a prebindgen_flat::TypeInfo<JniType, data::JniField>> {
172    types
173        .get(reference.key())
174        .ok_or_else(|| format!("missing JNI source type entry `{}`", reference.key()))
175}
176pub(crate) fn source_data_class<'a>(
177    types: &'a JniTypes,
178    reference: &prebindgen_flat::TypeRef,
179) -> Result<(&'a prebindgen_flat::Record<data::JniField>, &'a Class)> {
180    let source = source_type(types, reference)?;
181    let class = source
182        .metadata
183        .class
184        .as_deref()
185        .filter(|class| matches!(class.representation, ClassRepresentation::Data { .. }))
186        .ok_or("record conversion requires a declared data class")?;
187    match &source.definition {
188        prebindgen_flat::TypeDefinition::Record(record) => Ok((record, class)),
189        _ => Err(format!("`{}` is not a source record", reference.key())),
190    }
191}
192pub(crate) fn source_enum<'a>(
193    types: &'a JniTypes,
194    reference: &prebindgen_flat::TypeRef,
195) -> Result<&'a prebindgen_flat::PayloadEnum<(), data::JniField>> {
196    match &source_type(types, reference)?.definition {
197        prebindgen_flat::TypeDefinition::PayloadEnum(enumeration) => Ok(enumeration),
198        _ => Err(format!(
199            "`{}` is not a source payload enum",
200            reference.key()
201        )),
202    }
