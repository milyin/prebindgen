12// `slice_or_vec_elem` lived here: it matched `&[T]` / `Vec<T>` off the SPELLING
13// and returned the element. `vec_build_elem` was its only caller and now reads
14// the same two facts off the model (`sequence_elem`, through `borrow_target`),
15// where a `Box<Vec<T>>` answers as `Vec<T>` does instead of failing a
16// last-path-segment test. Its one non-structural rule — `&mut [T]` is refused,
17// because mutate-back semantics keep the `input_vec` path — survives as the
18// `mutable: false` guard on that match.
19
20/// What a slice/`Vec` parameter contributes to the Vec-build path: the
21/// **canonical** element, whether the run is borrowed, and the element wrappers
22/// the storage does not carry.
23///
24/// A named return rather than a tuple because the third field is only meaningful
25/// against the first: `elem` is the element with every transparent wrapper taken
26/// off, and `elem_wrappers` is what has to go back on. Reading `(a, b, c)` at
27/// three call sites would not say that.
28pub(crate) struct VecBuildElem {
29    /// The element with its transparent wrappers removed — what the helper trio
30    /// stores, what its name is derived from, and what the flatten plan is built
31    /// for. `Vec<Payload>` and `Vec<Box<Payload>>` answer with the same node,
32    /// which is the whole mechanism: one trio per Kotlin class.
33    pub elem: TypeRef,
34    /// The run is a `&[T]`/`&Vec<T>` borrow rather than an owned value.
35    pub by_ref: bool,
36    /// The wrappers taken off the element, outermost first — empty for the
37    /// ordinary case. Applied per element where the Vec is consumed, since the
38    /// storage holds the canonical type. A list and not a `TypeRef`: it is the
39    /// only part a rebuild uses, and it rides in [`KotlinParamOp::VecBuild`], whose
40    /// size every variant pays.
41    pub elem_wrappers: Vec<&'static str>,
42    /// The canonical helper ABI shared by every function using this element
43    /// and by both the Rust and Kotlin synthetic-helper writers.
44    pub helpers: std::rc::Rc<VecBuildHelpers>,
45}

47/// `Some(..)` when `arg_ty` is a slice/`Vec` input whose element is a
48/// **flattenable `data_class`** — i.e. it decomposes into the conservative leaf
49/// set [`build_flat_input_plan`] accepts, so each element can cross as decoupled
50/// raw params and be rebuilt on the Rust side with no `env.get_field(...)`.
51/// `None` for any other shape (opaque handles, enums, nested-`Option` structs),
52/// which keep the `input_vec` path.
53///
54/// This is the single detection seam shared by `emit_input_param`, the param
55/// classifier, `render_extern_decl`, and the synthetic-extern emitter so all
56/// four sites agree on which params take the handle path.
57pub(crate) fn vec_build_elem(
58    ext: &Declarations,
59    registry: &(impl prebindgen_registry::Conversions + ?Sized),
60    arg: &TypeRef,
61) -> Option<VecBuildElem> {
62    // The run and its element off the MODEL. `&mut [T]` is still refused —
63    // mutate-back semantics keep the `input_vec` path — and that is the one
64    // fact the layer accessors do not carry, so the borrow's mutability is read
65    // off the kind directly.
66    //
67    // The conversion follows the SYNTAX, and must: this path builds a Rust-side
68    // `Vec<T>` and hands the source fn a borrow of it (or `mem::take`s it), so
69    // the referent has to be a form that built value satisfies. `&[T]`
70    // deref-coerces and `Vec<T>` is the thing itself; `Box<Vec<T>>` and
71    // `Cow<'_, [T]>` classify identically and cannot be rebuilt from the local.
72    // `decoded_vec_satisfies` in `selector.rs` is the same rule guarding the
73    // general converter path — asked here of the model, which holds both halves.
74    //
75    // BEFORE the peel as well as after, and the order is the whole point: the
76    // erasure happens OUTSIDE the layer it wraps, so `Box<&Vec<T>>` classifies
77    // as `Ref` and interpreting `kind` first would replace `arg` with the inner
78    // sequence — whose spelling is a clean `Vec<T>` — and let the outer `Box`
79    // through unseen. Every layer is checked on the way down, the way
80    // `rebuildable_target` does it.
81    let (run, by_ref) = match arg.unwrapped().kind() {
82        flat::TypeKind::Ref {
83            mutable: false,
84            inner,
85            ..
86        } => (&**inner, true),
87        _ => (arg, false),
88    };
89    // A wrapper over the RUN is buildable only on the by-value path, and the
90    // reason is a cost rather than a type error. By value the local is owned, so
91    // `Box<Vec<T>>` is `Box::new(mem::take(..))` — free. Borrowed, the local is
92    // a borrow of the Vec the Kotlin side owns, and there is no way to put a
93    // `Box` between that borrow and the callee without **copying** the run
94    // (`&Box::new(v.clone())`) — which needs a `T: Clone` nothing here
95    // guarantees, and silently adds a per-call copy to a path whose entire point
96    // is not having one.
97    //
98    // Definitive, not deferred: the borrowed shape keeps the `input_vec` path,
99    // which is correct. If a binding ever wants the copy, it is a decision to
100    // make on purpose, at the declaration.
101    let wrapped_run = !arg.erased_wrappers().is_empty() || !run.erased_wrappers().is_empty();
102    if wrapped_run {
103        if by_ref {
104            return None;
105        }
106        // By value: both layers must be buildable (`Cow` still declines).
107        build_through_erased_wrappers(arg, quote!(__probe))?;
108        build_through_erased_wrappers(run, quote!(__probe))?;
109    }
110    let elem = run.sequence_elem()?;
111    // A wrapped ELEMENT is served here. #294 refused it and called the refusal
112    // definitive; #296 re-read it as reserved and this closes it.
113    //
114    // The trio stores a `Vec<#elem>` and its base name comes from the element's
115    // **Kotlin class** (`Payload` → `payloadVec`), so keying it on the SPELLING
116    // would make `Vec<Payload>` and `Vec<Box<Payload>>` two storages wanting one
117    // name. Keying it on the CANONICAL element removes the question: one trio
118    // per Kotlin class, storage `Vec<Payload>`, and the element's wrapper goes
119    // back on where the Vec is CONSUMED — `.into_iter().map(Box::new).collect()`
120    // in `emit_input_param`. No Rust wrapper reaches a JNI symbol, which is the
121    // representation leak this layer exists to prevent, and nothing collides.
122    //
123    // What that buys: a `Box` the model erases no longer downgrades the crossing
124    // from raw scalar leaves to a per-element `JObject` plus a field read per
125    // field. Correctness was never the issue — the general converter serves the
126    // shape (see the transparent-wrapper plan) — the cost was, and it was silent.
127    let elem_wrappers = elem.erased_wrappers();
128    if !elem_wrappers.is_empty() {
129        // **Definitive on the BORROWED path.** By value the local is owned, so
130        // the per-element wrap is one O(n) `Box::new` over a Vec already being
131        // moved. Borrowed, the local is a borrow of the Vec the Kotlin side
132        // owns, and a `Vec<Box<T>>` is not a `Vec<T>` viewed differently — it is
133        // a different allocation. Serving it would mean `mem::take`ing that Vec
134        // and borrowing a fresh one, which turns the single arm whose whole
135        // point is not copying into one that consumes its input.
136        //
137        // So the shapes coincide with the wrapped-run refusal above and the
138        // reasons do not: that one cannot copy without a `T: Clone`, this one
139        // could copy and must not.
140        if by_ref {
141            return None;
142        }
143        // `Cow` still declines, and by its own policy rather than this path's:
144        // `Cow::Owned` is well-typed, and refusing it is what keeps a binding
145        // from silently removing the borrow path. See its `WRAPPER_OPS` recipe.
146        build_through_wrappers(&elem_wrappers, quote!(__probe))?;
147    }
148    // Canonical from here down: the flatten plan, the trio's storage and its
149    // name are all the element's, and the wrapper is the consumption site's
150    // business. `unwrapped` and not `stripped_syntax` because this stays a model
151    // node — it still has to answer `key`, `kind` and its own spelling.
152    let elem = elem.unwrapped();
153    let helpers = ext.vec_build_helpers(registry, elem)?;
154    Some(VecBuildElem {
155        elem: elem.clone(),
156        by_ref,
157        elem_wrappers,
158        helpers,
159    })
160}

162/// Every distinct flattenable element type `T` that a scanned, declared function
163/// takes as a `&[T]`/`Vec<T>` input — the set the synthetic `…VecNew/Push/Free`
164/// externs are emitted for (once per type, shared across all such functions).
165/// Deduped by [`TypeKey`] and sorted for deterministic output (mirrors
166/// [`crate::jni::plan_handle_destructors`]).
167///
168/// The key is the **canonical** element's, since that is what
169/// [`vec_build_elem`] answers with — so `Vec<Payload>` and `Vec<Box<Payload>>`
170/// land on one entry and get one trio, rather than two storages contending for
171/// the name `payloadVec` (#296).
172pub(crate) fn collect_vec_build_helpers(ext: &Declarations) -> Vec<std::rc::Rc<VecBuildHelpers>> {
173    ext.generation
174        .as_deref()
175        .expect("Vec helper emission requires a frozen JNI generation plan")
176        .vec_builds()
177}

179/// One element type's `…VecNew/Push/Free` helper trio: the flatten plan whose
180/// leaves are the per-element push params, plus the camelCase base name
181/// (`payloadVec`) the Kotlin methods and Rust JNI symbols share.
182pub(crate) struct VecBuildHelpers {
183    /// Canonical element reading stored by the Rust helper trio.
184    pub elem: TypeRef,
185    /// camelCase base, e.g. `"payloadVec"` (Kotlin method = `<base>New/Push/Free`).
186    pub base: String,
187    /// Element flatten plan (built with the synthetic param ident `e`).
188    pub plan: FlatInputPlan,
189}
190
191/// During resolution, build the helper descriptor for one flattenable element
192/// type, or `None` if it doesn't flatten (caller keeps the `input_vec` path).
193/// The artifact writers enumerate the frozen generation plan directly; they
194/// cannot call this constructor and resume planning. The base name is derived
195/// from the element's **Kotlin** data-class short name (first char lowercased)
196/// so the generated methods read naturally (`Payload` → `payloadVec`).
197impl Declarations {
198    pub(crate) fn vec_build_helpers(
199        &self,
200        registry: &(impl prebindgen_registry::Conversions + ?Sized),
201        elem: &TypeRef,
202    ) -> Option<std::rc::Rc<VecBuildHelpers>> {
203        assert!(
204            self.generation.is_none(),
205            "Vec helper construction is resolution-only"
206        );
207        let key = elem.key();
208        if let Some(hit) = self.vec_build_plans.borrow().get(&key).cloned() {
209            return Some(hit);
210        }
211        let plan = build_flat_input_plan(self, registry, &format_ident!("e"), elem)
212            .ok()
213            .flatten()?;
214        if plan.contains_composed_child
215            || plan.leaves.iter().any(|l| {
216                l.is_present_flag()
217                    || l.wire.handle_target.is_some()
218                    || l.entry().is_none()
219                    || l.wire.staged()
220            })
221        {
222            return None;
223        }
224        let kt_fqn = self
225            .types
226            .get(&key)
227            .and_then(|c| c.name_spec.as_ref())
228            .map(|s| self.fqn_of(s))?;
229        let short = kt_fqn.rsplit('.').next().unwrap_or(&kt_fqn);
230        let mut chars = short.chars();
231        let base_lc = match chars.next() {
232            Some(f) => format!("{}{}", f.to_lowercase(), chars.as_str()),
233            None => short.to_string(),
234        };
235        let helper = std::rc::Rc::new(VecBuildHelpers {
236            elem: elem.clone(),
237            base: format!("{base_lc}Vec"),
238            plan,
239        });
240        self.vec_build_plans
241            .borrow_mut()
242            .insert(key, helper.clone());
243        Some(helper)
244    }
245}

247/// Kotlin `external fun` short name for a vec helper (`payloadVecNew`), routed
248/// through the method mangler like every other `JNINative` extern. The Rust JNI symbol
249/// (see [`vec_helper_symbol`]) and the Kotlin call site both use this, so they
250/// agree.
251pub(crate) fn vec_helper_method_name(ext: &Declarations, base: &str, suffix: &str) -> String {
252    ext.mangle_jni_method(&format!("{base}{suffix}"))
253}

255/// Full Rust JNI symbol for a vec helper — the same spec-escaped
256/// `Java_<pkg>_<JNINative>_…` scheme function wrappers use via the plan's
257/// `native_symbol` (see `symbol`, #86); these helpers live on the
258/// `JNINative` object, so they share its class path.
259fn vec_helper_symbol(ext: &Declarations, base: &str, suffix: &str) -> String {
260    ext.native_method_symbol(&vec_helper_method_name(ext, base, suffix))
261}

263/// One `#[no_mangle] extern "C"` `…VecNew/Push/Free` trio per flattenable
264/// element type used as a `&[T]`/`Vec<T>` input — the Rust half of the
265/// build-the-Vec-incrementally path. Modeled on [`crate::jni::plan_handle_destructors`]
266/// (deterministic symbol sort, emitted only for element types a scanned function
267/// actually takes by slice/Vec).
268///
269/// `Push` is **infallible**: every leaf but a `String` is a primitive (the
270/// converter can't fail), and a `String?` passed straight from Kotlin always
271/// decodes. The only way a converter errs here is a JNI-internal fault (OOM /
272/// pending exception), which can't arise from a valid argument — so on the cold
273/// `Err` path it logs and skips the element rather than threading a per-caller
274/// error sink (the sink's typed `run` descriptor varies by caller, and `Push` is
275/// shared across all callers of a given element type). This keeps the Kotlin
276/// push loop free of a per-element failure check.
277pub(crate) fn plan_vec_build_helpers(ext: &Declarations) -> Vec<crate::jni::generation::JVecBuild> {
278    // Planned while the generation plan is being frozen, which is before that
279    // plan is installed on the declarations — so the helpers come from the
280    // planning store rather than through `collect_vec_build_helpers`, which
281    // every post-freeze reader uses. Order comes from the exported symbols
282    // below, so the store's own order does not matter here.
283    let helpers: Vec<_> = ext.vec_build_plans.borrow().values().cloned().collect();
284    let mut planned: Vec<crate::jni::generation::JVecBuild> = helpers
285        .into_iter()
286        .map(|helpers| {
287            let base = helpers.base.clone();
288            crate::jni::generation::JVecBuild::new(
289                helpers,
290                vec_helper_symbol(ext, &base, "New"),
291                vec_helper_symbol(ext, &base, "Push"),
292                vec_helper_symbol(ext, &base, "Free"),
293            )
294        })
295        .collect();
296    planned.sort_by(|a, b| a.sort_key().cmp(b.sort_key()));
297    planned
298}

300/// The trio's Rust items, in symbol order.
301pub(crate) fn render_vec_build_helpers(
302    plan: &crate::jni::generation::JVecBuild,
303    emit: &prebindgen_registry::RustWriter,
304) -> Vec<syn::Item> {
305    let mut named: Vec<(String, syn::Item)> = Vec::new();
306    {
307        let h = plan.helpers();
308        let (new_sym, push_sym, free_sym) = (
309            plan.new_symbol().to_string(),
310            plan.push_symbol().to_string(),
311            plan.free_symbol().to_string(),
312        );
313        // Generated Rust spells `spell()`; the reading is what the plan
314        // and the key are taken from.
315        let elem = emit.emit_source_type(&h.elem);
316        let new_id = syn::Ident::new(&new_sym, Span::call_site());
317        let push_id = syn::Ident::new(&push_sym, Span::call_site());
318        let free_id = syn::Ident::new(&free_sym, Span::call_site());
319
320        named.push((
321            new_sym.clone(),
322            syn::parse_quote!(
323                #[no_mangle]
324                #[allow(non_snake_case, unused_variables)]
325                pub(crate) unsafe extern "C" fn #new_id(
326                    _env: jni::JNIEnv,
327                    _class: jni::objects::JClass,
328                    cap: jni::sys::jint,
329                ) -> jni::sys::jlong {
330                    let __cap = if cap > 0 { cap as usize } else { 0usize };
331                    Box::into_raw(Box::new(Vec::<#elem>::with_capacity(__cap))) as jni::sys::jlong
332                }
333            ),
334        ));
335
336        let leaf_params: Vec<TokenStream> = h
337            .plan
338            .leaves
339            .iter()
340            .filter(|l| !l.is_present_flag())
341            .map(|l| {
342                let id = &l.native_ident;
343                let ty = &l.native_wire_ty();
344                quote!(#id: #ty)
345            })
346            .collect();
347        let mut decodes: Vec<TokenStream> = Vec::new();
348        let mut inits: Vec<TokenStream> = Vec::new();
349        for l in h.plan.leaves.iter().filter(|l| !l.is_present_flag()) {
350            let conv = l.conv().expect("non-present leaf has a converter");
351            let conv = emit.operation_ident("jni", conv);
352            let wid = &l.native_ident;
353            let fid = format_ident!("{}", l.wire.field().expect("non-present leaf has a field"));
354            let tmp = format_ident!("__e_{}", fid);
355            decodes.push(quote!(
356                let #tmp = match #conv(&mut env, &#wid) {
357                    ::core::result::Result::Ok(__v) => __v,
358                    ::core::result::Result::Err(__e) => {
359                        tracing::error!("vecPush: decoding `{}`: {}", stringify!(#fid), __e);
360                        return;
361                    }
362                };
363            ));
364            inits.push(quote!(#fid: #tmp));
365        }
366        let module = &h.plan.struct_module;
367        let sid = &h.plan.struct_ident;
368
369        named.push((
370            push_sym.clone(),
371            syn::parse_quote!(
372                #[no_mangle]
373                #[allow(non_snake_case, unused_mut, unused_variables, dead_code)]
374                pub(crate) unsafe extern "C" fn #push_id<'a>(
375                    mut env: jni::JNIEnv<'a>,
376                    _class: jni::objects::JClass<'a>,
377                    handle: jni::sys::jlong,
378                    #(#leaf_params,)*
379                ) {
380                    if handle == 0 {
381                        return;
382                    }
383                    #(#decodes)*
384                    let __elem = #module::#sid { #(#inits),* };
385                    let __vec = &mut *(handle as *mut Vec<#elem>);
386                    __vec.push(__elem);
387                }
388            ),
389        ));
390
391        named.push((
392            free_sym.clone(),
393            syn::parse_quote!(
394                #[no_mangle]
395                #[allow(non_snake_case, unused_variables)]
396                pub(crate) unsafe extern "C" fn #free_id(
397                    _env: jni::JNIEnv,
398                    _class: jni::objects::JClass,
399                    handle: jni::sys::jlong,
400                ) {
401                    if handle != 0 {
402                        drop(Box::from_raw(handle as *mut Vec<#elem>));
403                    }
404                }
405            ),
406        ));
407    }
408    named.sort_by(|a, b| a.0.cmp(&b.0));
409    named.into_iter().map(|(_, item)| item).collect()
410}
