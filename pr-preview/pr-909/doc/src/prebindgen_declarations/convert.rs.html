11macro_rules! convert {($($ty:tt)+)=>{$crate::ConvertDecl::new($crate::declarations::parse_type(stringify!($($ty)+)))};}

25/// One function or direction-specific trait conversion source.
26#[derive(Clone, Debug)]
27pub struct ConvertSource {
28    source: ValueSource,
29    representation: Option<Type>,
30    input: Option<bool>,
31    result_value: bool,
32}
33impl ConvertSource {
34    /// Construct an input trait conversion (normally written with `from!`).
35    pub fn input(representation: Type, fallible: bool) -> Self {
36        Self {
37            source: ValueSource::Into { fallible },
38            representation: Some(representation),
39            input: Some(true),
40            result_value: false,
41        }
42    }
43    /// Construct an output trait conversion (normally written with `into!`).
44    pub fn output(representation: Type, fallible: bool) -> Self {
45        Self {
46            source: ValueSource::Into { fallible },
47            representation: Some(representation),
48            input: Some(false),
49            result_value: false,
50        }
51    }
52    pub(crate) fn resolve<S: Schema>(
53        &self,
54        flat: &Flat<S>,
55        ty: &Type,
56        input: bool,
57    ) -> Result<ValueConversion> {
58        if self.input.is_some_and(|i| i != input) {
59            return Err("conversion trait source has the wrong direction".into());
60        }
61        let representation = match &self.source {
62            ValueSource::Function(name) => {
63                let f = flat.function(name)?;
64                if f.params.len() != 1 {
65                    return Err(format!(
66                        "conversion `{name}` requires exactly one parameter"
67                    ));
68                }
69                let param = &f.params[0].ty.ty;
70                let param = match param {
71                    Type::Reference(r) if r.mutability.is_none() => &*r.elem,
72                    Type::Reference(_) => {
73                        return Err("conversion helpers cannot take mutable references".into())
74                    }
75                    _ => param,
76                };
77                let result = prebindgen_flat::result_parts(&f.ret.ty)
78                    .filter(|_| !(self.result_value || input && same_type(&f.ret.ty, ty)))
79                    .map_or(&f.ret.ty, |(ok, _)| ok);
80                if input {
81                    if !same_type(result, ty) {
82                        return Err(format!(
83                            "input conversion `{name}` must return `{}` or Result of that type",
84                            type_key(ty)
85                        ));
86                    }
87                    f.params[0].ty.ty.clone()
88                } else {
89                    if !same_type(param, ty) {
90                        return Err(format!(
91                            "output conversion `{name}` must take `{}` or its shared reference",
92                            type_key(ty)
93                        ));
94                    }
95                    result.clone()
96                }
97            }
98            ValueSource::Into { .. } => flat.resolve_type(self.representation.as_ref().unwrap())?,
99        };
100        if same_type(
101            match &representation {
102                Type::Reference(r) => &r.elem,
103                _ => &representation,
104            },
105            ty,
106        ) {
107            return Err(format!(
108                "conversion for `{}` must use a different representation type",
109                type_key(ty)
110            ));
111        }
112        Ok(ValueConversion {
113            representation,
114            source: self.source.clone(),
115        })
116    }
117}

130/// Canonical conversions for one source type. Directions are independent.
131#[derive(Clone, Debug)]
132pub struct ConvertDecl {
133    /// Source type whose representation is selected.
134    pub ty: Type,
135    pub(crate) input: Option<ConvertSource>,
136    pub(crate) output: Option<ConvertSource>,
137    duplicate: bool,
138}
139impl ConvertDecl {
140    /// Start a single-value conversion declaration for a source type.
141    pub fn new(ty: Type) -> Self {
142        Self {
143            ty,
144            input: None,
145            output: None,
146            duplicate: false,
147        }
148    }
149    /// Convert one representation into the source value.
150    pub fn input(mut self, source: impl Into<ConvertSource>) -> Self {
151        self.duplicate |= self.input.is_some();
152        self.input = Some(source.into());
153        self
154    }
155    /// Convert a source value into its representation.
156    pub fn output(mut self, source: impl Into<ConvertSource>) -> Self {
157        self.duplicate |= self.output.is_some();
158        self.output = Some(source.into());
159        self
160    }
161    pub(crate) fn validate(&self) -> Result<()> {
162        if self.duplicate {
163            return Err("duplicate conversion direction".into());
164        }
165        if self.input.is_none() && self.output.is_none() {
166            return Err("conversion declaration requires an input or output source".into());
167        }
168        Ok(())
169    }
170    /// Captured helper names needed when selecting a source stream.
171    pub fn functions(&self) -> Vec<String> {
172        [&self.input, &self.output]
173            .into_iter()
174            .flatten()
175            .filter_map(|s| match &s.source {
176                ValueSource::Function(f) => Some(f.clone()),
177                _ => None,
178            })
179            .collect()
180    }
181    /// Representation types explicitly named by trait conversions.
182    pub fn representation_types(&self) -> Vec<Type> {
183        [&self.input, &self.output]
184            .into_iter()
185            .flatten()
186            .filter_map(|s| s.representation.clone())
187            .collect()
188    }
189}

206/// Prepare canonical type conversions in the authoritative Flat type table.
207/// No slots are compiled here; declarations and independent codecs can use them.
208pub fn prepare_value_conversions<S: Schema>(
209    flat: &mut Flat<S>,
210    decls: Vec<ConvertDecl>,
211) -> Result<()> {
212    let mut selected = std::collections::BTreeMap::new();
213    for decl in decls {
214        decl.validate()?;
215        let ty = flat.resolve_type(&decl.ty)?;
216        if !matches!(ty, Type::Path(_))
217            || Primitive::from_type(&ty).is_some()
218            || prebindgen_flat::option_inner(&ty).is_some()
219            || prebindgen_flat::sequence_parts(&ty).is_some()
220            || matches!(type_key(&ty).as_str(), "String" | "str" | "()")
221            || flat.type_info(&ty).is_err()
222        {
223            return Err(
224                "canonical conversions require a named domain type, not a built-in representation"
225                    .into(),
226            );
227        }
228        let key = type_key(&ty);
229        let pair = selected.entry(key.clone()).or_insert_with(|| {
230            flat.type_info(&ty)
231                .map(|i| i.conversion.clone())
232                .unwrap_or_default()
233        });
234        for (input, source) in [(true, decl.input), (false, decl.output)] {
235            if let Some(source) = source {
236                let dest = if input {
237                    &mut pair.input
238                } else {
239                    &mut pair.output
240                };
241                if dest.is_some() {
242                    return Err(format!("duplicate conversion for `{key}`"));
243                }
244                *dest = Some(source.resolve(flat, &ty, input)?);
245            }
246        }
247    }
248    // Traverse the representation shape as well as its root. `A -> Vec<A>`
249    // is just as recursive as `A -> B -> A`, even when no function uses A.
250    fn check<S: Schema>(
251        flat: &Flat<S>,
252        selected: &std::collections::BTreeMap<String, prebindgen_flat::ValueConversions>,
253        ty: &Type,
254        input: bool,
255        active: &mut Vec<String>,
256    ) -> Result<()> {
257        let key = type_key(ty);
258        let pair = selected
259            .get(&key)
260            .cloned()
261            .or_else(|| flat.type_info(ty).ok().map(|i| i.conversion.clone()));
262        if let Some(conversion) = pair.and_then(|p| if input { p.input } else { p.output }) {
263            if active.contains(&key) {
264                return Err(format!("recursive canonical conversion `{key}`"));
265            }
266            active.push(key);
267            check(flat, selected, &conversion.representation, input, active)?;
268            active.pop();
269        } else if let Some(inner) = prebindgen_flat::option_inner(ty) {
270            check(flat, selected, inner, input, active)?;
271        } else if let Some((inner, _)) = prebindgen_flat::sequence_parts(ty) {
272            check(flat, selected, inner, input, active)?;
273        } else if !input {
274            if let Some(inner) = crate::output_referent(ty) {
275                check(flat, selected, inner, input, active)?;
276            }
277        }
278        Ok(())
279    }
280    for key in selected.keys() {
281        for input in [true, false] {
282            let ty = syn::parse_str(key).map_err(|e| format!("conversion type: {e}"))?;
283            check(flat, &selected, &ty, input, &mut Vec::new())?;
284        }
285    }
286    for (key, pair) in selected {
287        let ty = syn::parse_str(&key).map_err(|e| format!("conversion type: {e}"))?;
288        flat.type_info_mut(&ty)?.conversion = pair;
289    }
290    Ok(())
291}

295    #[allow(clippy::too_many_arguments)]
296    pub(crate) fn resolve_input_conversion<C: Codec>(
297        &self,
298        flat: &Flat<C::Schema>,
299        codec: &C,
300        ty: &Type,
301        conversion: ValueConversion,
302        child: &Input<InputRequest, OutputRequest>,
303        extra: &C::Extra,
304        stack: &mut Vec<String>,
305    ) -> Result<Input> {
306        let key = format!("value-input:{}", type_key(ty));
307        if stack.contains(&key) {
308            return Err(format!("recursive canonical conversion `{}`", type_key(ty)));
309        }
310        stack.push(key);
311        let input = self.input(flat, codec, &conversion.representation, child, extra, stack);
312        stack.pop();
313        Ok(Input::Unary {
314            operation: InputUnary::Convert {
315                conversion: Box::new(conversion),
316            },
317            child: Box::new(input?),
318        })
319    }
320    #[allow(clippy::too_many_arguments)]
321    pub(crate) fn resolve_output_conversion<C: Codec>(
322        &self,
323        flat: &Flat<C::Schema>,
324        codec: &C,
325        ty: &Type,
326        conversion: ValueConversion,
327        child: &Output<OutputRequest, InputRequest>,
328        extra: &C::Extra,
329        stack: &mut Vec<String>,
330    ) -> Result<Output> {
331        let key = format!("value-output:{}", type_key(ty));
332        if stack.contains(&key) {
333            return Err(format!("recursive canonical conversion `{}`", type_key(ty)));
334        }
335        stack.push(key);
336        let output = self.output(
337            flat,
338            codec,
339            &conversion.representation,
340            child,
341            extra,
342            stack,
343            true,
344        );
345        stack.pop();
346        Ok(Output::Unary {
347            operation: OutputUnary::Convert {
348                conversion: Box::new(conversion),
349            },
350            child: Box::new(output?),
351        })
352    }
353    pub(crate) fn resolve_input_value<C: Codec>(
354        &self,
355        flat: &Flat<C::Schema>,
356        codec: &C,
357        ty: &Type,
358        extra: &C::Extra,
359        stack: &mut Vec<String>,
360    ) -> Result<Input> {
361        if type_key(ty) == "()" {
362            return Ok(Input::Branch {
363                operation: InputBranch::Fields,
364                children: vec![],
365            });
366        }
367        if let Some(c) = flat
368            .type_info(ty)
369            .ok()
370            .and_then(|i| i.conversion.input.clone())
371        {
372            return self.resolve_input_conversion(
373                flat,
374                codec,
375                ty,
376                c,
377                &Input::value(),
378                extra,
379                stack,
380            );
381        }
382        if let Some(inner) = prebindgen_flat::option_inner(ty) {
383            return Ok(Input::Unary {
384                operation: InputUnary::Optional,
385                child: Box::new(self.resolve_input_value(flat, codec, inner, extra, stack)?),
386            });
387        }
388        if let Some((inner, _)) = prebindgen_flat::sequence_parts(ty)
389            .filter(|_| !codec.input_collection_is_leaf(flat, ty, self.input_role.get(), extra))
390        {
391            return Ok(Input::Unary {
392                operation: InputUnary::Sequence,
393                child: Box::new(self.resolve_input_value(flat, codec, inner, extra, stack)?),
394            });
395        }
396        self.input(
397            flat,
398            codec,
399            ty,
400            &Input::Leaf(InputLeafOp::Value),
401            extra,
402            stack,
403        )
404    }
405    #[allow(clippy::too_many_arguments)]
406    pub(crate) fn resolve_output_value<C: Codec>(
407        &self,
408        flat: &Flat<C::Schema>,
409        codec: &C,
410        ty: &Type,
411        extra: &C::Extra,
412        stack: &mut Vec<String>,
413        getter: bool,
414        policy: Option<&crate::Projection>,
415    ) -> Result<Output> {
416        if type_key(ty) == "()" {
417            return Ok(Output::Branch {
418                operation: OutputBranch::Fields,
419                children: vec![],
420            });
421        }
422        let c = flat
423            .type_info(ty)
424            .ok()
425            .and_then(|i| i.conversion.output.clone())
426            .or_else(|| {
427                crate::output_referent(ty)
428                    .and_then(|t| flat.type_info(t).ok())
429                    .and_then(|i| i.conversion.output.clone())
430            });
431        if let Some(c) = c {
432            if policy.is_some_and(|p| !p.fields.is_empty()) {
433                return Err(
434                    "field projections on a canonical value require an explicit output tree".into(),
435                );
436            }
437            if policy.is_some_and(|p| p.mode == crate::ProjectionMode::Borrowed)
438                && !conversion_borrows(flat, &c)?
439            {
440                return Err(
441                    "a borrowed projection requires a shared-reference conversion helper".into(),
442                );
443            }
444            return self.resolve_output_conversion(
445                flat,
446                codec,
447                ty,
448                c,
449                &Output::value(),
450                extra,
451                stack,
452            );
453        }
454        if let Some(inner) = prebindgen_flat::option_inner(ty) {
455            return Ok(Output::Unary {
456                operation: OutputUnary::Optional,
457                child: Box::new(
458                    self.resolve_output_value(flat, codec, inner, extra, stack, getter, policy)?,
459                ),
460            });
461        }
462        if let Some((inner, _)) = prebindgen_flat::sequence_parts(ty)
463            .filter(|_| !codec.output_collection_is_leaf(flat, ty, extra))
464        {
465            return Ok(Output::Unary {
466                operation: OutputUnary::Sequence {
467                    delivery: crate::SequenceDelivery::Value,
468                    mode: policy.map(|p| p.mode),
469                },
470                child: Box::new(
471                    self.resolve_output_value(flat, codec, inner, extra, stack, getter, policy)?,
472                ),
473            });
474        }
475        self.output(
476            flat,
477            codec,
478            ty,
479            &policy.map_or(Output::Leaf(OutputLeafOp::Value), |p| {
480                Output::Leaf(OutputLeafOp::Projected(p.clone()))
481            }),
482            extra,
483            stack,
484            getter,
485        )
486    }

489fn conversion_borrows<S: Schema>(flat: &Flat<S>, conversion: &ValueConversion) -> Result<bool> {
490    match &conversion.source {
491        ValueSource::Function(name) => Ok(flat
492            .function(name)?
493            .params
494            .first()
495            .is_some_and(|p| matches!(&p.ty.ty,Type::Reference(r) if r.mutability.is_none()))),
496        ValueSource::Into { .. } => Ok(false),
497    }
498}
499fn validate_conversion<S: Schema>(
500    flat: &Flat<S>,
501    ty: &Type,
502    conversion: &ValueConversion,
503    input: bool,
504) -> Result<()> {
505    let resolved = ConvertSource {
506        source: conversion.source.clone(),
507        representation: Some(conversion.representation.clone()),
508        input: None,
509        result_value: match &conversion.source {
510            ValueSource::Function(name) => {
511                let f = flat.function(name)?;
512                same_type(
513                    &f.ret.ty,
514                    if input {
515                        ty
516                    } else {
517                        &conversion.representation
518                    },
519                )
520            }
521            ValueSource::Into { .. } => false,
522        },
523    }
524    .resolve(flat, ty, input)?;
525    if !same_type(&resolved.representation, &conversion.representation) {
526        return Err("conversion representation does not match its helper signature".into());
527    }
528    Ok(())
529}

534impl<C: Codec> Compiler<'_, C> {
535    #[allow(clippy::too_many_arguments)]
536    pub(crate) fn input_conversion(
537        &self,
538        ty: &Type,
539        conversion: &ValueConversion,
540        input: &Input,
541        extra: &C::Extra,
542        active: &Condition,
543        names: Option<(&syn::Ident, &syn::Ident)>,
544    ) -> Result<CodecInput<C>> {
545        validate_conversion(self.flat, ty, conversion, true)?;
546        let target = names.map(|(_, v)| v.clone()).unwrap_or_else(|| self.name());
547        let repr = format_ident!("{target}_repr");
548        let decoded = format_ident!("{repr}_decoded");
549        let child_names = names.map(|(w, _)| (w, &decoded));
550        let mut child = self.input_named(
551            &conversion.representation,
552            input,
553            extra,
554            active,
555            child_names,
556        )?;
557        let (expr, fallible) = match &conversion.source {
558            ValueSource::Function(name) => {
559                let f = self.flat.function(name)?;
560                let path = &f.path;
561                (
562                    quote!(#path(#repr)),
563                    prebindgen_flat::result_parts(&f.ret.ty).is_some() && !same_type(&f.ret.ty, ty),
564                )
565            }
566            ValueSource::Into { fallible } => {
567                let r = &conversion.representation;
568                if *fallible {
569                    (
570                        quote!(<#r as ::core::convert::TryInto<#ty>>::try_into(#repr)),
571                        true,
572                    )
573                } else {
574                    (
575                        quote!(<#r as ::core::convert::Into<#ty>>::into(#repr)),
576                        false,
577                    )
578                }
579            }
580        };
581        let expr = if fallible {
582            quote!((#expr).map_err(|error|::std::string::ToString::to_string(&error))?)
583        } else {
584            expr
585        };
586        let rest: CodecInput<C> = prebindgen_conversions::InputPlan::new(
587            child.node.clone(),
588            quote!(let #target:#ty=#expr;),
589            quote!(#target),
590        );
591        let storage = std::mem::take(&mut child.storage);
592        let code = child.bind(&repr, rest.code)?;
593        Ok(prebindgen_conversions::InputPlan {
594            fallible: child.fallible || fallible,
595            node: child.node,
596            storage,
597            code,
598            context: Default::default(),
599        })
600    }
601    pub(crate) fn output_conversion(
602        &self,
603        ty: &Type,
604        conversion: &ValueConversion,
605        output: &Output,
606        extra: &C::Extra,
607        value: TokenStream,
608        borrowed: bool,
609    ) -> Result<CodecOutput<C>> {
610        validate_conversion(self.flat, ty, conversion, false)?;
611        let borrowed = borrowed || matches!(ty, Type::Reference(_));
612        let source = self.name();
613        let repr = self.name();
614        let (expr, fallible) = match &conversion.source {
615            ValueSource::Function(name) => {
616                let f = self.flat.function(name)?;
617                let path = &f.path;
618                let shared =
619                    matches!(&f.params[0].ty.ty,Type::Reference(r) if r.mutability.is_none());
620                if !shared && borrowed {
621                    return Err(format!("consuming conversion `{name}` cannot convert a borrowed source; use a shared-reference conversion"));
622                }
623                let argument = if shared && !borrowed {
624                    quote!(&#source)
625                } else {
626                    quote!(#source)
627                };
628                (
629                    quote!(#path(#argument)),
630                    prebindgen_flat::result_parts(&f.ret.ty).is_some()
631                        && !same_type(&f.ret.ty, &conversion.representation),
632                )
633            }
634            ValueSource::Into { fallible } => {
635                if borrowed {
636                    return Err("consuming trait conversion cannot convert a borrowed source; use a shared-reference helper".into());
637                }
638                let r = &conversion.representation;
639                if *fallible {
640                    (
641                        quote!(<#ty as ::core::convert::TryInto<#r>>::try_into(#source)),
642                        true,
643                    )
644                } else {
645                    (
646                        quote!(<#ty as ::core::convert::Into<#r>>::into(#source)),
647                        false,
648                    )
649                }
650            }
651        };
652        let expr = if fallible {
653            quote!((#expr).map_err(|error|::std::string::ToString::to_string(&error))?)
654        } else {
655            expr
656        };
657        let mut child = self.output_return(
658            &conversion.representation,
659            output,
660            extra,
661            quote!(#repr),
662            false,
663        )?;
664        self.codec.validate_getter_output(
665            self.flat,
666            ty,
667            borrowed && !self.temporary_receiver.get(),
668            &conversion.representation,
669            &child,
670            extra,
671        )?;
672        let prepared = child.statements;
673        child.statements = quote!(let #source=#value;let #repr=#expr;#prepared);
674        child.fallible |= fallible;
675        Ok(child)
676    }
