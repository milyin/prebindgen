31/// Conversion defaults indexed by canonical Rust type during preparation.
32/// Values are the same Input/Output trees accepted by per-function overrides.
33/// This environment is discarded after prepared conversions have been installed.
34pub struct Declarations {
35    pub(crate) input_role: std::cell::Cell<crate::InputRole>,
36    inputs: BTreeMap<String, Input<InputRequest, OutputRequest>>,
37    outputs: BTreeMap<String, Output<OutputRequest, InputRequest>>,
38    errors: BTreeMap<String, String>,
39}
40fn input_body(input: Input<InputRequest, OutputRequest>) -> Input<InputRequest, OutputRequest> {
41    match input {
42        Input::Branch {
43            operation: InputBranch::Selector,
44            children: mut arms,
45        } if arms.len() == 1 => arms.remove(0),
46        Input::Unary {
47            operation: InputUnary::Bulk,
48            child,
49        } => Input::Unary {
50            operation: InputUnary::Bulk,
51            child: Box::new(input_body(*child)),
52        },
53        Input::Unary {
54            operation: InputUnary::Temporary,
55            child,
56        } => Input::Unary {
57            operation: InputUnary::Temporary,
58            child: Box::new(input_body(*child)),
59        },
60        other => other,
61    }
62}
63fn output_body(output: Output<OutputRequest, InputRequest>) -> Output<OutputRequest, InputRequest> {
64    match output {
65        Output::Branch {
66            operation: OutputBranch::Combine,
67            children: mut parts,
68        } if parts.len() == 1 => parts.remove(0),
69        Output::Unary {
70            operation: OutputUnary::Bulk,
71            child,
72        } => Output::Unary {
73            operation: OutputUnary::Bulk,
74            child: Box::new(output_body(*child)),
75        },
76        other => other,
77    }
78}

80    /// Register typed Input/Output defaults and explicit error-message functions.
81    /// Typed trees use Input::for_type / Output::for_type. Production declaration
82    /// builders also convert directly into these trees with `.into()`.
83    pub fn new<M: prebindgen_flat::Schema>(
84        flat: &Flat<M>,
85        inputs: Vec<Input<InputRequest, OutputRequest>>,
86        outputs: Vec<Output<OutputRequest, InputRequest>>,
87        errors: Vec<(syn::Type, String)>,
88    ) -> Result<Self> {
89        let mut out = Self {
90            input_role: std::cell::Cell::new(crate::InputRole::Parameter),
91            inputs: BTreeMap::new(),
92            outputs: BTreeMap::new(),
93            errors: BTreeMap::new(),
94        };
95        for input in inputs {
96            let Input::Request(InputRequest::ForType { ty, input }) = input else {
97                return Err("input defaults require a source type".into());
98            };
99            let key = type_key(&crate::callback::input_type(flat, &ty)?);
100            if out.inputs.insert(key, input_body(*input)).is_some() {
101                return Err("duplicate input expansion".into());
102            }
103        }
104        for output in outputs {
105            let Output::Request(OutputRequest::ForType { ty, output }) = output else {
106                return Err("output defaults require a source type".into());
107            };
108            let key = type_key(&flat.resolve_type(&ty)?);
109            if out.outputs.insert(key, output_body(*output)).is_some() {
110                return Err("duplicate output expansion".into());
111            }
112        }
113        for (ty, function) in errors {
114            let key = type_key(&flat.resolve_type(&ty)?);
115            if out.errors.insert(key, function).is_some() {
116                return Err("duplicate error mapper for source type".into());
117            }
118        }
119        Ok(out)
120    }
121    /// Clone a source binding for a second explicit sequence delivery policy.
122    pub fn callback_variant<C: Codec>(
123        &self,
124        flat: &mut Flat<C::Schema>,
125        codec: &C,
126        decl: &FunctionDecl,
127        name: &str,
128        extra: &C::Extra,
129    ) -> Result<FunctionDecl>
130    where
131        <C::Schema as Schema>::Input: Clone,
132        <C::Schema as Schema>::Output: Clone,
133        <C::Schema as prebindgen_flat::Schema>::Function: Clone,
134    {
135        let source = flat.function(&decl.name)?;
136        let ty = result_parts(&source.ret.ty)
137            .filter(|_| !decl.result_value)
138            .map_or(&source.ret.ty, |(ok, _)| ok);
139        // Choose the delivery while expansion scopes are still intact. Bulk
140        // promotion for an ordinary return must not erase the element boundary
141        // before the callback variant selects that boundary.
142        let mut output = self.output(
143            flat,
144            codec,
145            ty,
146            decl.ret.as_ref().unwrap_or(&Output::auto()),
147            extra,
148            &mut vec![],
149            false,
150        )?;
151        fn select(output: &mut Output) -> usize {
152            match output {
153                Output::Unary {
154                    operation: OutputUnary::Sequence { delivery, .. },
155                    ..
156                } => {
157                    *delivery = SequenceDelivery::Callback;
158                    1
159                }
160                Output::Unary {
161                    operation: OutputUnary::Optional,
162                    child,
163                }
164                | Output::Unary {
165                    operation: OutputUnary::Bulk,
166                    child,
167                }
168                | Output::Unary {
169                    operation: OutputUnary::Named { .. },
170                    child,
171                    ..
172                }
173                | Output::Unary {
174                    operation: OutputUnary::Getter { .. },
175                    child,
176                    ..
177                }
178                | Output::Unary {
179                    operation: OutputUnary::Into { .. },
180                    child,
181                    ..
182                }
183                | Output::Unary {
184                    operation: OutputUnary::Convert { .. },
185                    child,
186                    ..
187                } => select(child),
188                Output::Branch {
189                    operation: OutputBranch::Combine,
190                    children,
191                }
192                | Output::Branch {
193                    operation: OutputBranch::Fields,
194                    children,
195                }
196                | Output::Branch {
197                    operation: OutputBranch::OwnedFields,
198                    children,
199                }
200                | Output::Branch {
201                    operation: OutputBranch::Record { .. },
202                    children,
203                    ..
204                } => children.iter_mut().map(select).sum(),
205                _ => 0,
206            }
207        }
208        if select(&mut output) != 1 {
209            fn structural(
210                ty: &syn::Type,
211                output: &Output,
212            ) -> Option<Output<OutputRequest, InputRequest>> {
213                match output {
214                    Output::Leaf(OutputLeafOp::Value)
215                        if prebindgen_flat::sequence_parts(ty).is_some() =>
216                    {
217                        Some(Output::Unary {
218                            operation: OutputUnary::Sequence {
219                                delivery: SequenceDelivery::Callback,
220                                mode: None,
221                            },
222                            child: Box::new(Output::auto()),
223                        })
224                    }
225                    Output::Unary {
226                        operation: OutputUnary::Optional,
227                        child,
228                    } => Some(Output::Unary {
229                        operation: OutputUnary::Optional,
230                        child: Box::new(structural(prebindgen_flat::option_inner(ty)?, child)?),
231                    }),
232                    _ => None,
233                }
234            }
235            if decl.ret.is_none() {
236                if let Some(tree) = structural(ty, &output) {
237                    output = self.output(flat, codec, ty, &tree, extra, &mut vec![], false)?;
238                } else {
239                    return Err("callback_return requires exactly one outer sequence".into());
240                }
241            } else {
242                return Err("callback_return requires exactly one outer sequence".into());
243            }
244        }
245        let mut function = flat.function(&decl.name)?.clone();
246        function.name = crate::ident(name)?;
247        function.ret.slot = None;
248        if flat.functions.contains_key(name) {
249            return Err("callback return function name collision".into());
250        }
251        flat.functions.insert(name.into(), function);
252        let mut variant = decl.clone();
253        variant.name = name.into();
254        variant.foreign = None;
255        variant.callback_return = None;
256        variant.ret = Some(output.into_preparation());
257        Ok(variant)
258    }
259    /// Resolve source-dependent operations, compile them, and atomically install one binding.
260    /// Existing converters are preserved unless explicitly replaced.
261    pub fn fill<C: Codec>(
262        &self,
263        flat: &mut Flat<C::Schema>,
264        codec: &C,
265        decl: &FunctionDecl,
266        extra: &C::Extra,
267    ) -> Result<()> {
268        let recipe = self
269            .resolve_function(flat, codec, decl, extra)
270            .map_err(|e| format!("{}: {e}", decl.name))?;
271        codec.compile_declarations(flat, extra)?;
272        crate::resolved::compile_function(flat, codec, &decl.name, recipe, extra)
273            .map_err(|e| format!("{}: {e}", decl.name))
274    }
275    pub(crate) fn resolve_function<C: Codec>(
276        &self,
277        flat: &Flat<C::Schema>,
278        codec: &C,
279        decl: &FunctionDecl,
280        extra: &C::Extra,
281    ) -> Result<crate::resolved::FunctionRecipe> {
282        let source = flat.function(&decl.name)?;
283        if decl.callback_return.is_some()
284            && source.ret.slot.is_some()
285            && decl.ret.is_none()
286            && decl.error.is_none()
287        {
288            return Err("callback_return cannot reconstruct a prebound output slot; supply an explicit output override".into());
289        }
290        for name in decl.params.keys() {
291            if !source.params.iter().any(|p| p.name == name.as_str()) {
292                return Err(format!("unknown parameter `{name}`"));
293            }
294        }
295        let mut inputs = vec![];
296        for p in &source.params {
297            let explicit = decl.params.get(&p.name.to_string());
298            inputs.push(if explicit.is_some() || p.ty.slot.is_none() {
299                let tree = self.resolve_input(
300                    flat,
301                    codec,
302                    &p.ty.ty,
303                    explicit.unwrap_or(&Input::auto()),
304                    extra,
305                )?;
306                Some(tree)
307            } else {
308                None
309            });
310        }
311        if decl.result_value && (decl.error.is_some() || result_parts(&source.ret.ty).is_none()) {
312            return Err(
313                "result_as_value requires Result<T, E> and cannot combine with on_error".into(),
314            );
315        }
316        if decl.error.is_some() && result_parts(&source.ret.ty).is_none() {
317            return Err("on_error requires a top-level Result<T, E> return".into());
318        }
319        let output = if decl.ret.is_some() || decl.error.is_some() || source.ret.slot.is_none() {
320            let tree = self.function_output(flat, codec, &source.ret.ty, decl, extra)?;
321            Some(tree)
322        } else {
323            None
324        };
325        Ok(crate::resolved::FunctionRecipe { inputs, output })
326    }
327    fn function_output<C: Codec>(
328        &self,
329        flat: &Flat<C::Schema>,
330        codec: &C,
331        ty: &syn::Type,
332        decl: &FunctionDecl,
333        extra: &C::Extra,
334    ) -> Result<Output> {
335        let tree = if let Some((_, error)) = result_parts(ty).filter(|_| !decl.result_value) {
336            Output::Unary {
337                operation: OutputUnary::Result {
338                    error: self.error_mapper_for(decl, error).map(str::to_owned),
339                },
340                child: Box::new(decl.ret.clone().unwrap_or_else(Output::auto)),
341            }
342        } else {
343            decl.ret.clone().unwrap_or_else(Output::auto)
344        };
345        self.resolve_output(flat, codec, ty, &tree, extra)
346    }

398    pub(crate) fn input<C: Codec>(
399        &self,
400        flat: &Flat<C::Schema>,
401        codec: &C,
402        ty: &syn::Type,
403        tree: &Input<InputRequest, OutputRequest>,
404        extra: &C::Extra,
405        stack: &mut Vec<String>,
406    ) -> Result<Input> {
407        let resolve =
408            |ty: &syn::Type, tree: &Input<InputRequest, OutputRequest>, stack: &mut Vec<String>| {
409                self.input(flat, codec, ty, tree, extra, stack)
410            };
411        match tree {
412            Input::None => {
413                if type_key(ty) == "()" {
414                    Ok(Input::None)
415                } else {
416                    Err("None input requires unit".into())
417                }
418            }
419            Input::Branch {
420                operation: InputBranch::Variant,
421                ..
422            } => Err("variant branch requires an enum parent".into()),
423            Input::Request(InputRequest::EnumFields(entries)) => {
424                self.enum_input_recipe(flat, codec, ty, extra, entries, false, stack)
425            }
426            Input::Branch {
427                operation: InputBranch::Enum { value },
428                children: variants,
429            } => {
430                let source = flat.payload_enum(ty)?;
431                if variants.len() != source.variants.len() {
432                    return Err("wrong enum variant count".into());
433                }
434                let variants = source
435                    .variants
436                    .iter()
437                    .zip(variants)
438                    .map(|(arm, fields)| {
439                        let fields = fields.variant_fields()?;
440                        if fields.len() != arm.fields.len() {
441                            return Err("wrong enum field count".into());
442                        }
443                        arm.fields
444                            .iter()
445                            .zip(fields)
446                            .map(|(f, r)| {
447                                let previous = self.input_role.get();
448                                if *value {
449                                    self.input_role.set(crate::InputRole::EnumField);
450                                }
451                                let result = resolve(&f.ty, r, stack);
452                                self.input_role.set(previous);
453                                result
454                            })
455                            .collect::<Result<Vec<_>>>()
456                    })
457                    .collect::<Result<Vec<_>>>()?;
458                Ok(Input::enum_variants(variants, *value))
459            }
460            Input::Request(InputRequest::Value) => {
461                self.resolve_input_value(flat, codec, ty, extra, stack)
462            }
463            Input::Request(InputRequest::Conversion(decl)) => {
464                decl.validate()?;
465                let source = decl
466                    .input
467                    .as_ref()
468                    .ok_or("local input conversion has no input source")?;
469                let conversion = source.resolve(flat, ty, true)?;
470                self.resolve_input_conversion(
471                    flat,
472                    codec,
473                    ty,
474                    conversion,
475                    &Input::value(),
476                    extra,
477                    stack,
478                )
479            }
480            Input::Unary {
481                operation: InputUnary::Convert { conversion },
482                child: input,
483            } => self.resolve_input_conversion(
484                flat,
485                codec,
486                ty,
487                (**conversion).clone(),
488                input,
489                extra,
490                stack,
491            ),
492            Input::Unary {
493                operation: InputUnary::Temporary,
494                child,
495            } => Ok(Input::Unary {
496                operation: InputUnary::Temporary,
497                child: Box::new(resolve(
498                    &crate::temporary_input_type(ty)?,
499                    &input_body((**child).clone()),
500                    stack,
501                )?),
502            }),
503            Input::Unary {
504                operation: InputUnary::Bulk,
505                child,
506            } => Ok(Input::Unary {
507                operation: InputUnary::Bulk,
508                child: Box::new(resolve(ty, &input_body((**child).clone()), stack)?),
509            }),
510            Input::Request(InputRequest::Auto) => {
511                let key = crate::callback::input_key(flat, ty)?;
512                if let Some(tree) = self.inputs.get(&key) {
513                    if stack.contains(&key) {
514                        return Err(format!("recursive input expansion `{key}`"));
515                    }
516                    stack.push(key);
517                    let result = resolve(ty, tree, stack);
518                    stack.pop();
519                    return result;
520                }
521                if let Some(inner) = option_inner(ty) {
522                    return Ok(Input::Unary {
523                        operation: InputUnary::Optional,
524                        child: Box::new(resolve(inner, &Input::auto(), stack)?),
525                    });
526                }
527                if let Some((element, _)) = sequence_parts(ty).filter(|_| {
528                    !codec.input_collection_is_leaf(flat, ty, self.input_role.get(), extra)
529                }) {
530                    return Ok(Input::Unary {
531                        operation: InputUnary::Sequence,
532                        child: Box::new(resolve(element, &Input::auto(), stack)?),
533                    });
534                }
535                if type_key(ty) == "()" {
536                    return Ok(Input::None);
537                }
538                resolve(ty, &Input::Leaf(InputLeafOp::Value), stack)
539            }
540            Input::Request(InputRequest::ForType { ty: target, input }) => {
541                let target = crate::callback::input_type(flat, target)?;
542                if crate::callback::input_key(flat, ty)? == type_key(&target) {
543                    return resolve(ty, &input_body((**input).clone()), stack);
544                }
545                if let Some(inner) = option_inner(ty) {
546                    return Ok(Input::Unary {
547                        operation: InputUnary::Optional,
548                        child: Box::new(resolve(inner, tree, stack)?),
549                    });
550                }
551                if let Some((element, _)) = sequence_parts(ty) {
552                    return Ok(Input::Unary {
553                        operation: InputUnary::Sequence,
554                        child: Box::new(resolve(element, tree, stack)?),
555                    });
556                }
557                Err("input expansion type does not match occurrence".into())
558            }
559            Input::Leaf(InputLeafOp::Value) => {
560                if flat
561                    .type_info(ty)
562                    .ok()
563                    .is_some_and(|i| i.conversion.input.is_none() && i.conversion.output.is_some())
564                {
565                    return Err(format!(
566                        "`{}` has no canonical input conversion",
567                        type_key(ty)
568                    ));
569                }
570                if let Some(conversion) = flat
571                    .type_info(ty)
572                    .ok()
573                    .and_then(|i| i.conversion.input.clone())
574                {
575                    return self.resolve_input_conversion(
576                        flat,
577                        codec,
578                        ty,
579                        conversion,
580                        &Input::value(),
581                        extra,
582                        stack,
583                    );
584                }
585
586                if prebindgen_flat::callback_signature(ty)?.is_some() {
587                    return resolve(
588                        ty,
589                        &Input::Request(InputRequest::Callback(Box::new(
590                            crate::CallbackDecl::new(ty.clone()),
591                        ))),
592                        stack,
593                    );
594                }
595                if let Some(inner) = option_inner(ty) {
596                    return Ok(Input::Unary {
597                        operation: InputUnary::Optional,
598                        child: Box::new(resolve(inner, &Input::Leaf(InputLeafOp::Value), stack)?),
599                    });
600                }
601                if let Some((element, _)) = sequence_parts(ty).filter(|_| {
602                    !codec.input_collection_is_leaf(flat, ty, self.input_role.get(), extra)
603                }) {
604                    return Ok(Input::Unary {
605                        operation: InputUnary::Sequence,
606                        child: Box::new(resolve(element, &Input::auto(), stack)?),
607                    });
608                }
609                if let Some(replacement) = codec.expand_input(flat, ty, tree, extra)? {
610                    return resolve(ty, &replacement, stack);
611                }
612                Ok(Input::Leaf(InputLeafOp::Value))
613            }
614            Input::Arrow {
615                operation: callback,
616                domain,
617                codomain,
618            } => {
619                callback.validate_for(flat, ty)?;
620                let signature = prebindgen_flat::callback_signature(ty)?
621                    .ok_or("arrow requires a callable source")?;
622                let arguments = domain.argument_nodes()?;
623                if arguments.len() != signature.arguments.len() {
624                    return Err("wrong arrow domain arity".into());
625                }
626                Ok(Input::Arrow {
627                    operation: callback.clone(),
628                    domain: Box::new(Output::arguments(
629                        arguments
630                            .iter()
631                            .zip(&signature.arguments)
632                            .map(|(tree, ty)| {
633                                self.output(flat, codec, ty, tree, extra, &mut vec![], true)
634                            })
635                            .collect::<Result<Vec<_>>>()?,
636                    )),
637                    codomain: Box::new(resolve(&signature.return_type, codomain, stack)?),
638                })
639            }
640            Input::Request(InputRequest::Callback(declaration)) => {
641                let canonical = crate::callback::input_type(flat, ty)?;
642                if !same_type(
643                    &canonical,
644                    &crate::callback::input_type(flat, &declaration.ty)?,
645                ) {
646                    return Err("callback declaration type does not match occurrence".into());
647                }
648                let recursion_key = format!("callback:{}", type_key(&canonical));
649                if stack.contains(&recursion_key) {
650                    return Err("recursive callback conversion".into());
651                }
652                stack.push(recursion_key);
653                let result = (|| {
654                    let signature = prebindgen_flat::callback_signature(&canonical)?
655                        .ok_or("callback declaration requires impl Fn(...) -> R")?;
656                    let mut selected = match self.inputs.get(&type_key(&canonical)) {
657                        Some(Input::Request(InputRequest::Callback(default))) => {
658                            (**default).clone()
659                        }
660                        _ => crate::CallbackDecl::new(canonical.clone()),
661                    };
662                    if declaration.foreign.is_some() {
663                        selected.foreign = declaration.foreign.clone();
664                    }
665                    if declaration.namespace.is_some() {
666                        selected.namespace = declaration.namespace.clone();
667                    }
668                    selected.params.extend(declaration.params.clone());
669                    if declaration.ret.is_some() {
670                        selected.ret = declaration.ret.clone();
671                    }
672                    if selected
673                        .params
674                        .keys()
675                        .any(|&i| i >= signature.arguments.len())
676                    {
677                        return Err("callback parameter override index is out of range".into());
678                    }
679                    let mut arguments = Vec::with_capacity(signature.arguments.len());
680                    for (index, source) in signature.arguments.iter().enumerate() {
681                        // Retain the source reference for exact type overrides.
682                        // As with getters, referent defaults may expand a borrow;
683                        // compilation preserves its lifetime and ownership.
684                        fn subject(ty: &syn::Type) -> &syn::Type {
685                            match ty {
686                                syn::Type::Reference(r) => subject(&r.elem),
687                                _ => option_inner(ty).map(subject).unwrap_or(ty),
688                            }
689                        }
690                        let exposed = crate::output_referent(source).unwrap_or(source);
691                        let scoped = !selected.params.contains_key(&index)
692                            && !self.outputs.contains_key(&type_key(exposed))
693                            && !self.outputs.contains_key(&type_key(subject(exposed)));
694                        let ownership = if scoped {
695                            crate::CallbackArgumentOwnership::Scoped
696                        } else {
697                            crate::CallbackArgumentOwnership::Transferred
698                        };
699                        arguments.push(
700                            self.output(
701                                flat,
702                                codec,
703                                source,
704                                selected.params.get(&index).unwrap_or(&Output::auto()),
705                                extra,
706                                &mut vec![],
707                                true,
708                            )?
709                            .callback_argument(ownership),
710                        );
711                    }
712                    let ret = resolve(
713                        &signature.return_type,
714                        selected.ret.as_ref().unwrap_or(&Input::auto()),
715                        stack,
716                    )?;
717                    Ok(Input::Arrow {
718                        domain: Box::new(Output::arguments(arguments)),
719                        codomain: Box::new(ret),
720                        operation: Box::new(crate::CallbackOperation {
721                            ty: canonical,
722                            foreign: selected.foreign,
723                            namespace: selected.namespace,
724                            explicit_return: selected.ret.is_some(),
725                        }),
726                    })
727                })();
728                stack.pop();
729                result
730            }
731            Input::Request(InputRequest::Call(function)) => {
732                let f = flat.function(function)?;
733                resolve(
734                    ty,
735                    &Input::Branch {
736                        operation: InputBranch::Constructor {
737                            function: function.clone(),
738                        },
739                        children: vec![Input::auto(); f.params.len()],
740                    },
741                    stack,
742                )
743            }
744            Input::Branch {
745                operation: InputBranch::Constructor { function },
746                children: args,
747            } => {
748                let f = flat.function(function)?;
749                if !same_type(&f.ret.ty, ty) {
750                    return Err(format!(
751                        "constructor `{function}` returns `{}`, expected `{}`",
752                        type_key(&f.ret.ty),
753                        type_key(ty)
754                    ));
755                }
756                if args.len() != f.params.len() {
757                    return Err(format!("constructor `{function}`: wrong argument count"));
758                }
759                Ok(Input::Branch {
760                    operation: InputBranch::Constructor {
761                        function: function.clone(),
762                    },
763                    children: f
764                        .params
765                        .iter()
766                        .zip(args)
767                        .map(|(p, child)| resolve(&p.ty.ty, child, stack))
768                        .collect::<Result<_>>()?,
769                })
770            }
771            Input::Unary {
772                operation: InputUnary::Optional,
773                child,
774            } => {
775                let inner = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
776                Ok(Input::Unary {
777                    operation: InputUnary::Optional,
778                    child: Box::new(resolve(inner, child, stack)?),
779                })
780            }
781            Input::Unary {
782                operation: InputUnary::Sequence,
783                child,
784            } => {
785                let (element, _) =
786                    sequence_parts(ty).ok_or("sequence input requires Vec<T>, &[T] or &Vec<T>")?;
787                Ok(Input::Unary {
788                    operation: InputUnary::Sequence,
789                    child: Box::new(resolve(element, child, stack)?),
790                })
791            }
792            Input::Branch {
793                operation: InputBranch::Fields,
794                children,
795            }
796            | Input::Branch {
797                operation: InputBranch::Record,
798                children,
799            } => {
800                if type_key(ty) == "()" && children.is_empty() {
801                    return Ok(Input::None);
802                }
803                let fields = &flat.record(ty)?.fields;
804                if fields.len() != children.len() {
805                    return Err(format!(
806                        "`{}`: expected {} field converters, got {}",
807                        type_key(ty),
808                        fields.len(),
809                        children.len()
810                    ));
811                }
812                let children = fields
813                    .iter()
814                    .zip(children)
815                    .map(|(f, c)| resolve(&f.ty, c, stack))
816                    .collect::<Result<_>>()?;
817                Ok(
818                    if matches!(
819                        tree,
820                        Input::Branch {
821                            operation: InputBranch::Record,
822                            children: _
823                        }
824                    ) {
825                        Input::Branch {
826                            operation: InputBranch::Record,
827                            children,
828                        }
829                    } else {
830                        Input::Branch {
831                            operation: InputBranch::Fields,
832                            children,
833                        }
834                    },
835                )
836            }
837            Input::Branch {
838                operation: InputBranch::Selector,
839                children: arms,
840            } => {
841                if arms.is_empty() {
842                    return Err("an input selector needs at least one alternative".into());
843                }
844                Ok(Input::Branch {
845                    operation: InputBranch::Selector,
846                    children: arms
847                        .iter()
848                        .map(|child| resolve(ty, child, stack))
849                        .collect::<Result<_>>()?,
850                })
851            }
852        }
853    }
854    /// Resolve every output operation, source reference and ownership choice before compilation.
855    pub fn resolve_output<C: Codec>(
856        &self,
857        flat: &Flat<C::Schema>,
858        codec: &C,
859        ty: &syn::Type,
860        tree: &Output<OutputRequest, InputRequest>,
861        extra: &C::Extra,
862    ) -> Result<Output> {
863        self.output(flat, codec, ty, tree, extra, &mut vec![], false)
864            .map(lift_output_bulk)
865    }
866    #[allow(clippy::too_many_arguments)] // Getter defaults differ from the existing plain-reference ABI.
867    pub(crate) fn output<C: Codec>(
868        &self,
869        flat: &Flat<C::Schema>,
870        codec: &C,
871        ty: &syn::Type,
872        tree: &Output<OutputRequest, InputRequest>,
873        extra: &C::Extra,
874        stack: &mut Vec<String>,
875        getter_result: bool,
876    ) -> Result<Output> {
877        let resolve = |ty: &syn::Type,
878                       tree: &Output<OutputRequest, InputRequest>,
879                       stack: &mut Vec<String>| {
880            self.output(flat, codec, ty, tree, extra, stack, getter_result)
881        };
882        if let Some(referent) = crate::output_referent(ty) {
883            if (getter_result || !matches!(tree, Output::Leaf(OutputLeafOp::Value)))
884                && !matches!(
885                    tree,
886                    Output::Request(OutputRequest::Auto | OutputRequest::ForType { .. })
887                )
888            {
889                return resolve(referent, tree, stack);
890            }
891        }
892        match tree {
893            Output::Unary {
894                operation: OutputUnary::CallbackArgument { ownership },
895                child,
896            } => Ok(resolve(ty, child, stack)?.callback_argument(*ownership)),
897            Output::None => {
898                if type_key(ty) == "()" {
899                    Ok(Output::None)
900                } else {
901                    Err("None output requires unit".into())
902                }
903            }
904            Output::Arrow { operation, .. } => match *operation {},
905            Output::Branch {
906                operation: OutputBranch::Variant | OutputBranch::Arguments,
907                ..
908            } => Err("variant/argument branch requires its parent operation".into()),
909            Output::Request(OutputRequest::EnumFields(entries)) => self.enum_output_recipe(
910                flat,
911                codec,
912                crate::output_referent(ty).unwrap_or(ty),
913                extra,
914                entries,
915                false,
916                stack,
917                getter_result,
918            ),
919            Output::Branch {
920                operation: OutputBranch::Enum { value, mode },
921                children: variants,
922            } => {
923                let source = flat.payload_enum(crate::output_referent(ty).unwrap_or(ty))?;
924                if variants.len() != source.variants.len() {
925                    return Err("wrong enum variant count".into());
926                }
927                let variants = source
928                    .variants
929                    .iter()
930                    .zip(variants)
931                    .map(|(arm, fields)| {
932                        let fields = fields.variant_fields()?;
933                        if fields.len() != arm.fields.len() {
934                            return Err("wrong enum field count".into());
935                        }
936                        arm.fields
937                            .iter()
938                            .zip(fields)
939                            .map(|(f, r)| resolve(&f.ty, r, stack))
940                            .collect::<Result<Vec<_>>>()
941                    })
942                    .collect::<Result<Vec<_>>>()?;
943                Ok(Output::enum_variants(variants, *value, *mode))
944            }
945            Output::Request(OutputRequest::Value { projection }) => self.resolve_output_value(
946                flat,
947                codec,
948                ty,
949                extra,
950                stack,
951                getter_result,
952                projection.as_ref(),
953            ),
954            Output::Request(OutputRequest::Conversion(decl)) => {
955                decl.validate()?;
956                let source = decl
957                    .output
958                    .as_ref()
959                    .ok_or("local output conversion has no output source")?;
960                let conversion = source.resolve(flat, ty, false)?;
961                self.resolve_output_conversion(
962                    flat,
963                    codec,
964                    ty,
965                    conversion,
966                    &Output::value(),
967                    extra,
968                    stack,
969                )
970            }
971            Output::Unary {
972                operation: OutputUnary::Convert { conversion },
973                child: output,
974            } => self.resolve_output_conversion(
975                flat,
976                codec,
977                ty,
978                (**conversion).clone(),
979                output,
980                extra,
981                stack,
982            ),
983            Output::Unary {
984                operation: OutputUnary::Bulk,
985                child,
986            } => Ok(Output::Unary {
987                operation: OutputUnary::Bulk,
988                child: Box::new(resolve(ty, &output_body((**child).clone()), stack)?),
989            }),
990            Output::Leaf(OutputLeafOp::Discard) => Ok(Output::Leaf(OutputLeafOp::Discard)),
991            Output::Unary {
992                operation: OutputUnary::Named { name },
993                child: output,
994            } => Ok(Output::Unary {
995                operation: OutputUnary::Named { name: name.clone() },
996                child: Box::new(resolve(ty, output, stack)?),
997            }),
998            Output::Request(OutputRequest::Auto) => {
999                let key = type_key(ty);
1000                if let Some(tree) = self.outputs.get(&key) {
1001                    if stack.contains(&key) {
1002                        return Err(format!("recursive output expansion `{key}`"));
1003                    }
1004                    stack.push(key);
1005                    let result = resolve(ty, tree, stack);
1006                    stack.pop();
1007                    return result;
1008                }
1009                if getter_result {
1010                    if let Some(referent) = crate::output_referent(ty) {
1011                        return resolve(referent, tree, stack);
1012                    }
1013                }
1014                if let Some(inner) = option_inner(ty) {
1015                    return Ok(Output::Unary {
1016                        operation: OutputUnary::Optional,
1017                        child: Box::new(resolve(inner, &Output::auto(), stack)?),
1018                    });
1019                }
1020                if let Some((element, _)) =
1021                    sequence_parts(ty).filter(|_| !codec.output_collection_is_leaf(flat, ty, extra))
1022                {
1023                    return Ok(Output::Unary {
1024                        operation: OutputUnary::Sequence {
1025                            delivery: SequenceDelivery::Auto,
1026                            mode: None,
1027                        },
1028                        child: Box::new(resolve(element, &Output::auto(), stack)?),
1029                    });
1030                }
1031                if type_key(ty) == "()" {
1032                    return Ok(Output::None);
1033                }
1034                resolve(ty, &Output::Leaf(OutputLeafOp::Value), stack)
1035            }
1036            Output::Request(OutputRequest::ForType { ty: target, output }) => {
1037                let target = flat.resolve_type(target)?;
1038                if same_type(ty, &target) {
1039                    return resolve(ty, &output_body((**output).clone()), stack);
1040                }
1041                if let Some(referent) = crate::output_referent(ty) {
1042                    return resolve(referent, tree, stack);
1043                }
1044                if let Some(inner) = option_inner(ty) {
1045                    let child = resolve(inner, tree, stack)?;
1046                    return Ok(if matches!(child, Output::Leaf(OutputLeafOp::Discard)) {
1047                        Output::Leaf(OutputLeafOp::Discard)
1048                    } else {
1049                        Output::Unary {
1050                            operation: OutputUnary::Optional,
1051                            child: Box::new(child),
1052                        }
1053                    });
1054                }
1055                if let Some((element, _)) = sequence_parts(ty) {
1056                    let child = resolve(element, tree, stack)?;
1057                    return Ok(if matches!(child, Output::Leaf(OutputLeafOp::Discard)) {
1058                        Output::Leaf(OutputLeafOp::Discard)
1059                    } else {
1060                        Output::Unary {
1061                            operation: OutputUnary::Sequence {
1062                                delivery: SequenceDelivery::Auto,
1063                                mode: None,
1064                            },
1065                            child: Box::new(child),
1066                        }
1067                    });
1068                }
1069                Err("output expansion type does not match occurrence".into())
1070            }
1071            Output::Leaf(OutputLeafOp::Value) | Output::Leaf(OutputLeafOp::Projected(_)) => {
1072                let conversion_type = crate::output_referent(ty).unwrap_or(ty);
1073                if flat
1074                    .type_info(conversion_type)
1075                    .ok()
1076                    .is_some_and(|i| i.conversion.output.is_none() && i.conversion.input.is_some())
1077                {
1078                    return Err(format!(
1079                        "`{}` has no canonical output conversion",
1080                        type_key(ty)
1081                    ));
1082                }
1083                let converted = flat
1084                    .type_info(ty)
1085                    .ok()
1086                    .and_then(|i| i.conversion.output.clone())
1087                    .or_else(|| {
1088                        crate::output_referent(ty)
1089                            .and_then(|t| flat.type_info(t).ok())
1090                            .and_then(|i| i.conversion.output.clone())
1091                    });
1092                if let Some(conversion) = converted {
1093                    if let Output::Leaf(OutputLeafOp::Projected(policy)) = tree {
1094                        return self.resolve_output_value(
1095                            flat,
1096                            codec,
1097                            ty,
1098                            extra,
1099                            stack,
1100                            getter_result,
1101                            Some(policy),
1102                        );
1103                    }
1104                    return self.resolve_output_conversion(
1105                        flat,
1106                        codec,
1107                        ty,
1108                        conversion,
1109                        &Output::value(),
1110                        extra,
1111                        stack,
1112                    );
1113                }
1114
1115                if let Some(inner) = option_inner(ty) {
1116                    return Ok(Output::Unary {
1117                        operation: OutputUnary::Optional,
1118                        child: Box::new(resolve(inner, tree, stack)?),
1119                    });
1120                }
1121                if let Some((element, _)) =
1122                    sequence_parts(ty).filter(|_| !codec.output_collection_is_leaf(flat, ty, extra))
1123                {
1124                    let mode = if let Output::Leaf(OutputLeafOp::Projected(p)) = tree {
1125                        if !p.fields.is_empty() {
1126                            return Err(
1127                                "collection field projections belong on the element type".into()
1128                            );
1129                        }
1130                        Some(p.mode)
1131                    } else {
1132                        None
1133                    };
1134                    return Ok(Output::Unary {
1135                        operation: OutputUnary::Sequence {
1136                            delivery: SequenceDelivery::Value,
1137                            mode,
1138                        },
1139                        child: Box::new(resolve(element, &Output::auto(), stack)?),
1140                    });
1141                }
1142                if let Some(replacement) = codec.expand_output(flat, ty, tree, extra)? {
1143                    return resolve(ty, &replacement, stack);
1144                }
1145                Ok(match tree {
1146                    Output::Leaf(OutputLeafOp::Value) => Output::Leaf(OutputLeafOp::Value),
1147                    Output::Leaf(OutputLeafOp::Projected(policy)) => {
1148                        Output::Leaf(OutputLeafOp::Projected(policy.clone()))
1149                    }
1150                    _ => unreachable!(),
1151                })
1152            }
1153            Output::Unary {
1154                operation: OutputUnary::Optional,
1155                child,
1156            } => {
1157                let inner = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
1158                Ok(Output::Unary {
1159                    operation: OutputUnary::Optional,
1160                    child: Box::new(resolve(inner, child, stack)?),
1161                })
1162            }
1163            Output::Unary {
1164                operation: OutputUnary::Sequence { delivery, mode },
1165                child: element,
1166            } => {
1167                let (ty, _) =
1168                    sequence_parts(ty).ok_or("sequence output requires a collection source")?;
1169                Ok(Output::Unary {
1170                    operation: OutputUnary::Sequence {
1171                        delivery: *delivery,
1172                        mode: *mode,
1173                    },
1174                    child: Box::new(resolve(ty, element, stack)?),
1175                })
1176            }
1177            Output::Request(OutputRequest::NamedFields { overrides, mode }) => {
1178                let fields = &flat.record(ty)?.fields;
1179                for name in overrides.keys() {
1180                    if !fields.iter().any(|f| f.name == name.as_str()) {
1181                        return Err(format!("unknown output field `{name}`"));
1182                    }
1183                }
1184                let children = fields
1185                    .iter()
1186                    .map(|f| {
1187                        resolve(
1188                            &f.ty,
1189                            overrides
1190                                .get(&f.name.to_string())
1191                                .unwrap_or(&Output::auto()),
1192                            stack,
1193                        )
1194                    })
1195                    .collect::<Result<_>>()?;
1196                Ok(if *mode == ProjectionMode::Owned {
1197                    Output::Branch {
1198                        operation: OutputBranch::OwnedFields,
1199                        children,
1200                    }
1201                } else {
1202                    Output::Branch {
1203                        operation: OutputBranch::Fields,
1204                        children,
1205                    }
1206                })
1207            }
1208            Output::Branch {
1209                operation: OutputBranch::Fields,
1210                children,
1211            }
1212            | Output::Branch {
1213                operation: OutputBranch::OwnedFields,
1214                children,
1215            }
1216            | Output::Branch {
1217                operation: OutputBranch::Record { .. },
1218                children,
1219                ..
1220            } => {
1221                if type_key(ty) == "()" && children.is_empty() {
1222                    return Ok(Output::None);
1223                }
1224                let fields = &flat.record(ty)?.fields;
1225                if fields.len() != children.len() {
1226                    return Err("wrong output field count".into());
1227                }
1228                let children = fields
1229                    .iter()
1230                    .zip(children)
1231                    .map(|(f, c)| resolve(&f.ty, c, stack))
1232                    .collect::<Result<_>>()?;
1233                Ok(match tree {
1234                    Output::Branch {
1235                        operation: OutputBranch::Fields,
1236                        children: _,
1237                    } => Output::Branch {
1238                        operation: OutputBranch::Fields,
1239                        children,
1240                    },
1241                    Output::Branch {
1242                        operation: OutputBranch::OwnedFields,
1243                        children: _,
1244                    } => Output::Branch {
1245                        operation: OutputBranch::OwnedFields,
1246                        children,
1247                    },
1248                    Output::Branch {
1249                        operation: OutputBranch::Record { mode, .. },
1250                        ..
1251                    } => Output::Branch {
1252                        operation: OutputBranch::Record { mode: *mode },
1253                        children,
1254                    },
1255                    _ => unreachable!(),
1256                })
1257            }
1258            Output::Unary {
1259                operation: OutputUnary::Getter { function },
1260                child: output,
1261            }
1262            | Output::Unary {
1263                operation: OutputUnary::Into { function },
1264                child: output,
1265            } => {
1266                let f = flat.function(function)?;
1267                let expected = if matches!(
1268                    tree,
1269                    Output::Unary {
1270                        operation: OutputUnary::Into { .. },
1271                        ..
1272                    }
1273                ) {
1274                    ty.clone()
1275                } else if matches!(f.params.first().map(|p| &p.ty.ty), Some(syn::Type::Reference(r)) if r.mutability.is_some())
1276                {
1277                    syn::parse_quote!(&mut #ty)
1278                } else {
1279                    syn::parse_quote!(&#ty)
1280                };
1281                if f.params.len() != 1 || !same_type(&f.params[0].ty.ty, &expected) {
1282                    return Err(format!(
1283                        "getter `{function}` must take exactly `{}`",
1284                        type_key(&expected)
1285                    ));
1286                }
1287                // Mutable getters require a retained owner. Compilation decides
1288                // this once all sibling output views and the codec are known.
1289                let output =
1290                    Box::new(self.output(flat, codec, &f.ret.ty, output, extra, stack, true)?);
1291                Ok(
1292                    if matches!(
1293                        tree,
1294                        Output::Unary {
1295                            operation: OutputUnary::Into { .. },
1296                            ..
1297                        }
1298                    ) {
1299                        Output::Unary {
1300                            operation: OutputUnary::Into {
1301                                function: function.clone(),
1302                            },
1303                            child: output,
1304                        }
1305                    } else {
1306                        Output::Unary {
1307                            operation: OutputUnary::Getter {
1308                                function: function.clone(),
1309                            },
1310                            child: output,
1311                        }
1312                    },
1313                )
1314            }
1315            Output::Branch {
1316                operation: OutputBranch::Combine,
1317                children: parts,
1318            } => {
1319                if parts.is_empty() {
1320                    return Err("an output combination needs at least one part".into());
1321                }
1322                Ok(Output::Branch {
1323                    operation: OutputBranch::Combine,
1324                    children: parts
1325                        .iter()
1326                        .map(|child| resolve(ty, child, stack))
1327                        .collect::<Result<_>>()?,
1328                })
1329            }
1330            Output::Unary {
1331                operation: OutputUnary::Result { error },
1332                child: ok,
1333            } => {
1334                let (success, failure) =
1335                    result_parts(ty).ok_or("Result recipe requires Result<T, E>")?;
1336                if let Some(name) = error {
1337                    let f = flat.function(name)?;
1338                    let valid = f.params.len() == 1
1339                        && (same_type(&f.params[0].ty.ty, failure)
1340                            || matches!(&f.params[0].ty.ty,syn::Type::Reference(r) if r.mutability.is_none()&&same_type(&r.elem,failure)));
1341                    if !valid || type_key(&f.ret.ty) != "String" {
1342                        return Err(
1343                            "Result error mapper must take exactly E or &E and return String"
1344                                .into(),
1345                        );
1346                    }
1347                }
1348                Ok(Output::Unary {
1349                    operation: OutputUnary::Result {
1350                        error: error.clone(),
1351                    },
1352                    child: Box::new(resolve(success, ok, stack)?),
1353                })
1354            }
1355        }
1356    }

1410                && self.ret.is_none()
1411                && self.error.is_none()

1417            allow_name || self.foreign.is_none(),
1418            "this helper reference does not support renaming"

1422    /// Replace one parameter's default conversion using the production API.

1426        decl: impl Into<Input<InputRequest, OutputRequest>>,
1427    ) -> Self {

1429    }

1431    pub fn expand_return(self, decl: impl Into<Output<OutputRequest, InputRequest>>) -> Self {
1432        self.output(decl)
1433    }
1434    /// Deliver the whole source Result as data instead of routing its Err arm.
1435    /// Return overrides then address the complete Result. Incompatible with on_error.
1436    pub fn result_as_value(mut self) -> Self {
1437        self.result_value = true;
1438        self
1439    }
1440    /// Add callback delivery alongside the ordinary sequence return.
1441    ///
1442    /// C emits `<function>_each`, whose visitor receives each element's converted
1443    /// fields. Kotlin emits `<function>Fold(initial, ..., fold)`, whose folder
1444    /// receives an accumulator and each element's converted Kotlin values.
1445    /// Supports one projected sequence, optionally inside Option and an outer Result.
1446    /// Existing buffer/list returns are preserved. Adapter builders validate the
1447    /// return shape and generated-name collisions.
1448    ///
1449    /// On adapters supporting bulk, expansions stay within each callback element.
1450    /// The ordinary variant
1451    /// and callback variant are prepared separately from the same declaration.
1452    /// For a source returning `Vec<Row>`, this selects a frame per row for the
1453    /// callback variant; its ordinary return keeps a frame for the collection:
1454    /// ```
1455    /// use prebindgen_declarations::{fun, expand_return};
1456    /// let _ = fun!(rows)
1457    ///     .expand_return(expand_return!(Row).field_self_into().bulk())
1458    ///     .callback_return();
1459    /// ```
1460    /// For a source returning `Vec<Vec<Row>>`, target the inner collection to
1461    /// select one bulk frame per batch:
1462    /// ```
1463    /// use prebindgen_declarations::{fun, expand_return};
1464    /// let _ = fun!(batches)
1465    ///     .callback_return_only()
1466    ///     .expand_return(expand_return!(Vec<Row>).field_self_into().bulk());
1467    /// ```
1468    /// An explicit bulk wrapper around the entire callback sequence is invalid;
1469    /// apply bulk to its element expansion instead. This also applies to a bulk
1470    /// wrapper supplied by a whole-collection type default. Nested automatic
1471    /// sequences stay ordinary inside a callback element; explicit bulk opts in
1472    /// to packing them. Chain order does not affect these policies, and
1473    /// per-function expansions override type-wide defaults.
1474    pub fn callback_return(mut self) -> Self {
1475        self.callback_return = Some(CallbackReturn::Additional);
1476        self
1477    }
1478    /// Expose only callback delivery, without generating an ordinary return.
1479    ///
1480    /// This explicitly permits decomposed multi-value Kotlin elements without
1481    /// requiring a Kotlin data class. Generated names are the same as for
1482    /// `callback_return`: C `_each` and Kotlin `Fold`.
1483    pub fn callback_return_only(mut self) -> Self {
1484        self.callback_return = Some(CallbackReturn::Only);
1485        self
1486    }
1487    /// Map the error of a top-level `Result<T, E>` return with a source function.
1488    ///
1489    /// The helper must take exactly `E` or `&E` and return `String`. Only its
1490    /// source name is accepted; it is selected for capture but is not exported.
1491    /// The success value uses normal return defaults and `expand_return` overrides.
1492    /// Replaces any previously selected mapper. Non-Result returns reject this option.
1493    pub fn on_error(mut self, mapper: impl Into<FunctionDecl>) -> Self {
1494        self.error = Some(mapper.into().reference_name(false));
1495        self
1496    }
1497    /// Set a literal destination name, subject to adapter validation.
1498    ///
1499    /// The JNI adapter uses it instead of its default camel-case name.
1500    pub fn name(mut self, name: impl Into<String>) -> Self {
1501        self.foreign = Some(name.into());
1502        self
1503    }
1504    /// Replace the expansion for one source parameter.
1505    ///
1506    /// # Arguments
1507    ///
1508    /// * `name` — the parameter's name, as written in the source.
1509    /// * `decl` — how to build that argument instead of its type's default.
1510    ///
1511    /// # Panics
1512    /// Panics if that parameter already has an override. Resolution checks that the
1513    /// parameter exists and the expansion type matches.
1514    pub fn input(
1515        mut self,
1516        name: impl AsRef<str>,
1517        decl: impl Into<Input<InputRequest, OutputRequest>>,
1518    ) -> Self {

1520            self.params
1521                .insert(name.as_ref().into(), decl.into())
1522                .is_none(),
1523            "duplicate parameter override"
1524        );
1525        self
1526    }
1527    /// Replace the result expansion, including any previously assigned override.
1528    ///
1529    /// Raw structural nodes infer the preparation payload automatically. An explicitly
1530    /// resolved tree must use [`Output::into_preparation`] before registration.
1531    /// ```
1532    /// use prebindgen_declarations::{fun,fields,expand_return,Input,Output};
1533    /// let _ = fun!(run).input("x", Input::fields([])).output(Output::Leaf(prebindgen_declarations::OutputLeafOp::Value));
1534    /// let _ = fields!(parts).field_override("hidden", Output::Leaf(prebindgen_declarations::OutputLeafOp::Discard));
1535    /// let _ = expand_return!(Value).view(Output::getter("length", Output::Leaf(prebindgen_declarations::OutputLeafOp::Value)));
1536    /// let resolved: Output = Output::getter("length", Output::Leaf(prebindgen_declarations::OutputLeafOp::Value));
1537    /// let _ = fun!(run).output(resolved.into_preparation());
1538    /// ```
1539    pub fn output(mut self, decl: impl Into<Output<OutputRequest, InputRequest>>) -> Self {
1540        self.ret = Some(decl.into());

1556/// Collect source helper and representation-type names needed by `Flat::parse_selected`.
1557pub fn source_items<'a>(
1558    inputs: &[Input<InputRequest, OutputRequest>],
1559    outputs: &[Output<OutputRequest, InputRequest>],
1560    errors: &[(syn::Type, String)],
1561    functions: impl IntoIterator<Item = &'a FunctionDecl>,
1562) -> Vec<String> {
1563    fn representation_types(decl: &crate::ConvertDecl, names: &mut Vec<String>) {
1564        struct References<'a>(&'a mut Vec<String>);
1565        impl syn::visit_mut::VisitMut for References<'_> {
1566            fn visit_type_path_mut(&mut self, path: &mut syn::TypePath) {
1567                if path.qself.is_none() {
1568                    if let Some(segment) = path.path.segments.last() {
1569                        self.0.push(segment.ident.to_string());
1570                    }
1571                }
1572                syn::visit_mut::visit_type_path_mut(self, path);
1573            }
1574        }
1575        for mut ty in decl.representation_types() {
1576            syn::visit_mut::VisitMut::visit_type_mut(&mut References(names), &mut ty);
1577        }
1578    }
1579    fn input(tree: &Input<InputRequest, OutputRequest>, names: &mut Vec<String>) {
1580        match tree {
1581            Input::None => {}
1582            Input::Branch {
1583                operation: InputBranch::Variant,
1584                children,
1585            } => {
1586                for child in children {
1587                    input(child, names);
1588                }
1589            }
1590            Input::Branch {
1591                operation: InputBranch::Enum { .. },
1592                children: variants,
1593                ..
1594            } => {
1595                for tree in variants.iter() {
1596                    input(tree, names);
1597                }
1598            }
1599            Input::Request(InputRequest::EnumFields(variants)) => {
1600                for tree in variants.values().flat_map(|v| v.values()) {
1601                    input(tree, names);
1602                }
1603            }
1604            Input::Request(InputRequest::Conversion(d)) => {
1605                names.extend(d.functions());
1606                representation_types(d, names);
1607            }
1608            Input::Unary {
1609                operation: InputUnary::Convert { conversion },
1610                child,
1611            } => {
1612                if let prebindgen_flat::ValueSource::Function(f) = &conversion.source {
1613                    names.push(f.clone());
1614                }
1615                input(child, names);
1616            }
1617            Input::Arrow {
1618                operation: callback,
1619                domain,
1620                codomain,
1621            } => {
1622                let _ = callback;
1623                output(domain, names);
1624                input(codomain, names);
1625            }
1626            Input::Request(InputRequest::Callback(declaration)) => {
1627                for tree in declaration.params.values() {
1628                    output(tree, names);
1629                }
1630                if let Some(tree) = &declaration.ret {
1631                    input(tree, names);
1632                }
1633            }
1634            Input::Request(InputRequest::Call(name)) => names.push(name.clone()),
1635            Input::Branch {
1636                operation: InputBranch::Constructor { function },
1637                children: args,
1638            } => {
1639                names.push(function.clone());
1640                for child in args {
1641                    input(child, names);
1642                }
1643            }
1644            Input::Request(InputRequest::ForType { input: child, .. })
1645            | Input::Unary {
1646                operation: InputUnary::Optional,
1647                child,
1648            }
1649            | Input::Unary {
1650                operation: InputUnary::Sequence,
1651                child,
1652            }
1653            | Input::Unary {
1654                operation: InputUnary::Bulk,
1655                child,
1656            }
1657            | Input::Unary {
1658                operation: InputUnary::Temporary,
1659                child,
1660            } => input(child, names),
1661            Input::Branch {
1662                operation: InputBranch::Fields,
1663                children,
1664            }
1665            | Input::Branch {
1666                operation: InputBranch::Record,
1667                children,
1668            }
1669            | Input::Branch {
1670                operation: InputBranch::Selector,
1671                children,
1672            } => {
1673                for child in children {
1674                    input(child, names)
1675                }
1676            }
1677            Input::Leaf(InputLeafOp::Value)
1678            | Input::Request(InputRequest::Auto)
1679            | Input::Request(InputRequest::Value) => {}
1680        }
1681    }
1682    fn output(tree: &Output<OutputRequest, InputRequest>, names: &mut Vec<String>) {
1683        match tree {
1684            Output::Unary {
1685                operation: OutputUnary::CallbackArgument { .. },
1686                child,
1687            } => output(child, names),
1688            Output::None => {}
1689            Output::Arrow { operation, .. } => match *operation {},
1690            Output::Branch {
1691                operation: OutputBranch::Variant | OutputBranch::Arguments,
1692                children,
1693            } => {
1694                for child in children {
1695                    output(child, names);
1696                }
1697            }
1698            Output::Branch {
1699                operation: OutputBranch::Enum { .. },
1700                children: variants,
1701                ..
1702            } => {
1703                for tree in variants.iter() {
1704                    output(tree, names);
1705                }
1706            }
1707            Output::Request(OutputRequest::EnumFields(variants)) => {
1708                for tree in variants.values().flat_map(|v| v.values()) {
1709                    output(tree, names);
1710                }
1711            }
1712            Output::Request(OutputRequest::Conversion(d)) => {
1713                names.extend(d.functions());
1714                representation_types(d, names);
1715            }
1716            Output::Unary {
1717                operation: OutputUnary::Convert { conversion },
1718                child,
1719            } => {
1720                if let prebindgen_flat::ValueSource::Function(f) = &conversion.source {
1721                    names.push(f.clone());
1722                }
1723                output(child, names);
1724            }
1725            Output::Unary {
1726                operation: OutputUnary::Named { .. },
1727                child,
1728                ..
1729            } => output(child, names),
1730            Output::Leaf(OutputLeafOp::Discard) => {}
1731            Output::Unary {
1732                operation: OutputUnary::Getter { function },
1733                child,
1734            }
1735            | Output::Unary {
1736                operation: OutputUnary::Into { function },
1737                child,
1738            } => {
1739                names.push(function.clone());
1740                output(child, names);
1741            }
1742            Output::Request(OutputRequest::ForType { output: child, .. })
1743            | Output::Unary {
1744                operation: OutputUnary::Optional,
1745                child,
1746            }
1747            | Output::Unary {
1748                operation: OutputUnary::Sequence { .. },
1749                child,
1750                ..
1751            }
1752            | Output::Unary {
1753                operation: OutputUnary::Bulk,
1754                child,
1755            } => output(child, names),
1756            Output::Unary {
1757                operation: OutputUnary::Result { error },
1758                child: ok,
1759            } => {
1760                output(ok, names);
1761                if let Some(name) = error {
1762                    names.push(name.clone());
1763                }
1764            }
1765            Output::Branch {
1766                operation: OutputBranch::Fields,
1767                children,
1768            }
1769            | Output::Branch {
1770                operation: OutputBranch::OwnedFields,
1771                children,
1772            }
1773            | Output::Branch {
1774                operation: OutputBranch::Record { .. },
1775                children,
1776                ..
1777            }
1778            | Output::Branch {
1779                operation: OutputBranch::Combine,
1780                children,
1781            } => {
1782                for child in children {
1783                    output(child, names)
1784                }
1785            }
1786            Output::Request(OutputRequest::NamedFields { overrides, .. }) => {
1787                for child in overrides.values() {
1788                    output(child, names)
1789                }
1790            }
1791            Output::Request(OutputRequest::Auto)
1792            | Output::Request(OutputRequest::Value { .. })
1793            | Output::Leaf(OutputLeafOp::Value)
1794            | Output::Leaf(OutputLeafOp::Projected(_)) => {}
1795        }
1796    }
1797    let mut names = vec![];
1798    for tree in inputs {
1799        input(tree, &mut names)
1800    }
1801    for tree in outputs {
1802        output(tree, &mut names)
1803    }
1804    names.extend(errors.iter().map(|(_, name)| name.clone()));
1805    for f in functions {
1806        names.push(f.name.clone());
1807        for tree in f.params.values() {
1808            input(tree, &mut names)
1809        }
1810        if let Some(tree) = &f.ret {
1811            output(tree, &mut names)
1812        }
1813        if let Some(name) = &f.error {
1814            names.push(name.clone())
1815        }
1816    }
1817    names
1818}

1824fn lift_input_bulk(tree: Input) -> Input {
1825    match tree {
1826        Input::Branch {
1827            operation: InputBranch::Enum { value },
1828            children: variants,
1829        } => Input::Branch {
1830            operation: InputBranch::Enum { value },
1831            children: variants.into_iter().map(lift_input_bulk).collect(),
1832        },
1833        Input::Unary {
1834            operation: InputUnary::Convert { conversion },
1835            child: input,
1836        } => Input::Unary {
1837            operation: InputUnary::Convert { conversion },
1838            child: Box::new(lift_input_bulk(*input)),
1839        },
1840        Input::Unary {
1841            operation: InputUnary::Bulk,
1842            child: c,
1843        } => Input::Unary {
1844            operation: InputUnary::Bulk,
1845            child: Box::new(lift_input_bulk(*c)),
1846        },
1847        Input::Unary {
1848            operation: InputUnary::Temporary,
1849            child: c,
1850        } => Input::Unary {
1851            operation: InputUnary::Temporary,
1852            child: Box::new(lift_input_bulk(*c)),
1853        },
1854        Input::Unary {
1855            operation: InputUnary::Sequence,
1856            child: c,
1857        } => {
1858            let child = lift_input_bulk(*c);
1859            if input_has_bulk(&child) {
1860                Input::Unary {
1861                    operation: InputUnary::Bulk,
1862                    child: Box::new(Input::Unary {
1863                        operation: InputUnary::Sequence,
1864                        child: Box::new(child),
1865                    }),
1866                }
1867            } else {
1868                Input::Unary {
1869                    operation: InputUnary::Sequence,
1870                    child: Box::new(child),
1871                }
1872            }
1873        }
1874        Input::Unary {
1875            operation: InputUnary::Optional,
1876            child: c,
1877        } => match lift_input_bulk(*c) {
1878            Input::Unary {
1879                operation: InputUnary::Bulk,
1880                child: c,
1881            } => Input::Unary {
1882                operation: InputUnary::Bulk,
1883                child: Box::new(Input::Unary {
1884                    operation: InputUnary::Optional,
1885                    child: c,
1886                }),
1887            },
1888            c => Input::Unary {
1889                operation: InputUnary::Optional,
1890                child: Box::new(c),
1891            },
1892        },
1893        Input::Branch {
1894            operation: InputBranch::Fields,
1895            children: c,
1896        } => Input::Branch {
1897            operation: InputBranch::Fields,
1898            children: c.into_iter().map(lift_input_bulk).collect(),
1899        },
1900        Input::Branch {
1901            operation: InputBranch::Record,
1902            children: c,
1903        } => Input::Branch {
1904            operation: InputBranch::Record,
1905            children: c.into_iter().map(lift_input_bulk).collect(),
1906        },
1907        Input::Branch {
1908            operation: InputBranch::Selector,
1909            children: c,
1910        } => Input::Branch {
1911            operation: InputBranch::Selector,
1912            children: c.into_iter().map(lift_input_bulk).collect(),
1913        },
1914        Input::Branch {
1915            operation: InputBranch::Constructor { function },
1916            children: args,
1917        } => Input::Branch {
1918            operation: InputBranch::Constructor { function },
1919            children: args.into_iter().map(lift_input_bulk).collect(),
1920        },
1921        Input::Branch {
1922            operation,
1923            children,
1924        } => Input::Branch {
1925            operation,
1926            children: children.into_iter().map(lift_input_bulk).collect(),
1927        },
1928        c => c,
1929    }
