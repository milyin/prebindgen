1//! Declaration language and composition for preparing conversion slots.
2//!
3//! [`Declarations::resolve`] produces a request-free [`Flat<Resolved<C>>`](Resolved)
4//! that owns all data needed by [`compile_resolved`]. Input and output trees keep
5//! separate operation vocabularies; [`InputNode::Arrow`] and [`OutputNode::Arrow`]
6//! expose reverse-direction domains structurally. [`Declarations::fill`],
7//! [`bind_input`] and [`bind_output`] use the same resolver and compiler for
8//! individual occurrences. Compiled slots retain wires and Rust code for emission.
9//! Generation is owned by the language generator;
10//! this crate owns only declaration resolution and slot preparation.

44// Sized shared outputs use the referent's ordinary recipe, with borrowing
45// carried by the compiler. Keep unsized string/slice views as their own types:
46// adapters and sequence transport already understand those reference shapes.
47fn output_referent(ty: &Type) -> Option<&Type> {
48    match ty {
49        Type::Reference(r)
50            if r.mutability.is_none()
51                && !matches!(r.elem.as_ref(), Type::Slice(_))
52                && !matches!(r.elem.as_ref(), Type::Path(p) if p.path.is_ident("str")) =>
53        {
54            Some(&r.elem)
55        }
56        _ => None,
57    }
58}
59// The owned value deliberately constructed by InputUnary::Temporary. A Vec/String
60// descriptor must never be fabricated over memory merely borrowed from C.
61fn temporary_input_type(ty: &Type) -> Result<Type> {
62    let Type::Reference(reference) = ty else {
63        return Err("reconstruct() requires a shared-reference input; apply it to the optional payload if necessary".into());
64    };
65    if reference.mutability.is_some()
66        || reference
67            .lifetime
68            .as_ref()
69            .is_some_and(|l| l.ident == "static")
70    {
71        return Err("reconstruct() cannot supply a mutable or 'static reference".into());
72    }
73    Ok(match reference.elem.as_ref() {
74        Type::Slice(slice) => {
75            let element = &slice.elem;
76            syn::parse_quote!(Vec<#element>)
77        }
78        Type::Path(path) if path.path.is_ident("str") => syn::parse_quote!(String),
79        other => other.clone(),
80    })
81}

117pub enum InputRequest {
118    /// Expand enum fields, inheriting defaults for fields not explicitly overridden.
119    EnumFields(
120        std::collections::BTreeMap<
121            String,
122            std::collections::BTreeMap<String, Input<InputRequest, OutputRequest>>,
123        >,
124    ),
125    /// Resolve only the canonical value form, independently of boundary defaults.
126    Value,
127    /// Select a local single-value conversion.

129    /// Select the configured default for this source occurrence.
130    Auto,
131    /// Select or override a callback signature's argument and result conversions.
132    Callback(Box<CallbackDecl>),
133    /// Apply a type-specific override, including inside optional/sequence sources.
134    ForType {
135        /// Source type targeted by the override.
136        ty: Box<Type>,

139    },
140    /// Infer a constructor's argument conversions from its source signature.
141    Call(String),

144impl Input<InputRequest, OutputRequest> {
145    /// Expand the tag and selected fields. Keys are source variant and field
146    /// names (or tuple indices). Unspecified fields inherit boundary defaults.
147    pub fn enum_fields<'a>(
148        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
149    ) -> Self {
150        Input::Request(InputRequest::EnumFields(variants::overrides(overrides)))
151    }
152    /// Request the canonical value representation, bypassing boundary defaults.
153    pub fn value() -> Self {
154        Input::Request(InputRequest::Value)
155    }
156    /// Request this source type's configured default during preparation.
157    pub fn auto() -> Self {
158        Input::Request(InputRequest::Auto)
159    }
160    /// Associate a conversion with a source type for a default or nested override.
161    pub fn for_type(ty: Type, input: Self) -> Self {
162        Input::Request(InputRequest::ForType {
163            ty: Box::new(ty),
164            input: Box::new(input),
165        })
166    }
167    /// Infer a source constructor's argument conversions from registered defaults.
168    pub fn call(name: impl Into<String>) -> Self {
169        Input::Request(InputRequest::Call(name.into()))
170    }
171    /// Explicitly reconstruct an owned temporary for a shared-reference input.
172    /// Slices construct a Vec, str constructs a String, and sized references
173    /// construct their referent. Child recipes keep their ordinary ownership
174    /// semantics; this operation authorizes the temporary's allocation/copying.
175    pub fn reconstruct(self) -> Self {
176        match self {
177            Input::Request(InputRequest::ForType { ty, input }) => Input::for_type(
178                *ty,
179                Input::Unary {
180                    operation: InputUnary::Temporary,
181                    child: input,
182                },
183            ),
184            other => Input::Unary {
185                operation: InputUnary::Temporary,
186                child: Box::new(other),
187            },
188        }
189    }
190    /// Add a constructor alternative to this conversion tree.
191    pub fn variant(self, name: impl Into<String>) -> Self {
192        self.add_alternative(Input::call(name))
193    }
194    /// Add a direct codec alternative, bypassing this type's expansion default.
195    pub fn variant_self(self) -> Self {
196        self.add_alternative(Input::Leaf(InputLeafOp::Value))
197    }
198    fn add_alternative(self, arm: Self) -> Self {
199        match self {
200            Input::Request(InputRequest::ForType { ty, input }) => {
201                Input::for_type(*ty, input.add_alternative(arm))
202            }
203            Input::Branch {
204                operation: InputBranch::Selector,
205                children: mut arms,
206            } => {
207                arms.push(arm);
208                Input::Branch {
209                    operation: InputBranch::Selector,
210                    children: arms,
211                }
212            }
213            other => Input::Branch {
214                operation: InputBranch::Selector,
215                children: vec![other, arm],
216            },
217        }
218    }
219}
220impl<P, Q> Input<P, Q> {
221    /// Group this explicit expansion into a bulk carrier.
222    pub fn bulk(self) -> Self {
223        Input::Unary {
224            operation: InputUnary::Bulk,
225            child: Box::new(self),
226        }
227    }
228    /// Construct a record from field recipes, or unit from an empty list.
229    ///
230    /// `fields` holds one recipe per field of the source record, in declaration
231    /// order. Binding verifies the count against the record.
232    pub fn fields(fields: impl IntoIterator<Item = Input<P, Q>>) -> Self {
233        Input::Branch {
234            operation: InputBranch::Fields,
235            children: fields.into_iter().collect(),
236        }
237    }
238    /// Call a named source constructor with the supplied argument recipes.
239    ///
240    /// Binding checks its result type and argument count. Argument validation and
241    /// ownership acquisition precede the source call.
242    ///
243    /// # Arguments
244    ///
245    /// * `name` — unqualified name of a source function whose result type is the
246    ///   input's type.
247    /// * `args` — one recipe per parameter of that function, in order.
248    pub fn constructor(
249        name: impl Into<String>,
250        args: impl IntoIterator<Item = Input<P, Q>>,
251    ) -> Self {
252        Input::Branch {
253            operation: InputBranch::Constructor {
254                function: name.into(),
255            },
256            children: args.into_iter().collect(),
257        }
258    }
259    /// Choose between owned representations of the same input type.
260    ///
261    /// `arms` holds one recipe per alternative; each must produce the input's
262    /// type. The caller passes the zero-based index of the chosen alternative in an
263    /// extra wire ahead of all arms' wires.
264    ///
265    /// Requires at least one arm. Each arm retains its own wires; this does not
266    /// merge equivalent wire parameters or generate destination-language overloads.
267    pub fn selector(arms: impl IntoIterator<Item = Input<P, Q>>) -> Self {
268        Input::Branch {
269            operation: InputBranch::Selector,
270            children: arms.into_iter().collect(),
271        }
272    }

309impl Output<OutputRequest, InputRequest> {
310    /// Expand an enum's tag and active fields using source variant/member keys.
311    pub fn enum_fields<'a>(
312        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
313    ) -> Self {
314        Output::Request(OutputRequest::EnumFields(variants::overrides(overrides)))
315    }
316    /// Request the canonical value representation, bypassing boundary defaults.
317    pub fn value() -> Self {
318        Output::Request(OutputRequest::Value { projection: None })
319    }
320    /// Request this source type's configured output default during preparation.
321    pub fn auto() -> Self {
322        Output::Request(OutputRequest::Auto)
323    }
324    /// Associate a conversion with a source type for a default or nested override.
325    pub fn for_type(ty: Type, output: Self) -> Self {
326        Output::Request(OutputRequest::ForType {
327            ty: Box::new(ty),
328            output: Box::new(output),
329        })
330    }
331    /// Append another simultaneous output view.
332    pub fn view(self, view: impl Into<Self>) -> Self {
333        let view = view.into();
334        match self {
335            Output::Request(OutputRequest::ForType { ty, output }) => {
336                Output::for_type(*ty, output.view(view))
337            }
338            Output::Branch {
339                operation: OutputBranch::Combine,
340                children: mut parts,
341            } => {
342                parts.push(view);
343                Output::Branch {
344                    operation: OutputBranch::Combine,
345                    children: parts,
346                }
347            }
348            other => Output::Branch {
349                operation: OutputBranch::Combine,
350                children: vec![other, view],
351            },
352        }
353    }
354    /// Append a source getter whose result uses its registered conversion default.
355    pub fn field(self, name: impl Into<String>) -> Self {
356        self.view(Output::getter(name, Output::auto()))
357    }
358    /// Append the direct codec representation, bypassing this type's expansion default.
359    pub fn field_self(self) -> Self {
360        self.view(Output::Leaf(OutputLeafOp::Value))
361    }
362    /// Append an explicit borrowed projection.
363    pub fn field_self_ref(self) -> Self {
364        self.project_self(Projection::borrowed())
365    }
366    /// Append an explicit owned projection.
367    pub fn field_self_into(self) -> Self {
368        self.project_self(Projection::owned())
369    }
370    /// Append a projection with explicit ownership and nested field policies.
371    pub fn project_self(self, policy: Projection) -> Self {
372        self.view(Output::Leaf(OutputLeafOp::Projected(policy)))
373    }
374    /// Append a consuming accessor followed by owned field decomposition.
375    pub fn fields_into(self, view: Self) -> Self {
376        let Output::Unary {
377            operation: OutputUnary::Getter { function },
378            child: output,
379        } = view
380        else {
381            panic!("fields_into requires a fields! accessor")
382        };
383        let Output::Request(OutputRequest::NamedFields { overrides, .. }) = *output else {
384            panic!("fields_into requires named fields")
385        };
386        self.view(Output::Unary {
387            operation: OutputUnary::Into { function },
388            child: Box::new(Output::Request(OutputRequest::NamedFields {
389                overrides,
390                mode: ProjectionMode::Owned,
391            })),
392        })
393    }
394    /// Deprecated spelling of [`Output::fields_into`].
395    #[deprecated(note = "use fields_into instead")]
396    pub fn fields_self_into(self, view: Self) -> Self {
397        self.fields_into(view)
398    }
399    /// Override one named source field, preserving defaults for its siblings.
400    pub fn field_override(mut self, name: impl Into<String>, output: Self) -> Self {
401        match &mut self {
402            Output::Request(OutputRequest::NamedFields { overrides, .. }) => {
403                assert!(
404                    overrides.insert(name.into(), output).is_none(),
405                    "duplicate field override"
406                );
407            }
408            Output::Unary {
409                operation: OutputUnary::Getter { .. },
410                child: inner,
411                ..
412            }
413            | Output::Unary {
414                operation: OutputUnary::Into { .. },
415                child: inner,
416                ..
417            }
418            | Output::Request(OutputRequest::ForType { output: inner, .. }) => {
419                let old = std::mem::replace(inner, Box::new(Output::auto()));
420                **inner = old.field_override(name, output);
421            }
422            _ => panic!("field override requires named field decomposition"),
423        }
424        self
425    }
426}

475impl Output {
476    fn consumes<C: Codec>(
477        &self,
478        flat: &prebindgen_flat::Flat<C::Schema>,
479        ty: &Type,
480        extra: &C::Extra,
481        codec: &C,
482        borrowed: bool,
483    ) -> bool {
484        match self {
485            Output::Unary {
486                operation: OutputUnary::Convert { conversion, .. },
487                ..
488            } => match &conversion.source {
489                prebindgen_flat::ValueSource::Function(name) => {
490                    flat.function(name).is_ok_and(|f| {
491                        f.params
492                            .first()
493                            .is_some_and(|p| !matches!(p.ty.ty, Type::Reference(_)))
494                    })
495                }
496                prebindgen_flat::ValueSource::Into { .. } => true,
497            },
498            Output::Unary {
499                operation: OutputUnary::Sequence { mode, .. },
500                ..
501            } => match mode {
502                Some(mode) => *mode == ProjectionMode::Owned,
503                None => !borrowed && sequence_parts(ty).is_some_and(|(_, view)| view.is_owned()),
504            },
505            Output::Unary {
506                operation: OutputUnary::Bulk,
507                child,
508            } => child.consumes(flat, ty, extra, codec, borrowed),
509            Output::Unary {
510                operation: OutputUnary::Named { .. },
511                child,
512                ..
513            } => child.consumes(flat, ty, extra, codec, borrowed),
514            Output::Unary {
515                operation: OutputUnary::Optional,
516                child,
517            } => child.consumes(flat, option_inner(ty).unwrap_or(ty), extra, codec, borrowed),
518            Output::Leaf(OutputLeafOp::Value) => {
519                !borrowed && output_referent(ty).is_none() && codec.output_consumes(flat, ty, extra)
520            }
521            Output::Leaf(OutputLeafOp::Projected(p)) => p.mode == ProjectionMode::Owned,
522            Output::Branch {
523                operation: OutputBranch::OwnedFields,
524                children: _,
525            }
526            | Output::Unary {
527                operation: OutputUnary::Into { .. },
528                ..
529            }
530            | Output::Unary {
531                operation: OutputUnary::Result { .. },
532                ..
533            } => true,
534            Output::Branch {
535                operation: OutputBranch::Record { mode, .. },
536                ..
537            } => mode.map_or(!borrowed && output_referent(ty).is_none(), |m| {
538                m == ProjectionMode::Owned
539            }),
540            Output::Branch {
541                operation: OutputBranch::Enum { mode, .. },
542                ..
543            } => mode.map_or(!borrowed && output_referent(ty).is_none(), |m| {
544                m == ProjectionMode::Owned
545            }),
546            Output::Request(never) => match *never {},
547            Output::Branch {
548                operation: OutputBranch::Combine,
549                children: p,
550            } => p
551                .iter()
552                .any(|p| p.consumes(flat, ty, extra, codec, borrowed)),
553            _ => false,
554        }
555    }
556}

563/// A flat model whose slots contain only conversion data for wrapper emission.
564///
565/// Unbound source helpers have `None` slots. Bound functions need `Some` for their
566/// result and every parameter; [`prebindgen_flat::validate_bindings`] checks this before emission.
567/// Record fields retain their canonical source information and metadata. Descriptions and [`Codec::Extra`]
568/// are construction inputs, not stored in this model.
569///
570/// Output conversion data cannot be attached to an input position:
571/// ```compile_fail
572/// use prebindgen_declarations::Bindings;
573/// fn swap(flat: &mut Bindings<syn::Type>) {
574///     let function = flat.function_mut("foo").unwrap();
575///     function.params[0].ty.slot = function.ret.slot.clone();
576/// }
577/// ```
578/// Nor can a recipe be stored in a slot:
579/// ```compile_fail
580/// use prebindgen_declarations::{Bindings, Input};
581/// fn configure(flat: &mut Bindings<syn::Type>) {
582///     flat.function_mut("foo").unwrap().params[0].ty.slot = Some(Input::Leaf(prebindgen_declarations::InputLeafOp::Value));
583/// }
584/// ```
585pub type Bindings<W, C = (), N = (), B = ()> = Flat<prebindgen_conversions::PlanSchema<W, C, N, B>>;

587/// Compile a request-free description for one named source parameter.
588/// The model keeps its fixed schema; this returns a converter without modifying its slot.
589/// ```
590/// use prebindgen_declarations::{compile_input,Codec,Input};
591/// use prebindgen_flat::Flat;
592/// fn resolved<C:Codec>(flat:&Flat<C::Schema>,codec:&C,extra:&C::Extra) {
593///     compile_input(flat,codec,"f","x",&Input::Leaf(prebindgen_declarations::InputLeafOp::Value),extra).unwrap();
594/// }
595/// ```
596/// ```compile_fail
597/// use prebindgen_declarations::{compile_input,Codec,Input};
598/// use prebindgen_flat::Flat;
599/// fn pending<C:Codec>(flat:&Flat<C::Schema>,codec:&C,extra:&C::Extra) {
600///     compile_input(flat,codec,"f","x",&Input::auto(),extra).unwrap();
601/// }
602/// ```
603pub fn compile_input<C: Codec>(
604    flat: &Flat<C::Schema>,
605    codec: &C,
606    function: &str,
607    parameter: &str,
608    recipe: &Input,
609    extra: &C::Extra,
610) -> Result<<C::Schema as Schema>::Input> {
611    let f = flat.function(function)?;
612    let index = f
613        .params
614        .iter()
615        .position(|p| p.name == parameter)
616        .ok_or_else(|| format!("`{function}`: unknown parameter `{parameter}`"))?;
617    let mut plan = Compiler::new(flat, codec, format!("p{index}")).input(
618        &f.params[index].ty.ty,
619        recipe,
620        extra,
621        &Condition::default(),
622    )?;
623    codec.finish_input(&mut plan, extra)?;
624    codec.validate_input(&plan)?;
625    Ok(plan.into())
626}

712pub fn compile_callback_argument<C: Codec>(
713    flat: &Flat<C::Schema>,
714    codec: &C,
715    ty: &Type,
716    recipe: &Output,
717    index: usize,
718    borrowed: bool,
719    extra: &C::Extra,
720) -> Result<<C::Schema as Schema>::Output> {
721    compile_output_at(
722        flat,
723        codec,
724        ty,
725        recipe,
726        extra,

728    )

731enum OutputPosition {

733    CallbackArgument { index: usize, borrowed: bool },

736fn compile_output_at<C: Codec>(
737    flat: &Flat<C::Schema>,
738    codec: &C,
739    ty: &Type,
740    recipe: &Output,
741    extra: &C::Extra,
742    position: OutputPosition,
743) -> Result<<C::Schema as Schema>::Output> {
744    let value = prebindgen_conversions::source_result_expr();
745    let mut plan = match position {
746        OutputPosition::Return => Compiler::new(flat, codec, "ret".into())
747            .output_root(ty, recipe, extra, value, false, true)?,

753    finalize_output_schema(codec, &mut plan, extra)?;
754    codec.finish_output(&mut plan, extra)?;
755    codec.validate_output(&plan)?;
756    Ok(plan.into())

759/// Build and attach a converter to a named function parameter.
760///
761/// Resolves the occurrence's source type from `flat`, checks recipe compatibility,
762/// and recursively composes wires and code. Nested conversions use the supplied
763/// `extra`, not data from referenced constructors or getters. Recipes and `extra`
764/// are borrowed only for this call. A failure leaves the existing slot unchanged.
765/// Rebinding an occurrence replaces its converter without changing other slots.
766/// Generated names are scoped by parameter position, so separate calls cannot
767/// collide within the wrapper. Keep the function signature fixed after binding.
768/// Check completeness with [`prebindgen_flat::validate_bindings`].
769///
770/// # Arguments
771///
772/// * `flat` — the model holding the function; the parameter's slot is replaced.
773/// * `codec` — the adapter's leaf conversions.
774/// * `function` — unqualified name of the source function.
775/// * `parameter` — name of the parameter, as written in the source.
776/// * `recipe` — preparation tree describing how to build the argument from boundary inputs.
777///   To bind an already resolved [`Input`], pass `&resolved.into_preparation()`.
778/// * `extra` — occurrence data passed to every codec call for this conversion.
779///
780/// Input binding rejects output recipes:
781/// ```compile_fail
782/// use prebindgen_declarations::{bind_input, Codec, Output};
783/// use prebindgen_flat::Flat;
784/// fn wrong<C: Codec>(flat: &mut Flat<C::Schema>, codec: &C, extra: &C::Extra) {
785///     bind_input(flat, codec, "foo", "x", &Output::Leaf(prebindgen_declarations::OutputLeafOp::Value), extra).unwrap();
786/// }
787/// ```
788pub fn bind_input<C: Codec>(
789    flat: &mut Flat<C::Schema>,
790    codec: &C,
791    function: &str,
792    parameter: &str,
793    recipe: &Input<InputRequest, OutputRequest>,
794    extra: &C::Extra,
795) -> Result<()> {
796    let f = flat.function(function)?;
797    let index = f
798        .params
799        .iter()
800        .position(|p| p.name == parameter)
801        .ok_or_else(|| format!("`{function}`: unknown parameter `{parameter}`"))?;
802    let defaults = Declarations::new(flat, vec![], vec![], vec![])?;
803    let recipe = defaults.resolve_input(flat, codec, &f.params[index].ty.ty, recipe, extra)?;
804    let conversion = compile_input(flat, codec, function, parameter, &recipe, extra)?;
805    flat.function_mut(function)?.params[index].ty.slot = Some(conversion);
806    Ok(())
807}

809/// Build and attach a converter to a function result, replacing any existing one.
810///
811/// Uses the same construction and failure rules as [`bind_input`]. Output code
812/// uses [`prebindgen_conversions::source_result_expr`] for the source call's result local.
813/// The recipe is checked for compatible types and signatures, and for repeated
814/// consumption of the same value. Unit results use `Output::fields([])`.
815///
816/// # Arguments
817///
818/// * `flat` — the model holding the function; the result's slot is replaced.
819/// * `codec` — the adapter's leaf conversions.
820/// * `function` — unqualified name of the source function.
821/// * `recipe` — preparation tree describing how to deliver the result as boundary outputs.
822///   To bind an already resolved [`Output`], pass `&resolved.into_preparation()`.
823/// * `extra` — occurrence data passed to every codec call for this conversion.
824///
825/// Output binding rejects input recipes:
826/// ```compile_fail
827/// use prebindgen_declarations::{bind_output, Codec, Input};
828/// use prebindgen_flat::Flat;
829/// fn wrong<C: Codec>(flat: &mut Flat<C::Schema>, codec: &C, extra: &C::Extra) {
830///     bind_output(flat, codec, "foo", &Input::Leaf(prebindgen_declarations::InputLeafOp::Value), extra).unwrap();
831/// }
832/// ```
833pub fn bind_output<C: Codec>(
834    flat: &mut Flat<C::Schema>,
835    codec: &C,
836    function: &str,
837    recipe: &Output<OutputRequest, InputRequest>,
838    extra: &C::Extra,
839) -> Result<()> {
840    let defaults = Declarations::new(flat, vec![], vec![], vec![])?;
841    let recipe =
842        defaults.resolve_output(flat, codec, &flat.function(function)?.ret.ty, recipe, extra)?;
843    let conversion = compile_output(flat, codec, function, &recipe, extra)?;
844    flat.function_mut(function)?.ret.slot = Some(conversion);
845    Ok(())
846}

848fn finalize_output_schema<C: Codec>(
849    codec: &C,
850    plan: &mut CodecOutput<C>,
851    extra: &C::Extra,
852) -> Result<()> {
853    if let Some(error) = codec.binding_failure_layout(extra)? {
854        plan.outcomes.binding_failure = Some(error);
855    }
856    codec.validate_output(plan)
857}

859struct Compiler<'a, C: Codec> {
860    input_role: std::cell::Cell<InputRole>,
861    bulk: std::cell::Cell<bool>,
862    // Explicit borrowed collection/record projections authorize owning copies
863    // of borrowed leaves, without changing conversions of owned getter results.
864    copy_borrowed: std::cell::Cell<bool>,
865    // Row temporaries may borrow only the enclosing sequence's cached frame,
866    // never a separate bulk field decoded inside an ordinary JVM object row.
867    ordinary_collections: std::cell::Cell<bool>,
868    // A Combine observation borrows a Rust temporary, not persistent foreign storage.
869    temporary_receiver: std::cell::Cell<bool>,
870    flat: &'a Flat<C::Schema>,
871    codec: &'a C,
872    prefix: String,
873    next: std::cell::Cell<usize>,
874}
875impl<'a, C: Codec> Compiler<'a, C> {
876    fn new(flat: &'a Flat<C::Schema>, codec: &'a C, prefix: String) -> Self {
877        Self {
878            input_role: std::cell::Cell::new(InputRole::Parameter),
879            bulk: std::cell::Cell::new(false),
880            copy_borrowed: std::cell::Cell::new(false),
881            ordinary_collections: std::cell::Cell::new(false),
882            temporary_receiver: std::cell::Cell::new(false),
883            flat,
884            codec,
885            prefix,
886            next: std::cell::Cell::new(0),
887        }
888    }
889    fn with_temporary_receiver<T>(
890        &self,
891        temporary: bool,
892        prepare: impl FnOnce() -> Result<T>,
893    ) -> Result<T> {
894        let previous = self.temporary_receiver.get();
895        self.temporary_receiver.set(previous || temporary);
896        let result = prepare();
897        self.temporary_receiver.set(previous);
898        result
899    }
900    fn with_copy_permission<T>(
901        &self,
902        explicit: bool,
903        prepare: impl FnOnce() -> Result<T>,
904    ) -> Result<T> {
905        let previous = self
906            .copy_borrowed
907            .replace(self.copy_borrowed.get() || explicit);
908        let result = prepare();
909        self.copy_borrowed.set(previous);
910        result
911    }
912    fn name(&self) -> syn::Ident {
913        let next = self.next.get();
914        let n = format_ident!("{}_w{next}", self.prefix);
915        self.next.set(next + 1);
916        n
917    }
918    #[allow(clippy::type_complexity)] // Preserve the codec metadata types while walking its tree.
919    fn prefix(
920        &self,
921        wires: &mut prebindgen_flat::WireNode<C::Wire, C::Chain, C::Branch>,
922        segment: ComponentSegment,
923    ) {
924        self.codec.prefix_layout(wires, segment);
925    }
926    fn input(
927        &self,
928        ty: &Type,
929        recipe: &Input,
930        extra: &C::Extra,
931        active: &Condition,
932    ) -> Result<CodecInput<C>> {
933        self.input_named(ty, recipe, extra, active, None)
934    }
935    fn input_named(
936        &self,
937        ty: &Type,
938        recipe: &Input,
939        extra: &C::Extra,
940        active: &Condition,
941        names: Option<(&syn::Ident, &syn::Ident)>,
942    ) -> Result<CodecInput<C>> {
943        self.input_unprepared_named(ty, recipe, extra, active, names)
944    }
945
946    fn input_unprepared_named(
947        &self,
948        ty: &Type,
949        recipe: &Input,
950        extra: &C::Extra,
951        active: &Condition,
952        names: Option<(&syn::Ident, &syn::Ident)>,
953    ) -> Result<CodecInput<C>> {
954        if prebindgen_flat::callback_signature(ty)?.is_some()
955            && !matches!(
956                recipe,
957                Input::Leaf(InputLeafOp::Value) | Input::Arrow { .. }
958            )
959        {
960            return Err("source callback parameters only support leaf inputs".into());
961        }
962        let var = names
963            .map(|(_, var)| var.clone())
964            .unwrap_or_else(|| self.name());
965        match recipe {
966            Input::None => {
967                if type_key(ty) != "()" {
968                    return Err("None input requires unit".into());
969                }
970                Ok(InputPlan::new(
971                    WireNode::Branch {
972                        controls: vec![],
973                        children: vec![],
974                        meta: self.codec.group(GroupKind::Fields)?,
975                    },
976                    quote!(),
977                    quote!(()),
978                ))
979            }
980            Input::Branch {
981                operation: InputBranch::Variant,
982                ..
983            } => Err("variant branch requires an enum parent".into()),
984            Input::Branch {
985                operation: InputBranch::Enum { value },
986                children: variants,
987            } => self.enum_input(ty, variants, *value, extra, active, names),
988            Input::Unary {
989                operation: InputUnary::Convert { conversion },
990                child: input,
991            } => self.input_conversion(ty, conversion, input, extra, active, names),
992            Input::Unary {
993                operation: InputUnary::Temporary,
994                child: inner,
995            } => {
996                let owned = temporary_input_type(ty)?;
997                let mut child = self.input_named(&owned, inner, extra, active, names)?;
998                let storage = self.name();
999                let target = format_ident!("{storage}_owned");
1000                let mut backing = std::mem::take(&mut child.storage);
1001                backing.extend(quote!(#[allow(unused_mut,unused_assignments)] let mut #storage:Option<#owned>=None;));
1002                let build = child.bind(&target, quote!(#storage=Some(#target);))?;
1003                let Type::Reference(reference) = ty else {
1004                    unreachable!()
1005                };
1006                let elem = &reference.elem;
1007                let value = if matches!(elem.as_ref(), Type::Slice(_))
1008                    || matches!(elem.as_ref(),Type::Path(p) if p.path.is_ident("str"))
1009                {
1010                    quote!(::core::convert::AsRef::<#elem>::as_ref(#storage.as_ref().expect("active temporary")))
1011                } else {
1012                    quote!(#storage.as_ref().expect("active temporary"))
1013                };
1014                let mut plan = InputPlan::new(
1015                    WireNode::Branch {
1016                        controls: vec![],
1017                        children: vec![child.node],
1018                        meta: self.codec.group(GroupKind::Values)?,
1019                    },
1020                    quote!(#build;),
1021                    value,
1022                );
1023                plan.storage = backing;
1024                plan.fallible = child.fallible;
1025                Ok(plan)
1026            }
1027            Input::Unary {
1028                operation: InputUnary::Bulk,
1029                child: inner,
1030            } => {
1031                require_bulk(self.codec, Direction::Input)?;
1032                if self.bulk.get() {
1033                    return self.input_named(ty, inner, extra, active, names);
1034                }
1035                let wire = self.name();
1036                self.bulk.set(true);
1037                let child = self.input_named(ty, inner, extra, active, names);
1038                self.bulk.set(false);
1039                self.codec
1040                    .bulk_input(self.flat, ty, extra, wire, child?, active)
1041            }
1042            Input::Arrow {
1043                operation: callback,
1044                domain,
1045                codomain,
1046            } => {
1047                callback.validate_for(self.flat, ty)?;
1048                let signature = prebindgen_flat::callback_signature(ty)?
1049                    .ok_or("arrow requires a callable source")?;
1050                if domain.argument_nodes()?.len() != signature.arguments.len() {
1051                    return Err("wrong arrow domain arity".into());
1052                }
1053                self.codec.input_arrow(InputArrow {
1054                    flat: self.flat,
1055                    operation: callback,
1056                    domain,
1057                    codomain,
1058                    extra,
1059                    wire: names
1060                        .map(|(wire, _)| wire.clone())
1061                        .unwrap_or_else(|| self.name()),
1062                    target: var,
1063                    active,
1064                })
1065            }
1066            Input::Unary {
1067                operation: InputUnary::Sequence,
1068                child: element_recipe,
1069            } => {
1070                if matches!(ty, Type::Reference(_)) {
1071                    let next = self.next.get();
1072                    let wire = names
1073                        .map(|(wire, _)| wire.clone())
1074                        .unwrap_or_else(|| self.name());
1075                    if let Some(plan) = self.codec.input_borrowed_sequence(
1076                        self.flat,
1077                        ty,
1078                        extra,
1079                        wire,
1080                        var.clone(),
1081                        element_recipe,
1082                        active,
1083                    )? {
1084                        return Ok(plan);
1085                    }
1086                    self.next.set(next);
1087                }
1088                if !self.bulk.get() {
1089                    if let Input::Unary {
1090                        operation: InputUnary::Bulk,
1091                        child: inner,
1092                    } = element_recipe.as_ref()
1093                    {
1094                        return self.input_named(
1095                            ty,
1096                            &Input::Unary {
1097                                operation: InputUnary::Bulk,
1098                                child: Box::new(Input::Unary {
1099                                    operation: InputUnary::Sequence,
1100                                    child: inner.clone(),
1101                                }),
1102                            },
1103                            extra,
1104                            active,
1105                            names,
1106                        );
1107                    }
1108                }
1109                let (element, _) =
1110                    sequence_parts(ty).ok_or("sequence input requires Vec<T>, &[T] or &Vec<T>")?;
1111                let wire = self.name();
1112                let length = self.name();
1113                let index = self.name();
1114                let child = self.input(element, element_recipe, extra, &Condition::default());
1115                let child = child?;
1116                sequence::input(
1117                    self.codec,
1118                    ty,
1119                    extra,
1120                    wire,
1121                    var,
1122                    length,
1123                    index,
1124                    active,
1125                    child,
1126                    self.bulk.get(),
1127                )
1128            }
1129            Input::Unary {
1130                operation: InputUnary::Optional,
1131                child: inner,
1132            } => {
1133                let ty = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
1134                let tag = names
1135                    .map(|(wire, _)| format_ident!("{wire}_present"))
1136                    .unwrap_or_else(|| self.name());
1137                let child = if let Some((wire, var)) = names {
1138                    self.input_named(
1139                        ty,
1140                        inner,
1141                        extra,
1142                        &active.arm(tag.clone(), 1),
1143                        Some((&format_ident!("{wire}_some"), &format_ident!("{var}_some"))),
1144                    )?
1145                } else {
1146                    self.input(ty, inner, extra, &active.arm(tag.clone(), 1))?
1147                };
1148                optional_input(self.codec, extra, tag, var, active, child)
1149            }
1150            Input::Leaf(InputLeafOp::Value) => {
1151                let w = names
1152                    .map(|(wire, _)| wire.clone())
1153                    .unwrap_or_else(|| self.name());
1154                let representation = if self.bulk.get() {
1155                    Representation::Bulk
1156                } else {
1157                    Representation::Ordinary
1158                };
1159                if representation == Representation::Bulk {
1160                    require_bulk(self.codec, Direction::Input)?;
1161                }
1162                self.codec.input_leaf(InputLeaf {
1163                    flat: self.flat,
1164                    ty,
1165                    extra,
1166                    wire: w,
1167                    target: var,
1168                    active,
1169                    representation,
1170                    role: self.input_role.get(),
1171                })
1172            }
1173            Input::Request(never) => match *never {},
1174            Input::Branch {
1175                operation: InputBranch::Fields,
1176                children,
1177            }
1178            | Input::Branch {
1179                operation: InputBranch::Record,
1180                children,
1181            } => {
1182                if type_key(ty) == "()" && children.is_empty() {
1183                    return Ok(InputPlan::new(
1184                        WireNode::Branch {
1185                            controls: vec![],
1186                            children: vec![],
1187                            meta: self.codec.group(GroupKind::Fields)?,
1188                        },
1189                        quote!(),
1190                        quote!(()),
1191                    ));
1192                }
1193                let r = self.flat.record(ty)?;
1194                if r.fields.len() != children.len() {
1195                    return Err(format!(
1196                        "`{}`: expected {} field converters, got {}",
1197                        type_key(ty),
1198                        r.fields.len(),
1199                        children.len()
1200                    ));
1201                }
1202                let grouped = matches!(
1203                    recipe,
1204                    Input::Branch {
1205                        operation: InputBranch::Record,
1206                        children: _
1207                    }
1208                );
1209                let wire =
1210                    grouped.then(|| names.map(|(w, _)| w.clone()).unwrap_or_else(|| self.name()));
1211                let mut plan = record_input(
1212                    self.codec,
1213                    ty,
1214                    var.clone(),
1215                    r.fields
1216                        .iter()
1217                        .zip(children)
1218                        .enumerate()
1219                        .map(|(index, (field, child))| {
1220                            let plan = if let Some(wire) = &wire {
1221                                self.input_named(
1222                                    &field.ty,
1223                                    child,
1224                                    extra,
1225                                    active,
1226                                    Some((
1227                                        &format_ident!("{wire}_f{index}"),
1228                                        &format_ident!("{var}_f{index}"),
1229                                    )),
1230                                )?
1231                            } else {
1232                                self.input(&field.ty, child, extra, active)?
1233                            };
1234                            Ok((field.name.clone(), plan))
1235                        }),
1236                )?;
1237                if let Some(wire) = wire {
1238                    self.codec
1239                        .input_record_metadata(self.flat, ty, &wire, active, &mut plan, extra)?;
1240                }
1241                Ok(plan)
1242            }
1243            Input::Branch {
1244                operation: InputBranch::Constructor { function },
1245                children: args,
1246            } => {
1247                let f = self.flat.function(function)?;
1248                if !same_type(&f.ret.ty, ty) {
1249                    return Err(format!(
1250                        "constructor `{function}` returns `{}`, expected `{}`",
1251                        type_key(&f.ret.ty),
1252                        type_key(ty)
1253                    ));
1254                }
1255                if f.params.len() != args.len() {
1256                    return Err(format!("constructor `{function}`: wrong argument count"));
1257                }
1258                let mut plans = Vec::new();
1259                let mut call_args = Vec::new();
1260                let mut wires = Vec::new();
1261                let mut storage = TokenStream::new();
1262                let mut fallible = false;
1263                for (i, (param, child)) in f.params.iter().zip(args).enumerate() {
1264                    let mut p = self.input(&param.ty.ty, child, extra, active)?;
1265                    self.prefix(
1266                        &mut p.node,
1267                        ComponentSegment::Argument(param.name.to_string()),
1268                    );
1269                    fallible |= p.fallible;
1270                    storage.extend(std::mem::take(&mut p.storage));
1271                    let target = format_ident!("{var}_argument_{i}");
1272                    call_args.push(target.clone());
1273                    plans.push((target, p));
1274                }
1275                let path = &f.path;
1276                let mut plan = InputPlan::new(
1277                    WireNode::default(),
1278                    quote!(let #var:#ty=#path(#(#call_args),*);),
1279                    quote!(#var),
1280                );
1281                for (target, p) in plans.iter().rev() {
1282                    plan.code = p.bind(target, plan.code)?;
1283                }
1284                for (_, p) in plans {
1285                    wires.push(p.node);
1286                }
1287                plan.node = WireNode::Branch {
1288                    controls: vec![],
1289                    children: wires,
1290                    meta: self.codec.group(GroupKind::Values)?,
1291                };
1292                plan.storage = storage;
1293                plan.fallible = fallible;
1294                Ok(plan)
1295            }
1296            Input::Branch {
1297                operation: InputBranch::Selector,
1298                children: arms,
1299            } => {
1300                let mut fallible = self.codec.validates_inputs();
1301                let mut storage = TokenStream::new();
1302                let mut validation = TokenStream::new();
1303                if arms.is_empty() {
1304                    return Err("an input selector needs at least one alternative".into());
1305                }
1306                if matches!(ty, Type::Reference(_)) {
1307                    return Err("selector results must be owned in this subset".into());
1308                }
1309                let tag = self.name();
1310                let mut selector_layout = prebindgen_flat::WireNode::Leaf(vec![Wire {
1311                    name: tag.clone(),
1312                    kind: self.codec.selector(extra)?,
1313                }]);
1314                self.prefix(&mut selector_layout, ComponentSegment::Selector);
1315                let WireNode::Leaf(mut selector_values) = selector_layout else {
1316                    return Err("selector prefixing must retain its value".into());
1317                };
1318                if selector_values.len() != 1 {
1319                    return Err("selector prefixing must retain its value".into());
1320                }
1321                let selector = selector_values.remove(0);
1322                let mut alternatives = Vec::new();
1323                let count = arms.len() as i32;
1324                let invalid = active.and(quote!(!(0..#count).contains(&#tag)));
1325                if self.codec.validates_inputs() {
1326                    validation.extend(quote!(if #invalid {return Err("invalid selector".into());}));
1327                }
1328                let mut cases = Vec::new();
1329                for (i, arm) in arms.iter().enumerate() {
1330                    let mut p = self.input(ty, arm, extra, &active.arm(tag.clone(), i as i32))?;
1331                    self.prefix(&mut p.node, ComponentSegment::Alternative(i));
1332                    fallible |= p.fallible;
1333                    storage.extend(std::mem::take(&mut p.storage));
1334                    let target = format_ident!("{var}_alternative_{i}");
1335                    let body = p.bind(&target, quote!(#target))?;
1336                    alternatives.push(p.node);
1337                    let i = i as i32;
1338                    cases.push(quote!(#i=>#body));
1339                }
1340                let metadata = self.codec.alternatives(Alternatives {
1341                    selector: &selector,
1342                    arms: &mut alternatives,
1343                })?;
1344                let node = WireNode::Branch {
1345                    controls: vec![selector],
1346                    meta: metadata,
1347                    children: alternatives,
1348                };
1349
1350                let invalid = if self.codec.validates_inputs() {
1351                    quote!(unreachable!("validated selector"))
1352                } else {
1353                    quote!(unsafe { ::core::hint::unreachable_unchecked() })
1354                };
1355                let mut plan = InputPlan::new(
1356                    node,
1357                    quote!(#validation let #var:#ty=match #tag {#(#cases,)* _=>#invalid};),
1358                    quote!(#var),
1359                );
1360                plan.storage = storage;
1361                plan.fallible = fallible;
1362                Ok(plan)
1363            }
1364        }
1365    }

1367    fn output_root(
1368        &self,
1369        ty: &Type,
1370        recipe: &Output,
1371        extra: &C::Extra,
1372        value: TokenStream,
1373        borrowed: bool,
1374        reference_return: bool,
1375    ) -> Result<CodecOutput<C>> {
1376        if let Output::Unary {
1377            operation: OutputUnary::Result { error },
1378            child: ok,
1379        } = recipe
1380        {
1381            if borrowed {
1382                return Err("Result error routing requires an owned Result occurrence".into());
1383            }
1384            let (success_ty, error_ty) =
1385                result_parts(ty).ok_or("Result recipe requires Result<T, E>")?;
1386            let payload = self.name();
1387            let failure = self.name();
1388            let mapped = if let Some(error) = error {
1389                let mapper = self.flat.function(error)?;
1390                let mapper_arg = mapper.params.first().map(|p| &p.ty.ty);
1391                let shared = matches!(mapper_arg, Some(Type::Reference(r)) if r.mutability.is_none() && same_type(&r.elem, error_ty));
1392                if mapper.params.len() != 1
1393                    || !(mapper_arg.is_some_and(|ty| same_type(ty, error_ty)) || shared)
1394                    || type_key(&mapper.ret.ty) != "String"
1395                {
1396                    return Err(
1397                        "Result error mapper must take exactly E or &E and return String".into(),
1398                    );
1399                }
1400                let path = &mapper.path;
1401                let argument = if shared {
1402                    quote!(&#failure)
1403                } else {
1404                    quote!(#failure)
1405                };
1406                Some(quote!(#path(#argument)))
1407            } else {
1408                None
1409            };
1410            let error = self.codec.source_error(SourceError {
1411                flat: self.flat,
1412                ty: error_ty,
1413                value: quote!(#failure),
1414                message: mapped,
1415                extra,
1416            })?;
1417            let message = error.propagate;
1418            let mut child = if reference_return {
1419                self.output_return(success_ty, ok, extra, quote!(#payload), false)?
1420            } else {
1421                self.output(success_ty, ok, extra, quote!(#payload), false)?
1422            };
1423            let prepare = child.statements;
1424            child.statements = quote! {
1425                let #payload = match #value {
1426                    ::core::result::Result::Ok(value) => value,
1427                    ::core::result::Result::Err(#failure) => return Err(#message),
1428                };
1429                #prepare
1430            };
1431
1432            child.outcomes.source_error = Some(error.payload);
1433            return Ok(child);
1434        }
1435        if reference_return {
1436            self.output_return(ty, recipe, extra, value, borrowed)
1437        } else {
1438            self.output(ty, recipe, extra, value, borrowed)
1439        }
1440    }
1441    // Preserve reference semantics for source returns and getter results.
1442    // The codec checks whether a getter's owner survives the generated call.
1443    fn output_return(
1444        &self,
1445        ty: &Type,
1446        recipe: &Output,
1447        extra: &C::Extra,
1448        value: TokenStream,
1449        borrowed: bool,
1450    ) -> Result<CodecOutput<C>> {
1451        if !borrowed {
1452            if matches!(ty, Type::Reference(_)) {
1453                if let Output::Branch {
1454                    operation: OutputBranch::Combine,
1455                    children: parts,
1456                } = recipe
1457                {
1458                    if parts.is_empty() {
1459                        return Err("an output combination needs at least one part".into());
1460                    }
1461                    if mutable_reference(ty) && parts.len() > 1 {
1462                        return Err(
1463                            "a mutable reference cannot supply multiple output views".into()
1464                        );
1465                    }
1466                    let mut result = OutputPlan {
1467                        fallible: false,
1468                        context: Default::default(),
1469                        delivery: Default::default(),
1470                        commit: TokenStream::new(),
1471                        statements: TokenStream::new(),
1472                        values: vec![],
1473                        outcomes: prebindgen_flat::Outcomes {
1474                            success: WireNode::Branch {
1475                                controls: vec![],
1476                                children: vec![],
1477                                meta: self.codec.group(GroupKind::Values)?,
1478                            },
1479                            source_error: None,
1480                            binding_failure: None,
1481                        },
1482                    };
1483                    for (index, part) in parts.iter().enumerate() {
1484                        let mut child =
1485                            self.output_return(ty, part, extra, value.clone(), false)?;
1486                        self.prefix(&mut child.outcomes.success, ComponentSegment::View(index));
1487                        append(&mut result, child)?;
1488                    }
1489                    return Ok(result);
1490                }
1491            }
1492            if let (Type::Reference(reference), Output::Leaf(OutputLeafOp::Value)) = (ty, recipe) {
1493                let wire = self.name();
1494                if let Some(plan) = self.codec.output_return_reference(
1495                    self.flat,
1496                    reference,
1497                    extra,
1498                    wire.clone(),
1499                    value.clone(),
1500                )? {
1501                    return Ok(plan);
1502                }
1503                return self.output_named(ty, recipe, extra, value, false, Some(&wire));
1504            }
1505            if let (
1506                Some(payload_ty),
1507                Output::Unary {
1508                    operation: OutputUnary::Optional,
1509                    child: inner,
1510                },
1511            ) = (option_inner(ty), recipe)
1512            {
1513                let tag = self.name();
1514                let payload = self.name();
1515                let child =
1516                    self.output_return(payload_ty, inner, extra, quote!(#payload), false)?;
1517                return optional_output(self.codec, extra, tag, payload, value, false, child);
1518            }
1519        }
1520        self.output(ty, recipe, extra, value, borrowed)
1521    }
1522    fn output(
1523        &self,
1524        ty: &Type,
1525        recipe: &Output,
1526        extra: &C::Extra,
1527        value: TokenStream,
1528        borrowed: bool,
1529    ) -> Result<CodecOutput<C>> {
1530        self.output_named(ty, recipe, extra, value, borrowed, None)
1531    }
1532    fn output_named(
1533        &self,
1534        ty: &Type,
1535        recipe: &Output,
1536        extra: &C::Extra,
1537        value: TokenStream,
1538        borrowed: bool,
1539        wire_name: Option<&syn::Ident>,
1540    ) -> Result<CodecOutput<C>> {
1541        if let Some(referent) = output_referent(ty) {
1542            if matches!(recipe, Output::Leaf(OutputLeafOp::Projected(p)) if p.mode == ProjectionMode::Owned)
1543            {
1544                return Err("owned projection cannot consume a reference payload".into());
1545            }
1546            let value = if borrowed { quote!(*(#value)) } else { value };
1547            return self.output_named(referent, recipe, extra, value, true, wire_name);
1548        }
1549        let mut out = OutputPlan {
1550            fallible: false,
1551            context: Default::default(),
1552            delivery: Default::default(),
1553            commit: TokenStream::new(),
1554            outcomes: prebindgen_flat::Outcomes {
1555                success: WireNode::Branch {
1556                    controls: vec![],
1557                    children: vec![],
1558                    meta: self.codec.group(GroupKind::Values)?,
1559                },
1560                source_error: None,
1561                binding_failure: None,
1562            },
1563            statements: TokenStream::new(),
1564            values: Vec::new(),
1565        };
1566        match recipe {
1567            Output::Unary {
1568                operation: OutputUnary::CallbackArgument { .. },
1569                ..
1570            } => return Err("callback argument ownership requires an arrow domain".into()),
1571            Output::None => {
1572                if type_key(ty) != "()" {
1573                    return Err("None output requires unit".into());
1574                }
1575                out.outcomes.success = WireNode::Branch {
1576                    controls: vec![],
1577                    children: vec![],
1578                    meta: self.codec.group(GroupKind::Fields)?,
1579                };
1580            }
1581            Output::Arrow { operation, .. } => match *operation {},
1582            Output::Branch {
1583                operation: OutputBranch::Variant | OutputBranch::Arguments,
1584                ..
1585            } => return Err("variant/argument branch requires its parent operation".into()),
1586            Output::Branch {
1587                operation:
1588                    OutputBranch::Enum {
1589                        value: grouped,
1590                        mode,
1591                    },
1592                children: variants,
1593            } => {
1594                return self.enum_output(
1595                    ty, variants, *grouped, *mode, extra, value, borrowed, wire_name,
1596                )
1597            }
1598            Output::Unary {
1599                operation: OutputUnary::Convert { conversion },
1600                child: output,
1601            } => return self.output_conversion(ty, conversion, output, extra, value, borrowed),
1602            Output::Unary {
1603                operation: OutputUnary::Bulk,
1604                child: inner,
1605            } => {
1606                require_bulk(self.codec, Direction::Output)?;
1607                if self.bulk.get() {
1608                    return self.output_named(ty, inner, extra, value, borrowed, wire_name);
1609                }
1610                let wire = self.name();
1611                self.bulk.set(true);
1612                let child = self.output_named(ty, inner, extra, value, borrowed, wire_name);
1613                self.bulk.set(false);
1614                return self.codec.bulk_output(self.flat, ty, extra, wire, child?);
1615            }
1616            Output::Leaf(OutputLeafOp::Discard) => return Ok(out),
1617            Output::Unary {
1618                operation: OutputUnary::Named { name },
1619                child: output,
1620            } => {
1621                let mut plan = self.output_named(ty, output, extra, value, borrowed, wire_name)?;
1622                plan.outcomes.success = WireNode::Branch {
1623                    meta: self.codec.group(GroupKind::NamedOutput(name))?,
1624                    controls: vec![],
1625                    children: vec![plan.outcomes.success],
1626                };
1627                return Ok(plan);
1628            }
1629            Output::Unary {
1630                operation: OutputUnary::Sequence { delivery, mode },
1631                child: element,
1632            } => {
1633                if self.bulk.get() && *delivery == SequenceDelivery::Callback {
1634                    return Err("bulk cannot enclose callback delivery; apply bulk to the callback element expansion instead".into());
1635                }
1636                if *delivery == SequenceDelivery::Auto
1637                    && !self.bulk.get()
1638                    && !self.ordinary_collections.get()
1639                    && self.codec.default_output_sequence_is_bulk(self.flat, extra)
1640                {
1641                    return self.output_named(
1642                        ty,
1643                        &Output::Unary {
1644                            operation: OutputUnary::Bulk,
1645                            child: Box::new(recipe.clone()),
1646                        },
1647                        extra,
1648                        value,
1649                        borrowed,
1650                        wire_name,
1651                    );
1652                }
1653                if !self.bulk.get() && *delivery != SequenceDelivery::Callback {
1654                    if let Output::Unary {
1655                        operation: OutputUnary::Bulk,
1656                        child: inner,
1657                    } = element.as_ref()
1658                    {
1659                        return self.output_named(
1660                            ty,
1661                            &Output::Unary {
1662                                operation: OutputUnary::Bulk,
1663                                child: Box::new(Output::Unary {
1664                                    operation: OutputUnary::Sequence {
1665                                        delivery: *delivery,
1666                                        mode: *mode,
1667                                    },
1668                                    child: inner.clone(),
1669                                }),
1670                            },
1671                            extra,
1672                            value,
1673                            borrowed,
1674                            wire_name,
1675                        );
1676                    }
1677                }
1678                let (element_type, _) =
1679                    sequence_parts(ty).ok_or("sequence output requires a collection")?;
1680                let explicit_copy = *mode == Some(ProjectionMode::Borrowed);
1681                let mode = mode.unwrap_or_else(|| {
1682                    if borrowed || !sequence_parts(ty).unwrap().1.is_owned() {
1683                        ProjectionMode::Borrowed
1684                    } else {
1685                        ProjectionMode::Owned
1686                    }
1687                });
1688                let wire = self.name();
1689                if mode == ProjectionMode::Borrowed && !explicit_copy && !self.copy_borrowed.get() {
1690                    if let Some(plan) = self.codec.output_borrowed_sequence(
1691                        self.flat,
1692                        ty,
1693                        extra,
1694                        wire.clone(),
1695                        value.clone(),
1696                        element,
1697                        *delivery,
1698                        self.temporary_receiver.get(),
1699                    )? {
1700                        return Ok(plan);
1701                    }
1702                }
1703                let payload = self.name();
1704                let length = self.name();
1705                let index = self.name();
1706                // An explicit ordinary collection (or a callback row) keeps
1707                // nested automatic sequences ordinary. Explicit Bulk still wins.
1708                let ordinary = self.ordinary_collections.replace(
1709                    self.ordinary_collections.get() || *delivery != SequenceDelivery::Auto,
1710                );
1711                let child = self.with_copy_permission(explicit_copy, || {
1712                    self.with_temporary_receiver(
1713                        mode == ProjectionMode::Borrowed
1714                            && !borrowed
1715                            && sequence_parts(ty).unwrap().1.is_owned(),
1716                        || {
1717                            self.output(
1718                                element_type,
1719                                element,
1720                                extra,
1721                                quote!(#payload),
1722                                mode == ProjectionMode::Borrowed,
1723                            )
1724                        },
1725                    )
1726                });
1727                self.ordinary_collections.set(ordinary);
1728                let child = child?;
1729                return sequence::output(
1730                    self.codec,
1731                    ty,
1732                    extra,
1733                    wire,
1734                    payload,
1735                    length,
1736                    index,
1737                    value,
1738                    borrowed,
1739                    mode,
1740                    child,
1741                    if *delivery == SequenceDelivery::Auto {
1742                        SequenceDelivery::Value
1743                    } else {
1744                        *delivery
1745                    },
1746                    self.bulk.get(),
1747                );
1748            }
1749            Output::Unary {
1750                operation: OutputUnary::Result { .. },
1751                ..
1752            } => {
1753                return Err("Result error routing is supported only at the outermost return".into())
1754            }
1755            Output::Unary {
1756                operation: OutputUnary::Optional,
1757                child: inner,
1758            } => {
1759                let ty = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
1760                let tag = wire_name
1761                    .map(|w| format_ident!("{w}_present"))
1762                    .unwrap_or_else(|| self.name());
1763                let payload = wire_name
1764                    .map(|w| format_ident!("{w}_payload"))
1765                    .unwrap_or_else(|| self.name());
1766                let child_wire = wire_name.map(|w| format_ident!("{w}_some"));
1767                let child = self.output_named(
1768                    ty,
1769                    inner,
1770                    extra,
1771                    quote!(#payload),
1772                    borrowed,
1773                    child_wire.as_ref(),
1774                )?;
1775                return optional_output(self.codec, extra, tag, payload, value, borrowed, child);
1776            }
1777            Output::Request(never) => match *never {},
1778            Output::Leaf(OutputLeafOp::Value) => {
1779                if self.bulk.get() {
1780                    require_bulk(self.codec, Direction::Output)?;
1781                }
1782                let request = OutputLeaf {
1783                    flat: self.flat,
1784                    ty,
1785                    extra,
1786                    wire: wire_name.cloned().unwrap_or_else(|| self.name()),
1787                    value,
1788                    access: if borrowed {
1789                        ValueAccess::Shared
1790                    } else {
1791                        ValueAccess::Owned
1792                    },
1793                    representation: if self.bulk.get() {
1794                        Representation::Bulk
1795                    } else {
1796                        Representation::Ordinary
1797                    },
1798                };
1799                return if borrowed && self.copy_borrowed.get() {
1800                    self.codec
1801                        .output_projected(request, &Projection::borrowed())
1802                } else {
1803                    self.codec.output_leaf(request)
1804                };
1805            }
1806            Output::Leaf(OutputLeafOp::Projected(projection)) => {
1807                if self.bulk.get() {
1808                    require_bulk(self.codec, Direction::Output)?;
1809                }
1810                if borrowed && projection.mode == ProjectionMode::Owned {
1811                    return Err(
1812                        "owned projection cannot consume a shared-reference occurrence".into(),
1813                    );
1814                }
1815                let value = if !borrowed && projection.mode == ProjectionMode::Borrowed {
1816                    quote!(&(#value))
1817                } else {
1818                    value
1819                };
1820                return self.codec.output_projected(
1821                    OutputLeaf {
1822                        flat: self.flat,
1823                        ty,
1824                        extra,
1825                        wire: wire_name.cloned().unwrap_or_else(|| self.name()),
1826                        value,
1827                        access: if projection.mode == ProjectionMode::Borrowed {
1828                            ValueAccess::Shared
1829                        } else {
1830                            ValueAccess::Owned
1831                        },
1832                        representation: if self.bulk.get() {
1833                            Representation::Bulk
1834                        } else {
1835                            Representation::Ordinary
1836                        },
1837                    },
1838                    projection,
1839                );
1840            }
1841            Output::Branch {
1842                operation: OutputBranch::Fields,
1843                children,
1844            }
1845            | Output::Branch {
1846                operation: OutputBranch::OwnedFields,
1847                children,
1848            }
1849            | Output::Branch {
1850                operation: OutputBranch::Record { .. },
1851                children,
1852                ..
1853            } => {
1854                let grouped = matches!(
1855                    recipe,
1856                    Output::Branch {
1857                        operation: OutputBranch::Record { .. },
1858                        ..
1859                    }
1860                );
1861                let wire = grouped.then(|| wire_name.cloned().unwrap_or_else(|| self.name()));
1862                let owned = match recipe {
1863                    Output::Branch {
1864                        operation: OutputBranch::OwnedFields,
1865                        children: _,
1866                    } => true,
1867                    Output::Branch {
1868                        operation: OutputBranch::Record { mode, .. },
1869                        ..
1870                    } => mode.map_or(!borrowed, |m| m == ProjectionMode::Owned),
1871                    _ => false,
1872                };
1873                if owned && borrowed {
1874                    return Err("owned fields cannot consume a shared-reference occurrence".into());
1875                }
1876                if type_key(ty) == "()" && children.is_empty() {
1877                    out.outcomes.success = WireNode::Branch {
1878                        controls: vec![],
1879                        meta: self.codec.group(GroupKind::Fields)?,
1880                        children: vec![],
1881                    };
1882
1883                    return Ok(out);
1884                }
1885                let r = self.flat.record(ty)?;
1886                if r.fields.len() != children.len() {
1887                    return Err("wrong output field count".into());
1888                }
1889                let convert = |index: usize, value| {
1890                    let child_wire = wire.as_ref().map(|w| format_ident!("{w}_f{index}"));
1891                    self.output_named(
1892                        &r.fields[index].ty,
1893                        &children[index],
1894                        extra,
1895                        value,
1896                        !owned,
1897                        child_wire.as_ref(),
1898                    )
1899                };
1900                let var = wire
1901                    .as_ref()
1902                    .map(|w| format_ident!("{w}_record"))
1903                    .unwrap_or_else(|| self.name());
1904                let explicit_copy = matches!(
1905                    recipe,
1906                    Output::Branch {
1907                        operation: OutputBranch::Record {
1908                            mode: Some(ProjectionMode::Borrowed),
1909                            ..
1910                        },
1911                        ..
1912                    }
1913                );
1914                let mut plan = self.with_copy_permission(explicit_copy, || {
1915                    self.with_temporary_receiver(!owned && !borrowed, || {
1916                        if owned {
1917                            record_output_owned(
1918                                self.codec,
1919                                var,
1920                                value,
1921                                r.fields.iter().map(|f| f.name.clone()),
1922                                convert,
1923                            )
1924                        } else {
1925                            record_output(
1926                                self.codec,
1927                                var,
1928                                value,
1929                                r.fields.iter().map(|f| f.name.clone()),
1930                                convert,
1931                            )
1932                        }
1933                    })
1934                })?;
1935                if let Some(wire) = wire {
1936                    self.codec
1937                        .output_record_metadata(self.flat, ty, &wire, &mut plan, extra)?;
1938                }
1939                return Ok(plan);
1940            }
1941            Output::Unary {
1942                operation: OutputUnary::Getter { function },
1943                child: output,
1944            }
1945            | Output::Unary {
1946                operation: OutputUnary::Into { function },
1947                child: output,
1948            } => {
1949                let consuming = matches!(
1950                    recipe,
1951                    Output::Unary {
1952                        operation: OutputUnary::Into { .. },
1953                        ..
1954                    }
1955                );
1956                if consuming && borrowed {
1957                    return Err(
1958                        "consuming accessor cannot consume a shared-reference occurrence".into(),
1959                    );
1960                }
1961                let f = self.flat.function(function)?;
1962                let expected: Type = if consuming {
1963                    ty.clone()
1964                } else {
1965                    syn::parse_quote!(&#ty)
1966                };
1967                if f.params.len() != 1 || !same_type(&f.params[0].ty.ty, &expected) {
1968                    return Err(format!(
1969                        "getter `{function}` must take exactly `{}`",
1970                        type_key(&expected)
1971                    ));
1972                }
1973                if matches!(&f.ret.ty, Type::Reference(r) if r.mutability.is_some()) {
1974                    return Err("mutable-reference getter results are unsupported".into());
1975                }
1976                let v = self.name();
1977                let path = &f.path;
1978                let arg = if borrowed || consuming {
1979                    value
1980                } else {
1981                    quote!(&#value)
1982                };
1983                out.statements = quote!(let #v=#path(#arg););
1984                let mut child = self.output_return(&f.ret.ty, output, extra, quote!(#v), false)?;
1985                self.codec.validate_getter_output(
1986                    self.flat,
1987                    ty,
1988                    borrowed && !self.temporary_receiver.get(),
1989                    &f.ret.ty,
1990                    &child,
1991                    extra,
1992                )?;
1993                self.prefix(
1994                    &mut child.outcomes.success,
1995                    ComponentSegment::Getter(function.clone()),
1996                );
1997                append(&mut out, child)?;
1998            }
1999            Output::Branch {
2000                operation: OutputBranch::Combine,
2001                children: parts,
2002            } => {
2003                if parts.is_empty() {
2004                    return Err("an output combination needs at least one part".into());
2005                }
2006                if parts
2007                    .iter()
2008                    .filter(|p| p.consumes(self.flat, ty, extra, self.codec, borrowed))
2009                    .count()
2010                    > 1
2011                {
2012                    return Err("a combination cannot consume the same value twice; use field_self_ref() for an explicit copy view".into());
2013                }
2014                if !borrowed {
2015                    if let Some(plan) = self.output_owner_views(ty, parts, extra, value.clone())? {
2016                        return Ok(plan);
2017                    }
2018                }
2019                let mut bodies = Vec::new();
2020                for (index, child) in parts.iter().enumerate() {
2021                    let consuming = child.consumes(self.flat, ty, extra, self.codec, borrowed);
2022                    let arg = if borrowed || consuming {
2023                        value.clone()
2024                    } else {
2025                        quote!(&#value)
2026                    };
2027                    let mut p = self.with_temporary_receiver(!borrowed && !consuming, || {
2028                        self.output(ty, child, extra, arg, borrowed || !consuming)
2029                    })?;
2030                    self.prefix(&mut p.outcomes.success, ComponentSegment::View(index));
2031                    bodies.push((consuming, std::mem::take(&mut p.statements)));
2032                    append(&mut out, p)?;
2033                }
2034                // Wire order follows the declaration. Observations execute before the move.
2035                bodies.sort_by_key(|(consumes, _)| *consumes);
2036                for (_, body) in bodies {
2037                    out.statements.extend(body);
2038                }
2039            }
2040        }
2041
2042        Ok(out)
2043    }

2045    fn output_owner_views(
2046        &self,
2047        ty: &Type,
2048        parts: &[Output],
2049        extra: &C::Extra,
2050        value: TokenStream,
2051    ) -> Result<Option<CodecOutput<C>>> {
2052        let owner = parts.iter().position(|part| {
2053            matches!(part,
2054            Output::Leaf(OutputLeafOp::Projected(p)) if p.mode == ProjectionMode::Owned && p.fields.is_empty())
2055                || matches!(part, Output::Leaf(OutputLeafOp::Value)) && self.codec.output_consumes(self.flat, ty, extra)
2056        });
2057        let Some(owner) = owner else {
2058            return Ok(None);
2059        };
2060        if parts.len() < 2 {
2061            return Ok(None);
2062        }
2063        // Explicit copy projections use observation-before-move ordering. They
2064        // do not require an address-preserving loan from the delivered owner.
2065        if parts
2066            .iter()
2067            .any(|part| matches!(part,Output::Leaf(OutputLeafOp::Projected(p)) if p.mode==ProjectionMode::Borrowed))
2068        {
2069            return Ok(None);
2070        }
2071        // Reserve the same owner wire that the ordinary consuming leaf uses.
2072        let next = self.next.get();
2073        let wire = self.name();
2074        let Some((mut owned, pointer)) =
2075            self.codec.output_owner(self.flat, ty, extra, wire, value)?
2076        else {
2077            self.next.set(next);
2078            return Ok(None);
2079        };
2080        if parts.iter().enumerate().any(|(i, part)| {
2081            i != owner
2082                && !matches!(
2083                    part,
2084                    Output::Unary {
2085                        operation: OutputUnary::Getter { .. },
2086                        ..
2087                    }
2088                )
2089        }) {
2090            return Err("an owner-backed output combination supports only the owning identity and getter views".into());
2091        }
2092        let prepare_owner = std::mem::take(&mut owned.statements);
2093        let mut result = OutputPlan {
2094            fallible: false,
2095            context: Default::default(),
2096            delivery: Default::default(),
2097            commit: TokenStream::new(),
2098            statements: TokenStream::new(),
2099            values: vec![],
2100            outcomes: prebindgen_flat::Outcomes {
2101                success: WireNode::Branch {
2102                    controls: vec![],
2103                    children: vec![],
2104                    meta: self.codec.group(GroupKind::Values)?,
2105                },
2106                source_error: None,
2107                binding_failure: None,
2108            },
2109        };
2110        let mut owner_plan = Some(owned);
2111        for (index, part) in parts.iter().enumerate() {
2112            let mut plan = if index == owner {
2113                owner_plan.take().unwrap()
2114            } else {
2115                let Output::Unary {
2116                    operation: OutputUnary::Getter { function },
2117                    child: output,
2118                } = part
2119                else {
2120                    unreachable!()
2121                };
2122                let f = self.flat.function(function)?;
2123                if f.params.len() != 1 {
2124                    return Err(format!(
2125                        "getter `{function}` must borrow its returned owner"
2126                    ));
2127                }
2128                let receiver = &f.params[0].ty.ty;
2129                let mutable_receiver =
2130                    matches!(receiver, Type::Reference(r) if r.mutability.is_some());
2131                let mutable = mutable_receiver || mutable_reference(&f.ret.ty);
2132                if mutable && parts.len() != 2 {
2133                    return Err("a mutable owner getter must be the only getter view; simultaneous owner borrows could invalidate earlier views".into());
2134                }
2135                let expected: Type = if mutable_receiver {
2136                    syn::parse_quote!(&mut #ty)
2137                } else {
2138                    syn::parse_quote!(&#ty)
2139                };
2140                if !same_type(receiver, &expected) {
2141                    return Err(format!(
2142                        "getter `{function}` must borrow its returned owner"
2143                    ));
2144                }
2145                let source = self.name();
2146                let path = &f.path;
2147                let argument = if mutable_receiver {
2148                    quote!(unsafe { &mut *(#pointer) })
2149                } else {
2150                    quote!(unsafe { &*(#pointer) })
2151                };
2152                let mut plan =
2153                    self.output_return(&f.ret.ty, output, extra, quote!(#source), false)?;
2154                self.codec
2155                    .validate_owner_getter(self.flat, ty, function, &plan)?;
2156                plan.statements = {
2157                    let body = plan.statements;
2158                    quote!(let #source = #path(#argument); #body)
2159                };
2160                self.prefix(
2161                    &mut plan.outcomes.success,
2162                    ComponentSegment::Getter(function.clone()),
2163                );
2164                plan
2165            };
2166            self.prefix(&mut plan.outcomes.success, ComponentSegment::View(index));
2167            append(&mut result, plan)?;
2168        }
2169        result.statements = {
2170            let views = result.statements;
2171            quote!(#prepare_owner #views)
2172        };
2173        Ok(Some(result))
2174    }
2175}
2176fn mutable_reference(ty: &Type) -> bool {
2177    matches!(ty, Type::Reference(r) if r.mutability.is_some())
2178        || option_inner(ty).is_some_and(mutable_reference)
2179}
2180fn validate_nested_output<W: prebindgen_flat::WireKind, C, N, B>(
2181    plan: &OutputPlan<W, C, N, B>,
2182) -> Result<()> {
2183    if plan.outcomes.source_error.is_some() || plan.outcomes.binding_failure.is_some() {
2184        return Err("completion channels belong to the outermost output conversion".into());
2185    }
2186    Ok(())
2187}
2188fn append<W: prebindgen_flat::WireKind, C, N, B>(
2189    to: &mut OutputPlan<W, C, N, B>,
2190    from: OutputPlan<W, C, N, B>,
2191) -> Result<()> {
2192    if !from.delivery.statements.is_empty()
2193        || from.delivery.result.to_string() != quote!(()).to_string()
2194    {
2195        return Err("composed output delivery must be installed by Codec::finish_output".into());
2196    }
2197    validate_nested_output(&from)?;
2198    let WireNode::Branch {
2199        children: elements, ..
2200    } = &mut to.outcomes.success
2201    else {
2202        return Err("output components require an ordered group".into());
2203    };
2204    elements.push(from.outcomes.success);
2205    to.fallible |= from.fallible;
2206    to.commit.extend(from.commit);
2207    to.statements.extend(from.statements);
2208    to.values.extend(from.values);
2209    prebindgen_conversions::validate_value_bindings(&to.values)
2210}
