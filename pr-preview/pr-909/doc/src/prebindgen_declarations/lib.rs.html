1//! Declaration language and composition for preparing conversion slots.
2//!
3//! [`DeclDefaults::resolve`] produces a request-free [`Flat<ResolvedSchema<C>>`](ResolvedSchema)
4//! that owns all data needed by [`compile_resolved`]. Input and output trees keep
5//! separate operation vocabularies; [`InputNode::Arrow`] and [`OutputNode::Arrow`]
6//! expose reverse-direction domains structurally. [`DeclDefaults::fill`],
7//! [`bind_input`] and [`bind_output`] use the same resolver and compiler for
8//! individual occurrences. Compiled slots retain wires and Rust code for emission.
9//! Generation is owned by the language generator;
10//! this crate owns only declaration resolution and slot preparation.
11//! Each [`Wire`] carries the adapter's own description in [`Wire::kind`].
12//! Reusable value and boxed-allocation leaves live in `prebindgen-conversions`.

51// Sized shared outputs use the referent's ordinary recipe, with borrowing
52// carried by the compiler. Keep unsized string/slice views as their own types:
53// adapters and sequence transport already understand those reference shapes.
54/// Referent used when observing a sized shared reference.
55pub fn output_referent(ty: &Type) -> Option<&Type> {
56    match ty {
57        Type::Reference(r)
58            if r.mutability.is_none()
59                && !matches!(r.elem.as_ref(), Type::Slice(_))
60                && !matches!(r.elem.as_ref(), Type::Path(p) if p.path.is_ident("str")) =>
61        {
62            Some(&r.elem)
63        }
64        _ => None,
65    }
66}
67// The owned value deliberately constructed by DeclInputUnary::Temporary. A Vec/String
68// descriptor must never be fabricated over memory merely borrowed from C.
69/// Owned backing type reconstructed for an explicitly requested temporary borrow.
70pub fn temporary_input_type(ty: &Type) -> Result<Type> {
71    let Type::Reference(reference) = ty else {
72        return Err("reconstruct() requires a shared-reference input; apply it to the optional payload if necessary".into());
73    };
74    if reference.mutability.is_some()
75        || reference
76            .lifetime
77            .as_ref()
78            .is_some_and(|l| l.ident == "static")
79    {
80        return Err("reconstruct() cannot supply a mutable or 'static reference".into());
81    }
82    Ok(match reference.elem.as_ref() {
83        Type::Slice(slice) => {
84            let element = &slice.elem;
85            syn::parse_quote!(Vec<#element>)
86        }
87        Type::Path(path) if path.path.is_ident("str") => syn::parse_quote!(String),
88        other => other.clone(),
89    })
90}

124/// Source-dependent choices accepted only while preparing an input tree.
125#[derive(Clone, Debug)]
126pub enum DeclInputRequest {
127    /// Expand enum fields, inheriting defaults for fields not explicitly overridden.
128    EnumFields(
129        std::collections::BTreeMap<
130            String,
131            std::collections::BTreeMap<String, DeclInput<DeclInputRequest, DeclOutputRequest>>,
132        >,
133    ),
134    /// Resolve only the canonical value form, independently of boundary defaults.
135    Value,
136    /// Select a local single-value conversion.
137    Conversion(Box<DeclConvert>),
138    /// Select the configured default for this source occurrence.
139    Auto,
140    /// Select or override a callback signature's argument and result conversions.
141    Callback(Box<DeclCallback>),
142    /// Apply a type-specific override, including inside optional/sequence sources.
143    ForType {
144        /// Source type targeted by the override.
145        ty: Box<Type>,
146        /// The requested conversion for that type.
147        input: Box<DeclInput<DeclInputRequest, DeclOutputRequest>>,
148    },
149    /// Infer a constructor's argument conversions from its source signature.
150    Call(String),
151}

153impl DeclInput<DeclInputRequest, DeclOutputRequest> {
154    /// Expand the tag and selected fields. Keys are source variant and field
155    /// names (or tuple indices). Unspecified fields inherit boundary defaults.
156    pub fn enum_fields<'a>(
157        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
158    ) -> Self {
159        DeclInput::Request(DeclInputRequest::EnumFields(variants::overrides(overrides)))
160    }
161    /// Request the canonical value representation, bypassing boundary defaults.
162    pub fn value() -> Self {
163        DeclInput::Request(DeclInputRequest::Value)
164    }
165    /// Request this source type's configured default during preparation.
166    pub fn auto() -> Self {
167        DeclInput::Request(DeclInputRequest::Auto)
168    }
169    /// Associate a conversion with a source type for a default or nested override.
170    pub fn for_type(ty: Type, input: Self) -> Self {
171        DeclInput::Request(DeclInputRequest::ForType {
172            ty: Box::new(ty),
173            input: Box::new(input),
174        })
175    }
176    /// Infer a source constructor's argument conversions from registered defaults.
177    pub fn call(name: impl Into<String>) -> Self {
178        DeclInput::Request(DeclInputRequest::Call(name.into()))
179    }
180    /// Explicitly reconstruct an owned temporary for a shared-reference input.
181    /// Slices construct a Vec, str constructs a String, and sized references
182    /// construct their referent. Child recipes keep their ordinary ownership
183    /// semantics; this operation authorizes the temporary's allocation/copying.
184    pub fn reconstruct(self) -> Self {
185        match self {
186            DeclInput::Request(DeclInputRequest::ForType { ty, input }) => DeclInput::for_type(
187                *ty,
188                DeclInput::Unary {
189                    operation: DeclInputUnary::Temporary,
190                    child: input,
191                },
192            ),
193            other => DeclInput::Unary {
194                operation: DeclInputUnary::Temporary,
195                child: Box::new(other),
196            },
197        }
198    }
199    /// Add a constructor alternative to this conversion tree.
200    pub fn variant(self, name: impl Into<String>) -> Self {
201        self.add_alternative(DeclInput::call(name))
202    }
203    /// Add a direct codec alternative, bypassing this type's expansion default.
204    pub fn variant_self(self) -> Self {
205        self.add_alternative(DeclInput::Leaf(DeclInputLeafOp::Value))
206    }
207    fn add_alternative(self, arm: Self) -> Self {
208        match self {
209            DeclInput::Request(DeclInputRequest::ForType { ty, input }) => {
210                DeclInput::for_type(*ty, input.add_alternative(arm))
211            }
212            DeclInput::Branch {
213                operation: DeclInputBranch::Selector,
214                children: mut arms,
215            } => {
216                arms.push(arm);
217                DeclInput::Branch {
218                    operation: DeclInputBranch::Selector,
219                    children: arms,
220                }
221            }
222            other => DeclInput::Branch {
223                operation: DeclInputBranch::Selector,
224                children: vec![other, arm],
225            },
226        }
227    }
228}
229impl<InRequest, OutRequest> DeclInput<InRequest, OutRequest> {
230    /// Group this explicit expansion into a bulk carrier.
231    pub fn bulk(self) -> Self {
232        DeclInput::Unary {
233            operation: DeclInputUnary::Bulk,
234            child: Box::new(self),
235        }
236    }
237    /// Construct a record from field recipes, or unit from an empty list.
238    ///
239    /// `fields` holds one recipe per field of the source record, in declaration
240    /// order. Binding verifies the count against the record.
241    pub fn fields(fields: impl IntoIterator<Item = DeclInput<InRequest, OutRequest>>) -> Self {
242        DeclInput::Branch {
243            operation: DeclInputBranch::Fields,
244            children: fields.into_iter().collect(),
245        }
246    }
247    /// Call a named source constructor with the supplied argument recipes.
248    ///
249    /// Binding checks its result type and argument count. Argument validation and
250    /// ownership acquisition precede the source call.
251    ///
252    /// # Arguments
253    ///
254    /// * `name` — unqualified name of a source function whose result type is the
255    ///   input's type.
256    /// * `args` — one recipe per parameter of that function, in order.
257    pub fn constructor(
258        name: impl Into<String>,
259        args: impl IntoIterator<Item = DeclInput<InRequest, OutRequest>>,
260    ) -> Self {
261        DeclInput::Branch {
262            operation: DeclInputBranch::Constructor {
263                function: name.into(),
264            },
265            children: args.into_iter().collect(),
266        }
267    }
268    /// Choose between owned representations of the same input type.
269    ///
270    /// `arms` holds one recipe per alternative; each must produce the input's
271    /// type. The caller passes the zero-based index of the chosen alternative in an
272    /// extra wire ahead of all arms' wires.
273    ///
274    /// Requires at least one arm. Each arm retains its own wires; this does not
275    /// merge equivalent wire parameters or generate destination-language overloads.
276    pub fn selector(arms: impl IntoIterator<Item = DeclInput<InRequest, OutRequest>>) -> Self {
277        DeclInput::Branch {
278            operation: DeclInputBranch::Selector,
279            children: arms.into_iter().collect(),
280        }
281    }
282}

285pub enum DeclOutputRequest {
286    /// Select per-element delivery after resolving defaults and local expansion.

308    Auto,

314        output: Box<DeclOutput<DeclInputRequest, DeclOutputRequest>>,
315    },
316    /// Resolve source record fields by name and fill unspecified conversions.
317    NamedFields {
318        /// Explicit conversions keyed by source field name.

321        /// Ownership of field decomposition.
322        mode: ProjectionMode,
323    },

326impl DeclOutput<DeclInputRequest, DeclOutputRequest> {
327    /// Expand an enum's tag and active fields using source variant/member keys.
328    pub fn enum_fields<'a>(
329        overrides: impl IntoIterator<Item = (&'a str, Vec<(&'a str, Self)>)>,
330    ) -> Self {
331        DeclOutput::Request(DeclOutputRequest::EnumFields(variants::overrides(
332            overrides,
333        )))
334    }
335    /// Request the canonical value representation, bypassing boundary defaults.
336    pub fn value() -> Self {
337        DeclOutput::Request(DeclOutputRequest::Value { projection: None })
338    }
339    /// Request this source type's configured output default during preparation.
340    pub fn auto() -> Self {
341        DeclOutput::Request(DeclOutputRequest::Auto)
342    }
343    /// Associate a conversion with a source type for a default or nested override.
344    pub fn for_type(ty: Type, output: Self) -> Self {
345        DeclOutput::Request(DeclOutputRequest::ForType {
346            ty: Box::new(ty),
347            output: Box::new(output),
348        })
349    }
350    /// Append another simultaneous output view.
351    pub fn view(self, view: impl Into<Self>) -> Self {
352        let view = view.into();
353        match self {
354            DeclOutput::Request(DeclOutputRequest::ForType { ty, output }) => {
355                DeclOutput::for_type(*ty, output.view(view))
356            }
357            DeclOutput::Branch {
358                operation: DeclOutputBranch::Combine,
359                children: mut parts,
360            } => {
361                parts.push(view);
362                DeclOutput::Branch {
363                    operation: DeclOutputBranch::Combine,
364                    children: parts,
365                }
366            }
367            other => DeclOutput::Branch {
368                operation: DeclOutputBranch::Combine,
369                children: vec![other, view],
370            },
371        }
372    }
373    /// Append a source getter whose result uses its registered conversion default.
374    pub fn field(self, name: impl Into<String>) -> Self {
375        self.view(DeclOutput::getter(name, DeclOutput::auto()))
376    }
377    /// Append the direct codec representation, bypassing this type's expansion default.
378    pub fn field_self(self) -> Self {
379        self.view(DeclOutput::Leaf(DeclOutputLeafOp::Value))
380    }
381    /// Append an explicit borrowed projection.
382    pub fn field_self_ref(self) -> Self {
383        self.project_self(Projection::borrowed())
384    }
385    /// Append an explicit owned projection.
386    pub fn field_self_into(self) -> Self {
387        self.project_self(Projection::owned())
388    }
389    /// Append a projection with explicit ownership and nested field policies.
390    pub fn project_self(self, policy: Projection) -> Self {
391        self.view(DeclOutput::Leaf(DeclOutputLeafOp::Projected(policy)))
392    }
393    /// Append a consuming accessor followed by owned field decomposition.
394    pub fn fields_into(self, view: Self) -> Self {
395        let DeclOutput::Unary {
396            operation: DeclOutputUnary::Getter { function },
397            child: output,
398        } = view
399        else {
400            panic!("fields_into requires a fields! accessor")
401        };
402        let DeclOutput::Request(DeclOutputRequest::NamedFields { overrides, .. }) = *output else {
403            panic!("fields_into requires named fields")
404        };
405        self.view(DeclOutput::Unary {
406            operation: DeclOutputUnary::Into { function },
407            child: Box::new(DeclOutput::Request(DeclOutputRequest::NamedFields {
408                overrides,
409                mode: ProjectionMode::Owned,
410            })),
411        })
412    }
413    /// Deprecated spelling of [`DeclOutput::fields_into`].
414    #[deprecated(note = "use fields_into instead")]
415    pub fn fields_self_into(self, view: Self) -> Self {
416        self.fields_into(view)
417    }
418    /// Override one named source field, preserving defaults for its siblings.
419    pub fn field_override(mut self, name: impl Into<String>, output: Self) -> Self {
420        match &mut self {
421            DeclOutput::Request(DeclOutputRequest::NamedFields { overrides, .. }) => {
422                assert!(
423                    overrides.insert(name.into(), output).is_none(),
424                    "duplicate field override"
425                );
426            }
427            DeclOutput::Unary {
428                operation: DeclOutputUnary::Getter { .. },
429                child: inner,
430                ..
431            }
432            | DeclOutput::Unary {
433                operation: DeclOutputUnary::Into { .. },
434                child: inner,
435                ..
436            }
437            | DeclOutput::Request(DeclOutputRequest::ForType { output: inner, .. }) => {
438                let old = std::mem::replace(inner, Box::new(DeclOutput::auto()));
439                **inner = old.field_override(name, output);
440            }
441            _ => panic!("field override requires named field decomposition"),
442        }
443        self
444    }
445}

504/// A flat model whose slots contain only conversion data for wrapper emission.
505///
506/// Unbound source helpers have `None` slots. Bound functions need `Some` for their
507/// result and every parameter; [`prebindgen_flat::validate_bindings`] checks this before emission.
508/// Record fields retain their canonical source information and metadata. Descriptions and [`Codec::Extra`]
509/// are construction inputs, not stored in this model.
510///
511/// Output conversion data cannot be attached to an input position:
512/// ```compile_fail
513/// use prebindgen_declarations::PlanBindings;
514/// fn swap(flat: &mut PlanBindings<syn::Type>) {
515///     let function = flat.function_mut("foo").unwrap();
516///     function.params[0].ty.slot = function.ret.slot.clone();
517/// }
518/// ```
519/// Nor can a recipe be stored in a slot:
520/// ```compile_fail
521/// use prebindgen_declarations::{PlanBindings, DeclInput};
522/// fn configure(flat: &mut PlanBindings<syn::Type>) {
523///     flat.function_mut("foo").unwrap().params[0].ty.slot = Some(DeclInput::Leaf(prebindgen_declarations::frontend::DeclInputLeafOp::Value));
524/// }
525/// ```
526pub type PlanBindings<WireType, Context = (), Chain = (), Branch = ()> =
527    Flat<prebindgen_conversions::PlanSchema<WireType, Context, Chain, Branch>>;
528
529/// Compile a request-free description for one named source parameter.
530/// The model keeps its fixed schema; this returns a converter without modifying its slot.
531/// ```
532/// use prebindgen_declarations::{compile_input,Codec,ResolvedInput};
533/// use prebindgen_flat::Flat;
534/// fn resolved<C:Codec>(flat:&Flat<C::Schema>,codec:&C,tree:&ResolvedInput<C>,extra:&C::Extra) {
535///     compile_input(flat,codec,"f","x",tree,extra).unwrap();
536/// }
537/// ```
538/// ```compile_fail
539/// use prebindgen_declarations::{compile_input,Codec,DeclInput};
540/// use prebindgen_flat::Flat;
541/// fn pending<C:Codec>(flat:&Flat<C::Schema>,codec:&C,extra:&C::Extra) {
542///     compile_input(flat,codec,"f","x",&DeclInput::auto(),extra).unwrap();
543/// }
544/// ```
545pub fn compile_input<C: Codec>(
546    flat: &Flat<C::Schema>,
547    codec: &C,
548    function: &str,
549    parameter: &str,
550    recipe: &ResolvedInput<C>,
551    extra: &C::Extra,
552) -> Result<<C::Schema as Schema>::Input> {
553    let f = flat.function(function)?;
554    let index = f
555        .params
556        .iter()
557        .position(|p| p.name == parameter)
558        .ok_or_else(|| format!("`{function}`: unknown parameter `{parameter}`"))?;
559    let mut plan = Compiler::new(flat, codec, format!("p{index}")).input(
560        &f.params[index].ty.ty,
561        recipe,
562        extra,
563        &Condition::default(),
564    )?;
565    codec.finish_input(&mut plan, extra)?;
566    codec.validate_input(&plan)?;
567    Ok(plan.into())
568}

633    flat: &Flat<C::Schema>,
634    codec: &C,
635    ty: &Type,
636    recipe: &ResolvedInput<C>,
637    extra: &C::Extra,
638) -> Result<<C::Schema as Schema>::Input> {
639    let compiler = Compiler {
640        input_role: std::cell::Cell::new(InputRole::CallbackResult),
641        ..Compiler::new(flat, codec, "callback_return".into())
642    };
643    let mut plan = compiler.input(ty, recipe, extra, &Condition::default())?;
644    codec.finish_input(&mut plan, extra)?;
645    codec.validate_input(&plan)?;
646    Ok(plan.into())
647}

649/// Compile one Rust-to-foreign callback argument with independent local names.
650///
651/// `index` is its position in the callback signature. The adapter supplies callback
652/// delivery instead of the ordinary function-result packet; the same resolved
653/// projection and ownership operations prepare the value in either position.
654pub fn compile_callback_argument<C: Codec>(
655    flat: &Flat<C::Schema>,
656    codec: &C,
657    ty: &Type,
658    recipe: &ResolvedOutput<C>,
659    index: usize,
660    borrowed: bool,
661    extra: &C::Extra,
662) -> Result<<C::Schema as Schema>::Output> {
663    compile_output_at(
664        flat,
665        codec,
666        ty,
667        recipe,
668        extra,
669        OutputPosition::CallbackArgument { index, borrowed },
670    )
671}
672
673enum OutputPosition {
674    Return,
675    CallbackArgument { index: usize, borrowed: bool },
676}
677
678fn compile_output_at<C: Codec>(
679    flat: &Flat<C::Schema>,
680    codec: &C,
681    ty: &Type,
682    recipe: &ResolvedOutput<C>,
683    extra: &C::Extra,
684    position: OutputPosition,
685) -> Result<<C::Schema as Schema>::Output> {
686    let value = prebindgen_conversions::source_result_expr();
687    let mut plan = match position {
688        OutputPosition::Return => Compiler::new(flat, codec, "ret".into())
689            .output_root(ty, recipe, extra, value, false, true)?,
690        OutputPosition::CallbackArgument { index, borrowed } => {
691            Compiler::new(flat, codec, format!("callback_arg_{index}"))
692                .output_root(ty, recipe, extra, value, borrowed, true)?
693        }
694    };
695    finalize_output_schema(codec, &mut plan, extra)?;
696    codec.finish_output(&mut plan, extra)?;
697    codec.validate_output(&plan)?;
698    Ok(plan.into())
699}

701/// Build and attach a converter to a named function parameter.
702///
703/// Resolves the occurrence's source type from `flat`, checks recipe compatibility,
704/// and recursively composes wires and code. Nested conversions use the supplied
705/// `extra`, not data from referenced constructors or getters. Recipes and `extra`
706/// are borrowed only for this call. A failure leaves the existing slot unchanged.
707/// Rebinding an occurrence replaces its converter without changing other slots.
708/// Generated names are scoped by parameter position, so separate calls cannot
709/// collide within the wrapper. Keep the function signature fixed after binding.
710/// Check completeness with [`prebindgen_flat::validate_bindings`].
711///
712/// # Arguments
713///
714/// * `flat` — the model holding the function; the parameter's slot is replaced.
715/// * `codec` — the adapter's leaf conversions.
716/// * `function` — unqualified name of the source function.
717/// * `parameter` — name of the parameter, as written in the source.
718/// * `recipe` — preparation tree describing how to build the argument from boundary inputs.
719///   Adapter-resolved trees go directly to [`compile_input`].
720/// * `extra` — occurrence data passed to every codec call for this conversion.
721///
722/// Input binding rejects output recipes:
723/// ```compile_fail
724/// use prebindgen_declarations::{bind_input, DeclCodec, DeclOutput};
725/// use prebindgen_flat::Flat;
726/// fn wrong<C: DeclCodec>(flat: &mut Flat<C::Schema>, codec: &C, extra: &C::Extra) {
727///     bind_input(flat, codec, "foo", "x", &DeclOutput::value(), extra).unwrap();
728/// }
729/// ```
730pub fn bind_input<C: DeclCodec>(
731    flat: &mut Flat<C::Schema>,
732    codec: &C,
733    function: &str,
734    parameter: &str,
735    recipe: &DeclInput<DeclInputRequest, DeclOutputRequest>,
736    extra: &C::Extra,
737) -> Result<()> {
738    let f = flat.function(function)?;
739    let index = f
740        .params
741        .iter()
742        .position(|p| p.name == parameter)
743        .ok_or_else(|| format!("`{function}`: unknown parameter `{parameter}`"))?;
744    let defaults = DeclDefaults::new(flat, vec![], vec![], vec![])?;
745    let recipe = defaults.resolve_input(flat, codec, &f.params[index].ty.ty, recipe, extra)?;
746    let conversion = compile_input(flat, codec, function, parameter, &recipe, extra)?;
747    flat.function_mut(function)?.params[index].ty.slot = Some(conversion);
748    Ok(())
749}

751/// Build and attach a converter to a function result, replacing any existing one.
752///
753/// Uses the same construction and failure rules as [`bind_input`]. Output code
754/// uses [`prebindgen_conversions::source_result_expr`] for the source call's result local.
755/// The recipe is checked for compatible types and signatures, and for repeated
756/// consumption of the same value. Unit results use `DeclOutput::fields([])`.
757///
758/// # Arguments
759///
760/// * `flat` — the model holding the function; the result's slot is replaced.
761/// * `codec` — the adapter's leaf conversions.
762/// * `function` — unqualified name of the source function.
763/// * `recipe` — preparation tree describing how to deliver the result as boundary outputs.
764///   Adapter-resolved trees go directly to [`compile_output`].
765/// * `extra` — occurrence data passed to every codec call for this conversion.
766///
767/// Output binding rejects input recipes:
768/// ```compile_fail
769/// use prebindgen_declarations::{bind_output, DeclCodec, DeclInput};
770/// use prebindgen_flat::Flat;
771/// fn wrong<C: DeclCodec>(flat: &mut Flat<C::Schema>, codec: &C, extra: &C::Extra) {
772///     bind_output(flat, codec, "foo", &DeclInput::value(), extra).unwrap();
773/// }
774/// ```
775pub fn bind_output<C: DeclCodec>(
776    flat: &mut Flat<C::Schema>,
777    codec: &C,
778    function: &str,
779    recipe: &DeclOutput<DeclInputRequest, DeclOutputRequest>,
780    extra: &C::Extra,
781) -> Result<()> {
782    let defaults = DeclDefaults::new(flat, vec![], vec![], vec![])?;
783    let recipe =
784        defaults.resolve_output(flat, codec, &flat.function(function)?.ret.ty, recipe, extra)?;
785    let conversion = compile_output(flat, codec, function, &recipe, extra)?;
786    flat.function_mut(function)?.ret.slot = Some(conversion);
787    Ok(())
788}

790fn finalize_output_schema<C: Codec>(
791    codec: &C,
792    plan: &mut CodecOutputPlan<C>,
793    extra: &C::Extra,
794) -> Result<()> {
795    if let Some(error) = codec.binding_failure_layout(extra)? {
796        plan.outcomes.binding_failure = Some(error);
797    }
798    codec.validate_output(plan)
799}

801/// Generation context shared by nested adapter conversion operations.
802/// It holds source metadata and a deterministic name scope, not an operation vocabulary.
803pub struct Compiler<'a, C: Codec> {
804    /// Whether the input supplies a function parameter or callback result.
805    pub input_role: std::cell::Cell<InputRole>,
806    copy_borrowed: std::cell::Cell<bool>,
807    temporary_receiver: std::cell::Cell<bool>,
808    /// Source declarations and adapter metadata.
809    pub flat: &'a Flat<C::Schema>,
810    /// Interpreter for this adapter’s operations.
811    pub codec: &'a C,
812    prefix: String,
813    next: std::cell::Cell<usize>,
814    /// Adapter-specific state for this compilation scope.
815    pub state: C::CompileState,
816}
817#[allow(missing_docs, clippy::too_many_arguments)]
818impl<'a, C: Codec> Compiler<'a, C> {
819    pub fn new(flat: &'a Flat<C::Schema>, codec: &'a C, prefix: String) -> Self {
820        Self {
821            input_role: std::cell::Cell::new(InputRole::Parameter),
822            copy_borrowed: std::cell::Cell::new(false),
823            temporary_receiver: std::cell::Cell::new(false),
824            flat,
825            codec,
826            prefix,
827            next: std::cell::Cell::new(0),
828            state: Default::default(),
829        }
830    }
831    pub fn input(
832        &self,
833        ty: &Type,
834        recipe: &ResolvedInput<C>,
835        extra: &C::Extra,
836        active: &Condition,
837    ) -> Result<CodecInputPlan<C>> {
838        self.input_named(ty, recipe, extra, active, None)
839    }
840    pub fn input_named(
841        &self,
842        ty: &Type,
843        recipe: &ResolvedInput<C>,
844        extra: &C::Extra,
845        active: &Condition,
846        names: Option<(&syn::Ident, &syn::Ident)>,
847    ) -> Result<CodecInputPlan<C>> {
848        self.codec
849            .compile_input_tree(self, ty, recipe, extra, active, names)
850    }
851    pub fn output_root(
852        &self,
853        ty: &Type,
854        recipe: &ResolvedOutput<C>,
855        extra: &C::Extra,
856        value: TokenStream,
857        borrowed: bool,
858        reference_return: bool,
859    ) -> Result<CodecOutputPlan<C>> {
860        self.codec
861            .compile_output_root(self, ty, recipe, extra, value, borrowed, reference_return)
862    }
863    pub fn output_return(
864        &self,
865        ty: &Type,
866        recipe: &ResolvedOutput<C>,
867        extra: &C::Extra,
868        value: TokenStream,
869        borrowed: bool,
870    ) -> Result<CodecOutputPlan<C>> {
871        self.codec
872            .compile_output_return(self, ty, recipe, extra, value, borrowed)
873    }
874    pub fn output(
875        &self,
876        ty: &Type,
877        recipe: &ResolvedOutput<C>,
878        extra: &C::Extra,
879        value: TokenStream,
880        borrowed: bool,
881    ) -> Result<CodecOutputPlan<C>> {
882        self.output_named(ty, recipe, extra, value, borrowed, None)
883    }
884    pub fn output_named(
885        &self,
886        ty: &Type,
887        recipe: &ResolvedOutput<C>,
888        extra: &C::Extra,
889        value: TokenStream,
890        borrowed: bool,
891        wire: Option<&syn::Ident>,
892    ) -> Result<CodecOutputPlan<C>> {
893        self.codec
894            .compile_output_tree(self, ty, recipe, extra, value, borrowed, wire)
895    }
896}
897/// Adapter-specific input operations stored in a resolved Flat.
898pub type ResolvedInput<C> = InputNode<<C as Codec>::InOps, <C as Codec>::OutOps>;
899/// Adapter-specific output operations stored in a resolved Flat.
900pub type ResolvedOutput<C> = OutputNode<<C as Codec>::InOps, <C as Codec>::OutOps>;
901fn mutable_reference(ty: &Type) -> bool {
902    matches!(ty, Type::Reference(r) if r.mutability.is_some())
903        || option_inner(ty).is_some_and(mutable_reference)
904}
905fn validate_nested_output<W: prebindgen_flat::WireKind, C, N, B>(
906    plan: &OutputPlan<W, C, N, B>,
907) -> Result<()> {
908    if plan.outcomes.source_error.is_some() || plan.outcomes.binding_failure.is_some() {
909        return Err("completion channels belong to the outermost output conversion".into());
910    }
911    Ok(())
912}
913/// Append one output view while preserving its preparation and ownership commit.
914pub fn append<W: prebindgen_flat::WireKind, C, N, B>(
915    to: &mut OutputPlan<W, C, N, B>,
916    from: OutputPlan<W, C, N, B>,
917) -> Result<()> {
918    if !from.delivery.statements.is_empty()
919        || from.delivery.result.to_string() != quote!(()).to_string()
920    {
921        return Err("composed output delivery must be installed by Codec::finish_output".into());
922    }
923    validate_nested_output(&from)?;
924    let WireNode::Branch {
925        children: elements, ..
926    } = &mut to.outcomes.success
927    else {
928        return Err("output components require an ordered group".into());
929    };
930    elements.push(from.outcomes.success);
931    to.fallible |= from.fallible;
932    to.commit.extend(from.commit);
933    to.statements.extend(from.statements);
934    to.values.extend(from.values);
935    prebindgen_conversions::validate_value_bindings(&to.values)
936}

938fn require_bulk<C: Codec + ?Sized>(codec: &C, direction: Direction) -> Result<()> {
939    if codec.supports_bulk(direction) {
940        Ok(())
941    } else {
942        Err(match direction {
943            Direction::Input => "bulk inputs are unsupported by this codec",
944            Direction::Output => "bulk outputs are unsupported by this codec",
945        }
946        .into())
947    }
948}
