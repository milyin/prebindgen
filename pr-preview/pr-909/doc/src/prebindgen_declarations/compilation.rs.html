7    pub fn with_temporary_receiver<T>(
8        &self,
9        temporary: bool,
10        prepare: impl FnOnce() -> Result<T>,
11    ) -> Result<T> {
12        let previous = self.temporary_receiver.get();
13        self.temporary_receiver.set(previous || temporary);
14        let result = prepare();
15        self.temporary_receiver.set(previous);
16        result
17    }
18    pub fn with_copy_permission<T>(
19        &self,
20        explicit: bool,
21        prepare: impl FnOnce() -> Result<T>,
22    ) -> Result<T> {
23        let previous = self
24            .copy_borrowed
25            .replace(self.copy_borrowed.get() || explicit);
26        let result = prepare();
27        self.copy_borrowed.set(previous);
28        result
29    }
30    pub fn name(&self) -> syn::Ident {
31        let next = self.next.get();
32        let n = format_ident!("{}_w{next}", self.prefix);
33        self.next.set(next + 1);
34        n
35    }

79    pub fn input_temporary(
80        &self,
81        ty: &Type,
82        extra: &C::Extra,
83        active: &Condition,
84        names: Option<(&syn::Ident, &syn::Ident)>,
85        inner: &ResolvedInput<C>,
86    ) -> Result<CodecInput<C>> {
87        {
88            let owned = temporary_input_type(ty)?;
89            let mut child = self.input_named(&owned, inner, extra, active, names)?;
90            let storage = self.name();
91            let target = format_ident!("{storage}_owned");
92            let mut backing = std::mem::take(&mut child.storage);
93            backing . extend (quote!(# [allow (unused_mut , unused_assignments)] let mut # storage : Option <# owned >= None ;)) ;
94            let build = child.bind(&target, quote!(# storage = Some (# target) ;))?;
95            let Type::Reference(reference) = ty else {
96                unreachable!()
97            };
98            let elem = &reference.elem;
99            let value = if matches!(elem.as_ref(), Type::Slice(_))
100                || matches!(elem . as_ref () , Type :: Path (p) if p . path . is_ident ("str"))
101            {
102                quote!(:: core :: convert :: AsRef ::<# elem >:: as_ref (# storage . as_ref () . expect ("active temporary")))
103            } else {
104                quote!(# storage . as_ref () . expect ("active temporary"))
105            };
106            let mut plan = InputPlan::new(
107                WireNode::Branch {
108                    controls: vec![],
109                    children: vec![child.node],
110                    meta: self.codec.group(GroupKind::Values)?,
111                },
112                quote!(# build ;),
113                value,
114            );
115            plan.storage = backing;
116            plan.fallible = child.fallible;
117            Ok(plan)
118        }
119    }
120    pub fn input_optional(
121        &self,
122        ty: &Type,
123        extra: &C::Extra,
124        active: &Condition,
125        names: Option<(&syn::Ident, &syn::Ident)>,
126        var: syn::Ident,
127        inner: &ResolvedInput<C>,
128    ) -> Result<CodecInput<C>> {
129        {
130            let ty = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
131            let tag = names
132                .map(|(wire, _)| format_ident!("{wire}_present"))
133                .unwrap_or_else(|| self.name());
134            let child = if let Some((wire, var)) = names {
135                self.input_named(
136                    ty,
137                    inner,
138                    extra,
139                    &active.arm(tag.clone(), 1),
140                    Some((&format_ident!("{wire}_some"), &format_ident!("{var}_some"))),
141                )?
142            } else {
143                self.input(ty, inner, extra, &active.arm(tag.clone(), 1))?
144            };
145            optional_input(self.codec, extra, tag, var, active, child)
146        }
147    }
148    pub fn input_fields(
149        &self,
150        ty: &Type,
151        extra: &C::Extra,
152        active: &Condition,
153        names: Option<(&syn::Ident, &syn::Ident)>,
154        var: syn::Ident,
155        children: &[ResolvedInput<C>],
156        grouped: bool,
157    ) -> Result<CodecInput<C>> {
158        {
159            if type_key(ty) == "()" && children.is_empty() {
160                return Ok(InputPlan::new(
161                    WireNode::Branch {
162                        controls: vec![],
163                        children: vec![],
164                        meta: self.codec.group(GroupKind::Fields)?,
165                    },
166                    quote!(),
167                    quote!(()),
168                ));
169            }
170            let r = self.flat.record(ty)?;
171            if r.fields.len() != children.len() {
172                return Err(format!(
173                    "`{}`: expected {} field converters, got {}",
174                    type_key(ty),
175                    r.fields.len(),
176                    children.len()
177                ));
178            }
179            let wire =
180                grouped.then(|| names.map(|(w, _)| w.clone()).unwrap_or_else(|| self.name()));
181            let mut plan = record_input(
182                self.codec,
183                ty,
184                var.clone(),
185                r.fields
186                    .iter()
187                    .zip(children)
188                    .enumerate()
189                    .map(|(index, (field, child))| {
190                        let plan = if let Some(wire) = &wire {
191                            self.input_named(
192                                &field.ty,
193                                child,
194                                extra,
195                                active,
196                                Some((
197                                    &format_ident!("{wire}_f{index}"),
198                                    &format_ident!("{var}_f{index}"),
199                                )),
200                            )?
201                        } else {
202                            self.input(&field.ty, child, extra, active)?
203                        };
204                        Ok((field.name.clone(), plan))
205                    }),
206            )?;
207            if let Some(wire) = wire {
208                self.codec
209                    .input_record_metadata(self.flat, ty, &wire, active, &mut plan, extra)?;
210            }
211            Ok(plan)
212        }
213    }
214    pub fn input_constructor(
215        &self,
216        ty: &Type,
217        extra: &C::Extra,
218        active: &Condition,
219        var: syn::Ident,
220        function: &String,
221        args: &[ResolvedInput<C>],
222    ) -> Result<CodecInput<C>> {
223        {
224            let f = self.flat.function(function)?;
225            if !same_type(&f.ret.ty, ty) {
226                return Err(format!(
227                    "constructor `{function}` returns `{}`, expected `{}`",
228                    type_key(&f.ret.ty),
229                    type_key(ty)
230                ));
231            }
232            if f.params.len() != args.len() {
233                return Err(format!("constructor `{function}`: wrong argument count"));
234            }
235            let mut plans = Vec::new();
236            let mut call_args = Vec::new();
237            let mut wires = Vec::new();
238            let mut storage = TokenStream::new();
239            let mut fallible = false;
240            for (i, (param, child)) in f.params.iter().zip(args).enumerate() {
241                let mut p = self.input(&param.ty.ty, child, extra, active)?;
242                self.prefix(
243                    &mut p.node,
244                    ComponentSegment::Argument(param.name.to_string()),
245                );
246                fallible |= p.fallible;
247                storage.extend(std::mem::take(&mut p.storage));
248                let target = format_ident!("{var}_argument_{i}");
249                call_args.push(target.clone());
250                plans.push((target, p));
251            }
252            let path = &f.path;
253            let mut plan = InputPlan::new(
254                WireNode::default(),
255                quote!(let # var :# ty =# path (# (# call_args) ,*) ;),
256                quote!(# var),
257            );
258            for (target, p) in plans.iter().rev() {
259                plan.code = p.bind(target, plan.code)?;
260            }
261            for (_, p) in plans {
262                wires.push(p.node);
263            }
264            plan.node = WireNode::Branch {
265                controls: vec![],
266                children: wires,
267                meta: self.codec.group(GroupKind::Values)?,
268            };
269            plan.storage = storage;
270            plan.fallible = fallible;
271            Ok(plan)
272        }
273    }
274    pub fn input_selector(
275        &self,
276        ty: &Type,
277        extra: &C::Extra,
278        active: &Condition,
279        var: syn::Ident,
280        arms: &[ResolvedInput<C>],
281    ) -> Result<CodecInput<C>> {
282        {
283            let mut fallible = self.codec.validates_inputs();
284            let mut storage = TokenStream::new();
285            let mut validation = TokenStream::new();
286            if arms.is_empty() {
287                return Err("an input selector needs at least one alternative".into());
288            }
289            if matches!(ty, Type::Reference(_)) {
290                return Err("selector results must be owned in this subset".into());
291            }
292            let tag = self.name();
293            let mut selector_layout = prebindgen_flat::WireNode::Leaf(vec![Wire {
294                name: tag.clone(),
295                kind: self.codec.selector(extra)?,
296            }]);
297            self.prefix(&mut selector_layout, ComponentSegment::Selector);
298            let WireNode::Leaf(mut selector_values) = selector_layout else {
299                return Err("selector prefixing must retain its value".into());
300            };
301            if selector_values.len() != 1 {
302                return Err("selector prefixing must retain its value".into());
303            }
304            let selector = selector_values.remove(0);
305            let mut alternatives = Vec::new();
306            let count = arms.len() as i32;
307            let invalid = active.and(quote!(! (0 ..# count) . contains (&# tag)));
308            if self.codec.validates_inputs() {
309                validation
310                    .extend(quote!(if # invalid { return Err ("invalid selector" . into ()) ; }));
311            }
312            let mut cases = Vec::new();
313            for (i, arm) in arms.iter().enumerate() {
314                let mut p = self.input(ty, arm, extra, &active.arm(tag.clone(), i as i32))?;
315                self.prefix(&mut p.node, ComponentSegment::Alternative(i));
316                fallible |= p.fallible;
317                storage.extend(std::mem::take(&mut p.storage));
318                let target = format_ident!("{var}_alternative_{i}");
319                let body = p.bind(&target, quote!(# target))?;
320                alternatives.push(p.node);
321                let i = i as i32;
322                cases.push(quote!(# i =># body));
323            }
324            let metadata = self.codec.alternatives(Alternatives {
325                selector: &selector,
326                arms: &mut alternatives,
327            })?;
328            let node = WireNode::Branch {
329                controls: vec![selector],
330                meta: metadata,
331                children: alternatives,
332            };
333            let invalid = if self.codec.validates_inputs() {
334                quote!(unreachable!("validated selector"))
335            } else {
336                quote!(unsafe { ::core::hint::unreachable_unchecked() })
337            };
338            let mut plan = InputPlan::new(
339                node,
340                quote!(# validation let # var :# ty = match # tag { # (# cases ,) * _ =># invalid } ;),
341                quote!(# var),
342            );
343            plan.storage = storage;
344            plan.fallible = fallible;
345            Ok(plan)
346        }
347    }

441    pub fn output_projection(
442        &self,
443        ty: &Type,
444        extra: &C::Extra,
445        value: TokenStream,
446        borrowed: bool,
447        wire_name: Option<&syn::Ident>,
448        projection: &Projection,
449        representation: Representation,
450    ) -> Result<CodecOutput<C>> {
451        {
452            if borrowed && projection.mode == ProjectionMode::Owned {
453                return Err("owned projection cannot consume a shared-reference occurrence".into());
454            }
455            let value = if !borrowed && projection.mode == ProjectionMode::Borrowed {
456                quote!(& (# value))
457            } else {
458                value
459            };
460            self.codec.output_projected(
461                OutputLeaf {
462                    flat: self.flat,
463                    ty,
464                    extra,
465                    wire: wire_name.cloned().unwrap_or_else(|| self.name()),
466                    value,
467                    access: if projection.mode == ProjectionMode::Borrowed {
468                        ValueAccess::Shared
469                    } else {
470                        ValueAccess::Owned
471                    },
472                    representation,
473                },
474                projection,
475            )
476        }
477    }
478    pub fn output_fields(
479        &self,
480        ty: &Type,
481        extra: &C::Extra,
482        value: TokenStream,
483        borrowed: bool,
484        wire_name: Option<&syn::Ident>,
485        children: &[ResolvedOutput<C>],
486        grouped: bool,
487        owned: bool,
488        explicit_copy: bool,
489        mut out: CodecOutput<C>,
490    ) -> Result<CodecOutput<C>> {
491        {
492            let wire = grouped.then(|| wire_name.cloned().unwrap_or_else(|| self.name()));
493            if owned && borrowed {
494                return Err("owned fields cannot consume a shared-reference occurrence".into());
495            }
496            if type_key(ty) == "()" && children.is_empty() {
497                out.outcomes.success = WireNode::Branch {
498                    controls: vec![],
499                    meta: self.codec.group(GroupKind::Fields)?,
500                    children: vec![],
501                };
502                return Ok(out);
503            }
504            let r = self.flat.record(ty)?;
505            if r.fields.len() != children.len() {
506                return Err("wrong output field count".into());
507            }
508            let convert = |index: usize, value| {
509                let child_wire = wire.as_ref().map(|w| format_ident!("{w}_f{index}"));
510                self.output_named(
511                    &r.fields[index].ty,
512                    &children[index],
513                    extra,
514                    value,
515                    !owned,
516                    child_wire.as_ref(),
517                )
518            };
519            let var = wire
520                .as_ref()
521                .map(|w| format_ident!("{w}_record"))
522                .unwrap_or_else(|| self.name());
523            let mut plan = self.with_copy_permission(explicit_copy, || {
524                self.with_temporary_receiver(!owned && !borrowed, || {
525                    if owned {
526                        record_output_owned(
527                            self.codec,
528                            var,
529                            value,
530                            r.fields.iter().map(|f| f.name.clone()),
531                            convert,
532                        )
533                    } else {
534                        record_output(
535                            self.codec,
536                            var,
537                            value,
538                            r.fields.iter().map(|f| f.name.clone()),
539                            convert,
540                        )
541                    }
542                })
543            })?;
544            if let Some(wire) = wire {
545                self.codec
546                    .output_record_metadata(self.flat, ty, &wire, &mut plan, extra)?;
547            }
548            Ok(plan)
549        }
550    }
551    pub fn output_getter(
552        &self,
553        ty: &Type,
554        extra: &C::Extra,
555        value: TokenStream,
556        borrowed: bool,
557        function: &String,
558        output: &ResolvedOutput<C>,
559        consuming: bool,
560        mut out: CodecOutput<C>,
561    ) -> Result<CodecOutput<C>> {
562        {
563            if consuming && borrowed {
564                return Err(
565                    "consuming accessor cannot consume a shared-reference occurrence".into(),
566                );
567            }
568            let f = self.flat.function(function)?;
569            let expected: Type = if consuming {
570                ty.clone()
571            } else {
572                syn::parse_quote!(&# ty)
573            };
574            if f.params.len() != 1 || !same_type(&f.params[0].ty.ty, &expected) {
575                return Err(format!(
576                    "getter `{function}` must take exactly `{}`",
577                    type_key(&expected)
578                ));
579            }
580            if matches!(& f . ret . ty , Type :: Reference (r) if r . mutability . is_some ()) {
581                return Err("mutable-reference getter results are unsupported".into());
582            }
583            let v = self.name();
584            let path = &f.path;
585            let arg = if borrowed || consuming {
586                value
587            } else {
588                quote!(&# value)
589            };
590            out.statements = quote!(let # v =# path (# arg) ;);
591            let mut child = self.output_return(&f.ret.ty, output, extra, quote!(# v), false)?;
592            self.codec.validate_getter_output(
593                self.flat,
594                ty,
595                borrowed && !self.temporary_receiver.get(),
596                &f.ret.ty,
597                &child,
598                extra,
599            )?;
600            self.prefix(
601                &mut child.outcomes.success,
602                ComponentSegment::Getter(function.clone()),
603            );
604            append(&mut out, child)?;
605            Ok(out)
606        }
607    }

609    #[allow(clippy::too_many_arguments)]
610    pub fn input_conversion(
611        &self,
612        ty: &Type,
613        conversion: &ValueConversion,
614        input: &ResolvedInput<C>,
615        extra: &C::Extra,
616        active: &Condition,
617        names: Option<(&syn::Ident, &syn::Ident)>,
618    ) -> Result<CodecInput<C>> {
619        crate::convert::validate_conversion(self.flat, ty, conversion, true)?;
620        let target = names.map(|(_, v)| v.clone()).unwrap_or_else(|| self.name());
621        let repr = format_ident!("{target}_repr");
622        let decoded = format_ident!("{repr}_decoded");
623        let child_names = names.map(|(w, _)| (w, &decoded));
624        let mut child = self.input_named(
625            &conversion.representation,
626            input,
627            extra,
628            active,
629            child_names,
630        )?;
631        let (expr, fallible) = match &conversion.source {
632            ValueSource::Function(name) => {
633                let f = self.flat.function(name)?;
634                let path = &f.path;
635                (
636                    quote!(#path(#repr)),
637                    prebindgen_flat::result_parts(&f.ret.ty).is_some() && !same_type(&f.ret.ty, ty),
638                )
639            }
640            ValueSource::Into { fallible } => {
641                let r = &conversion.representation;
642                if *fallible {
643                    (
644                        quote!(<#r as ::core::convert::TryInto<#ty>>::try_into(#repr)),
645                        true,
646                    )
647                } else {
648                    (
649                        quote!(<#r as ::core::convert::Into<#ty>>::into(#repr)),
650                        false,
651                    )
652                }
653            }
654        };
655        let expr = if fallible {
656            quote!((#expr).map_err(|error|::std::string::ToString::to_string(&error))?)
657        } else {
658            expr
659        };
660        let rest: CodecInput<C> = prebindgen_conversions::InputPlan::new(
661            child.node.clone(),
662            quote!(let #target:#ty=#expr;),
663            quote!(#target),
664        );
665        let storage = std::mem::take(&mut child.storage);
666        let code = child.bind(&repr, rest.code)?;
667        Ok(prebindgen_conversions::InputPlan {
668            fallible: child.fallible || fallible,
669            node: child.node,
670            storage,
671            code,
672            context: Default::default(),
673        })
674    }
675    pub fn output_conversion(
676        &self,
677        ty: &Type,
678        conversion: &ValueConversion,
679        output: &ResolvedOutput<C>,
680        extra: &C::Extra,
681        value: TokenStream,
682        borrowed: bool,
683    ) -> Result<CodecOutput<C>> {
684        crate::convert::validate_conversion(self.flat, ty, conversion, false)?;
685        let borrowed = borrowed || matches!(ty, Type::Reference(_));
686        let source = self.name();
687        let repr = self.name();
688        let (expr, fallible) = match &conversion.source {
689            ValueSource::Function(name) => {
690                let f = self.flat.function(name)?;
691                let path = &f.path;
692                let shared =
693                    matches!(&f.params[0].ty.ty,Type::Reference(r) if r.mutability.is_none());
694                if !shared && borrowed {
695                    return Err(format!("consuming conversion `{name}` cannot convert a borrowed source; use a shared-reference conversion"));
696                }
697                let argument = if shared && !borrowed {
698                    quote!(&#source)
699                } else {
700                    quote!(#source)
701                };
702                (
703                    quote!(#path(#argument)),
704                    prebindgen_flat::result_parts(&f.ret.ty).is_some()
705                        && !same_type(&f.ret.ty, &conversion.representation),
706                )
707            }
708            ValueSource::Into { fallible } => {
709                if borrowed {
710                    return Err("consuming trait conversion cannot convert a borrowed source; use a shared-reference helper".into());
711                }
712                let r = &conversion.representation;
713                if *fallible {
714                    (
715                        quote!(<#ty as ::core::convert::TryInto<#r>>::try_into(#source)),
716                        true,
717                    )
718                } else {
719                    (
720                        quote!(<#ty as ::core::convert::Into<#r>>::into(#source)),
721                        false,
722                    )
723                }
724            }
725        };
726        let expr = if fallible {
727            quote!((#expr).map_err(|error|::std::string::ToString::to_string(&error))?)
728        } else {
729            expr
730        };
731        let mut child = self.output_return(
732            &conversion.representation,
733            output,
734            extra,
735            quote!(#repr),
736            false,
737        )?;
738        self.codec.validate_getter_output(
739            self.flat,
740            ty,
741            borrowed && !self.temporary_receiver.get(),
742            &conversion.representation,
743            &child,
744            extra,
745        )?;
746        let prepared = child.statements;
747        child.statements = quote!(let #source=#value;let #repr=#expr;#prepared);
748        child.fallible |= fallible;
749        Ok(child)
750    }

752    #[allow(clippy::too_many_arguments)]
753    pub fn enum_input(
754        &self,
755        ty: &Type,
756        recipes: &[&[ResolvedInput<C>]],
757        value: bool,
758        extra: &C::Extra,
759        active: &Condition,
760        names: Option<(&syn::Ident, &syn::Ident)>,
761    ) -> Result<CodecInput<C>> {
762        if matches!(ty, Type::Reference(_)) {
763            return Err("enum-field input constructs an owned value; select reconstruct() to borrow a temporary".into());
764        }
765        let source = self.flat.payload_enum(ty)?;
766        if source.variants.len() != recipes.len() {
767            return Err("wrong enum variant count".into());
768        }
769        i32::try_from(recipes.len()).map_err(|_| "too many enum variants")?;
770        let root = names.map(|(w, _)| w.clone()).unwrap_or_else(|| self.name());
771        let var = names.map(|(_, v)| v.clone()).unwrap_or_else(|| self.name());
772        let tag = format_ident!("{root}_tag");
773        let mut selector = Wire {
774            name: tag.clone(),
775            kind: self.codec.selector(extra)?,
776        };
777        let mut arms = vec![];
778        let mut cases = vec![];
779        let mut storage = TokenStream::new();
780        let mut fallible = self.codec.validates_inputs();
781        for (index, (arm, recipes)) in source.variants.iter().zip(recipes).enumerate() {
782            let recipes = *recipes;
783            if arm.fields.len() != recipes.len() {
784                return Err("wrong enum field count".into());
785            }
786            let mut plans = vec![];
787            let mut arguments = vec![];
788            let mut children = vec![];
789            let condition = active.arm(tag.clone(), index as i32);
790            for (field_index, (field, recipe)) in arm.fields.iter().zip(recipes).enumerate() {
791                let wire = format_ident!("{root}_a{index}_f{field_index}");
792                let target = format_ident!("{var}_a{index}_f{field_index}");
793                let converted = format_ident!("{target}_value");
794                let previous = self.input_role.get();
795                if value {
796                    self.input_role.set(InputRole::EnumField);
797                }
798                let result = self.input_named(
799                    &field.ty,
800                    recipe,
801                    extra,
802                    &condition,
803                    Some((&wire, &converted)),
804                );
805                self.input_role.set(previous);
806                let mut child = result
807                    .map_err(|e| format!("{}::{}.{}: {e}", type_key(ty), arm.name, field.key()))?;
808                let name = format_ident!("{}", field.property());
809                if arm.fields.len() != 1 || field.key() != "0" {
810                    self.prefix(&mut child.node, ComponentSegment::Field(field.property()));
811                }
812                child.node = WireNode::Branch {
813                    controls: vec![],
814                    children: vec![child.node],
815                    meta: self.codec.group(GroupKind::Field(&name))?,
816                };
817                storage.extend(std::mem::take(&mut child.storage));
818                fallible |= child.fallible;
819                arguments.push(quote!(#target));
820                children.push(child.node.clone());
821                plans.push((target, child));
822            }
823            let constructed = source.spell(index, &arguments)?;
824            let mut body = quote!(#constructed);
825            for (target, child) in plans.iter().rev() {
826                body = child.bind(target, body)?;
827            }
828            let index = index as i32;
829            cases.push(quote!(#index=>{#body}));
830            let mut arm_node = WireNode::Branch {
831                controls: vec![],
832                children,
833                meta: self.codec.group(GroupKind::Fields)?,
834            };
835            self.prefix(
836                &mut arm_node,
837                ComponentSegment::Named(arm.name.to_string().trim_start_matches("r#").into()),
838            );
839            arms.push(arm_node);
840        }
841        let encoding = self.codec.enum_variants(EnumVariants {
842            flat: self.flat,
843            ty,
844            extra,
845            selector: &mut selector,
846            arms: &mut arms,
847            form: if value {
848                EnumForm::Value
849            } else {
850                EnumForm::Fields
851            },
852            direction: Direction::Input,
853        })?;
854        let invalid = if self.codec.validates_inputs() {
855            quote!(return Err("invalid enum tag".into()))
856        } else {
857            quote!(unsafe { ::core::hint::unreachable_unchecked() })
858        };
859        let mut plan = InputPlan::new(
860            WireNode::Branch {
861                controls: vec![selector],
862                children: arms,
863                meta: encoding.metadata,
864            },
865            quote!(let #var:#ty=match #tag {#(#cases,)* _=>#invalid};),
866            quote!(#var),
867        );
868        plan.storage = storage;
869        plan.fallible = fallible;
870        if value {
871            self.codec.input_enum(
872                self.flat,
873                ty,
874                extra,
875                root,
876                plan,
877                active,
878                self.input_role.get(),
879            )
880        } else {
881            Ok(plan)
882        }
883    }

885    #[allow(clippy::too_many_arguments)]
886    pub fn enum_output(
887        &self,
888        ty: &Type,
889        recipes: &[&[ResolvedOutput<C>]],
890        grouped: bool,
891        mode: Option<ProjectionMode>,
892        extra: &C::Extra,
893        value: TokenStream,
894        borrowed: bool,
895        wire: Option<&syn::Ident>,
896    ) -> Result<CodecOutput<C>> {
897        let owned = mode.map_or(!borrowed, |m| m == ProjectionMode::Owned);
898        if owned && borrowed {
899            return Err("owned enum fields cannot consume a shared-reference occurrence".into());
900        }
901        let temporary_observation = !owned && !borrowed;
902        let observed = format_ident!(
903            "__enum_observed_{}",
904            wire.cloned().unwrap_or_else(|| self.name())
905        );
906        let prepare = if !owned && !borrowed {
907            quote!(let #observed=#value;)
908        } else {
909            TokenStream::new()
910        };
911        let value = if !owned && !borrowed {
912            quote!(&#observed)
913        } else {
914            value
915        };
916        let borrowed = !owned;
917        let source = self.flat.payload_enum(ty)?;
918        if source.variants.len() != recipes.len() {
919            return Err("wrong enum variant count".into());
920        }
921        i32::try_from(recipes.len()).map_err(|_| "too many enum variants")?;
922        let root = wire.cloned().unwrap_or_else(|| self.name());
923        let tag = format_ident!("{root}_tag");
924        let mut selector = Wire {
925            name: tag.clone(),
926            kind: self.codec.selector(extra)?,
927        };
928        let mut plans = vec![];
929        let mut patterns = vec![];
930        for (index, (arm, recipes)) in source.variants.iter().zip(recipes).enumerate() {
931            let recipes = *recipes;
932            if arm.fields.len() != recipes.len() {
933                return Err("wrong enum field count".into());
934            }
935            let bindings = (0..arm.fields.len())
936                .map(|i| format_ident!("__{root}_a{index}_b{i}"))
937                .collect::<Vec<_>>();
938            patterns.push(source.spell(
939                index,
940                &bindings.iter().map(|b| quote!(#b)).collect::<Vec<_>>(),
941            )?);
942            let mut child = OutputPlan {
943                fallible: false,
944                context: Default::default(),
945                delivery: Default::default(),
946                commit: TokenStream::new(),
947                statements: TokenStream::new(),
948                values: vec![],
949                outcomes: prebindgen_flat::Outcomes {
950                    success: WireNode::Branch {
951                        controls: vec![],
952                        children: vec![],
953                        meta: self.codec.group(GroupKind::Fields)?,
954                    },
955                    source_error: None,
956                    binding_failure: None,
957                },
958            };
959            for (field_index, (field, recipe)) in arm.fields.iter().zip(recipes).enumerate() {
960                let binding = &bindings[field_index];
961                let wire = format_ident!("{root}_a{index}_f{field_index}");
962                let mut field_plan = self
963                    .with_copy_permission(mode == Some(ProjectionMode::Borrowed), || {
964                        self.with_temporary_receiver(temporary_observation, || {
965                            self.output_named(
966                                &field.ty,
967                                recipe,
968                                extra,
969                                quote!(#binding),
970                                borrowed,
971                                Some(&wire),
972                            )
973                        })
974                    })
975                    .map_err(|e| format!("{}::{}.{}: {e}", type_key(ty), arm.name, field.key()))?;
976                let name = format_ident!("{}", field.property());
977                if arm.fields.len() != 1 || field.key() != "0" {
978                    self.prefix(
979                        &mut field_plan.outcomes.success,
980                        ComponentSegment::Field(field.property()),
981                    );
982                }
983                field_plan.outcomes.success = WireNode::Branch {
984                    controls: vec![],
985                    children: vec![field_plan.outcomes.success],
986                    meta: self.codec.group(GroupKind::Field(&name))?,
987                };
988                append(&mut child, field_plan)?;
989            }
990            self.prefix(
991                &mut child.outcomes.success,
992                ComponentSegment::Named(arm.name.to_string().trim_start_matches("r#").into()),
993            );
994            plans.push(child);
995        }
996        let mut arms = plans
997            .iter_mut()
998            .map(|p| std::mem::take(&mut p.outcomes.success))
999            .collect::<Vec<_>>();
1000        let encoding = self.codec.enum_variants(EnumVariants {
1001            flat: self.flat,
1002            ty,
1003            extra,
1004            selector: &mut selector,
1005            arms: &mut arms,
1006            form: if grouped {
1007                EnumForm::Value
1008            } else {
1009                EnumForm::Fields
1010            },
1011            direction: Direction::Output,
1012        })?;
1013        if encoding.inactive.len() != plans.len() {
1014            return Err("missing inactive enum arm values".into());
1015        }
1016        let mut defaults = vec![];
1017        for (child, absent) in plans.iter().zip(encoding.inactive) {
1018            let mut absent = absent
1019                .into_iter()
1020                .map(|(n, v)| (n.to_string(), v))
1021                .collect::<BTreeMap<_, _>>();
1022            defaults.push(
1023                child
1024                    .values
1025                    .iter()
1026                    .map(|(n, _)| {
1027                        absent
1028                            .remove(&n.to_string())
1029                            .ok_or_else(|| format!("missing inactive enum value `{n}`"))
1030                    })
1031                    .collect::<Result<Vec<_>>>()?,
1032            );
1033            if !absent.is_empty() {
1034                return Err("unexpected inactive enum output".into());
1035            }
1036        }
1037        // One statically typed active-owner guard, with no boxing or per-arm
1038        // Option storage. A failed later conversion drops the selected guard.
1039        let guard_type = format_ident!("__PbEnumCommit_{root}");
1040        let guard = format_ident!("__{root}_commit");
1041        let cases = (0..plans.len())
1042            .map(|i| format_ident!("A{i}"))
1043            .collect::<Vec<_>>();
1044        let parameters = (0..plans.len())
1045            .map(|i| format_ident!("F{i}"))
1046            .collect::<Vec<_>>();
1047        let root_ref = &root;
1048        let locals = plans
1049            .iter()
1050            .enumerate()
1051            .flat_map(|(i, p)| {
1052                (0..p.values.len()).map(move |j| format_ident!("__{root_ref}_a{i}_value{j}"))
1053            })
1054            .collect::<Vec<_>>();
1055        let values = std::iter::once((tag.clone(), quote!(#tag)))
1056            .chain(
1057                plans
1058                    .iter()
1059                    .flat_map(|p| p.values.iter().map(|(n, _)| n.clone()))
1060                    .zip(locals.iter().map(|n| quote!(#n))),
1061            )
1062            .collect();
1063        let mut bodies = vec![];
1064        let mut fallible = false;
1065        for (index, (plan, pattern)) in plans.iter().zip(patterns).enumerate() {
1066            let tag = index as i32;
1067            let body = &plan.statements;
1068            let commit = &plan.commit;
1069            let case = &cases[index];
1070            let fields = plans
1071                .iter()
1072                .enumerate()
1073                .flat_map(|(i, p)| {
1074                    if i == index {
1075                        p.values.iter().map(|(_, v)| v.clone()).collect::<Vec<_>>()
1076                    } else {
1077                        defaults[i].clone()
1078                    }
1079                })
1080                .collect::<Vec<_>>();
1081            bodies.push(quote!(#pattern => { #body let values=(#(#fields,)*); (#tag,values,#guard_type::#case(move ||{#commit})) }));
1082            fallible |= plan.fallible;
1083        }
1084        let statements = quote! {
1085            #prepare
1086            #[allow(non_camel_case_types)] enum #guard_type<#(#parameters),*> { #(#cases(#parameters)),* }
1087            let (#tag,(#(#locals,)*),#guard)=match #value {#(#bodies,)*};
1088        };
1089        let commit = quote!(match #guard {#(#guard_type::#cases(action)=>action(),)*});
1090        let plan = OutputPlan {
1091            fallible,
1092            context: Default::default(),
1093            delivery: Default::default(),
1094            statements,
1095            commit,
1096            values,
1097            outcomes: prebindgen_flat::Outcomes {
1098                success: WireNode::Branch {
1099                    controls: vec![selector],
1100                    children: arms,
1101                    meta: encoding.metadata,
1102                },
1103                source_error: None,
1104                binding_failure: None,
1105            },
1106        };
1107        if grouped {
1108            self.codec.output_enum(self.flat, ty, extra, root, plan)
1109        } else {
1110            Ok(plan)
1111        }
1112    }
1113}

1116impl<C: Codec> Compiler<'_, C> {
1117    pub fn output_result(
1118        &self,
1119        ty: &Type,
1120        ok: &ResolvedOutput<C>,
1121        error: &Option<String>,
1122        extra: &C::Extra,
1123        value: TokenStream,
1124        borrowed: bool,
1125        reference_return: bool,
1126    ) -> Result<CodecOutput<C>> {
1127        if borrowed {
1128            return Err("Result error routing requires an owned Result occurrence".into());
1129        }
1130        let (success_ty, error_ty) =
1131            result_parts(ty).ok_or("Result recipe requires Result<T, E>")?;
1132        let payload = self.name();
1133        let failure = self.name();
1134        let mapped = if let Some(error) = error {
1135            let mapper = self.flat.function(error)?;
1136            let mapper_arg = mapper.params.first().map(|p| &p.ty.ty);
1137            let shared = matches!(mapper_arg , Some (Type :: Reference (r)) if r . mutability . is_none () && same_type (& r . elem , error_ty));
1138            if mapper.params.len() != 1
1139                || !(mapper_arg.is_some_and(|ty| same_type(ty, error_ty)) || shared)
1140                || type_key(&mapper.ret.ty) != "String"
1141            {
1142                return Err(
1143                    "Result error mapper must take exactly E or &E and return String".into(),
1144                );
1145            }
1146            let path = &mapper.path;
1147            let argument = if shared {
1148                quote!(&# failure)
1149            } else {
1150                quote!(# failure)
1151            };
1152            Some(quote!(# path (# argument)))
1153        } else {
1154            None
1155        };
1156        let error = self.codec.source_error(SourceError {
1157            flat: self.flat,
1158            ty: error_ty,
1159            value: quote!(# failure),
1160            message: mapped,
1161            extra,
1162        })?;
1163        let message = error.propagate;
1164        let mut child = if reference_return {
1165            self.output_return(success_ty, ok, extra, quote!(# payload), false)?
1166        } else {
1167            self.output(success_ty, ok, extra, quote!(# payload), false)?
1168        };
1169        let prepare = child.statements;
1170        child.statements = quote! { let # payload = match # value { :: core :: result :: Result :: Ok (value) => value , :: core :: result :: Result :: Err (# failure) => return Err (# message) , } ; # prepare };
1171        child.outcomes.source_error = Some(error.payload);
1172        Ok(child)
1173    }
1174    pub fn input_sequence(
1175        &self,
1176        ty: &Type,
1177        element_recipe: &ResolvedInput<C>,
1178        extra: &C::Extra,
1179        active: &Condition,
1180        names: Option<(&syn::Ident, &syn::Ident)>,
1181        var: syn::Ident,
1182        bulk: bool,
1183    ) -> Result<CodecInput<C>> {
1184        {
1185            if matches!(ty, Type::Reference(_)) {
1186                let next = self.next.get();
1187                let wire = names
1188                    .map(|(wire, _)| wire.clone())
1189                    .unwrap_or_else(|| self.name());
1190                if let Some(plan) = self.codec.input_borrowed_sequence(
1191                    self.flat,
1192                    ty,
1193                    extra,
1194                    wire,
1195                    var.clone(),
1196                    element_recipe,
1197                    active,
1198                )? {
1199                    return Ok(plan);
1200                }
1201                self.next.set(next);
1202            }
1203            let (element, _) =
1204                sequence_parts(ty).ok_or("sequence input requires Vec<T>, &[T] or &Vec<T>")?;
1205            let wire = self.name();
1206            let length = self.name();
1207            let index = self.name();
1208            let child = self.input(element, element_recipe, extra, &Condition::default());

1215    pub fn output_sequence(
1216        &self,
1217        ty: &Type,
1218        element: &ResolvedOutput<C>,
1219        extra: &C::Extra,
1220        value: TokenStream,
1221        borrowed: bool,
1222        delivery: &SequenceDelivery,
1223        mode: &Option<ProjectionMode>,
1224        bulk: bool,
1225    ) -> Result<CodecOutput<C>> {
1226        {
1227            let (element_type, _) =
1228                sequence_parts(ty).ok_or("sequence output requires a collection")?;
1229            let explicit_copy = *mode == Some(ProjectionMode::Borrowed);
1230            let mode = mode.unwrap_or_else(|| {
1231                if borrowed || !sequence_parts(ty).unwrap().1.is_owned() {
1232                    ProjectionMode::Borrowed
1233                } else {
1234                    ProjectionMode::Owned
1235                }
1236            });
1237            let wire = self.name();
1238            if mode == ProjectionMode::Borrowed && !explicit_copy && !self.copy_borrowed.get() {
1239                if let Some(plan) = self.codec.output_borrowed_sequence(
1240                    self.flat,
1241                    ty,
1242                    extra,
1243                    wire.clone(),
1244                    value.clone(),
1245                    element,
1246                    *delivery,
1247                    self.temporary_receiver.get(),
1248                )? {
1249                    return Ok(plan);
1250                }
1251            }
1252            let payload = self.name();
1253            let length = self.name();
1254            let index = self.name();
1255            let child = self.with_copy_permission(explicit_copy, || {
1256                self.with_temporary_receiver(
1257                    mode == ProjectionMode::Borrowed
1258                        && !borrowed
1259                        && sequence_parts(ty).unwrap().1.is_owned(),
1260                    || {
1261                        self.output(
1262                            element_type,
1263                            element,
1264                            extra,
1265                            quote!(# payload),
1266                            mode == ProjectionMode::Borrowed,
1267                        )
1268                    },
1269                )
1270            });
1271            let child = child?;
1272            sequence::output(
1273                self.codec,
1274                ty,
1275                extra,
1276                wire,
1277                payload,
1278                length,
1279                index,
1280                value,
1281                borrowed,
1282                mode,
1283                child,
1284                if *delivery == SequenceDelivery::Auto {
1285                    SequenceDelivery::Value
1286                } else {
1287                    *delivery
1288                },
1289                bulk,
1290            )
1291        }
1292    }
1293    pub fn output_owner_getters(
1294        &self,
1295        ty: &Type,
1296        parts: &[Option<(&str, &ResolvedOutput<C>)>],
1297        owner: usize,
1298        extra: &C::Extra,
1299        value: TokenStream,
1300    ) -> Result<Option<CodecOutput<C>>> {
1301        {
1302            let next = self.next.get();
1303            let wire = self.name();
1304            let Some((mut owned, pointer)) =
1305                self.codec.output_owner(self.flat, ty, extra, wire, value)?
1306            else {
1307                self.next.set(next);
1308                return Ok(None);
1309            };
1310            if parts
1311                .iter()
1312                .enumerate()
1313                .any(|(i, part)| i != owner && part.is_none())
1314            {
1315                return Err("an owner-backed output combination supports only the owning identity and getter views".into());
1316            }
1317            let prepare_owner = std::mem::take(&mut owned.statements);
1318            let mut result = OutputPlan {
1319                fallible: false,
1320                context: Default::default(),
1321                delivery: Default::default(),
1322                commit: TokenStream::new(),
1323                statements: TokenStream::new(),
1324                values: vec![],
1325                outcomes: prebindgen_flat::Outcomes {
1326                    success: WireNode::Branch {
1327                        controls: vec![],
1328                        children: vec![],
1329                        meta: self.codec.group(GroupKind::Values)?,
1330                    },
1331                    source_error: None,
1332                    binding_failure: None,
1333                },
1334            };
1335            let mut owner_plan = Some(owned);
1336            for (index, part) in parts.iter().enumerate() {
1337                let mut plan = if index == owner {
1338                    owner_plan.take().unwrap()
1339                } else {
1340                    let (function, output) = part.expect("validated getter");
1341                    let f = self.flat.function(function)?;
1342                    if f.params.len() != 1 {
1343                        return Err(format!(
1344                            "getter `{function}` must borrow its returned owner"
1345                        ));
1346                    }
1347                    let receiver = &f.params[0].ty.ty;
1348                    let mutable_receiver =
1349                        matches!(receiver , Type :: Reference (r) if r . mutability . is_some ());
1350                    let mutable = mutable_receiver || mutable_reference(&f.ret.ty);
1351                    if mutable && parts.len() != 2 {
1352                        return Err ("a mutable owner getter must be the only getter view; simultaneous owner borrows could invalidate earlier views" . into ()) ;
1353                    }
1354                    let expected: Type = if mutable_receiver {
1355                        syn::parse_quote!(& mut # ty)
1356                    } else {
1357                        syn::parse_quote!(&# ty)
1358                    };
1359                    if !same_type(receiver, &expected) {
1360                        return Err(format!(
1361                            "getter `{function}` must borrow its returned owner"
1362                        ));
1363                    }
1364                    let source = self.name();
1365                    let path = &f.path;
1366                    let argument = if mutable_receiver {
1367                        quote!(unsafe { & mut * (# pointer) })
1368                    } else {
1369                        quote!(unsafe { &* (# pointer) })
1370                    };
1371                    let mut plan =
1372                        self.output_return(&f.ret.ty, output, extra, quote!(# source), false)?;
1373                    self.codec
1374                        .validate_owner_getter(self.flat, ty, function, &plan)?;
1375                    plan.statements = {
1376                        let body = plan.statements;
1377                        quote!(let # source = # path (# argument) ; # body)
1378                    };
1379                    self.prefix(
1380                        &mut plan.outcomes.success,
1381                        ComponentSegment::Getter(function.to_owned()),
1382                    );
1383                    plan
1384                };
1385                self.prefix(&mut plan.outcomes.success, ComponentSegment::View(index));
1386                append(&mut result, plan)?;
1387            }
1388            result.statements = {
1389                let views = result.statements;
1390                quote!(# prepare_owner # views)
1391            };
1392            Ok(Some(result))
1393        }
1394    }
1395}
1396
1397#[allow(missing_docs)]
1398impl<C: Codec> Compiler<'_, C> {
1399    pub fn output_reference_views(
1400        &self,
1401        ty: &Type,
1402        parts: &[ResolvedOutput<C>],
1403        extra: &C::Extra,
1404        value: TokenStream,
1405    ) -> Result<CodecOutput<C>> {
1406        if parts.is_empty() {
1407            return Err("an output combination needs at least one part".into());
1408        }
1409        if mutable_reference(ty) && parts.len() > 1 {
1410            return Err("a mutable reference cannot supply multiple output views".into());
1411        }
1412        let mut result = OutputPlan {
1413            fallible: false,
1414            context: Default::default(),
1415            delivery: Default::default(),
1416            commit: TokenStream::new(),
1417            statements: TokenStream::new(),
1418            values: vec![],
1419            outcomes: prebindgen_flat::Outcomes {
1420                success: WireNode::Branch {
1421                    controls: vec![],
1422                    children: vec![],
1423                    meta: self.codec.group(GroupKind::Values)?,
1424                },
1425                source_error: None,
1426                binding_failure: None,
1427            },
1428        };
1429        for (index, part) in parts.iter().enumerate() {
1430            let mut child = self.output_return(ty, part, extra, value.clone(), false)?;
1431            self.prefix(&mut child.outcomes.success, ComponentSegment::View(index));
1432            append(&mut result, child)?;
1433        }
1434        Ok(result)
1435    }
