1//! Source structure of data-carrying enum variants.
2use super::*;
3use quote::quote;

5/// A Rust enum represented by a selected variant and recursively converted fields.
6#[derive(Clone, Debug)]
7pub struct PayloadEnum<V = (), F = ()> {
8    /// Qualified source type, including the arguments of a standard Result.
9    pub ty: Type,
10    /// Source variants in wire-tag order, starting at zero.
11    pub variants: Vec<PayloadVariant<V, F>>,
12}
13
14/// Source spelling used to construct or match an enum alternative.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum VariantStyle {
17    /// `Variant`.
18    Unit,
19    /// `Variant(...)`, including an empty tuple constructor.
20    Tuple,
21    /// `Variant { ... }`, including an empty named constructor.
22    Named,
23}
24
25/// One alternative of a data-carrying enum.
26#[derive(Clone, Debug)]
27pub struct PayloadVariant<V = (), F = ()> {
28    /// Generator declaration metadata.
29    pub metadata: V,
30    /// Source variant name, independent of a foreign display name.
31    pub name: syn::Ident,
32    /// Source constructor delimiters.
33    pub style: VariantStyle,
34    /// Payload fields in source order.
35    pub fields: Vec<PayloadField<F>>,
36}
37
38/// One named or positional enum payload member.
39#[derive(Clone, Debug)]
40pub struct PayloadField<F = ()> {
41    /// Generator field metadata, as on record fields.
42    pub metadata: F,
43    /// Source member: a named field or tuple position.
44    pub member: syn::Member,
45    /// Qualified source type.
46    pub ty: Type,
47}
48impl<F> PayloadField<F> {
49    /// Source field key used by occurrence overrides.
50    pub fn key(&self) -> String {
51        match &self.member {
52            syn::Member::Named(name) => name.to_string().trim_start_matches("r#").into(),
53            syn::Member::Unnamed(index) => index.index.to_string(),
54        }
55    }
56    /// Public default property name, using `v0`, `v1`, ... for tuple fields.
57    pub fn property(&self) -> String {
58        match &self.member {
59            syn::Member::Named(_) => self.key(),
60            syn::Member::Unnamed(index) => format!("v{}", index.index),
61        }
62    }
63}
64impl<V, F> PayloadEnum<V, F> {
65    /// Spell a source constructor or pattern with the supplied field expressions.
66    /// A standard Result uses its generic constructor path with inferred arguments.
67    pub fn spell(
68        &self,
69        variant: usize,
70        values: &[proc_macro2::TokenStream],
71    ) -> Result<proc_macro2::TokenStream> {
72        let arm = self
73            .variants
74            .get(variant)
75            .ok_or("unknown payload variant")?;
76        if arm.fields.len() != values.len() {
77            return Err("wrong enum payload field count".into());
78        }
79        let Type::Path(source) = &self.ty else {
80            return Err("enum source must be a path".into());
81        };
82        let mut path = source.path.clone();
83        if result_parts(&self.ty).is_some() {
84            path.segments.last_mut().unwrap().arguments = syn::PathArguments::None;
85        }
86        let name = &arm.name;
87        Ok(match arm.style {
88            VariantStyle::Unit => quote!(#path::#name),
89            VariantStyle::Tuple => quote!(#path::#name(#(#values),*)),
90            VariantStyle::Named => {
91                let members = arm.fields.iter().map(|f| &f.member);
92                quote!(#path::#name { #(#members: #values),* })
93            }
94        })
95    }
96}

98impl<M: Schema> Flat<M> {
99    pub(crate) fn parse_payload(
100        &self,
101        ty: Type,
102        source: syn::ItemEnum,
103    ) -> Result<PayloadEnum<M::Variant, M::Field>> {
104        if source.variants.is_empty() || i32::try_from(source.variants.len()).is_err() {
105            return Err("payload enum needs 1..=i32::MAX variants".into());
106        }
107        let mut names = std::collections::BTreeSet::new();
108        let variants = source
109            .variants
110            .into_iter()
111            .map(|arm| {
112                if !names.insert(arm.ident.to_string().trim_start_matches("r#").to_owned()) {
113                    return Err("duplicate enum variant".into());
114                }
115                if arm.attrs.iter().any(|a| {
116                    a.path().is_ident("cfg")
117                        || a.path().is_ident("cfg_attr")
118                        || a.path().is_ident("non_exhaustive")
119                }) {
120                    return Err(
121                        "conditional or non-exhaustive enum variants are unsupported".into(),
122                    );
123                }
124                let style = match &arm.fields {
125                    syn::Fields::Unit => VariantStyle::Unit,
126                    syn::Fields::Unnamed(_) => VariantStyle::Tuple,
127                    syn::Fields::Named(_) => VariantStyle::Named,
128                };
129                let fields = arm
130                    .fields
131                    .into_iter()
132                    .enumerate()
133                    .map(|(index, field)| {
134                        if field
135                            .attrs
136                            .iter()
137                            .any(|a| a.path().is_ident("cfg") || a.path().is_ident("cfg_attr"))
138                        {
139                            return Err("conditional enum payload fields are unsupported".into());
140                        }
141                        validate_input_lifetimes(&field.ty)?;
142                        Ok(PayloadField {
143                            metadata: Default::default(),
144                            member: field
145                                .ident
146                                .map(syn::Member::Named)
147                                .unwrap_or_else(|| syn::Member::Unnamed(syn::Index::from(index))),
148                            ty: self.resolve_type(&field.ty)?,
149                        })
150                    })
151                    .collect::<Result<Vec<_>>>()?;
152                Ok(PayloadVariant {
153                    metadata: Default::default(),
154                    name: arm.ident,
155                    style,
156                    fields,
157                })
158            })
159            .collect::<Result<Vec<_>>>()?;
160        Ok(PayloadEnum { ty, variants })
161    }
162
163    /// Look up a payload enum, including a concrete standard `Result<T, E>`.
164    pub fn payload_enum(&self, ty: &Type) -> Result<&PayloadEnum<M::Variant, M::Field>> {
165        self.payload_enums
166            .get(&type_key(ty))
167            .ok_or_else(|| format!("`{}` is not a payload enum", type_key(ty)))
168    }
169
170    pub(crate) fn register_result(&mut self, ty: &Type) {
171        let Some((ok, error)) = result_parts(ty) else {
172            return;
173        };
174        let key = type_key(ty);
175        if self.payload_enums.contains_key(&key) {
176            return;
177        }
178        self.types.entry(key.clone()).or_insert_with(|| TypeInfo {
179            ty: ty.clone(),
180            metadata: Default::default(),
181            conversion: Default::default(),
182        });
183        let variants = [("Ok", ok), ("Err", error)]
184            .into_iter()
185            .map(|(name, child)| PayloadVariant {
186                metadata: Default::default(),
187                name: syn::Ident::new(name, proc_macro2::Span::call_site()),
188                style: VariantStyle::Tuple,
189                fields: vec![PayloadField {
190                    metadata: Default::default(),
191                    member: syn::Member::Unnamed(syn::Index::from(0)),
192                    ty: child.clone(),
193                }],
194            })
195            .collect();
196        self.payload_enums.insert(
197            key,
198            PayloadEnum {
199                ty: ty.clone(),
200                variants,
201            },
202        );
203    }
204
205    pub(crate) fn register_results(&mut self) {
206        struct Results(Vec<Type>);
207        impl VisitMut for Results {
208            fn visit_type_mut(&mut self, ty: &mut Type) {
209                if result_parts(ty).is_some() {
210                    self.0.push(ty.clone());
211                }
212                syn::visit_mut::visit_type_mut(self, ty);
213            }
214        }
215        let mut roots = self
216            .functions
217            .values()
218            .flat_map(|f| {
219                f.params
220                    .iter()
221                    .map(|p| p.ty.ty.clone())
222                    .chain(std::iter::once(f.ret.ty.clone()))
223            })
224            .collect::<Vec<_>>();
225        roots.extend(
226            self.records
227                .values()
228                .flat_map(|r| r.fields.iter().map(|f| f.ty.clone())),
229        );
230        roots.extend(self.payload_enums.values().flat_map(|e| {
231            e.variants
232                .iter()
233                .flat_map(|v| v.fields.iter().map(|f| f.ty.clone()))
234        }));
235        let mut results = Results(vec![]);
236        for mut ty in roots {
237            results.visit_type_mut(&mut ty);
238        }
239        for ty in results.0 {
240            self.register_result(&ty);
241        }
242    }
243}

245#[cfg(test)]
246mod tests {
247    use super::*;
248    fn parse(source: &str) -> Result<Flat<()>> {
249        Flat::parse(
250            syn::parse_file(source)
251                .unwrap()
252                .items
253                .into_iter()
254                .map(|i| (i, Default::default())),
255        )
256    }
257    #[test]
258    fn payload_syntax_preserves_empty_constructor_shapes() {
259        let flat = parse("pub enum Empty { Unit, Tuple(), Named{} }").unwrap();
260        let source = flat.payload_enums.values().next().unwrap();
261        assert_eq!(
262            source.variants.iter().map(|v| v.style).collect::<Vec<_>>(),
263            [VariantStyle::Unit, VariantStyle::Tuple, VariantStyle::Named]
264        );
265        assert_eq!(
266            source.spell(1, &[]).unwrap().to_string(),
267            "crate :: Empty :: Tuple ()"
268        );
269    }
270    #[test]
271    fn conditional_fields_and_generic_enums_are_diagnostics() {
272        for source in [
273            "pub enum E{ A{#[cfg(any())] value:i64} }",
274            "pub enum E<T>{A(T)}",
275            "#[non_exhaustive]pub enum E{A(i64)}",
276        ] {
277            assert!(parse(source).is_err(), "{source}");
278        }
279    }
280    #[test]
281    fn result_source_variants_do_not_change_function_error_routing() {
282        let flat =
283            parse("pub fn f(value:Option<Result<i64,String>>)->Result<(),String>{}").unwrap();
284        let result = flat
285            .resolve_type(&syn::parse_quote!(Result<i64,String>))
286            .unwrap();
287        assert!(flat.payload_enums.contains_key(&type_key(&result)));
288        let f = flat.function("f").unwrap();
289        assert!(result_parts(&f.ret.ty).is_some());
290    }
