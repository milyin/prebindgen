8#[derive(Clone, Debug)]
9pub(crate) struct ErrorBinding {
10    pub(crate) helper: syn::Ident,
11    ty: syn::Type,
12    pub output: Option<JniOutput>,
13    recipe: Option<prebindgen_declarations::Output>,
14    pub handler: String,
15}
16impl ErrorBinding {
17    pub(crate) fn output(&self) -> Result<&JniOutput> {
18        self.output
19            .as_ref()
20            .ok_or_else(|| format!("JNI error `{}` has no compiled output", self.handler))
21    }
22
23    pub fn throws(&self, exception: Option<&str>) -> String {
24        exception.map_or_else(String::new, |class| {
25            format!("@kotlin.jvm.Throws({}::class)\n", class.replace('$', "."))
26        })
27    }
28    pub fn parameter(&self, ret: &str, exception: Option<&str>) -> String {
29        let default = exception.map_or_else(String::new, |class| {
30            format!(
31                " = {} {{ message -> throw {}(message) }}",
32                self.handler,
33                class.replace('$', ".")
34            )
35        });
36        format!("onError: {}<{ret}>{default}", self.handler)
37    }
38}

40/// Bind a prepared function's typed domain-error output converter.
41/// The result slot must declare matching error wires. The helper and Kotlin
42/// handler are local to the source function, so overloads and other functions
43/// sharing the same Rust error type retain independent conversion policies.
44pub fn bind_error_output<I, O>(
45    flat: &mut Flat<JniSchema<I, O>>,
46    function: &str,
47    output: impl KotlinOutputConversion + 'static,
48) -> Result<()> {
49    let function = flat.function_mut(function)?;
50    let identity = format!(
51        "{}::{}",
52        names::path_identity(&function.path),
53        function.name
54    );
55    let hash = prebindgen_reprc::schema_id(&identity);
56    let label = function.name.to_string();
57    let label = label.trim_start_matches("r#");
58    let metadata = function
59        .metadata
60        .as_mut()
61        .ok_or("function has no export metadata")?;
62    let previous = metadata
63        .error
64        .as_ref()
65        .ok_or("function has no prepared domain-error converter")?;
66    let mut error = (**previous).clone();
67    error.helper = format_ident!("__pb_jni_error_{label}_local_{hash:016x}");
68    let package = error
69        .handler
70        .rsplit_once('.')
71        .ok_or("domain-error handler has no package")?
72        .0;
73    error.handler = format!("{package}.LocalErrorHandler_{label}_{hash:016x}");
74    error.output = Some(JniOutput::new(output));
75    metadata.error = Some(std::rc::Rc::new(error));
76    Ok(())
77}
78fn helper_name(names: &mut names::Names, ty: &syn::Type, mapped: bool) -> Result<syn::Ident> {
79    let mode = if mapped { "mapped" } else { "typed" };
80    Ok(format_ident!(
81        "{}",
82        names.name(
83            "__pb_jni_error",
84            &format!("{}_{mode}", names::type_label(ty)),
85            &[&names::type_identity(ty), mode]
86        )?
87    ))
88}

90/// Prepare each error once on its type and attach each function's selected binding.
91pub(crate) fn resolve(
92    flat: &mut Bindings,
93    codec: &JniCodec,
94    declarations: &prebindgen_declarations::Declarations,
95    functions: &[FunctionDecl],
96) -> Result<()> {
97    let mut names = names::Names::default();
98    for ty in flat.types.values() {
99        for (mapped, binding) in &ty.metadata.errors {
100            names.reserve(
101                &binding.helper.to_string(),
102                "__pb_jni_error",
103                &[
104                    &names::type_identity(&ty.ty),
105                    if *mapped { "mapped" } else { "typed" },
106                ],
107            )?;
108        }
109    }
110    for function in functions {
111        if function.result_value {
112            continue;
113        }
114        let ret = flat.function(&function.name)?.ret.ty.clone();
115        let Some((_, error)) = prebindgen_flat::result_parts(&ret) else {
116            continue;
117        };
118        flat.type_metadata_mut(error)?;
119        let canonical_error = flat.type_info(error)?.ty.clone();
120        let mapped = declarations.error_mapper_for(function, error).is_some();
121        let exception = flat.type_info(error)?.metadata.error_class.clone();
122        if exception.is_some() && !mapped {
123            return Err(
124                "error_class requires an explicit message mapper for its default Kotlin handler"
125                    .into(),
126            );
127        }
128        if let Some(binding) = flat.type_info(error)?.metadata.errors.get(&mapped).cloned() {
129            flat.function_mut(&function.name)?
130                .metadata
131                .as_mut()
132                .ok_or_else(|| format!("JNI function `{}` has no export metadata", function.name))?
133                .error = Some(binding);
134            continue;
135        }
136        let (ty, recipe) = if mapped {
137            (
138                syn::parse_quote!(String),
139                prebindgen_declarations::Output::Leaf(OutputLeafOp::Value),
140            )
141        } else {
142            (
143                canonical_error.clone(),
144                declarations.output_recipe(flat, codec, &canonical_error, &())?,
145            )
146        };
147        let helper = helper_name(&mut names, &canonical_error, mapped)?;
148        let pkg = flat
149            .type_info(error)?
150            .metadata
151            .class
152            .as_ref()
153            .map_or(flat.metadata.package.as_str(), |c| c.pkg.as_str());
154        let suffix = if mapped { "MessageHandler" } else { "Handler" };
155        let short = if let syn::Type::Path(path) = &canonical_error {
156            if path.qself.is_none()
157                && path
158                    .path
159                    .segments
160                    .iter()
161                    .all(|segment| matches!(segment.arguments, syn::PathArguments::None))
162            {
163                Some(format!(
164                    "{}{suffix}",
165                    path.path.segments.last().unwrap().ident
166                ))
167            } else {
168                None
169            }
170        } else {
171            None
172        };
173        let short = if let Some(short) = short {
174            short
175        } else {
176            let mode = if mapped { "mapped" } else { "typed" };
177            names.name(
178                "ErrorHandler",
179                &format!("{}_{mode}", names::type_label(&canonical_error)),
180                &[&names::type_identity(&canonical_error), mode],
181            )?
182        };
183        let handler = format!("{pkg}.{short}");
184        if foreign_name_exists(flat, &handler) {
185            return Err(format!("duplicate foreign name `{handler}`"));
186        }
187        let binding = std::rc::Rc::new(ErrorBinding {
188            helper,
189            ty,
190            output: None,
191            recipe: Some(recipe),
192            handler,
193        });
194        flat.type_metadata_mut(error)?
195            .errors
196            .insert(mapped, binding.clone());
197        flat.function_mut(&function.name)?
198            .metadata
199            .as_mut()
200            .ok_or_else(|| format!("JNI function `{}` has no export metadata", function.name))?
201            .error = Some(binding);
202    }
203    Ok(())
204}

206pub(crate) fn compile(flat: &mut Bindings, codec: &JniCodec) -> Result<()> {
207    let pending = flat
208        .types
209        .iter()
210        .flat_map(|(key, t)| {
211            t.metadata
212                .errors
213                .iter()
214                .filter(|(_, e)| e.recipe.is_some())
215                .map(|(mapped, e)| (key.clone(), *mapped, e.clone()))
216        })
217        .collect::<Vec<_>>();
218    for (key, mapped, old) in pending {
219        let mut binding = (*old).clone();
220        let recipe = binding.recipe.take().unwrap();
221        binding.output = Some(
222            prebindgen_declarations::compile_output_value(flat, codec, &binding.ty, &recipe, &())
223                .map_err(|error| {
224                let function = flat.functions.values().find(|function| {
225                    function
226                        .metadata
227                        .as_ref()
228                        .and_then(|m| m.error.as_ref())
229                        .is_some_and(|binding| std::rc::Rc::ptr_eq(binding, &old))
230                });
231                match function {
232                    Some(function) => format!("JNI error `{key}` in `{}`: {error}", function.name),
233                    None => format!("JNI error `{key}`: {error}"),
234                }
235            })?,
236        );
237        let binding = std::rc::Rc::new(binding);
238        flat.types
239            .get_mut(&key)
240            .unwrap()
241            .metadata
242            .errors
243            .insert(mapped, binding.clone());
244        for function in flat.functions.values_mut() {
245            if let Some(metadata) = &mut function.metadata {
246                if metadata
247                    .error
248                    .as_ref()
249                    .is_some_and(|e| std::rc::Rc::ptr_eq(e, &old))
250                {
251                    metadata.error = Some(binding.clone());
252                }
253            }
254        }
255    }
256    Ok(())
257}

259pub(crate) fn emit(errors: &BTreeMap<String, &ErrorBinding>) -> Result<TokenStream> {
260    let mut code = TokenStream::new();
261    let mut emitted = BTreeSet::new();
262    for error in errors.values() {
263        if !emitted.insert(error.helper.to_string()) {
264            continue;
265        }
266        let helper = &error.helper;
267        let ty = &error.ty;
268        let source = format_ident!("source");
269        let context = JniContext {
270            input_transfers: prebindgen_conversions::InputTransfers::Forbidden,
271            return_delivery: crate::JniReturnDelivery::Packet,
272            prepared_delivery: None,
273            environment: format_ident!("env"),
274            domain_output: format_ident!("reported"),
275            return_sink: None,
276            is_domain_error: true,
277            output_committed: quote!(*reported = Some(result.as_raw());),
278        };
279        let body_for = |context: &JniContext| -> Result<TokenStream> {
280            let mut prepared_context = context.clone();
281            returns::prepare(error.output()?, &mut prepared_context)?;
282            let context = &prepared_context;
283            let commit = error.output()?.code_fragment_slot(
284                OutputFragment::Commit {
285                    context,
286                    source: &source,
287                },
288                &syn::parse_quote!(()),
289            )?;
290            let body = error.output()?.code_fragment_slot(
291                OutputFragment::Deliver {
292                    context,
293                    source: &source,
294                    next: quote!({let _: () = #commit; Ok(())}),
295                },
296                &syn::parse_quote!(::prebindgen_runtime::Result<()>),
297            )?;
298            error.output()?.code_fragment_slot(
299                OutputFragment::Prepare {
300                    context,
301                    source: &source,
302                    next: body,
303                },
304                &syn::parse_quote!(::prebindgen_runtime::Result<()>),
305            )
306        };
307        let body = body_for(&context)?;
308        let callback_context = JniContext {
309            input_transfers: prebindgen_conversions::InputTransfers::Forbidden,
310            return_delivery: JniReturnDelivery::Capture,
311            prepared_delivery: None,
312            return_sink: Some(format_ident!("__pb_return")),
313            output_committed: quote!(*reported = Some(std::ptr::null_mut());),
314            ..context
315        };
316        let callback_body = if returns::select(error.output()?).is_some() {
317            body_for(&callback_context)?
318        } else {
319            quote!(Err("custom error output requires packet delivery".into()))
320        };
321        code.extend(quote! {
322            #[allow(non_snake_case, clippy::needless_borrow, clippy::let_unit_value, clippy::unnecessary_cast, clippy::useless_conversion)]
323            fn #helper(mut env: &mut ::prebindgen_jni_runtime::jni::JNIEnv<'_>, source: #ty, reported: &mut Option<::prebindgen_jni_runtime::jni::sys::jobjectArray>, return_sink: Option<&::prebindgen_jni_runtime::jni::objects::JObject<'_>>) -> ::prebindgen_runtime::Result<()> {
324                if let Some(__pb_return) = return_sink { #callback_body } else { #body }
325            }
326        });
327    }
328    Ok(code)
329}

331pub(crate) fn interfaces(prefix: &str, files: &mut BTreeMap<String, String>) -> Result<()> {
332    let binding_name = format!("{}/JniErrorHandler.kt", prefix.replace('.', "/"));
333    if files.insert(binding_name, format!("package {prefix}\n\n/** Binding failure callback, invoked after the native call and handle locks finish. */\npublic fun interface JniErrorHandler<out R> {{ public fun run(message: String?): R }}\n")).is_some() {
334        return Err("Kotlin error interface collision".into());
335    }
336    Ok(())
337}

339pub(crate) fn handler_files(
340    errors: &BTreeMap<String, &ErrorBinding>,
341    context: &KotlinContext<'_>,
342    files: &mut BTreeMap<String, String>,
343) -> Result<()> {
344    let mut seen = BTreeSet::new();
345    for error in errors.values() {
346        if !seen.insert(error.handler.clone()) {
347            continue;
348        }
349        let (pkg, name) = error.handler.rsplit_once('.').unwrap();
350        let output = error.output()?.kotlin(context)?;
351        let slots: usize = output
352            .result_types
353            .iter()
354            .map(|ty| {
355                if matches!(ty.as_str(), "Long" | "Double" | "ULong") {
356                    2
357                } else {
358                    1
359                }
360            })
361            .sum();
362        if slots > 254 {
363            return Err(format!(
364                "domain handler `{}` exceeds the JVM parameter limit",
365                error.handler
366            ));
367        }
368        let params = output.callback_parameter_declarations()?;
369        let text = format!("package {pkg}\n\n/** Domain-error callback. Called after JNI and handle locks finish. Handle arguments are closed after run; call take() to retain an owner. */\npublic fun interface {name}<out R> {{ public fun run({params}): R }}\n");
370        if files
371            .insert(format!("{}/{}.kt", pkg.replace('.', "/"), name), text)
372            .is_some()
373        {
374            return Err("Kotlin error interface collision".into());
375        }
376    }
377    Ok(())
378}

380pub(crate) fn delivery(error: &ErrorBinding, context: &KotlinContext<'_>) -> Result<String> {
381    let context = KotlinContext {
382        cleanup_failure: Some("cleanupFailure"),
383        ..*context
384    };
385    let mut output = error.output()?.kotlin(&context)?;
386    kotlin::remap_output(
387        &mut output,
388        context.result,
389        context.result,
390        1,
391        crate::group::result_packet_fields(&error.output()?.output_outcomes().success).len(),
392    );
393    let call = format!("onError.run({})", output.delivered.join(", "));
394    if output.cleanup.is_empty() {
395        return Ok(format!(
396            "{}{}\n{call}",
397            output.output_locals, output.materialize
398        ));
399    }
400    let cleanup = output.cleanup;
401    Ok(format!("{}\nvar cleanupFailure: Throwable? = null\nvar callbackFailed = false\ntry {{\n{}\n{call}\n}} catch (failure: Throwable) {{ callbackFailed = true; cleanupFailure = failure; throw failure }} finally {{\n{cleanup}\nif (!callbackFailed) cleanupFailure?.let {{ throw it }}\n}}", output.output_locals, output.materialize))
