27/// Resolved incoming conversion. Configuration-only payloads are uninhabited.
28/// A configuration tree cannot be passed as a final tree:
29/// ```compile_fail
30/// use prebindgen_c::{CInput,CInputConfig,prelude::*};
31/// let _:CInput=CInputConfig::auto();
32/// ```
33/// Even a nested child must have the final operation sets:
34/// ```compile_fail
35/// use prebindgen_c::{CInput,CInputConfig,CInputUnary,prelude::*};
36/// let _:CInput=CInput::Unary {
37///     operation:CInputUnary::Sequence,
38///     child:Box::new(CInputConfig::auto()),
39/// };
40/// ```
41/// Configuration-only leaf payloads are directly Infallible:
42/// ```compile_fail
43/// use prebindgen_c::{CInput,CInputLeafOp};
44/// let _:CInput=CInput::Leaf(CInputLeafOp::ConfigDefault(()));
45/// ```
46/// The same applies to unary scopes and both partial enum operations:
47/// ```compile_fail
48/// use prebindgen_c::{CInput,CInputUnary};
49/// let _:CInput=CInput::Unary {
50///     operation:CInputUnary::ConfigForType {ty:Box::new(syn::parse_quote!(i64))},
51///     child:Box::new(CInput::None),
52/// };
53/// ```
54/// ```compile_fail
55/// use prebindgen_c::{CInput,CInputBranch};
56/// let _:CInput=CInput::Branch {
57///     operation:CInputBranch::ConfigEnumFields(()),children:vec![],
58/// };
59/// ```
60/// ```compile_fail
61/// use prebindgen_c::{CInput,CInputBranch,CVariantFields};
62/// let _:CInput=CInput::Branch {
63///     operation:CInputBranch::ConfigVariantFields(CVariantFields {variant:"Row".into(),fields:vec![]}),
64///     children:vec![],
65/// };
66/// ```
67pub type CInput = InputNode<CInputOps, COutputOps>;
68/// Resolved outgoing conversion. An optional configuration payload cannot
69/// leave a constructible None in the final leaf type:
70/// ```compile_fail
71/// use prebindgen_c::{COutput,COutputLeafOp};
72/// let _:COutput=COutput::Leaf(COutputLeafOp::ConfigValue(None));
73/// ```
74pub type COutput = OutputNode<CInputOps, COutputOps>;
75/// Incoming conversion configuration.
76pub type CInputConfig = InputNode<CInputConfigOps, COutputConfigOps>;
77/// Outgoing conversion configuration.
78pub type COutputConfig = OutputNode<CInputConfigOps, COutputConfigOps>;

330pub(crate) fn sequence_callback(
331    cx: &mut crate::CResolve<'_>,
332    ty: &syn::Type,
333    mut output: COutput,
334    structural_default: bool,
335) -> Result<COutput> {
336    fn select(output: &mut COutput) -> usize {
337        match output {
338            COutput::Unary {
339                operation: COutputUnary::Sequence { delivery, .. },
340                ..
341            } => {
342                *delivery = SequenceDelivery::Callback;
343                1
344            }
345            COutput::Unary {
346                operation: COutputUnary::Optional,
347                child,
348            }
349            | COutput::Unary {
350                operation: COutputUnary::Named { .. },
351                child,
352                ..
353            }
354            | COutput::Unary {
355                operation: COutputUnary::Getter { .. },
356                child,
357                ..
358            }
359            | COutput::Unary {
360                operation: COutputUnary::Into { .. },
361                child,
362                ..
363            }
364            | COutput::Unary {
365                operation: COutputUnary::Convert { .. },
366                child,
367                ..
368            } => select(child),
369            COutput::Branch {
370                operation: COutputBranch::Combine,
371                children,
372            }
373            | COutput::Branch {
374                operation: COutputBranch::Fields,
375                children,
376            }
377            | COutput::Branch {
378                operation: COutputBranch::OwnedFields,
379                children,
380            } => children.iter_mut().map(select).sum(),
381            _ => 0,
382        }
383    }
384    if select(&mut output) != 1 {
385        fn structural(ty: &syn::Type, output: &COutput) -> Option<crate::COutputConfig> {
386            match output {
387                COutput::Leaf(COutputLeafOp::Value)
388                    if prebindgen_flat::sequence_parts(ty).is_some() =>
389                {
390                    Some(crate::COutputConfig::Unary {
391                        operation: crate::COutputUnary::Sequence {
392                            delivery: SequenceDelivery::Callback,
393                            mode: None,
394                        },
395                        child: Box::new(crate::COutputConfig::auto()),
396                    })
397                }
398                COutput::Unary {
399                    operation: COutputUnary::Optional,
400                    child,
401                } => Some(crate::COutputConfig::Unary {
402                    operation: crate::COutputUnary::Optional,
403                    child: Box::new(structural(prebindgen_flat::option_inner(ty)?, child)?),
404                }),
405                _ => None,
406            }
407        }
408        if structural_default {
409            if let Some(tree) = structural(ty, &output) {
410                output = cx.output(ty, &tree)?;
411            } else {
412                return Err("callback_return requires exactly one outer sequence".into());
413            }
414        } else {
415            return Err("callback_return requires exactly one outer sequence".into());
416        }
417    }
418    Ok(output)
419}

469            CInputBranch::Constructor { function } => Self::Constructor { function },
470            CInputBranch::Enum { value } => Self::Enum { value },
471            CInputBranch::Variant => Self::Variant,
472        }
473    }
