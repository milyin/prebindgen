1//! The leaves a decomposed value crosses as, and the walk over them.
2//!
3//! A **leaf** is one value that crosses once a returned struct has been taken
4//! apart as far as it goes. Here are the leaf model — where a leaf is reached
5//! from, which arms it is live inside, what it converts through — and the walk
6//! grouping a leaf list into the arms an emitter renders as a `match`. Which
7//! **delivery** those leaves take, a returned value against a foreign callback,
8//! is deliberately not here: that follows from what the target language can
9//! receive, so it is the adapter's (#680 review).
10
11/// Outer shape wrapping the [core decomposition](`UnfoldShape::Base`).
12/// The output-side analog of [`crate::expand::FoldShape`], on the
13/// unified [`Shape`](crate::shape::Shape) layer stack:
14///   * `Base` — run the accessor's records on the value, producing all
15///     leaves and invoking the builder once;
16///   * `Optional((), inner)` — `Option<T>`/`Option<&T>` return: `None` ⇒ a null
17///     result (builder skipped), `Some` ⇒ decompose the inner;
18///   * `Iterable(inner)` — `Vec<T>` return: fold the elements through an
19///     accumulator `(acc, …) -> acc`. Each element is delivered either WHOLE (via
20///     its own output converter + projection — see an adapter's plan `element`) or
21///     DECOMPOSED into per-element leaves (explicit accessors, or a synthesized
22///     `data_class` — see an adapter's plan `fixed_builder`); inner is `Base`.
23///
24/// The `()` payload is unused here — only the JNI adapter's
25/// `Shape<NullableKind>` carries per-layer data.
26pub use crate::shape::Shape as UnfoldShape;

28/// Identity of the deconstructor **declaration** a plan's records came from.
29/// A `run`-signature artifact (e.g. a generated callback interface) is fully
30/// determined by the declaration, so adapters key such artifacts on this —
31/// functions selecting the same declaration share one artifact; differently
32/// declared decompositions of the same type get distinct ones. The first
33/// field is always the target type's canonical [`TypeKey`](crate::TypeKey)
34/// string.
35#[derive(Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
36pub enum DeconId {
37    /// The type's default (`expand_return!`-declared) deconstructor.
38    Default(String),
39    /// Per-fn inline records (`.expand_return`) — unique to the
40    /// function (second field = the fn ident).
41    PerFn(String, String),
42}

44/// The declaration-default decomposition of one deconstructor: its leaf
45/// list resolved ONCE from the declaration's records with **normalized**
46/// inputs (borrowed identity form, no outer shape), so the content is
47/// independent of which functions use the declaration and in what order.
48/// Stored in `Registry::decon_plans`; the single source language adapters
49/// derive declaration-keyed signature artifacts (e.g. generated callback
50/// interfaces) from. Per-function aspects (`by_ref`, shape, delivery) live on
51/// each function's an adapter's decomposition plan, which points here via an adapter's plan `decon`.
52///
53/// Normalization detail: the identity leaf's `out_ty` is always the borrowed
54/// `&Source` form (an owned-return function's own plan carries owned `Source`
55/// instead) — both resolve to the same projection/class, and adapters reading
56/// the spec must tolerate whichever form their type tables resolved.
57#[derive(Clone)]
58pub struct DeconSpec {
59    /// The decomposed type as first encountered. A **reading**, so "compare via
60    /// [`TypeKey`](crate::TypeKey), not syntactically" is
61    /// the type rather than an instruction: `source.key()` is the identity and
62    /// `emit.emit_source_type(&source)` is what an emission callback writes.
63    pub source: crate::flat::TypeRef,
64    /// Flattened leaves in declared record order — names, types, paths,
65    /// nullability all declaration-fixed.
66    pub leaves: Vec<UnfoldLeaf>,
67}

69/// One step of a leaf's [`UnfoldLeaf::path`] — how to get from the value
70/// reached so far to the next one.
71///
72/// A step is typed rather than a bare ident because a single path may **mix**
73/// the two: an `expand_return!(T).fields(fields!(t_to_struct))` leaf calls the
74/// value-form accessor, reads a struct field, and may then call that field
75/// type's own accessor — `Call(t_to_struct)`, `Field(key_expr)`,
76/// `Call(keyexpr_as_str)`. [`LeafSource`] still says what *kind* of leaf sits
77/// at the end of the path; the steps say how it is reached.
78///
79/// Each step also records whether it is **optional** — its accessor returns
80/// `Option<…>`, or its field is typed `Option<…>`. A `true` on a step *before*
81/// the last makes it a nullable nesting step: the emitter matches on it and the
82/// `None` arm short-circuits the whole leaf to null. The flag is carried rather
83/// than re-derived so both kinds answer the question the same way and the
84/// emitter needs no type walk (an accessor's `Option` was already peeled where
85/// the step was built).
86#[derive(Clone, PartialEq, Eq, Debug)]
87pub enum PathStep {
88    /// Call a `#[prebindgen]` accessor on the value reached so far:
89    /// `source_module::f(&value)`.
90    Call {
91        ident: syn::Ident,
92        optional: bool,
93        /// Whether the value this call yields (its return with any `Option`
94        /// peeled) is **owned** rather than a borrow — `f(..) -> T` / `-> Option<T>`
95        /// against `-> &T` / `-> Option<&T>`.
96        ///
97        /// Only an OPTIONAL step's payload can reach an emitter as a bare
98        /// binding, so this is only ever consulted there: what a `Some` arm
99        /// binds is the accessor's own value, and everything downstream of it —
100        /// the next step's receiver, the value form's argument — needs to know
101        /// whether it may be moved or has to be borrowed. Recorded here, at the
102        /// one place the signature is in hand, because deriving it later from
103        /// the path alone is not possible.
104        owned: bool,
105    },
106    /// Read a struct field of the value reached so far: `value.f`.
107    Field { ident: syn::Ident, optional: bool },
108}
109
110impl PathStep {
111    /// An accessor call step. `owned` says whether its (`Option`-peeled) return
112    /// is an owned value rather than a borrow — see [`Self::Call::owned`].
113    pub fn call(ident: syn::Ident, optional: bool, owned: bool) -> Self {
114        Self::Call {
115            ident,
116            optional,
117            owned,
118        }
119    }
120
121    /// Whether the value this step yields is owned. A field read composes as
122    /// `&(e).f`, so it is a borrow by construction.
123    pub fn yields_owned(&self) -> bool {
124        matches!(self, Self::Call { owned: true, .. })
125    }
126
127    /// A struct-field read step.
128    pub fn field(ident: syn::Ident, optional: bool) -> Self {
129        Self::Field { ident, optional }
130    }
131
132    /// The step's ident, whichever kind it is.
133    pub fn ident(&self) -> &syn::Ident {
134        match self {
135            Self::Call { ident, .. } | Self::Field { ident, .. } => ident,
136        }
137    }
138
139    /// Whether the step yields an `Option` — a nullable nesting step when it is
140    /// not the last on the path.
141    pub fn is_optional(&self) -> bool {
142        match self {
143            Self::Call { optional, .. } | Self::Field { optional, .. } => *optional,
144        }
145    }
146
147    /// Whether the step is a plain (non-optional) field read — a path made only
148    /// of these renders as `value.a.b`, needing no nesting `match`.
149    pub fn is_plain_field(&self) -> bool {
150        matches!(
151            self,
152            Self::Field {
153                optional: false,
154                ..
155            }
156        )
157    }
158
159    /// Whether the step is a field read, `Option` or not.
160    pub fn is_field(&self) -> bool {
161        matches!(self, Self::Field { .. })
162    }
163}

165/// Whether a run of steps can be **moved** out of the value it hangs off:
166/// field reads only, with an `Option` allowed on the last one — a `None` arm
167/// still hands over the whole `Option` by value, while an `Option` in the
168/// middle would have to be unwrapped and so can only be borrowed through.
169///
170/// This is the one place the rule is written: the resolver uses it to decide
171/// whether a leaf OWNS what it reaches (its `out_ty` then being the owned type
172/// rather than a borrow), and the emitters use it to project that place. Two
173/// readings of it would drift, and the disagreement would be a borrow handed to
174/// an owning converter.
175pub fn steps_are_movable(steps: &[PathStep]) -> bool {
176    steps
177        .iter()
178        .enumerate()
179        .all(|(i, s)| s.is_field() && (!s.is_optional() || i + 1 == steps.len()))
180}

182/// How a leaf's [`UnfoldLeaf::path`] is reached from the decomposed value.
183#[derive(Clone, PartialEq, Eq, Debug, Default)]
184pub enum LeafSource {
185    /// The value is **reached off** the decomposed value: an accessor chain, a
186    /// field chain, or the two mixed, which is what
187    /// [`UnfoldLeaf::path`] spells.
188    ///
189    /// One variant rather than two. `Accessor` and `Field` were separate while
190    /// an adapter branched on which it was; none does — the difference is
191    /// whether the path ENDS in a field read, which the path itself says. Two
192    /// spellings of one fact are two things to keep in step, and this is the
193    /// one that had no reader left.
194    #[default]
195    Reach,
196    /// The **synthesized selector** of a decomposed sum: an `i32` naming which
197    /// alternative is live. It is not read off the value at all — the emitter
198    /// assigns it per `match` arm — so it has no path. Emitted once, ahead of
199    /// the groups it selects between (see
200    /// the adapter's sum deconstruction).
201    ///
202    /// Its [`out_ty`](UnfoldLeaf::out_ty) is **the sum**, not the `i32` — it
203    /// carries *which* sum it chooses between, which is how the emitter finds
204    /// the enum to `match`. That type is **registered and not required** (#282):
205    /// it gets a table cell like every other leaf's, but no root, because a sum
206    /// has no whole-value output converter and demanding one would fail
207    /// resolution over a type that never crosses whole. The reading comes from
208    /// the declaration — [`Variant::type_ref`](crate::flat::Variant::type_ref)
209    /// — never from an adapter composing one out of a name.
210    SumTag,
211    /// A payload field of ONE alternative of a decomposed sum, reached through
212    /// a **variant pattern** rather than a path: the emitter binds `member`
213    /// inside `variant`'s `match` arm. The leaf is live only when
214    /// [`UnfoldLeaf::groups`] ends in the value's tag; in every other arm its
215    /// slot carries the wire default.
216    ///
217    /// This is the selector [`Reach`](Self::Reach) deliberately lacks — a
218    /// reached value is a deterministic product, every one of them
219    /// contributing unconditionally.
220    VariantField {
221        /// The variant's ident as declared in the source enum.
222        variant: syn::Ident,
223        /// How the payload field is addressed in the arm's pattern.
224        member: syn::Member,
225    },
226    /// The **synthesized presence** of an optional value the decomposition
227    /// looks through: a boolean saying whether the leaves that follow carry
228    /// anything.
229    ///
230    /// The selector [`SumTag`](Self::SumTag) is for a value that is one of
231    /// several alternatives; this is for a value that is either there or not,
232    /// and the difference is not a two-alternative sum: absence has no
233    /// alternative of its own to name, and the group it gates is the value's
234    /// own leaves rather than one arm's payload. Like a tag it is not read off
235    /// the value — the emitter assigns it — so [`UnfoldLeaf::path`] reaches
236    /// the OPTIONAL value it tests, not a place holding a boolean.
237    Presence,
238}

240/// One hoisted value form: where it sits, and whether it **consumes** the value
241/// it decomposes.
242#[derive(Clone)]
243pub struct Hoist {
244    /// The path prefix to bind, ending in the value form's
245    /// [`PathStep::Call`] (`DeconRecord::Fields`).
246    pub prefix: Vec<PathStep>,
247    /// `true` when the accessor takes its receiver **by value**
248    /// (`f(v: T) -> TStruct`), so the value is moved in and each field can be
249    /// moved *out* into its leaf instead of cloned — the whole point of a
250    /// consuming value form.
251    ///
252    /// Carried on the hoist rather than on [`PathStep::Call`] because only a
253    /// value-form root can consume: the ordinary accessor-chain steps are
254    /// always borrows.
255    pub consuming: bool,
256}

258/// One flattened output leaf of a decomposed return value.
259#[derive(Clone)]
260pub struct UnfoldLeaf {
261    /// The author-supplied leaf name, used **literally** (no casing / stripping /
262    /// keyword escaping). Nested records prefix the child's name with their own
263    /// name, joined by the reserved `"__"` separator (`"sample"` splicing
264    /// `"keyExpr"` → `"sample__keyExpr"`); a root identity leaf is `"handle"`.
265    /// Names are unique within a deconstructor (a duplicate is a hard error).
266    pub name: String,
267    /// Reach chain from the root value (`[]` = the identity/root itself;
268    /// `[Call(f)]` = `f(&root)`; longer = nested records, M3). Steps of both
269    /// kinds may mix — see [`PathStep`].
270    pub path: Vec<PathStep>,
271    /// The **reading** of the type whose resolved output converter encodes this
272    /// leaf — a reference type for accessors (`&str`, `&F`), `&Source` for the
273    /// identity leaf (so the borrowed-opaque clone converter / projection is
274    /// reused). Spell it with `emit.emit_source_type(&out_ty)` in an emission callback.
275    ///
276    /// A reading rather than a spelling because a consumer asking what this
277    /// leaf's type *means* had to hand the spelling back to the registry and
278    /// hope for a cell — the round trip #263 removed from `api/core`, surviving
279    /// in the plans, and answering "no layer" for a type it had never seen
280    /// (#275). The composed ones (`&Source`) are built by
281    /// [`TypeRef::borrowed`](crate::flat::TypeRef::borrowed), which
282    /// pairs the kind with its own spelling.
283    pub out_ty: crate::flat::TypeRef,
284    /// `true` for the move/clone-the-value handle leaf, emitted **last** (after
285    /// every reference leaf's JVM conversion has ended its borrow).
286    pub identity: bool,
287    /// `true` when a nesting accessor on [`Self::path`] returns `Option` (M3):
288    /// the reached value may be absent, so the leaf is nullable on the
289    /// destination side (e.g. a Kotlin `?` type); emit wraps the encode in a
290    /// `match Some/None`.
291    pub nullable: bool,
292    /// How [`Self::path`] is reached from the value — an accessor-fn chain
293    /// (default), a struct-field chain (synthesized `data_class`), a variant
294    /// pattern binding (decomposed sum), or one of the two synthesized
295    /// selectors, which are assigned rather than reached.
296    pub source: LeafSource,
297    /// **Group membership**, as the path of arms this leaf sits inside —
298    /// outermost first. Empty for a leaf that is always live. A non-empty path
299    /// marks the leaf as belonging to the group a **selector** chooses: live
300    /// only when that selector says so, wire-defaulted otherwise.
301    ///
302    /// Each element is one arm, and what an arm number means is the selector's
303    /// own answer:
304    ///
305    /// * a [`SumTag`](LeafSource::SumTag) chooses among alternatives, and its
306    ///   arm is the alternative's tag;
307    /// * a [`Presence`](LeafSource::Presence) chooses between "the value is
308    ///   there" and "it is not", and its arm is `0` — the one group a presence
309    ///   flag gates, carried by the leaves of the value it speaks for.
310    ///
311    /// Grouping is what turns a leaf list into a `match`: leaves sharing a
312    /// group are emitted together in one arm instead of as independent
313    /// per-leaf expressions.
314    ///
315    /// **A selector carries the path it is nested in, not its own arms.** An
316    /// unconditional selector's path is empty; one inside a group carries that
317    /// group's path, the same as any other member — which is what lets a
318    /// selector own another. Its own members extend that path by one, so
319    /// "member of the outer group" and "selector of an inner one" are two
320    /// different lengths of the same path rather than two meanings of one
321    /// number, and [`segments`] reads one nesting
322    /// level at a time (#602).
323    pub groups: Vec<i32>,
324}
325
326impl UnfoldLeaf {
327    /// Whether this leaf's [`out_ty`](Self::out_ty) needs a resolved **output
328    /// converter**. False for a synthesized **selector** — a
329    /// [`SumTag`](LeafSource::SumTag) or a [`Presence`](LeafSource::Presence):
330    /// each is assigned by the emitter rather than converted, so requiring
331    /// a converter for one would make every sum depend on an unrelated `i32`
332    /// crossing existing in the binding.
333    ///
334    /// **This is the root question, not the registration question.** Every
335    /// leaf's `out_ty` gets a table cell; this decides which of them the
336    /// binding additionally *demands* a converter for. A cell says the type
337    /// entered the pipeline, a root says the binding asked for it directly, and
338    /// an entry says one resolved — three separate claims, and a `SumTag` leaf
339    /// makes only the first (#282).
340    pub fn has_converter(&self) -> bool {
341        // A presence flag is synthesized like a tag: it says whether the
342        // leaves after it carry anything, and the value it tests crosses
343        // through those leaves rather than through this one.
344        !matches!(self.source, LeafSource::SumTag | LeafSource::Presence)
345    }
346}

353/// What the walk needs to know about one delivered leaf.
354///
355/// The adapter's own leaf carries far more — its wire type, its JNI or C
356/// encoding, its projection — and none of that reaches here.
357pub trait DecomposedLeaf {
358    /// Steps from the decomposed value to this leaf.
359    fn reach(&self) -> &[PathStep];
360
361    /// The leaf's own name, used in diagnostics and in local names.
362    fn name(&self) -> &str;
363
364    /// Whether this leaf *is* the decomposed value rather than a part of it.
365    fn identity(&self) -> bool;
366
367    /// The reading the leaf delivers.
368    fn source(&self) -> &TypeRef;
369
370    /// Whether this leaf is a synthesized **selector** — a value naming which
371    /// group of the leaves after it is live. It chooses between groups rather
372    /// than joining one, so its own [`Self::groups`] is the path it sits in,
373    /// not one of the arms it chooses.
374    ///
375    /// Two kinds select: a sum's tag, which names the live alternative, and an
376    /// optional value's presence, which says whether its one group carries
377    /// anything.
378    fn selects(&self) -> bool;
379
380    /// The arms this leaf sits inside, outermost first — empty for a leaf that
381    /// is always live. A selector carries the path it is nested in, and its own
382    /// members extend that path by one.
383    ///
384    /// See [`UnfoldLeaf::groups`](UnfoldLeaf::groups) for what
385    /// an arm number means and why the path is what lets one selector own
386    /// another.
387    fn groups(&self) -> &[i32];
388
389    /// Whether the last step reaching it is a field read rather than a call.
390    ///
391    /// Derived from [`Self::reach`], so no implementation can disagree with
392    /// its own path — and the answer decides whether the reached place is
393    /// cloned, which is an ownership decision rather than a spelling.
394    fn is_field_read(&self) -> bool {
395        matches!(self.reach().last(), Some(PathStep::Field { .. }))
396    }
397}

409/// A leaf's reach, rendered **around** an encoding body: the walk hands the
410/// body the reached Rust expression and gets back the encoded value, so an
411/// absent value short-circuits to the adapter's own absence rather than to one
412/// the walk chose. `dyn` on both halves because each side is written where it
413/// is known and neither can name the other's closure.
414pub type Reach<'a> = dyn Fn(&dyn Fn(TokenStream) -> TokenStream) -> TokenStream + 'a;

416/// The adapter's half of a delivery.
417///
418/// The walk in this module owns how a decomposed value is reached — which
419/// hoist a leaf sits under, what is owned, what is borrowed. What a leaf
420/// becomes in the target language, and how the delivered values are handed to
421/// the call, is the adapter's, and this is where it says so. Stated over
422/// [`DecomposedLeaf`] rather than over an adapter's own leaf type, so the walk
423/// can call it.
424pub trait DeliveryBridge {
425    /// The adapter's leaf.
426    type Leaf: DecomposedLeaf;
427
428    /// Where the source item declaring `ident` is qualified from — the one
429    /// fact about a captured path this crate cannot answer for itself.
430    fn qualify(&self, ident: &syn::Ident) -> syn::Path;
431
432    /// The sum a selector leaf names: the qualified source path of the type,
433    /// and its Flat shape.
434    fn sum(&self, leaf: &Self::Leaf) -> (syn::Path, &crate::flat::Variant);
435
436    /// Encode one leaf into the value the delivery call receives, and bind
437    /// that value to `slot`.
438    ///
439    /// The adapter emits the binding rather than the walk, because the slot's
440    /// type is the adapter's: one leaf may cross as a scalar and the next as a
441    /// reference. `index` is the leaf's position in the delivery, which is
442    /// what any temporary the encode needs is named from. An encoding that can
443    /// fail routes its failure through `fail`.
444    fn encode(
445        &self,
446        leaf: &Self::Leaf,
447        index: usize,
448        slot: &syn::Ident,
449        reach: &Reach<'_>,
450        fail: &dyn Fn(TokenStream) -> TokenStream,
451        emit: &crate::RustWriter,
452    ) -> TokenStream;
453
454    /// The expression that passes `slot` as one argument of the delivery call.
455    fn argument(&self, leaf: &Self::Leaf, slot: &syn::Ident) -> TokenStream;
456
457    /// What one leaf's slot holds when the walk fills it without running the
458    /// leaf's encode — the `None` arm of a gate, where a slot exists but the
459    /// value behind it does not.
460    fn slot(&self, leaf: &Self::Leaf) -> Slot;
461
462    /// What a leaf delivers when an optional step on the way to it found
463    /// nothing. [`reach_leaf`] gates on this rather than on a value of its
464    /// own: absence is a target-language representation, and this crate has
465    /// none.
466    fn absent(&self) -> TokenStream;
467}

469/// Compose one [`PathStep`] onto the reference expression reached so far.
470/// A `Call` applies its accessor (origin-qualified); a `Field` reads the field
471/// and re-borrows, so the result is a reference either way and steps chain
472/// uniformly.
473pub fn compose_step(
474    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
475    step: &PathStep,
476    e: TokenStream,
477) -> TokenStream {
478    match step {
479        PathStep::Call { ident, .. } => {
480            let m = qualify(ident);
481            quote!(#m::#ident(#e))
482        }
483        PathStep::Field { ident, .. } => quote!(&(#e).#ident),
484    }
485}
486/// Fold a run of steps onto `e`, borrowing wherever ownership demands it.
487///
488/// [`compose_step`] hands a `Call` its receiver as written, and an accessor
489/// takes that receiver **by reference** — so an owned value in hand has to be
490/// borrowed before the next call composes onto it. A value is in hand whenever
491/// the previous step returned one (`f(..) -> T` rather than `-> &T`), which is
492/// what [`PathStep::yields_owned`] records; `owned` says whether `e` itself
493/// started that way.
494///
495/// A `Field` step needs no borrow either way — it composes as `&(e).f`, which
496/// reads through a value and a reference alike.
497///
498/// This is the ONE place the rule lives, so every fold — a leaf's reach, a
499/// conditional hoist's prefix, a sum's matched value — answers it the same way.
500/// Splitting it produced exactly the bug it exists to prevent: ownership was
501/// handled at the optional binding and at the value form, and lost at every
502/// ordinary call in between.
503pub fn fold_steps(
504    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
505    steps: &[PathStep],
506    mut e: TokenStream,
507    mut owned: bool,
508) -> TokenStream {
509    for step in steps {
510        if owned && matches!(step, PathStep::Call { .. }) {
511            e = quote!(&#e);
512        }
513        e = compose_step(qualify, step, e);
514        owned = step.yields_owned();
515    }
516    e
517}
518/// Compose a value form's OWN call — the one step in the whole system whose
519/// receiver may be by value.
520///
521/// Four cases, from what the fold ended up holding crossed with what the
522/// accessor takes. A CONSUMING form takes its receiver by value: hand it the
523/// value when that is ours, clone when it is not — the same cost the borrowing
524/// form of the accessor would have paid, which keeps one declaration usable by
525/// both owned and `&T` roots. A borrowing form takes a reference, so an owned
526/// value is borrowed for it.
527///
528/// The decision is made from the fold's RESULT, never from where the fold
529/// began: that is what lets a consuming form sit behind ordinary accessors,
530/// where the chain in front borrows and the form itself still moves.
531///
532/// One function because both hoist paths — the conditional binding and the
533/// ordinary one — need exactly this rule, and stating it twice is what turned
534/// each new shape into a new defect.
535fn compose_value_form_call(
536    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
537    call: &PathStep,
538    e: TokenStream,
539    e_owned: bool,
540    consuming: bool,
541) -> TokenStream {
542    match (consuming, e_owned) {
543        (true, owned) => {
544            let (m, f) = (qualify(call.ident()), call.ident());
545            // Parenthesized: the clone applies to whatever the fold holds, and
546            // `&x.clone()` would parse as `&(x.clone())`.
547            let arg = if owned { e } else { quote!((#e).clone()) };
548            quote!(#m::#f(#arg))
549        }
550        (false, true) => compose_step(qualify, call, quote!(&#e)),
551        (false, false) => compose_step(qualify, call, e),
552    }
553}
554/// Start a reach from `base`, projecting the **leading run of plain field
555/// steps** directly (`&base.a.b`) instead of through a borrow of the base
556/// (`&(&base).a.b`). Returns the expression and how many steps it consumed.
557///
558/// The two forms name the same value, but the second borrows the base **as a
559/// whole**, which the borrow checker rejects once a sibling leaf has moved a
560/// different field out of it. Projecting directly makes each leaf's borrow
561/// disjoint, so a consuming value form's field moves are order-independent
562/// rather than compiling only while the borrowing leaves happen to be declared
563/// first.
564pub fn project_leading_fields(
565    base: &TokenStream,
566    base_is_ref: bool,
567    path: &[PathStep],
568) -> (TokenStream, usize) {
569    if base_is_ref {
570        return (base.clone(), 0);
571    }
572    let n = path.iter().take_while(|s| s.is_plain_field()).count();
573    if n == 0 {
574        return (quote!(&#base), 0);
575    }
576    let segs: Vec<&syn::Ident> = path[..n].iter().map(PathStep::ident).collect();
577    (quote!(&#base #(.#segs)*), n)
578}

580/// Where one leaf's reach starts, once the hoists are bound.
581///
582/// Produced by `Hoisted::place`, which is the one place that decides it: the
583/// innermost value form the leaf sits under, the name a conditional form's
584/// `Some` arm binds, or the delivered value itself.
585pub struct LeafPlace {
586    /// What to reach from.
587    pub base: TokenStream,
588    /// Whether `base` is already a reference.
589    pub base_is_ref: bool,
590    /// The steps left from `base` — the leaf's own path with the prefix that
591    /// bound the hoist already consumed.
592    pub path: Vec<PathStep>,
593    /// Whether the form that produced `base` gave its value away.
594    pub consuming: bool,
595    /// The conditional hoist whose `Some` arm this leaf belongs in. Its
596    /// statements go there rather than beside the others, because the arm is
597    /// where the binding they reach off exists.
598    pub conditional: Option<usize>,
599}
600
601impl LeafPlace {
602    /// The place this leaf may MOVE out of, or `None` when what it reaches is
603    /// not the delivery's to give away, or is not a place a move can name.
604    ///
605    /// The same question [`reach_leaf`] answers on its way to the terminal
606    /// treatment, asked separately by a leaf whose encode needs the place
607    /// itself rather than an expression reaching it — a handle that is boxed
608    /// rather than converted.
609    pub fn owned<L: DecomposedLeaf>(&self, leaf: &L) -> Option<TokenStream> {
610        (reached_is_ours(leaf, self.consuming) && steps_are_movable(&self.path)).then(|| {
611            let base = &self.base;
612            let segs: Vec<&syn::Ident> = self.path.iter().map(PathStep::ident).collect();
613            quote!(#base #(.#segs)*)
614        })
615    }
616}

618/// Whether what a leaf reaches is OURS, and so is moved rather than borrowed
619/// or cloned. The two leaf kinds say it differently:
620///
621/// * an IDENTITY leaf carries the answer in its own reading — the plan
622///   resolved that to the owned type exactly when the value is the plan's to
623///   give away (an owned root, or a field of a CONSUMING value form), and that
624///   is also what selected the owning converter, which boxes the move rather
625///   than cloning a borrow;
626/// * every other leaf reads a field, whose reading is the field type as
627///   written and owned either way, so ownership is the enclosing form's: only
628///   a consuming one gives its fields away.
629pub fn reached_is_ours<L: DecomposedLeaf>(leaf: &L, consuming: bool) -> bool {
630    if leaf.identity() {
631        !matches!(leaf.source().kind(), crate::flat::TypeKind::Ref { .. })
632    } else {
633        consuming
634    }
635}

637/// One leaf, and the place a reach of it starts from.
638///
639/// A caller that has already bound a hoist hands over the local it bound and
640/// the steps that are left, which is why `path` is a suffix of the leaf's own
641/// reach rather than always equal to it.
642pub struct LeafAt<'a, L> {
643    /// The leaf being reached.
644    pub leaf: &'a L,
645    /// The steps still to walk from `base`.
646    pub path: &'a [PathStep],
647    /// What the walk starts from — the delivered value, or a local a hoist or
648    /// a gate bound.
649    pub base: TokenStream,
650    /// Whether `base` is already a reference.
651    pub base_is_ref: bool,
652    /// Whether the form that produced `base` gave its value away, which is
653    /// what lets a field of it be moved rather than cloned.
654    pub consuming: bool,
655    /// Whether a FINAL optional step is gated too. An identity leaf delivers
656    /// the reached value itself, so it is; every other leaf hands the final
657    /// step's full type to its own converter, `Option` and all.
658    pub unwrap_last: bool,
659}

661/// Reach one leaf from `base` and hand what it reached to `body`.
662///
663/// Three things happen on the way, and all three are facts about Rust values
664/// rather than about any target language:
665///
666/// * **Gating.** An optional step before the leaf becomes a `match` whose
667///   `None` arm yields `absent()` — the adapter's own absence, since this
668///   crate has none to offer. `unwrap_last` says whether a FINAL optional step
669///   is gated too: an identity leaf delivers the reached value itself, so it
670///   is, while every other leaf hands the final step's full type to its own
671///   converter, `Option` and all.
672/// * **Refusal.** `absent` is `None` at a site that cannot express absence at
673///   all — a single delivered return value, which has no arm to put one in.
674///   An optional step before the end is then refused rather than composed into
675///   code the consumer's crate cannot type-check.
676/// * **Ownership**: the reached place is moved when it is ours, cloned when
677///   it is a field read of a place that is not, and borrowed otherwise.
678///
679/// One derivation, used by every delivery an adapter renders and by the
680/// single-leaf shortcut a decomposed return takes. They drifted once while
681/// they were two — the shortcut was missing the field clone and handed `&F` to
682/// an `F` converter — and a second pair, ungated here and gated in JniGen,
683/// drifted the same way until #607 folded them together.
684pub fn reach_leaf<L: DecomposedLeaf>(
685    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
686    at: LeafAt<'_, L>,
687    absent: Option<&dyn Fn() -> TokenStream>,
688    body: &dyn Fn(TokenStream) -> TokenStream,
689) -> TokenStream {
690    reach_leaf_at(qualify, at, absent, 0, body)
691}
692
693fn reach_leaf_at<L: DecomposedLeaf>(
694    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
695    at: LeafAt<'_, L>,
696    absent: Option<&dyn Fn() -> TokenStream>,
697    depth: usize,
698    body: &dyn Fn(TokenStream) -> TokenStream,
699) -> TokenStream {
700    let LeafAt {
701        path,
702        ref base,
703        base_is_ref,
704        unwrap_last,
705        ..
706    } = at;
707    // Every optional step on the way to the leaf becomes a `match` whose
708    // `None` arm is the adapter's absent value. A site with no absent value to
709    // give — a single delivered return, which has no arm to put one in —
710    // passes `None`, and an optional step before the end is then refused
711    // rather than composed into code the consumer's crate cannot type-check.
712    if let Some(absent) = absent {
713        let limit = if unwrap_last {
714            path.len()
715        } else {
716            // A non-identity leaf's converter takes the final step's FULL type,
717            // `Option` included, so only the steps before it are nesting.
718            path.len().saturating_sub(1)
719        };
720        let (projected, lead) = project_leading_fields(base, base_is_ref, path);
721        if let Some(k) = (lead..limit).find(|&i| path[i].is_optional()) {
722            // Through the optional step INCLUSIVE: the same fold, so the borrow
723            // in front of it is the ordinary rule rather than a second
724            // statement of it.
725            let opt_e = fold_steps(qualify, &path[lead..=k], projected, false);
726            let nested = format_ident!("__n{}", depth);
727            // What the arm binds is the step's OWN value: an owned payload is
728            // a bare `T`, so composing the next step onto it directly would
729            // hand `T` to an accessor typed for `&T`. Say it is not a
730            // reference and let `project_leading_fields` borrow it; a borrowed
731            // payload is already one and passes through. With no steps left
732            // the binding goes to `body` untouched, which is what lets an
733            // owned payload be moved rather than borrowed straight back.
734            //
735            // The same rule [`reach_optional`] states for a conditional
736            // hoist's prefix. This used to be a hardcoded `true` in the gated
737            // reach JniGen owned, where an `Option<T>` accessor followed by
738            // another accessor emitted `next(__n0)` against a `&T` receiver.
739            let rest = &path[k + 1..];
740            let inner = reach_leaf_at(
741                qualify,
742                LeafAt {
743                    path: rest,
744                    base: quote!(#nested),
745                    base_is_ref: rest.is_empty() || !path[k].yields_owned(),
746                    ..at
747                },
748                Some(absent),
749                depth + 1,
750                body,
751            );
752            let gone = absent();
753            // A FIELD read composes to a borrow (`&(e).f`), so it goes through
754            // a coercion site and the destructuring stops caring which
755            // representation the source spelled the optional as (#268).
756            //
757            // A CALL composes to the accessor's own returned value, which is
758            // owned and whose payload downstream may move. Borrowing it to
759            // coerce would change that ownership, so it keeps its direct match.
760            if path[k].is_field() {
761                let opt_bind = format_ident!("__o{}", depth);
762                return quote! {
763                    {
764                        let #opt_bind: &::core::option::Option<_> = #opt_e;
765                        match #opt_bind {
766                            ::core::option::Option::Some(#nested) => { #inner }
767                            ::core::option::Option::None => #gone,
768                        }
769                    }
770                };
771            }
772            return quote! {
773                match #opt_e {
774                    ::core::option::Option::Some(#nested) => { #inner }
775                    ::core::option::Option::None => #gone,
776                }
777            };
778        }
779        return body(reached_place(qualify, &at));
780    }
781    // An optional step BEFORE the last one needs a `match` whose `None` arm has
782    // somewhere to go. This derivation has none — it yields a plain Rust value,
783    // not a representation that can carry absence — so the shape is refused
784    // here rather than composed into code that cannot type-check in the
785    // consumer's crate.
786    //
787    // Asked of the leaf's OWN path, not of `path`. The caller may hand a
788    // suffix: `wrapper.rs` rebases onto a hoisted local, and `Hoisted::innermost`
789    // strips the prefix that bound it — including any optional step inside it.
790    // Checking the parameter would therefore pass exactly when the hoist is the
791    // conditional one, which is the case that cannot compose (an `Option<T>`
792    // local with a field read hung off it). The full path is what the shape
793    // question is about.
794    let own_path = at.leaf.reach();
795    assert!(
796        !own_path.iter().rev().skip(1).any(PathStep::is_optional),
797        "unfold: leaf `{}` reaches through an optional step but is \
798         delivered as a single return value, which has no `None` arm — this \
799         shape needs callback delivery",
800        at.leaf.name(),
801    );
802    // Whether what this leaf reaches is OURS, and so is moved rather than
803    // borrowed or cloned. The two leaf kinds say it differently:
804    //
805    // * an IDENTITY leaf carries the answer in its `out_ty` — the plan resolved
806    //   it to the owned type exactly when the value is the plan's to give away
807    //   (`place_is_owned`: an owned root, or a field of a CONSUMING value form),
808    //   and that is also what selected the owning converter, which boxes the
809    //   move rather than cloning a borrow;
810    // * a FIELD leaf's `out_ty` is the field type as written, owned either way,
811    //   so ownership is the enclosing form's: only a consuming one gives its
812    //   fields away.
813    //
814    // How to project that place is `steps_are_movable`'s question, and it is
815    // asked there rather than restated here. This used to spell it
816    // `all(is_plain_field)`, defending the restatement on the grounds that a
817    // trailing `Option` cannot reach return delivery anyway — true, and enforced
818    // in `single_return` (`core/unfold.rs`), which is precisely why a local
819    // restatement could disagree with the rule for as long as the invariant held
820    // somewhere else. `plan.rs` says two readings would drift and the
821    // disagreement would be a borrow handed to an owning converter; this is the
822    // second reading, removed.
823    body(reached_place(qualify, &at))
824}

826/// The place a leaf reaches, with the terminal treatment its ownership calls
827/// for: moved out when it is ours and the path projects a place, cloned out
828/// when it is a field read of a place that is not, and borrowed otherwise.
829fn reached_place<L: DecomposedLeaf>(
830    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
831    at: &LeafAt<'_, L>,
832) -> TokenStream {
833    let LeafAt {
834        leaf,
835        path,
836        base,
837        base_is_ref,
838        consuming,
839        ..
840    } = at;
841    let (base_is_ref, consuming) = (*base_is_ref, *consuming);
842    // How to project a movable place is `steps_are_movable`'s question, asked
843    // there rather than restated here. This used to spell it
844    // `all(is_plain_field)`, defending the restatement on the grounds that a
845    // trailing `Option` cannot reach return delivery anyway — true, and
846    // enforced in `single_return`, which is precisely why a local restatement
847    // could disagree with the rule for as long as the invariant held somewhere
848    // else. `plan.rs` says two readings would drift and the disagreement would
849    // be a borrow handed to an owning converter; this is the second reading,
850    // removed.
851    if reached_is_ours(*leaf, consuming) && steps_are_movable(path) {
852        let segs: Vec<&syn::Ident> = path.iter().map(PathStep::ident).collect();
853        return quote!(#base #(.#segs)*);
854    }
855    let (e, lead) = project_leading_fields(base, base_is_ref, path);
856    let e = fold_steps(qualify, &path[lead..], e, false);
857    // Cloned out of the BORROW the reach yields, not out of the place behind
858    // it. The two agree for a field of an owned type and disagree for a field
859    // that is itself a reference: `place.clone()` there resolves through the
860    // reference and deep-clones the pointee, where the converter takes the
861    // field type as written. JniGen's own reach spelled it the short way and
862    // no declared field exercised the difference — the drift this step exists
863    // to close.
864    if leaf.is_field_read() {
865        quote!((#e).clone())
866    } else {
867        e
868    }
869}
870/// The **segments** of a decomposition: each selector leaf, together with the
871/// group leaves that follow it.
872///
873/// A segment's leaves are not independent — one alternative of a sum is live
874/// per value, and a gated group is live only when its value is there — so a
875/// segment is emitted as one `match` rather than as a leaf at a time. The plan
876/// already says which leaves those are: a selector says so of itself, and each
877/// of the leaves after it carries the group it belongs to until one does not.
878///
879/// **One nesting level at a time.** A group's leaves may themselves include a
880/// selector — an optional nested class whose own fields choose, an `Option` of
881/// a sum — and the ranges returned here would overlap if such an inner selector
882/// opened a segment of its own. So this answers for one level: at `depth`, the
883/// selectors whose arm path is exactly that deep, each with everything nested
884/// under it. A renderer that has entered an arm asks again one level down, and
885/// the inner selectors become the segments of that answer.
886///
887/// A decomposition may carry several segments, or none: a sum that IS the
888/// delivered value is one segment covering everything, and a value form
889/// contributes one per sum-typed field.
890///
891/// Not recognising a segment is silent rather than a compile error — the
892/// leaves are all there, and a delivery that treats them as independent reads
893/// a dead alternative's fields — which is why this is the registry's answer
894/// and not an adapter's.
895pub fn segments<L: DecomposedLeaf>(leaves: &[L]) -> Vec<std::ops::Range<usize>> {
896    segments_at(leaves, 0)
897}
898
899/// [`segments`] one nesting level down: the selectors whose arm path is exactly
900/// `depth` long, with the leaves nested under each.
901///
902/// `depth` is how many arms the caller has already entered, so a leaf deeper
903/// than that belongs to the segment it follows rather than opening one.
904pub fn segments_at<L: DecomposedLeaf>(leaves: &[L], depth: usize) -> Vec<std::ops::Range<usize>> {
905    let n = leaves.len();
906    let nested = |i: usize| leaves[i].groups().len() > depth;
907    // Exactly this deep, not "no deeper": a selector shallower than `depth`
908    // encloses the level being asked about rather than sitting on it, and
909    // answering with it would return the enclosing segment a second time.
910    (0..n)
911        .filter(|&i| leaves[i].selects() && leaves[i].groups().len() == depth)
912        .map(|start| {
913            let end = (start + 1..n)
914                .take_while(|&i| nested(i))
915                .last()
916                .map_or(start + 1, |i| i + 1);
917            start..end
918        })
919        .collect()
920}

922/// Render one segment off `place`, gating the whole of it when the selector
923/// reaches the value it selects on through an optional step.
924///
925/// A segment's **selected value** is what its selector chooses over: the sum a
926/// tag names an alternative of, or the optional value a presence flag says is
927/// there. `group` renders the segment's own encode from the expression naming
928/// it — that half is the adapter's, since what a group of slots is filled with
929/// is a target-language question.
930///
931/// **An optional value gates the segment, not each slot.** A segment's leaves
932/// are not independent, so absence cannot be the per-leaf absent value
933/// [`reach_leaf`] gives an ordinary optional field: it is one tuple bind whose
934/// `None` arm carries every slot's default. That is the same shape a
935/// conditional value form's hoist emits, applied to an optional step inside
936/// the segment's own path (#220).
937///
938/// `index` names the segment's locals, so two segments of one delivery cannot
939/// collide.
940pub fn segment<B: DeliveryBridge>(
941    bridge: &B,
942    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
943    place: &LeafPlace,
944    index: usize,
945    leaves: &[B::Leaf],
946    slots: &[syn::Ident],
947    group: &dyn Fn(TokenStream) -> TokenStream,
948) -> TokenStream {
949    let path = &place.path;
950    // A plain field chain is borrowed DIRECTLY (`&base.a.b`) rather than
951    // through the base (`&(&base).a.b`). The two are the same value, but the
952    // second borrows the base as a whole, which the borrow checker rejects
953    // once a sibling leaf has moved another field out of it — and borrowing
954    // this field while sibling fields move is exactly what a consuming value
955    // form does.
956    let (projected, lead) = project_leading_fields(&place.base, place.base_is_ref, path);
957    // The selector's own path reaches the selected value (empty when that
958    // value IS the delivered one), and a step on it MAY be optional — which is
959    // always so for a presence flag, whose whole subject is that step.
960    let Some(k) = (lead..path.len()).find(|&i| path[i].is_optional()) else {
961        return group(fold_steps(qualify, &path[lead..], projected, false));
962    };
963    // Through the optional step INCLUSIVE, then the rest off the binding — the
964    // same split [`reach_leaf`] makes, so the borrow in front of it stays the
965    // ordinary rule rather than a second statement of it.
966    let opt_e = fold_steps(qualify, &path[lead..=k], projected, false);
967    let bind = format_ident!("__sg{}", index);
968    // ONE optional step is what this gate handles. A second one in the tail
969    // would compose `match &Option<..>` against the patterns the group's own
970    // encode writes — an E0308 in the consumer's crate — so the named
971    // diagnostic is raised here instead.
972    //
973    // `assert!`, not `debug_assert!`: a build script inherits the consumer's
974    // profile, so a debug-only check is absent from exactly the release build
975    // where a mis-emission costs the most to diagnose. Same rule, same
976    // phrasing, as the single-return optional-step assert in [`reach_leaf`].
977    //
978    // The condition is what actually breaks, not the stronger fact that
979    // happens to hold: a selector's path stops AT the value it selects on, so
980    // the tail is empty today, but a NON-optional tail composes correctly
981    // through [`fold_steps`] — refusing it would refuse a shape that works.
982    assert!(
983        !path[k + 1..].iter().any(PathStep::is_optional),
984        "unfold: leaf `{}` reaches the value it selects on through TWO optional \
985         steps — the segment gate has one `None` arm, so the second would be \
986         matched as if it were that value itself",
987        leaves
988            .first()
989            .expect("a segment has at least its selector")
990            .name(),
991    );
992    // What the `match` binds, asked of the step rather than assumed: the FIELD
993    // branch scrutinizes `&Option<_>`, so ergonomics binds a borrow of the
994    // selected value. Only an owned-yielding CALL binds an owned one.
995    let inner = group(fold_steps(
996        qualify,
997        &path[k + 1..],
998        quote!(#bind),
999        path[k].yields_owned(),
1000    ));
1001    let filled: Vec<Slot> = leaves.iter().map(|leaf| bridge.slot(leaf)).collect();
1002    let tys = filled.iter().map(|slot| &slot.ty);
1003    let defaults = filled.iter().map(|slot| &slot.default);
1004    // A FIELD step composes to a borrow, so it goes through a coercion site
1005    // and the destructure stops caring which representation the source spelled
1006    // the optional as (#268). A CALL yields its own owned value, whose payload
1007    // downstream may move, so it keeps the direct match — the same division
1008    // [`reach_leaf`] makes.
1009    let (prelude, scrutinee) = if path[k].is_field() {
1010        let opt_bind = format_ident!("__so{}", index);
1011        (bind_as_option(&opt_e, &opt_bind), quote!(#opt_bind))
1012    } else {
1013        (TokenStream::new(), opt_e)
1014    };
1015    quote! {
1016        let (#(#slots,)*): (#(#tys,)*) = {
1017            #prelude
1018            match #scrutinee {
1019                ::core::option::Option::Some(#bind) => {
1020                    #inner
1021                    (#(#slots,)*)
1022                }
1023                ::core::option::Option::None => (#(#defaults,)*),
1024            }
1025        };
1026    }
1027}

1029/// The `match` a **conditional** value form's leaves share.
1030///
1031/// A hoist under an optional step ran only where the value was present, so its
1032/// leaves cannot be independent statements: their slots exist either way, and
1033/// only one arm computes them. The `Some` arm runs `body` — the statements
1034/// those leaves contributed — and yields their slots as a tuple; the `None`
1035/// arm yields each slot's default, the same shape [`segment`] gives an absent
1036/// sum. Binding the tuple outside the `match` keeps the slots in scope for the
1037/// call's argument list, which is indifferent to how a slot was filled.
1038///
1039/// Matched BY VALUE: the local is this arm's alone, every leaf under the hoist
1040/// being in it, so a consuming value form's fields move out here exactly as
1041/// they do at an unconditional one.
1042pub fn conditional_arm<B: DeliveryBridge>(
1043    bridge: &B,
1044    hoisted: &Hoisted,
1045    index: usize,
1046    leaves: &[B::Leaf],
1047    slots: &[syn::Ident],
1048    body: TokenStream,
1049) -> TokenStream {
1050    let local = hoisted.local(index);
1051    let bind = format_ident!("__u{}", index);
1052    let under: Vec<usize> = (0..leaves.len())
1053        .filter(|&k| {
1054            hoisted
1055                .conditional(leaves[k].reach())
1056                .is_some_and(|(j, ..)| j == index)
1057        })
1058        .collect();
1059    let ids: Vec<&syn::Ident> = under.iter().map(|&k| &slots[k]).collect();
1060    let filled: Vec<Slot> = under.iter().map(|&k| bridge.slot(&leaves[k])).collect();
1061    let tys = filled.iter().map(|slot| &slot.ty);
1062    let defaults = filled.iter().map(|slot| &slot.default);
1063    quote! {
1064        let (#(#ids,)*): (#(#tys,)*) = match #local {
1065            ::core::option::Option::Some(#bind) => { #body (#(#ids,)*) }
1066            ::core::option::Option::None => (#(#defaults,)*),
1067        };
1068    }
1069}

1071/// Bind `e` so it can be destructured as an `Option` **whatever Rust
1072/// representation the source used for it**.
1073///
1074/// The model says a position is optional; it deliberately does not say whether
1075/// Rust spells that `Option<T>`, `Box<Option<T>>`, or something else — the
1076/// side interpreting the classification is the side that must accept any
1077/// representation. Matching the reached place directly assumed one, which is
1078/// "classify off the model, spell from the model" broken in the direction
1079/// nothing was watching: `Box<Option<T>>` then produced
1080/// `match &place { Some(..) => .. }` and an E0308 (#268).
1081///
1082/// A type-ascribed `let` is a coercion site, and deref coercion is transitive
1083/// **and** a no-op when the types already match — so this one shape serves
1084/// every representation, and the plain `Option<T>` case is unchanged.
1085///
1086/// `e` is expected to be a **reference** already — [`compose_step`] composes a
1087/// field read as `&(e).f` — so nothing is borrowed here. Borrowing only: an
1088/// owned position cannot be made representation-agnostic this way, because
1089/// deref coercion applies to references and moving out of a wrapper is
1090/// something only some of them permit (`Box` does, `Rc` cannot). A site that
1091/// must MOVE the payload keeps its direct match.
1092pub fn bind_as_option(e: &TokenStream, bind: &syn::Ident) -> TokenStream {
1093    quote! { let #bind: &::core::option::Option<_> = #e; }
1094}

1096/// Every value form on a plan, evaluated **once** and bound to a local
1097/// (`__vf0`, `__vf1`, …), so a struct is built once per delivery rather than
1098/// once per field. The bound prefixes come back with the statements, since
1099/// reaching a leaf means starting from the innermost local it sits under.
1100///
1101/// Shared by both delivery paths — the multi-leaf encoder below and the
1102/// single-leaf `Delivery::Return` shortcut in `emit/wrapper.rs`. The shortcut
1103/// used to compose its reach straight off the raw value, which for a consuming
1104/// value form emitted `f(&v)` against a by-value receiver: ill-typed Rust in
1105/// the consumer's crate. One binder, so the two cannot disagree about what a
1106/// hoist is or who owns it.
1107pub struct Hoisted {
1108    /// The `let __vfN = …;` bindings, outermost-first.
1109    pub stmts: TokenStream,
1110    /// Each hoist's path prefix and the local it was bound to.
1111    bound: Vec<(Vec<PathStep>, syn::Ident)>,
1112    /// Whether each bound hoist consumed the value it decomposed.
1113    consuming: Vec<bool>,
1114    /// Whether each bound local is `Option<TStruct>` rather than `TStruct` —
1115    /// the hoist sits under an optional step, so the value form ran only where
1116    /// the value was present. Its leaves cannot be emitted as independent
1117    /// statements: they share ONE `match` on the local (see
1118    /// [`encode_plan_leaves`]), taken by value, so a consuming form's fields
1119    /// still move out inside the arm.
1120    optional: Vec<bool>,
1121}
1122impl Hoisted {
1123    /// Index of the innermost bound hoist `path` sits under, with that prefix
1124    /// already consumed. `None` for a leaf under no value form at all — a
1125    /// sibling `.field()` / `.field_self()`, which still reaches from the value
1126    /// itself.
1127    fn innermost(&self, path: &[PathStep]) -> Option<(usize, Vec<PathStep>)> {
1128        self.bound
1129            .iter()
1130            .enumerate()
1131            .filter(|(_, (p, _))| p.len() < path.len() && path.starts_with(p))
1132            .max_by_key(|(_, (p, _))| p.len())
1133            .map(|(i, (p, _))| (i, path[p.len()..].to_vec()))
1134    }
1135
1136    /// The innermost bound local `path` sits under, with that prefix already
1137    /// consumed, and whether that hoist gave its value away.
1138    pub fn rebase(&self, path: &[PathStep]) -> Option<(syn::Ident, Vec<PathStep>, bool)> {
1139        self.innermost(path)
1140            .map(|(i, rest)| (self.bound[i].1.clone(), rest, self.consuming[i]))
1141    }
1142
1143    /// The innermost **conditional** hoist `path` sits under: its index, the
1144    /// local holding the `Option`, the name its `Some` arm binds, and the steps
1145    /// left to reach the leaf from there. `None` when the leaf's innermost
1146    /// hoist is unconditional (or there is none) — then [`Self::rebase`]
1147    /// applies and the leaf is an ordinary standalone statement.
1148    pub fn conditional(
1149        &self,
1150        path: &[PathStep],
1151    ) -> Option<(usize, syn::Ident, syn::Ident, Vec<PathStep>)> {
1152        let (i, rest) = self.innermost(path)?;
1153        self.optional[i].then(|| (i, self.bound[i].1.clone(), format_ident!("__u{}", i), rest))
1154    }
1155
1156    /// Where a leaf's reach starts, once these hoists are bound.
1157    ///
1158    /// Three cases, and the leaf's own path decides which.
1159    ///
1160    /// A leaf under a **conditional** value form reaches off the name that
1161    /// form's `Some` arm binds. That arm matches the hoist's `Option` by
1162    /// value — every leaf under the hoist is inside it, so the local is the
1163    /// arm's alone — which makes the binding an owned payload: a consuming
1164    /// form's fields move out of it exactly as they do at an unconditional
1165    /// one, and [`LeafPlace::owned`] says so. The leaf's statements belong in
1166    /// that arm, because that is where the binding exists.
1167    ///
1168    /// A leaf under an **ordinary** value form reaches off the local that form
1169    /// was bound to, with the prefix already consumed and the form's own
1170    /// ownership carried along.
1171    ///
1172    /// A leaf under **no** value form — a sibling accessor of the delivered
1173    /// value — reaches from that value.
1174    pub fn place<L: DecomposedLeaf>(
1175        &self,
1176        leaf: &L,
1177        value: &TokenStream,
1178        by_ref: bool,
1179    ) -> LeafPlace {
1180        let path = leaf.reach();
1181        if let Some((i, _, bind, rest)) = self.conditional(path) {
1182            return LeafPlace {
1183                base: quote!(#bind),
1184                base_is_ref: false,
1185                path: rest,
1186                consuming: self.consumed(i),
1187                conditional: Some(i),
1188            };
1189        }
1190        match self.rebase(path) {
1191            Some((local, rest, consuming)) => LeafPlace {
1192                base: quote!(#local),
1193                base_is_ref: false,
1194                path: rest,
1195                consuming,
1196                conditional: None,
1197            },
1198            None => LeafPlace {
1199                base: value.clone(),
1200                base_is_ref: by_ref,
1201                path: path.to_vec(),
1202                consuming: false,
1203                conditional: None,
1204            },
1205        }
1206    }
1207
1208    /// The local a hoist was bound to.
1209    pub fn local(&self, i: usize) -> syn::Ident {
1210        self.bound[i].1.clone()
1211    }
1212
1213    /// Whether a hoist consumed the value it decomposed.
1214    pub fn consumed(&self, i: usize) -> bool {
1215        self.consuming[i]
1216    }
1217}
1218/// Fold `path` over `base` the way an adapter's own gated reach does, but
1219/// yielding an `Option<…>` rather than the adapter's absent value: the optional
1220/// steps become a `map`/`and_then` chain, so an absent value short-circuits to
1221/// `None` instead of to whatever that adapter uses for absence. `body` renders
1222/// the innermost reached expression as a BARE value — the chain's last link
1223/// wraps it.
1224///
1225/// The gated reach this mirrors is [`reach_leaf`], which takes the absent
1226/// value from the adapter rather than choosing one. This one yields an
1227/// `Option` instead, because what it binds is a hoist rather than a delivered
1228/// leaf: the value form runs where the value is present, and the `None` the
1229/// chain produces is the local's own.
1230///
1231/// This is how a CONDITIONAL value form is bound — the accessor runs only where
1232/// the value it decomposes is actually present.
1233fn reach_optional(
1234    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
1235    path: &[PathStep],
1236    base: TokenStream,
1237    base_is_ref: bool,
1238    depth: usize,
1239    body: &dyn Fn(TokenStream) -> TokenStream,
1240) -> TokenStream {
1241    let (e, lead) = project_leading_fields(&base, base_is_ref, path);
1242    match (lead..path.len()).find(|&i| path[i].is_optional()) {
1243        None => body(fold_steps(qualify, &path[lead..], e, false)),
1244        Some(k) => {
1245            // Through the optional step INCLUSIVE: the same fold, so the
1246            // borrow in front of it is the ordinary rule rather than a second
1247            // statement of it.
1248            let opt_e = fold_steps(qualify, &path[lead..=k], e, false);
1249            let bind = format_ident!("__hb{}", depth);
1250            // What the arm binds is the step's own value: an OWNED payload is a
1251            // bare `T`, so composing the next step onto it directly would hand
1252            // `T` to an accessor typed for `&T`. Say it is not a reference and
1253            // let `project_leading_fields` borrow it; a borrowed payload is
1254            // already one and passes through.
1255            //
1256            // With NO steps left the binding goes to `body` untouched — that is
1257            // what lets a consuming value form MOVE an owned payload rather than
1258            // borrow it straight back, so the terminal case stays "already a
1259            // reference" whatever the payload is.
1260            let rest = &path[k + 1..];
1261            let inner = reach_optional(
1262                qualify,
1263                rest,
1264                quote!(#bind),
1265                rest.is_empty() || !path[k].yields_owned(),
1266                depth + 1,
1267                body,
1268            );
1269            // `map` when this is the LAST optional step (the body yields a bare
1270            // value) and `and_then` when another follows (the recursion yields
1271            // an `Option` that must not nest). The equivalent `match` reads the
1272            // same but generated code runs through the consumer's lints, where
1273            // `clippy::manual_map` is a denial.
1274            let combinator = if rest.iter().any(PathStep::is_optional) {
1275                format_ident!("and_then")
1276            } else {
1277                format_ident!("map")
1278            };
1279            quote! {
1280                #opt_e.#combinator(|#bind| #inner)
1281            }
1282        }
1283    }
1284}
1285pub fn bind_hoists(
1286    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
1287    hoists: &[Hoist],
1288    value: &TokenStream,
1289    by_ref: bool,
1290) -> Hoisted {
1291    let mut out = Hoisted {
1292        stmts: TokenStream::new(),
1293        bound: Vec::new(),
1294        consuming: Vec::new(),
1295        optional: Vec::new(),
1296    };
1297    // Value forms COMPOSE, so each hoist is built from the longest hoist that
1298    // is already a proper prefix of it (they arrive outermost-first), and from
1299    // `value` otherwise.
1300    for (i, h) in hoists.iter().enumerate() {
1301        let local = format_ident!("__vf{}", i);
1302        // A hoist under an optional step binds `Option<TStruct>`: the value
1303        // form runs in the `Some` arm only. Core refuses to nest these, so the
1304        // enclosing value is always the plan's own — no rebase to consider.
1305        if h.prefix.iter().any(PathStep::is_optional) {
1306            let (last, lead) = h
1307                .prefix
1308                .split_last()
1309                .expect("a hoist prefix ends in its value-form call");
1310            let consuming = h.consuming;
1311            // The value form is handed the payload only when the optional step
1312            // is the LAST thing before it; any step in between composes as a
1313            // borrow, so what arrives is a reference either way.
1314            let owned = lead.last().is_some_and(PathStep::yields_owned);
1315            let expr = reach_optional(qualify, lead, value.clone(), by_ref, 0, &|reached| {
1316                compose_value_form_call(qualify, last, reached, owned, consuming)
1317            });
1318            out.stmts.extend(quote! { let #local = #expr; });
1319            out.bound.push((h.prefix.clone(), local));
1320            out.consuming.push(h.consuming);
1321            out.optional.push(true);
1322            continue;
1323        }
1324        // Where the fold starts, and whether what it starts from is OWNED. The
1325        // value form's own boundary is decided below, from what the fold ends
1326        // up holding — never from where it began.
1327        let (from, start, start_owned) = match out.rebase(&h.prefix) {
1328            // A NESTED consuming form is handed the parent's field by MOVE: a
1329            // hoisted value form is an owned struct and its fields are
1330            // disjoint, so moving one out leaves every sibling leaf readable.
1331            // `compose_step` borrows (`&(e).f`), so a plain field run to that
1332            // field is projected here instead of going through it.
1333            Some((outer, rest, _))
1334                if h.consuming && rest[..rest.len() - 1].iter().all(PathStep::is_plain_field) =>
1335            {
1336                let lead = &rest[..rest.len() - 1];
1337                let segs: Vec<&syn::Ident> = lead.iter().map(PathStep::ident).collect();
1338                (h.prefix.len() - 1, quote!(#outer #(.#segs)*), true)
1339            }
1340            // Any other rebased hoist: project its own leading field run
1341            // DIRECTLY off the parent local rather than reaching it through a
1342            // borrow of the parent. A sibling hoist may already have moved a
1343            // different field out — that is what a consuming value form does —
1344            // and `&(&__vf0).wrapper` borrows the partially moved parent as a
1345            // whole where `&__vf0.wrapper` is a disjoint borrow that survives.
1346            // Same invariant `project_leading_fields` states for leaf reaches,
1347            // and the same reason.
1348            Some((outer, rest, _)) => {
1349                let (e, lead) = project_leading_fields(&quote!(#outer), false, &rest);
1350                (h.prefix.len() - rest.len() + lead, e, false)
1351            }
1352            None if by_ref => (0, value.clone(), false),
1353            None => (0, value.clone(), true),
1354        };
1355        // Everything before the value form is an ordinary accessor chain.
1356        let last = h.prefix.len() - 1;
1357        let head = &h.prefix[from..last];
1358        let e = fold_steps(qualify, head, start, start_owned);
1359        let e_owned = head.last().map_or(start_owned, PathStep::yields_owned);
1360        // The value form itself. A CONSUMING one takes its receiver BY VALUE —
1361        // that is the move the whole declaration exists for — so it is handed
1362        // what the fold holds when that is ours, and a clone when it is not:
1363        // the same cost the borrowing form of the accessor would have paid,
1364        // which keeps one declaration usable by both owned and `&T` returns.
1365        // A borrowing one takes a reference, so an owned value is borrowed.
1366        //
1367        // Deciding this from the fold's RESULT rather than from its start is
1368        // what lets a consuming form sit behind ordinary accessors: the chain
1369        // in front borrows, the form itself still moves.
1370        let expr = compose_value_form_call(qualify, &h.prefix[last], e, e_owned, h.consuming);
1371        out.stmts.extend(quote! { let #local = #expr; });
1372        out.bound.push((h.prefix.clone(), local));
1373        out.consuming.push(h.consuming);
1374        out.optional.push(false);
1375    }
1376    out
1377}
