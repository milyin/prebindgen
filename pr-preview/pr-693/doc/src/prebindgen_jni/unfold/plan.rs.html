1//! The plan that carries the registry's leaves under one of JniGen's two
2//! deliveries.
3//!
4//! Not beside the leaves, because which delivery a decomposition takes follows
5//! from what the target can receive: JniGen returns one non-nullable leaf and
6//! hands everything else to a callback, since a nullable leaf crosses as a boxed
7//! `Long` and a JVM `null`. A C binding could deliver several leaves as
8//! out-parameters (#680 review).
9
10use prebindgen_registry::leaf::{DeconId, Hoist, UnfoldLeaf, UnfoldShape};
11
12/// How the decomposed value(s) are delivered to the foreign side. Derived
13/// from the resolved leaf count (1 ⇒ `Return`, N ⇒ `Callback`); errors are
14/// always `Callback`-shaped.
15#[derive(Clone, Copy, PartialEq, Eq, Debug)]
16pub enum Delivery {
17    /// Deliver the leaves to a foreign **callback** (builder / fold). Any
18    /// leaf count.
19    Callback,
20    /// **Return**/deliver the single decomposed value (no builder). Requires
21    /// exactly one leaf and a non-`Iterable` shape.
22    Return,
23}
24
25/// A resolved output expansion for one function.
26#[derive(Clone)]
27pub struct UnfoldPlan {
28    /// Owned core type the records decompose — the function's return after
29    /// peeling `&` / `Option` / `Vec`.
30    pub source: prebindgen_registry::flat::TypeRef,
31    /// Which deconstructor declaration produced [`Self::leaves`] — the
32    /// identity adapters key signature artifacts on. `None` only for the
33    /// whole-element `Iterable` arm (no declaration involved).
34    pub decon: Option<DeconId>,
35    /// True when the return was `&T` / `Option<&T>`: the identity leaf clones
36    /// the borrow; otherwise it moves the owned value.
37    pub by_ref: bool,
38    /// Outer shape over the core decomposition (`Decompose` for a plain
39    /// `T`/`&T` return).
40    pub shape: UnfoldShape,
41    /// Flattened output leaves, in builder-argument order. Populated for
42    /// `Decompose`/`Optional` (accessor decomposition) and for a **decomposed**
43    /// `Iterable` fold (per-element leaves — explicit-accessor or a synthesized
44    /// `data_class` [`Self::fixed_builder`]); **empty** only for a
45    /// **whole-element** `Iterable`, which delivers each element via
46    /// [`Self::element`].
47    pub leaves: Vec<UnfoldLeaf>,
48    /// For a **whole-element** `Iterable` plan: the owned/ref element type,
49    /// delivered to the fold via its own output converter + projection (not
50    /// decomposed). `None` for `Decompose`/`Optional` and for a **decomposed**
51    /// `Iterable` fold (which uses [`Self::leaves`]).
52    pub element: Option<prebindgen_registry::flat::TypeRef>,
53    /// Callback (`deconstruct_output`) vs return-value (`convert_output`)
54    /// delivery.
55    pub delivery: Delivery,
56    /// For [`Delivery::Return`]: the single leaf's `out_ty` lifted through the
57    /// shape (`Decompose` ⇒ `out_ty`, `Optional` ⇒ `Option<out_ty>`). The
58    /// wrapper returns this value through its ordinary output converter (no
59    /// callback). `None` for [`Delivery::Callback`].
60    pub convert_out_ty: Option<prebindgen_registry::flat::TypeRef>,
61    /// `true` for a synthesized by-value `data_class` decomposition (see
62    /// [`ValueDecon`](crate::unfold::ValueDecon)): the builder/folder
63    /// is a **fixed, hoisted** foreign singleton that reconstructs the concrete
64    /// class (the wrapper takes no caller `build`/`fold` param and is not
65    /// generic over `R`/`A` — it returns the concrete type). `false` for the
66    /// accessor-declared deconstructors, whose builder is caller-supplied.
67    pub fixed_builder: bool,
68    /// Value forms that must be evaluated **once** and bound to a local. Every
69    /// leaf below one reaches off that local — otherwise each field would
70    /// rebuild the whole struct, cloning all of it once per leaf.
71    ///
72    /// A list rather than a single accessor because value forms **compose**: a
73    /// field may splice a child type whose own boundary is derived from *its*
74    /// value form, and that child call is a second hoist nested under the
75    /// first. Ordered outermost-first, so a hoist can be composed from the
76    /// longest already-bound prefix of itself.
77    pub hoists: Vec<Hoist>,
78}
79
80impl UnfoldPlan {
81    /// Whether this plan is exactly one Optional layer over one scalar/product
82    /// decomposition. Iterable and future composed inner shapes answer false.
83    pub fn is_optional_base(&self) -> bool {
84        matches!(
85            &self.shape,
86            UnfoldShape::Optional((), inner) if matches!(**inner, UnfoldShape::Base)
87        )
88    }
89}
