1//! Rendering captured Rust syntax through a pipeline-owned capability.
2//!
3//! The flat crate owns the captured syntax and therefore owns this rendering
4//! protocol. A collecting pipeline owns the concrete key that implements it.
5//! `prebindgen-registry`, for example, hands its unconstructable
6//! `prebindgen_registry::Emit` only to Rust-emission callbacks.
7//!
8//! This direction preserves the crate pipeline:
9//!
10//! ```text
11//! prebindgen -> prebindgen-flat -> prebindgen-registry -> adapters
12//!   extract          parse              collect          convert
13//! ```
14//!
15//! A different collector can use `prebindgen-flat` independently and
16//! deliberately implement `RustEmitter` for its own callback key. Merely
17//! holding a flat model node does not reveal captured syntax without explicitly
18//! establishing another rendering boundary.
19//!
20//! # Direct syntax doors remain closed
21//!
22//! ```compile_fail
23//! # use prebindgen_flat::flat;
24//! fn leak(t: &flat::TypeRef) -> proc_macro2::TokenStream { t.spell() }
25//! ```
26//!
27//! ```compile_fail
28//! # use prebindgen_flat::flat;
29//! fn leak(f: &flat::Function) -> proc_macro2::TokenStream { f.origin.spell() }
30//! ```
31//!
32//! ```compile_fail
33//! # use prebindgen_flat::{Element, flat};
34//! fn leak(e: &Element) -> syn::Item { e.as_syn() }
35//! ```
36//!
37//! ```compile_fail
38//! # use prebindgen_flat::flat;
39//! fn leak(t: &flat::TypeRef) -> &syn::Type { t.as_syn() }
40//! ```
41//!
42//! ```compile_fail
43//! # use prebindgen_flat::flat;
44//! fn leak(t: &flat::TypeRef) -> syn::Type { t.stripped_syntax() }
45//! ```
46//!
47//! ```compile_fail
48//! # use prebindgen_flat::flat;
49//! fn leak(k: &flat::TypeKind) -> syn::Type { k.to_syn() }
50//! ```
51

56/// Rendering operations supplied by a pipeline-owned callback key.
57///
58/// All methods are renderings: they answer what the source wrote, never what
59/// it means. Classification uses the flat model (`TypeRef::kind`, keys and
60/// structural readings) and does not need this protocol.
61///
62/// This trait intentionally has no provided concrete key. Implementing it is a
63/// collector's explicit decision to establish an emission boundary; adapters
64/// using that collector receive its key only where that collector chooses.
65/// The trait is object-safe so a collector wrapper can expose the full API by
66/// delegation without reproducing every method.
67///
68/// An independent collector establishes its own boundary by implementing the
69/// protocol for its own key:
70///
71/// ```
72/// use prebindgen_flat::{Flat, RustEmitter};
73///
74/// struct MyCollectorKey;
75/// impl RustEmitter for MyCollectorKey {}
76///
77/// let flat = Flat::builder().build()?;
78/// let syntax: syn::Type = syn::parse_quote!(Option<String>);
79/// let reading = flat.classify(&syntax)?;
80/// assert_eq!(
81///     MyCollectorKey.spell(&reading).to_string(),
82///     "Option < String >"
83/// );
84/// # Ok::<(), Box<dyn std::error::Error>>(())
85/// ```
86pub trait RustEmitter {
87    /// Spell a type exactly as captured.
88    fn spell(&self, ty: &TypeRef) -> TokenStream {
89        ty.spell()
90    }
91
92    /// Spell a type as a `syn::Type`.
93    fn spell_ty(&self, ty: &TypeRef) -> syn::Type {
94        let tokens = ty.spell();
95        syn::parse_quote!(#tokens)
96    }
97
98    /// Spell the type below every transparent wrapper.
99    fn spell_stripped(&self, ty: &TypeRef) -> syn::Type {
100        ty.stripped_syntax()
101    }
102
103    /// Re-emit a constant as an alias to its source module.
104    fn const_alias(&self, item: &super::Constant, source_module: &syn::Path) -> syn::ItemConst {
105        let ident = &item.name;
106        let ty = self.spell(&item.ty);
107        let docs: Vec<syn::Attribute> = item
108            .docs()
109            .into_iter()
110            .flat_map(|docs| {
111                docs.lines()
112                    .map(|line| {
113                        let doc = format!(" {line}");
114                        syn::parse_quote!(#[doc = #doc])
115                    })
116                    .collect::<Vec<_>>()
117            })
118            .collect();
119        syn::parse_quote!(#(#docs)* pub const #ident: #ty = #source_module::#ident;)
120    }
121
122    /// Re-emit an anonymous feature guard.
123    fn guard(&self, guard: &super::Guard) -> syn::ItemConst {
124        guard.origin.as_syn().clone()
125    }
126
127    /// Return an enum discriminant exactly as written.
128    fn discriminant(&self, value: &EnumValue) -> Option<TokenStream> {
129        value
130            .origin
131            .as_syn()
132            .discriminant
133            .as_ref()
134            .map(|(_, expr)| quote::quote!(#expr))
135    }
136
137    /// Render a struct pattern or constructor with its captured delimiters.
138    fn shape_struct(&self, item: &Struct, head: TokenStream, parts: &[TokenStream]) -> TokenStream {
139        super::spell::fields(super::spell::Shaped::shape(item), head, parts)
140    }
141
142    /// Render an enum alternative with its captured delimiters.
143    fn shape_alternative(
144        &self,
145        item: &Alternative,
146        head: TokenStream,
147        parts: &[TokenStream],
148    ) -> TokenStream {
149        super::spell::fields(super::spell::Shaped::shape(item), head, parts)
150    }
151
152    /// Render an enum value with its captured delimiters.
153    fn shape_enum_value(
154        &self,
155        item: &EnumValue,
156        head: TokenStream,
157        parts: &[TokenStream],
158    ) -> TokenStream {
159        super::spell::fields(super::spell::Shaped::shape(item), head, parts)
160    }
161}
