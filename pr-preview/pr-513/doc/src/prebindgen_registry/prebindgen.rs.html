1//! `Prebindgen` — what a generator still hands the emitter.
2//!
3//! One method per `#[prebindgen]` item kind (`on_function`, `on_struct`,
4//! `on_enum`, `on_const`) returning typed Rust items to emit, plus the
5//! items they depend on (`prerequisites`) and two invariant checks.
6//!
7//! **Conversion is not here.** A generator builds those itself, one per
8//! crossing, inside `RegistryBuilder::convert_with` — so there is no
9//! `on_input_type`, no deferral, and no fixed-point loop retrying until it
10//! converges.
11//!
12//! [`ConverterImpl::converter`] is the stable identity of the wire-facing
13//! converter. Its executable artifact is retained separately by the adapter's
14//! frozen generation plan; this shared registry carrier never stores rendered
15//! Rust syntax.
16

24/// One link in a converter's [stage chain](`ConverterImpl::pre_stages`) —
25/// a value-inspecting step that sits between the rust value the
26/// `#[prebindgen]` fn yields/receives and the wire-facing
27/// [`ConverterImpl::converter`].
28///
29/// Each stage is a fallible `In → Result<Out, Err>` function. The core
30/// pipeline only ever orders [`Self::converter`]; how a
31/// stage's `Err` arm is surfaced to the foreign side — throw an exception,
32/// return an error code, set `errno`, … — is entirely up to the
33/// destination-language adapter and is described by [`Self::metadata`].
34#[derive(Clone)]
35pub struct Stage<M = ()> {
36    /// Stable identity of this stage's separately retained executable
37    /// artifact.
38    pub converter: OperationId,
39    /// Adapter-specific extras for this stage — the same type as the owning
40    /// converter's ([`ConverterImpl::metadata`]). The core never
41    /// inspects this; the adapter's emitter reads it to decide how the
42    /// stage's `Err` arm is surfaced (e.g. a JNI adapter stores the JVM
43    /// exception class and `throw_*` fn to call here; a C adapter might
44    /// store the error-code sentinel). Defaults to `()`.
45    pub metadata: M,
46}

48/// Result of resolving one converter — the wire (destination) type the rest
49/// of the registry sees, plus the identity of its separately retained
50/// executable artifact.
51///
52/// `converter` is registry-owned semantic identity. Composed parents retain
53/// that identity directly; they never recompute it from a Rust type or wire
54/// type. The writer turns it into a private Rust symbol only through
55/// [`crate::RustWriter`] during final file assembly.
56#[derive(Clone)]
57pub struct ConverterImpl<M = ()> {
58    /// Wire/destination type. Other converters that ask "what's the wire
59    /// form of this rust type?" read this. The actual function may return
60    /// a wrapped form (e.g. an adapter's own `Result`-like envelope) — that
61    /// is the adapter's internal calling convention; `destination` is the
62    /// value the wire carries on success.
63    pub destination: syn::Type,
64    /// Stable identity of the **wire-facing** stage. The adapter's frozen
65    /// generation plan owns its parameter list, return type, body, and other
66    /// Rust emission policy.
67    /// For input direction this is the FIRST stage in execution order
68    /// (it takes the wire); for output direction this is the LAST stage
69    /// (it produces the wire).
70    pub converter: OperationId,
71    /// **Rust-side** stages that compose with [`Self::converter`] to form
72    /// the full conversion chain. Default empty — a 1-stage converter
73    /// is just `function`.
74    ///
75    /// Order is rust-side-first → function-side-last. Concretely:
76    /// * **Input** (wire → rust): chain runs `wire → function →
77    ///   pre_stages[0] → pre_stages[1] → … → pre_stages[N-1] → rust`.
78    /// * **Output** (rust → wire): chain runs `rust → pre_stages[N-1] →
79    ///   … → pre_stages[1] → pre_stages[0] → function → wire`.
80    ///
81    /// Each stage is fallible; how its `Err` arm is surfaced is adapter
82    /// specific and carried in [`Stage::metadata`].
83    pub pre_stages: Vec<Stage<M>>,
84    /// Bit-patterns the wire type can represent but this converter never
85    /// produces (output) and rejects (input). Wrapper handlers like
86    /// `Option<_>` consume one slot for their own discriminant and
87    /// re-export the rest — see [`Niches`] for the cascade model.
88    /// Default is empty (no niche optimisation).
89    pub niches: Niches,
90    /// Adapter-specific extras carried alongside the converter. Filled by
91    /// the same handler that produces `destination` / `function` / `niches`,
92    /// and read by the adapter's language-side emitters off the fragment this
93    /// conversion belongs to. Set this where you build the converter, not in a
94    /// side channel.
95    pub metadata: M,
96    /// Inner crossings this converter composed from — the ones the adapter
97    /// looked up to build a wrapper (`Option<X>` → `[X]`, `Result<T,E>` →
98    /// `[T, E]`, `&T` → `[&T]`). Empty for a terminal converter (scalar, opaque
99    /// handle, string) and for a callback's own converter (callback args are
100    /// cross-direction — their required-ness flows through the registry's
101    /// type-graph edges, not here).
102    ///
103    /// This is what an adapter reports back as an
104    /// [`Answer`](crate::Answer), and what `propagate_required` walks to mark
105    /// reachable crossings required.
106    ///
107    /// **Identities, not spellings.** They are looked up and walked, never
108    /// emitted. An adapter that composed the inner type keys it
109    /// (`TypeKey::from_type`); one that read it off the model asks the reading
110    /// (`TypeRef::key`), and names no escape to do it.
111    pub subs: Vec<crate::registry::TypeKey>,
112}

114/// What an emitter asks of a conversion it holds.
115impl<M> ConverterImpl<M> {
116    /// Semantic identity of the wire-facing converter function.
117    pub fn converter_id(&self) -> &OperationId {
118        &self.converter
119    }
120
121    /// Wire type this conversion carries on success.
122    pub fn wire_type(&self) -> &syn::Type {
123        &self.destination
124    }
125
126    /// Rust-side stages in input execution order, after the wire-facing
127    /// converter has decoded the wire value.
128    pub fn input_stage_order(&self) -> impl Iterator<Item = (usize, &Stage<M>)> {
129        self.pre_stages.iter().enumerate().rev()
130    }
131
132    /// Rust-side stages in output execution order, before the wire-facing
133    /// converter encodes the final wire value.
134    pub fn output_stage_order(&self) -> impl Iterator<Item = (usize, &Stage<M>)> {
135        self.pre_stages.iter().enumerate()
136    }
137}

139/// The single extension point of the pipeline: implement this trait once per
140/// **destination language** (C/cbindgen, JNI/Kotlin, Swift, Python, …) to teach
141/// the language-agnostic [`Registry`] how that language represents Rust types
142/// on the wire and what wrapper code to emit.
143///
144/// The trait has no language-specific concepts of its own, and — since the
145/// registry stopped asking it questions — one job left: **per-item emission**.
146/// The file emitter calls `on_function` / `on_struct` / `on_enum` / `on_const`
147/// to produce the per-item wrapper code, plus `prerequisites` and the two
148/// `validate` hooks for adapter invariants.
149///
150/// What used to be here and is not any more: which items to build, how
151/// composites decompose, and the wire form of each type. A generator states the
152/// first two into the builder (`RegistryBuilder::export`,
153/// `RegistryBuilder::decompose`) and answers the third inside
154/// `RegistryBuilder::convert_with` — so nothing in core calls back to ask. Moving emission out too is what would delete this
155/// trait entirely (prebindgen#251 phase E).
156///
157/// Anything language-specific the rest of the pipeline must carry — a JNI
158/// adapter's Kotlin class names and exception info, a C adapter's header names,
159/// etc. — rides in [`ConverterImpl::metadata`], and is read back by the
160/// adapter's own emitter off the fragment that conversion belongs to. The
161/// registry neither stores it nor names its type.
162///
163/// # The rule an adapter must obey
164///
165/// "Analyze the Flat model; generate Rust only" tells an adapter where every
166/// decision comes from. Final source-type output is an inert fragment produced
167/// by [`RustWriter`](crate::RustWriter), never a typed view of retained syntax. It is silent
168/// on the question adapters actually face — what the **destination language**
169/// ends up seeing.
170/// That one has its own answer:
171///
172/// > **Same `kind` ⇒ same destination-language type.** The *wire* is the
173/// > generator's to choose, and may differ per spelling.
174///
175/// The weaker-sounding half is the important one. It is tempting to write "same
176/// `kind` ⇒ same wire", and that is **false** — prebindgen's own adapters
177/// violate it deliberately:
178///
179/// | Rust | `kind` | Kotlin type | wire |
180/// |---|---|---|---|
181/// | `&[Payload]` | `Ref(Slice)` | `List<Payload>` | `Long` — a handle to a Rust-side `Vec` |
182/// | `Vec<Box<Payload>>` | `Vec(Boxed)` | `List<Payload>` | `JObject` — a Java `List<Payload>` |
183///
184/// Two wires, one surface. Choosing a wire is exactly the generator's job, and
185/// the destination-language wrapper absorbs the difference; a caller cannot
186/// tell. What a caller *can* tell — and what
187/// [`unwrapped`](crate::flat::TypeRef::unwrapped) exists to prevent — is the
188/// **type** changing because the source spelled a `Box`.
189///
190/// The rule scopes to **converted** positions: those where a converter stands
191/// between the Rust value and the destination and is therefore free to bridge.
192/// It cannot apply to a **layout mirror**, where the destination type is
193/// reinterpreted from the source struct's bytes and is a *layout* fact rather
194/// than a surface choice — there `Box<T>` (a pointer) genuinely is a different
195/// destination type from `T` (inline), the spelling is load-bearing by
196/// construction, and no erasure can apply. The C adapter's `repr_c_struct` is
197/// the one such position in-tree, and its own documentation carries that half.
198///
199/// Reusing a mirror's spelling test in a converted position is how the rule
200/// gets broken (prebindgen#230, #292).
201pub trait Prebindgen {
202    /// Rust items the adapter's emitted converters depend on (helper
203    /// structs, type aliases, runtime-support code). Emitted at the top
204    /// of the destination file, before all auto-generated converters.
205    ///
206    /// Default: none. Wrapper adapters that compose a base adapter should
207    /// forward to or extend the base's `prerequisites()`. The resolved
208    /// `registry` is supplied so prerequisites can be gated on what the
209    /// (feature-aware) scan actually contains — e.g. emitting a
210    /// per-opaque-handle item only for handles a scanned `#[prebindgen]`
211    /// fn references.
212    fn prerequisites(&self, _registry: &Registry, _emit: &crate::RustWriter) -> Vec<syn::Item> {
213        Vec::new()
214    }
215
216    // ── Declaration queries ────────────────────────────────────────
217
218    /// Adapter-invariant checks that need registry **signatures** — the
219    /// earliest they can run (decl objects are built before any source is
220    /// read). Called by `RegistryBuilder::validate_with` right after the declaration
221    /// scan (so a missing fn has already hard-errored; validate sees only
222    /// indexed items) and before plan application. An `Err` aborts the
223    /// resolve as `ScanError::AdapterInvariant` with the message verbatim
224    /// — e.g. jnigen rejects a `.fun()` member whose target has no
225    /// receiver parameter of the class type.
226    ///
227    /// Default: no checks.
228    fn validate(&self, _binding: &crate::registry::Building<'_>) -> Result<(), String> {
229        Ok(())
230    }
231
232    /// Post-**resolve** validation boundary — the counterpart of
233    /// [`Self::validate`] that sees the fully resolved registry (converters,
234    /// plans, metadata). Every artifact writer calls it before writing
235    /// anything, so an invalid binding fails cleanly — with every problem
236    /// reported at once — instead of panicking midway after a sibling
237    /// artifact already reached disk. Deterministic over `(self, registry)`;
238    /// it runs once per write call, which keeps artifact writes
239    /// order-independent.
240    ///
241    /// Default: no checks.
242    fn validate_resolved(&self, _registry: &Registry) -> Result<(), String> {
243        Ok(())
244    }
245
246    /// Absolute path under which the source crate's items are reachable
247    /// from the generated file (e.g. `zenoh_flat`), for adapters that
248    /// qualify emitted references against one. An adapter may also use it in
249    /// its required [`Self::on_const`] policy to generate a path-alias to the
250    /// source item instead of copying initializer tokens. Default: `None`.
251    fn source_module(&self) -> Option<&syn::Path> {
252        None
253    }
254
255    // ── Item methods ───────────────────────────────────────────────
256    //
257    // Each takes the **element**, not the `syn` item it was parsed from.
258    //
259    // The element is the model's own node: its types are `TypeRef`s, already
260    // classified. An adapter handed one therefore cannot ask what a type means
261    // and be told "no reading" — the question a `&syn::ItemFn` forced it to ask
262    // the registry, and which answered wrongly for a type that never entered
263    // the pipeline (#275). Final Rust source types are generated by `RustWriter` from
264    // those model facts; retained type syntax is unavailable here.
265
266    /// Wrap a `#[prebindgen]` fn into the destination-language wrapper
267    /// (e.g. JNI `extern "C"` fn).
268    fn on_function(
269        &self,
270        f: &prebindgen_flat::flat::Function,
271        registry: &Registry,
272        emit: &crate::RustWriter,
273    ) -> Vec<syn::Item>;
274
275    /// Per-struct emission. Typically empty for languages that get
276    /// everything they need from auto-generated converters.
277    fn on_struct(
278        &self,
279        s: &prebindgen_flat::flat::Struct,
280        registry: &Registry,
281        emit: &crate::RustWriter,
282    ) -> Vec<syn::Item>;
283
284    /// Per-sum emission — an `enum` whose alternatives carry payloads.
285    ///
286    /// Separate from [`Self::on_enum`] because the model separates them: the
287    /// two are numbered differently and consumed as different constructs. An
288    /// adapter with nothing to say about one shape returns an empty vector, as
289    /// both in-tree adapters do for both.
290    fn on_variant(
291        &self,
292        v: &prebindgen_flat::flat::Variant,
293        registry: &Registry,
294        emit: &crate::RustWriter,
295    ) -> Vec<syn::Item>;
296
297    /// Per-enum emission — the fieldless shape, a named set of integers.
298    fn on_enum(
299        &self,
300        e: &prebindgen_flat::flat::Enum,
301        registry: &Registry,
302        emit: &crate::RustWriter,
303    ) -> Vec<syn::Item>;
304
305    /// Per-const emission. There is deliberately no default: Flat models the
306    /// const's name, type and documentation, but not an arbitrary Rust
307    /// initializer expression. Each adapter must therefore state its value
308    /// policy explicitly, commonly by generating a path-alias to a source
309    /// module. Initializer tokens are never copied or parsed.
310    ///
311    /// A const reaching here is always named: prebindgen's own injected feature
312    /// checks are [`Guard`](prebindgen_flat::flat::Guard)s, not consts, so this
313    /// never has to recognise one.
314    fn on_const(
315        &self,
316        c: &prebindgen_flat::flat::Constant,
317        registry: &Registry,
318        emit: &crate::RustWriter,
319    ) -> Vec<syn::Item>;
