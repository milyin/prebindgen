10/// Emit the output-expansion delivery body (output phase) for a function
11/// marked `.expand_output()`. The return value (`__out`) is decomposed by the
12/// plan's accessor leaves, each encoded into a JVM `Object`, and all delivered
13/// to the foreign builder lambda (`__builder`) in a single `invoke` call whose
14/// `JObject` result becomes the wrapper's return.
15///
16/// **Borrow ordering** (the user's zero-copy rationale): the reference
17/// (non-identity) accessor leaves are encoded **first** — each leaf's converter
18/// performs the single JVM copy (`&str -> jstring`), ending its borrow into
19/// `__out` — then the identity/handle leaf is emitted **last**: an owned `T`
20/// return is **moved** into the handle (`Box::into_raw(Box::new(__out))`, no
21/// clone) once the borrows are gone; a `&T` return is **cloned** via the
22/// borrowed-opaque output converter. The builder args are assembled in declared
23/// leaf order regardless of encode order.
24///
25/// Shape handling: [`UnfoldShape::Base`] decomposes the returned value
26/// directly; [`UnfoldShape::Optional`] uses the registry's Optional-over-Product
27/// chain when available, yielding one presence wire plus the child's wires.
28/// Absence skips the builder and delivers null. Leaf wires may be object
29/// (JString/JByteArray/JObject — cast via `.into()`) or primitive (boxed to
30/// `java.lang.*` via the cached `box_helper_for_wire` runtime helpers).
31///
32/// [`UnfoldShape::Base`]: prebindgen_registry::unfold::UnfoldShape::Base
33/// [`UnfoldShape::Optional`]: prebindgen_registry::unfold::UnfoldShape::Optional
34pub(crate) fn emit_unfold_delivery(
35    plan: &prebindgen_registry::unfold::UnfoldPlan,
36    output: &crate::jni::fn_plan::UnfoldOutputPlan,
37    call_expr: &TokenStream,
38    on_err: &TokenStream,
39    emit: &prebindgen_registry::RustWriter,
40) -> TokenStream {
41    use prebindgen_registry::unfold::UnfoldShape;
42    let context = &output.delivery;
43
44    let n = plan.leaves.len();
45
46    // Builder-arg locals, one per leaf in declared order. The builder is a
47    // generated typed `<Source>Builder<out R>` fun interface — its `run`
48    // method ID is resolved once per process on the interface class
49    // ([`CachedIfaceMethod`]); primitives cross as raw typed jvalues.
50    let obj_idents: Vec<syn::Ident> = (0..n).map(|i| format_ident!("__obj{}", i)).collect();
51
52    // Return-site error path: route the message to the error sink, then return
53    // the wrapper's sentinel. Threads through the shared leaf encoder.
54    let fail = |msg: TokenStream| -> TokenStream {
55        quote! {
56            signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &#msg);
57            return #on_err;
58        }
59    };
60
61    // Encode a value's leaves (`__out`, a `Some`-bound `__inner`, or a Vec
62    // `__elem`) into `__obj0…__objN` (shared with the callback trampoline),
63    // yielding the per-leaf typed jvalue arg expressions.
64    let encode_leaves = |value: &TokenStream, optional: bool| {
65        let mut delivered = Delivered::planned(plan, output.wires.clone(), output.chain.clone());
66        delivered.optional = optional;
67        encode_plan_leaves(context, delivered, &obj_idents, value, &fail, emit)
68    };
69
70    // Cached-interface call statics for the builder / folder `run`.
71    let iface_statics = |spec: &IfaceSpec| -> TokenStream {
72        let fqn_lit = syn::LitStr::new(&spec.raw_slash_fqn(), Span::call_site());
73        let descr_lit = syn::LitStr::new(&spec.descr, Span::call_site());
74        quote! {
75            #[allow(non_upper_case_globals)]
76            static __CB_MID: ::prebindgen_jni_runtime::CachedIfaceMethod =
77                ::prebindgen_jni_runtime::CachedIfaceMethod::new();
78            const __CB_FQN: &str = #fqn_lit;
79            const __CB_DESCR: &str = #descr_lit;
80        }
81    };
82
83    // Common builder-invoke (typed `run`, `Object` return = the erased `R`).
84    // Used by `Decompose`/`Optional`; its success arm yields the result
85    // `JObject`, error arms route to the sink + return the wrapper's sentinel.
86    let builder_invoke = |arg_exprs: &[TokenStream]| -> TokenStream {
87        quote! {
88            match __CB_MID.call_object(
89                &mut env, __CB_FQN, "run", __CB_DESCR, &__builder, &[#(#arg_exprs),*],
90            ) {
91                ::core::result::Result::Ok(__o) => __o,
92                ::core::result::Result::Err(__e) => {
93                    // Clears any pending JVM exception so the sink call is safe.
94                    let _ = env.exception_describe();
95                    let __e2 = <__JniErr as ::core::convert::From<String>>::from(__e.to_string());
96                    signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &__e2.to_string());
97                    #on_err
98                }
99            }
100        }
101    };
102
103    // Decompose a value into leaves then invoke the builder once (`Decompose`/
104    // `Optional`).
105    let emit_decompose = |value: &TokenStream| -> TokenStream {
106        let (leaves, arg_exprs, _) = encode_leaves(value, false);
107        let invoke = builder_invoke(&arg_exprs);
108        quote! { #leaves #invoke }
109    };
110
111    // Iterable (fold) delivery — possibly wrapped in ONE `Optional` layer.
112    // `Vec<T>` folds the elements through the typed `<Element>Folder<A>.run(acc,
113    // …)`, threading `__acc` and returning the final accumulator; per element the
114    // fold args are either the element WHOLE (M4) or its decomposed leaves (M5),
115    // with `acc` the erased `A` (`Object`). `Option<Vec<T>>` additionally yields a
116    // null result for `None` (the fold is skipped).
117    let opt_iterable = match &plan.shape {
118        UnfoldShape::Iterable(_) => Some(false),
119        UnfoldShape::Optional((), inner) if matches!(**inner, UnfoldShape::Iterable(_)) => {
120            Some(true)
121        }
122        _ => None,
123    };
124    if let Some(optional) = opt_iterable {
125        let statics = iface_statics(
126            output
127                .iface
128                .as_deref()
129                .expect("folder interface spec derivable for a resolved plan"),
130        );
131        let fold_invoke = |arg_exprs: &[TokenStream]| -> TokenStream {
132            quote! {
133                __acc = match __CB_MID.call_object(
134                    &mut env, __CB_FQN, "run", __CB_DESCR, &__fold,
135                    &[jni::sys::jvalue { l: __acc.as_raw() }, #(#arg_exprs),*],
136                ) {
137                    ::core::result::Result::Ok(__o) => __o,
138                    ::core::result::Result::Err(__e) => {
139                        let _ = env.exception_describe();
140                        let __e2 = <__JniErr as ::core::convert::From<String>>::from(__e.to_string());
141                        signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &__e2.to_string());
142                        return #on_err;
143                    }
144                };
145            }
146        };
147
148        let loop_body = if plan.element.is_some() {
149            // Whole-element (M4): encode the element via its own converter —
150            // a raw typed jvalue for a primitive-wire element, a JObject
151            // otherwise (mirrors `leaf_is_prim`; the folder interface
152            // declares the matching typed param).
153            let pipeline = output
154                .element_pipeline
155                .as_ref()
156                .expect("whole-element fold carries its frozen output pipeline");
157            let elem_call = pipeline.invoke(quote!(__elem), emit);
158            let elem_conv = quote! {
159                match #elem_call {
160                    ::core::result::Result::Ok(__w) => __w,
161                    ::core::result::Result::Err(__e) => {
162                        signal_binding_error(&mut env, &__error_sink, &__SINK_MID, __SINK_FQN, __SINK_DESCR, &__e.to_string());
163                        return #on_err;
164                    }
165                }
166            };
167            let elem_wire = pipeline.wire().clone();
168            // Primitive-wire elements (including an opaque **handle**, whose wire
169            // is `jlong`) cross as a raw typed jvalue; object wires (String,
170            // arrays) cross as a `JObject`. Keyed purely on the wire shape —
171            // a handle's `Some(Handle)` projection still rides its `jlong`, and
172            // the folder interface declares the matching `Long` (raw) param.
173            let elem_is_prim = matches!(jni_field_access(&elem_wire), Some((_, _, false)));
174            let enc = format_ident!("__enc");
175            let (bind_obj, arg_expr) = if elem_is_prim {
176                let letter = jni_field_access(&elem_wire).unwrap().1;
177                (
178                    TokenStream::new(),
179                    quote!(jni::sys::jvalue { #letter: __enc }),
180                )
181            } else {
182                let cast = cast_wire_to_jobject(&enc, &elem_wire, &fail);
183                (
184                    quote! { let __obj: jni::objects::JObject = #cast; },
185                    quote!(jni::sys::jvalue { l: __obj.as_raw() }),
186                )
187            };
188            let invoke = fold_invoke(&[arg_expr]);
189            quote! {
190                let __enc = #elem_conv;
191                #bind_obj
192                #invoke
193            }
194        } else {
195            // Decomposed (M5): encode each element's leaves, fold over them.
196            let (leaves, arg_exprs, _) = encode_leaves(&quote!(__elem), false);
197            let invoke = fold_invoke(&arg_exprs);
198            quote! {
199                #leaves
200                #invoke
201            }
202        };
203        // Fold the elements of `__vec` into `__acc` (`into_iter()` yields the
204        // element type exactly as written — owned `T`, or `&T` for a borrow).
205        let fold = quote! {
206            let mut __acc = __acc;
207            for __elem in __vec.into_iter() {
208                #loop_body
209            }
210            __acc
211        };
212        // `Option<Vec<T>>`: `None` ⇒ null result; `Some(vec)` ⇒ fold. A bare
213        // `Vec<T>` folds the returned value directly.
214        return if optional {
215            quote! {
216                #statics
217                let __out = #call_expr;
218                match __out {
219                    ::core::option::Option::Some(__vec) => { #fold }
220                    ::core::option::Option::None => #on_err,
221                }
222            }
223        } else {
224            quote! {
225                #statics
226                let __vec = #call_expr;
227                #fold
228            }
229        };
230    }
231
232    match &plan.shape {
233        UnfoldShape::Base => {
234            let statics = iface_statics(
235                output
236                    .iface
237                    .as_deref()
238                    .expect("builder interface spec derivable for a registered declaration"),
239            );
240            let body = emit_decompose(&quote!(__out));
241            quote! {
242                #statics
243                let __out = #call_expr;
244                #body
245            }
246        }
247        UnfoldShape::Optional((), inner) => {
248            match **inner {
249                UnfoldShape::Base => {}
250                _ => panic!(
251                    "emit_unfold_delivery: Optional inner must be Base (scalar) or \
252                     Iterable (`Option<Vec<T>>`, handled above)"
253                ),
254            }
255            let statics = iface_statics(
256                output
257                    .iface
258                    .as_deref()
259                    .expect("builder interface spec derivable for a registered declaration"),
260            );
261            let (leaves, arg_exprs, present) = encode_leaves(&quote!(__out), true);
262            if let Some(present) = present {
263                let invoke = builder_invoke(&arg_exprs);
264                quote! {
265                    #statics
266                    let __out = #call_expr;
267                    #leaves
268                    if #present != 0 {
269                        #invoke
270                    } else {
271                        #on_err
272                    }
273                }
274            } else {
275                // A non-composed declaration keeps the established delivery
276                // path until its shape has a registry recipe.
277                let body = emit_decompose(&quote!(__inner));
278                quote! {
279                    #statics
280                    let __out = #call_expr;
281                    match __out {
282                        ::core::option::Option::Some(__inner) => { #body }
283                        ::core::option::Option::None => #on_err,
284                    }
285                }
286            }
287        }
288        UnfoldShape::Iterable(_) => {
289            unreachable!("Iterable delivery is handled by the `opt_iterable` branch above")
290        }
291    }
292}

294/// Cast an encoded wire local to a `JObject` for the erased `invoke`: object
295/// wires pass through / `.into()`; primitive wires box to `java.lang.*`.
296/// `fail(msg)` — `msg` an expression yielding `String` — produces the
297/// diverging on-error statements (sink + sentinel at a return site, `Err` in
298/// the trampoline). Returns an expression yielding `JObject`.
299pub(crate) fn cast_wire_to_jobject(
300    enc: &syn::Ident,
301    wire: &syn::Type,
302    fail: &dyn Fn(TokenStream) -> TokenStream,
303) -> TokenStream {
304    if is_jobject_wire(wire) {
305        quote!(#enc)
306    } else if matches!(jni_field_access(wire), Some((_, _, true))) {
307        quote!(#enc.into())
308    } else if let Some(helper) = box_helper_for_wire(wire) {
309        let on_fail = fail(quote!(__e));
310        quote! {
311            match ::prebindgen_jni_runtime::#helper(&mut env, #enc) {
312                ::core::result::Result::Ok(__o) => __o,
313                ::core::result::Result::Err(__e) => {
314                    #on_fail
315                }
316            }
317        }
318    } else {
319        panic!(
320            "jnigen unfold: leaf has unsupported wire `{}`",
321            wire.to_token_stream()
322        )
323    }
324}

326/// Compose one [`PathStep`] onto the reference expression reached so far.
327/// A `Call` applies its accessor (origin-qualified); a `Field` reads the field
328/// and re-borrows, so the result is a reference either way and steps chain
329/// uniformly.
330pub(crate) fn compose_step(
331    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
332    step: &PathStep,
333    e: TokenStream,
334) -> TokenStream {
335    match step {
336        PathStep::Call { ident, .. } => {
337            let m = qualify(ident);
338            quote!(#m::#ident(#e))
339        }
340        PathStep::Field { ident, .. } => quote!(&(#e).#ident),
341    }
342}

344/// Fold a run of steps onto `e`, borrowing wherever ownership demands it.
345///
346/// [`compose_step`] hands a `Call` its receiver as written, and an accessor
347/// takes that receiver **by reference** — so an owned value in hand has to be
348/// borrowed before the next call composes onto it. A value is in hand whenever
349/// the previous step returned one (`f(..) -> T` rather than `-> &T`), which is
350/// what [`PathStep::yields_owned`] records; `owned` says whether `e` itself
351/// started that way.
352///
353/// A `Field` step needs no borrow either way — it composes as `&(e).f`, which
354/// reads through a value and a reference alike.
355///
356/// This is the ONE place the rule lives, so every fold — a leaf's reach, a
357/// conditional hoist's prefix, a sum's matched value — answers it the same way.
358/// Splitting it produced exactly the bug it exists to prevent: ownership was
359/// handled at the optional binding and at the value form, and lost at every
360/// ordinary call in between.
361fn fold_steps(
362    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
363    steps: &[PathStep],
364    mut e: TokenStream,
365    mut owned: bool,
366) -> TokenStream {
367    for step in steps {
368        if owned && matches!(step, PathStep::Call { .. }) {
369            e = quote!(&#e);
370        }
371        e = compose_step(qualify, step, e);
372        owned = step.yields_owned();
373    }
374    e
375}

377/// Compose a value form's OWN call — the one step in the whole system whose
378/// receiver may be by value.
379///
380/// Four cases, from what the fold ended up holding crossed with what the
381/// accessor takes. A CONSUMING form takes its receiver by value: hand it the
382/// value when that is ours, clone when it is not — the same cost the borrowing
383/// form of the accessor would have paid, which keeps one declaration usable by
384/// both owned and `&T` roots. A borrowing form takes a reference, so an owned
385/// value is borrowed for it.
386///
387/// The decision is made from the fold's RESULT, never from where the fold
388/// began: that is what lets a consuming form sit behind ordinary accessors,
389/// where the chain in front borrows and the form itself still moves.
390///
391/// One function because both hoist paths — the conditional binding and the
392/// ordinary one — need exactly this rule, and stating it twice is what turned
393/// each new shape into a new defect.
394fn compose_value_form_call(
395    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
396    call: &PathStep,
397    e: TokenStream,
398    e_owned: bool,
399    consuming: bool,
400) -> TokenStream {
401    match (consuming, e_owned) {
402        (true, owned) => {
403            let (m, f) = (qualify(call.ident()), call.ident());
404            // Parenthesized: the clone applies to whatever the fold holds, and
405            // `&x.clone()` would parse as `&(x.clone())`.
406            let arg = if owned { e } else { quote!((#e).clone()) };
407            quote!(#m::#f(#arg))
408        }
409        (false, true) => compose_step(qualify, call, quote!(&#e)),
410        (false, false) => compose_step(qualify, call, e),
411    }
412}

414/// Start a reach from `base`, projecting the **leading run of plain field
415/// steps** directly (`&base.a.b`) instead of through a borrow of the base
416/// (`&(&base).a.b`). Returns the expression and how many steps it consumed.
417///
418/// The two forms name the same value, but the second borrows the base **as a
419/// whole**, which the borrow checker rejects once a sibling leaf has moved a
420/// different field out of it. Projecting directly makes each leaf's borrow
421/// disjoint, so a consuming value form's field moves are order-independent
422/// rather than compiling only while the borrowing leaves happen to be declared
423/// first.
424fn project_leading_fields(
425    base: &TokenStream,
426    base_is_ref: bool,
427    path: &[PathStep],
428) -> (TokenStream, usize) {
429    if base_is_ref {
430        return (base.clone(), 0);
431    }
432    let n = path.iter().take_while(|s| s.is_plain_field()).count();
433    if n == 0 {
434        return (quote!(&#base), 0);
435    }
436    let segs: Vec<&syn::Ident> = path[..n].iter().map(PathStep::ident).collect();
437    (quote!(&#base #(.#segs)*), n)
438}

440/// Fold a leaf's whole `path` over `base` with no optional-step handling, then
441/// apply the terminal treatment its [`LeafSource`] calls for: a `Field` leaf is
442/// **cloned** out of the place it reached, because its converter takes the field
443/// type as written (owned); every other leaf keeps the borrow its converter
444/// expects.
445///
446/// This is the derivation the single-leaf [`Delivery::Return`] shortcut uses
447/// (`emit/wrapper.rs`). It exists here, beside [`reach_leaf`], so the two are
448/// read and changed together: they drifted once already — the shortcut was
449/// missing the `Field` clone and handed `&F` to an `F` converter.
450///
451/// [`Delivery::Return`]: prebindgen_registry::unfold::Delivery::Return
452pub(crate) fn reach_leaf_flat(
453    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
454    leaf: &crate::jni::compile::OutWire,
455    path: &[PathStep],
456    base: TokenStream,
457    base_is_ref: bool,
458    consuming: bool,
459) -> TokenStream {
460    // An optional step BEFORE the last one needs a `match` whose `None` arm has
461    // somewhere to go. This derivation has none — it yields a plain Rust value,
462    // not a `JObject` that could be null — so the shape is refused here rather
463    // than composed into code that cannot type-check in the consumer's crate.
464    //
465    // Asked of the leaf's OWN path, not of `path`. The caller may hand a
466    // suffix: `wrapper.rs` rebases onto a hoisted local, and `Hoisted::innermost`
467    // strips the prefix that bound it — including any optional step inside it.
468    // Checking the parameter would therefore pass exactly when the hoist is the
469    // conditional one, which is the case that cannot compose (an `Option<T>`
470    // local with a field read hung off it). The full path is what the shape
471    // question is about.
472    let own_path = leaf.reach();
473    assert!(
474        !own_path.iter().rev().skip(1).any(PathStep::is_optional),
475        "jnigen unfold: leaf `{}` reaches through an optional step but is \
476         delivered as a single return value, which has no `None` arm — this \
477         shape needs callback delivery",
478        leaf.name,
479    );
480    // Whether what this leaf reaches is OURS, and so is moved rather than
481    // borrowed or cloned. The two leaf kinds say it differently:
482    //
483    // * an IDENTITY leaf carries the answer in its `out_ty` — the plan resolved
484    //   it to the owned type exactly when the value is the plan's to give away
485    //   (`place_is_owned`: an owned root, or a field of a CONSUMING value form),
486    //   and that is also what selected the owning converter, which boxes the
487    //   move rather than cloning a borrow;
488    // * a FIELD leaf's `out_ty` is the field type as written, owned either way,
489    //   so ownership is the enclosing form's: only a consuming one gives its
490    //   fields away.
491    //
492    // How to project that place is `steps_are_movable`'s question, and it is
493    // asked there rather than restated here. This used to spell it
494    // `all(is_plain_field)`, defending the restatement on the grounds that a
495    // trailing `Option` cannot reach return delivery anyway — true, and enforced
496    // in `single_return` (`core/unfold.rs`), which is precisely why a local
497    // restatement could disagree with the rule for as long as the invariant held
498    // somewhere else. `plan.rs` says two readings would drift and the
499    // disagreement would be a borrow handed to an owning converter; this is the
500    // second reading, removed.
501    let reached_is_ours = if leaf.identity {
502        !matches!(
503            leaf.out_ty.kind(),
504            prebindgen_registry::flat::TypeKind::Ref { .. }
505        )
506    } else {
507        consuming
508    };
509    if reached_is_ours && steps_are_movable(path) {
510        let segs: Vec<&syn::Ident> = path.iter().map(PathStep::ident).collect();
511        return quote!(#base #(.#segs)*);
512    }
513    let (e, lead) = project_leading_fields(&base, base_is_ref, path);
514    let e = fold_steps(qualify, &path[lead..], e, false);
515    if leaf.is_field_read() {
516        quote!((#e).clone())
517    } else {
518        e
519    }
520}

522/// One delivery site, as the encoder sees it: what it hands out, and the value
523/// forms it evaluates once on the way.
524///
525/// Not an `UnfoldPlan`. A plan carries the delivery's shape as well — whether
526/// it folds, whether it is optional, what the builder's generic is — and none
527/// of that reaches this encoder, which writes only the statements that fill the
528/// slots. Naming exactly what it reads is what lets the caller assemble one
529/// from a recipe.
530pub(crate) struct Delivered<'a> {
531    /// The values handed out, in the order the builder receives them.
532    pub(crate) wires: Vec<crate::jni::compile::OutWire>,
533    /// The value forms to evaluate once and bind to a local, outermost first.
534    ///
535    /// Per **site** rather than per value: two values reached through one
536    /// accessor share its call, and that is the whole point of a hoist.
537    pub(crate) hoists: &'a [prebindgen_registry::unfold::Hoist],
538    /// Whether the delivered value is reached through a borrow.
539    pub(crate) by_ref: bool,
540    /// True when these leaves are the model-derived data-class Product itself.
541    pub(crate) fixed_product: bool,
542    /// Whether the fixed Product is wrapped in one outer Optional shape.
543    pub(crate) optional: bool,
544
545    /// Exact composed child handed down by an enclosing registry recipe.
546    pub(crate) chain: Option<crate::jni::compile::ComposedChain>,
547}

569/// Callback delivery facts frozen while the registry is available. Rust
570/// source types remain as opaque readings inside the wires/pipelines; the only
571/// syntax retained here is origin qualification and Flat alternative shape,
572/// both consumed with the writer by the final Invoke renderer.
573#[derive(Clone)]
574pub(crate) struct FrozenDelivery {
575    wires: Vec<crate::jni::compile::OutWire>,
576    hoists: Vec<prebindgen_registry::unfold::Hoist>,
577    by_ref: bool,
578    fixed_product: bool,
579    optional: bool,
580    chain: Option<crate::jni::compile::ComposedChain>,
581    modules: std::collections::BTreeMap<String, syn::Path>,
582    sums: std::collections::BTreeMap<String, FrozenSum>,
583}
584
585impl FrozenDelivery {
586    /// Every converter encoding these leaves calls: each leaf's own pipeline,
587    /// and the composed converter when the delivery has one.
588    pub(crate) fn calls(&self, out: &mut Vec<prebindgen_registry::write::ArtifactKey>) {
589        for wire in &self.wires {
590            wire.calls(out);
591        }
592        if let Some(chain) = &self.chain {
593            out.push(prebindgen_registry::write::ArtifactKey::Operation(
594                chain.operation.clone(),
595            ));
596        }
597    }
598
599    pub(crate) fn new(
600        ext: &Declarations,
601        registry: &impl Conversions,
602        plan: &prebindgen_registry::unfold::UnfoldPlan,
603        wires: Vec<crate::jni::compile::OutWire>,
604        chain: Option<crate::jni::compile::ComposedChain>,
605    ) -> Self {
606        let mut modules = std::collections::BTreeMap::new();
607        for step in plan
608            .hoists
609            .iter()
610            .flat_map(|hoist| &hoist.prefix)
611            .chain(wires.iter().flat_map(|wire| wire.reach()))
612        {
613            if let prebindgen_registry::unfold::PathStep::Call { ident, .. } = step {
614                modules
615                    .entry(ident.to_string())
616                    .or_insert_with(|| ext.fn_module(registry, ident));
617            }
618        }
619        let mut sums = std::collections::BTreeMap::new();
620        for wire in wires.iter().filter(|wire| wire.is_tag()) {
621            let prebindgen_registry::flat::TypeKind::Named { id, .. } =
622                wire.out_ty.unwrapped().kind()
623            else {
624                panic!("jnigen sum unfold: selector type is not named")
625            };
626            let ident = id.ident().unwrap_or_else(|| {
627                panic!(
628                    "jnigen sum unfold: selector type `{}` is not an identifier",
629                    id.name
630                )
631            });
632            let Some(prebindgen_registry::flat::Type::Variant(sum)) =
633                registry.flat().declared_type(&ident)
634            else {
635                panic!("jnigen sum unfold: no indexed sum `{ident}`")
636            };
637            let module = ext.fn_module(registry, &ident);
638            sums.entry(id.name.clone()).or_insert_with(|| FrozenSum {
639                source: syn::parse_quote!(#module::#ident),
640                model: sum.clone(),
641            });
642        }
643        Self {
644            wires,
645            hoists: plan.hoists.clone(),
646            by_ref: plan.by_ref,
647            fixed_product: plan.fixed_builder,
648            optional: plan.is_optional_base(),
649            chain,
650            modules,
651            sums,
652        }
653    }
654
655    pub(crate) fn delivered(&self) -> Delivered<'_> {
656        Delivered {
657            wires: self.wires.clone(),
658            hoists: &self.hoists,
659            by_ref: self.by_ref,
660            fixed_product: self.fixed_product,
661            optional: self.optional,
662            chain: self.chain.clone(),
663        }
664    }
665
666    pub(crate) fn wire_count(&self) -> usize {
667        self.wires.len()
668    }

721/// Every value form on a plan, evaluated **once** and bound to a local
722/// (`__vf0`, `__vf1`, …), so a struct is built once per delivery rather than
723/// once per field. The bound prefixes come back with the statements, since
724/// reaching a leaf means starting from the innermost local it sits under.
725///
726/// Shared by both delivery paths — the multi-leaf encoder below and the
727/// single-leaf `Delivery::Return` shortcut in `emit/wrapper.rs`. The shortcut
728/// used to compose its reach straight off the raw value, which for a consuming
729/// value form emitted `f(&v)` against a by-value receiver: ill-typed Rust in
730/// the consumer's crate. One binder, so the two cannot disagree about what a
731/// hoist is or who owns it.
732pub(crate) struct Hoisted {
733    /// The `let __vfN = …;` bindings, outermost-first.
734    pub(crate) stmts: TokenStream,
735    /// Each hoist's path prefix and the local it was bound to.
736    bound: Vec<(Vec<PathStep>, syn::Ident)>,
737    /// Whether each bound hoist consumed the value it decomposed.
738    consuming: Vec<bool>,
739    /// Whether each bound local is `Option<TStruct>` rather than `TStruct` —
740    /// the hoist sits under an optional step, so the value form ran only where
741    /// the value was present. Its leaves cannot be emitted as independent
742    /// statements: they share ONE `match` on the local (see
743    /// [`encode_plan_leaves`]), taken by value, so a consuming form's fields
744    /// still move out inside the arm.
745    optional: Vec<bool>,
746}
747
748impl Hoisted {
749    /// Index of the innermost bound hoist `path` sits under, with that prefix
750    /// already consumed. `None` for a leaf under no value form at all — a
751    /// sibling `.field()` / `.field_self()`, which still reaches from the value
752    /// itself.
753    fn innermost(&self, path: &[PathStep]) -> Option<(usize, Vec<PathStep>)> {
754        self.bound
755            .iter()
756            .enumerate()
757            .filter(|(_, (p, _))| p.len() < path.len() && path.starts_with(p))
758            .max_by_key(|(_, (p, _))| p.len())
759            .map(|(i, (p, _))| (i, path[p.len()..].to_vec()))
760    }
761
762    /// The innermost bound local `path` sits under, with that prefix already
763    /// consumed, and whether that hoist gave its value away.
764    pub(crate) fn rebase(&self, path: &[PathStep]) -> Option<(syn::Ident, Vec<PathStep>, bool)> {
765        self.innermost(path)
766            .map(|(i, rest)| (self.bound[i].1.clone(), rest, self.consuming[i]))
767    }
768
769    /// The innermost **conditional** hoist `path` sits under: its index, the
770    /// local holding the `Option`, the name its `Some` arm binds, and the steps
771    /// left to reach the leaf from there. `None` when the leaf's innermost
772    /// hoist is unconditional (or there is none) — then [`Self::rebase`]
773    /// applies and the leaf is an ordinary standalone statement.
774    pub(crate) fn conditional(
775        &self,
776        path: &[PathStep],
777    ) -> Option<(usize, syn::Ident, syn::Ident, Vec<PathStep>)> {
778        let (i, rest) = self.innermost(path)?;
779        self.optional[i].then(|| (i, self.bound[i].1.clone(), format_ident!("__u{}", i), rest))
780    }
781
782    /// The local a hoist was bound to.
783    pub(crate) fn local(&self, i: usize) -> syn::Ident {
784        self.bound[i].1.clone()
785    }
786
787    /// Whether a hoist consumed the value it decomposed.
788    pub(crate) fn consumed(&self, i: usize) -> bool {
789        self.consuming[i]
790    }
791}

793/// Fold `path` over `base` the way [`reach_leaf`] does, but yielding an
794/// `Option<…>` rather than a `JObject`: the optional steps become a
795/// `map`/`and_then` chain, so an absent value short-circuits to `None` instead
796/// of to a null object. `body` renders the innermost reached expression as a
797/// BARE value — the chain's last link wraps it.
798///
799/// This is how a CONDITIONAL value form is bound — the accessor runs only where
800/// the value it decomposes is actually present.
801fn reach_optional(
802    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
803    path: &[PathStep],
804    base: TokenStream,
805    base_is_ref: bool,
806    depth: usize,
807    body: &dyn Fn(TokenStream) -> TokenStream,
808) -> TokenStream {
809    let (e, lead) = project_leading_fields(&base, base_is_ref, path);
810    match (lead..path.len()).find(|&i| path[i].is_optional()) {
811        None => body(fold_steps(qualify, &path[lead..], e, false)),
812        Some(k) => {
813            // Through the optional step INCLUSIVE: the same fold, so the
814            // borrow in front of it is the ordinary rule rather than a second
815            // statement of it.
816            let opt_e = fold_steps(qualify, &path[lead..=k], e, false);
817            let bind = format_ident!("__hb{}", depth);
818            // What the arm binds is the step's own value: an OWNED payload is a
819            // bare `T`, so composing the next step onto it directly would hand
820            // `T` to an accessor typed for `&T`. Say it is not a reference and
821            // let `project_leading_fields` borrow it; a borrowed payload is
822            // already one and passes through.
823            //
824            // With NO steps left the binding goes to `body` untouched — that is
825            // what lets a consuming value form MOVE an owned payload rather than
826            // borrow it straight back, so the terminal case stays "already a
827            // reference" whatever the payload is.
828            let rest = &path[k + 1..];
829            let inner = reach_optional(
830                qualify,
831                rest,
832                quote!(#bind),
833                rest.is_empty() || !path[k].yields_owned(),
834                depth + 1,
835                body,
836            );
837            // `map` when this is the LAST optional step (the body yields a bare
838            // value) and `and_then` when another follows (the recursion yields
839            // an `Option` that must not nest). The equivalent `match` reads the
840            // same but generated code runs through the consumer's lints, where
841            // `clippy::manual_map` is a denial.
842            let combinator = if rest.iter().any(PathStep::is_optional) {
843                format_ident!("and_then")
844            } else {
845                format_ident!("map")
846            };
847            quote! {
848                #opt_e.#combinator(|#bind| #inner)
849            }
850        }
851    }
852}

854pub(crate) fn bind_hoists(
855    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
856    hoists: &[prebindgen_registry::unfold::Hoist],
857    value: &TokenStream,
858    by_ref: bool,
859) -> Hoisted {
860    let mut out = Hoisted {
861        stmts: TokenStream::new(),
862        bound: Vec::new(),
863        consuming: Vec::new(),
864        optional: Vec::new(),
865    };
866    // Value forms COMPOSE, so each hoist is built from the longest hoist that
867    // is already a proper prefix of it (they arrive outermost-first), and from
868    // `value` otherwise.
869    for (i, h) in hoists.iter().enumerate() {
870        let local = format_ident!("__vf{}", i);
871        // A hoist under an optional step binds `Option<TStruct>`: the value
872        // form runs in the `Some` arm only. Core refuses to nest these, so the
873        // enclosing value is always the plan's own — no rebase to consider.
874        if h.prefix.iter().any(PathStep::is_optional) {
875            let (last, lead) = h
876                .prefix
877                .split_last()
878                .expect("a hoist prefix ends in its value-form call");
879            let consuming = h.consuming;
880            // The value form is handed the payload only when the optional step
881            // is the LAST thing before it; any step in between composes as a
882            // borrow, so what arrives is a reference either way.
883            let owned = lead.last().is_some_and(PathStep::yields_owned);
884            let expr = reach_optional(qualify, lead, value.clone(), by_ref, 0, &|reached| {
885                compose_value_form_call(qualify, last, reached, owned, consuming)
886            });
887            out.stmts.extend(quote! { let #local = #expr; });
888            out.bound.push((h.prefix.clone(), local));
889            out.consuming.push(h.consuming);
890            out.optional.push(true);
891            continue;
892        }
893        // Where the fold starts, and whether what it starts from is OWNED. The
894        // value form's own boundary is decided below, from what the fold ends
895        // up holding — never from where it began.
896        let (from, start, start_owned) = match out.rebase(&h.prefix) {
897            // A NESTED consuming form is handed the parent's field by MOVE: a
898            // hoisted value form is an owned struct and its fields are
899            // disjoint, so moving one out leaves every sibling leaf readable.
900            // `compose_step` borrows (`&(e).f`), so a plain field run to that
901            // field is projected here instead of going through it.
902            Some((outer, rest, _))
903                if h.consuming && rest[..rest.len() - 1].iter().all(PathStep::is_plain_field) =>
904            {
905                let lead = &rest[..rest.len() - 1];
906                let segs: Vec<&syn::Ident> = lead.iter().map(PathStep::ident).collect();
907                (h.prefix.len() - 1, quote!(#outer #(.#segs)*), true)
908            }
909            // Any other rebased hoist: project its own leading field run
910            // DIRECTLY off the parent local rather than reaching it through a
911            // borrow of the parent. A sibling hoist may already have moved a
912            // different field out — that is what a consuming value form does —
913            // and `&(&__vf0).wrapper` borrows the partially moved parent as a
914            // whole where `&__vf0.wrapper` is a disjoint borrow that survives.
915            // Same invariant `project_leading_fields` states for leaf reaches,
916            // and the same reason.
917            Some((outer, rest, _)) => {
918                let (e, lead) = project_leading_fields(&quote!(#outer), false, &rest);
919                (h.prefix.len() - rest.len() + lead, e, false)
920            }
921            None if by_ref => (0, value.clone(), false),
922            None => (0, value.clone(), true),
923        };
924        // Everything before the value form is an ordinary accessor chain.
925        let last = h.prefix.len() - 1;
926        let head = &h.prefix[from..last];
927        let e = fold_steps(qualify, head, start, start_owned);
928        let e_owned = head.last().map_or(start_owned, PathStep::yields_owned);
929        // The value form itself. A CONSUMING one takes its receiver BY VALUE —
930        // that is the move the whole declaration exists for — so it is handed
931        // what the fold holds when that is ours, and a clone when it is not:
932        // the same cost the borrowing form of the accessor would have paid,
933        // which keeps one declaration usable by both owned and `&T` returns.
934        // A borrowing one takes a reference, so an owned value is borrowed.
935        //
936        // Deciding this from the fold's RESULT rather than from its start is
937        // what lets a consuming form sit behind ordinary accessors: the chain
938        // in front borrows, the form itself still moves.
939        let expr = compose_value_form_call(qualify, &h.prefix[last], e, e_owned, h.consuming);
940        out.stmts.extend(quote! { let #local = #expr; });
941        out.bound.push((h.prefix.clone(), local));
942        out.consuming.push(h.consuming);
943        out.optional.push(false);
944    }
945    out
946}

948/// Bind `e` so it can be destructured as an `Option` **whatever Rust
949/// representation the source used for it**.
950///
951/// `kind` says a position is optional; it deliberately does not say whether
952/// Rust spells that `Option<T>`, `Box<Option<T>>`, or something else — the flat
953/// model states the destination-language invariant, and the side interpreting
954/// it is the side that must accept any representation. Matching the reached
955/// place directly assumed one, which is `classify off kind, spell with spell()`
956/// broken in the direction nothing was watching: the classification was right
957/// and the *spelling* came from it too. `Box<Option<T>>` then produced
958/// `match &place { Some(..) => .. }` and `E0308` (#268).
959///
960/// A type-ascribed `let` is a coercion site, and deref coercion is transitive
961/// **and** a no-op when the types already match — so this one shape serves
962/// every representation, and the plain `Option<T>` case is unchanged in
963/// behaviour. The payload stays `_`: what it is, is the source's business.
964///
965/// `e` is expected to be a **reference** already — [`compose_step`] composes a
966/// field read as `&(e).f` — so nothing is borrowed here. Borrowing only: an
967/// owned position cannot be made representation-agnostic this way, because
968/// deref coercion applies to references and moving out of a wrapper is
969/// something only some of them permit (`Box` does, `Rc` cannot). A site that
970/// must MOVE the payload keeps its direct match; see `owned_place` below.
971pub(crate) fn bind_as_option(e: &TokenStream, bind: &syn::Ident) -> TokenStream {
972    quote! { let #bind: &::core::option::Option<_> = #e; }
973}

975/// Reach a leaf's input by folding its `path` over `base`, then hand the
976/// reached expression to `body` (which renders the encode and yields
977/// `JObject`). Every optional nesting step becomes a `match`: its `None` arm
978/// short-circuits the whole leaf to `JObject::null()` (the value is absent ⇒
979/// the leaf is null) — any number of optional steps on the path nest.
980/// With `unwrap_last == false` the final path element composes directly — a
981/// non-identity leaf's converter takes the final step's **full** type
982/// (`Option` included), so only the steps *before* it are nesting. An
983/// identity leaf (`unwrap_last == true`) delivers the reached value itself, so
984/// a final `Option` step unwraps too.
985fn reach_leaf(
986    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
987    path: &[PathStep],
988    base: TokenStream,
989    base_is_ref: bool,
990    unwrap_last: bool,
991    depth: usize,
992    body: &dyn Fn(TokenStream) -> TokenStream,
993) -> TokenStream {
994    let limit = if unwrap_last {
995        path.len()
996    } else {
997        path.len().saturating_sub(1)
998    };
999    let (e, lead) = project_leading_fields(&base, base_is_ref, path);
1000    match (lead..limit).find(|&i| path[i].is_optional()) {
1001        // No (more) optional nesting steps: compose the rest plainly.
1002        None => body(fold_steps(qualify, &path[lead..], e, false)),
1003        Some(k) => {
1004            // Through the optional step INCLUSIVE: the same fold, so the
1005            // borrow in front of it is the ordinary rule rather than a second
1006            // statement of it.
1007            let opt_e = fold_steps(qualify, &path[lead..=k], e, false);
1008            let nested = format_ident!("__n{}", depth);
1009            let inner = reach_leaf(
1010                qualify,
1011                &path[k + 1..],
1012                quote!(#nested),
1013                true,
1014                unwrap_last,
1015                depth + 1,
1016                body,
1017            );
1018            // A FIELD read composes to a borrow (`&(e).f`), so it goes through
1019            // a coercion site and the destructuring stops caring which
1020            // representation the source spelled the optional as.
1021            //
1022            // A CALL composes to the accessor's own returned value, which is
1023            // owned and whose payload downstream may move. Borrowing it to
1024            // coerce would change that ownership, so it keeps its direct match
1025            // — and an owned position could not be made representation-agnostic
1026            // this way regardless (see [`bind_as_option`]).
1027            if path[k].is_field() {
1028                let opt_bind = format_ident!("__o{}", depth);
1029                let coerce = bind_as_option(&opt_e, &opt_bind);
1030                quote! {
1031                    {
1032                        #coerce
1033                        match #opt_bind {
1034                            ::core::option::Option::Some(#nested) => { #inner }
1035                            ::core::option::Option::None => jni::objects::JObject::null(),
1036                        }
1037                    }
1038                }
1039            } else {
1040                quote! {
1041                    match #opt_e {
1042                        ::core::option::Option::Some(#nested) => { #inner }
1043                        ::core::option::Option::None => jni::objects::JObject::null(),
1044                    }
1045                }
1046            }
1047        }
1048    }
1049}

1051/// Encode a plan's leaves off `value` (`__out`, a `Some`-bound `__inner`, a Vec
1052/// `__elem`, an owned callback arg, or a domain error `__de`) into the
1053/// `obj_idents` locals, in declared-leaf order. Reference (non-identity)
1054/// leaves are encoded first — ending their borrow into the value — and the
1055/// identity leaf last (move owned / clone `&T`). Each leaf's value is reached
1056/// by folding its accessor `path` over `value`; every `Option`-returning
1057/// nesting step on the path wraps the rest in a `match Some/None` (`None` ⇒ a
1058/// null leaf) — see [`reach_leaf`]. Error arms are produced by `fail` (see
1059/// [`cast_wire_to_jobject`]). Shared by the return-delivery site
1060/// ([`emit_unfold_delivery`]), the callback trampoline, and the domain-error
1061/// arm of fallible externs (whose `fail` routes an encoding failure to the
1062/// binding-error channel).
1063pub(crate) fn encode_plan_leaves(
1064    context: &impl DeliveryContext,
1065    site: Delivered<'_>,
1066    obj_idents: &[syn::Ident],
1067    value: &TokenStream,
1068    fail: &dyn Fn(TokenStream) -> TokenStream,
1069    emit: &prebindgen_registry::RustWriter,
1070) -> (TokenStream, Vec<TokenStream>, Option<syn::Ident>) {
1071    let Delivered {
1072        wires,
1073        hoists,
1074        by_ref,
1075        chain,
1076        fixed_product,
1077        optional,
1078    } = site;
1079    // Per-fn origin qualification: each accessor call is prefixed with the
1080    // module of the crate that defines it (multi-source bindings).
1081    let qualify = |id: &syn::Ident| -> syn::Path { context.qualify(id) };
1082    let n = wires.len();
1083
1084    // Typed `jvalue` argument expression per leaf, in leaf order: a non-null
1085    // primitive-wire leaf passes its raw primitive (`__objN` IS the jvalue);
1086    // every other leaf is a `JObject` local whose raw pointer rides the `l`
1087    // slot. Matches the descriptor [`crate::jni::iface`]
1088    // derives for the same leaf (primitive chunk vs object chunk).
1089    let mut arg_exprs: Vec<TokenStream> = Vec::with_capacity(n);
1090    for (idx, leaf) in wires.iter().enumerate() {
1091        let obj_ident = &obj_idents[idx];
1092        if context.leaf_is_prim(leaf) {
1093            arg_exprs.push(quote!(#obj_ident));
1094        } else {
1095            arg_exprs.push(quote!(jni::sys::jvalue { l: #obj_ident.as_raw() }));
1096        }
1097    }
1098
1099    // A fixed decomposition has one Product, Optional or Choice intermediate.
1100    // Invoke its converter once, then adapt the intermediate leaves to the JNI
1101    // call ABI. Callback and ordinary output delivery therefore share the
1102    // same Rust-value walk; only the final delivery remains JNI-specific.
1103    if fixed_product && hoists.is_empty() {
1104        let chain = chain;
1105        if let Some(chain) = chain.filter(|chain| match &chain.layout {
1106            crate::jni::compile::JLayout::Optional(inner) => optional && inner.leaf_count() == n,
1107            layout => !optional && layout.leaf_count() == n,
1108        }) {
1109            chain.activate();

1135                        let #object = jni::sys::jvalue { i: #encoded };

1142                        .expect("a primitive Product leaf has a JNI jvalue member");

1144                        let #object = jni::sys::jvalue { #member: #encoded };

1157    let mut stmts = hoisted.stmts.clone();

1159    // Reach a leaf off the innermost value form it sits under, with that
1160    // prefix's steps already consumed, and say whether that form CONSUMED its
1161    // value — so the leaf owns its field and may move it out rather than clone
1162    // it. A leaf under no value form at all (a sibling `.field()` /
1163    // `.field_self()`) still reaches from the value itself.
1164    // A leaf under a CONDITIONAL value form reaches off the name that form's
1165    // `Some` arm binds — a borrow of the struct, so nothing moves out of it —
1166    // and its statements are collected into that arm rather than emitted here.
1167    let rebase = |leaf: &crate::jni::compile::OutWire| -> (TokenStream, bool, Vec<PathStep>, bool) {
1168        let path = leaf.reach();
1169        if let Some((i, _, bind, rest)) = hoisted.conditional(path) {
1170            return (quote!(#bind), false, rest, hoisted.consumed(i));
1171        }
1172        match hoisted.rebase(path) {
1173            Some((local, rest, consuming)) => (quote!(#local), false, rest, consuming),
1174            None => (value.clone(), by_ref, path.to_vec(), false),
1175        }
1176    };
1177
1178    // A decomposed **sum** is the one shape whose leaves are not independent:
1179    // only one group is live per value, so its whole segment — the selector
1180    // leaf plus the group leaves that follow it — is emitted as ONE `match`
1181    // instead of per-leaf expressions. A plan may carry several: a sum that IS
1182    // the returned value is the degenerate case of one segment covering
1183    // everything, while a value form contributes one per sum-typed field.
1184    let sum_segments: Vec<std::ops::Range<usize>> = (0..n)
1185        .filter(|&i| wires[i].is_tag())
1186        .map(|start| {
1187            let end = (start + 1..n)
1188                .take_while(|&i| wires[i].group.is_some())
1189                .last()
1190                .map_or(start + 1, |i| i + 1);
1191            start..end
1192        })
1193        .collect();

1195    // Leaves under a conditional value form are collected per hoist and emitted
1196    // below as ONE `match` on its `Option` local — the same treatment a sum's
1197    // groups get, and for the same reason: their slots exist unconditionally
1198    // but only one arm computes them. Built BEFORE the sum pass, because a
1199    // conditional form may carry a sum field and that segment has to land in
1200    // the arm too: emitted ahead of it, its `match` would reach a binding the
1201    // arm has not introduced yet.
1202    let mut cond_stmts: std::collections::BTreeMap<usize, TokenStream> = hoists
1203        .iter()
1204        .enumerate()
1205        .filter_map(|(i, _)| {
1206            wires
1207                .iter()
1208                .any(|w| hoisted.conditional(w.reach()).is_some_and(|(j, ..)| j == i))
1209                .then_some((i, TokenStream::new()))
1210        })
1211        .collect();

1213    for seg in &sum_segments {

1215        let (base, base_is_ref, path, _) = rebase(leaf);
1216        // The value to `match` on. The selector's own path reaches the sum
1217        // (empty when the sum IS the value), and a step on it MAY be optional:
1218        // the refusal that used to guarantee otherwise is gone (#220), which is
1219        // what the gate below exists for.
1220        //
1221        // A plain field chain is borrowed DIRECTLY (`&base.a.b`) rather than
1222        // through the base (`&(&base).a.b`). The two are the same value, but
1223        // the second borrows the base as a whole, which the borrow checker
1224        // rejects once a sibling leaf has moved another field out of it — and
1225        // borrowing this field while sibling fields move is exactly what a
1226        // consuming value form does.

1228        // An `Option<sum>` field gates the WHOLE segment, not each slot (#220).
1229        // A sum's leaves are not independent — only one group is live per value
1230        // — so absence cannot be the per-leaf `null` `reach_leaf` gives an
1231        // ordinary optional field. It is one tuple bind whose `None` arm carries
1232        // every slot's default, which is the shape a conditional value form's
1233        // hoist already emits below; this applies it to an optional step inside
1234        // the segment's own path.

1236        let (matched, gate) = match opt_at {
1237            None => (fold_steps(&qualify, &path[lead..], projected, false), None),
1238            Some(k) => {
1239                // Through the optional step INCLUSIVE, then the rest off the
1240                // binding — the same split `reach_leaf` makes, so the borrow in
1241                // front of it stays the ordinary rule rather than a second
1242                // statement of it.
1243                let opt_e = fold_steps(&qualify, &path[lead..=k], projected, false);
1244                let bind = format_ident!("__sg{}", seg.start);
1245                // ONE optional step is what the gate below handles. A second
1246                // one in the tail would compose `match &Option<..>` against bare
1247                // variant patterns — the E0308 in the consumer's crate that the
1248                // deleted `builder.rs` assert used to pre-empt by name, so the
1249                // named diagnostic keeps a home here.
1250                //
1251                // `assert!`, not `debug_assert!`: a build script inherits the
1252                // consumer's profile, so a debug-only check is absent from
1253                // exactly the release build where a mis-emission costs the most
1254                // to diagnose. Same rule, same phrasing, as the single-return
1255                // optional-step assert in `reach_leaf` above.
1256                //
1257                // The condition is what actually breaks, not the stronger fact
1258                // that happens to hold: every sum leaf's path stops AT the sum,
1259                // so the tail is empty today, but a NON-optional tail composes
1260                // correctly through `fold_steps` — refusing it would refuse a
1261                // shape that works.
1262                assert!(
1263                    !path[k + 1..].iter().any(PathStep::is_optional),
1264                    "jnigen unfold: leaf `{}` reaches its sum through TWO optional \
1265                     steps — the segment gate has one `None` arm, so the second \
1266                     would be matched as if it were the sum itself",
1267                    leaf.name,
1268                );
1269                // What the `match` binds, asked of the step rather than assumed:
1270                // the FIELD branch scrutinizes `&Option<_>` (that is what
1271                // `bind_as_option` is for), so ergonomics binds `&Sum` — a
1272                // borrow. Only an owned-yielding CALL binds an owned value. The
1273                // literal `true` disagreed with `reach_leaf`, which passes
1274                // `false` for its analogous recursion.
1275                let inner = fold_steps(
1276                    &qualify,
1277                    &path[k + 1..],
1278                    quote!(#bind),
1279                    path[k].yields_owned(),
1280                );
1281                (inner, Some((k, opt_e, bind)))
1282            }
1283        };
1284        let (group_stmts, group_args) = encode_sum_group(
1285            context,
1286            &wires[seg.clone()],
1287            &obj_idents[seg.clone()],
1288            matched,
1289            fail,
1290            emit,
1291        );

1293            None => group_stmts,
1294            Some((k, opt_e, bind)) => {
1295                let ids: Vec<&syn::Ident> = obj_idents[seg.clone()].iter().collect();

1297                    .iter()
1298                    .map(|l| leaf_slot(context, l))
1299                    .collect();

1301                let defaults = slots.iter().map(|s| &s.default);
1302                // A FIELD step composes to a borrow, so it goes through a
1303                // coercion site and the destructure stops caring which
1304                // representation the source spelled the optional as (#268). A
1305                // CALL yields its own owned value, whose payload downstream may
1306                // move, so it keeps the direct match — the same division
1307                // `reach_leaf` makes.

1309                    let opt_bind = format_ident!("__so{}", seg.start);
1310                    (bind_as_option(&opt_e, &opt_bind), quote!(#opt_bind))
1311                } else {
1312                    (TokenStream::new(), opt_e)
1313                };

1316                        #prelude

1318                            ::core::option::Option::Some(#bind) => {
1319                                #group_stmts
1320                                (#(#ids,)*)
1321                            }

1325                }
1326            }

1332                .get_mut(&i)
1333                .expect("a conditional leaf's hoist has a bucket")
1334                .extend(group_stmts),
1335            None => stmts.extend(group_stmts),
1336        }
1337        for (k, e) in group_args.into_iter().enumerate() {
1338            arg_exprs[seg.start + k] = e;
1339        }

1353        // below lands in the right place without knowing which case it is in.
1354        let stmts: &mut TokenStream = match hoisted.conditional(leaf.reach()) {
1355            Some((i, ..)) => cond_stmts.get_mut(&i).expect("collected above"),
1356            None => &mut stmts,
1357        };

1380                    ::core::result::Result::Err(__e) => {

1382                    }
1383                }
1384            }

1415            // rather than being cloned through the borrowed converter (which

1420            // what selected the owning converter. `steps_are_movable` then says

1444                        });

1446                        (&owned_place, path.last().is_some_and(PathStep::is_optional))

1449                        // leaf hangs off a conditional value form, so the absent
1450                        // case is the enclosing `match`'s other arm, not an
1451                        // `Option` here. Move into the Box and box the jlong, so
1452                        // both arms fill the slot with the same shape.
1453                        let box_fail = fail(quote!(__e.to_string()));

1455                            obj_ident,
1456                            quote! {{
1457                                let #handle_ident: jni::sys::jlong =
1458                                    std::boxed::Box::into_raw(std::boxed::Box::new(#place))
1459                                        as jni::sys::jlong;
1460                                match ::prebindgen_jni_runtime::box_jlong(&mut env, #handle_ident) {
1461                                    ::core::result::Result::Ok(__o) => __o,

1464                                    }
1465                                }
1466                            }},
1467                        ));

1470                        // the present handle is moved into its Box, boxed
1471                        // `java.lang.Long` when present / JVM null when absent.
1472                        // Matching `&place` here is what used to clone it back
1473                        // through the borrowed converter.
1474                        let box_fail = fail(quote!(__e.to_string()));
1475                        stmts.extend(bind_obj(
1476                            obj_ident,
1477                            quote! {{
1478                                match #place {
1479                                    ::core::option::Option::Some(__n) => {
1480                                        let #handle_ident: jni::sys::jlong =
1481                                            std::boxed::Box::into_raw(std::boxed::Box::new(__n))
1482                                                as jni::sys::jlong;
1483                                        match ::prebindgen_jni_runtime::box_jlong(&mut env, #handle_ident) {
1484                                            ::core::result::Result::Ok(__o) => __o,
1485                                            ::core::result::Result::Err(__e) => {
1486                                                #box_fail
1487                                            }
1488                                        }
1489                                    }
1490                                    ::core::option::Option::None => jni::objects::JObject::null(),
1491                                }
1492                            }},
1493                        ));

1499                            &path,
1500                            value.clone(),
1501                            by_ref,

1596        // every `Option` nesting step (`None` ⇒ a null leaf). A `Field` leaf
1597        // (synthesized `data_class`) reaches a struct field and clones it
1598        // (`value.a.b.clone()`); the converter (`Option<Box<String>>` → nullable
1599        // String, …) carries any nullability, so there is no path `Option` to
1600        // unwrap. `reach(body)` dispatches on the source and feeds the reached
1601        // Rust expression to `body`.

1604        // reach in a field step, an accessor's ends at the call. That is what
1605        // `LeafSource` said, and saying it off the reach keeps one answer.

1607            match leaf.is_field_read() {

1610                // is **moved** out; the whole point of consuming is that this
1611                // clone disappears. Each field is read by exactly one leaf, so
1612                // the partial moves are disjoint — but nothing may then borrow
1613                // the local as a whole, which is why the reach below projects
1614                // the field directly rather than through `&(&local)`.

1620                    }
1621                }

1623                // accessor, so the path can be mixed: compose it like an
1624                // accessor leaf — the final step composes directly, since the
1625                // converter takes the field type as written (`Option` and all)
1626                // — and clone the reached borrow, which is what a field leaf
1627                // delivers.
1628                true => reach_leaf(
1629                    &qualify,
1630                    &path,
1631                    value.clone(),
1632                    by_ref,
1633                    false,
1634                    0,
1635                    &|reached| body(quote!((#reached).clone())),

1638        };
1639
1640        // A non-null primitive-wire leaf delivers its raw primitive as a typed
1641        // `jvalue` — no boxing, no JNI call at all (the typed `run` descriptor
1642        // declares the primitive). Everything else (object wires, and nullable
1643        // leaves whose `None` arm must yield a JVM null) encodes the reached
1644        // value with the leaf's output converter and casts to JObject.
1645        let enc_ident = format_ident!("__enc{}", idx);
1646        if context.leaf_is_prim(leaf) {
1647            let letter = jni_field_access(&wire)
1648                .expect("leaf_is_prim guarantees a primitive wire")
1649                .1;
1650            let expr = reach(&|reached| {
1651                let __encoded = conv(quote!(#reached));
1652                quote! {{
1653                    let #enc_ident = #__encoded;
1654                    jni::sys::jvalue { #letter: #enc_ident }
1655                }}
1656            });
1657            stmts.extend(quote! {
1658                let #obj_ident: jni::sys::jvalue = #expr;
1659            });
1660            continue;
1661        }
1662        let cast = cast_wire_to_jobject(&enc_ident, &wire, fail);
1663        let expr = reach(&|reached| {
1664            let __encoded = conv(quote!(#reached));
1665            quote! {{
1666                let #enc_ident = #__encoded;
1667                #cast
1668            }}
1669        });
1670        stmts.extend(bind_obj(obj_ident, expr));
1671    }
1672
1673    // One `match` per conditional value form: the `Some` arm runs the leaves
1674    // that hang off it and yields their locals as a tuple; the `None` arm
1675    // yields the same wire defaults an inert sum group carries. Binding the
1676    // tuple outside the match is what keeps the leaves' locals in scope for the
1677    // argument expressions, which are indifferent to how the slot was filled.
1678    for (i, body) in cond_stmts {
1679        let local = hoisted.local(i);
1680        let bind = format_ident!("__u{}", i);
1681        let idxs: Vec<usize> = (0..n)
1682            .filter(|&k| {
1683                hoisted
1684                    .conditional(wires[k].reach())
1685                    .is_some_and(|(j, ..)| j == i)
1686            })
1687            .collect();
1688        let ids: Vec<&syn::Ident> = idxs.iter().map(|&k| &obj_idents[k]).collect();
1689        let tys = idxs.iter().map(|&k| leaf_slot(context, &wires[k]).ty);
1690        let defaults = idxs.iter().map(|&k| leaf_slot(context, &wires[k]).default);
1691        // Matched BY VALUE: the local is this arm's alone (every leaf under the
1692        // hoist is in it), so a consuming value form's fields move out here
1693        // exactly as they do at an unconditional one.
1694        stmts.extend(quote! {
1695            let (#(#ids,)*): (#(#tys,)*) = match #local {
1696                ::core::option::Option::Some(#bind) => { #body (#(#ids,)*) }
1697                ::core::option::Option::None => (#(#defaults,)*),
1698            };
1699        });
1700    }
1701    (stmts, arg_exprs, None)
1702}

1704fn frozen_leaf_wire(leaf: &crate::jni::compile::OutWire) -> syn::Type {
1705    match &leaf.abi {
1706        Some(crate::jni::compile::OutAbi::Tag) => syn::parse_quote!(jni::sys::jint),
1707        Some(crate::jni::compile::OutAbi::Value(value)) => value.pipeline.wire().clone(),
1708        None => panic!("frozen JNI delivery leaf `{}` has no output ABI", leaf.name),
1709    }
1710}

1712/// True when a plan leaf crosses the typed `run` as a **raw primitive**
1713/// `jvalue`: non-nullable, no projection (not a handle), and a
1714/// primitive JNI wire. Must agree with the descriptor chunk
1715/// [`crate::jni::iface`] derives for the same leaf — a
1716/// nullable primitive boxes (object chunk), object wires pass as objects.
1717///
1718/// The synthesized sum selector is a `jint` by definition — it is assigned,
1719/// never converted — unless it is NULLABLE: the sum then sits under a
1720/// conditional value form, and the absent case needs a representation the
1721/// tag's own variants do not provide. A raw `jint` has none (zero is a real
1722/// variant), so the selector boxes like any other nullable leaf.
1723fn frozen_leaf_is_prim(leaf: &crate::jni::compile::OutWire) -> bool {
1724    if leaf.is_tag() {
1725        return !leaf.nullable;
1726    }
1727    if leaf.nullable {
1728        return false;
1729    }
1730    let Some(crate::jni::compile::OutAbi::Value(value)) = &leaf.abi else {
1731        panic!("frozen JNI delivery leaf `{}` has no output ABI", leaf.name)
1732    };
1733    let proj_ok = match &value.projection {
1734        None => true,
1735        Some(projection) => matches!(
1736            projection.kind,
1737            ProjectionKind::Handle | ProjectionKind::Unsigned64
1738        ),
1739    };
1740    proj_ok && matches!(jni_field_access(value.pipeline.wire()), Some((_, _, false)))
1741}
1742
1743/// The wire half of [`leaf_is_prim`]: does a leaf of this type occupy a **raw
1744/// primitive** slot? Split out so the interface derivation can ask the question
1745/// about a leaf whose own `nullable` flag it is in the middle of computing (an
1746/// inert sum group slot).
1747pub(crate) fn leaf_ty_is_prim(
1748    ext: &Declarations,
1749    out_ty: &prebindgen_registry::flat::TypeRef,
1750) -> bool {
1751    let Some(entry) = ext.out_frag(out_ty) else {
1752        return false;
1753    };
1754    // No projection (plain primitive/enum wire) — or an opaque HANDLE, whose
1755    // converter's wire is the raw `jlong` the typed `run` declares as `Long`
1756    // (`J`): the receiver constructs the typed class in bytecode. A nullable
1757    // handle boxes to `java.lang.Long` instead (object chunk). Fixed-size
1758    // arrays and other object wires stay objects (for example, `[B`).
1759    let proj_ok = match &entry.metadata.projection {
1760        None => true,
1761        Some(p) => matches!(p.kind, ProjectionKind::Handle | ProjectionKind::Unsigned64),
1762    };
1763    proj_ok && matches!(jni_field_access(&entry.destination), Some((_, _, false)))
1764}
