5/// Internal collector that accumulates `syn::Item` objects (Rust AST items)
6/// and writes them as `prettyplease`-formatted Rust source code to a file.
7///
8/// This is the final step of `Registry::write_rust`, where the generated FFI
9/// items are materialized into the bindings file included by the consumer crate.
10///
11/// The [`write`](Self::write) method handles path resolution automatically:
12/// - Relative paths are resolved relative to the `OUT_DIR` environment variable
13/// - Absolute paths are used as-is
14pub struct Destination {
15    file: syn::File,
16}
17
18impl FromIterator<syn::Item> for Destination {
19    /// Creates a `Destination` from an iterator of `syn::Item` objects.
20    fn from_iter<T: IntoIterator<Item = syn::Item>>(iter: T) -> Self {
21        Self {
22            file: syn::File {
23                shebang: None,
24                attrs: vec![],
25                items: iter.into_iter().collect(),
26            },
27        }
28    }
29}
30
31impl FromIterator<(syn::Item, SourceLocation)> for Destination {
32    /// Creates a `Destination` from an iterator of `(syn::Item, SourceLocation)` tuples.
33    ///
34    /// The source location information is discarded during collection, keeping only
35    /// the `syn::Item` objects for code generation.
36    fn from_iter<T: IntoIterator<Item = (syn::Item, SourceLocation)>>(iter: T) -> Self {
37        Self {
38            file: syn::File {
39                shebang: None,
40                attrs: vec![],
41                items: iter.into_iter().map(|(item, _)| item).collect(),
42            },
43        }
44    }
45}

47impl Destination {
48    /// Writes the collected Rust items to a file and returns the absolute path.
49    ///
50    /// This method formats the collected `syn::Item` objects into valid Rust source code
51    /// using `prettyplease` and writes it to the specified file path.
52    ///
53    /// # Path Resolution
54    ///
55    /// - **Relative paths**: Resolved relative to the `OUT_DIR` environment variable
56    /// - **Absolute paths**: Used as-is without modification
57    ///
58    /// # Arguments
59    ///
60    /// * `filename` - The target file path (relative or absolute)
61    ///
62    /// # Returns
63    ///
64    /// The absolute path where the file was written.
65    ///
66    /// # Panics
67    ///
68    /// - If the `OUT_DIR` environment variable is not set (when using relative paths)
69    /// - If the file cannot be written (e.g., permission denied, disk full)
70    pub fn write<P: AsRef<Path>>(self, filename: P) -> std::path::PathBuf {
71        let file_path = if filename.as_ref().is_relative() {
72            let out_dir = env::var("OUT_DIR").expect("OUT_DIR environment variable not set");
73            std::path::PathBuf::from(out_dir).join(filename)
74        } else {
75            filename.as_ref().to_path_buf()
76        };
77
78        let content = prettyplease::unparse(&self.file);
79        fs::write(&file_path, content).unwrap_or_else(|e| {
80            panic!("Failed to write file {}: {}", file_path.display(), e);
81        });
82
83        file_path
84    }
85}
86
87impl std::fmt::Display for Destination {
88    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
89        write!(f, "{}", prettyplease::unparse(&self.file))
90    }
