22/// Provides access to prebindgen data generated by the `#[prebindgen]` macro
23///
24/// The `Source` struct reads prebindgen data files from a directory that was
25/// initialized by [`init_prebindgen_out_dir`](crate::init_prebindgen_out_dir) in
26/// the source project's `build.rs`. This allows destination projects to access
27/// and process the collected FFI interface definitions.
28///
29/// # Workflow
30///
31/// 1. **Source project**: Calls `init_prebindgen_out_dir()` in `build.rs` and uses `#[prebindgen]` macro
32/// 2. **Destination project**: Uses `Source::new()` to read the collected data
33///
34/// # Groups
35///
36/// Items are organized into groups based on the first parameter of the `#[prebindgen]`
37/// macro. For example, `#[prebindgen("structs")]` creates items in the "structs" group.
38/// Items without an explicit group are placed in the "default" group.
39///
40/// # Example
41///
42/// The `PREBINDGEN_OUT_DIR` constant is defined in the source FFI crate using the
43/// `prebindgen_out_dir!()` macro:
44///
45/// ```rust
46/// // In source_ffi/src/lib.rs
47/// use prebindgen_proc_macro::{features, prebindgen, prebindgen_out_dir};
48///
49/// pub const PREBINDGEN_OUT_DIR: &str = prebindgen_out_dir!();
50/// pub const FEATURES: &str = features!();
51///
52/// #[prebindgen]
53/// pub fn my_function() -> i32 { 42 }
54/// ```
55///
56/// Then in the destination project's `build.rs`:
57///
58/// ```
59/// # prebindgen::Source::init_doctest_simulate();
60/// let source = prebindgen::Source::new("source_ffi");
61///
62/// // Process all items
63/// for (item, location) in source.items_all() {
64///     // Process each syn::Item...
65/// }
66/// ```
67#[derive(Clone)]
68pub struct Source {
69    crate_name: String,
70    items: HashMap<String, Vec<(syn::Item, SourceLocation)>>,
71    // Configuration needed to build a CfgFilter at iteration time
72    features_constant: Option<String>,
73    target_triple: Option<String>,
74    features_list: Vec<String>, // normalized list from features.txt
75}

94impl Source {
95    fn build_internal(
96        input_dir: &Path,
97        features_constant: Option<String>,
98        target_triple: Option<String>,
99        crate_name_override: Option<String>,
100    ) -> Self {
101        if let Some(source) = DOCTEST_SOURCE.with(|source| (*source.borrow()).clone()) {
102            return source;
103        }
104        if !input_dir.is_dir() {
105            panic!(
106                "Input directory {} does not exist or is not a directory",
107                input_dir.display()
108            );
109        }
110        // The stored name (CARGO_PKG_NAME at capture time) doubles as the
111        // "directory was initialized" check, so it is read even when
112        // overridden. The override wins: it is the name THIS crate
113        // references the source crate by (a Cargo.toml dependency rename).
114        let stored_crate_name = read_stored_crate_name(input_dir).unwrap_or_else(|| {
115            panic!(
116                "The directory {} was not initialized with init_prebindgen_out_dir(). \
117                Please ensure that init_prebindgen_out_dir() is called in the build.rs of the source crate.",
118                input_dir.display()
119            )
120        });
121        let crate_name = crate_name_override.unwrap_or(stored_crate_name);
122
123        let groups = Self::discover_groups(input_dir);
124        let mut items = HashMap::new();
125        for group in groups {
126            let records = Self::read_group(input_dir, &group);
127            let group_items = records
128                .iter()
129                .map(|r| {
130                    // Stamp the origin crate into every item's location: the
131                    // captured JSONL doesn't carry it (the proc-macro runs
132                    // inside the crate), but from here on the item streams
133                    // are self-describing — streams from several sources can
134                    // be chained into one `Flat::builder().items(..)` call
135                    // without losing per-item origins.
136                    let (item, mut loc) = r.parse();
137                    loc.crate_name = Some(crate_name.clone());
138                    (item, loc)
139                })
140                .collect::<Vec<_>>();
141            items.insert(group, group_items);
142        }
143
144        // Read features list once and store normalized list
145        let features_list = read_features_from_out_dir(input_dir);
146
147        Self {
148            crate_name,
149            items,
150            features_constant,
151            features_list,
152            target_triple,
153        }
154    }

213    /// Returns an iterator over items from specific groups
214    ///
215    /// # Parameters
216    ///
217    /// * `groups` - Array of group names to include
218    ///
219    /// # Example
220    ///
221    /// ```
222    /// # prebindgen::Source::init_doctest_simulate();
223    /// let source = prebindgen::Source::new("source_ffi");
224    /// // Process only items from "structs" and "functions" groups
225    /// let items = source.items_in_groups(&["structs"]).collect::<Vec<_>>();
226    /// assert_eq!(items.len(), 1); // only TestStruct should be present
227    /// ```
228    pub fn items_in_groups<'a>(
229        &'a self,
230        groups: &'a [&'a str],
231    ) -> impl Iterator<Item = (syn::Item, SourceLocation)> + 'a {
232        // Build a cfg filter and apply it lazily with itertools::batching
233        let mut filter = self.build_cfg_filter();
234        groups
235            .iter()
236            .filter_map(|group| self.items.get(*group))
237            .flat_map(|records| records.iter().cloned())
238            .batching(move |iter| filter.call(iter))
239    }

241    /// Returns an iterator over items excluding specific groups
242    ///
243    /// # Parameters
244    ///
245    /// * `groups` - Array of group names to exclude
246    ///
247    /// # Example
248    ///
249    /// ```
250    /// # prebindgen::Source::init_doctest_simulate();
251    /// let source = prebindgen::Source::new("source_ffi");
252    /// let items = source.items_except_groups(&["structs"]).collect::<Vec<_>>();
253    /// assert_eq!(items.len(), 1); // only test_function should be present
254    /// ```
255    pub fn items_except_groups<'a>(
256        &'a self,
257        groups: &'a [&'a str],
258    ) -> impl Iterator<Item = (syn::Item, SourceLocation)> + 'a {
259        // Build a cfg filter and apply it lazily with itertools::batching
260        let mut filter = self.build_cfg_filter();
261        self.items
262            .iter()
263            .filter(|(group, _)| !groups.contains(&group.as_str()))
264            .flat_map(|(_, records)| records.iter().cloned())
265            .batching(move |iter| filter.call(iter))
266    }

268    /// Returns an iterator over all items from all groups
269    ///
270    /// This is the most commonly used method for processing all prebindgen items.
271    ///
272    /// # Example
273    ///
274    /// ```
275    /// # prebindgen::Source::init_doctest_simulate();
276    /// let source = prebindgen::Source::new("source_ffi");
277    /// let items: Vec<_> = source.items_all().collect();
278    /// assert_eq!(items.len(), 2); // should contain TestStruct and test_function
279    /// ```
280    pub fn items_all(&self) -> impl Iterator<Item = (syn::Item, SourceLocation)> + '_ {
281        // Build a cfg filter and apply it lazily with itertools::batching
282        let mut filter = self.build_cfg_filter();
283        self.items
284            .values()
285            .flat_map(|records| records.iter().cloned())
286            .batching(move |iter| filter.call(iter))
287    }

289    /// Internal: construct a CfgFilter from the stored configuration and features file
290    fn build_cfg_filter(&self) -> cfg_filter::CfgFilter {
291        let mut builder = cfg_filter::CfgFilter::builder();
292        if let Some(const_name) = &self.features_constant {
293            // If the provided constant isn't fully qualified, qualify it with the crate name
294            let qualified_const = if const_name.contains("::") {
295                const_name.clone()
296            } else {
297                format!("{}::{}", self.crate_name.replace('-', "_"), const_name)
298            };
299            // Prepend each feature with "crate_name/" and join with spaces
300            let features_list = self
301                .features_list
302                .iter()
303                .map(|f| format!("{}/{}", self.crate_name, f))
304                .join(" ");
305            builder = builder.predefined_features(qualified_const, features_list);
306        }
307        if let Some(target) = &self.target_triple {
308            let target_triple = TargetTriple::parse(target)
309                .unwrap_or_else(|e| panic!("Failed to parse target triple '{}': {}", target, e));
310            if let Some(arch) = target_triple.arch() {
311                builder = builder.enable_target_arch(arch);
312            }
313            if let Some(vendor) = target_triple.vendor() {
314                builder = builder.enable_target_vendor(vendor);
315            }
316            if let Some(os) = target_triple.os() {
317                builder = builder.enable_target_os(os);
318            }
319            if let Some(env) = target_triple.env() {
320                builder = builder.enable_target_env(env);
321            }
322        }
323        builder.build()
324    }

326    /// Internal method to read all exported files matching the group name pattern `<group>_*`
327    fn read_group<P: AsRef<Path>>(input_dir: P, group: &str) -> Vec<Record> {
328        let pattern = format!("{group}_");
329        let mut record_map = HashMap::new();
330
331        // Read the directory and find all matching files
332        if let Ok(entries) = fs::read_dir(&input_dir) {
333            for entry in entries.flatten() {
334                let path = entry.path();
335                if let Some(file_name) = path.file_name().and_then(|n| n.to_str()) {
336                    if file_name.starts_with(&pattern) && file_name.ends_with(JSONL_EXTENSION) {
337                        #[cfg(feature = "debug")]
338                        println!("Reading exported file: {}", path.display());
339                        let path_clone = path.clone();
340
341                        match read_jsonl_file(&path) {
342                            Ok(records) => {
343                                for record in records {
344                                    // Use a HashMap to deduplicate records by name and cfg
345                                    let key = if let Some(cfg) = &record.cfg {
346                                        format!("{}#{}", record.name, cfg)
347                                    } else {
348                                        record.name.clone()
349                                    };
350                                    record_map.insert(key, record);
351                                }
352                            }
353                            Err(e) => {
354                                panic!("Failed to read {}: {}", path_clone.display(), e);
355                            }
356                        }
357                    }
358                }
359            }
360        }
361
362        // Return deduplicated records for this group
363        record_map.into_values().collect::<Vec<_>>()
364    }

366    /// Internal method to discover all available groups from the directory
367    fn discover_groups<P: AsRef<Path>>(input_dir: P) -> HashSet<String> {
368        let mut groups = HashSet::new();
369
370        // Discover all available groups
371        if let Ok(entries) = fs::read_dir(input_dir) {
372            for entry in entries.flatten() {
373                let path = entry.path();
374                if let Some(file_name) = path.file_name().and_then(|n| n.to_str()) {
375                    if file_name.ends_with(JSONL_EXTENSION) {
376                        // Extract group name from filename (everything before the first underscore)
377                        if let Some(underscore_pos) = file_name.find('_') {
378                            let group_name = &file_name[..underscore_pos];
379                            groups.insert(group_name.to_string());
380                        }
381                    }
382                }
383            }
384        }
385
386        groups
387    }

390/// Read the crate name from the stored file
391fn read_stored_crate_name(input_dir: &Path) -> Option<String> {
392    let crate_name_path = input_dir.join(CRATE_NAME_FILE);
393    fs::read_to_string(crate_name_path)
394        .ok()
395        .map(|s| s.trim().to_string())
396}

398/// Read enabled features list from FEATURES_FILE and normalize into a sorted, deduplicated Vec<String>
399fn read_features_from_out_dir(input_dir: &Path) -> Vec<String> {
400    let features_path = input_dir.join(FEATURES_FILE);
401    let Some(contents) = fs::read_to_string(features_path).ok() else {
402        return Vec::new();
403    };
404    let mut features: Vec<String> = contents
405        .lines()
406        .map(str::trim)
407        .filter(|l| !l.is_empty())
408        .map(|s| s.to_string())
409        .collect();
410    features.sort();
411    features.dedup();
412    features
413}

415/// Builder for constructing a `Source` with custom options
416pub struct Builder {
417    input_dir: PathBuf,
418    features_constant: Option<String>,
419    target_triple: Option<String>,
420    crate_name: Option<String>,
421}
422
423impl Builder {
424    fn new<P: AsRef<Path>>(input_dir: P) -> Self {
425        let target = std::env::var("TARGET").unwrap_or_default();
426        let host = std::env::var("HOST").unwrap_or_default();
427        let target_triple = if !target.is_empty() {
428            Some(target)
429        } else if !host.is_empty() {
430            Some(host)
431        } else {
432            None
433        };
434        Self {
435            input_dir: input_dir.as_ref().to_path_buf(),
436            features_constant: Some("FEATURES".to_string()),
437            target_triple,
438            crate_name: None,
439        }
440    }

442    /// Override the crate name recorded by the source crate's build script
443    /// (`CARGO_PKG_NAME` at capture time) with the name **this** crate
444    /// references the source crate by. Needed when the dependency is
445    /// renamed in Cargo.toml:
446    ///
447    /// ```toml
448    /// [dependencies]
449    /// myflat = { package = "some-flat-crate", version = "..." }
450    /// ```
451    ///
452    /// The name is stamped into every yielded item's `SourceLocation`, so
453    /// it is what generated code qualifies references with
454    /// (`myflat::…`) and what qualifies the features-guard constant. Being
455    /// per-`Source`, it composes with multi-source registries — each
456    /// chained stream carries its own (possibly overridden) origin.
457    pub fn crate_name(mut self, name: impl Into<String>) -> Self {
458        self.crate_name = Some(name.into());
459        self
460    }

462    /// Enables or disables filtering by features when extracting collected data.
463    /// Accepts name of the constant with the list of features in the source crate.
464    ///
465    /// Pass `None` to disable feature filtering.
466    ///
467    /// Pass, for example, `Some("source_crate::FEATURES")` or `Some("FEATURES")`
468    /// to enable filtering. The value is the name of the features constant
469    /// from the source crate.
470    ///
471    /// It's important to note that the set of features to filter is determined
472    /// not by this constant, but by file "features.txt" in
473    /// prebindgen output directory. These are features which the source crate was
474    /// built with *as build.rs dependency*. The constant contains the features which
475    /// the source crate was built *as library dependency*.
476    /// The purpose of the constant is to use it in the assert in the
477    /// generated code to ensure that both feature lists are the same.
478    /// If it's not the case, compilation fails and it's developer's
479    /// job to ensure that the source crate is configured in the same way for
480    /// both `[dependencies]` and `[build-dependencies]`.
481    ///
482    /// Filtering is enabled by default; the default constant name is `FEATURES`.
483    ///
484    /// The features constant should be defined in the source crate as follows:
485    /// ```rust
486    /// const FEATURES: &str = prebindgen_proc_macro::features!();
487    /// ```
488    #[roxygen]
489    pub fn enable_feature_filtering(
490        mut self,
491        /// Full name of the constant with features in the source crate
492        name: Option<impl Into<String>>,
493    ) -> Self {
494        self.features_constant = name.map(|n| n.into());
495        self
496    }

498    /// Enables filtering source code by target-triple. This is
499    /// useful if some binding generators or custom tools over them
500    /// encounter problems with build conditions in the source code.
501    ///
502    /// Accepts the target triple string.
503    ///
504    /// Default value is the value of the environment variable
505    /// `TARGET` or `HOST` if `TARGET` is not set.
506    ///
507    /// Pass `None` to disable target filtering.
508    ///
509    /// When enabled, the code will be filtered by the target triple specified
510    /// (architecture, vendor, os, env).
511    ///
512    /// It's useful to mention that it's hard to imagine a scenario
513    /// when it's necessary to change the target triple.
514    /// The `Source` object is used in the final crate, the one which builds the
515    /// no-mangle library. On this stage the `TARGET` environment variable contains
516    /// real target triple for cross-compilation scenarios.
517    ///
518    /// There is the case when build.rs is not aware of the cross-compilation
519    /// target: the build.rs for library in [build-dependencies] section.
520    /// This is the situation which the "ffi" crate found itself in. So sometimes
521    /// for correct code generation it's necessary to inform the underlying "ffi"
522    /// crate about the real target in a way like this:
523    ///
524    /// ```sh
525    /// CROSS_TARGET=x86_64-pc-windows-gnu cargo build --target x86_64-pc-windows-gnu
526    /// ```
527    ///
528    /// But in typical situation this doesn't affect the `Source` object which
529    /// reads the data prepared by "ffi" crate on the following stage.
530    ///
531    /// So this `Some(<target-triple>)` is added mostly for API completeness,
532    /// typical usage if this function will be with `None` argument to disable
533    /// filtering by target triple if it's necessary to see the original
534    /// collected code.
535    #[roxygen]
536    pub fn enable_target_filtering(
537        mut self,
538        /// The target triple to filter collected sources
539        name: Option<impl Into<String>>,
540    ) -> Self {
541        self.target_triple = name.map(|n| n.into());
542        self
543    }
544
545    /// Build the `Source` instance
546    pub fn build(self) -> Source {
547        Source::build_internal(
548            &self.input_dir,
549            self.features_constant,
550            self.target_triple,
551            self.crate_name,
552        )
553    }
