9/// Direction relative to the flat Rust value, independently of its location.
10///
11/// A callback argument usually leaves Rust, even though it is a parameter.
12/// Defaults and overrides are keyed by this direction, not by parameter/return.
13#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
14pub enum Direction {
15    /// Destination value → wires → flat Rust value.
16    IntoRust,
17    /// Flat Rust value → wires → destination value.
18    OutOfRust,
19}
20
21/// A selected conversion for one occurrence of a flat type.
22///
23/// `W` is the generator's closed wire enum; `M` describes the destination
24/// value as a whole (a class, constructor, builder, or an adapter-owned tree).
25/// The [`Form`] records shared conversion structure; neither writer needs to
26/// look up defaults again. Input and output plans are resolved independently.
27///
28/// Constructors validate wire-name uniqueness. The resolver also validates
29/// the root type and direction. These checks validate the plan's structure,
30/// not arbitrary Rust token semantics or the existence of trait impls: compile
31/// generated Rust to verify those.
32#[derive(Clone, Debug)]
33pub struct ConversionPlan<W, M = ()> {
34    code: Code<W>,
35    metadata: M,
36}
37
38#[derive(Clone, Debug)]
39enum Code<W> {
40    Input {
41        conversion: Input<W>,
42        setup: TokenStream,
43    },
44    Output {
45        conversion: Output<W>,
46        source: syn::Ident,
47    },
48}
49
50impl<W: WireType, M> ConversionPlan<W, M> {
51    /// Select a wires-to-Rust conversion with destination metadata.
52    pub fn input(conversion: Input<W>, metadata: M) -> Result<Self, String> {
53        Self::input_with_setup(conversion, TokenStream::new(), metadata)
54    }
55
56    /// Select an input that needs backing storage or a runtime guard.
57    ///
58    /// `setup` runs once before the converted value is bound. It may use `?`
59    /// in the enclosing error scope. Storage and RAII guards declared here
60    /// remain alive through the body passed to [`Self::with_input`], including
61    /// its failure paths. The conversion expression may borrow that storage.
62    ///
63    /// This iteration supports setup at the plan root. Conditional backing
64    /// storage inside optional/choice children and explicit non-RAII cleanup
65    /// are left to the generator; tools does not synthesize them yet.
66    pub fn input_with_setup(
67        conversion: Input<W>,
68        setup: TokenStream,
69        metadata: M,
70    ) -> Result<Self, String> {
71        check_wires(&conversion.form)?;
72        Ok(Self {
73            code: Code::Input { conversion, setup },
74            metadata,
75        })
76    }
77
78    /// Select a Rust-to-wires conversion.
79    ///
80    /// Its expression reads the flat value through `source`. [`Self::emit_output`]
81    /// binds that local once, before evaluating the conversion.
82    pub fn output(source: syn::Ident, conversion: Output<W>, metadata: M) -> Result<Self, String> {
83        check_wires(&conversion.form)?;
84        Ok(Self {
85            code: Code::Output { conversion, source },
86            metadata,
87        })
88    }
89
90    /// The original flat type at this occurrence, including borrowing.
91    pub fn source_type(&self) -> &TypeRef {
92        &self.form().ty
93    }
94
95    /// The direction this plan implements.
96    pub fn direction(&self) -> Direction {
97        match self.code {
98            Code::Input { .. } => Direction::IntoRust,
99            Code::Output { .. } => Direction::OutOfRust,
100        }
101    }
102
103    /// The shared relation tree, already selected for this occurrence.
104    pub fn form(&self) -> &Form<W> {
105        match &self.code {
106            Code::Input { conversion, .. } => &conversion.form,
107            Code::Output { conversion, .. } => &conversion.form,
108        }
109    }
110
111    /// Boundary slots in conversion order; zero, one, or several.
112    pub fn wires(&self) -> Vec<&Wire<W>> {
113        self.form().wires()
114    }
115
116    /// Destination-language meaning of the converted value as a whole.
117    ///
118    /// Per-slot semantics belong in `W`; structural/class/builder information
119    /// belongs here. An adapter can retain child metadata in its own `M` tree.
120    pub fn metadata(&self) -> &M {
121        &self.metadata
122    }
123
124    /// Consume a selected child for an input combinator, retaining its metadata.
125    ///
126    /// A child with setup storage cannot be flattened by the current expression
127    /// combinators; emit it through `with_input` at the element level instead.
128    pub fn into_input(self) -> Result<(Input<W>, M), String> {
129        match self.code {
130            Code::Input { conversion, setup } if setup.is_empty() => {
131                Ok((conversion, self.metadata))
132            }
133            Code::Input { .. } => {
134                Err("input setup requires with_input; child setup composition is deferred".into())
135            }
136            Code::Output { .. } => Err("into_input requires an IntoRust plan".into()),
137        }
138    }
139
140    /// Consume a selected child for an output combinator, retaining its metadata.
141    ///
142    /// Bind the child's flat `value` once to the source local its policy chose.
143    pub fn into_output(self, value: impl ToTokens) -> Result<(Output<W>, M), String> {
144        match self.code {
145            Code::Output {
146                mut conversion,
147                source,
148            } => {
149                let (value, expr) = (value.to_token_stream(), &conversion.expr);
150                conversion.expr = quote!({ let #source = #value; #expr });
151                Ok((conversion, self.metadata))
152            }
153            Code::Input { .. } => Err("into_output requires an OutOfRust plan".into()),
154        }
155    }
156
157    /// Emit setup, bind the converted input once, and use it within that scope.
158    ///
159    /// The enclosing generator supplies the Rust error scope. This deliberately
160    /// does not put the conversion in a separate `Result` closure: a borrowed
161    /// value must remain with its storage through the flat call. `body` should
162    /// consume the borrow within this scope, rather than return it.
163    /// Names for `local`, setup locals, and wires must be distinct.
164    /// Returns an error when called on an output plan.
165    pub fn with_input(
166        &self,
167        local: &syn::Ident,
168        body: impl FnOnce(TokenStream) -> TokenStream,
169    ) -> Result<TokenStream, String> {
170        let Code::Input { conversion, setup } = &self.code else {
171            return Err("with_input requires an IntoRust plan".into());
172        };
173        let expr = &conversion.expr;
174        let body = body(local.to_token_stream());
175        Ok(quote!({ #setup let #local = #expr; #body }))
176    }
177
178    /// Emit a conversion that evaluates `value` once and produces its wires.
179    ///
180    /// Zero wires produce `()`, one produces a bare value, several a tuple.
181    /// `?` propagates to the enclosing generator's error scope. The source
182    /// local shadows names only inside this block, after `value` is evaluated.
183    /// Returns an error when called on an input plan.
184    pub fn emit_output(&self, value: impl ToTokens) -> Result<TokenStream, String> {
185        let Code::Output { conversion, source } = &self.code else {
186            return Err("emit_output requires an OutOfRust plan".into());
187        };
188        let (value, expr) = (value.to_token_stream(), &conversion.expr);
189        Ok(quote!({ let #source = #value; #expr }))
190    }
191}
192
193fn check_wires<W>(form: &Form<W>) -> Result<(), String> {
194    let mut names = HashSet::new();
195    for wire in form.wires() {
196        if !names.insert(names::bare(&wire.name)) {
197            return Err(format!("duplicate wire `{}`", wire.name));
198        }
199    }
200    Ok(())
201}
