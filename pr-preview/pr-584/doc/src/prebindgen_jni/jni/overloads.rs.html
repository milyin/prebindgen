1//! #52: idiomatic Kotlin overloads for expanded params.
2//!
3//! A multi-variant `expand_param!(T)` crosses the wire as a selector tuple
4//! (`expectedSel: Int, expected00: Long?, …`); the raw call site passes magic
5//! ints and null-padding. Two mechanisms turn that into idiomatic Kotlin:
6//!
7//! * **Proactive splittability check** ([`Declarations::validate_split_declarations`]):
8//!   every multi-variant `expand_param!` declaration (type-level or per-fn) is
9//!   verified up front to be *splittable* — its arms surface as pairwise-distinct
10//!   JVM signatures — so a function can safely request overloads. A collision is
11//!   a hard build error attributed to the declaration; `.no_split()` opts out.
12//! * **Per-function emission** ([`render_param_overloads`], driven by
13//!   [`FunctionDecl::split_on_param`](prebindgen_registry::fun)): for the named split params the
14//!   generator emits, alongside the selector wrapper, the **cartesian product** of
15//!   their arms as typed overloads, each delegating to the selector form. The
16//!   concrete product must have no two combinations sharing a JVM signature.
17//!
18//! Only flat arms are splittable; an explicit `.split_on_param` on a
19//! recursively-built / single-variant / unknown parameter is a hard error.
20//!
21//! **Optional params (nullable-arm rule)** — an `Option<…>` parameter is
22//! splittable through its **single-leaf** arms only (the identity arm, or a
23//! build arm with exactly one constructor input): the overload keeps the arm's
24//! nullable type (`T?`) and `null` selects absence (selector `-1`), mirroring
25//! the "one nullable leaf decides presence" convention of optional single-ctor
26//! expansion. Multi-leaf arms have no nullable single type and stay
27//! selector-only; an optional param with no single-leaf arm is a hard error.
28//! With two qualifying nullable arms a bare `null` argument is ambiguous at
29//! Kotlin call sites (both mean absent) and needs a cast — signatures are
30//! still pairwise-distinct per the checks above.

42impl Declarations {
43    /// Proactively verify every multi-variant `expand_param!` declaration is
44    /// splittable (its arms have pairwise-distinct JVM-erased signatures), so
45    /// [`FunctionDecl::split_on_param`](prebindgen_registry::fun) can emit unambiguous
46    /// overloads. Runs regardless of whether any function actually splits the
47    /// type, so authorship errors surface early. `.no_split()` opts a decl out.
48    /// A collision errors with a message attributed to the declaration
49    /// (surfaced through the [`validate_resolved`] boundary before any
50    /// artifact is written).
51    ///
52    /// [`validate_resolved`]: prebindgen_registry::Prebindgen::validate_resolved
53    pub(crate) fn validate_split_declarations(&self, registry: &Registry) -> Result<(), String> {
54        let type_level = self
55            .param_expand_decls
56            .iter()
57            .map(|d| (d.key().as_str().to_string(), d));
58        let per_fn = self
59            .fn_param_expands
60            .iter()
61            .map(|(func, param, d)| (format!("fun `{func}` param `{param}`"), d));
62        for (site, decl) in type_level.chain(per_fn) {
63            if decl.is_no_split() || decl.variants().len() < 2 {
64                continue;
65            }
66            let target = decl.rust_type().key();
67            let sigs: Vec<(String, Vec<ErasedJvmType>)> = decl
68                .variants()
69                .iter()
70                .map(|v| {
71                    let ctor = match v {
72                        LocalVariant::Ctor(c) => Some(c),
73                        LocalVariant::SelfIdentity => None,
74                    };
75                    (
76                        ctor.map(|c| c.to_string())
77                            .unwrap_or_else(|| "variant_self()".to_string()),
78                        arm_erased_sig(self, registry, &target, ctor),
79                    )
80                })
81                .collect();
82            for i in 0..sigs.len() {
83                for j in (i + 1)..sigs.len() {
84                    if sigs[i].1 == sigs[j].1 {
85                        return Err(format!(
86                            "expand_param!({t}) [{site}]: variants {a} and {b} both surface as \
87                             `({sig})` — a split would emit two overloads with the same JVM \
88                             signature; disambiguate the constructors or add .no_split()",
89                            t = decl.key().as_str(),
90                            a = sigs[i].0,
91                            b = sigs[j].0,
92                            sig = sigs[i]
93                                .1
94                                .iter()
95                                .map(|e| e.to_string())
96                                .collect::<Vec<_>>()
97                                .join(", "),
98                        ));
99                    }
100                }
101            }
102        }
103        Ok(())
104    }
105}

107/// The JVM-erased type list of one arm: the constructor's parameter types
108/// (build arm), or the single target type (`variant_self`, `ctor == None`).
109/// Uses the shared [`erase_kt_type`] model (issue #89 stage 2) so the split
110/// ambiguity check and the whole-artifact overload table agree on erasure.
111fn arm_erased_sig(
112    ext: &Declarations,
113    registry: &Registry,
114    target: &TypeKey,
115    ctor: Option<&syn::Ident>,
116) -> Vec<ErasedJvmType> {
117    match ctor {
118        Some(cf) => match registry.flat().function(&cf) {
119            Some(f) => f
120                .params
121                .iter()
122                .map(|p| rust_type_erased(ext, &p.ty))
123                .collect(),
124            None => Vec::new(),
125        },
126        // The identity, because the two callers reach it differently: one holds
127        // a plan's reading, the other a `.expand_param(...)` DECLARATION, which
128        // the build script wrote and the model may never have interned. Where
129        // it did, the reading answers as before; where it did not, the erased
130        // form is the identity's own canonical spelling — which is all a
131        // declaration has to be told apart by.
132        None => vec![match registry.reading(target) {
133            Some(reading) => rust_type_erased(ext, &reading),
134            None => ErasedJvmType::raw(target.as_str().to_string()),
135        }],
136    }
137}

139/// The [`ErasedJvmType`] a Rust arm type surfaces as: map it to its Kotlin
140/// surface type (a declared class's FQN, else the resolved converter's Kotlin
141/// name) and run the shared [`erase_kt_type`]; a plain class folds to its FQN
142/// there. Falls back to the token string for a
143/// type with no resolved surface. References are peeled first (`&T` erases
144/// like `T`).
145fn rust_type_erased(ext: &Declarations, ty: &prebindgen_registry::flat::TypeRef) -> ErasedJvmType {
146    // The spelling's own outermost borrow, as `TypeKind::Ref` — not
147    // `borrow_target`, which reaches through the erased wrappers (#S31).
148    let peeled = match ty.kind() {
149        prebindgen_registry::flat::TypeKind::Ref { inner, .. } => inner,
150        _ => ty,
151    };
152    let key = peeled.key();
153    if ext.types.get(&key).is_some_and(|c| c.name_spec.is_some()) {
154        if let Some(fqn) = ext.kotlin_fqn(&key) {
155            return erase_kt_type(&[], &KtType::cls(fqn));
156        }
157    }
158    if let Some(kt) = ext
159        .in_frag(peeled)
160        .and_then(|e| e.metadata.kotlin_name.clone())
161    {
162        return erase_kt_type(&[], &kt);
163    }
164    // The type's IDENTITY, not its spelling. `ErasedJvmType` is a string
165    // compared with `==` — "two parameters collide as overloads iff their
166    // tokens are equal" — so keying it on what the source wrote meant two
167    // spellings of one type (`Box<Payload>` and `Payload`, `crate::Foo` and
168    // `Foo`) erased to different tokens and were judged NOT to collide. The
169    // JVM would then reject the emitted class for a clash this check exists
170    // to catch first. A `TypeKey` is canonical, so equal types compare equal.
171    ErasedJvmType::raw(peeled.key().as_str())
172}

174/// Whether `plan` is a multi-variant expansion that can be turned into
175/// overloads: a selector (≥2 arms) and no recursively-built arm. An `Option`
176/// outer shape is in scope — its arms are filtered to single-leaf ones by
177/// [`resolve_split`] (nullable-arm rule, see module docs).
178fn plan_in_scope(plan: &FoldPlan) -> bool {
179    plan.selector.is_some()
180        && !plan
181            .variants
182            .iter()
183            .any(|v| v.inputs.iter().any(|a| matches!(a, FoldArg::Build(_))))
184}

186/// One resolved split parameter of a function, positioned against the rendered
187/// selector wrapper.
188struct Split<'a> {
189    /// The original Rust parameter ident (e.g. `expected`).
190    param: syn::Ident,
191    plan: &'a FoldPlan,
192    /// Start of this param's contiguous leaf block in `sel_fun.params`.
193    start: usize,
194    /// Block length (`plan.leaves.len()`).
195    len: usize,
196    /// Selector-leaf index within the block.
197    sel_idx: usize,
198    /// `Option<…>` parameter — overload arms keep their nullable type and the
199    /// delegated selector is conditional (`null` = absent = `-1`).
200    optional: bool,
201    /// Overloadable arms: original variant index (the selector value) plus the
202    /// arm's typed params as `(param, leaf-index-within-block)`. For an
203    /// optional param this is the single-leaf subset of the variants.
204    arms: Vec<(usize, Vec<(KtParam, usize)>)>,
205}

208/// The Kotlin names of a constructor's parameters.

210/// Off `Function::params`, where a parameter **is** a name and a reading. The

213/// already required.
214fn ctor_param_names(f: &prebindgen_registry::flat::Function) -> Vec<String> {
215    f.params
216        .iter()
217        .map(|p| kt_param_name(&p.name.to_string()))
218        .collect()

221/// `origin` + Capitalized `name` (`primary` + `count` → `primaryCount`).
222fn prefixed(origin: &str, name: &str) -> String {
223    let mut c = name.chars();
224    match c.next() {
225        Some(first) => format!("{origin}{}{}", first.to_uppercase(), c.as_str()),
226        None => origin.to_string(),
227    }
228}

230/// Clear selector-dispatch nullability (`T?` → `T`). Genuinely optional
231/// constructor parameters bypass this helper and retain their rendered `T?`.
232fn non_null(mut ty: KtType) -> KtType {
233    match &mut ty {
234        KtType::Named { nullable, .. } | KtType::Function { nullable, .. } => *nullable = false,
235    }
236    ty
237}

239/// The typed overload params of one variant arm, paired with the leaf index
240/// each fills. `origin`/`multi` drive name disambiguation (build-arm params are
241/// prefixed with the origin parameter name when the function splits more than
242/// one parameter). `optional_plan` keeps every slot's rendered nullability —
243/// for an `Option<…>` parameter `null` encodes absence (nullable-arm rule).
244/// Returns `None` if any input is not a flat leaf.
245fn variant_typed_params(
246    registry: &impl Conversions,
247    variant: &prebindgen_registry::expand::FoldVariant,
248    origin: &syn::Ident,
249    block: &[KtParam],
250    multi: bool,
251    optional_plan: bool,
252) -> Option<Vec<(KtParam, usize)>> {
253    let origin_kt = kt_param_name(&origin.to_string());
254    let (names, optional): (Vec<String>, Vec<bool>) = match &variant.ctor {
255        Some(cf) => {
256            let f = registry.flat().function(&cf)?;
257            // `Optional` off the kind, not `is_option` off a path: the same
258            // question, asked of the grammar the source wrote.
259            let optional = f
260                .params
261                .iter()
262                .map(|p| {
263                    matches!(
264                        p.ty.kind(),
265                        prebindgen_registry::flat::TypeKind::Optional(_)
266                    )
267                })
268                .collect();
269            (ctor_param_names(f), optional)
270        }
271        // Identity arm: one parameter, the value itself, named after the origin
272        // parameter (already unique across split params — never prefixed).
273        None => (vec![origin_kt.clone()], vec![false]),
274    };
275    let mut out = Vec::new();
276    for (m, arg) in variant.inputs.iter().enumerate() {
277        let FoldArg::Leaf(idx, _) = arg else {
278            return None;
279        };
280        let slot = block.get(*idx)?;
281        let base = names.get(m).cloned().unwrap_or_else(|| slot.name.clone());
282        let name = if multi && variant.ctor.is_some() {
283            prefixed(&origin_kt, &base)
284        } else {
285            base
286        };
287        let ty = if optional_plan || optional.get(m).copied().unwrap_or(false) {
288            slot.ty.clone()
289        } else {
290            non_null(slot.ty.clone())
291        };
292        out.push((KtParam::new(&name, ty), *idx));
293    }
294    Some(out)
295}

297/// Locate a param's contiguous leaf block in the selector wrapper's parameter
298/// list, matching by leaf name in order.
299fn find_block(params: &[KtParam], leaf_names: &[String]) -> Option<usize> {
300    if leaf_names.is_empty() || params.len() < leaf_names.len() {
301        return None;
302    }
303    (0..=params.len() - leaf_names.len()).find(|&s| {
304        params[s..s + leaf_names.len()]
305            .iter()
306            .zip(leaf_names)
307            .all(|(p, n)| &p.name == n)
308    })
309}

311/// Resolve one `.split_on_param` request against the rendered selector wrapper,
312/// validating the parameter is expandable, multi-variant, and in-scope (all
313/// hard errors — the user explicitly asked to split it).
314fn resolve_split<'a>(
315    registry: &'a Registry,
316    fplan: &'a JniFunctionPlan,
317    f: &prebindgen_registry::flat::Function,
318    sel_fun: &KtFun,
319    param_name: &str,
320    multi: bool,
321) -> Split<'a> {
322    let param = syn::Ident::new(param_name, Span::call_site());
323    let plan = fplan
324        .params
325        .iter()
326        .find(|planned| planned.ident == param)
327        .and_then(|planned| match &planned.form {
328            ParamForm::Expanded { plan, .. } => Some(plan.fold()),
329            ParamForm::Single(_) => None,
330        })
331        .unwrap_or_else(|| {
332            panic!(
333                "fun!({}).split_on_param(\"{param_name}\"): `{param_name}` is not an expandable \
334                 parameter (it has no `expand_param!` variants)",
335                f.name
336            )
337        });
338    assert!(
339        plan.selector.is_some(),
340        "fun!({}).split_on_param(\"{param_name}\"): `{param_name}` has a single variant — there \
341         is nothing to split (it already flattens to one signature)",
342        f.name
343    );
344    assert!(
345        plan_in_scope(plan),
346        "fun!({}).split_on_param(\"{param_name}\"): `{param_name}` has a recursively-built arm — \
347         it cannot be overloaded; keep the selector form",
348        f.name
349    );
350    let leaf_names: Vec<String> = plan
351        .leaves
352        .iter()
353        .map(|l| kt_param_name(&l.name.to_string()))
354        .collect();
355    let len = leaf_names.len();
356    let start = find_block(&sel_fun.params, &leaf_names).unwrap_or_else(|| {
357        panic!(
358            "fun!({}).split_on_param(\"{param_name}\"): could not locate the parameter's leaf \
359             block in the generated wrapper",
360            f.name
361        )
362    });
363    let block = &sel_fun.params[start..start + len];
364    let sel_idx = plan.selector.expect("selector present");
365    // Nullable-arm rule: an `Option<…>` param is overloadable through its
366    // single-leaf arms only — the arm's one nullable param doubles as the
367    // presence flag (`null` = absent). Multi-leaf arms stay selector-only.
368    let optional = plan.produces_option();
369    let arms: Vec<(usize, Vec<(KtParam, usize)>)> = plan
370        .variants
371        .iter()
372        .enumerate()
373        .filter(|(_, v)| !optional || v.inputs.len() == 1)
374        .map(|(vi, v)| {
375            let typed = variant_typed_params(registry, v, &param, block, multi, optional)
376                .unwrap_or_else(|| {
377                    panic!(
378                        "fun!({}).split_on_param(\"{param_name}\"): an arm has a non-flat input; \
379                         it cannot be overloaded",
380                        f.name
381                    )
382                });
383            (vi, typed)
384        })
385        .collect();
386    assert!(
387        !arms.is_empty(),
388        "fun!({}).split_on_param(\"{param_name}\"): `{param_name}` is an `Option<_>` parameter \
389         and none of its arms is a single leaf — its overload has no clean nullable type; keep \
390         the selector form",
391        f.name
392    );
393    Split {
394        param,
395        plan,
396        start,
397        len,
398        sel_idx,
399        optional,
400        arms,
401    }
402}

404/// The overloads for one function, delegating to its already-rendered selector
405/// wrapper `sel_fun`. Empty unless the function has `.split_on_param` requests.
406/// Emits the cartesian product of the named params' arms; panics (a build
407/// error) if the product has two combinations with the same JVM signature.
408pub(crate) fn render_param_overloads(
409    ext: &Declarations,
410    f: &prebindgen_registry::flat::Function,
411    registry: &Registry,
412    sel_fun: &KtFun,
413) -> Vec<KtFun> {
414    // Requested split params for this function, in signature order.
415    let requested: Vec<String> = {
416        let want: std::collections::HashSet<&str> = ext
417            .fn_split_params
418            .iter()
419            .filter(|(func, _)| func == &f.name)
420            .map(|(_, p)| p.as_str())
421            .collect();
422        if want.is_empty() {
423            return Vec::new();
424        }
425        f.params
426            .iter()
427            .filter(|p| want.contains(p.name.to_string().as_str()))
428            .map(|p| p.name.to_string())
429            .collect()
430    };
431    // Any requested name that didn't match a real parameter is a typo — surface
432    // it rather than silently dropping.
433    for (func, p) in &ext.fn_split_params {
434        if func == &f.name && !requested.iter().any(|r| r == p) {
435            panic!(
436                "fun!({}).split_on_param(\"{p}\"): no parameter named `{p}` on this function",
437                f.name
438            );
439        }
440    }
441
442    let fplan = ext.fn_plan(registry, f).unwrap_or_else(|_| {
443        panic!(
444            "fun!({}): its frozen JNI generation plan is unavailable",
445            f.name
446        )
447    });
448    let multi = requested.len() > 1;
449    let splits: Vec<Split> = requested
450        .iter()
451        .map(|name| resolve_split(registry, &fplan, f, sel_fun, name, multi))
452        .collect();
453
454    // Cartesian product of arm indices across all split params.
455    let combos = cartesian(&splits.iter().map(|s| s.arms.len()).collect::<Vec<_>>());
456
457    // Product-global JVM-signature collision check (fixed params are identical
458    // across every overload, so only the split-arm lists can collide).
459    let sigs: Vec<Vec<ErasedJvmType>> = combos
460        .iter()
461        .map(|combo| {
462            splits
463                .iter()
464                .zip(combo)
465                .flat_map(|(s, &ai)| {
466                    let ctor = s.plan.variants[s.arms[ai].0].ctor.as_ref();
467                    arm_erased_sig(ext, registry, &s.plan.target.key(), ctor)
468                })
469                .collect()
470        })
471        .collect();
472    for i in 0..sigs.len() {
473        for j in (i + 1)..sigs.len() {
474            if sigs[i] == sigs[j] {
475                panic!(
476                    "fun!({}): split_on_param product is ambiguous — combinations {} and {} both \
477                     surface as `({})`; add .no_split() intent is not enough here, disambiguate \
478                     the constructors or drop one .split_on_param",
479                    f.name,
480                    combo_label(&splits, &combos[i]),
481                    combo_label(&splits, &combos[j]),
482                    sigs[i]
483                        .iter()
484                        .map(|e| e.to_string())
485                        .collect::<Vec<_>>()
486                        .join(", "),
487                );
488            }
489        }
490    }
491
492    // Emit one overload per product combination.
493    let n = sel_fun.params.len();
494    let mut out = Vec::with_capacity(combos.len());
495    for combo in &combos {
496        // Per-split delegation slots, keyed by block start.
497        let mut params: Vec<KtParam> = Vec::new();
498        let mut call_args: Vec<String> = Vec::new();
499        let mut pos = 0usize;
500        while pos < n {
501            if let Some((si, s)) = splits.iter().enumerate().find(|(_, s)| s.start == pos) {
502                // Replace this param's whole leaf block with the chosen arm's
503                // typed params; fill the block's delegation slots.
504                let (vi, typed) = &s.arms[combo[si]];
505                let mut leaf_arg: Vec<String> = vec!["null".to_string(); s.len];
506                leaf_arg[s.sel_idx] = if s.optional {
507                    // Nullable-arm rule: the arm's single param doubles as the
508                    // presence flag — `null` delegates absence (selector -1).
509                    format!("if ({} != null) {vi} else -1", typed[0].0.name)
510                } else {
511                    vi.to_string()
512                };
513                for (p, lidx) in typed {
514                    params.push(p.clone());
515                    leaf_arg[*lidx] = p.name.clone();
516                }
517                call_args.extend(leaf_arg);
518                pos += s.len;
519            } else {
520                // A fixed (or non-split expanded) param — passes through.
521                let p = &sel_fun.params[pos];
522                params.push(p.clone());
523                call_args.push(p.name.clone());
524                pos += 1;
525            }
526        }
527
528        // Guard against a param-name clash (e.g. an arm name colliding with a
529        // fixed param) rather than emitting uncompilable Kotlin.
530        let mut seen = std::collections::HashSet::new();
531        for p in &params {
532            assert!(
533                seen.insert(p.name.clone()),
534                "fun!({}): split overload has a duplicate parameter name `{}` — rename the \
535                 constructor parameter",
536                f.name,
537                p.name
538            );
539        }
540
541        out.push(overload_shell(
542            sel_fun,
543            params,
544            KtCode::new().line(format!("{}({})", sel_fun.name, call_args.join(", "))),
545        ));
546    }
547    out
548}

550/// An overload derived from the selector wrapper by a signature-preserving
551/// transform: generics, annotations, modifiers, visibility, and return type
552/// carry over unchanged (a wrapper generic over `R`/`A` — builder/fold
553/// delivery — keeps its `fun <R> …` declaration, #87); only the parameter
554/// list and the delegating body are replaced. Kdoc stays on the selector
555/// form — overloads are bare.
556fn overload_shell(sel_fun: &KtFun, params: Vec<KtParam>, body: KtCode) -> KtFun {
557    let mut ov = sel_fun.clone();
558    ov.params = params;
559    ov.kdoc = None;
560    ov.body = KtBody::Expr(body);
561    ov
562}

564/// Cartesian product of index ranges `0..counts[k]`, as a list of index
565/// tuples. `[2, 2]` → `[[0,0],[0,1],[1,0],[1,1]]`.
566fn cartesian(counts: &[usize]) -> Vec<Vec<usize>> {
567    let mut acc = vec![Vec::new()];
568    for &c in counts {
569        acc = acc
570            .into_iter()
571            .flat_map(|prefix| {
572                (0..c).map(move |i| {
573                    let mut next = prefix.clone();
574                    next.push(i);
575                    next
576                })
577            })
578            .collect();
579    }
580    acc
581}

583/// A `param=variant` label for one product combination, for error messages.
584fn combo_label(splits: &[Split], combo: &[usize]) -> String {
585    let parts: Vec<String> = splits
586        .iter()
587        .zip(combo)
588        .map(|(s, &ai)| {
589            let v = match &s.plan.variants[s.arms[ai].0].ctor {
590                Some(c) => c.to_string(),
591                None => "variant_self()".to_string(),
592            };
593            format!("{}={v}", s.param)
594        })
595        .collect();
596    format!("({})", parts.join(", "))
597}
