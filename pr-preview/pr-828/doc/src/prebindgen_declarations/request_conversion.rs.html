1//! Embed explicit operations into preparation trees.
2use crate::{Input, InputRequest, Output, OutputRequest};
3
4impl Input {
5    /// Embed this subtree in preparation. This keeps its explicit operations;
6    /// source-dependent normalization still runs when the tree is resolved.
7    pub fn into_preparation(self) -> Input<InputRequest> {
8        match self {
9            Input::Request(never) => match never {},
10            Input::Leaf => Input::Leaf,
11            Input::Optional(child) => Input::Optional(Box::new(child.into_preparation())),
12            Input::Sequence(child) => Input::Sequence(Box::new(child.into_preparation())),
13            Input::Fields(children) => {
14                Input::Fields(children.into_iter().map(Input::into_preparation).collect())
15            }
16            Input::Record(children) => {
17                Input::Record(children.into_iter().map(Input::into_preparation).collect())
18            }
19            Input::Selector(children) => {
20                Input::Selector(children.into_iter().map(Input::into_preparation).collect())
21            }
22            Input::Constructor { function, args } => Input::Constructor {
23                function,
24                args: args.into_iter().map(Input::into_preparation).collect(),
25            },
26        }
27    }
28}
29impl Output {
30    /// Embed this subtree in preparation. This keeps its explicit operations;
31    /// source-dependent normalization still runs when the tree is resolved.
32    pub fn into_preparation(self) -> Output<OutputRequest> {
33        match self {
34            Output::Request(never) => match never {},
35            Output::Leaf => Output::Leaf,
36            Output::Discard => Output::Discard,
37            Output::Projected(policy) => Output::Projected(policy),
38            Output::Optional(child) => Output::Optional(Box::new(child.into_preparation())),
39            Output::Named { name, output } => Output::Named {
40                name,
41                output: Box::new(output.into_preparation()),
42            },
43            Output::Sequence {
44                element,
45                mode,
46                delivery,
47            } => Output::Sequence {
48                element: Box::new(element.into_preparation()),
49                mode,
50                delivery,
51            },
52            Output::Result { ok, error } => Output::Result {
53                ok: Box::new(ok.into_preparation()),
54                error,
55            },
56            Output::Fields(children) => {
57                Output::Fields(children.into_iter().map(Output::into_preparation).collect())
58            }
59            Output::OwnedFields(children) => {
60                Output::OwnedFields(children.into_iter().map(Output::into_preparation).collect())
61            }
62            Output::Combine(children) => {
63                Output::Combine(children.into_iter().map(Output::into_preparation).collect())
64            }
65            Output::Record { fields, mode } => Output::Record {
66                fields: fields.into_iter().map(Output::into_preparation).collect(),
67                mode,
68            },
69            Output::Getter { function, output } => Output::Getter {
70                function,
71                output: Box::new(output.into_preparation()),
72            },
73            Output::Into { function, output } => Output::Into {
74                function,
75                output: Box::new(output.into_preparation()),
76            },
77        }
78    }
79}
