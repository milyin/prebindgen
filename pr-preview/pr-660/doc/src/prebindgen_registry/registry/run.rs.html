1//! Bind a finished registry to the generator that filled it.
2
3use super::*;
4
5impl Registry {
6    pub(super) fn apply_adapter_plans(
7        &mut self,
8        declared: &mut Declared,
9    ) -> Result<(), WriteRustError> {
10        // The set of declared fns drives `.default()` auto-apply: a defaulted
11        // constructor/deconstructor is synthesized for every matching declared
12        // fn. `accessors` is the `.fun_accessor` subset: excluded from
13        // constructor composition and the only fns a decomposer record may
14        // reference.
15        let d = &mut declared.decompositions;
16        if let Some(exp) = &d.expansions {
17            crate::expand::apply(
18                self,
19                exp,
20                &declared.functions,
21                &declared.accessors,
22                &declared.method_receivers,
23            )?;
24        }
25        // What the adapter's decompositions asked the output tables for,
26        // replayed in the order it asked. Building the plans reads the model
27        // alone (see `unfold::Unfolding`), so this is every effect they have on
28        // a registry — after `expand::apply`, which is where they were made.
29        for requirement in &d.requirements {
30            match requirement {
31                crate::unfold::Requirement::Output(reading) => self.require_output(reading),
32                crate::unfold::Requirement::Reference(reading) => self.reference_output(reading),
33                crate::unfold::Requirement::Unrequire(reading) => self.unrequire_output(reading),
34            }
35        }
36        // Every crossing these types make is now covered by a plan, so the
37        // scan-time direct converter requirement is stale — and typically
38        // unresolvable, since such a type has no destination representation.
39        // Drop it both ways; the cell stays, so a converter is still produced
40        // if one happens to resolve.
41        for key in declared.decompositions.replaces.keys() {
42            // The key is what a root flag is stored under, so it goes straight
43            // in — no `to_type()` round trip to be re-keyed on the far side.
44            self.clear_root(Direction::Construct, key);
45            self.clear_root(Direction::Deconstruct, key);
46        }
47        Ok(())
48    }
49}
