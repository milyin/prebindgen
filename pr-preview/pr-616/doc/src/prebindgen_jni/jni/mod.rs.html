57/// Short name of the generated `@RequiresOptIn` marker annotation class.
58///
59/// The Kotlin half of the raw-pointer guard: it marks every generated entry
60/// point that hands a raw native pointer to, or takes one from, safe Kotlin
61/// and cannot be hidden outright — `NativeHandle.peek()` and the `fromParts`
62/// factories, both reached from Rust by JNI reflection, so renaming them is
63/// not an option. [`JVM_SYNTHETIC`] is the other half, for Java.
64///
65/// See [`Declarations::unsafe_marker_fqn`] for the qualified form.
66pub(crate) const UNSAFE_MARKER: &str = "UnsafeNativeApi";
67
68/// `kotlin.jvm.JvmSynthetic` — the JVM half of the raw-pointer guard.
69///
70/// `internal` and `@RequiresOptIn` are Kotlin-source constructs and stop at
71/// the Kotlin compiler: `internal` members are emitted as **public** JVM
72/// methods under a mangled name, and javac neither knows nor enforces an
73/// opt-in marker. `@JvmSynthetic` sets `ACC_SYNTHETIC`, which javac skips when
74/// resolving a call, so a Java consumer cannot name the member at all.
75///
76/// It leaves the name and the JVM signature alone, so JNI's own lookup
77/// (`GetMethodID` / `GetStaticMethodID`, which ignore the flag) still finds
78/// `peek`, `fromParts`, and the `external fun` externs. That is exactly what
79/// `internal` could not offer. It is **not** applicable to a constructor
80/// (Kotlin rejects the target), which is why a handle's constructor is instead
81/// `private` behind a synthetic factory — see [`HANDLE_FACTORY`].
82pub(crate) const JVM_SYNTHETIC: &str = "JvmSynthetic";
83
84/// Name of the generated per-handle factory that replaces the raw-pointer
85/// constructor: `Storage.fromRawPtr(p)` rather than `Storage(p)`.
86///
87/// A handle's constructor is `private` — `internal` would still be a public
88/// JVM constructor and `@JvmSynthetic` cannot be applied to one, so `new
89/// Storage(0xdeadbeefL)` compiled fine from Java. The factory is `internal` +
90/// [`JVM_SYNTHETIC`], reachable from generated Kotlin and from nowhere else.
91pub(crate) const HANDLE_FACTORY: &str = "fromRawPtr";
92
93/// Spell the construction of a typed handle from a raw pointer expression.
94///
95/// Every generated site that mints a handle goes through here, so the choice
96/// of [`HANDLE_FACTORY`] over a constructor is made once.
97pub(crate) fn handle_from_raw(short: &str, raw: &str) -> String {
98    format!("{short}.{HANDLE_FACTORY}({raw})")

101/// An `internal` member of the generated surface, hidden from Java too.
102///
103/// Always pair the two: `internal` alone leaves a public JVM method behind a
104/// mangled name, and `handle.setPtr$mymodule(0xdeadbeefL)` from Java would
105/// then repoint a live handle at an address of the caller's choosing, which
106/// the next generated call happily frees. See [`JVM_SYNTHETIC`].
107pub(crate) fn internal_fun(name: &str) -> kotlin_codegen::KtFun {
108    kotlin_codegen::KtFun::new(name)
109        .vis(kotlin_codegen::KtVis::Internal)
110        .annotation(JVM_SYNTHETIC)
111}

113/// Use-site targets for [`JVM_SYNTHETIC`] on an `internal` property: a bare
114/// annotation would land on the backing field and leave the accessors — the
115/// part javac actually resolves — visible.
116pub(crate) fn internal_prop(p: kotlin_codegen::KtProperty) -> kotlin_codegen::KtProperty {
117    p.vis(kotlin_codegen::KtVis::Internal)
118        .annotation(format!("get:{JVM_SYNTHETIC}"))
119        .annotation(format!("set:{JVM_SYNTHETIC}"))
120}

122/// [`internal_prop`] for a `val`, which has no setter to hide.
123pub(crate) fn internal_val(p: kotlin_codegen::KtProperty) -> kotlin_codegen::KtProperty {
124    p.vis(kotlin_codegen::KtVis::Internal)

132/// Options of a type declared with `ptr_class!` — the payload of
133/// [`DeclaredKind::Ptr`], which is what marks the type as an opaque handle.
134/// The unified Kotlin emitter writes a typed-handle `.kt` file (and the Rust
135/// side its `freePtr` destructor) for every such type.
136///
137/// The typed-handle Kotlin FQN (e.g. `"io.zenoh.jni.JNISession"`) lives
138/// in the surrounding [`TypeConfig::name_spec`] slot — FQN-consumers
139/// (typed-handle class emission, `instanceof` dispatch,
140/// return-value constructor wrap) read it from there. The
141/// value-context Kotlin name for the same type (`"Long"`) is produced
142/// independently by the rank-0 opaque handler in [`KotlinMeta`], so
143/// the two roles don't collide despite sharing the `TypeConfig`.
144#[derive(Clone, Default)]
145pub(crate) struct OpaqueConfig {
146    /// `ptr_class!(X).gc_managed()`: the typed handle stores its pointer in
147    /// a separate atomic cell and registers a `Cleaner` action that frees the
148    /// native box if no other release path (close/take/consumption) won the
149    /// untagged→tagged CAS ticket first.
150    pub gc_managed: bool,
151}
152
153/// Options of a type declared with `enum_class!` — the payload of
154/// [`DeclaredKind::Enum`], which is what marks a `#[prebindgen]`-scanned
155/// `enum` as a Kotlin enum class. There are none yet: the declaration itself
156/// carries all the information, so this is the empty slot future enum options
157/// go in.
158///
159/// The rank-0 converter arms emit `jint ↔ Rust enum` bodies (via
160/// `TryFrom<i32>` for input and `as jni::sys::jint` for output), and the
161/// Kotlin emitter writes an `enum class` file with SCREAMING_SNAKE_CASE
162/// variants and a discriminant-keyed `fromInt(...)` companion. The Kotlin FQN
163/// lives in the surrounding [`TypeConfig::name_spec`] slot, same as
164/// [`OpaqueConfig`].
165#[derive(Clone, Default)]
166pub(crate) struct EnumConfig {}
167
168/// Per-variant naming of a type declared with `sealed_class!` — the payload
169/// of [`DeclaredKind::Sealed`], which is what marks a `#[prebindgen]`
170/// **data-carrying** enum as mirrored by a
171/// Kotlin `sealed interface`. The tag/leaf-group structure itself is read
172/// from the model's [`Variant`](prebindgen_registry::flat::Variant) — its
173/// `alternatives` in declaration order, indexed as they are tagged — and only
174/// what the declaration adds lives here.
175#[derive(Clone, Default)]
176pub(crate) struct SumConfig {
177    /// Per-variant Kotlin class-name overrides, keyed by the Rust variant
178    /// ident. Undeclared variants keep their Rust ident.
179    pub variant_names: HashMap<String, String>,
180}
181
182/// One registered package-level `.fun(...)` entry. The Rust identifier is captured
183/// at build-script time via `syn::parse_quote` (i.e. `pq!(rust_fn_name)`); the
184/// optional override sets the Kotlin-side name when the default
185/// `snake_to_camel(rust_ident)` derivation isn't what the user wants.
186#[derive(Clone, Debug)]
187pub struct FunctionEntry {
188    /// Rust function ident — must match a `#[prebindgen]`-marked free
189    /// function in the registered source module. Looked up by
190    /// `registry.flat().function(ident)`.
191    pub rust_ident: syn::Ident,
192    /// Kotlin-side name override, set by chaining `.name("...")` after
193    /// the entry's registration. `None` = derive from `rust_ident` via
194    /// `snake_to_camel`, then apply the target package's function hook.
195    pub kotlin_name_override: Option<String>,
196}

198impl FunctionEntry {
199    pub fn new(rust_ident: syn::Ident) -> Self {
200        Self {
201            rust_ident,
202            kotlin_name_override: None,
203        }
204    }

207/// Which of the five class declarators a type is registered under, **with
208/// that declarator's own options**. A type gets exactly one — this is a sum,
209/// not a set of independent markers: two declarators would emit two Kotlin
210/// declarations for the same FQN, so the second is a hard error (see
211/// [`DeclaredKind::merge`]). Reopening the *same* declarator stays legal —
212/// `merge` folds the incoming payload into the stored one.
213///
214/// Adding a sixth class kind is one variant here plus its emitter: there is
215/// no flag to add and no precedence chain to extend, because every consumer
216/// reads this one field (via [`TypeConfig`]'s accessors, [`JniGenBuilder::type_kind`],
217/// or a direct match).
218#[derive(Clone)]
219pub(crate) enum DeclaredKind {
220    /// `ptr_class!` — an opaque-handle type: jlong wire,
221    /// `Box::into_raw`/`Box::from_raw` conventions, instanceof dispatch, and
222    /// Kotlin typed-handle class emission.
223    Ptr(OpaqueConfig),
224    /// `enum_class!` — a `#[prebindgen]` fieldless enum mirrored as a Kotlin
225    /// `enum class`: jint wire (input + output via `TryFrom<i32>` / `as
226    /// jint`) and a generated `.kt` file.
227    Enum(EnumConfig),
228    /// `sealed_class!` — a `#[prebindgen]` data-carrying enum mirrored as a
229    /// Kotlin `sealed interface`: a tag plus one leaf group per variant.
230    Sealed(SumConfig),
231    /// `data_class!` — a `#[prebindgen]` struct mirrored as a Kotlin `data
232    /// class`, flattened field-by-field at the boundary. The kind with no
233    /// options of its own.
234    Data,
235}

244/// All configuration the structured builder accumulates for one
245/// canonical Rust type key. The declared kind is mandatory (an entry exists
246/// only because some declarator created it); the remaining, cross-kind
247/// options default to unset and are populated by the decl that carries them.
248#[derive(Clone)]
249pub(crate) struct TypeConfig {
250    /// The class declarator this type is registered under, carrying that
251    /// kind's own options. Every entry in [`JniGenBuilder::types`] has one: entries
252    /// are created only by a class declarator (see
253    /// `JniGenBuilder::register_class`), which is why presence in the table *is*
254    /// "declared as a class" — declared classes are required in **both**
255    /// directions at scan (their converters always resolve both ways),
256    /// unlike a wrapper registration, which is required per **usage**
257    /// direction.
258    pub kind: DeclaredKind,
259    /// The type this declaration was **written with**, e.g. the `Foo` in
260    /// `ptr_class!(Foo)`.
261    ///
262    /// A class declarator receives a real `syn::Type` and used to keep only the
263    /// key derived from it, so every later site that needed the tokens back had
264    /// to ask the key for them. That is the wrong direction: the declaration is
265    /// where the type came from, and this is where it stays (#291).
266    pub rust_type: Origin<syn::Type>,
267    /// Raw naming spec of the type as declared — verbatim Kotlin type or
268    /// settings-derived class name. Required for any type emitted in
269    /// Kotlin; the concrete FQN (`Sample` → `"io.zenoh.jni.Sample"`,
270    /// `Vec<u8>` → `"ByteArray"`) is materialized only at read time via
271    /// [`JniGenBuilder::fqn_of`], which is what makes the `set_*` settings
272    /// order-independent w.r.t. declarations.
273    pub name_spec: Option<NameSpec>,
274    /// Explicit opt-in for a `data_class` to cross Kotlin → Rust as one
275    /// `JObject`. Unmarked data classes are required to flatten completely;
276    /// this flag is sticky across reopened declarations.
277    pub jobject_input: bool,
278    /// Emit the generated interface mirroring the class's public instance
279    /// surface, and make the class implement it with `override` on every
280    /// class-body member (`.interface()` / implied by `.interface_name()`).
281    pub interface_enabled: bool,
282    /// Per-decl literal interface name (`.interface_name(...)`, relative, no
283    /// dots) — bypasses the `set_interface_name_mangle` hook.
284    pub interface_name_override: Option<String>,
285    /// Kotlin interfaces added to the generated class's supertype list
286    /// (`.implements(...)`, any class kind) — the class implements them
287    /// nominally; the class body and lifecycle members are unaffected.
288    /// Orthogonal to [`Self::interface_enabled`].
289    pub interfaces: Vec<String>,
290}

292impl TypeConfig {
293    /// A freshly declared type: the declarator's kind and naming spec, every
294    /// cross-kind option unset. Reopening the same declarator goes through
295    /// [`DeclaredKind::merge`] instead.
296    pub(crate) fn new(
297        kind: DeclaredKind,
298        name_spec: NameSpec,
299        rust_type: Origin<syn::Type>,
300    ) -> Self {
301        Self {
302            kind,
303            rust_type,
304            name_spec: Some(name_spec),
305            jobject_input: false,
306            interface_enabled: false,
307            interface_name_override: None,
308            interfaces: Vec::new(),
309        }
310    }
311
312    /// The opaque-handle options if this type is a `ptr_class`, else `None`.
313    pub(crate) fn opaque(&self) -> Option<&OpaqueConfig> {
314        match &self.kind {
315            DeclaredKind::Ptr(o) => Some(o),
316            _ => None,
317        }
318    }
319
320    /// `true` if this type is a `ptr_class`-declared opaque handle.
321    pub(crate) fn is_opaque(&self) -> bool {
322        matches!(self.kind, DeclaredKind::Ptr(_))
323    }
324
325    /// `true` if this type is an `enum_class`-declared Kotlin `enum class`.
326    pub(crate) fn is_enum_class(&self) -> bool {
327        matches!(self.kind, DeclaredKind::Enum(_))
328    }
329
330    /// The sum options if this type is a `sealed_class`, else `None`.
331    pub(crate) fn sum(&self) -> Option<&SumConfig> {
332        match &self.kind {
333            DeclaredKind::Sealed(s) => Some(s),
334            _ => None,
335        }
336    }
337}

382/// One `#[prebindgen]` function attached to a declared class (`ptr_class` /
383/// `data_class`) via a declaration's `.method(...)` /
384/// `.constructor(...)`. Methods become **instance methods** (receiver
385/// dropped→`this`); constructors become **companion factory** methods. Each
386/// is also a real `#[prebindgen]` wrapper (Rust extern + `JNINative` extern +
387/// JSONL).

389pub(crate) struct ClassMember {

392    /// Per-member `.name()` override, stored RAW — the effective Kotlin

395    /// camelCase ident), keeping `set_method_name_mangle` order-independent. An
396    /// `expand_return!` `.field` referencing the same underlying function
397    /// inherits the effective name unless it sets its own `.name()`;
398    /// `expand_param!` variants reference the fn by ident only.
399    pub kotlin_name_override: Option<String>,
400    /// Member kind (method / constructor).
401    pub kind: MemberKind,
402}

420/// JNI back-end. Global settings are applied with the order-insensitive
421/// `set_*` methods; declarations are accepted as pre-built objects
422/// (`PackageDecl`, `ExpandParamDecl`, `ExpandReturnDecl`,
423/// `ConvertDecl` — see `decl.rs`) built
424/// independently of `JniGenBuilder` itself; there is no fluent typestate cursor.
425///
426/// ```
427/// use prebindgen_jni::JniGenBuilder;
428///
429/// let jni = JniGenBuilder::new()
430///     .set_package_prefix("io.test.jni")
431///     .package(
432///         prebindgen_jni::package!("keyexpr")
433///             .class(prebindgen_jni::ptr_class!(KeyExpr)
434///                 .method(prebindgen_registry::fun!(keyexpr_get_str).name("getStr"))
435///                 .constructor(prebindgen_registry::fun!(keyexpr_new_try_from).name("tryFrom"))),
436///     )
437///     // A KeyExpr param accepts EITHER a String (built via tryFrom) OR an
438///     // existing handle; a returned KeyExpr decomposes into its string form.
439///     .expand(
440///         prebindgen_registry::expand_param!(KeyExpr)
441///             .variant(prebindgen_registry::fun!(keyexpr_new_try_from))
442///             .variant_self(),
443///     )
444///     .expand(prebindgen_registry::expand_return!(KeyExpr).field(prebindgen_registry::fun!(keyexpr_get_str)));
445/// ```
446/// A resolved JNI binding: every crossing has a conversion, and the artifacts
447/// can be written.
448///
449/// Built by [`JniGenBuilder::build`]. Read-only — the registry inside it is
450/// complete, which is what lets every `write_*` be a pure emission that can run
451/// in any order, or not at all.
452pub struct JniGen {
453    /// What the binding declared. The emitters read it for names, classes and
454    /// decompositions.
455    ///
456    /// A [`Declarations`], not the [`JniGenBuilder`] it came from: the builder's
457    /// mutators would otherwise ride into the finished object, and "finished"
458    /// would be a comment rather than a type.
459    decls: Declarations,
460    /// Every crossing this binding needs, each with its conversion.
461    registry: prebindgen_registry::Registry,
462}

480/// How a reach renders: its field steps, joined — the accessor call every leaf
481/// hangs off is the site's, not the value's, so it is dropped.
482#[cfg(test)]
483fn reach_of(path: &[prebindgen_registry::unfold::PathStep]) -> String {
484    use prebindgen_registry::unfold::PathStep;
485    path.iter()
486        .filter_map(|step| match step {
487            PathStep::Field { ident, .. } => Some(ident.to_string()),
488            PathStep::Call { .. } => None,
489        })
490        .collect::<Vec<_>>()
491        .join(".")
492}
493
494/// How the Rust side reaches one plan leaf, in the form a wire renders it —
495/// the accessor call every leaf hangs off is the site's, not the value's, so it
496/// is dropped to line the two up.
497#[cfg(test)]
498fn leaf_reach(leaf: &prebindgen_registry::unfold::UnfoldLeaf) -> String {
499    use prebindgen_registry::unfold::LeafSource;
500    match &leaf.source {
501        LeafSource::SumTag => "tag".to_string(),
502        LeafSource::VariantField { variant, member } => format!(
503            "{variant}.{}",
504            crate::jni::struct_plan::sum_field_prop_name(member)
505        ),
506        _ => reach_of(&leaf.path),
507    }
508}

535    fn render_plan_for_test(
536        &self,
537        ty: syn::Type,
538        direction: prebindgen_registry::recipe::Direction,
539        parts: bool,
540    ) -> Result<(bool, String, String), String> {
541        use prebindgen_registry::recipe::{Compiler, Crossing};
542
543        let reading = self
544            .registry
545            .flat()
546            .classify(&ty)
547            .map_err(|e| format!("the model does not classify {}: {e}", quote::quote!(#ty)))?;
548        let crossing = Crossing::new(reading, direction);
549        let mut compiler = Compiler::resume(
550            self.registry.flat(),
551            self.decls.recipe_table(),
552            self.decls.site_bindings(),
553            self.generation_plan().conversions().clone(),
554        );
555        let mut adapter = crate::jni::compile::JCompile {
556            decls: &self.decls,
557            registry: &self.registry,
558            declared_return: None,
559            site: None,
560        };
561        let fragment = if parts {
562            compiler.recipe_of(&mut adapter, &crossing, &crate::jni::recipes::parts())
563        } else {
564            compiler.crossing(&mut adapter, &crossing)
565        }
566        .map_err(|e| e.to_string())?;
567        fragment.rust.mark_reachable();
568        let rendered =
569            fragment
570                .rust
571                .render_fn(&prebindgen_registry::RustWriter::for_registry_test(
572                    &self.registry,
573                ));
574        Ok((
575            fragment.composed_only,
576            rendered.sig.ident.to_string(),
577            quote::quote!(#rendered).to_string(),
578        ))
579    }

893    /// Write the generated Rust file — the JNI externs and the converters they
894    /// call. `out_path` may be relative (resolved against `OUT_DIR`) or
895    /// absolute; returns the path actually written.
896    pub fn write_rust(
897        &self,
898        out_path: impl AsRef<std::path::Path>,
899    ) -> Result<std::path::PathBuf, prebindgen_registry::WriteRustError> {
900        // Every binding a test writes also checks that a data class's two
901        // leaf derivations agree wherever both exist — the property #603
902        // measured and #602 needs before it can unify them.
903        #[cfg(test)]
904        self.assert_leaf_derivations_agree();
905
906        // Every binding a test writes also checks that the assembly's
907        // dependency edges name every call its artifacts render — the
908        // completeness the emission-time check reasons from.
909        #[cfg(test)]
910        prebindgen_registry::write::assert_edges_cover_rendered_calls(
911            self.decls
912                .generation
913                .as_ref()
914                .expect("resolved JniGen has no frozen generation plan")
915                .assembly(),
916            "jni",
917        );
918        Ok(prebindgen_registry::write::write_rust(
919            self.decls
920                .generation
921                .as_ref()
922                .expect("resolved JniGen has no frozen generation plan")
923                .assembly(),
924            out_path,
925        )?)
926    }

928    /// A declared data class is decomposed twice — by `struct_out_wires_of`,
929    /// whose leaves the registry receives, and by `StructPlan`, whose leaves
930    /// the whole-object encode emits and Kotlin's `fromParts` declares. Where
931    /// both exist they must name the same leaves in the same order.
932    ///
933    /// #603 records why the two are not one derivation: the first refuses
934    /// shapes the second supports, so unifying them is the feature #602
935    /// proposes. Until then this is what says they still agree — the
936    /// precondition that unification rests on.
937    ///
938    /// Both are available only after resolve, which is why this runs here and
939    /// not while planning.
940    #[cfg(test)]
941    fn assert_leaf_derivations_agree(&self) {
942        let emit = prebindgen_registry::RustWriter::for_registry_test(&self.registry);
943        for item in self.registry.flat().types() {
944            let prebindgen_registry::flat::Type::Struct(item) = item else {
945                continue;
946            };
947            // Only a declared data class has a frozen plan; the lookup is
948            // panic-backed for anything else, and an opaque handle's struct
949            // decomposes without ever having one.
950            let declared = self
951                .decls
952                .types
953                .get(&item.type_ref().key())
954                .is_some_and(|cfg| !cfg.special_decl() && cfg.name_spec.is_some());
955            if !declared {
956                continue;
957            }
958            let Some(wires) = self.decls.struct_out_wires_of(&self.registry, &item.name) else {
959                continue;
960            };
961            let Some(plan) = self.decls.struct_plan(&self.registry, item, 0) else {
962                continue;
963            };
964            // The two spell a leaf for different audiences: the decomposition
965            // names it as Kotlin will see it (`maybeLong`), the encode as Rust
966            // wrote it (`maybe_long`), and a keyword-colliding name keeps a
967            // trailing underscore on one side only. The property here is which
968            // leaves, in which order — so the comparison drops the spelling.
969            // Two comparisons, because a leaf has two properties that must
970            // agree and one that need not. Its *name* is spelled for two
971            // audiences — the decomposition names it as Kotlin will see it
972            // (`maybeLong`), the encode as Rust wrote it (`maybe_long`), and a
973            // keyword-colliding name keeps a trailing underscore on one side —
974            // so the comparison drops the spelling. Its *nesting* is structure:
975            // the decomposition joins an inlined class's path with the reserved
976            // `__` separator, and the encode counts the same inlining as depth.
977            let plain = |name: &str| name.replace('_', "").to_lowercase();
978            let decomposed: Vec<String> = wires.iter().map(|wire| plain(&wire.name)).collect();
979            let decomposed_depth: Vec<usize> = wires
980                .iter()
981                .map(|wire| wire.name.matches("__").count())
982                .collect();
983            let leaves = crate::jni::emit::encode_leaves(&plan, &emit);
984            let encoded: Vec<String> = leaves.iter().map(|(name, _)| plain(name)).collect();
985            let encoded_depth: Vec<usize> = leaves.iter().map(|(_, depth)| *depth).collect();
986            assert_eq!(
987                encoded, decomposed,
988                "`{}`: the whole-object encode and the registry-facing \
989                 decomposition name different leaves",
990                item.name
991            );
992            assert_eq!(
993                encoded_depth, decomposed_depth,
994                "`{}`: the whole-object encode and the registry-facing \
995                 decomposition flatten nesting differently",
996                item.name
997            );
998        }
999    }

1021/// Everything a binding **declared**, once it is done declaring.
1022///
1023/// The read-only half of what used to be one `JniGenBuilder`. Splitting it is what
1024/// keeps the phase separation now that a generator owns its own registry: before
1025/// [#253](https://github.com/milyin/prebindgen/pull/253) a build script held a
1026/// `RegistryBuilder` and then a `Registry`, and that type split *was* the
1027/// enforcement. Once `JniGen::builder().source(..).build()` moved both inside the
1028/// generator, the built object was left holding the builder — mutators and all.
1029///
1030/// So the mutators live on [`JniGenBuilder`] and nothing else, and this type — the
1031/// one a [`JniGen`] keeps and every emitter reads — **has no `&mut self` method at
1032/// all**. Not "the obvious ones were removed": none, which
1033/// `a_built_jnigen_exposes_no_mutation` checks by reading the source, because a
1034/// one-line grep once missed a multi-line signature and let `Registry::supply`
1035/// survive two commits that claimed it was gone.

1073    /// Structured per-type configuration keyed by canonical Rust type.
1074    /// One entry per declared class; populated when accepting a `ClassDecl`,
1075    /// through the table's single writer `JniGenBuilder::register_class` — so
1076    /// presence here *is* "declared as a class", and each entry's
1077    /// [`TypeConfig::kind`] is the one representation of which declarator it

1246/// Describe a JNI binding: state the Kotlin surface, then [`build`](Self::build).
1247///
1248/// Holds the [`Declarations`] being filled and the sources to parse, and it is the
1249/// only type with mutators. `build` consumes it, hands the declarations to the
1250/// registry, and stores them in a [`JniGen`] — from where nothing can declare
1251/// anything again, because this type is gone by then.
1252#[derive(Clone, Default)]
1253pub struct JniGenBuilder {
1254    /// What has been declared so far. Moved out whole by [`Self::build`].
1255    pub(crate) decls: Declarations,
1256
1257    /// Where the `#[prebindgen]` items come from.
1258    ///
1259    /// A [`FlatBuilder`](prebindgen_registry::flat::FlatBuilder), stated with the same
1260    /// three feeders it has — so a build script says where the source is in the
1261    /// vocabulary the model already uses, and never names a `Flat` or a
1262    /// `Registry` itself.
1263    ///
1264    /// Not a declaration, which is why it stays here rather than moving across:
1265    /// it is *input to* building, and keeping it out is what makes
1266    /// [`Declarations`] mean one thing.
1267    pub(crate) sources: prebindgen_registry::flat::FlatBuilder,
1268}
