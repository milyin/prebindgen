1//! Constructors and fields for generated immutable JVM output rows.
2use crate::{jni, Result};
3use jni::{
4    objects::{GlobalRef, JClass, JFieldID, JMethodID, JObject},
5    JNIEnv,
6};
7use std::sync::OnceLock;
8
9/// One generated row schema, registered from Kotlin before native calls begin.
10/// Passing the actual class avoids application-class lookups on native workers.
11#[derive(Default)]
12pub struct OutputRowType(OnceLock<OutputRowConstructor>);
13/// A pinned row class with constructor and raw-field IDs resolved at registration.
14/// Borrow once per collection; constructing rows performs no metadata lookup.
15pub struct OutputRowConstructor {
16    class: GlobalRef,
17    constructor: JMethodID,
18    signature: Box<str>,
19    fields: Box<[JFieldID]>,
20}
21impl OutputRowType {
22    /// Create an unregistered generated row type.
23    pub const fn new() -> Self {
24        Self(OnceLock::new())
25    }
26    /// Register the generated class's raw constructor and `c0`, `c1`, … fields.
27    /// JVM descriptors must follow the constructor's field order.
28    pub fn initialize(
29        &self,
30        env: &mut JNIEnv<'_>,
31        class: &JClass<'_>,
32        descriptors: &[&str],
33    ) -> Result<()> {
34        let signature = format!("({})V", descriptors.concat());
35        if let Some(previous) = self.0.get() {
36            return previous.check_registration(env, class, &signature);
37        }
38        let constructor = env
39            .get_method_id(class, "<init>", &signature)
40            .map_err(|e| e.to_string())?;
41        let fields = descriptors
42            .iter()
43            .enumerate()
44            .map(|(i, descriptor)| {
45                env.get_field_id(class, format!("c{i}"), descriptor)
46                    .map_err(|e| e.to_string())
47            })
48            .collect::<Result<Vec<_>>>()?
49            .into_boxed_slice();
50        let class = env.new_global_ref(class).map_err(|e| e.to_string())?;
51        if let Err(candidate) = self.0.set(OutputRowConstructor {
52            class,
53            constructor,
54            signature: signature.into_boxed_str(),
55            fields,
56        }) {
57            // Another initializer may have won while IDs were being resolved.
58            // Accept only the same schema and class, never silently another loader.
59            let class: &JClass<'_> = candidate.class.as_obj().into();
60            self.0
61                .get()
62                .unwrap()
63                .check_registration(env, class, &candidate.signature)?;
64        }
65        Ok(())
66    }
67    /// Borrow registered metadata before converting a collection.
68    pub fn get(&self) -> Result<&OutputRowConstructor> {
69        self.0
70            .get()
71            .ok_or_else(|| "output row type is not initialized".into())
72    }
73}
74impl OutputRowConstructor {
75    fn check_registration(
76        &self,
77        env: &mut JNIEnv<'_>,
78        class: &JClass<'_>,
79        signature: &str,
80    ) -> Result<()> {
81        if self.signature.as_ref() != signature
82            || !env
83                .is_same_object(self.class.as_obj(), class)
84                .map_err(|e| e.to_string())?
85        {
86            return Err("output row type registered with a different class or schema".into());
87        }
88        Ok(())
89    }
90    /// Allocate a row using already prepared raw JVM arguments.
91    ///
92    /// # Safety
93    /// Arguments must match the registered constructor descriptors and stay live
94    /// through this invocation. Construction does not transfer native ownership.
95    #[inline]
96    pub unsafe fn construct<'local>(
97        &self,
98        env: &mut JNIEnv<'local>,
99        arguments: &[jni::sys::jvalue],
100    ) -> Result<JObject<'local>> {
101        let class: &JClass<'_> = self.class.as_obj().into();
102        env.new_object_unchecked(class, self.constructor, arguments)
103            .map_err(|e| e.to_string())
104    }
105    /// Field ID for a trusted generated row's raw field, used on rollback.
106    pub fn field(&self, index: usize) -> JFieldID {
107        self.fields[index]
108    }
109}
