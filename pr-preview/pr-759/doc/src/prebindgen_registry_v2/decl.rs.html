1//! What a binding declared, in terms neither language owns.
2//!
3//! A frontend turns its own declaration storage into these — one
4//! [`Declaration`] per thing the user asked for, inside the
5//! [`BindingRequests`](crate::BindingRequests) it hands the engine. The
6//! declaration is what the report accounts for; the request carries the target's
7//! configuration for it.
8
9use prebindgen_flat::flat::{Element, Flat};
10use serde::Serialize;
11
12/// What a [`Declaration`] declares: a function, a type, a constant, a callback
13/// or a conversion.
14///
15/// This is the coarse, language-neutral category — the five things any binding
16/// can be made of. It is deliberately coarser than an adapter's own vocabulary:
17/// `prebindgen-c` declares a type with `opaque_ptr` or `data_struct`, and
18/// `prebindgen-jni` with `ptr_class` or `data_class`, but all four produce a
19/// `Type`. The word the adapter used is kept separately in
20/// [`Declaration::representation`] and printed back verbatim.
21///
22/// Three things depend on the category:
23///
24/// * **What the engine plans.** [`generate`](crate::generate) dispatches on it,
25///   one lowering per kind: a `Function` is planned as an exported wrapper, a
26///   `Type` as a record.
27/// * **Identity.** A [`DeclarationId`] pairs the kind with the origin, and the
28///   kind is what keeps the origins' several naming spaces apart: an origin may
29///   be a captured item's name, a type key, a callback's signature or a name
30///   the binding coined, so `type:Foo` and `conversion:Foo` are two
31///   declarations about one Rust type.
32///   [`SurfaceSpec::requires`](crate::SurfaceSpec) says by id which of them a
33///   wrapper needs emitted.
34/// * **Report layout.** [`Report`](crate::Report) groups and sorts by it, so a
35///   report reads types first, then conversions, callbacks, constants and
36///   functions.
37///
38/// It says what the *target* language gets, not what the captured Rust source
39/// held — a Kotlin constant may be backed by a captured Rust function. That
40/// second question is [`Declaration::source`] / [`SourceKind`].
41#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize)]
42#[serde(rename_all = "snake_case")]
43pub enum DeclarationKind {
44    /// A `#[prebindgen]` function the binding exports a wrapper for.
45    Function,
46    /// A `#[prebindgen]` type the binding gives a foreign representation.
47    Type,
48    /// A `#[prebindgen]` constant the binding exposes as a foreign constant.
49    Const,
50    /// A callback signature the binding exports as a foreign callable.
51    Callback,
52    /// A declared conversion between a Rust type and its wire form.
53    Conversion,
54}
55
56impl DeclarationKind {
57    /// The id prefix and report label.
58    pub fn as_str(self) -> &'static str {
59        match self {
60            DeclarationKind::Function => "fn",
61            DeclarationKind::Type => "type",
62            DeclarationKind::Const => "const",
63            DeclarationKind::Callback => "callback",
64            DeclarationKind::Conversion => "conversion",
65        }
66    }
67}

69/// A declaration's stable identity: its [`DeclarationKind`] and the name its
70/// Rust origin goes by.
71///
72/// Stable across runs and across pipelines, so a report, a build script and a
73/// capability-selected test section can all name the same declaration. Derived from
74/// what the *source* calls the thing, not from what the target does — a rename
75/// on the foreign side must not silently retire a test's requirement.
76///
77/// The two parts are stored, readable ([`Self::kind`], [`Self::origin`]) and
78/// fixed at construction. `<kind>:<origin>` — `type:Stamp`, `fn:stamp_sum` —
79/// is how an id prints and how the report writes it, and that spelling is a
80/// rendering: nothing reads an id back out of it.
81#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
82pub struct DeclarationId {
83    kind: DeclarationKind,
84    origin: String,
85}
86
87impl DeclarationId {
88    /// The id of `origin` declared as `kind`.
89    pub fn new(kind: DeclarationKind, origin: impl AsRef<str>) -> Self {
90        DeclarationId {
91            kind,
92            origin: origin.as_ref().to_string(),
93        }
94    }
95
96    /// Which kind of declaration this names.
97    pub fn kind(&self) -> DeclarationKind {
98        self.kind
99    }
100
101    /// What the Rust source calls it — see [`Declaration::rust_origin`].
102    pub fn origin(&self) -> &str {
103        &self.origin
104    }
105}
106
107impl std::fmt::Display for DeclarationId {
108    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
109        write!(f, "{}:{}", self.kind.as_str(), self.origin)
110    }
111}
112
113impl Serialize for DeclarationId {
114    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
115        serializer.collect_str(self)
116    }
117}

119/// One declaration, as the report accounts for it.
120///
121/// Built by the adapter with [`Self::new`] and read back through the accessors.
122/// Its kind and origin live in the [`DeclarationId`], which the run is keyed
123/// by, so they are fixed once declared and read out of the id rather than
124/// stored beside it.
125#[derive(Clone, Debug, PartialEq, Eq)]
126pub struct Declaration {
127    id: DeclarationId,
128    placement: String,
129    representation: String,
130    source: SourceKind,
131}
132
133impl Serialize for Declaration {
134    /// Flat, and with the identity spelled out: the report carries `id`,
135    /// `kind` and `rust_origin` as separate columns, and a reader of the JSON
136    /// should not have to split the id to get at the last two.
137    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
138        use serde::ser::SerializeStruct;
139        let mut entry = serializer.serialize_struct("Declaration", 6)?;
140        entry.serialize_field("id", &self.id)?;
141        entry.serialize_field("kind", &self.id.kind())?;
142        entry.serialize_field("rust_origin", self.id.origin())?;
143        entry.serialize_field("placement", &self.placement)?;
144        entry.serialize_field("representation", &self.representation)?;
145        entry.serialize_field("source", &self.source)?;
146        entry.end()
147    }
148}

150/// What a declaration's Rust origin must name in the captured source.
151///
152/// Captured items live in one flat namespace holding functions, types and
153/// constants, and a declaration means one of the three. Naming the wrong one —
154/// `.fun(fun!(x))` where the source captured `const x` — is an error in the
155/// binding, and the engine fails the run over it instead of reporting a skip.
156#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
157#[serde(rename_all = "snake_case")]
158pub enum SourceKind {
159    /// A captured `#[prebindgen]` function.
160    Function,
161    /// A captured `#[prebindgen]` type.
162    Type,
163    /// A captured `#[prebindgen]` constant.
164    Const,
165    /// Nothing: the binding defines this itself. A callback signature, a
166    /// binding-local conversion helper, or a type the target represents
167    /// although the source never exported it (`String` as an opaque handle).
168    BindingLocal,
169}
170
171impl SourceKind {
172    /// The captured element an origin of this kind names, if the model holds it.
173    ///
174    /// The namespace is flat, so the name alone finds any element; the kind is
175    /// what says whether the element found is the one the declaration meant.
176    /// [`Self::BindingLocal`] names no captured item and so finds none — which
177    /// is not the same as a missing one, and [`Self::missing_from`] is the
178    /// question to ask about presence.
179    pub(crate) fn element<'f>(self, flat: &'f Flat, name: &str) -> Option<&'f Element> {
180        let element = flat.element(name)?;
181        matches!(
182            (self, element),
183            (SourceKind::Function, Element::Function(_))
184                | (SourceKind::Type, Element::Type(_))
185                | (SourceKind::Const, Element::Constant(_))
186        )
187        .then_some(element)
188    }
189
190    /// Whether the model lacks what an origin of this kind must name.
191    ///
192    /// False for a binding-local origin, which requires nothing of the model.
193    pub(crate) fn missing_from(self, flat: &Flat, name: &str) -> bool {
194        self != SourceKind::BindingLocal && self.element(flat, name).is_none()
195    }
196
197    /// The word a refusal uses for this kind.
198    pub(crate) fn describe(self) -> &'static str {
199        match self {
200            SourceKind::Function => "function",
201            SourceKind::Type => "type",
202            SourceKind::Const => "constant",
203            SourceKind::BindingLocal => "binding-local item",
204        }
205    }
206}

208impl Declaration {
209    /// Declare `rust_origin` as a `kind` the target places at `placement`.
210    pub fn new(
211        kind: DeclarationKind,
212        rust_origin: impl Into<String>,
213        placement: impl Into<String>,
214        representation: impl Into<String>,
215    ) -> Self {
216        Declaration {
217            id: DeclarationId::new(kind, rust_origin.into()),
218            placement: placement.into(),
219            representation: representation.into(),
220            // The usual case: the declaration is named after the item it is built
221            // from. `sourced_as` states the exceptions.
222            source: match kind {
223                DeclarationKind::Function => SourceKind::Function,
224                DeclarationKind::Type => SourceKind::Type,
225                DeclarationKind::Const => SourceKind::Const,
226                DeclarationKind::Callback | DeclarationKind::Conversion => SourceKind::BindingLocal,
227            },
228        }
229    }
230
231    /// The same, for something the binding defines itself rather than something
232    /// it selects out of the captured source — see [`SourceKind::BindingLocal`].
233    pub fn local(self) -> Self {
234        self.sourced_as(SourceKind::BindingLocal)
235    }
236
237    /// The same, for a declaration whose target kind and source kind differ — see
238    /// [`Self::source`].
239    pub fn sourced_as(mut self, source: SourceKind) -> Self {
240        self.source = source;
241        self
242    }
243
244    /// Stable identity — see [`DeclarationId`].
245    pub fn id(&self) -> &DeclarationId {
246        &self.id
247    }
248
249    /// Which kind of declaration it is.
250    pub fn kind(&self) -> DeclarationKind {
251        self.id.kind()
252    }
253
254    /// What the Rust source calls it (`Calculator`, `calculator_new`), or the
255    /// signature for a callback that has no name of its own.
256    pub fn rust_origin(&self) -> &str {
257        self.id.origin()
258    }
259
260    /// Where it is meant to land in the target language, spelled the way that
261    /// language spells it: `calculator_t`, `io.zenoh.jni.Session`.
262    pub fn placement(&self) -> &str {
263        &self.placement
264    }
265
266    /// The declarator that produced it (`opaque_ptr`, `data_class`, `fun`, …) —
267    /// the adapter's own word, printed back verbatim.
268    pub fn representation(&self) -> &str {
269        &self.representation
270    }
271
272    /// What [`Self::rust_origin`] must name in the captured source.
273    ///
274    /// Not the same question as [`Self::kind`], which says what the *target*
275    /// gets: a Kotlin `val` declared with `constant!(X).fun(fun!(f))` is a
276    /// [`DeclarationKind::Const`] backed by a captured **function**. Stated by
277    /// the adapter, because only the adapter knows which of its own
278    /// declaration forms produced this one.
279    pub fn source(&self) -> SourceKind {
280        self.source
281    }
282}
