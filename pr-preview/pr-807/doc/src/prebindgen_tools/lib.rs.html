1//! Conversion composition. Adapters supply leaf codecs; wrappers consume conversion slots.
2//!
3//! [`Declarations::fill`], [`bind_input`] and [`bind_output`] build conversions
4//! directly in a [`Bindings`] model. Recipes and codec input data are used only
5//! during construction; slots retain wires and code needed by wrapper writers.
6//! Before emission, call [`validate_bindings`] and follow the execution ordering
7//! documented on [`InputConversion`] and [`OutputConversion`].
8//! Each [`Wire`] carries the adapter's own description in [`Wire::kind`].
9//! [`value`] provides conversions for values copied across the boundary and
10//! [`boxed`] for values kept in Rust-owned allocations.
11#![warn(missing_docs)]
12use prebindgen_flat::{same_type, type_key, Flat, Result};
13use proc_macro2::TokenStream;
14use quote::{format_ident, quote};
15use syn::Type;

17/// A recipe for converting boundary inputs into one source argument.
18///
19/// Binding expands the recipe into wires and Rust conversion code.
20#[derive(Clone, Debug)]
21pub enum Input {
22    /// Convert the occurrence type itself with [`Codec::input_leaf`].
23    Leaf,
24    /// Construct a named record from children in field order; an empty list builds unit.
25    Fields(Vec<Input>),
26    /// Call a source function after preparing its arguments with the child recipes.
27    Constructor {
28        /// Unqualified source function name; its result must match the occurrence type.
29        function: String,
30        /// Recipes in the constructor's parameter order.
31        args: Vec<Input>,
32    },
33    /// Choose an owned input using an `i32` arm index followed by all arms' wires.
34    /// Inactive arms are neither validated nor acquired; indices are zero-based.
35    Selector(Vec<Input>),
36}
37impl Input {
38    /// Construct a record from field recipes, or unit from an empty list.
39    ///
40    /// `fields` holds one recipe per field of the source record, in declaration
41    /// order. Binding verifies the count against the record.
42    pub fn fields(fields: impl IntoIterator<Item = Input>) -> Self {
43        Self::Fields(fields.into_iter().collect())
44    }
45    /// Call a named source constructor with the supplied argument recipes.
46    ///
47    /// Binding checks its result type and argument count. Argument validation and
48    /// ownership acquisition precede the source call.
49    ///
50    /// # Arguments
51    ///
52    /// * `name` — unqualified name of a source function whose result type is the
53    ///   input's type.
54    /// * `args` — one recipe per parameter of that function, in order.
55    pub fn constructor(name: impl Into<String>, args: impl IntoIterator<Item = Input>) -> Self {
56        Self::Constructor {
57            function: name.into(),
58            args: args.into_iter().collect(),
59        }
60    }
61    /// Choose between owned representations of the same input type.
62    ///
63    /// `arms` holds one recipe per alternative; each must produce the input's
64    /// type. The caller passes the zero-based index of the chosen alternative in an
65    /// extra wire ahead of all arms' wires.
66    ///
67    /// Requires at least one arm. Each arm retains its own wires; this does not
68    /// merge equivalent wire parameters or generate destination-language overloads.
69    pub fn selector(arms: impl IntoIterator<Item = Input>) -> Self {
70        Self::Selector(arms.into_iter().collect())
71    }
72}
73/// A result conversion recipe, interpreted together with its source type and codec input data.
74///
75/// Field projections are observations rather than moves. Adapters decide how to
76/// copy or otherwise expose a foreign value observed through a shared reference.
77#[derive(Clone, Debug)]
78pub enum Output {
79    /// Convert the occurrence type itself with [`Codec::output_leaf`].
80    Leaf,
81    /// Observe named fields in declaration order; an empty list represents unit.
82    Fields(Vec<Output>),
83    /// A shared-borrow accessor, followed by the output conversion of its result.
84    Getter {
85        /// Source getter taking exactly one shared reference to the current type.
86        function: String,
87        /// Conversion of the getter's owned result; borrowed results are unsupported.
88        output: Box<Output>,
89    },
90    /// Produce several views of one value. At most one child may consume it.
91    /// Observations run before consumption, while wire order follows this list.
92    Combine(Vec<Output>),
93}
94impl Output {
95    /// Observe record fields in declaration order; use an empty list for unit.
96    ///
97    /// `fields` holds one recipe per field of the source record, in declaration order.
98    pub fn fields(fields: impl IntoIterator<Item = Output>) -> Self {
99        Self::Fields(fields.into_iter().collect())
100    }
101    /// Observe a value through a source getter and convert the getter's result.
102    ///
103    /// The getter must take exactly `&T` for the current type `T`, and return an
104    /// owned value. Binding rejects mismatched signatures.
105    ///
106    /// # Arguments
107    ///
108    /// * `name` — unqualified name of the source getter.
109    /// * `output` — recipe for the getter's result.
110    pub fn getter(name: impl Into<String>, output: Output) -> Self {
111        Self::Getter {
112            function: name.into(),
113            output: Box::new(output),
114        }
115    }
116    /// Produce multiple outputs from one value.
117    ///
118    /// `parts` holds one recipe per view of the value; their wires appear in this
119    /// order. Binding rejects an empty list or more than one consuming part. The
120    /// framework uses [`Codec::consumes`] to order observations before a move.
121    pub fn combine(parts: impl IntoIterator<Item = Output>) -> Self {
122        Self::Combine(parts.into_iter().collect())
123    }
124    fn consumes<C: Codec>(&self, ty: &Type, extra: &C::Extra, codec: &C) -> bool {
125        match self {
126            Self::Leaf => codec.consumes(ty, extra),
127            Self::Combine(p) => p.iter().any(|p| p.consumes(ty, extra, codec)),
128            _ => false,
129        }
130    }
131}

133/// One boundary parameter or result component used by wrapper emission.
134///
135/// ```
136/// use prebindgen_tools::{Condition, Wire};
137/// let wire = Wire {
138///     name: syn::parse_quote!(value),
139///     kind: "i64",
140///     active: Condition::default(),
141/// };
142/// assert_eq!(wire.kind, "i64");
143/// ```
144#[derive(Clone, Debug)]
145pub struct Wire<W> {
146    /// Generated identifier used by conversion code and wrapper signatures.
147    pub name: syn::Ident,
148    /// Adapter description of the carrier, supplied by [`Codec::input_leaf`],
149    /// [`Codec::output_leaf`] or [`Codec::selector`].
150    /// The framework carries it into the conversion without interpreting it.
151    pub kind: W,
152    /// Input selector path; outputs normally use the always-active default.
153    pub active: Condition,
154}

156/// Language-specific leaf conversions. Recursive composition stays in tools.
157///
158/// A leaf is a source type the adapter converts itself: a value copied across
159/// the boundary (see [`value`]) or an object kept in Rust (see [`boxed`]). The
160/// adapter chooses by type; a recipe only says where the leaves are.
161///
162/// Input leaves must validate without consuming in `validation`, then
163/// acquire ownership without failure or source calls in `acquire`. Each acquired
164/// value needs an ordinary Rust owner so later failures drop it. `statements`
165/// can construct arguments from those owners. Every input wire must retain
166/// the supplied activation condition. Output leaves keep ownership in Rust
167/// locals until their infallible `commit` runs after successful delivery.
168///
169/// # Implementing a codec
170///
171/// This minimal codec uses Rust values as its carriers. It illustrates converter
172/// construction without a foreign ABI or object storage. A language adapter
173/// additionally writes wrappers and chooses native and destination-language
174/// carrier types for its wire kinds.
175///
176/// ```
177/// use prebindgen_flat::Result;
178/// use prebindgen_tools::{
179///     value, Codec, Condition, Declarations, InputConversion, OutputConversion,
180///     Bindings, fun, expand_param, expand_return, bind_input, Input, validate_bindings,
181/// };
182/// use proc_macro2::TokenStream;
183/// use quote::quote;
184/// use syn::{Ident, Type};
185///
186/// struct RustCarriers;
187/// impl Codec for RustCarriers {
188///     type Extra = ();
189///     type Wire = Option<value::Value>; // `None` marks a selector.
190///
191///     fn selector(&self, _: &()) -> Self::Wire { None }
192///     fn consumes(&self, _: &Type, _: &()) -> bool { false }
193///     fn input_leaf(
194///         &self, ty: &Type, _: &(), wire: Ident, var: Ident, active: &Condition,
195///     ) -> Result<InputConversion<Self::Wire>> {
196///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
197///         let decode = quote!(#wire.clone());
198///         value::input(ty, wire, var, active, decode, Some(kind))
199///     }
200///     fn output_leaf(
201///         &self, ty: &Type, _: &(), wire: Ident, v: TokenStream, borrowed: bool,
202///     ) -> Result<OutputConversion<Self::Wire>> {
203///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
204///         value::output(ty, wire, v, borrowed, |v| v, Some(kind))
205///     }
206/// }
207///
208/// let source = syn::parse_file(r#"
209///     pub fn from_i32(value: i32) -> i64 { value as i64 }
210///     pub fn to_i32(value: &i64) -> i32 { *value as i32 }
211///     pub fn echo(value: i64) -> i64 { value }
212/// "#)?;
213/// let mut flat = Bindings::<Option<value::Value>>::parse(
214///     source.items.into_iter().map(|item| (item, Default::default())),
215/// )?;
216/// let defaults = Declarations::new(&flat, vec![
217///     expand_param!(i64).variant(fun!(from_i32)).variant_self().into(),
218///     expand_return!(i64).field(fun!(to_i32)).field_self().into(),
219/// ])?;
220/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
221/// let function = flat.function("echo")?;
222/// // Input: selector, constructor argument, direct value. Output: two values.
223/// assert_eq!(function.params[0].ty.slot.as_ref().unwrap().wires.len(), 3);
224/// assert_eq!(function.ret.slot.as_ref().unwrap().wires.len(), 2);
225/// // Replace just this parameter's converter; later default filling preserves it.
226/// bind_input(&mut flat, &RustCarriers, "echo", "value", &Input::Leaf, &())?;
227/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
228/// validate_bindings(&flat)?;
229/// let input = flat.function("echo")?.params[0].ty.slot.as_ref().unwrap();
230/// assert_eq!(input.wires.len(), 1);
231/// assert!(matches!(input.wires[0].kind, Some(value::Value::Scalar(_))));
232/// assert!(input.resources.is_empty()); // No foreign ownership in this codec.
233/// # Ok::<(), Box<dyn std::error::Error>>(())
234/// ```
235pub trait Codec {
236    /// Input data borrowed while building one occurrence's conversion; use `()` if unused.
237    /// The same value is passed through nested conversions. It is not stored in Flat;
238    /// hooks retain only the data needed for emission in their code and wires.
239    type Extra;
240    /// The adapter's description of one wire, stored in [`Wire::kind`], such as
241    /// its carrier type or a destination type name.
242    type Wire: Clone + std::fmt::Debug;
243    /// Describe the `i32` arm index wire that tools create for [`Input::Selector`].
244    ///
245    /// `extra` is the occurrence data of the conversion that contains the selector.
246    fn selector(&self, extra: &Self::Extra) -> Self::Wire;
247    /// Whether a leaf output consumes the current source value.
248    ///
249    /// This must agree with [`Self::output_leaf`]. It governs move ordering and
250    /// double-consumption rejection in [`Output::Combine`], not input ownership.
251    /// A non-consuming leaf receives a shared reference with `borrowed` set.
252    ///
253    /// # Arguments
254    ///
255    /// * `ty` — owned source type of the value being delivered.
256    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`].
257    fn consumes(&self, ty: &Type, extra: &Self::Extra) -> bool;
258    /// Prepare a leaf input of source type `ty` using occurrence data `extra`.
259    ///
260    /// All fallible checks belong in [`InputConversion::validation`]; do not
261    /// consume values there. Return a generation error when this type or access
262    /// mode is unsupported.
263    ///
264    /// # Arguments
265    ///
266    /// * `ty` — the source parameter type at this position, as written: `T`,
267    ///   `&T` or `&mut T`.
268    /// * `extra` — occurrence data of the conversion, as passed to [`bind_input`]
269    ///   or [`Declarations::fill`].
270    /// * `wire` — fresh identifier for the boundary parameter this leaf adds.
271    ///   A leaf that needs more wires or locals derives their names from it.
272    /// * `var` — fresh identifier for the Rust local that holds the converted value.
273    ///   [`InputConversion::argument`] normally refers to it.
274    /// * `active` — the selector arms under which this leaf is used. Copy it to every
275    ///   returned wire and resource, and guard validation and acquisition with it.
276    fn input_leaf(
277        &self,
278        ty: &Type,
279        extra: &Self::Extra,
280        wire: syn::Ident,
281        var: syn::Ident,
282        active: &Condition,
283    ) -> Result<InputConversion<Self::Wire>>;
284    /// Prepare delivery of the Rust expression `value` as one or more wires.
285    ///
286    /// Keep unpublished allocations under Rust ownership and transfer them only in
287    /// [`OutputConversion::commit`]. Return a generation error for unsupported
288    /// conversions.
289    ///
290    /// # Arguments
291    ///
292    /// * `ty` — the owned source type of the value.
293    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`]
294    ///   or [`Declarations::fill`].
295    /// * `wire` — fresh identifier for the delivered value. A leaf that needs more
296    ///   wires or locals derives their names from it.
297    /// * `value` — Rust expression producing the value to deliver.
298    /// * `borrowed` — `true` when `value` is a shared reference to `ty`, such as a
299    ///   record field observed in place; do not move out of it. `false` when
300    ///   `value` is owned.
301    fn output_leaf(
302        &self,
303        ty: &Type,
304        extra: &Self::Extra,
305        wire: syn::Ident,
306        value: TokenStream,
307        borrowed: bool,
308    ) -> Result<OutputConversion<Self::Wire>>;
309}
310/// An input conversion split into call-wide validation, acquisition and construction.
311///
312/// Wrappers must validate **all** active inputs and [`Resource`] conflicts before
313/// executing any acquisition, then acquire all consuming inputs before evaluating
314/// argument construction. This preserves caller ownership on validation failure.
315#[derive(Clone, Debug)]
316pub struct InputConversion<W> {
317    /// Flattened boundary parameters in signature order.
318    pub wires: Vec<Wire<W>>,
319    /// Validate and prepare every active leaf before any ownership transfer.
320    pub validation: TokenStream,
321    /// Infallible acquisition of all owned inputs, before constructors can run.
322    pub acquire: TokenStream,
323    /// Active foreign resources whose identities must be checked across all arguments.
324    pub resources: Vec<Resource>,
325    /// Argument construction after all acquisitions, including source constructor calls.
326    pub statements: TokenStream,
327    /// Rust expression passed to the source function after `statements`.
328    pub argument: TokenStream,
329}
330/// Output conversion whose owned temporaries survive until successful delivery.
331///
332/// Wrappers evaluate `statements`, deliver `values` in `wires` order, then execute
333/// `commit` only after every fallible delivery operation succeeds. Values use
334/// the representation produced by the codec's output leaves.
335#[derive(Clone, Debug)]
336pub struct OutputConversion<W> {
337    /// Boundary results in delivery order; one entry per expression in `values`.
338    pub wires: Vec<Wire<W>>,
339    /// Transfer ownership only after all foreign delivery operations succeed.
340    pub commit: TokenStream,
341    /// Code producing owned output temporaries and evaluating observations before moves.
342    pub statements: TokenStream,
343    /// Delivery expressions in wire order; they must remain valid until commit.
344    pub values: Vec<TokenStream>,
345}
346/// A flat model whose slots contain only conversion data for wrapper emission.
347///
348/// Unbound source helpers have `None` slots. Bound functions need `Some` for their
349/// result and every parameter; [`validate_bindings`] checks this before emission.
350/// Record fields retain only names and source types. Recipes and [`Codec::Extra`]
351/// are construction inputs, not stored in this model.
352///
353/// Output conversion data cannot be attached to an input position:
354/// ```compile_fail
355/// use prebindgen_tools::Bindings;
356/// fn swap(flat: &mut Bindings<()>) {
357///     let function = flat.function_mut("foo").unwrap();
358///     function.params[0].ty.slot = function.ret.slot.clone();
359/// }
360/// ```
361/// Nor can a recipe be stored in a slot:
362/// ```compile_fail
363/// use prebindgen_tools::{Bindings, Input};
364/// fn configure(flat: &mut Bindings<()>) {
365///     flat.function_mut("foo").unwrap().params[0].ty.slot = Some(Input::Leaf);
366/// }
367/// ```
368pub type Bindings<W> = Flat<Option<InputConversion<W>>, Option<OutputConversion<W>>>;
369
370/// Build and attach a converter to a named function parameter.
371///
372/// Resolves the occurrence's source type from `flat`, checks recipe compatibility,
373/// and recursively composes wires and code. Nested conversions use the supplied
374/// `extra`, not data from referenced constructors or getters. Recipes and `extra`
375/// are borrowed only for this call. A failure leaves the existing slot unchanged.
376/// Rebinding an occurrence replaces its converter without changing other slots.
377/// Generated names are scoped by parameter position, so separate calls cannot
378/// collide within the wrapper. Keep the function signature fixed after binding.
379/// Check completeness with [`validate_bindings`].
380///
381/// # Arguments
382///
383/// * `flat` — the model holding the function; the parameter's slot is replaced.
384/// * `codec` — the adapter's leaf conversions.
385/// * `function` — unqualified name of the source function.
386/// * `parameter` — name of the parameter, as written in the source.
387/// * `recipe` — how to build the argument from boundary inputs.
388/// * `extra` — occurrence data passed to every codec call for this conversion.
389///
390/// Input binding rejects output recipes:
391/// ```compile_fail
392/// use prebindgen_tools::{bind_input, Bindings, Codec, Output};
393/// fn wrong<C: Codec>(flat: &mut Bindings<C::Wire>, codec: &C, extra: &C::Extra) {
394///     bind_input(flat, codec, "foo", "x", &Output::Leaf, extra).unwrap();
395/// }
396/// ```
397pub fn bind_input<C: Codec>(
398    flat: &mut Bindings<C::Wire>,
399    codec: &C,
400    function: &str,
401    parameter: &str,
402    recipe: &Input,
403    extra: &C::Extra,
404) -> Result<()> {
405    let f = flat.function(function)?;
406    let index = f
407        .params
408        .iter()
409        .position(|p| p.name == parameter)
410        .ok_or_else(|| format!("`{function}`: unknown parameter `{parameter}`"))?;
411    let conversion = Compiler::new(flat, codec, format!("p{index}")).input(
412        &f.params[index].ty.ty,
413        recipe,
414        extra,
415        &Condition::default(),
416    )?;
417    flat.function_mut(function)?.params[index].ty.slot = Some(conversion);
418    Ok(())
419}

421/// Build and attach a converter to a function result, replacing any existing one.
422///
423/// Uses the same construction and failure rules as [`bind_input`]. Output code
424/// refers to the wrapper-local `__result` containing the source call's result.
425/// The recipe is checked for compatible types and signatures, and for repeated
426/// consumption of the same value. Unit results use `Output::fields([])`.
427///
428/// # Arguments
429///
430/// * `flat` — the model holding the function; the result's slot is replaced.
431/// * `codec` — the adapter's leaf conversions.
432/// * `function` — unqualified name of the source function.
433/// * `recipe` — how to deliver the result as boundary outputs.
434/// * `extra` — occurrence data passed to every codec call for this conversion.
435///
436/// Output binding rejects input recipes:
437/// ```compile_fail
438/// use prebindgen_tools::{bind_output, Bindings, Codec, Input};
439/// fn wrong<C: Codec>(flat: &mut Bindings<C::Wire>, codec: &C, extra: &C::Extra) {
440///     bind_output(flat, codec, "foo", &Input::Leaf, extra).unwrap();
441/// }
442/// ```
443pub fn bind_output<C: Codec>(
444    flat: &mut Bindings<C::Wire>,
445    codec: &C,
446    function: &str,
447    recipe: &Output,
448    extra: &C::Extra,
449) -> Result<()> {
450    let conversion = Compiler::new(flat, codec, "ret".into()).output(
451        &flat.function(function)?.ret.ty,
452        recipe,
453        extra,
454        quote!(__result),
455        false,
456    )?;
457    flat.function_mut(function)?.ret.slot = Some(conversion);
458    Ok(())
459}

461/// Check that every bound function has an input converter for each parameter
462/// and an output converter, including unit results.
463///
464/// Functions with entirely empty slots remain source-only helpers. This check
465/// neither builds conversions nor changes the model. Adapters call it before
466/// writing wrappers. It does not validate code fragments supplied directly by callers.
467pub fn validate_bindings<W>(flat: &Bindings<W>) -> Result<()> {
468    for f in flat.functions.values() {
469        if f.ret.slot.is_some() {
470            for p in &f.params {
471                if p.ty.slot.is_none() {
472                    return Err(format!(
473                        "`{}`: parameter `{}` has no input converter",
474                        f.name, p.name
475                    ));
476                }
477            }
478        } else if f.params.iter().any(|p| p.ty.slot.is_some()) {
479            return Err(format!(
480                "`{}`: bound inputs have no output converter",
481                f.name
482            ));
483        }
484    }
485    Ok(())
486}
487struct Compiler<'a, C: Codec> {
488    flat: &'a Bindings<C::Wire>,
489    codec: &'a C,
490    prefix: String,
491    next: std::cell::Cell<usize>,
492}
493impl<'a, C: Codec> Compiler<'a, C> {
494    fn new(flat: &'a Bindings<C::Wire>, codec: &'a C, prefix: String) -> Self {
495        Self {
496            flat,
497            codec,
498            prefix,
499            next: std::cell::Cell::new(0),
500        }
501    }
502    fn name(&self) -> syn::Ident {
503        let next = self.next.get();
504        let n = format_ident!("{}_w{next}", self.prefix);
505        self.next.set(next + 1);
506        n
507    }
508    fn input(
509        &self,
510        ty: &Type,
511        recipe: &Input,
512        extra: &C::Extra,
513        active: &Condition,
514    ) -> Result<InputConversion<C::Wire>> {
515        let var = self.name();
516        let mut wires = Vec::new();
517        let mut validation = TokenStream::new();
518        let mut acquire = TokenStream::new();
519        let mut resources = Vec::new();
520        let mut statements = TokenStream::new();
521        let argument = quote!(#var);
522        match recipe {
523            Input::Leaf => {
524                let w = self.name();
525                return self.codec.input_leaf(ty, extra, w, var, active);
526            }
527            Input::Fields(children) => {
528                if type_key(ty) == "()" && children.is_empty() {
529                    return Ok(InputConversion {
530                        wires,
531                        validation,
532                        acquire,
533                        resources,
534                        statements,
535                        argument: quote!(()),
536                    });
537                }
538                let r = self.flat.record(ty)?;
539                if r.fields.len() != children.len() {
540                    return Err(format!(
541                        "`{}`: expected {} field converters, got {}",
542                        type_key(ty),
543                        r.fields.len(),
544                        children.len()
545                    ));
546                }
547                let mut fields = Vec::new();
548                for (field, child) in r.fields.iter().zip(children) {
549                    let p = self.input(&field.ty, child, extra, active)?;
550                    validation.extend(p.validation);
551                    acquire.extend(p.acquire);
552                    resources.extend(p.resources);
553                    statements.extend(p.statements);
554                    wires.extend(p.wires);
555                    let name = &field.name;
556                    let arg = p.argument;
557                    fields.push(quote!(#name:#arg));
558                }
559                statements.extend(quote!(let #var:#ty=#ty{#(#fields),*};));
560            }
561            Input::Constructor { function, args } => {
562                let f = self.flat.function(function)?;
563                if !same_type(&f.ret.ty, ty) {
564                    return Err(format!(
565                        "constructor `{function}` returns `{}`, expected `{}`",
566                        type_key(&f.ret.ty),
567                        type_key(ty)
568                    ));
569                }
570                if f.params.len() != args.len() {
571                    return Err(format!("constructor `{function}`: wrong argument count"));
572                }
573                let mut call_args = Vec::new();
574                for (param, child) in f.params.iter().zip(args) {
575                    let p = self.input(&param.ty.ty, child, extra, active)?;
576                    validation.extend(p.validation);
577                    acquire.extend(p.acquire);
578                    resources.extend(p.resources);
579                    statements.extend(p.statements);
580                    wires.extend(p.wires);
581                    call_args.push(p.argument);
582                }
583                let path = &f.path;
584                statements.extend(quote!(let #var:#ty=#path(#(#call_args),*);));
585            }
586            Input::Selector(arms) => {
587                if arms.is_empty() {
588                    return Err("an input selector needs at least one alternative".into());
589                }
590                if matches!(ty, Type::Reference(_)) {
591                    return Err("selector results must be owned in this subset".into());
592                }
593                let tag = self.name();
594                wires.push(Wire {
595                    name: tag.clone(),
596                    kind: self.codec.selector(extra),
597                    active: active.clone(),
598                });
599                let count = arms.len() as i32;
600                let invalid = active.and(quote!(!(0..#count).contains(&#tag)));
601                validation.extend(quote!(if #invalid {return Err("invalid selector".into());}));
602                let mut cases = Vec::new();
603                for (i, arm) in arms.iter().enumerate() {
604                    let p = self.input(ty, arm, extra, &active.arm(tag.clone(), i as i32))?;
605                    validation.extend(p.validation);
606                    acquire.extend(p.acquire);
607                    resources.extend(p.resources);
608                    let body = p.statements;
609                    let value = p.argument;
610                    wires.extend(p.wires);
611                    let i = i as i32;
612                    cases.push(quote!(#i=>{#body #value}));
613                }
614                statements = quote!(let #var:#ty=match #tag{#(#cases,)* _=>unreachable!("validated selector")};);
615            }
616        }
617        Ok(InputConversion {
618            wires,
619            validation,
620            acquire,
621            resources,
622            statements,
623            argument,
624        })
625    }
626    fn output(
627        &self,
628        ty: &Type,
629        recipe: &Output,
630        extra: &C::Extra,
631        value: TokenStream,
632        borrowed: bool,
633    ) -> Result<OutputConversion<C::Wire>> {
634        let mut out = OutputConversion {
635            commit: TokenStream::new(),
636            wires: Vec::new(),
637            statements: TokenStream::new(),
638            values: Vec::new(),
639        };
640        match recipe {
641            Output::Leaf => {
642                let w = self.name();
643                return self.codec.output_leaf(ty, extra, w, value, borrowed);
644            }
645            Output::Fields(children) => {
646                if type_key(ty) == "()" && children.is_empty() {
647                    return Ok(out);
648                }
649                let r = self.flat.record(ty)?;
650                if r.fields.len() != children.len() {
651                    return Err("wrong output field count".into());
652                }
653                for (field, child) in r.fields.iter().zip(children) {
654                    let name = &field.name;
655                    let p = self.output(&field.ty, child, extra, quote!(&(#value).#name), true)?;
656                    append(&mut out, p);
657                }
658            }
659            Output::Getter { function, output } => {
660                let f = self.flat.function(function)?;
661                let expected: Type = syn::parse_quote!(&#ty);
662                if f.params.len() != 1 || !same_type(&f.params[0].ty.ty, &expected) {
663                    return Err(format!(
664                        "getter `{function}` must take exactly `&{}`",
665                        type_key(ty)
666                    ));
667                }
668                if matches!(f.ret.ty, Type::Reference(_)) {
669                    return Err("borrowed getter results are outside this subset".into());
670                }
671                let v = self.name();
672                let path = &f.path;
673                let arg = if borrowed { value } else { quote!(&#value) };
674                out.statements = quote!(let #v=#path(#arg););
675                append(
676                    &mut out,
677                    self.output(&f.ret.ty, output, extra, quote!(#v), false)?,
678                );
679            }
680            Output::Combine(parts) => {
681                if parts.is_empty() {
682                    return Err("an output combination needs at least one part".into());
683                }
684                if parts
685                    .iter()
686                    .filter(|p| p.consumes(ty, extra, self.codec))
687                    .count()
688                    > 1
689                {
690                    return Err("a combination cannot consume the same value twice".into());
691                }
692                let mut bodies = Vec::new();
693                for child in parts {
694                    let consuming = child.consumes(ty, extra, self.codec);
695                    let arg = if borrowed || consuming {
696                        value.clone()
697                    } else {
698                        quote!(&#value)
699                    };
700                    let mut p = self.output(ty, child, extra, arg, borrowed || !consuming)?;
701                    bodies.push((consuming, std::mem::take(&mut p.statements)));
702                    append(&mut out, p);
703                }
704                // Wire order follows the declaration. Observations execute before the move.
705                bodies.sort_by_key(|(consumes, _)| *consumes);
706                for (_, body) in bodies {
707                    out.statements.extend(body);
708                }
709            }
710        }
711        Ok(out)
712    }
713}
714fn append<W>(to: &mut OutputConversion<W>, from: OutputConversion<W>) {
715    to.commit.extend(from.commit);
716    to.wires.extend(from.wires);
717    to.statements.extend(from.statements);
718    to.values.extend(from.values);
719}

722///
723/// `tokens` must be a sequence of items, such as a whole generated module. Returns a syntax error if the tokens do not form a file. This does not

728}

745/// A leaf is active only on its selected path. Adapters use the same condition
746/// for foreign-side resource locking and native conversion code.
747#[derive(Clone, Debug, Default)]
748pub struct Condition(
749    /// Conjunction of `(selector identifier, required arm index)` terms.
750    /// An empty vector is always active.
751    pub Vec<(syn::Ident, i32)>,
752);
753impl Condition {
754    /// Return a new condition additionally requiring one selector arm.
755    ///
756    /// Does not modify the original condition; no additional runtime flag is generated.
757    ///
758    /// # Arguments
759    ///
760    /// * `tag` — identifier of the selector's index wire.
761    /// * `index` — zero-based index of the required arm.
762    pub fn arm(&self, tag: syn::Ident, index: i32) -> Self {
763        let mut result = self.clone();
764        result.0.push((tag, index));
765        result
766    }
767    /// Emit this condition AND a Rust predicate, short-circuiting the predicate
768    /// when the selector path is inactive. `predicate` is a Rust `bool` expression.
769    pub fn and(&self, predicate: TokenStream) -> TokenStream {
770        if self.0.is_empty() {
771            predicate
772        } else {
773            let condition = self.rust();
774            quote!((#condition) && (#predicate))
775        }
776    }
777    /// Emit the conjunction as a Rust boolean expression; an empty condition emits `true`.
778    pub fn rust(&self) -> TokenStream {
779        self.0
780            .iter()
781            .map(|(tag, i)| quote!(#tag == #i))
782            .reduce(|a, b| quote!((#a) && (#b)))
783            .unwrap_or_else(|| quote!(true))
784    }
785}
786/// Access to one foreign resource during an exported call.
787#[derive(Clone, Copy, Debug, PartialEq, Eq)]
788pub enum Access {
789    /// Read-only borrow; multiple active shared accesses may alias.
790    Shared,
791    /// Mutable borrow; conflicts with every other active use of the resource.
792    Exclusive,
793    /// Ownership transfer; conflicts with every other active use of the resource.
794    Move,
795}
796/// The adapter supplies identity; tools only enforce call-wide access conflicts.
797#[derive(Clone, Debug)]
798pub struct Resource {
799    /// Side-effect-free Rust expression identifying a live resource, such as an address.
800    /// All identities compared within a call must share an equality-compatible domain.
801    pub identity: TokenStream,
802    /// Borrow or ownership requirement imposed by this input conversion.
803    pub access: Access,
804    /// Path on which this resource is accessed; inactive resources do not conflict.
805    pub active: Condition,
806}
807/// Emit pairwise alias checks across all active input resources.
808///
809/// `resources` holds the [`InputConversion::resources`] of all of one exported
810/// function's parameters together, so that aliases across parameters are found.
811///
812/// Run this code after leaf validation and before **any** ownership acquisition.
813/// Shared/shared pairs and mutually exclusive selector arms do not conflict.
814/// Conflicts return `Err("conflicting input aliases".into())` from the enclosing
815/// wrapper. This does not validate pointer liveness, synchronize threads, or
816/// acquire resources; those remain adapter/caller responsibilities.
817pub fn validate_resources(resources: &[Resource]) -> TokenStream {
818    let mut result = TokenStream::new();
819    for (i, a) in resources.iter().enumerate() {
820        for b in &resources[i + 1..] {
821            if a.access == Access::Shared && b.access == Access::Shared {
822                continue;
823            }
824            let x = &a.identity;
825            let y = &b.identity;
826            // Different alternatives of the same selector cannot conflict.
827            if a.active
828                .0
829                .iter()
830                .any(|(tag, arm)| b.active.0.iter().any(|(t, i)| t == tag && i != arm))
831            {
832                continue;
833            }
834            let mut active = a.active.clone();
835            for term in &b.active.0 {
836                if !active.0.contains(term) {
837                    active.0.push(term.clone());
838                }
839            }
840            let conflict = active.and(quote!((#x) == (#y)));
841            result.extend(quote!(if #conflict {return Err("conflicting input aliases".into());}));
842        }
843    }
844    result
845}
