1//! Panic containment and pointer validation for foreign adapters.
2#![warn(missing_docs)]
3/// Runtime failure represented as a message for the adapter's error channel.
4pub type Result<T> = std::result::Result<T, String>;
5
6/// Run a boundary operation, converting unwinding panics into a generic error.
7///
8/// `f` is the body of the exported call; its error message passes through to
9/// the adapter's error channel. Ordinary errors pass through unchanged. This does not roll back side effects
10/// or ownership transfers, and cannot catch a panic configured to abort.
11///
12/// ```
13/// use prebindgen_runtime::boundary;
14/// assert_eq!(boundary(|| Ok(42)), Ok(42));
15/// assert_eq!(boundary::<()>(|| Err("invalid input".into())),
16///            Err("invalid input".into()));
17/// ```
18pub fn boundary<T>(f: impl FnOnce() -> Result<T>) -> Result<T> {
19    std::panic::catch_unwind(std::panic::AssertUnwindSafe(f))
20        .unwrap_or_else(|_| Err("Rust operation panicked".into()))
21}
22
23/// Check that `pointer`, an allocation address read from a boundary wire, is
24/// non-null and aligned for `T`.
25///
26/// Check encoding only; the adapter/caller must guarantee type, liveness and
27/// synchronization. This helper neither registers objects nor tracks ownership.
28/// It never dereferences the pointer. Success does not establish that dereferencing
29/// or freeing it is safe; zero-sized pointees are not rejected by this check.
30pub fn validate_pointer<T>(pointer: *const T) -> Result<()> {
31    if pointer.is_null() || !pointer.is_aligned() {
32        return Err("null or misaligned allocation pointer".into());
33    }
34    Ok(())
35}

37#[cfg(test)]
38mod tests {
39    use super::*;
40    #[test]
41    fn panics_do_not_escape() {
42        assert!(super::boundary::<()>(|| panic!("test")).is_err());
43    }
44    #[test]
45    fn pointer_encoding_checks() {
46        let value = 1u64;
47        let pointer = &value as *const u64;
48        assert!(validate_pointer(pointer).is_ok());
49        assert!(validate_pointer((pointer as usize | 1) as *const u64).is_err());
50        assert!(validate_pointer::<u64>(std::ptr::null()).is_err());
51    }
