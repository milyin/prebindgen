1//! What became of a declaration, and why.
2//!
3//! Four outcomes, and they are not interchangeable. A capability v2 has not
4//! implemented is a [`Skipped`](Outcome::Skipped) declaration with a code and a
5//! path to the site that could not be lowered. Malformed input, a missing
6//! declared item, or a bug in the engine are **errors** — never skips — because
7//! a build that quietly reports its own bug as an unsupported feature is a
8//! build that cannot be trusted to say what it generated.
9
10use serde::Serialize;
11
12use crate::decl::{DeclarationId, Origin};
13
14/// A stable code naming *what* is not implemented, dotted from general to
15/// specific: `unsupported.string`, `unsupported.handle.borrowed_input`.
16///
17/// Stable because reports are diffed and CI gates name codes. The readable
18/// explanation beside it is free to change; this is not.
19#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize)]
20#[serde(transparent)]
21pub struct Capability(String);
22
23impl Capability {
24    pub fn new(code: impl Into<String>) -> Self {
25        Capability(code.into())
26    }
27
28    pub fn as_str(&self) -> &str {
29        &self.0
30    }
31}
32
33impl std::fmt::Display for Capability {
34    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
35        f.write_str(&self.0)
36    }
37}
38
39/// Why one declaration was not generated.
40#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
41pub struct Skip {
42    /// The unimplemented capability — see [`Capability`].
43    pub capability: Capability,
44    /// One readable sentence a build script can print as it stands.
45    pub explanation: String,
46    /// Root → failing site, one step per line of the printed path: the declaration,
47    /// then how the walk reached the thing that could not be lowered. A direct
48    /// skip has just the root.
49    pub dependency_path: Vec<String>,
50}
51
52impl Skip {
53    /// A skip at the declaration itself, with no dependency to walk through.
54    pub fn direct(
55        capability: impl Into<String>,
56        explanation: impl Into<String>,
57        root: impl Into<String>,
58    ) -> Self {
59        Skip {
60            capability: Capability::new(capability),
61            explanation: explanation.into(),
62            dependency_path: vec![root.into()],
63        }
64    }
65
66    /// This skip as `a -> b -> c`.
67    pub fn path(&self) -> String {
68        self.dependency_path.join(" -> ")
69    }
70}
71
72/// What became of one declaration.
73#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
74#[serde(tag = "outcome", rename_all = "snake_case")]
75pub enum Outcome {
76    /// Fully generated, with its dependencies and its declared semantics.
77    Emitted,
78    /// Not generated: a capability is missing, here or in something this
79    /// declaration needs.
80    Skipped(Skip),
81    /// The binding asked for it to be left alone.
82    Ignored,
83}
84
85impl Outcome {
86    /// The skip, when this is one.
87    pub fn skip(&self) -> Option<&Skip> {
88        match self {
89            Outcome::Skipped(skip) => Some(skip),
90            Outcome::Emitted | Outcome::Ignored => None,
91        }
92    }
93}

95/// The engine failed, and no report describes the run.
96///
97/// Distinct from a skip on purpose: everything here means the *input* or the
98/// *engine* is wrong, not that a feature is missing.
99#[derive(Debug)]
100pub enum EngineError {
101    /// The declared sources could not be parsed into the model.
102    Source(prebindgen_flat::flat::ParseError),
103    /// Declared declarations that match no captured `#[prebindgen]` item. A
104    /// declaration is a statement of intent; its target being absent is a typo
105    /// or a source-crate rename, and never a capability question. All of them
106    /// are collected before failing.
107    DeclaredNotFound {
108        entries: Vec<(DeclarationId, Origin)>,
109    },
110    /// Two declarations answering to one id. An id identifies a declaration,
111    /// and a report that gave one id to two entries could not account for
112    /// either — so this is a contradiction in the declarations, not a gap in
113    /// what v2 implements.
114    DuplicateDeclaration { entries: Vec<DeclarationId> },
115    /// Planning refused the input, or an internal contract was violated. A
116    /// missing capability is never one of these — that is a [`Skip`].
117    Planning(crate::target::PlanningError),
118    /// Writing an artifact or a report failed.
119    Io(std::io::Error),
120}
121
122impl std::fmt::Display for EngineError {
123    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
124        match self {
125            EngineError::Source(error) => write!(f, "v2: {error}"),
126            EngineError::DeclaredNotFound { entries } => {
127                writeln!(
128                    f,
129                    "v2: {} declared item(s) match no captured `#[prebindgen]` item:",
130                    entries.len()
131                )?;
132                for (id, origin) in entries {
133                    writeln!(
134                        f,
135                        "  {id} — no captured {} `{}`",
136                        origin.describe_captured(),
137                        origin.name()
138                    )?;
139                }
140                write!(
141                    f,
142                    "check the spelling in build.rs, and that the source crate still \
143                     exports them"
144                )
145            }
146            EngineError::DuplicateDeclaration { entries } => {
147                writeln!(
148                    f,
149                    "v2: {} declaration id(s) were declared more than once:",
150                    entries.len()
151                )?;
152                for id in entries {
153                    writeln!(f, "  {id}")?;
154                }
155                write!(f, "each declaration answers to one id")
156            }
157            EngineError::Planning(error) => write!(f, "v2: {error}"),
158            EngineError::Io(error) => write!(f, "v2: {error}"),
159        }
160    }
161}
162
163impl std::error::Error for EngineError {}
164
165impl From<std::io::Error> for EngineError {
166    fn from(error: std::io::Error) -> Self {
167        EngineError::Io(error)
168    }
169}
170
171impl From<prebindgen_flat::flat::ParseError> for EngineError {
172    fn from(error: prebindgen_flat::flat::ParseError) -> Self {
173        EngineError::Source(error)
174    }
