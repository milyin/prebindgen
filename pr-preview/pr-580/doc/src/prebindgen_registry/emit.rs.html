5/// The capability handed to Rust-emission callbacks.
6///
7/// The flat crate owns the rendering protocol because it owns the syntax. The
8/// registry owns this concrete key and its private constructor, so ordinary
9/// registry users receive an `Emit` only during final `write_rust` callbacks.
10///
11/// ```compile_fail
12/// use prebindgen_registry::Emit;
13/// let emit = Emit::new();
14/// ```
15///
16/// ```compile_fail
17/// use prebindgen_registry::Emit;
18/// let emit = Emit(());
19/// ```
20///
21/// A registry-only adapter cannot name the flat rendering protocol through
22/// the registry's model re-export:
23///
24/// ```compile_fail
25/// use prebindgen_registry::flat::emit::RustEmitter;
26/// ```
27///
28/// An adapter receives only generate-only operations, not retained spelling:
29///
30/// ```compile_fail
31/// # use prebindgen_registry::{Emit, flat::TypeRef};
32/// use prebindgen_flat::RustEmitter;
33/// # fn leak(emit: &Emit, ty: &TypeRef) {
34/// let _ = emit.spell(ty);
35/// # }
36/// ```
37#[derive(Debug)]
38pub struct Emit {
39    flat: FlatEmitKey,
40    source_modules: HashMap<String, syn::Path>,
41    default_module: syn::Path,
42}
43
44#[derive(Debug)]
45struct FlatEmitKey;
46
47impl prebindgen_flat::RustEmitter for FlatEmitKey {}
48
49impl Emit {
50    pub(crate) fn new(registry: &crate::Registry, source_module: Option<&syn::Path>) -> Self {
51        let default_module = source_module
52            .cloned()
53            .or_else(|| registry.default_module())
54            .unwrap_or_else(|| syn::parse_quote!(crate));
55        let source_modules = registry
56            .named_item_idents()
57            .map(|ident| {
58                let module = registry
59                    .origin_module(ident)
60                    .unwrap_or_else(|| default_module.clone());
61                (ident.to_string(), module)
62            })
63            .collect();
64        Self {
65            flat: FlatEmitKey,
66            source_modules,
67            default_module,
68        }
69    }

91    /// Emit one source-type fragment from Flat model facts.
92    ///
93    /// Qualification is driven by Flat nominal/extent facts and the registry's
94    /// declaration-module map. No token text is inspected to decide what the
95    /// type means.
96    pub fn emit_source_type(
97        &self,
98        ty: &prebindgen_flat::flat::TypeRef,
99    ) -> proc_macro2::TokenStream {
100        prebindgen_flat::RustEmitter::emit_source_type(
101            &self.flat,
102            ty,
103            &self.source_modules,
104            &self.default_module,
105        )
106    }

108    /// Generate a public const alias to the source declaration.
109    pub fn const_alias(
110        &self,
111        item: &prebindgen_flat::flat::Constant,
112        source_module: &syn::Path,
113    ) -> syn::ItemConst {
114        let ident = &item.name;
115        let ty = self.emit_source_type(&item.ty);
116        let docs: Vec<syn::Attribute> = item
117            .docs()
118            .into_iter()
119            .flat_map(|docs| {
120                docs.lines()
121                    .map(|line| {
122                        let doc = format!(" {line}");
123                        syn::parse_quote!(#[doc = #doc])
124                    })
125                    .collect::<Vec<_>>()
126            })
127            .collect();
128        syn::parse_quote!(#(#docs)* pub const #ident: #ty = #source_module::#ident;)
129    }

164    /// Allocate the final private Rust symbol for a registry-owned operation.
165    ///
166    /// `namespace` is adapter vocabulary (for example `"jni"`), not a Rust
167    /// type spelling. This method lives on the emission capability so neither
168    /// the registry plan nor a language adapter can turn model identity into a
169    /// Rust identifier before final file assembly. The emitted name keeps a
170    /// bounded semantic stem for readability and a stable hash for uniqueness.
171    pub fn operation_ident(
172        &self,
173        namespace: &str,
174        operation: &crate::generation::OperationId,
175    ) -> syn::Ident {
176        let semantic = ident_component(&operation.semantic_label());
177        let (direction, semantic) = match operation.direction() {
178            crate::recipe::Direction::Construct => ("in", format!("wire_to_{semantic}")),
179            crate::recipe::Direction::Deconstruct => ("out", format!("{semantic}_to_wire")),
180        };
181        let role = match operation.role() {
182            crate::generation::OperationRole::Converter => "convert".to_owned(),
183            crate::generation::OperationRole::Stage(index) => format!("stage_{index}"),
184        };
185        quote::format_ident!(
186            "__{namespace}_{direction}_{role}_{semantic}_{:016x}",
187            operation.stable_fingerprint()
188        )
189    }
190}

192/// Turn diagnostic model vocabulary into a readable, bounded identifier
193/// component. This formats identity during final emission; it does not parse
194/// the text or use it to make a generation decision.
195fn ident_component(label: &str) -> String {
196    const MAX_CHARS: usize = 96;
197    if label.trim() == "()" {
198        return "unit".to_owned();
199    }
200    let mut out = String::new();
201    let mut separator = false;
202    for ch in label.chars() {
203        if ch.is_ascii_alphanumeric() {
204            out.push(ch);
205            separator = false;
206        } else if !out.is_empty() && !separator {
207            out.push('_');
208            separator = true;
209        }
210        if out.len() >= MAX_CHARS {
211            break;
212        }
213    }
214    while out.ends_with('_') {
215        out.pop();
216    }
217    if out.is_empty() {
218        "operation".to_owned()
219    } else {
220        out
221    }
222}

224#[cfg(test)]
225mod tests {
226    use super::{ident_component, Emit};
227    use crate::{ArtifactId, Direction, OperationId};

229    #[test]
230    fn operation_symbols_are_stable_and_writer_scoped() {
231        let operation = OperationId::shared(
232            ArtifactId::new("test-codec", "owned").unwrap(),
233            Direction::Construct,
234        );
235        let emit = Emit::for_test();
236
237        let first = emit.operation_ident("test", &operation);
238        let second = emit.operation_ident("test", &operation);
239
240        assert_eq!(first, second);
241        assert!(first
242            .to_string()
243            .starts_with("__test_in_convert_wire_to_test_codec_owned_"));
244    }
245
246    #[test]
247    fn operation_symbol_stems_are_readable_and_bounded() {
248        assert_eq!(
249            ident_component("impl Fn(ZSample) + Send + Sync + 'static"),
250            "impl_Fn_ZSample_Send_Sync_static"
251        );
252        assert_eq!(ident_component("()"), "unit");
253        assert!(ident_component(&"word-".repeat(100)).len() <= 96);
254    }
