1//! The common Rust writer: one component, both targets.
2//!
3//! It owns the shape of the generated Rust — the locals and their order, the
4//! branches on failure, the construction of the source value, the call, the
5//! return — and allocates every temporary from the plan, so two operations
6//! rendered into one wrapper cannot collide over a name. A target contributes
7//! one expression per operation of its own ([`Target::write_operation`]) and
8//! the declarations its carriers need ([`Target::write_carrier`]); it
9//! contributes no control flow.
10//!
11//! Source types and struct shapes are generated through Flat's emission
12//! capability, never from retained syntax and never spelled by an adapter.
13

29/// This engine's emission key.
30///
31/// Implementing [`RustEmitter`] is a collector's deliberate decision to
32/// establish an emission boundary; v2 establishes its own rather than borrowing
33/// v1's.
34pub(crate) struct Writer;

36impl RustEmitter for Writer {}

38/// How generated code names each entity: the module it is reached through.
39///
40/// A captured item is reached through the source module the frontend
41/// configured for the run. An item the binding defines itself is reached
42/// where the binding said it is — `crate::helpers::x` — which its origin
43/// records as its crate stamp, since for a binding-local item that is what
44/// the stamp means. One value carries both so the writer asks one question.
45pub(crate) struct Reach {
46    /// Name → module, for every entity that is not reached through the
47    /// default. The shape [`RustEmitter::emit_source_type`] takes.
48    modules: HashMap<String, syn::Path>,
49    default: syn::Path,
50}
51
52impl Reach {
53    pub(crate) fn new(flat: &Flat, default: syn::Path) -> Self {
54        let modules = flat
55            .elements()
56            .filter_map(|element| Some((element.name()?, element)))
57            .filter(|(name, _)| flat.is_binding_local(*name))
58            .filter_map(|(name, element)| {
59                let module = element.location().crate_name.as_deref()?;
60                Some((name.to_string(), syn::parse_str(module).ok()?))
61            })
62            .collect();
63        Reach { modules, default }
64    }
65
66    /// The module the entity with this name is reached through.
67    fn of(&self, name: &str) -> &syn::Path {
68        self.modules.get(name).unwrap_or(&self.default)
69    }
70}

72/// The whole generated Rust file, as text.
73#[allow(clippy::too_many_arguments)]
74pub(crate) fn render<T: Target>(
75    flat: &Flat,
76    target: &T,
77    binding: &Binding<T>,
78    reach: &Reach,
79    retained: &[Retained],
80    nodes: &[ValuePlan],
81    primitives: &[Applied<T::Op>],
82    functions: &[FunctionPlan<T::Op>],
83) -> String {
84    // The guards come first, and they are emitted whatever else this run
85    // retained. Each one is a compile-time assertion the capture reader
86    // injected — today, that the source crate's features match the set the
87    // capture was filtered by — and a binding that dropped it would compile
88    // happily against a source crate it disagrees with. It belongs to no
89    // declaration, so nothing in retention decides whether to keep it.
90    let guards = flat.guards().map(|guard| Writer.guard(guard));
91    let carriers = carriers(flat, target, binding, retained, nodes);
92    // A helper an operation needs is emitted once, before the wrappers, in
93    // the order the wrappers first need it.
94    let mut helpers: Vec<Artifact> = Vec::new();
95    let wrappers: Vec<TokenStream> = functions
96        .iter()
97        .map(|function| {
98            wrapper(
99                flat,
100                target,
101                binding,
102                reach,
103                primitives,
104                function,
105                &mut helpers,
106            )
107        })
108        .collect();
109    let helpers = helpers.iter().map(|helper| &helper.rust);
110    let tokens = quote! {
111        #(#guards)*
112        #(#carriers)*
113        #(#helpers)*
114        #(#wrappers)*
115    };
116    match syn::parse2::<syn::File>(tokens.clone()) {
117        Ok(file) => prettyplease::unparse(&file),
118        // Unparseable output is a defect in this writer, and hiding it behind
119        // pretty printing would only move the compiler error somewhere less
120        // useful.
121        Err(_) => tokens.to_string(),
122    }
123}

125/// The declarations the carriers of every retained type need, each once, in
126/// the order the types were declared.
127///
128/// A carrier's declaration exists only where the source item its type names
129/// does — the `repr(C)` mirror of a conditional struct is itself conditional —
130/// so every item the target writes carries that item's conditions.
131fn carriers<T: Target>(
132    flat: &Flat,
133    target: &T,
134    binding: &Binding<T>,
135    retained: &[Retained],
136    nodes: &[ValuePlan],
137) -> Vec<TokenStream> {
138    let mut written: std::collections::HashSet<CarrierId> = std::collections::HashSet::new();
139    let mut items = Vec::new();
140    for values in retained {
141        let OutputForm::Type { representation, .. } = binding.form_of(values.output) else {
142            continue;
143        };
144        let conditions = values
145            .declaration
146            .captured(flat)
147            .map(|element| Writer.conditions(Conditioned::Item(element)))
148            .unwrap_or_default();
149        let unit = values
150            .declaration
151            .entity_name()
152            .and_then(|name| flat.unit_enum(&name));
153        let root = values.inputs.first().map(|root| &nodes[root.0]);
154        let used: Vec<(CarrierId, bool)> = match binding.representation_of(*representation) {
155            Representation::Terminal {
156                into_rust,
157                out_of_rust,
158                ..
159            } => into_rust
160                .iter()
161                .chain(out_of_rust.iter())
162                .map(|codec| (codec.carrier, false))
163                .collect(),
164            Representation::Product { carrier, .. } => vec![(*carrier, true)],
165            Representation::Unsupported(_) => Vec::new(),
166        };
167        for (carrier, product) in used {
168            if !written.insert(carrier) {
169                continue;
170            }
171            let members = match (product, root) {
172                (true, Some(root)) => root
173                    .relation
174                    .parts()
175                    .iter()
176                    .zip(&root.children)
177                    .map(|(part, child)| (part, binding.carrier_of(nodes[child.0].carrier)))
178                    .collect(),
179                _ => Vec::new(),
180            };
181            let feed = CarrierFeed {
182                carrier: binding.carrier_of(carrier),
183                members,
184                unit,
185            };
186            for item in target.write_carrier(&feed) {
187                items.push(quote!(#(#conditions)* #item));
188            }
189        }
190    }
191    items
192}

194/// One exported function.
195fn wrapper<T: Target>(
196    flat: &Flat,
197    target: &T,
198    binding: &Binding<T>,
199    reach: &Reach,
200    primitives: &[Applied<T::Op>],
201    function: &FunctionPlan<T::Op>,
202    helpers: &mut Vec<Artifact>,
203) -> TokenStream {
204    let mut names: HashMap<ValueId, syn::Ident> = HashMap::new();
205    // Parameters are named by the form, because a convention that requires an
206    // environment operand has to say what it is called. Everything else is
207    // named here, and around those names: a temporary that shadowed a live
208    // parameter would compile and read the wrong value.
209    // Reserved by their canonical spelling: `r#v0` and `v0` are one name to
210    // Rust, so reserving the raw form would leave the plain one free to shadow
211    // it.
212    let mut taken: std::collections::HashSet<String> = function
213        .params
214        .iter()
215        .map(|(_, p)| unraw(&p.name))
216        .collect();
217    let params: Vec<TokenStream> = function
218        .params
219        .iter()
220        .map(|(id, param)| {
221            names.insert(*id, param.name.clone());
222            let name = &param.name;
223            let ty = &param.ty;
224            let mutable = param.mutable.then(|| quote!(mut));
225            quote!(#mutable #name: #ty)
226        })
227        .collect();
228    let ok_binding = free_name("value", &mut taken);
229    let error_binding = free_name("error", &mut taken);
230    // Temporaries are numbered by definition order, which is why the same
231    // operation used twice cannot collide with itself.
232    let mut next_local = 0;
233    let mut local = |names: &mut HashMap<ValueId, syn::Ident>,
234                     taken: &mut std::collections::HashSet<String>,
235                     id: ValueId| {
236        let name = loop {
237            let candidate = format!("v{next_local}");
238            next_local += 1;
239            if taken.insert(candidate.clone()) {
240                break format_ident!("{candidate}");
241            }
242        };
243        names.insert(id, name.clone());
244        name
245    };
246
247    // A wrapper names source items — the function it calls, and every struct it
248    // constructs — and compiles only where all of them exist. So it inherits
249    // each one's condition: a `#[cfg]` the capture reader could not answer and
250    // rewrote back onto the item. Rust conjoins repeated `#[cfg]` attributes on
251    // one item, so carrying them side by side settles the wrapper's own
252    // condition without anything here reading what they say. Two items carrying
253    // the same condition state it once, which conjunction makes cosmetic and
254    // review makes worth doing.
255    //
256    // The instructions are where a wrapper spells a source item: its signature
257    // carries carriers only, and the one operation that spells a source type —
258    // a handle taken back or released, cast to `*mut source::Ledger` — is a
259    // standard one the registry writes, so it is found here too.
260    let mut conditions = Vec::new();
261    let mut conditioned: std::collections::HashSet<String> = std::collections::HashSet::new();
262    for step in &function.instrs {
263        let named = match &step.instr {
264            Instr::Construct { name, .. } => name.clone(),
265            Instr::Call { function, .. } => function.clone(),
266            Instr::Apply { primitive, .. } => {
267                let applied = &primitives[primitive.0];
268                match &applied.implementation {
269                    Implementation::Standard(StandardOp::FromRaw | StandardOp::Release) => {
270                        match applied.subject.kind() {
271                            prebindgen_flat::flat::TypeKind::Named { id, .. } => id.name.clone(),
272                            _ => continue,
273                        }
274                    }
275                    _ => continue,
276                }
277            }
278        };
279        // Planning names only items the model declares.
280        let Some(element) = flat.element(named.as_str()) else {
281            continue;
282        };
283        for condition in Writer.conditions(Conditioned::Item(element)) {
284            if conditioned.insert(condition.to_string()) {
285                conditions.push(condition);
286            }
287        }
288    }
289
290    let mut statements = Vec::new();
291    for step in &function.instrs {
292        // What a field's condition holds the statement to: a member declared
293        // only sometimes is read only sometimes, and the initializer that
294        // consumes the read exists only then. The statement is one `let`, so
295        // the attribute goes in front of it.
296        let guarded = &step.conditions;
297        match &step.instr {
298            Instr::Apply {
299                primitive,
300                operands,
301                result,
302            } => {
303                let applied = &primitives[primitive.0];
304                let operand_names: Vec<syn::Ident> = operands
305                    .iter()
306                    .map(|operand| operand_name(&names, function, operand))
307                    .collect();
308                let expression =
309                    operation(target, binding, reach, applied, &operand_names, helpers);
310                let statement = match (&applied.failure, result) {
311                    (None, Some(result)) => {
312                        let name = local(&mut names, &mut taken, *result);
313                        quote!(let #name = #expression;)
314                    }
315                    (None, None) => quote!(#expression;),
316                    (Some(failure), Some(result)) => {
317                        let name = local(&mut names, &mut taken, *result);
318                        let route = failure_arm(
319                            target,
320                            function,
321                            failure.category,
322                            &error_binding,
323                            helpers,
324                        );
325                        let ok = &ok_binding;
326                        let error = error_pattern(function, failure.category, &error_binding);
327                        quote! {
328                            let #name = match #expression {
329                                Ok(#ok) => #ok,
330                                Err(#error) => { #route }
331                            };
332                        }
333                    }
334                    (Some(failure), None) => {
335                        let route = failure_arm(
336                            target,
337                            function,
338                            failure.category,
339                            &error_binding,
340                            helpers,
341                        );
342                        let error = error_pattern(function, failure.category, &error_binding);
343                        quote! {
344                            if let Err(#error) = #expression { #route }
345                        }
346                    }
347                };
348                statements.push(quote!(#(#guarded)* #statement));
349            }
350            Instr::Construct {
351                name,
352                parts,
353                result,
354            } => {
355                let item = flat
356                    .struct_type(name.as_str())
357                    .expect("a construction names a struct the model declares");
358                let ident = &item.name;
359                let module = reach.of(&ident.to_string());
360                let head = quote!(#module::#ident);
361                // An initializer carries its own field's condition: the value
362                // it names was bound by a statement under the same one, so a
363                // field the source does not have is neither read nor filled in.
364                //
365                // Read from the model here, while the statements got theirs
366                // from `Part::conditions` at plan time. The two agree because
367                // both read the same field.
368                let bound: Vec<TokenStream> = item
369                    .fields
370                    .iter()
371                    .zip(parts)
372                    .map(|(field, value)| {
373                        let conditions = Writer.conditions(Conditioned::Field(field));
374                        let bound = field.bind(&names[value]);
375                        quote!(#(#conditions)* #bound)
376                    })
377                    .collect();
378                let value = Writer.shape_struct(item, head, &bound);
379                let name = local(&mut names, &mut taken, *result);
380                statements.push(quote!(#(#guarded)* let #name = #value;));
381            }
382            Instr::Call {
383                function: callee,
384                args,
385                result,
386            } => {
387                let module = reach.of(callee);
388                let callee = format_ident!("{callee}");
389                let args: Vec<syn::Ident> = args.iter().map(|value| names[value].clone()).collect();
390                let call = quote!(#module::#callee(#(#args),*));
391                // A source function returning nothing produces no value, and a
392                // `let` over it would name one nobody can use.
393                let statement = match result {
394                    Some(result) => {
395                        let name = local(&mut names, &mut taken, *result);
396                        quote!(let #name = #call;)
397                    }
398                    None => quote!(#call;),
399                };
400                statements.push(quote!(#(#guarded)* #statement));
401            }
402        }
403    }
404
405    let tail = match function.result {
406        Some(result) => {
407            let name = &names[&result];
408            quote!(#name)
409        }
410        None => quote!(),
411    };
412    let symbol = format_ident!("{}", function.symbol);
413    let abi = &function.abi;
414    let attrs = &function.attrs;
415    // A carrier whose bits are read as an integer holds whatever the caller
416    // put in it. A foreign caller passes a number, which the match checks;
417    // safe Rust could pass storage never initialized, which no check can
418    // look at. So the wrapper is `unsafe`, and says what it is owed.
419    let reads_bits = function.instrs.iter().any(|step| match &step.instr {
420        Instr::Apply { primitive, .. } => matches!(
421            primitives[primitive.0].implementation,
422            Implementation::Standard(StandardOp::EnumIn { bits: Some(_), .. })
423        ),
424        _ => false,
425    });
426    let unsafety = (function.unsafety || reads_bits).then(|| quote!(unsafe));
427    let safety = reads_bits.then(|| {
428        quote! {
429            /// # Safety
430            ///
431            /// Every argument that carries an enum as its bits must be
432            /// initialized. The wrapper reads those bits as an integer and
433            /// refuses a number no value has, but reading storage that was
434            /// never initialized is undefined behaviour.
435        }
436    });
437    let ret = function
438        .ret
439        .as_ref()
440        .map(|ty| quote!(-> #ty))
441        .unwrap_or_default();
442    quote! {
443        #safety
444        #(#conditions)*
445        #[no_mangle]
446        #(#attrs)*
447        pub #unsafety extern #abi fn #symbol(#(#params),*) #ret {
448            #(#statements)*
449            #tail
450        }
451    }
452}

454/// Keep `written`'s helpers, each once.
455fn keep(helpers: &mut Vec<Artifact>, written: Written) -> TokenStream {
456    for helper in written.helpers {
457        if !helpers.iter().any(|kept| kept.name == helper.name) {
458            helpers.push(helper);
459        }
460    }
461    written.text
462}

464/// One operation's expression: the registry writes its own, the target writes
465/// its own from what the plan feeds it.
466fn operation<T: Target>(
467    target: &T,
468    binding: &Binding<T>,
469    reach: &Reach,
470    applied: &Applied<T::Op>,
471    operands: &[syn::Ident],
472    helpers: &mut Vec<Artifact>,
473) -> TokenStream {
474    // The handle operations spell a source type, which is what makes them the
475    // registry's: an adapter has no way to, and no business doing it.
476    let source_type = |ty| Writer.emit_source_type(ty, &reach.modules, &reach.default);
477    let carrier_type = |slot: Option<Slot>| match slot {
478        Some(Slot::Carrier(carrier)) => {
479            let ty = &binding.carrier_of(carrier).rust;
480            quote!(#ty)
481        }
482        _ => source_type(&applied.subject),
483    };
484    let value = &operands[0];
485    match &applied.implementation {
486        Implementation::Standard(StandardOp::Identity) => quote!(#value),
487        Implementation::Standard(StandardOp::ReadMember) => {
488            let member = applied
489                .part
490                .as_ref()
491                .expect("a member read is applied to a part")
492                .member();
493            quote!(#value.#member)
494        }
495        Implementation::Standard(StandardOp::IntoRaw) => {
496            let carrier = carrier_type(applied.result);
497            quote!(Box::into_raw(Box::new(#value)) as #carrier)
498        }
499        Implementation::Standard(StandardOp::FromRaw) => {
500            let ty = source_type(&applied.subject);
501            let message = format!("null `{}` handle", applied.subject.key());
502            quote! {
503                ::core::ptr::NonNull::new(#value as *mut #ty)
504                    .map(|handle| unsafe { *Box::from_raw(handle.as_ptr()) })
505                    .ok_or_else(|| String::from(#message))
506            }
507        }
508        Implementation::Standard(StandardOp::EnumOut { values }) => {
509            let ty = source_type(&applied.subject);
510            let arms = values.iter().map(|arm| {
511                let pattern = enum_value(&ty, arm);
512                let carried = &arm.carried;
513                quote!(#pattern => #carried)
514            });
515            quote!(match #value { #(#arms),* })
516        }
517        Implementation::Standard(StandardOp::EnumIn {
518            values,
519            invalid,
520            bits,
521        }) => {
522            let value = match bits {
523                Some(bits) => quote!(unsafe { ::core::mem::transmute_copy::<_, #bits>(&(#value)) }),
524                None => quote!(#value),
525            };
526            let ty = source_type(&applied.subject);
527            let arms = values.iter().map(|arm| {
528                let carried = &arm.carried;
529                let constructed = enum_value(&ty, arm);
530                match invalid {
531                    Some(_) => quote!(#carried => ::core::result::Result::Ok(#constructed)),
532                    None => quote!(#carried => #constructed),
533                }
534            });
535            match invalid {
536                Some(message) => quote! {
537                    match #value {
538                        #(#arms,)*
539                        other => ::core::result::Result::Err(::std::format!(#message, other)),
540                    }
541                },
542                None => quote!(match #value { #(#arms),* }),
543            }
544        }
545        Implementation::Standard(StandardOp::Release) => {
546            let ty = source_type(&applied.subject);
547            quote! {
548                drop(::core::ptr::NonNull::new(#value as *mut #ty)
549                    .map(|handle| unsafe { Box::from_raw(handle.as_ptr()) }))
550            }
551        }
552        Implementation::Target(op) => {
553            let fed = |slot: Slot| match slot {
554                Slot::Source => Fed::Source(&applied.subject),
555                Slot::Carrier(carrier) => Fed::Carrier(binding.carrier_of(carrier)),
556            };
557            let feed = OperationFeed {
558                value: Some((value.clone(), fed(applied.value))),
559                contexts: applied
560                    .contexts
561                    .iter()
562                    .cloned()
563                    .zip(operands[1..].iter().cloned())
564                    .collect(),
565                error: None,
566                result: applied.result.map(fed),
567                part: applied.part.as_ref(),
568            };
569            keep(helpers, target.write_operation(op, &feed))
570        }
571    }
572}

574/// One value of a fieldless enum, spelled as the source declared it.
575///
576/// A pattern and a constructor are the same text for a fieldless value, so
577/// this serves both sides of the two enum operations. `Add` stays `Add`, and
578/// `Add()` and `Mul {}` keep their delimiters — the model's speller decides,
579/// from the shape it captured.
580fn enum_value(ty: &TokenStream, arm: &crate::target::EnumArm) -> TokenStream {
581    let name = &arm.name;
582    arm.shape.spell_fieldless(quote!(#ty::#name))
583}

585/// The pattern a failure arm binds the error with: the name when a reporter
586/// reads it, `_` when the route only terminates.
587fn error_pattern<Op>(
588    function: &FunctionPlan<Op>,
589    category: FailureCategory,
590    error: &syn::Ident,
591) -> TokenStream {
592    let reported = function
593        .routes
594        .iter()
595        .any(|route| route.category == category && route.report.is_some());
596    if reported {
597        quote!(#error)
598    } else {
599        quote!(_)
600    }
601}

603/// What one failure category does at this boundary: report if the form has an
604/// operation for it, then terminate.
605fn failure_arm<T: Target>(
606    target: &T,
607    function: &FunctionPlan<T::Op>,
608    category: FailureCategory,
609    error: &syn::Ident,
610    helpers: &mut Vec<Artifact>,
611) -> TokenStream {
612    let route = function
613        .routes
614        .iter()
615        .find(|route| route.category == category)
616        .expect("an unrouted failure category is refused before assembly");
617    let report = route.report.as_ref().map(|report| {
618        let contexts: Vec<(String, syn::Ident)> = report
619            .operation
620            .context
621            .iter()
622            .map(|name| (name.clone(), context_name(function, name)))
623            .collect();
624        let expression = match &report.operation.implementation {
625            Implementation::Target(op) => {
626                let feed = OperationFeed::<T> {
627                    value: None,
628                    contexts,
629                    error: Some(error.clone()),
630                    result: None,
631                    part: None,
632                };
633                keep(helpers, target.write_operation(op, &feed))
634            }
635            // A standard operation converts a value, and a report has none to
636            // convert; a binding that stated one is refused before assembly.
637            Implementation::Standard(_) => {
638                unreachable!("a reporting operation is the target's own")
639            }
640        };
641        let on_failure = terminal(&route.on_report_failure);
642        match report.operation.failure {
643            None => quote!(#expression;),
644            Some(_) => quote! {
645                if #expression.is_err() { #on_failure }
646            },
647        }
648    });
649    let terminate = terminal(&route.terminate);
650    quote! {
651        #report
652        #terminate
653    }
654}

656fn terminal(terminal: &Terminal) -> TokenStream {
657    match terminal {
658        Terminal::Abort => quote!(std::process::abort();),
659        Terminal::Return(expr) => quote!(return #expr;),
660    }
661}

663fn operand_name<Op>(
664    names: &HashMap<ValueId, syn::Ident>,
665    function: &FunctionPlan<Op>,
666    operand: &Operand,
667) -> syn::Ident {
668    match operand {
669        Operand::Value(id) => names[id].clone(),
670        Operand::Context(name) => context_name(function, name),
671    }

674/// An identifier's canonical spelling: `r#type` and `type` name one binding.
675fn unraw(ident: &syn::Ident) -> String {
676    ident.to_string().trim_start_matches("r#").to_string()
677}

679/// A name nothing in this wrapper already uses.
680fn free_name(preferred: &str, taken: &mut std::collections::HashSet<String>) -> syn::Ident {
681    if taken.insert(preferred.to_string()) {
682        return format_ident!("{preferred}");
683    }
684    for suffix in 1.. {
685        let candidate = format!("{preferred}_{suffix}");
686        if taken.insert(candidate.clone()) {
687            return format_ident!("{candidate}");
688        }
689    }
690    unreachable!("the search above only ends by returning")
691}

693/// The wrapper parameter supplying a runtime context.
694fn context_name<Op>(function: &FunctionPlan<Op>, context: &str) -> syn::Ident {
695    function
696        .params
697        .iter()
698        .find(|(_, param)| matches!(&param.role, ParamRole::Context(name) if name == context))
699        .map(|(_, param)| param.name.clone())
700        .expect("a missing runtime context is refused before assembly")
701}
