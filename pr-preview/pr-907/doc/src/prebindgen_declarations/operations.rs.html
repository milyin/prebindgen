4/// Wire tree prepared by one codec.
5pub type CodecNode<C> = WireNode<<C as Codec>::Wire, <C as Codec>::Chain, <C as Codec>::Branch>;
6/// Prepared expressions keyed by their wire binding names.
7pub type WireValues = Vec<(syn::Ident, TokenStream)>;
8
9/// Direction in which a source value crosses the boundary.
10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11pub enum Direction {
12    /// Foreign values become a Rust argument or callback result.
13    Input,
14    /// A Rust value becomes foreign result or callback argument values.
15    Output,
16}
17/// Physical representation selected by the declaration compiler.
18#[derive(Clone, Copy, Debug, PartialEq, Eq)]
19pub enum Representation {
20    /// The adapter's normal representation, also inside ordinary sequence rows.
21    Ordinary,
22    /// Native fields inside an explicitly selected bulk frame.
23    Bulk,
24}
25/// Source operation for which an input is being prepared.
26#[derive(Clone, Copy, Debug, PartialEq, Eq)]
27pub enum InputRole {
28    /// One ordinary function parameter, including its nested fields.
29    Parameter,
30    /// An owned value returned by a retained source callback.
31    CallbackResult,
32}
33/// Access to the Rust expression supplied to an output leaf.
34#[derive(Clone, Copy, Debug, PartialEq, Eq)]
35pub enum ValueAccess {
36    /// The expression has the declared type T; a reference T does not own its pointee.
37    Owned,
38    /// The expression observes the declared source value through &T.
39    Shared,
40}
41impl ValueAccess {
42    /// Whether the supplied expression is a shared observation.
43    pub fn is_shared(self) -> bool {
44        self == Self::Shared
45    }
46}

48/// Prepare one source input leaf in an explicit representation and scope.
49pub struct InputLeaf<'a, C: Codec + ?Sized> {
50    /// Authoritative source and binding declarations.
51    pub flat: &'a Flat<C::Schema>,
52    /// Source Rust type at this position.
53    pub ty: &'a Type,
54    /// Borrowed occurrence configuration.
55    pub extra: &'a C::Extra,
56    /// Unique boundary binding stem.
57    pub wire: syn::Ident,
58    /// Rust binding for the constructed argument.
59    pub target: syn::Ident,
60    /// Branches under which this value is active.
61    pub active: &'a Condition,
62    /// Ordinary values or native bulk fields.
63    pub representation: Representation,
64    /// Ordinary parameter or source callback result.
65    pub role: InputRole,
66}
67impl<'a, C: Codec + ?Sized> InputLeaf<'a, C> {
68    /// Request an ordinary leaf for an enclosing function-parameter conversion.
69    pub fn new(
70        flat: &'a Flat<C::Schema>,
71        ty: &'a Type,
72        extra: &'a C::Extra,
73        wire: syn::Ident,
74        target: syn::Ident,
75        active: &'a Condition,
76    ) -> Self {
77        Self {
78            flat,
79            ty,
80            extra,
81            wire,
82            target,
83            active,
84            representation: Representation::Ordinary,
85            role: InputRole::Parameter,
86        }
87    }
88}

129/// Meaning of a group in the adapter's wire tree.
130#[derive(Clone, Copy, Debug)]
131pub enum GroupKind<'a> {
132    /// Ordered constructor arguments or combined output observations.
133    Values,
134    /// Source fields, including unit's empty group.
135    Fields,
136    /// One named source field and its entire subtree.

138    /// An explicitly named output view.
139    NamedOutput(&'a str),

142/// Choose an optional encoding and update its payload's remaining niche offers.
143pub struct Optional<'a, C: Codec + ?Sized> {
144    /// Borrowed occurrence configuration.
145    pub extra: &'a C::Extra,
146    /// Temporary payload tree; its native scope determines available discriminators.
147    pub payload: &'a mut CodecNode<C>,
148    /// Requested normalized 0/1 presence binding.
149    pub presence: syn::Ident,
150    /// Input needs a presence test; output needs inactive expressions.
151    pub direction: Direction,
152}
153/// Presence encoding selected by one optional preparation operation.
154pub enum Presence<W: prebindgen_flat::WireKind> {
155    /// One i32-compatible 0/1 control, named by the requested presence binding.
156    Flag(Wire<W>),
157    /// No added wire. Input requires a total, side-effect-free boolean test;
158    /// output requires None because it branches on the source Rust Option.
159    Niche {
160        /// Test supplied only in input direction.
161        input_test: Option<TokenStream>,
162    },
163}
164/// Complete physical optional encoding, without duplicating its payload tree.
165pub struct OptionalEncoding<C: Codec + ?Sized> {
166    /// Separate flag or a reserved representation in an existing payload wire.
167    pub presence: Presence<C::Wire>,
168    /// Adapter metadata retaining the selected discriminator/reservation.
169    pub metadata: C::Chain,
170    /// Output expressions for every inactive payload wire. The niche donor must
171    /// produce its reserved value. Empty for input.
172    pub absent_values: WireValues,
173}
174/// Prepare metadata and inactive arm offers after the compiler names the selector.
175pub struct Alternatives<'a, C: Codec + ?Sized> {
176    /// The selector after its source path has already been prefixed.
177    pub selector: &'a Wire<C::Wire>,
178    /// Prepared arm trees; retire offers in inactive values within each native scope.
179    pub arms: &'a mut [CodecNode<C>],
180}
