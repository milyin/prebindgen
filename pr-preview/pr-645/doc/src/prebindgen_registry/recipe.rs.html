1//! The table of crossings, and the recipes that answer them.
2//!
3//! A **crossing** is one Rust type plus one of two directions — *construct* a Rust
4//! value out of the wire values that arrived, or *deconstruct* one into the
5//! wire values that leave. [`Recipes`] is the table that answers it, and a
6//! **recipe** is one answer: which `#[prebindgen]` constructor assembles the
7//! value, or which accessors take it apart.
8//!
9//! A recipe says nothing about the wire. How many wire values a crossing costs,
10//! how they are encoded and what they cost to clean up is the adapter's answer,
11//! and no part of it is stored here.
12//!
13//! A recipe also says nothing the model already says. Part count, part types,
14//! ownership, names and fallibility are all read off [`flat`](crate::flat), so
15//! a recipe cannot contradict the Rust it describes. What is left is short: a
16//! whole product is one identifier.
17//!
18//! [`RecipeName`] is reusable adapter vocabulary such as `whole` or `parts`.
19//! [`RecipeKey`] pairs that name with a [`CrossingKey`] and uniquely identifies
20//! one row throughout the table. Resolved bindings and compiled fragments carry
21//! the full key rather than a context-dependent name.
22//!
23//! # What a crossing is keyed by
24//!
25//! A borrow is not part of the crossing. `Sample`, `&Sample` and `Box<Sample>`
26//! are one recipe, because the same recipe assembles all three; whether the value
27//! is handed over or reached through a borrow is [`Crossing::mode`], read off
28//! the type. So an adapter declares `Sample` once and every site finds it.
29//!
30//! # Recipes nobody declares
31//!
32//! A type with no declared recipe still gets one, derived from its kind:
33//! `Option<T>` yields [`Shape::Optional`], `Vec<T>` and `&[T]` and `[T; N]`
34//! yield [`Shape::Sequence`], and everything else yields [`Shape::Atomic`].
35//! Nesting needs no rule of its own: a recipe names one layer, and the layer
36//! inside it is a crossing with a recipe of its own.
37//!
38//! A callback type yields [`Shape::Invoke`], and that is the only shape such a
39//! crossing takes: converting the arguments is what makes the callable
40//! callable, so there is no second answer to choose between.
41//! [`RecipesBuilder::build`] refuses any other shape there, and an `Invoke` recipe's
42//! parts are the callback's arguments, which take the other direction.
43

67/// Which of the two directions a crossing is, as a value.
68#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
69pub enum Direction {
70    /// Build a Rust value out of the wire values that arrived.
71    Construct,
72    /// Take a Rust value apart into the wire values that leave.
73    Deconstruct,
74}
75
76impl Direction {
77    /// The other direction. Only a callback swaps: the Rust side holds the values
78    /// arguments carry and pushes them out through the call.
79    pub fn swap(self) -> Self {
80        match self {
81            Direction::Construct => Direction::Deconstruct,
82            Direction::Deconstruct => Direction::Construct,
83        }
84    }
85}
86
87impl fmt::Display for Direction {
88    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
89        f.write_str(match self {
90            Direction::Construct => "construct",
91            Direction::Deconstruct => "deconstruct",
92        })
93    }
94}

96/// The same two directions at the type level, stated on the operations themselves.
97///
98/// An adapter never writes this bound and never names an implementor: `OP` is
99/// inferred from the shape handed to [`RecipesBuilder::declare`], which is what
100/// files a recipe under the right direction without anything stating it twice.
101pub trait Operation: Sized {
102    /// Which direction an operation of this type takes.
103    const DIRECTION: Direction;
104
105    /// Erase the direction so the table can hold both. Not part of the surface an
106    /// adapter writes against.
107    #[doc(hidden)]
108    fn into_recipe(shape: Shape<Self>) -> Recipe;
109}

111/// What assembles a value from its parts.
112#[derive(Clone, Debug)]
113pub enum Construct {
114    /// Call this `#[prebindgen]` constructor. **Its parameters are the parts** —
115    /// how many, in what order, named what, typed what, taken by value or by
116    /// reference. All of that is the model's
117    /// [`Function`], so none of it is restated here, and
118    /// whether the call is fallible is its return type.
119    Call(syn::Ident),
120    /// Write the value's own fields. **They are the parts**, in the model's
121    /// order, and every one of them contributes.
122    ///
123    /// The adapter-synthesized case, where nothing is called: a `#[repr(C)]`
124    /// mirror of a struct with public fields is rebuilt as a struct literal,
125    /// with no constructor in the source crate to name. Unlike
126    /// [`Deconstruct::Fields`] it names no list, because a value cannot be
127    /// built with one of its fields missing, and which fields there are is the
128    /// model's answer.
129    ///
130    /// Inside a [`Shape::Choice`] arm the fields are that alternative's.
131    Fields,
132}
133
134impl Operation for Construct {
135    const DIRECTION: Direction = Direction::Construct;
136
137    fn into_recipe(shape: Shape<Self>) -> Recipe {
138        Recipe::Constructing(shape)
139    }
140}

142/// What takes a value apart.
143#[derive(Clone, Debug)]
144pub enum Deconstruct {
145    /// Read the parts off the value where it stands.
146    Fields(Vec<Reach>),
147    /// Call this accessor once, bind its result, and read the parts off that
148    /// binding. Whether the accessor consumes the value is read from its own
149    /// first parameter, not stated again.
150    ValueForm {
151        /// The accessor to call, whose first parameter is the recipe's own type.
152        func: syn::Ident,
153        /// Where each part of the bound result comes from.
154        parts: Vec<Reach>,
155    },
156}
157
158impl Operation for Deconstruct {
159    const DIRECTION: Direction = Direction::Deconstruct;
160
161    fn into_recipe(shape: Shape<Self>) -> Recipe {
162        Recipe::Deconstructing(shape)
163    }
164}

166/// Where one part comes from.
167///
168/// Each form yields the part's type, its ownership and its name, so a recipe
169/// never repeats any of the three.
170#[derive(Clone, Debug)]
171pub enum Reach {
172    /// Field `index` of the model's [`Struct`](crate::flat::Struct) — or,
173    /// inside an arm, of that alternative. The adapter-synthesized case, where
174    /// nothing is called.
175    Field(usize),
176    /// A `#[prebindgen]` accessor `fn(&T) -> P`, or `fn(T) -> P` to consume.
177    Accessor(syn::Ident),
178    /// This position contributes nothing.
179    Omit,
180    /// The value itself, as one part — cloned from a borrow, moved from an
181    /// owned receiver. The form `DeconRecord::Identity` states and no reach
182    /// could: `Field` indexes into a product and `Accessor` calls out of one,
183    /// while a handle leaf is the whole value with nothing between (#613 step
184    /// 10).
185    ///
186    /// The part's type is the receiver, so it resolves through the crossing's
187    /// DEFAULT row rather than the row being compiled — otherwise the compiler
188    /// re-enters the same crossing and recurses until the stack runs out. That
189    /// default is the value's own converter, which is exactly what a handle
190    /// leaf delivers.
191    ///
192    /// So an identity row works **beside** a default row and not instead of
193    /// one: `whole` stays the default and `handle` states `Identity`, and a
194    /// site asks for `handle`. A type whose ONLY row is the identity one has a
195    /// default that is itself, which is genuinely circular and is refused as a
196    /// cycle.
197    Identity,
198}

200/// Whether a value is handed over, or reached through a borrow.
201#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
202pub enum Mode {
203    /// Handed over. The receiver may keep it.
204    Owned,
205    /// Reached through `&T`.
206    Shared,
207    /// Reached through `&mut T`.
208    Exclusive,
209}
210
211impl Mode {
212    /// How a value of mode `self` is held when it is reached **through** a
213    /// container held as `outer`.
214    ///
215    /// Both layers decide it, and neither alone: reading through a shared
216    /// `&Option<T>` can only ever lend its value, however the value is spelled,
217    /// and a `Vec<&T>` hands its elements over while what it hands over is a
218    /// borrow. Composing them is what a nested reading needs and what asking
219    /// either layer on its own gets wrong in both directions — refusing the one
220    /// correct fragment, or accepting one the hook cannot be fed.
221    ///
222    /// | held through | this mode | reached as |
223    /// |---|---|---|
224    /// | owned | any | itself — the container gives its contents up |
225    /// | `&` | any | `&` — a shared view yields nothing stronger |
226    /// | `&mut` | `&` | `&` — an exclusive view of a shared reference is still shared |
227    /// | `&mut` | owned or `&mut` | `&mut` |
228    pub fn through(self, outer: Mode) -> Mode {
229        match outer {
230            Mode::Owned => self,
231            Mode::Shared => Mode::Shared,
232            Mode::Exclusive => match self {
233                Mode::Shared => Mode::Shared,
234                Mode::Owned | Mode::Exclusive => Mode::Exclusive,
235            },
236        }
237    }
238
239    /// Whether a part produced in this mode can be consumed where `wanted` is
240    /// required.
241    ///
242    /// **Owning it is enough for anything**: a value handed over can be
243    /// consumed, lent, or lent mutably, so an owned production serves every
244    /// edge. A borrow serves only its own kind — a `&T` cannot be consumed and
245    /// cannot be written through, and a `&mut T` is not what an edge asking for
246    /// `&T` was written against.
247    ///
248    /// The rule fires only where an adapter's fragment **disagrees** with how
249    /// its crossing is spelled, since both sides are otherwise read off the same
250    /// type. `prebindgen-jni`'s borrowed opaque output is the live example: it
251    /// clones its referent into a fresh handle, so it produces owned where the
252    /// crossing says `&T`.
253    pub fn satisfies(self, wanted: Mode) -> bool {
254        self == Mode::Owned || self == wanted
255    }
256}
257
258impl fmt::Display for Mode {
259    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
260        f.write_str(match self {
261            Mode::Owned => "owned",
262            Mode::Shared => "&",
263            Mode::Exclusive => "&mut",
264        })
265    }
266}

270/// One Rust type and one of the two directions: the question the table answers.
271#[derive(Clone, Debug)]
272pub struct Crossing {
273    ty: TypeRef,
274    direction: Direction,
275}
276
277impl Crossing {
278    /// The crossing of `ty`, doing `direction`.
279    ///
280    /// `ty` is kept exactly as the site spelled it, borrow and transparent
281    /// wrappers included; what is normalized away is only the key
282    /// [`Self::key`] derives from it.
283    pub fn new(ty: TypeRef, direction: Direction) -> Self {
284        Self { ty, direction }
285    }
286
287    /// The type as the site spelled it.
288    pub fn spelled(&self) -> &TypeRef {
289        &self.ty
290    }
291
292    /// The Rust value that crosses: the spelled type with a borrow peeled off.
293    pub fn value(&self) -> &TypeRef {
294        self.ty.borrow_target().unwrap_or(&self.ty)
295    }
296
297    /// Which direction this crossing takes.
298    pub fn direction(&self) -> Direction {
299        self.direction
300    }
301
302    /// Whether the value is handed over or reached through a borrow.
303    pub fn mode(&self) -> Mode {
304        match self.ty.unwrapped().kind() {
305            TypeKind::Ref { mutable: true, .. } => Mode::Exclusive,
306            TypeKind::Ref { .. } => Mode::Shared,
307            _ => Mode::Owned,
308        }
309    }
310
311    /// The erased form, for maps and diagnostics.
312    pub fn key(&self) -> CrossingKey {
313        CrossingKey {
314            ty: self.value().stripped_key(),
315            direction: self.direction,
316        }
317    }
318
319    /// The row named `name` under this crossing's normalized key.
320    pub fn row(&self, name: RecipeName) -> RecipeKey {
321        self.key().row(name)
322    }
323}

325/// A crossing identified rather than described.

327pub struct CrossingKey {

378///
379/// A [`RecipeKey`] identifies that position whether or not the table currently

418/// How a value gets across, in terms of its parts.
419///
420/// `OP` is the one thing a recipe states differently between the two directions:
421/// what assembles the parts into a value, or takes them out of it. Nothing
422/// below the top restates the direction, because a part, an optional's value and
423/// a sequence's element all take the same direction as the recipe.
424#[derive(Clone, Debug)]
425pub enum Shape<OP> {
426    /// No parts. The adapter emits the conversion itself; how many wire values
427    /// that costs is the adapter's business and the table never asks.
428    Atomic,
429    /// Absent, or the value the `Option` wraps.
430    ///
431    /// The inner type is the crossing's own — the payload of `Option<T>` is
432    /// `T` — so it is read off the type rather than stated here.
433    Optional,
434    /// A run of the collection's element.
435    ///
436    /// The element type is the crossing's own, and whether iterating yields
437    /// owned values or borrows is the collection's business too: `Vec<T>` gives
438    /// its elements up, `&[T]` and `Cow<'_, [T]>` lend them.
439    Sequence,
440    /// Every part contributes.
441    Product(OP),
442    /// Exactly one arm is live at run time. Every arm still compiles.
443    Choice {
444        /// One entry per alternative that crosses.
445        arms: Vec<Arm<OP>>,
446    },
447    /// A callable the foreign side supplied, taken apart into the values that
448    /// pass through it.
449    ///
450    /// The only shape a crossing of a callback type may be declared with:
451    /// converting the arguments is what makes the callable callable, so there
452    /// is no second way for such a crossing to be answered. It names no
453    /// arguments — they are the ones the crossing's own type carries — and
454    /// their direction is the recipe's swapped, which the table applies rather than any
455    /// declaration stating it.
456    Invoke,
457}
458
459/// One alternative of the model's [`Variant`](crate::flat::Variant).
460///
461/// The alternative's name and its position come with it, and how the foreign
462/// side is told which arm is live is the adapter's choice, which is why no tag
463/// is stated here.
464#[derive(Clone, Debug)]
465pub struct Arm<OP> {
466    /// Position of the alternative within its sum.
467    pub alternative: usize,
468    /// What assembles this arm's payload from its parts, or takes it out.
469    pub op: OP,
470}

472/// A recipe that builds a Rust value out of what arrived.
473pub type Constructing = Shape<Construct>;
474/// A recipe that takes a Rust value apart into what leaves.
475pub type Deconstructing = Shape<Deconstruct>;
476
477/// One recipe, under whichever direction it takes.
478///
479/// The table holds recipes in this form so both directions share one map. An adapter
480/// writes a [`Constructing`] or a [`Deconstructing`] and never builds or
481/// matches on a `Recipe`.
482#[derive(Clone, Debug)]
483pub enum Recipe {
484    /// A recipe filed under [`Direction::Construct`].
485    Constructing(Constructing),
486    /// A recipe filed under [`Direction::Deconstruct`].
487    Deconstructing(Deconstructing),
488}
489
490impl Recipe {
491    /// Whether this recipe takes a callable apart — the one shape whose parts do
492    /// the other direction.
493    pub fn is_invoke(&self) -> bool {
494        matches!(
495            self,
496            Recipe::Constructing(Shape::Invoke) | Recipe::Deconstructing(Shape::Invoke)
497        )
498    }
499
500    /// Which direction this recipe takes.
501    pub fn direction(&self) -> Direction {
502        match self {
503            Recipe::Constructing(_) => Direction::Construct,
504            Recipe::Deconstructing(_) => Direction::Deconstruct,
505        }
506    }
507}

511#[derive(Clone, Debug)]
512struct Entry {
513    key: RecipeKey,
514    recipe: Recipe,
515    /// The type as the declaration spelled it, which is what the checks read
516    /// fields and alternatives off.
517    ty: TypeRef,
518    default: bool,
519}
520
521/// The resolved table. Built by [`RecipesBuilder`], checked once, then
522/// immutable.
523#[derive(Clone, Debug, Default)]
524pub struct Recipes {
525    recipes: HashMap<CrossingKey, Vec<Entry>>,
526}
527
528impl Recipes {
529    /// Start describing a table.
530    pub fn builder() -> RecipesBuilder {
531        RecipesBuilder::default()
532    }
533
534    /// Which recipe names this crossing has, in declaration order.
535    ///
536    /// Empty for a crossing nobody declared, which still has the recipe
537    /// [`Self::recipe`] derives.
538    pub fn names_of(&self, key: &CrossingKey) -> Vec<&RecipeName> {
539        self.recipes
540            .get(key)
541            .map(|recipes| recipes.iter().map(|e| e.key.name()).collect())
542            .unwrap_or_default()
543    }
544
545    /// The key of one named row under a crossing, or `None` if it was never declared.
546    pub fn key_of(&self, crossing: &CrossingKey, name: &RecipeName) -> Option<&RecipeKey> {
547        self.recipes
548            .get(crossing)?
549            .iter()
550            .find(|e| e.key.name() == name)
551            .map(|e| &e.key)
552    }
553
554    /// One row by its globally unique key.
555    pub fn get(&self, key: &RecipeKey) -> Option<&Recipe> {
556        self.recipes
557            .get(key.crossing())?
558            .iter()
559            .find(|e| &e.key == key)
560            .map(|e| &e.recipe)
561    }
562
563    /// The recipe a site uses when it names none.
564    ///
565    /// With one declared recipe that recipe is the default; with several it is the
566    /// one declared through [`RecipesBuilder::declare_default`]. `None` for a
567    /// crossing nobody declared.
568    pub fn default_of(&self, key: &CrossingKey) -> Option<&RecipeKey> {
569        let recipes = self.recipes.get(key)?;
570        match recipes.as_slice() {
571            [only] => Some(&only.key),
572            many => many.iter().find(|e| e.default).map(|e| &e.key),
573        }
574    }
575
576    /// The default recipe for a crossing: the declared one, or the recipe derived
577    /// from the type's kind.
578    ///
579    /// A callback type derives [`Shape::Invoke`], the only shape such a
580    /// crossing takes, so an adapter reaches one here the same way it reaches
581    /// every other recipe.
582    pub fn recipe(&self, crossing: &Crossing) -> (RecipeKey, Cow<'_, Recipe>) {
583        let key = crossing.key();
584        match self.default_of(&key) {
585            Some(recipe_key) => {
586                let recipe_key = recipe_key.clone();
587                let recipe = self
588                    .get(&recipe_key)
589                    .expect("default names a declared recipe");
590                (recipe_key, Cow::Borrowed(recipe))
591            }
592            None => (RecipeKey::derived(key), Cow::Owned(derive(crossing))),
593        }
594    }
595}
596
597/// The arity recipe a crossing gets when nobody declared one.
598fn derive(crossing: &Crossing) -> Recipe {
599    let value = crossing.value();
600    let kind = if value.callback_args().is_some() {
601        DerivedKind::Invoke
602    } else if value.optional_inner().is_some() {
603        DerivedKind::Optional
604    } else if sequence_elem(value).is_some() {
605        DerivedKind::Sequence
606    } else {
607        DerivedKind::Atomic
608    };
609    match crossing.direction {
610        Direction::Construct => Recipe::Constructing(kind.shape()),
611        Direction::Deconstruct => Recipe::Deconstructing(kind.shape()),
612    }
613}

615/// The shape a type's own kind implies, before either direction is chosen. Shared so
616/// the two arms of [`derive`] cannot drift apart.
617#[derive(Copy, Clone)]
618enum DerivedKind {
619    Atomic,
620    Optional,
621    Sequence,
622    Invoke,
623}
624
625impl DerivedKind {
626    fn shape<OP>(self) -> Shape<OP> {
627        match self {
628            DerivedKind::Atomic => Shape::Atomic,
629            DerivedKind::Optional => Shape::Optional,
630            DerivedKind::Sequence => Shape::Sequence,
631            DerivedKind::Invoke => Shape::Invoke,
632        }
633    }
634}

636/// The element of a run, including the fixed-size array the model's own
637/// [`TypeRef::sequence_elem`](crate::flat::TypeRef::sequence_elem) leaves out.
638pub(crate) fn sequence_elem(ty: &TypeRef) -> Option<&TypeRef> {
639    if let Some(elem) = ty.sequence_elem() {
640        return Some(elem);
641    }
642    match ty.unwrapped().kind() {
643        TypeKind::Array { elem, .. } => Some(elem),
644        _ => None,
645    }
646}

648/// Describes a [`Recipes`] table.
649#[derive(Default)]
650pub struct RecipesBuilder {
651    recipes: HashMap<CrossingKey, Vec<Entry>>,
652    /// A recipe declared twice under one name, reported by [`Self::build`] rather
653    /// than by overwriting silently.
654    duplicates: Vec<RecipeKey>,
655}
656
657impl RecipesBuilder {
658    /// Add one recipe for `ty`.
659    ///
660    /// Which direction the recipe is filed under is the shape's own, so nothing states
661    /// it twice and the two cannot disagree. Declaring a second recipe for one
662    /// crossing is how a type offers a choice, and the site is what picks
663    /// between them — at which point one of the recipes has to be declared through
664    /// [`Self::declare_default`].
665    pub fn declare<OP: Operation>(
666        &mut self,
667        ty: TypeRef,
668        name: RecipeName,
669        shape: Shape<OP>,
670    ) -> &mut Self {
671        self.insert(ty, name, OP::into_recipe(shape), false)
672    }
673
674    /// Add one recipe and make it the recipe used where a site names none.
675    ///
676    /// Needed only once a crossing has more than one recipe: with a single recipe
677    /// that recipe is the default, so the common case never says so.
678    pub fn declare_default<OP: Operation>(
679        &mut self,
680        ty: TypeRef,
681        name: RecipeName,
682        shape: Shape<OP>,
683    ) -> &mut Self {
684        self.insert(ty, name, OP::into_recipe(shape), true)
685    }
686
687    fn insert(
688        &mut self,
689        ty: TypeRef,
690        name: RecipeName,
691        recipe: Recipe,
692        default: bool,
693    ) -> &mut Self {
694        let crossing = Crossing::new(ty.clone(), recipe.direction()).key();
695        let key = crossing.row(name);
696        let entries = self.recipes.entry(crossing).or_default();
697        if entries.iter().any(|e| e.key == key) {
698            self.duplicates.push(key);
699            return self;
700        }
701        entries.push(Entry {
702            key,
703            recipe,
704            ty,
705            default,
706        });
707        self
708    }
709
710    /// Check the table and freeze it.
711    ///
712    /// Every problem is reported, not just the first. The checks are the ones
713    /// no type can express: whether a recipe names something the model has,
714    /// whether a crossing with several recipes says which of them wins, and
715    /// whether a recipe reaches its own crossing.
716    pub fn build(self, model: &Flat) -> Result<Recipes, Vec<RecipeError>> {
717        let table = Recipes {
718            recipes: self.recipes,
719        };
720        let mut errors: Vec<RecipeError> = self
721            .duplicates
722            .into_iter()
723            .map(|recipe| RecipeError::Duplicate { recipe })
724            .collect();
725
726        for (key, entries) in &table.recipes {
727            if entries.len() > 1 {
728                let defaults: Vec<RecipeName> = entries
729                    .iter()
730                    .filter(|e| e.default)
731                    .map(|e| e.key.name().clone())
732                    .collect();
733                if defaults.len() != 1 {
734                    errors.push(RecipeError::NoDefault {
735                        crossing: key.clone(),
736                        defaults,
737                    });
738                }
739            }
740            for entry in entries {
741                let crossing = Crossing::new(entry.ty.clone(), entry.recipe.direction());
742                // A callback type takes `Invoke` and nothing else: converting
743                // the arguments is what makes the callable callable, so there
744                // is no second answer for such a crossing.
745                if crossing.value().callback_args().is_some() && !entry.recipe.is_invoke() {
746                    errors.push(RecipeError::CallbackShape {
747                        recipe: entry.key.clone(),
748                    });
749                    continue;
750                }
751                let mut check = Check {
752                    model,
753                    recipe: &entry.key,
754                    errors: &mut errors,
755                    arm_fields: None,
756                };
757                check.recipe(&entry.ty, &entry.recipe);
758            }
759        }
760
761        cycles(model, &table, &mut errors);
762
763        if errors.is_empty() {
764            Ok(table)
765        } else {
766            Err(errors)
767        }
768    }
769}

771// ── Checking one recipe ──────────────────────────────────────────────────────

773struct Check<'a, 'e> {
774    model: &'a Flat,
775    recipe: &'a RecipeKey,
776    errors: &'e mut Vec<RecipeError>,
777    /// The payload fields of the arm being checked, which is what a
778    /// [`Reach::Field`] indexes inside a [`Shape::Choice`].
779    arm_fields: Option<Vec<Field>>,
780}
781
782impl<'a> Check<'a, '_> {
783    fn push(&mut self, error: RecipeError) {
784        self.errors.push(error);
785    }
786
787    fn out_of_range(&mut self, index: usize, len: usize) {
788        self.push(RecipeError::OutOfRange {
789            recipe: self.recipe.clone(),
790            index,
791            len,
792        });
793    }
794
795    fn not_a_product(&mut self) {
796        self.push(RecipeError::NotAProduct {
797            recipe: self.recipe.clone(),
798        });
799    }
800
801    fn not_a_callback(&mut self) {
802        self.push(RecipeError::WrongShape {
803            recipe: self.recipe.clone(),
804            shape: "Invoke",
805            wanted: "a callback type",
806        });
807    }
808
809    fn wrong_arity(&mut self, optional: bool) {
810        let (shape, wanted) = if optional {
811            ("Optional", "an `Option`")
812        } else {
813            ("Sequence", "a `Vec`, slice or array")
814        };
815        self.push(RecipeError::WrongShape {
816            recipe: self.recipe.clone(),
817            shape,
818            wanted,
819        });
820    }
821
822    /// Every part crossing this recipe reaches, checking what it names on the way.
823    fn recipe(&mut self, ty: &TypeRef, recipe: &Recipe) -> Vec<Crossing> {
824        let direction = recipe.direction();
825        // The one place the two directions swap: the Rust side holds the values a
826        // callback's arguments carry and pushes them out through the call.
827        if recipe.is_invoke() {
828            let value = Crossing::new(ty.clone(), direction);
829            let Some(args) = value.value().callback_args().map(<[_]>::to_vec) else {
830                self.not_a_callback();
831                return Vec::new();
832            };
833            return args
834                .into_iter()
835                .map(|ty| Crossing::new(ty, direction.swap()))
836                .collect();
837        }
838        let parts = match recipe {
839            Recipe::Constructing(shape) => self.constructing(ty, shape),
840            Recipe::Deconstructing(shape) => self.deconstructing(ty, shape),
841        };
842        parts
843            .into_iter()
844            .map(|ty| Crossing::new(ty, direction))
845            .collect()
846    }
847
848    /// The inner crossing an arity shape reaches, or a refusal naming the
849    /// shape the type does not have.
850    ///
851    /// Stated nowhere in the recipe: the payload of `Option<T>` is `T` and the
852    /// element of `Vec<T>` is `T`, so declaring either would only be a fact
853    /// that could disagree with the type.
854    fn arity_inner(&mut self, ty: &TypeRef, optional: bool) -> Vec<TypeRef> {
855        let inner = if optional {
856            ty.optional_inner()
857        } else {
858            sequence_elem(ty)
859        };
860        match inner {
861            Some(inner) => vec![inner.clone()],
862            None => {
863                self.wrong_arity(optional);
864                Vec::new()
865            }
866        }
867    }
868
869    fn constructing(&mut self, ty: &TypeRef, shape: &Constructing) -> Vec<TypeRef> {
870        match shape {
871            Shape::Atomic => Vec::new(),
872            Shape::Optional => self.arity_inner(ty, true),
873            Shape::Sequence => self.arity_inner(ty, false),
874            // Reached through `is_invoke` above, never here.
875            Shape::Invoke => Vec::new(),
876            Shape::Product(op) => self.construct(ty, op),
877            Shape::Choice { arms } => {
878                let Some(alternatives) = self.alternatives(ty) else {
879                    self.not_a_product();
880                    return Vec::new();
881                };
882                let mut parts = Vec::new();
883                for arm in arms {
884                    let Some(fields) = alternatives.get(arm.alternative) else {
885                        self.out_of_range(arm.alternative, alternatives.len());
886                        continue;
887                    };
888                    // The arm's payload stands in for the sum's own fields,
889                    // which the model gives a sum none of.
890                    let outer = self.arm_fields.replace(fields.clone());
891                    parts.extend(self.construct(ty, &arm.op));
892                    self.arm_fields = outer;
893                }
894                parts
895            }
896        }
897    }
898
899    fn deconstructing(&mut self, ty: &TypeRef, shape: &Deconstructing) -> Vec<TypeRef> {
900        match shape {
901            Shape::Atomic => Vec::new(),
902            Shape::Optional => self.arity_inner(ty, true),
903            Shape::Sequence => self.arity_inner(ty, false),
904            // Reached through `is_invoke` above, never here.
905            Shape::Invoke => Vec::new(),
906            Shape::Product(op) => self.deconstruct(ty, op),
907            Shape::Choice { arms } => {
908                let Some(alternatives) = self.alternatives(ty) else {
909                    self.not_a_product();
910                    return Vec::new();
911                };
912                let mut parts = Vec::new();
913                for arm in arms {
914                    let Some(fields) = alternatives.get(arm.alternative) else {
915                        self.out_of_range(arm.alternative, alternatives.len());
916                        continue;
917                    };
918                    // The arm's payload stands in for the sum's own fields,
919                    // which the model gives a sum none of.
920                    let outer = self.arm_fields.replace(fields.clone());
921                    parts.extend(self.deconstruct(ty, &arm.op));
922                    self.arm_fields = outer;
923                }
924                parts
925            }
926        }
927    }
928
929    fn construct(&mut self, ty: &TypeRef, op: &Construct) -> Vec<TypeRef> {
930        match op {
931            Construct::Call(func) => match self.function(func) {
932                Some(f) => {
933                    let parts = f.params.iter().map(|p| p.ty.clone()).collect();
934                    if !constructor_of(f, ty) {
935                        self.push(RecipeError::NotAConstructor {
936                            recipe: self.recipe.clone(),
937                            func: func.clone(),
938                        });
939                    }
940                    parts
941                }
942                None => Vec::new(),
943            },
944            Construct::Fields => match self.fields(ty) {
945                Some(fields) => fields.into_iter().map(|f| f.ty).collect(),
946                None => {
947                    self.not_a_product();
948                    Vec::new()
949                }
950            },
951        }
952    }
953
954    fn deconstruct(&mut self, ty: &TypeRef, op: &Deconstruct) -> Vec<TypeRef> {
955        match op {
956            Deconstruct::Fields(reaches) => self.reaches(ty, reaches),
957            Deconstruct::ValueForm { func, parts } => {
958                let Some(f) = self.function(func) else {
959                    return Vec::new();
960                };
961                let bound = f.ret.clone();
962                if !accessor_of(f, ty) {
963                    self.push(RecipeError::NotAnAccessor {
964                        recipe: self.recipe.clone(),
965                        func: func.clone(),
966                    });
967                }
968                // Parts are read off the bound result, so the arm's payload no
969                // longer stands in for them.
970                let outer = self.arm_fields.take();
971                let parts = self.reaches(&bound, parts);
972                self.arm_fields = outer;
973                parts
974            }
975        }
976    }
977
978    fn reaches(&mut self, ty: &TypeRef, reaches: &[Reach]) -> Vec<TypeRef> {
979        let mut out = Vec::new();
980        for reach in reaches {
981            match reach {
982                Reach::Omit => {}
983                // The part IS the receiver, so it reaches no new type: there
984                // is no accessor to check and nothing further to resolve. It
985                // must not push `ty` either — that reads as the crossing
986                // depending on itself, and validation reports a cycle. The
987                // value's own converter comes from its `whole` row, which is a
988                // different recipe.
989                Reach::Identity => {}
990                Reach::Accessor(func) => {
991                    let Some(f) = self.function(func) else {
992                        continue;
993                    };
994                    let ret = f.ret.clone();
995                    if !accessor_of(f, ty) {
996                        self.push(RecipeError::NotAnAccessor {
997                            recipe: self.recipe.clone(),
998                            func: func.clone(),
999                        });
1000                    }
1001                    out.push(ret);
1002                }
1003                Reach::Field(index) => {
1004                    let Some(fields) = self.fields(ty) else {
1005                        self.not_a_product();
1006                        continue;
1007                    };
1008                    match fields.get(*index) {
1009                        Some(field) => out.push(field.ty.clone()),
1010                        None => self.out_of_range(*index, fields.len()),
1011                    }
1012                }
1013            }
1014        }
1015        out
1016    }
1017
1018    fn function(&mut self, name: &syn::Ident) -> Option<&'a Function> {
1019        match self.model.function(name) {
1020            Some(f) => Some(f),
1021            None => {
1022                let error = RecipeError::UnknownFunction {
1023                    recipe: self.recipe.clone(),
1024                    func: name.clone(),
1025                };
1026                self.push(error);
1027                None
1028            }
1029        }
1030    }
1031
1032    /// The fields a [`Reach::Field`] indexes: an arm's payload where one is in
1033    /// scope, else the type's own.
1034    fn fields(&self, ty: &TypeRef) -> Option<Vec<Field>> {
1035        match &self.arm_fields {
1036            Some(fields) => Some(fields.clone()),
1037            None => match declared(self.model, ty)? {
1038                Type::Struct(s) => Some(s.fields.clone()),
1039                _ => None,
1040            },
1041        }
1042    }
1043
1044    fn alternatives(&self, ty: &TypeRef) -> Option<Vec<Vec<Field>>> {
1045        match declared(self.model, ty)? {
1046            Type::Variant(v) => Some(v.alternatives.iter().map(|a| a.fields.clone()).collect()),
1047            _ => None,
1048        }
1049    }
1050}
1051
1052/// Whether `f` reads a value of `ty` — `fn(&T) -> P` or `fn(T) -> P`.
1053fn accessor_of(f: &Function, ty: &TypeRef) -> bool {
1054    let Some(first) = f.params.first() else {
1055        return false;
1056    };
1057    value_key(&first.ty) == value_key(ty)
1058}
1059
1060/// Whether `f` builds a value of `ty` — `fn(..) -> T`, or `fn(..) -> Result<T, E>`
1061/// for a construction that can fail.
1062///
1063/// The constructing twin of [`accessor_of`]. A recipe names the function and the
1064/// model supplies everything else, so the one thing left to check is that the
1065/// function it names produces the type the recipe is filed under.
1066fn constructor_of(f: &Function, ty: &TypeRef) -> bool {
1067    // A fallible constructor is recognised by its return type, which is the same
1068    // place the parts' fallibility is read from — the recipe never states it.
1069    let built = match f.ret.fallible_parts() {
1070        Some((ok, _err)) => ok,
1071        None => &f.ret,
1072    };
1073    value_key(built) == value_key(ty)
1074}
1075
1076/// The type's identity with a borrow and any transparent wrapper gone — what
1077/// [`Crossing::key`] keys a recipe by, without a direction attached.
1078fn value_key(ty: &TypeRef) -> TypeKey {
1079    ty.borrow_target().unwrap_or(ty).stripped_key()
1080}
1081
1082/// The `#[prebindgen]` type declaration `ty` names, if it names one.
1083fn declared<'a>(model: &'a Flat, ty: &TypeRef) -> Option<&'a Type> {
1084    let value = ty.borrow_target().unwrap_or(ty).unwrapped();
1085    match value.kind() {
1086        TypeKind::Named { id, .. } => model.resolve(id),
1087        _ => None,
1088    }
1089}

1091// ── The one structural rule ───────────────────────────────────────────────
1092
1093/// A recipe may not reach its own crossing, directly or through any chain of
1094/// parts.
1095///
1096/// The traversal unrolls the table's graph statically, so a cycle would not
1097/// terminate. That is a mechanical limit rather than a semantic one, which is
1098/// why a self-referential Rust type is refused here and is not otherwise
1099/// ill-behaved.
1100fn cycles(model: &Flat, table: &Recipes, errors: &mut Vec<RecipeError>) {
1101    let mut settled: HashSet<CrossingKey> = HashSet::new();
1102    let mut reported: HashSet<CrossingKey> = HashSet::new();
1103    let roots: Vec<Crossing> = table
1104        .recipes
1105        .values()
1106        .flat_map(|entries| {
1107            entries
1108                .iter()
1109                .map(|e| Crossing::new(e.ty.clone(), e.recipe.direction()))
1110        })
1111        .collect();
1112    for root in roots {
1113        let mut path = Vec::new();
1114        walk(
1115            model,
1116            table,
1117            &root,
1118            &mut path,
1119            &mut settled,
1120            &mut reported,
1121            errors,
1122        );
1123    }
1124}
1125
1126fn walk(
1127    model: &Flat,
1128    table: &Recipes,
1129    crossing: &Crossing,
1130    path: &mut Vec<CrossingKey>,
1131    settled: &mut HashSet<CrossingKey>,
1132    reported: &mut HashSet<CrossingKey>,
1133    errors: &mut Vec<RecipeError>,
1134) {
1135    let key = crossing.key();
1136    if settled.contains(&key) {
1137        return;
1138    }
1139    if let Some(start) = path.iter().position(|k| k == &key) {
1140        if reported.insert(key.clone()) {
1141            let mut cycle = path[start..].to_vec();
1142            cycle.push(key);
1143            errors.push(RecipeError::Cycle { path: cycle });
1144        }
1145        return;
1146    }
1147    path.push(key.clone());
1148    for part in successors(model, table, crossing) {
1149        walk(model, table, &part, path, settled, reported, errors);
1150    }
1151    path.pop();
1152    settled.insert(key);
1153}
1154
1155/// Every crossing a crossing's recipes reach.
1156///
1157/// Every recipe counts, not only the default: a site may name any of them, so a
1158/// cycle through the recipe nobody happens to default to is still a cycle.
1159fn successors(model: &Flat, table: &Recipes, crossing: &Crossing) -> Vec<Crossing> {
1160    let key = crossing.key();
1161    let recipes: Vec<(RecipeKey, TypeRef, Recipe)> = match table.recipes.get(&key) {
1162        Some(entries) => entries
1163            .iter()
1164            .map(|e| (e.key.clone(), e.ty.clone(), e.recipe.clone()))
1165            .collect(),
1166        None => vec![(
1167            key.row(RecipeName::derived()),
1168            crossing.spelled().clone(),
1169            derive(crossing),
1170        )],
1171    };
1172    // Anything a recipe names wrongly is reported by the per-recipe pass; here the
1173    // walk only needs the parts a well-formed reading of the recipe reaches.
1174    let mut discarded = Vec::new();
1175    recipes
1176        .into_iter()
1177        .flat_map(|(recipe_key, ty, recipe)| {
1178            let mut check = Check {
1179                model,
1180                recipe: &recipe_key,
1181                errors: &mut discarded,
1182                arm_fields: None,
1183            };
1184            check.recipe(&ty, &recipe)
1185        })
1186        .collect()
1187}

1191/// A problem [`RecipesBuilder::build`] found.
1192#[derive(Debug)]
1193pub enum RecipeError {
1194    /// A recipe's parts reach the recipe's own crossing.
1195    ///
1196    /// The path is a chain of keys because it may pass through a callback, and
1197    /// so swap directions.
1198    Cycle {
1199        /// The chain, starting and ending at the crossing that repeats.
1200        path: Vec<CrossingKey>,
1201    },
1202    /// A crossing has several recipes and none, or more than one, was declared the
1203    /// default.
1204    NoDefault {
1205        /// The crossing whose recipes disagree.
1206        crossing: CrossingKey,
1207        /// The recipes that claimed to be the default.
1208        defaults: Vec<RecipeName>,
1209    },
1210    /// One row was declared twice.
1211    Duplicate {
1212        /// The row declared twice.
1213        recipe: RecipeKey,
1214    },
1215    /// A recipe named a constructor or accessor the model does not have.
1216    UnknownFunction {
1217        /// The row that named it.
1218        recipe: RecipeKey,
1219        /// The name that resolved to nothing.
1220        func: syn::Ident,
1221    },
1222    /// A constructor was named where what it returns is not the recipe type.
1223    ///
1224    /// The constructing twin of [`NotAnAccessor`](Self::NotAnAccessor). A
1225    /// `Result<T, E>` return counts as building a `T`, because that is where a
1226    /// construction's fallibility is read from.
1227    NotAConstructor {
1228        /// The row that named the function.
1229        recipe: RecipeKey,
1230        /// The function whose return type does not match.
1231        func: syn::Ident,
1232    },
1233    /// An accessor was named where its first parameter is not the recipe type.
1234    NotAnAccessor {
1235        /// The row that named the function.
1236        recipe: RecipeKey,
1237        /// The function whose first parameter does not match.
1238        func: syn::Ident,
1239    },
1240    /// A [`Reach::Field`] or an [`Arm::alternative`] is past the end of what the
1241    /// model holds for that type.
1242    OutOfRange {
1243        /// The row that names the index.
1244        recipe: RecipeKey,
1245        /// The index the recipe named.
1246        index: usize,
1247        /// How many the model holds.
1248        len: usize,
1249    },
1250    /// A shape was declared for a type that cannot take it: `Optional` on a
1251    /// type that is not an `Option`, `Sequence` on one that is not a run, or
1252    /// `Invoke` on one that is not a callback.
1253    ///
1254    /// The arity shapes read their inner type off the crossing rather than
1255    /// stating it, so this is where a mismatch surfaces.
1256    WrongShape {
1257        /// The row whose shape is wrong.
1258        recipe: RecipeKey,
1259        /// The shape as declared.
1260        shape: &'static str,
1261        /// What the type would have had to be.
1262        wanted: &'static str,
1263    },
1264    /// A product or choice recipe was declared for a type the model gives no parts.
1265    NotAProduct {
1266        /// The row whose type has no parts.
1267        recipe: RecipeKey,
1268    },
1269    /// A site asked for a recipe row that was never declared.
1270    UnknownRecipe {
1271        /// Where the value crosses.
1272        site: Site,
1273        /// The complete key the site asked for.
1274        recipe: RecipeKey,
1275    },
1276    /// A caller named a recipe row the table does not have.
1277    ///
1278    /// Distinct from [`UnknownRecipe`](Self::UnknownRecipe), which is a **site**
1279    /// asking for one. This is a caller compiling a named recipe directly through
1280    /// [`Compiler::recipe_of`](crate::recipe::Compiler::recipe_of), where there is no
1281    /// site to name — an adapter checking a recipe it declared conditionally, and
1282    /// getting the condition wrong.
1283    NoSuchRecipe {
1284        /// The complete key the caller asked for.
1285        recipe: RecipeKey,
1286    },
1287    /// Two declarations of equal precedence bound one site to different recipes.
1288    Rebound {
1289        /// The site both named.
1290        site: Site,
1291        /// The precedence they share.
1292        origin: Origin,
1293    },
1294    /// A part yields a different Rust type from the one its edge needs.
1295    ///
1296    /// The type half of the composition contract; [`Composition`](Self::Composition)
1297    /// is the ownership half. Both are checked at every part, this one first: a
1298    /// part producing the wrong value is wrong however it is held.
1299    ///
1300    /// **Every** part, in the full sense — a product's fields and constructor
1301    /// arguments, an optional's value, a run's element, and a callback's
1302    /// arguments. Each is a crossing the recipe reaches, and each is checked in
1303    /// the one place the driver reaches one.
1304    ComposedType {
1305        /// The part's own site.
1306        site: Site,
1307        /// Which part of the recipe.
1308        part: usize,
1309        /// What the edge requires: a constructor's parameter, a field, an
1310        /// accessor's receiver, an optional's value, a run's element, or a
1311        /// callback argument.
1312        wanted: TypeKey,
1313        /// What the part's fragment says it produces.
1314        got: TypeKey,
1315    },
1316    /// A part yields something the edge it feeds cannot consume.
1317    Composition {
1318        /// The part's own site.
1319        site: Site,
1320        /// Which part of the recipe.
1321        part: usize,
1322        /// How the edge requires it held: a product's declared parameter or
1323        /// field mode, an optional's value, the mode a collection lends its
1324        /// elements in, or a callback argument's own.
1325        wanted: Mode,
1326        /// What the part's fragment produces.
1327        got: Mode,
1328    },
1329    /// A site needs a value that outlives the call and got a borrowed one.
1330    Validity {
1331        /// Where the value crosses.
1332        site: Site,
1333        /// The weakest validity the site's role accepts.
1334        needed: crate::recipe::Validity,
1335        /// What the root fragment produces.
1336        got: crate::recipe::Validity,
1337    },
1338    /// A callback crossing was declared with a shape other than
1339    /// [`Shape::Invoke`].
1340    ///
1341    /// Converting the arguments is what makes a callable callable, so taking it
1342    /// apart is the only thing an adapter can do with one.
1343    CallbackShape {
1344        /// The invalid callback row.
1345        recipe: RecipeKey,
1346    },
1347}
1348
1349impl fmt::Display for RecipeError {
1350    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
1351        match self {
1352            RecipeError::Cycle { path } => {
1353                let chain: Vec<String> = path.iter().map(|k| k.to_string()).collect();
1354                write!(
1355                    f,
1356                    "a recipe reaches its own crossing: {}",
1357                    chain.join(" -> ")
1358                )
1359            }
1360            RecipeError::NoDefault { crossing, defaults } => write!(
1361                f,
1362                "{crossing} has several recipes and {} of them is the default; \
1363                 declare exactly one with `declare_default`",
1364                if defaults.is_empty() {
1365                    "none".to_owned()
1366                } else {
1367                    format!("{}", defaults.len())
1368                }
1369            ),
1370            RecipeError::Duplicate { recipe } => {
1371                write!(f, "{recipe} is declared twice")
1372            }
1373            RecipeError::UnknownFunction { recipe, func } => write!(
1374                f,
1375                "{recipe} names `{func}`, which no #[prebindgen] source declares"
1376            ),
1377            RecipeError::NotAConstructor { recipe, func } => write!(
1378                f,
1379                "{recipe} builds the value with `{func}`, which does not return that type"
1380            ),
1381            RecipeError::NotAnAccessor { recipe, func } => write!(
1382                f,
1383                "{recipe} reaches a part through `{func}`, whose first parameter is not that type"
1384            ),
1385            RecipeError::OutOfRange { recipe, index, len } => {
1386                write!(f, "{recipe} names index {index}, and the model holds {len}")
1387            }
1388            RecipeError::WrongShape {
1389                recipe,
1390                shape,
1391                wanted,
1392            } => write!(
1393                f,
1394                "{recipe} declares `{shape}`, but the type is not {wanted}"
1395            ),
1396            RecipeError::NotAProduct { recipe } => write!(
1397                f,
1398                "{recipe} takes its type apart, and the model gives that type no parts"
1399            ),
1400            RecipeError::NoSuchRecipe { recipe } => write!(
1401                f,
1402                "{recipe} does not exist — it was compiled by name, not through a site"
1403            ),
1404            RecipeError::UnknownRecipe { site, recipe } => {
1405                write!(f, "{site} asks for {recipe}, which does not exist")
1406            }
1407            RecipeError::Rebound { site, origin } => write!(
1408                f,
1409                "{site} is bound to two different recipes by {origin}; one of them has to \
1410                 be written at a higher precedence"
1411            ),
1412            RecipeError::ComposedType {
1413                site,
1414                part,
1415                wanted,
1416                got,
1417            } => write!(
1418                f,
1419                "part {part} of {site} needs a `{wanted}` and its fragment produces a `{got}`"
1420            ),
1421            RecipeError::Composition {
1422                site,
1423                part,
1424                wanted,
1425                got,
1426            } => write!(
1427                f,
1428                "part {part} of {site} needs `{wanted}` and its fragment produces `{got}`"
1429            ),
1430            RecipeError::Validity { site, needed, got } => write!(
1431                f,
1432                "{site} needs a {needed} value and its fragment produces a {got} one"
1433            ),
1434            RecipeError::CallbackShape { recipe } => write!(
1435                f,
1436                "{recipe} is not `Invoke`; a callback is always taken apart into its \
1437                 arguments, so that is the only shape such a crossing takes"
1438            ),
1439        }
1440    }
1441}
