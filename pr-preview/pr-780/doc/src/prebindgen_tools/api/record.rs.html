10/// A struct or one alternative of a sum: something with fields and
11/// delimiters.
12#[derive(Clone, Copy)]
13pub enum Record<'a> {
14    Struct(&'a Struct),
15    Alt(&'a Alternative),
16}
17
18impl<'a> Record<'a> {
19    pub fn fields(&self) -> &'a [Field] {
20        match self {
21            Record::Struct(s) => &s.fields,
22            Record::Alt(a) => &a.fields,
23        }
24    }
25
26    /// `head { a: v0, b: v1 }` / `head(v0, v1)` / `head`, with the
27    /// delimiters the source wrote.
28    pub fn construct(&self, head: &TokenStream, values: &[TokenStream]) -> TokenStream {
29        let parts: Vec<TokenStream> = self
30            .fields()
31            .iter()
32            .zip(values)
33            .map(|(f, v)| f.bind(v))
34            .collect();
35        self.shape(head, &parts)
36    }
37
38    /// The pattern binding each field to the matching ident in `binds`.
39    pub fn pattern(&self, head: &TokenStream, binds: &[syn::Ident]) -> TokenStream {
40        self.construct(
41            head,
42            &binds
43                .iter()
44                .map(|b| b.to_token_stream())
45                .collect::<Vec<_>>(),
46        )
47    }
48
49    /// Fresh binding names for the fields: `__f0`, `__f1`, …
50    pub fn binds(&self) -> Vec<syn::Ident> {
51        (0..self.fields().len())
52            .map(|i| format_ident!("__f{}", i))
53            .collect()
54    }
55
56    fn shape(&self, head: &TokenStream, parts: &[TokenStream]) -> TokenStream {
57        let emit = Emit::new();
58        match self {
59            Record::Struct(s) => emit.shape(*s, head.clone(), parts),
60            Record::Alt(a) => emit.shape(*a, head.clone(), parts),
61        }
62    }
63}
64
65/// The adapter's decision for one field.
66pub trait FieldCallbacks {
67    type Error;
68
69    /// The wires `field` arrives on, and how they rebuild it.
70    fn field_in(&mut self, field: &Field) -> Result<Input, Self::Error>;
71
72    /// The wires `field` leaves on, reading the field's value from `value`
73    /// (an owned binding).
74    fn field_out(&mut self, field: &Field, value: &TokenStream) -> Result<Output, Self::Error>;
75}
76
77/// A record rebuilt from its fields' inputs: wires concatenated in field
78/// order, values placed into the record's constructor `head`.
79pub fn record_in(record: Record<'_>, head: &TokenStream, fields: Vec<Input>) -> Input {
80    Input::combine(fields, |values| record.construct(head, &values))
81}
82
83/// A record taken apart into its fields' outputs. `src` is the record
84/// value; `field` builds each field's output from the binding holding it.
85pub fn record_out<E>(
86    record: Record<'_>,
87    head: &TokenStream,
88    src: &TokenStream,
89    mut field: impl FnMut(&Field, &TokenStream) -> Result<Output, E>,
90) -> Result<Output, E> {
91    let binds = record.binds();
92    let mut parts = Vec::new();
93    for (f, b) in record.fields().iter().zip(&binds) {
94        parts.push(field(f, &b.to_token_stream())?);
95    }
96    let pat = record.pattern(head, &binds);
97    Ok(Output::concat(quote!(let #pat = #src;), parts))
98}

100/// A `#[repr(C)]`-style mirror of a source struct and its two conversions.
101pub struct StructMirror {
102    /// The mirror's definition.
103    pub def: TokenStream,
104    /// The mirror's fields, in order.
105    pub wires: Vec<Wire>,
106    /// Mirror value bound to `v` → source value.
107    pub input: Input,
108    /// Source value bound to `v` → mirror value.
109    pub output: Output,
110}
111
112/// Writes a mirror struct for a source struct.
113pub struct StructWriter<'s> {
114    source: &'s Struct,
115    name: syn::Ident,
116    source_path: TokenStream,
117    attrs: Vec<TokenStream>,
118}
119
120impl<'s> StructWriter<'s> {
121    /// A mirror of `source` (named by `source_path` in generated code),
122    /// called `name`.
123    pub fn new(source: &'s Struct, source_path: impl ToTokens, name: syn::Ident) -> Self {
124        Self {
125            source,
126            name,
127            source_path: source_path.to_token_stream(),
128            attrs: Vec::new(),
129        }
130    }
131
132    /// An attribute on the mirror (`#[repr(C)]`, `#[derive(..)]`).
133    pub fn attr(mut self, attr: impl ToTokens) -> Self {
134        self.attrs.push(attr.to_token_stream());
135        self
136    }
137
138    pub fn write<C: FieldCallbacks>(self, cb: &mut C) -> Result<StructMirror, C::Error> {
139        let record = Record::Struct(self.source);
140        let mut ins = Vec::new();
141        let mut outs = Vec::new();
142        let binds = record.binds();
143        for (f, b) in self.source.fields.iter().zip(&binds) {
144            ins.push(cb.field_in(f)?);
145            outs.push(cb.field_out(f, &b.to_token_stream())?);
146        }
147        let name = &self.name;
148        let wires: Vec<Wire> = ins.iter().flat_map(|i| i.wires.clone()).collect();
149        let decls = wires.iter().map(|w| {
150            let d = w.decl();
151            quote!(pub #d)
152        });
153        let attrs = &self.attrs;
154        let def = quote! {
155            #(#attrs)*
156            pub struct #name { #(#decls),* }
157        };
158        let wire_names: Vec<&syn::Ident> = wires.iter().map(|w| &w.name).collect();
159        let head = &self.source_path;
160        let rebuilt = record_in(record, head, ins);
161        let input = Input {
162            wires: Vec::new(),
163            expr: {
164                let e = &rebuilt.expr;
165                quote!({ let #name { #(#wire_names),* } = v; #e })
166            },
167            fallible: rebuilt.fallible,
168            pass: None,
169        };
170        let pat = record.pattern(head, &binds);
171        let fallible = outs.iter().any(|o| o.fallible);
172        let out_binds: Vec<TokenStream> = outs.iter().map(Output::bind).collect();
173        let output = Output {
174            wires: Vec::new(),
175            expr: quote!({ let #pat = v; #(#out_binds)* #name { #(#wire_names),* } }),
176            fallible,
177        };
178        Ok(StructMirror {
179            def,
180            wires,
181            input,
182            output,
183        })
184    }
185}

187/// A mirror of a source sum: an enum whose alternatives carry the wires.
188pub struct SumMirror {
189    pub def: TokenStream,
190    /// Per alternative, its payload wires.
191    pub alternatives: Vec<Vec<Wire>>,
192    /// Mirror value bound to `v` → source value (a `match`).
193    pub input: Input,
194    /// Source value bound to `v` → mirror value (a `match`).
195    pub output: Output,
196}
197
198/// Writes a mirror enum for a source sum.
199pub struct SumWriter<'v> {
200    source: &'v Variant,
201    name: syn::Ident,
202    source_path: TokenStream,
203    attrs: Vec<TokenStream>,
204}
205
206impl<'v> SumWriter<'v> {
207    pub fn new(source: &'v Variant, source_path: impl ToTokens, name: syn::Ident) -> Self {
208        Self {
209            source,
210            name,
211            source_path: source_path.to_token_stream(),
212            attrs: Vec::new(),
213        }
214    }
215
216    pub fn attr(mut self, attr: impl ToTokens) -> Self {
217        self.attrs.push(attr.to_token_stream());
218        self
219    }
220
221    pub fn write<C: FieldCallbacks>(self, cb: &mut C) -> Result<SumMirror, C::Error> {
222        let name = &self.name;
223        let src = &self.source_path;
224        let mut variants = Vec::new();
225        let mut in_arms = Vec::new();
226        let mut out_arms = Vec::new();
227        let mut alternatives = Vec::new();
228        let mut in_fallible = false;
229        let mut out_fallible = false;
230        for alt in &self.source.alternatives {
231            let record = Record::Alt(alt);
232            let aname = &alt.name;
233            let binds = record.binds();
234            let mut ins = Vec::new();
235            let mut outs = Vec::new();
236            for (f, b) in alt.fields.iter().zip(&binds) {
237                ins.push(cb.field_in(f)?);
238                outs.push(cb.field_out(f, &b.to_token_stream())?);
239            }
240            // A mirror alternative has one mirror field per source field; a
241            // field that needs several wires is kept whole as a tuple would
242            // lose the per-field names C sees, so it is refused here by
243            // construction: the adapter hands one wire per field.
244            let wires: Vec<Wire> = ins.iter().flat_map(|i| i.wires.clone()).collect();
245            let mirror_head = quote!(#name::#aname);
246            let source_head = quote!(#src::#aname);
247            // Declaration: same delimiters as the source, wire types in place.
248            let tys: Vec<TokenStream> = wires.iter().map(|w| w.ty.clone()).collect();
249            let decl = record.construct(&quote!(#aname), &tys);
250            variants.push(decl);
251            // In: match the mirror alternative, binding its wires by name.
252            let wire_binds: Vec<syn::Ident> = wires.iter().map(|w| w.name.clone()).collect();
253            let mirror_pat = record.pattern(&mirror_head, &wire_binds);
254            let rebuilt = record_in(record, &source_head, ins);
255            in_fallible |= rebuilt.fallible;
256            let e = &rebuilt.expr;
257            in_arms.push(quote!(#mirror_pat => #e));
258            // Out: match the source alternative, produce the mirror one.
259            let source_pat = record.pattern(&source_head, &binds);
260            out_fallible |= outs.iter().any(|o| o.fallible);
261            let out_binds: Vec<TokenStream> = outs.iter().map(Output::bind).collect();
262            let values: Vec<TokenStream> = wires.iter().map(|w| w.name.to_token_stream()).collect();
263            let rebuilt_mirror = record.construct(&mirror_head, &values);
264            out_arms.push(quote!(#source_pat => { #(#out_binds)* #rebuilt_mirror }));
265            alternatives.push(wires);
266        }
267        let attrs = &self.attrs;
268        let def = quote! {
269            #(#attrs)*
270            pub enum #name { #(#variants),* }
271        };
272        Ok(SumMirror {
273            def,
274            alternatives,
275            input: Input {
276                wires: Vec::new(),
277                expr: quote!(match v { #(#in_arms),* }),
278                fallible: in_fallible,
279                pass: None,
280            },
281            output: Output {
282                wires: Vec::new(),
283                expr: quote!(match v { #(#out_arms),* }),
284                fallible: out_fallible,
285            },
286        })
287    }
