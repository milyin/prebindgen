1//! What a C binding declares.
2//!
3//! One [`Decls`] holds every declaration the binding makes: the types that
4//! cross, the functions it exports, the callback signatures it accepts, and the
5//! conversions it supplies. The set is **flat**, because an exported C function
6//! is not a member of anything: `calculator_apply(calculator_t *, …)` is a free
7//! function that happens to take a handle, and the header declares it beside
8//! the type rather than inside it. Nothing here groups a function under a type,
9//! because C has nothing for that to mean.
10//!
11//! What each declaration does carry is its own options — a handle's name base,
12//! a function's permission to panic, a callback's takeable arguments — rather
13//! than the builder carrying them for whatever was declared last. That is the
14//! difference from calling the declarators one after another, where
15//! `.base_name("value")` modified whichever declaration came before it, so
16//! moving a line changed what it meant and an option could attach to the wrong
17//! declaration with nothing to notice.
18//!
19//! ```
20//! use prebindgen_c::{data_type, decls, enum_type, fun, ptr_type};
21//!
22//! let _ = decls!()
23//!     .ptr_type(ptr_type!(Calculator))
24//!     .data_type(data_type!(Drawing))
25//!     .enum_type(enum_type!(Operation))
26//!     .fun(fun!(calculator_apply))
27//!     .fun(fun!(stamp_sum).abort_on_conversion_error());
28//! ```

32/// One exported function.
33///
34/// `abort_on_conversion_error` is the permission a wrapper needs when a
35/// conversion it performs can fail and it has no `Result` to report that
36/// through — an input the binding must reject, or a result it cannot encode.
37/// Without the permission such a function is a declaration error rather than a
38/// wrapper that terminates the process, which is what a C caller would observe.
39#[derive(Clone)]
40pub struct FunDecl {
41    pub(crate) ident: syn::Ident,
42    pub(crate) base: Option<String>,
43    pub(crate) abort_on_conversion_error: bool,
44}
45
46impl FunDecl {
47    /// Declare the `#[prebindgen]` function with this Rust name.
48    pub fn new(ident: syn::Ident) -> Self {
49        FunDecl {
50            ident,
51            base: None,
52            abort_on_conversion_error: false,
53        }
54    }
55
56    /// Override the token the function-name mangler receives, in place of the
57    /// Rust name.
58    pub fn base_name(mut self, base: impl Into<String>) -> Self {
59        self.base = Some(base.into());
60        self
61    }
62
63    /// Allow this wrapper to terminate the process when a conversion fails and
64    /// it has no error channel to report through.
65    ///
66    /// C observes termination, not a catchable Rust panic, which is why the
67    /// permission is named for what the caller sees.
68    pub fn abort_on_conversion_error(mut self) -> Self {
69        self.abort_on_conversion_error = true;
70        self
71    }
72}

98/// A payload-carrying enum crossing as a `#[repr(C)]` tag plus union.
99///
100/// # Who frees a payload
101///
102/// The union crosses **by value**, like a [`data_type!`](crate::data_type). If
103/// any variant's payload wire owns memory — a `char *`, an opaque pointer — the
104/// declaration also produces a typed `<base>_drop` that frees the **active
105/// arm**. An owning payload with no drop is a generation error rather than a
106/// leak.
107///
108/// Ownership follows the wire, so a payload that is itself a declared data type
109/// is owning when its own mirror has owning fields, even though that payload
110/// crosses as a struct by value rather than as a pointer. The drop reaches
111/// through it and releases each of them, nulling the slot so a second drop is a
112/// no-op. The reach goes one level further: a struct payload's field may itself
113/// be a declared tagged union with an owning arm, and the outer drop delegates
114/// to that union's typed drop, which nulls what it frees — so idempotence
115/// composes.
116///
117/// Nothing else can reach those bytes. At top level the data-type contract is
118/// that C releases each owning field itself, but a union arm is not a top-level
119/// field. One predicate decides both whether the drop is emitted and whether a
120/// containing struct calls it, so a nested union cannot be freed through a
121/// symbol that was never emitted.
122///
123/// The drop is a second C entry point into the same bytes, so it range-checks
124/// the tag exactly as an inbound conversion does, and treats an out-of-range
125/// one as nothing to release.
126#[derive(Clone)]
127pub struct TaggedUnionDecl {
128    pub(crate) ty: syn::Type,
129    pub(crate) base: Option<String>,
130}

277impl CallbackDecl {
278    /// Declare a callback signature, written as `impl Fn(..) + Send + Sync + 'static`.
279    pub fn new(ty: syn::Type) -> Self {
280        CallbackDecl {
281            ty,
282            base: None,
283            takeable: Vec::new(),
284        }
285    }
286
287    /// Deliver this argument as a takeable owned pointer rather than by value.
288    ///
289    /// The index is zero-based over the callback's own arguments, and one the
290    /// signature does not have is a declaration error: silently ignoring it
291    /// would leave the argument the author meant on ordinary delivery, with
292    /// nothing to say so.
293    pub fn takeable_param(mut self, index: usize) -> Self {
294        let arity = prebindgen_registry::flat::extract_fn_trait_args(&self.ty)
295            .map(|args| args.len())
296            .unwrap_or(0);
297        assert!(
298            index < arity,
299            "takeable_param({index}) on a callback with {arity} argument(s): the index is zero-based over the callback's own arguments, and this one has no argument to name"
300        );
301        self.takeable.push(index);
302        self
303    }
304}

329/// Everything one C binding declares, as one flat set.
330///
331/// Hand it to [`CbindgenBuilder::declare`](crate::CbindgenBuilder::declare).
332#[derive(Clone, Default)]
333pub struct Decls {
334    pub(crate) ptr_types: Vec<PtrTypeDecl>,
335    pub(crate) data_types: Vec<DataTypeDecl>,
336    pub(crate) enum_types: Vec<EnumTypeDecl>,
337    pub(crate) tagged_unions: Vec<TaggedUnionDecl>,
338    pub(crate) value_types: Vec<ValueTypeDecl>,
339    pub(crate) repr_c_types: Vec<ReprCTypeDecl>,
340    pub(crate) error_types: Vec<ErrorTypeDecl>,
341    pub(crate) callbacks: Vec<CallbackDecl>,
342    pub(crate) converts: Vec<ConvertTypeDecl>,
343    pub(crate) funs: Vec<FunDecl>,
344    pub(crate) ignored_funs: Vec<syn::Ident>,
345    pub(crate) ignored_types: Vec<syn::Type>,
346}
347
348impl Decls {
349    /// An empty declaration set.
350    pub fn new() -> Self {
351        Self::default()
352    }
353
354    /// Add an opaque-handle type.
355    pub fn ptr_type(mut self, decl: PtrTypeDecl) -> Self {
356        self.ptr_types.push(decl);
357        self
358    }
359
360    /// Add a by-value aggregate type.
361    pub fn data_type(mut self, decl: DataTypeDecl) -> Self {
362        self.data_types.push(decl);
363        self
364    }
365
366    /// Add a fieldless enum.
367    pub fn enum_type(mut self, decl: EnumTypeDecl) -> Self {
368        self.enum_types.push(decl);
369        self
370    }
371
372    /// Add a payload-carrying enum.
373    pub fn tagged_union(mut self, decl: TaggedUnionDecl) -> Self {
374        self.tagged_unions.push(decl);
375        self
376    }
377
378    /// Add a by-value opaque type and its counterpart.
379    pub fn value_type(mut self, decl: ValueTypeDecl) -> Self {
380        self.value_types.push(decl);
381        self
382    }
383
384    /// Add a source type that is already `#[repr(C)]`.
385    pub fn repr_c_type(mut self, decl: ReprCTypeDecl) -> Self {
386        self.repr_c_types.push(decl);
387        self
388    }
389
390    /// Add an opaque error type and its message accessor.
391    pub fn error_type(mut self, decl: ErrorTypeDecl) -> Self {
392        self.error_types.push(decl);
393        self
394    }
395
396    /// Add a callback signature.
397    pub fn callback(mut self, decl: CallbackDecl) -> Self {
398        self.callbacks.push(decl);
399        self
400    }
401
402    /// Add a declared conversion between a source type and its wire form.
403    ///
404    /// Takes the conversion itself, or one wrapped in a [`ConvertTypeDecl`]
405    /// when its generated constants need a naming base of their own.
406    pub fn convert(mut self, decl: impl Into<ConvertTypeDecl>) -> Self {
407        self.converts.push(decl.into());
408        self
409    }
410
411    /// Export a free function.
412    pub fn fun(mut self, decl: FunDecl) -> Self {
413        self.funs.push(decl);
414        self
415    }
416
417    /// Record that a captured function is deliberately not exported.
418    pub fn ignore_fun(mut self, ident: syn::Ident) -> Self {
419        self.ignored_funs.push(ident);
420        self
421    }
422
423    /// Record that a captured type is deliberately not exported.
424    pub fn ignore_type(mut self, ty: syn::Type) -> Self {
425        self.ignored_types.push(ty);
426        self
427    }
428}
