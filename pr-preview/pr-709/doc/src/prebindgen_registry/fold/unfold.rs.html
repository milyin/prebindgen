1//! The deconstructing half of the view: leaves read off a row that takes a
2//! value apart.
3//!
4//! The constructing half is in the parent module, and the two answer the same
5//! question in opposite directions — what a row costs on the foreign signature,
6//! in order, with names. What a row does not say, the target does, through an
7//! [`UnfoldPolicy`]: what a selector and a presence flag are, what one part is
8//! called, and how a nested part's name joins its parent's.
9//!
10//! Which parts are taken apart further is a **binding**, not a property of the
11//! row: a part whose type states a row still crosses whole unless something
12//! says to read it in pieces. That is #701's design decision 3, and it is what
13//! `Bindings` carries here.
14//!
15//! Only what a row can already state is here. `Reach::Path` and `Reach::Nested`
16//! are the two forms step 3 replaces with part bindings; until they are, a row
17//! using one is refused rather than read as something else.
18

27/// What a target decides when a row is taken apart into leaves.
28///
29/// A row states which parts there are and how each is reached. It does not
30/// state what any of them is *called* — an accessor's leaf name is the
31/// declaration's, not the function's — nor what a synthesized leaf carries.
32pub trait UnfoldPolicy {
33    /// The leaf saying which alternative of a sum is live.
34    ///
35    /// Its type is the sum itself rather than the integer it crosses as: what
36    /// the emitter needs is which sum it is choosing between.
37    fn selector(&self, source: &TypeRef) -> UnfoldLeaf;
38
39    /// The leaf saying whether an optional value the decomposition looks
40    /// through is present.
41    ///
42    /// `source` is the optional reading the flag reports on, which is what the
43    /// target needs to answer: the flag stands where that value does, and what
44    /// it carries is a choice about how absence crosses rather than a fact the
45    /// walk holds.
46    fn presence(&self, name: &str, source: &TypeRef) -> UnfoldLeaf;
47
48    /// Whether the parts being named are a value form's.
49    ///
50    /// A value form's parts are named by the declaration that lists them, and a
51    /// product's own by whatever the target calls the field. The two are
52    /// different lists, and only the caller knows which is being walked.
53    fn value_form_part(&self, _index: usize) -> Option<String> {
54        None
55    }
56
57    /// What the part reached by `reach` is called.
58    ///
59    /// The declaration's answer, which is why the row cannot give it: two
60    /// bindings may read one accessor under two names. `field` is the model's
61    /// name for the field a [`Reach::Field`] reads, which a target that names
62    /// its slots after the struct's own fields needs and one that names them
63    /// after the declaration ignores.
64    /// `owner` is the type whose row states this part — NOT the root the walk
65    /// started from. A row splices a child's decomposition into its parent's,
66    /// and the child's declaration is what names the child's parts, so a policy
67    /// asked under the root would miss every name the child declared.
68    fn part_name(
69        &self,
70        owner: &TypeRef,
71        reach: &Reach,
72        index: usize,
73        field: Option<&syn::Ident>,
74    ) -> String;
75
76    /// What one payload of one alternative is called.
77    ///
78    /// Not [`Self::part_name`] with the alternative prefixed: a target names an
79    /// arm's payload after what the alternative is called on ITS side, which is
80    /// not the Rust variant's name, and after how it addresses the field. Both
81    /// are the target's, and neither is in the row.
82    /// `sum` is the type the alternative belongs to, which is not the type the
83    /// walk started from: a sum reached as a part of something else carries its
84    /// own naming, and reading the root's would name a nested sum's slots after
85    /// whatever the outer type happens to say.
86    fn arm_part_name(
87        &self,
88        sum: &TypeRef,
89        variant: &syn::Ident,
90        member: &syn::Member,
91        index: usize,
92    ) -> String;
93
94    /// What the value itself is called when it is one of the parts.
95    fn identity_name(&self) -> String;
96
97    /// How a nested part's name joins the name of the part it was reached
98    /// through.
99    fn nest(&self, outer: &str, inner: &str) -> String;
100}

102/// Why a row could not be taken apart into leaves.
103#[derive(Debug)]
104pub enum UnfoldViewError {
105    /// A row named a function the model does not have.
106    UnknownAccessor(syn::Ident),
107    /// A row states a form the view does not read yet.
108    ///
109    /// Not a defect in the row: these are the shapes #701's step 3 is still
110    /// replacing, and a caller running the view beside the older path skips
111    /// them. Every other variant here IS a defect, which is why they are
112    /// separate — [`Self::is_not_yet_readable`] is what a caller may skip on,
113    /// and skipping on any error would let a bad row disable the very check
114    /// that found it.
115    NotYetReadable {
116        /// The crossing whose row was walked.
117        crossing: String,
118        /// The form it used.
119        form: &'static str,
120    },
121    /// More than one part of one level is the value itself.
122    ManyIdentities {
123        /// The value being taken apart.
124        ty: String,
125    },
126    /// Two leaves of one decomposition carry the same name.
127    DuplicateName {
128        /// The value being taken apart.
129        ty: String,
130        /// The name written twice.
131        name: String,
132    },
133    /// A part's row reaches the value being taken apart.
134    Cycle {
135        /// The type the walk came back to.
136        ty: String,
137    },
138}
139
140impl UnfoldViewError {
141    /// Whether this is a shape the view does not represent yet, rather than a
142    /// row that is wrong.
143    pub fn is_not_yet_readable(&self) -> bool {
144        matches!(self, UnfoldViewError::NotYetReadable { .. })
145    }
146
147    /// A short, stable name for what went wrong.
148    ///
149    /// Separate from [`Display`](std::fmt::Display) because the two are read by
150    /// different things. A message is for a person and says which type and
151    /// which row; a code is for a caller that records this outcome and compares
152    /// the record later, and such a record should not move when the wording
153    /// improves or when a type is spelled differently.
154    pub fn code(&self) -> &'static str {
155        match self {
156            UnfoldViewError::UnknownAccessor(_) => "unknown-accessor",
157            UnfoldViewError::NotYetReadable { form, .. } => form,
158            UnfoldViewError::ManyIdentities { .. } => "many-identities",
159            UnfoldViewError::DuplicateName { .. } => "duplicate-leaf-name",
160            UnfoldViewError::Cycle { .. } => "cycle",
161        }
162    }
163}
164
165impl std::fmt::Display for UnfoldViewError {
166    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
167        match self {
168            UnfoldViewError::UnknownAccessor(name) => {
169                write!(f, "no `#[prebindgen]` function named `{name}`")
170            }
171            UnfoldViewError::NotYetReadable { crossing, form } => write!(
172                f,
173                "`{crossing}`'s row uses `{form}`, which the leaf view does not read yet"
174            ),
175            UnfoldViewError::ManyIdentities { ty } => write!(
176                f,
177                "`{ty}` hands itself over more than once, and it has only one to give"
178            ),
179            UnfoldViewError::DuplicateName { ty, name } => {
180                write!(f, "`{ty}` gives two of its leaves the name `{name}`")
181            }
182            UnfoldViewError::Cycle { ty } => {
183                write!(
184                    f,
185                    "`{ty}` is taken apart into a part that is taken apart into it"
186                )
187            }
188        }
189    }
190}

192/// What one walk of a row did NOT read.
193///
194/// A caller running this beside an older decomposition needs to know where the
195/// two are not comparable, and needs to know it from the walk rather than by
196/// measuring the answers against each other. Measuring cannot tell a shape the
197/// rows do not state yet from a defect that made the read smaller — which is
198/// the same mistake in the small that a silent fallback is in the large.
199///
200/// Empty means the walk read everything the rows and bindings state. Anything
201/// else names a place it actually stopped.
202///
203/// **Not a list of possibilities.** A part no binding takes apart is not unread
204/// — crossing whole is its answer, and the walk read it. Only a place where the
205/// walk asked for parts and could not get them belongs here.
206#[derive(Clone, Debug, Default)]
207pub struct Coverage {
208    unread: Vec<&'static str>,
209}
210
211impl Coverage {
212    /// Whether the walk read the whole of what the row states.
213    pub fn is_complete(&self) -> bool {
214        self.unread.is_empty()
215    }
216
217    /// The places it stopped, for a caller that reports them.
218    pub fn unread(&self) -> &[&'static str] {
219        &self.unread
220    }
221
222    fn note(&mut self, what: &'static str) {
223        if !self.unread.contains(&what) {
224            self.unread.push(what);
225        }
226    }
227}

229/// What one walk carries: the target's answers, the row name a nested part is
230/// looked up under, what has been read so far, and the chain of types being
231/// taken apart, which is what catches a value reached from itself.
232struct Walk<'w> {
233    policy: &'w dyn UnfoldPolicy,
234    bindings: &'w crate::recipe::Bindings,
235    /// Whether the value this walk started from is lent rather than given.
236    lent: bool,
237    leaves: Vec<UnfoldLeaf>,
238    hoists: Vec<Hoist>,
239    reading: Vec<String>,
240    coverage: Coverage,
241}
242
243/// Where one level of the walk stands: the path reached so far, the name
244/// prefix, and whether anything above was optional.
245#[derive(Clone)]
246struct At {
247    path: Vec<PathStep>,
248    name: Option<String>,
249    nullable: bool,
250    /// Whether these parts are a value form's, which names them itself.
251    value_form: bool,
252    /// The row being read at this level, as a key.
253    ///
254    /// The KEY and not a name: a part's binding is written against the row its
255    /// product belongs to, and after a binding has taken a part to some other
256    /// row, that row is what its own parts are keyed by. Carrying one root name
257    /// down the walk would look for a grandchild's binding under the root's
258    /// name and miss it — while `Compiler::part_of` keys the same part by the
259    /// row it is actually at.
260    row: crate::recipe::RecipeKey,
261}
262
263impl Folding<'_> {
264    /// The leaves one value is taken apart into, and the value forms to bind
265    /// once on the way.
266    ///
267    /// `reading` is the value's own type. The row walked is the one named,
268    /// because taking a value apart is what a site opts into — the crossing's
269    /// default row is the value crossing whole.
270    pub fn unfold(
271        &self,
272        policy: &dyn UnfoldPolicy,
273        bindings: &crate::recipe::Bindings,
274        reading: &TypeRef,
275        row: &RecipeName,
276    ) -> Result<(Vec<UnfoldLeaf>, Vec<Hoist>, Coverage), UnfoldViewError> {
277        let shape = self
278            .deconstructing(reading, row)
279            .ok_or_else(|| UnfoldViewError::NotYetReadable {
280                crossing: reading.to_string(),
281                form: "no-parts-row",
282            })?
283            .clone();
284        let mut walk = Walk {
285            policy,
286            bindings,
287            lent: reading.borrow_target().is_some(),
288            leaves: Vec::new(),
289            hoists: Vec::new(),
290            reading: vec![reading.key().to_string()],
291            coverage: Coverage::default(),
292        };
293        let at = At {
294            path: Vec::new(),
295            name: None,
296            nullable: false,
297            value_form: false,
298            row: crate::recipe::Crossing::new(
299                reading.clone(),
300                crate::recipe::Direction::Deconstruct,
301            )
302            .row(row.clone()),
303        };
304        // The value the row describes, not the way it was reached. A `&T`
305        // return reads T's row, and `lent` above is where the borrow is already
306        // recorded — leaving it on the source too makes an identity leaf borrow
307        // a second time, `&&T` for a `&T` reading.
308        let value = reading.borrow_target().unwrap_or(reading);
309        self.level(&mut walk, value, &shape, &at)?;
310        unique_names(&walk.leaves, reading)?;
311        Ok((walk.leaves, walk.hoists, walk.coverage))
312    }
313
314    /// One row, at one place in the value.
315    fn level(
316        &self,
317        walk: &mut Walk<'_>,
318        source: &TypeRef,
319        shape: &Shape<Deconstruct>,
320        at: &At,
321    ) -> Result<(), UnfoldViewError> {
322        match shape {
323            Shape::Product(Deconstruct::Fields(reaches)) => self.parts(walk, source, reaches, at),
324            Shape::Product(Deconstruct::ValueForm { func, parts }) => {
325                // The form is called once and every part hangs off that one
326                // call, which is what a hoist records.
327                let mut root = at.path.clone();
328                root.push(PathStep::call(func.clone(), false, true));
329                walk.hoists.push(Hoist {
330                    prefix: root.clone(),
331                    consuming: self.consumes(func)?,
332                });
333                let inner = At {
334                    path: root,
335                    name: at.name.clone(),
336                    nullable: at.nullable,
337                    value_form: true,
338                    // A value form's parts belong to the row that states the
339                    // form, not to the struct the call returns.
340                    row: at.row.clone(),
341                };
342                // The parts are read off what the call RETURNS, not off the
343                // value it was called on: `Reach::Field(0)` here is the first
344                // field of the bound struct.
345                let bound = self.returns(func)?;
346                self.parts(walk, &bound, parts, &inner)
347            }
348            Shape::Choice { arms } => {
349                // The selector first, then one group per arm. Every arm's
350                // leaves are written, and the selector says which of them is
351                // live — an arm the value did not take fills its slots with the
352                // wire's default rather than being absent.
353                // The selector names WHICH sum it chooses between, so it takes
354                // the sum itself however the value was reached — a borrow is
355                // how this walk holds the value, not what the selector is about.
356                let mut tag = walk
357                    .policy
358                    .selector(source.borrow_target().unwrap_or(source));
359                // A sum reached AS A PART carries that part's path and name:
360                // its leaves are read out of the field the walk arrived
361                // through, not off the value the row belongs to.
362                tag.path = at.path.clone();
363                tag.nullable = at.nullable;
364                if let Some(outer) = &at.name {
365                    tag.name = walk.policy.nest(outer, &tag.name);
366                }
367                walk.leaves.push(tag);
368                for arm in arms {
369                    let Some(alternative) = arm.alternative else {
370                        return Err(UnfoldViewError::NotYetReadable {
371                            crossing: source.to_string(),
372                            form: "arm-without-alternative",
373                        });
374                    };
375                    let Deconstruct::Fields(reaches) = &arm.op else {
376                        return Err(UnfoldViewError::NotYetReadable {
377                            crossing: source.to_string(),
378                            form: "arm-through-value-form",
379                        });
380                    };
381                    let Some((variant, fields)) = self.alternative(source, alternative) else {
382                        return Err(UnfoldViewError::NotYetReadable {
383                            crossing: source.to_string(),
384                            form: "unknown-alternative",
385                        });
386                    };
387                    for (index, reach) in reaches.iter().enumerate() {
388                        let Reach::Field(field) = reach else {
389                            return Err(UnfoldViewError::NotYetReadable {
390                                crossing: source.to_string(),
391                                form: reach_form(reach),
392                            });
393                        };
394                        let Some(payload) = fields.get(*field) else {
395                            return Err(UnfoldViewError::NotYetReadable {
396                                crossing: source.to_string(),
397                                form: "unknown-payload-field",
398                            });
399                        };
400                        // Bound by a pattern in the arm rather than reached off
401                        // the value, so the leaf carries no path.
402                        // Named under the ALTERNATIVE: every arm writes its
403                        // own slots, so two arms reading their first payload
404                        // field would otherwise be two leaves of one name. What
405                        // that name is, is the target's — see
406                        // [`UnfoldPolicy::arm_part_name`].
407                        let member = member_of(payload, *field);
408                        let name = walk.policy.arm_part_name(
409                            source.borrow_target().unwrap_or(source),
410                            variant,
411                            &member,
412                            index,
413                        );
414                        let full = match &at.name {
415                            Some(outer) => walk.policy.nest(outer, &name),
416                            None => name,
417                        };
418                        // A payload may itself be taken apart, and a binding on
419                        // one part of one ARM is what says so — the case
420                        // `ArmKey` exists to tell from a part of the product
421                        // itself. `Compiler::part_of` follows the same key.
422                        let arm_key = ArmKey::Alternative(alternative);
423                        if let Some((row, child)) =
424                            self.bound_row(walk, at, Some(arm_key), index, &payload.ty)
425                        {
426                            let child = child.clone();
427                            let seen = payload.ty.key().to_string();
428                            if walk.reading.contains(&seen) {
429                                return Err(UnfoldViewError::Cycle { ty: seen });
430                            }
431                            walk.reading.push(seen);
432                            let inner = At {
433                                // Bound by a pattern in the arm, so the payload
434                                // is reached from the binding rather than from
435                                // the value: its own parts hang off no path.
436                                path: Vec::new(),
437                                name: Some(full),
438                                nullable: at.nullable,
439                                value_form: false,
440                                row,
441                            };
442                            let before = walk.leaves.len();
443                            self.level(walk, &payload.ty, &child, &inner)?;
444                            walk.reading.pop();
445                            // Everything the payload contributed is live only
446                            // in this arm, exactly as the payload itself would
447                            // have been.
448                            for leaf in &mut walk.leaves[before..] {
449                                leaf.groups.insert(0, alternative as i32);
450                            }
451                            continue;
452                        }
453                        walk.leaves.push(UnfoldLeaf {
454                            name: full,
455                            path: at.path.clone(),
456                            out_ty: payload.ty.clone(),
457                            identity: false,
458                            nullable: at.nullable,
459                            source: LeafSource::VariantField {
460                                variant: variant.clone(),
461                                member,
462                            },
463                            groups: vec![alternative as i32],
464                        });
465                    }
466                }
467                Ok(())
468            }
469            Shape::Optional => {
470                // The decomposition looks THROUGH an optional value: what comes
471                // apart is what the `Option` holds, and every leaf under it is
472                // null when the value is absent. The layer is the model's, so
473                // the row states it and the walk peels it rather than every row
474                // restating what it wraps.
475                let Some(inner) = source.optional_inner() else {
476                    return Err(UnfoldViewError::NotYetReadable {
477                        crossing: source.to_string(),
478                        form: "optional-layer-without-inner",
479                    });
480                };
481                let Some(child) = self.deconstructing(inner, at.row.name()).cloned() else {
482                    // The layer is peeled by now: what is missing is the row
483                    // for what it HOLDS. Naming the optional here would file
484                    // this under a shape the walk reads, and a later row
485                    // declaration closing it would then read as the optional
486                    // support having been finished by something else.
487                    return Err(UnfoldViewError::NotYetReadable {
488                        crossing: inner.to_string(),
489                        form: "no-parts-row",
490                    });
491                };
492                // A presence flag ahead of the leaves it gates, and the leaves
493                // in its group: the value's absence is one wire slot, and the
494                // leaves under it carry the wire default when it says absent.
495                // How absence is spent is the target's answer — see
496                // [`UnfoldPolicy::presence`] — and what it gates is this walk's.
497                let name = at.name.clone().unwrap_or_default();
498                let mut flag = walk.policy.presence(&name, source);
499                // The flag stands where the value does, and what it carries is
500                // the OPTIONAL as written: the emitter tests that value to set
501                // it, so the reading it names is the one it tests.
502                // The step that reaches the value is optional — the emitter
503                // unwraps it before going on — and the flag stands at the same
504                // place, because testing the value is reaching it.
505                let mut path = at.path.clone();
506                if let Some(last) = path.last_mut() {
507                    *last = match last {
508                        PathStep::Call { ident, owned, .. } => {
509                            PathStep::call(ident.clone(), true, *owned)
510                        }
511                        PathStep::Field { ident, .. } => PathStep::field(ident.clone(), true),
512                    };
513                }
514                // Whether the absence gets a WIRE SLOT of its own, or is
515                // carried by the leaves under it.
516                //
517                // A layer reached by CALLING an accessor is spent as
518                // nullability: the emitter calls the accessor, and every leaf
519                // under it is null when nothing came back. There is no second
520                // value to test, so a flag would be a slot nothing sets. A
521                // layer the value HOLDS — a field, or the reading itself — is
522                // one the emitter tests before it reaches anything, and that
523                // test is the flag.
524                let gated = !matches!(path.last(), Some(PathStep::Call { .. }));
525                if gated {
526                    flag.path = path.clone();
527                    walk.leaves.push(flag);
528                }
529                let inner_at = At {
530                    path,
531                    nullable: at.nullable || !gated,
532                    // The row the INNER crossing states, so its parts are keyed
533                    // by it: a binding written for a part of `Option<T>` is not
534                    // one written for a part of `T`, and the compiler keys them
535                    // apart the same way.
536                    row: crate::recipe::Crossing::new(
537                        inner.clone(),
538                        crate::recipe::Direction::Deconstruct,
539                    )
540                    .row(at.row.name().clone()),
541                    ..at.clone()
542                };
543                let before = walk.leaves.len();
544                self.level(walk, inner, &child, &inner_at)?;
545                // Everything the inner contributed is live only when the flag
546                // says the value is there.
547                // Group 0: an optional has one arm, the one the flag admits.
548                if gated {
549                    for leaf in &mut walk.leaves[before..] {
550                        leaf.groups.insert(0, 0);
551                    }
552                }
553                Ok(())
554            }
555            Shape::Atomic => {
556                // Asked for parts and got none: a placeholder a crossing
557                // carries so a site can select it, which says the adapter emits
558                // this conversion itself.
559                walk.coverage.note("row-states-no-parts");
560                Ok(())
561            }
562            other => Err(UnfoldViewError::NotYetReadable {
563                crossing: source.to_string(),
564                form: name_of(other),
565            }),
566        }
567    }

569    /// Every part of one product.
570    fn parts(
571        &self,
572        walk: &mut Walk<'_>,
573        source: &TypeRef,
574        reaches: &[Reach],
575        at: &At,
576    ) -> Result<(), UnfoldViewError> {
577        // One value to give, so one part may be it. It keeps the position the
578        // row puts it in: what goes last is its EMISSION, after every borrow
579        // taken off the value has ended, and that is the emitter's ordering
580        // rather than the leaf list's.
581        let mut seen_identity = false;
582        for (index, reach) in reaches.iter().enumerate() {
583            if !matches!(reach, Reach::Identity) {
584                self.part(walk, source, reach, index, at)?;
585                continue;
586            }
587            if seen_identity {
588                return Err(UnfoldViewError::ManyIdentities {
589                    ty: source.key().to_string(),
590                });
591            }
592            seen_identity = true;
593            let name = match &at.name {
594                Some(outer) => outer.clone(),
595                None => walk.policy.identity_name(),
596            };
597            walk.leaves.push(UnfoldLeaf {
598                name,
599                path: at.path.clone(),
600                // The value as it stands: owned where it is OURS to give — the
601                // root of a decomposition that was handed the value — and
602                // borrowed everywhere else, where the leaf clones through its
603                // own converter. A root reached by reference is lent, not
604                // given, which the reading says and the row cannot.
605                out_ty: if at.path.is_empty() && !walk.lent {
606                    source.clone()
607                } else {
608                    // A borrow of the VALUE. A part reached through a layer is
609                    // already spelled `&T` where the accessor lends it, and
610                    // borrowing that again gives `&&T` — the same double borrow
611                    // the walk entry peels for a `&T` reading.
612                    source.borrow_target().unwrap_or(source).borrowed()
613                },
614                identity: true,
615                nullable: at.nullable,
616                source: LeafSource::Reach,
617                groups: Vec::new(),
618            });
619        }
620        Ok(())
621    }

623    /// One part, spliced through its own row when it has one.
624    fn part(
625        &self,
626        walk: &mut Walk<'_>,
627        source: &TypeRef,
628        reach: &Reach,
629        index: usize,
630        at: &At,
631    ) -> Result<(), UnfoldViewError> {
632        let name = match at
633            .value_form
634            .then(|| walk.policy.value_form_part(index))
635            .flatten()
636        {
637            Some(name) => name,
638            // A value form that names no part at this position falls back to
639            // the ordinary answer, the same as a product's own part.
640            None => walk
641                .policy
642                .part_name(source, reach, index, self.field_name(source, reach)),
643        };
644        let full = match &at.name {
645            Some(outer) => walk.policy.nest(outer, &name),
646            None => name,
647        };
648        // `ty` is what this level goes on to WALK, delivery layers peeled off.
649        // `crossed` is what the part CROSSES, layers included — the type
650        // `Compiler::part_of` builds its crossing from, so the one a binding
651        // for this part is written against. They differ for an optional part,
652        // and resolving the binding under the walked value rather than the
653        // crossed one is why an optional part could not be bound: whichever
654        // type the adapter named, one of the two sides looked for the other.
655        let (step, ty, crossed) = match reach {
656            Reach::Omit => return Ok(()),
657            Reach::Accessor(func) => {
658                let ret = self.returns(func)?;
659                let optional = ret.optional_inner().is_some();
660                let core = ret.optional_inner().unwrap_or(&ret);
661                let owned = core.borrow_target().is_none();
662                (
663                    PathStep::call(func.clone(), optional, owned),
664                    core.borrow_target().unwrap_or(core).clone(),
665                    ret.clone(),
666                )
667            }
668            Reach::Field(field) => {
669                let ty = self.field_ty(source, *field).ok_or_else(|| {
670                    UnfoldViewError::NotYetReadable {
671                        crossing: source.to_string(),
672                        form: "unknown-field",
673                    }
674                })?;
675                let ident = self.field_ident(source, *field);
676                // A field's own `Option` rides the leaf's own converter: a part
677                // that crosses whole takes the `Option` with it. Looking
678                // THROUGH one — the nesting step a decomposition makes of an
679                // optional nested class — is the `Optional` part #701's
680                // decision 3 describes, and waits for it.
681                (PathStep::field(ident, false), ty.clone(), ty)
682            }
683            Reach::Identity => unreachable!("handled by the caller"),
684            other => {
685                return Err(UnfoldViewError::NotYetReadable {
686                    crossing: source.to_string(),
687                    form: reach_form(other),
688                })
689            }
690        };
691        let optional = step.is_optional();
692        let mut path = at.path.clone();
693        path.push(step);
694
695        // A part is taken apart further only where a binding says so. Having a
696        // row of its own is not enough — a `data_class` states one and still
697        // crosses whole as an accessor's return, which is the difference a
698        // binding carries and a row cannot (#701 decision 3).
699        if let Some((row, child)) = self.bound_row(walk, at, None, index, &crossed) {
700            let child = child.clone();
701            let seen = ty.key().to_string();
702            if walk.reading.contains(&seen) {
703                return Err(UnfoldViewError::Cycle { ty: seen });
704            }
705            walk.reading.push(seen);
706            let inner = At {
707                path,
708                name: Some(full),
709                nullable: at.nullable || optional,
710                value_form: false,
711                // The row the binding took this part to, so ITS parts are keyed
712                // by it rather than by the row above.
713                row,
714            };
715            // A row that states the layer is walked with the crossed value:
716            // the Optional row peels it itself, marking the step already on the
717            // path rather than adding one.
718            let layered = matches!(child, Shape::Optional);
719            self.level(walk, if layered { &crossed } else { &ty }, &child, &inner)?;
720            walk.reading.pop();
721            return Ok(());
722        }
723
724        // Nothing bound this part, so it crosses whole — which is an answer,
725        // not a stopping point. Whether some older statement takes it apart
726        // further is that statement's business, and a caller comparing the two
727        // knows its own declarations; this walk read what the rows say.
728        walk.leaves.push(UnfoldLeaf {
729            name: full,
730            path,
731            out_ty: self.leaf_ty(reach, source, index),
732            identity: false,
733            nullable: at.nullable,
734            source: LeafSource::Reach,
735            groups: Vec::new(),
736        });
737        Ok(())
738    }

740    /// The model's name for the field a reach reads, when it reads one.
741    fn field_name(&self, source: &TypeRef, reach: &Reach) -> Option<&syn::Ident> {
742        let Reach::Field(index) = reach else {
743            return None;
744        };
745        self.fields_of(source).get(*index)?.name.as_ref()
746    }
747
748    /// The row a part is taken apart by, if a binding names one.
749    ///
750    /// The site is the part's own position in the row being walked, which is
751    /// the key `Compiler::part_of` builds for the same part — so an adapter
752    /// writes one binding and both the compiler and this view find it.
753    fn bound_row(
754        &self,
755        walk: &Walk<'_>,
756        at: &At,
757        arm: Option<crate::recipe::ArmKey>,
758        index: usize,
759        part: &TypeRef,
760    ) -> Option<(crate::recipe::RecipeKey, &Shape<Deconstruct>)> {
761        use crate::recipe::{Crossing, Direction, Site};
762
763        // Keyed by the row this level is AT, which is what `Compiler::part_of`
764        // keys the same part by — so an adapter writes one binding and both
765        // find it.
766        let site = Site::arm_part(&at.row, arm, index);
767        // DECLARED, not merely resolved. `Bindings::resolve` answers for every
768        // site, falling back to the crossing's default row — which is the whole
769        // value crossing, and following it here would take a part apart because
770        // its type has a row rather than because something said to. That is the
771        // opposite of the rule: only a binding takes a part apart.
772        if !walk.bindings.is_declared(&site) {
773            return None;
774        }
775        let crossing = Crossing::new(part.clone(), Direction::Deconstruct);
776        let bound = walk.bindings.resolve(&site, &crossing, self.recipes())?;
777        match self.recipes().get(&bound.recipe)? {
778            Recipe::Deconstructing(shape) => Some((bound.recipe, shape)),
779            Recipe::Constructing(_) => None,
780        }
781    }

783    /// One alternative of a declared sum: its name, and its payload fields.
784    fn alternative(
785        &self,
786        source: &TypeRef,
787        index: usize,
788    ) -> Option<(&syn::Ident, &[prebindgen_flat::flat::Field])> {
789        let value = source.borrow_target().unwrap_or(source).unwrapped();
790        let prebindgen_flat::flat::TypeKind::Named { id, .. } = value.kind() else {
791            return None;
792        };
793        let prebindgen_flat::flat::Type::Variant(v) = self.model().resolve(id)? else {
794            return None;
795        };
796        let alternative = v.alternatives.get(index)?;
797        Some((&alternative.name, alternative.fields.as_slice()))
798    }

800    /// This crossing's deconstructing row under `row`, if it states one.
801    fn deconstructing(&self, ty: &TypeRef, row: &RecipeName) -> Option<&Shape<Deconstruct>> {
802        let crossing =
803            crate::recipe::Crossing::new(ty.clone(), crate::recipe::Direction::Deconstruct).key();
804        match self.recipes().get(self.recipes().key_of(&crossing, row)?)? {
805            Recipe::Deconstructing(shape) => Some(shape),
806            Recipe::Constructing(_) => None,
807        }
808    }
809
810    /// What a part's own leaf carries, when it is not spliced.
811    fn leaf_ty(&self, reach: &Reach, source: &TypeRef, index: usize) -> TypeRef {
812        match reach {
813            Reach::Accessor(func) => self.returns(func).unwrap_or_else(|_| source.clone()),
814            Reach::Field(field) => self
815                .field_ty(source, *field)
816                .unwrap_or_else(|| source.clone()),
817            _ => {
818                let _ = index;
819                source.clone()
820            }
821        }
822    }
823
824    fn returns(&self, func: &syn::Ident) -> Result<TypeRef, UnfoldViewError> {
825        self.model()
826            .function(func)
827            .map(|f| f.ret.clone())
828            .ok_or_else(|| UnfoldViewError::UnknownAccessor(func.clone()))
829    }
830
831    fn consumes(&self, func: &syn::Ident) -> Result<bool, UnfoldViewError> {
832        let f = self
833            .model()
834            .function(func)
835            .ok_or_else(|| UnfoldViewError::UnknownAccessor(func.clone()))?;
836        Ok(f.params
837            .first()
838            .is_some_and(|p| p.ty.borrow_target().is_none()))
839    }
840
841    fn field_ty(&self, source: &TypeRef, index: usize) -> Option<TypeRef> {
842        self.fields_of(source).get(index).map(|f| f.ty.clone())
843    }
844
845    /// How a field is addressed: its name, or its position for a tuple struct.
846    fn field_ident(&self, source: &TypeRef, index: usize) -> syn::Ident {
847        self.fields_of(source)
848            .get(index)
849            .and_then(|f| f.name.clone())
850            .unwrap_or_else(|| {
851                syn::Ident::new(&format!("_{index}"), proc_macro2::Span::call_site())
852            })
853    }
854}

856/// Two leaves of one decomposition may not share a name: the foreign signature
857/// names its arguments, and two of one name is one the emitter cannot write.
858fn unique_names(leaves: &[UnfoldLeaf], source: &TypeRef) -> Result<(), UnfoldViewError> {
859    let mut seen = std::collections::HashSet::new();
860    for leaf in leaves {
861        if !seen.insert(leaf.name.clone()) {
862            return Err(UnfoldViewError::DuplicateName {
863                ty: source.key().to_string(),
864                name: leaf.name.clone(),
865            });
866        }
867    }
868    Ok(())
869}

871/// The stable code for a shape the walk stops at.
872///
873/// Identifiers, not phrases, for the same reason [`UnfoldViewError::code`]
874/// exists: this reaches a caller's record of the outcome, and a record must not
875/// move when the wording does. `Display` renders the prose from the code.
876fn name_of(shape: &Shape<Deconstruct>) -> &'static str {
877    match shape {
878        Shape::Atomic => "atomic-row",
879        Shape::Optional => "optional-layer",
880        Shape::Sequence => "sequence-layer",
881        Shape::Invoke => "callable",
882        Shape::Product(_) => "product-row",
883        Shape::Choice { .. } => "choice-row",
884    }
885}
886
887/// How a payload field is addressed in its arm's pattern: by name, or by
888/// position for a tuple variant.
889fn member_of(field: &prebindgen_flat::flat::Field, index: usize) -> syn::Member {
890    match &field.name {
891        Some(name) => syn::Member::Named(name.clone()),
892        None => syn::Member::Unnamed(syn::Index::from(index)),
893    }
894}

896/// The code for a reach the view does not read, and the message for it.
897fn reach_form(reach: &Reach) -> &'static str {
898    match reach {
899        Reach::Path(_) => "field-of-a-field-chain",
900        Reach::Nested { .. } => "field-taken-apart-in-place",
901        _ => "unread-reach",
902    }
