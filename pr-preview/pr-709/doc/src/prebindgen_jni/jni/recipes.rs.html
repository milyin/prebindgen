1//! JniGen's Kotlin class declarations, lowered into the registry's recipe
2//! vocabulary.
3//!
4//! A build script writes `ptr_class!`, `data_class!`, `enum_class!` or
5//! `sealed_class!`, each naming the Kotlin type one Rust type becomes, and
6//! `convert!` for a conversion the binding supplies itself. This turns those
7//! into recipes in a [`Recipes`] table: the shared statement of **which parts** a
8//! value gets across in, with nothing about the JNI wire in it.
9//!
10//! One recipe per declared type and direction, and every crossing nobody declared
11//! takes
12//! the recipe the registry derives from its kind. The chain of guesses this
13//! replaced — try a terminal, then a `Result`, then an optional, then a run,
14//! then a borrow, then a transparent bridge — is a lookup for the declared
15//! half, and the derived half is the registry's.
16

29/// Every type one captured item names, so a reading nested inside it can be
30/// found without the caller knowing which kind of item it came from.
31fn element_types(element: &prebindgen_registry::Element) -> Vec<&TypeRef> {
32    use prebindgen_registry::flat::Type;
33    match element {
34        prebindgen_registry::Element::Function(f) => f
35            .params
36            .iter()
37            .map(|p| &p.ty)
38            .chain(std::iter::once(&f.ret))
39            .collect(),
40        prebindgen_registry::Element::Type(Type::Struct(s)) => {
41            s.fields.iter().map(|f| &f.ty).collect()
42        }
43        prebindgen_registry::Element::Type(Type::Variant(v)) => v
44            .alternatives
45            .iter()
46            .flat_map(|a| a.fields.iter().map(|f| &f.ty))
47            .collect(),
48        prebindgen_registry::Element::Constant(c) => vec![&c.ty],
49        _ => Vec::new(),
50    }
51}

53/// One arm per alternative of a declared sum, each assembled from its own
54/// payload fields.
55///
56/// Empty for anything the model does not hold as a data-carrying enum, which
57/// includes a `sealed_class!` declared over a type the scan dropped — the scan
58/// reports that, and a recipe over no arms would report it a second time.
59fn alternatives(model: &Flat, ty: &TypeRef) -> Vec<Arm<Construct>> {
60    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
61        return Vec::new();
62    };
63    let Some(prebindgen_registry::flat::Type::Variant(v)) =
64        id.ident().and_then(|ident| model.declared_type(&ident))
65    else {
66        return Vec::new();
67    };
68    v.alternatives
69        .iter()
70        .map(|alt| Arm {
71            alternative: Some(alt.index),
72            op: Construct::Fields,
73        })
74        .collect()
75}

77/// One arm per alternative of a declared sum, each taken apart into its own
78/// payload fields.
79///
80/// The deconstructing twin of [`alternatives`]. Separate because the two directions
81/// name different operations — building an alternative is `Construct::Fields`,
82/// reading one is `Deconstruct::Fields` over the reaches that get there — and
83/// a shared helper would have to be generic over an operation neither side
84/// chooses.
85fn out_alternatives(model: &Flat, ty: &TypeRef) -> Vec<Arm<Deconstruct>> {
86    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
87        return Vec::new();
88    };
89    let Some(prebindgen_registry::flat::Type::Variant(v)) =
90        id.ident().and_then(|ident| model.declared_type(&ident))
91    else {
92        return Vec::new();
93    };
94    v.alternatives
95        .iter()
96        .map(|alt| Arm {
97            alternative: Some(alt.index),
98            op: Deconstruct::Fields((0..alt.fields.len()).map(Reach::Field).collect()),
99        })
100        .collect()
101}

104    /// How many values this type hands out, or 0 if it hands out none.
105    ///
106    /// Model and declaration only, like the compositions it asks — so a
107    /// declaration-time caller may ask it, which is what binding a return site
108    /// needs.
109    fn out_wire_count(
110        &self,
111        registry: &(impl prebindgen_registry::Conversions + ?Sized),
112        ty: &TypeRef,
113    ) -> usize {
114        let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
115            return 0;
116        };
117        let Some(ident) = id.ident() else { return 0 };
118        match registry.flat().declared_type(&ident) {
119            Some(prebindgen_registry::flat::Type::Variant(_)) => self
120                .sum_out_wires(registry.flat(), &ident, ty)
121                .map_or(0, |w| w.len()),
122            Some(prebindgen_registry::flat::Type::Struct(_)) => {
123                self.struct_out_wires(registry, ty).map_or(0, |w| w.len())
124            }
125            _ => 0,
126        }
127    }

129    /// The accessor a type's value form names, and one reach per field of the
130    /// struct it returns.
131    ///
132    /// `None` unless the declaration is exactly one `.fields(fields!(f))` whose
133    /// every field is a plain leaf. A record that states its own decomposition
134    /// — a per-field `expand_return!` override, an inlined nested class, a sum
135    /// laid out as a selector and its groups — is a shape this recipe does not
136    /// state yet, and stating half of one would describe a boundary nothing
137    /// emits.
138    fn value_form_of(
139        &self,
140        model: &Flat,
141        registry: &(impl prebindgen_registry::Conversions + ?Sized),
142        ty: &TypeRef,
143    ) -> Option<(syn::Ident, Vec<Reach>)> {
144        use crate::unfold::{FieldDecon, FieldRecord};
145        let decl = self
146            .return_expand_decls
147            .iter()
148            .find(|d| *d.key() == ty.stripped_key())?;
149        let [crate::jni::LocalField::Fields(fields)] = decl.field_list() else {
150            return None;
151        };
152        let records: Vec<FieldRecord> = self.lower_value_form(registry, decl.key(), fields);
153        let ret = model.function(fields.func())?.ret.clone();
154        let ret = ret.borrow_target().unwrap_or(&ret);
155        let prebindgen_registry::flat::TypeKind::Named { id, .. } = ret.unwrapped().kind() else {
156            return None;
157        };
158        let prebindgen_registry::flat::Type::Struct(st) =
159            id.ident().and_then(|i| model.declared_type(&i))?
160        else {
161            return None;
162        };
163        let mut reaches = Vec::new();
164        for record in &records {
165            if !matches!(record.decon, FieldDecon::Default) {
166                return None;
167            }
168            // A record with several members is an INLINED nested class: each
169            // member is one hop of a field-access chain. `Reach::Path` states
170            // that; before it existed this row declined and fell back to a
171            // placeholder (#613 step 10).
172            let mut here = st;
173            let mut indices = Vec::with_capacity(record.members.len());
174            for member in &record.members {
175                let hop = here
176                    .fields
177                    .iter()
178                    .position(|f| f.name.as_ref() == Some(member))?;
179                indices.push(hop);
180                if indices.len() < record.members.len() {
181                    let next = here.fields[hop].ty.stripped_key().ident()?;
182                    let prebindgen_registry::flat::Type::Struct(next) =
183                        model.declared_type(&next)?
184                    else {
185                        return None;
186                    };
187                    here = next;
188                }
189            }
190            let [index] = indices.as_slice() else {
191                reaches.push(Reach::Path(indices));
192                continue;
193            };
194            let index = *index;
195            // A field that reaches the type being taken apart. The leaf
196            // synthesis treats `Box<T>` as a spelling of its own and stops
197            // there; a recipe keys a crossing by the value that crosses, so
198            // `Box<ZSample>` inside `ZSample`'s own value form is that recipe
199            // reaching itself. Refused here rather than by `Recipes::build`,
200            // which would refuse the whole binding over a shape the leaf
201            // synthesis handles.
202            if st.fields[index].ty.stripped_key() == ty.stripped_key() {
203                return None;
204            }
205            reaches.push(Reach::Field(index));
206        }
207        Some((fields.func().clone(), reaches))
208    }
209}

212    /// What a value form calls each of its fields, by the Rust field ident.
213    ///
214    /// The declaration's answer, so a `.name(..)` rename carries through to the
215    /// builder parameter. `None` for a type with no value form, or one whose
216    /// records this recipe does not state — the same test [`Self::value_form_of`]
217    /// makes when it declares the recipe.
218    pub(crate) fn value_form_names(
219        &self,
220        registry: &(impl prebindgen_registry::Conversions + ?Sized),
221        ty: &TypeRef,
222    ) -> Option<std::collections::HashMap<String, String>> {
223        use crate::unfold::FieldDecon;
224        let decl = self
225            .return_expand_decls
226            .iter()
227            .find(|d| *d.key() == ty.stripped_key())?;
228        let [crate::jni::LocalField::Fields(fields)] = decl.field_list() else {
229            return None;
230        };
231        self.lower_value_form(registry, decl.key(), fields)
232            .into_iter()
233            .map(|r| match (r.members.as_slice(), &r.decon) {
234                ([member], FieldDecon::Default) => Some((member.to_string(), r.name)),
235                _ => None,
236            })
237            .collect()
238    }

241/// One reach per field of a declared struct, in the model's order.
242///
243/// Empty for anything the model does not hold as a struct, which includes a
244/// `data_class!` over a type the scan dropped — the scan reports that, and a
245/// recipe over no fields would report it a second time.
246fn out_fields(model: &Flat, ty: &TypeRef) -> Vec<Reach> {
247    (0..fields_of(model, ty).len()).map(Reach::Field).collect()
248}

250/// The fields of a declared struct, or none for anything the model does not
251/// hold as one.
252fn fields_of<'a>(model: &'a Flat, ty: &TypeRef) -> &'a [prebindgen_registry::flat::Field] {
253    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
254        return &[];
255    };
256    match id.ident().and_then(|ident| model.declared_type(&ident)) {
257        Some(prebindgen_registry::flat::Type::Struct(s)) => &s.fields,
258        _ => &[],
259    }
260}

279/// A name of its own, because it sits on the same crossing as the type's

294/// Whether a reach's type mentions `owner` anywhere inside it.
295///
296/// Deliberately NOT a delivery peel. [`TypeRef::layer_stack`] answers "what
297/// does this crossing deliver", and is bounded because `Vec<Option<T>>` has
298/// the optional inside the ELEMENT rather than in the boundary. The question
299/// here is a different one — "would splicing this part make the row reach its
300/// own crossing" — and a self-reference nested out of position still closes the
301/// loop: `T`'s row reaches `Vec<Option<T>>`, whose element row reaches `T`.
302/// Answering it with a peel of either depth conflates the two.
303fn mentions(ty: &TypeRef, owner: &TypeKey) -> bool {
304    ty.walk().iter().any(|node| node.stripped_key() == *owner)
305}

308    /// Every recipe this binding's declarations state.
309    ///
310    /// A type declared but absent from the model is skipped rather than
311    /// refused: the scan already reports it, and reporting it twice in
312    /// different words helps nobody.
313    /// This type's `parts` decomposition, read off its `expand_return!`
314    /// declaration in the recipe table's own vocabulary.
315    ///
316    /// `None` when the declaration states a shape a `Deconstruct` cannot hold —
317    /// a value-form record mixed with others, since `ValueForm` is the whole
318    /// row while `LocalField::Fields` is one record among many. No declaration
319    /// in this workspace has that shape (measured: every `Fields` record stands
320    /// alone), so the caller's `Atomic` fallback is a guard rather than a path
321    /// anything takes.
322    fn parts_deconstruct(
323        &self,
324        model: &Flat,
325        registry: &(impl prebindgen_registry::Conversions + ?Sized),
326        ty: &TypeRef,
327    ) -> Option<Deconstruct> {
328        let decl = self
329            .return_expand_decls
330            .iter()
331            .find(|d| *d.key() == ty.stripped_key())?;
332        self.decon_of(model, registry, ty, decl)
333    }

335    /// [`Self::parts_deconstruct`] against ONE declaration, whichever states
336    /// it: the type's own, or the one a function wrote for its return.
337    fn decon_of(
338        &self,
339        model: &Flat,
340        registry: &(impl prebindgen_registry::Conversions + ?Sized),
341        ty: &TypeRef,
342        decl: &crate::jni::ExpandReturnDecl,
343    ) -> Option<Deconstruct> {
344        let fields = decl.field_list();
345        if matches!(fields, [crate::jni::LocalField::Fields(_)]) {
346            // `ValueForm` means "call this accessor, then read these parts off
347            // its result" — the parts are how the compiler reaches the returned
348            // struct's fields, so an empty list is a decomposition with nothing
349            // in it rather than one of everything (#638 review).
350            //
351            // `value_form_of` already computes exactly that mapping, and it is
352            // not a count: `lower_value_form` returns FLATTENED records, one of
353            // which can splice a child's fields, so `Report` yields six records
354            // over a five-field struct. It maps each record back to the struct
355            // field it names and declines the shapes a row cannot hold.
356            let (func, parts) = self.value_form_of(model, registry, ty)?;
357            return Some(Deconstruct::ValueForm { func, parts });
358        }
359        let mut reaches = Vec::with_capacity(fields.len());
360        for field in fields {
361            // A field that reaches the type being taken apart — the
362            // conditional-handle idiom, `fn(&T) -> Option<&T>`. Splicing it
363            // would make this row reach its own crossing; the field crosses as
364            // one nullable handle leaf instead. Refused here for the same
365            // reason `value_form_of` refuses its own case: `Recipes::build`
366            // would otherwise fail the whole binding over one field.
367            let reaches_self = match field {
368                crate::jni::LocalField::Local { sig, .. } => match &sig.output {
369                    syn::ReturnType::Type(_, ret) => model
370                        .classify(ret)
371                        .is_ok_and(|r| mentions(&r, &ty.stripped_key())),
372                    syn::ReturnType::Default => false,
373                },
374                crate::jni::LocalField::Named(func, _) => model
375                    .function(func)
376                    .is_some_and(|f| mentions(&f.ret, &ty.stripped_key())),
377                _ => false,
378            };
379            if reaches_self {
380                return None;
381            }
382            reaches.push(match field {
383                crate::jni::LocalField::Named(func, _) => Reach::Accessor(func.clone()),
384                // The handle itself — the leaf `Reach::Identity` was added for.
385                crate::jni::LocalField::SelfField => Reach::Identity,
386                // A binding-local accessor resolves by its path's last segment:
387                // `local_functions()` registers it under exactly that ident.
388                crate::jni::LocalField::Local { path, .. } => {
389                    Reach::Accessor(path.segments.last()?.ident.clone())
390                }
391                crate::jni::LocalField::Fields(_) => return None,
392            });
393        }
394        // A row names a part by the accessor that reaches it, so one accessor
395        // used twice — `.field(fun!(f).name("a")).field(fun!(f).name("b"))` —
396        // is two parts the row cannot tell apart. The declaration's name
397        // overrides say which is which and `Reach` has nowhere to put them, so
398        // the row declines rather than stating two leaves of one name.
399        let mut seen: Vec<&syn::Ident> = Vec::with_capacity(reaches.len());
400        for reach in &reaches {
401            if let Reach::Accessor(func) = reach {
402                if seen.contains(&func) {
403                    return None;
404                }
405                seen.push(func);
406            }
407        }
408        Some(Deconstruct::Fields(reaches))
409    }

411    pub(crate) fn recipes(
412        &self,
413        model: &Flat,
414        expansion_leaves: &[TypeRef],
415        registry: &(impl prebindgen_registry::Conversions + ?Sized),
416    ) -> Result<Recipes, Vec<RecipeError>> {
417        let mut recipes = Recipes::builder();
418        // How each `expand_param!`-declared type is built. Named rather than
419        // defaulted: the crossing's default stays the value's own conversion,
420        // which is what an identity arm is handed (#701 step 2).
421        // Which crossings already state a deconstructing `parts` row, so the
422        // decomposition block below adds one only where none was declared.
423        let mut parts_out: std::collections::HashSet<TypeKey> = std::collections::HashSet::new();
424        // Every Kotlin class declaration, and every `convert!`-declared
425        // conversion. The second matters as much as the first: a conversion may
426        // be declared on a type the registry would otherwise read as an arity
427        // layer, and `convert!(Option<Duration> => ..)` means the adapter emits
428        // that optional's conversion itself rather than wrapping `Duration`'s.
429        let mut declared: Vec<(TypeKey, Origin<syn::Type>)> = self
430            .declared_types()
431            .into_iter()
432            .chain(
433                self.convert_decls
434                    .iter()
435                    .map(|d| (d.key().clone(), d.rust_type().clone())),
436            )
437            .collect();
438        declared.sort_by(|a, b| a.0.as_str().cmp(b.0.as_str()));
439        declared.dedup_by(|a, b| a.0 == b.0);
440
441        for (_key, origin) in declared {
442            // The declarator's own tokens, re-parsed: a declared type may be an
443            // alias the model holds as an `Extern`, which carries no reading of
444            // its own to borrow.
445            let Ok(spelled) = syn::parse2::<syn::Type>(origin.declared_spelling()) else {
446                continue;
447            };
448            let Ok(ty) = model.classify(&spelled) else {
449                continue;
450            };
451            // Every declared Kotlin shape is one recipe with no parts today,
452            // including the two that plainly have some: a `data_class` arrives
453            // as separate JNI parameters and is rebuilt inside one generated
454            // function, and a `sealed_class` is a selector plus the live arm's
455            // slots inside another. `Atomic` is what a recipe says about that —
456            // the adapter emits the conversion itself, and how many wire values
457            // that costs is its own business — so it describes the adapter as it
458            // stands rather than as it will be.
459            // A `sealed_class` hands its value out as a tag plus every
460            // alternative's payloads, laid side by side — the deconstructing
461            // twin of the constructing recipe below, and the direction that
462            // actually has no alternative: a sum has no whole-value output
463            // conversion, because a tag plus groups is not one wire.
464            match self.types.get(&ty.key()).map(|c| &c.kind) {
465                Some(DeclaredKind::Sealed(_)) => {
466                    let arms = out_alternatives(model, &ty);
467                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
468                    if !arms.is_empty() {
469                        recipes.declare(ty.clone(), parts(), Deconstructing::Choice { arms });
470                        parts_out.insert(ty.stripped_key());
471                    }
472                }
473                // A `data_class` hands its value out as its fields too, so the
474                // foreign side reassembles the object and nothing builds one on
475                // the Rust side.
476                Some(DeclaredKind::Data) => {
477                    let reaches = out_fields(model, &ty);
478                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
479                    if !reaches.is_empty() {
480                        recipes.declare(
481                            ty.clone(),
482                            parts(),
483                            Deconstructing::Product(Deconstruct::Fields(reaches)),
484                        );
485                        parts_out.insert(ty.stripped_key());
486                    }
487                }
488                _ => {
489                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
490                }
491            }
492            // A value form: `expand_return!(T).fields(fields!(f))` calls `f`
493            // once and hands out the fields of the struct it returns. The one
494            // declaration shape that names its own accessor, and the reason
495            // `Deconstruct::ValueForm` exists.
496            if let Some((func, reaches)) = self.value_form_of(model, registry, &ty) {
497                recipes.declare(
498                    ty.clone(),
499                    parts(),
500                    Deconstructing::Product(Deconstruct::ValueForm {
501                        func,
502                        parts: reaches,
503                    }),
504                );
505                parts_out.insert(ty.stripped_key());
506            }
507            // A `data_class` also has a recipe that says what it is made of, so
508            // its constructing side names which of the two a site takes by
509            // default. See [`parts`] for why that is still `whole`.
510            match self.types.get(&ty.key()).map(|c| &c.kind) {
511                // A struct with no fields is nothing to be made of, so it
512                // states no `parts` recipe — the same condition the deconstructing
513                // side applies, and the reason the row is looked up by name before
514                // it is compiled.
515                Some(DeclaredKind::Data) if !fields_of(model, &ty).is_empty() => {
516                    recipes
517                        .declare_default(ty.clone(), whole(), Constructing::Atomic)
518                        .declare_unasked(ty, parts(), Constructing::Product(Construct::Fields));
519                }
520                // A `sealed_class` has one too, and it is a choice rather than
521                // a product: exactly one alternative is live, every one of them
522                // still crosses, and the tag says which. The arms are the
523                // model's — a recipe states which parts, never what they are.
524                Some(DeclaredKind::Sealed(_)) => {
525                    let arms = alternatives(model, &ty);
526                    recipes.declare_default(ty.clone(), whole(), Constructing::Atomic);
527                    if !arms.is_empty() {
528                        recipes.declare(ty, parts(), Constructing::Choice { arms });
529                    }
530                }
531                _ => {
532                    recipes.declare(ty, whole(), Constructing::Atomic);
533                }
534            }
535        }
536
537        // A type a callback delivers by taking it apart states that as a row,
538        // so the argument's site has a recipe to name. Without one the site
539        // could only be fabricated — a `Bound` no binding answered — which is
540        // what #622's first two attempts produced and why they were withdrawn.
541        //
542        // `Atomic` for the reason the declared-type loop above gives for the
543        // two shapes that plainly have parts: the adapter emits this
544        // conversion itself, and how many wire values that costs is its own
545        // business. The parts are stated by the deconstructor declaration the
546        // registry already resolved into an `UnfoldPlan`, which this row does
547        // not restate — it names the crossing that plan decomposes. Deleting
548        // that second statement is #613 step 5b's, not this row's.
549        //
550        // The source is the plan's own owned core, so a plan keyed under `&T`
551        // and one keyed under `T` state one row; a type that already declared
552        // `parts` — a `data_class`, a `sealed_class`, a value form — keeps it.
553        //
554        // A **whole-element fold** is not one of these and must not earn a row.
555        // `apply_leaf_vec_folds` files a plan for `impl Fn(&[T])` under `&[T]`
556        // whose `source` is the ELEMENT and whose `decon` is `None`: nothing is
557        // taken apart, each element crosses whole through its own converter.
558        // Declaring `parts` off it would state a decomposition of `T` that does
559        // not exist — and, worse, a scalar `T` argument elsewhere in the same
560        // model would then bind to that row and name a fragment compiled under
561        // a different recipe (#623 review). `decon` is the gate the model
562        // already carries for this: `None` only for the whole-element arm.
563        let mut decomposed: BTreeMap<TypeKey, TypeRef> = BTreeMap::new();
564        // EVERY decomposition, not only a callback argument's. A return and a
565        // domain error come apart by the same declaration — `Error` is one
566        // declaration reached from every fallible function in zenoh-flat-jni —
567        // so each states its row too. Without this the rows described a
568        // fraction of what a binding decomposes, which is what 68 entries of
569        // the parity inventory said (#701 step 3).
570        let unfolded = self.unfolded();
571        let every_plan = unfolded
572            .callback_arg_plans
573            .values()
574            .chain(unfolded.unfold_plans.values())
575            .chain(unfolded.error_plans.values());
576        for plan in every_plan {
577            if plan.decon.is_none() {
578                continue;
579            }
580            let key = plan.source.stripped_key();
581            if parts_out.contains(&key) {
582                continue;
583            }
584            decomposed.entry(key).or_insert_with(|| plan.source.clone());
585        }
586        for ty in decomposed.into_values() {
587            // A real row where the declaration can state one. #622 wrote
588            // `Atomic` here because `Reach` could not spell an identity leaf;
589            // it can now (#635), so the row says how the value comes apart
590            // instead of only existing to be selected (#613 step 10).
591            match self.parts_deconstruct(model, registry, &ty) {
592                Some(deconstruct) => {
593                    recipes.declare(ty, parts(), Deconstructing::Product(deconstruct));
594                }
595                None => {
596                    recipes.declare(ty, parts(), Deconstructing::Atomic);
597                }
598            }
599        }
600
601        // A per-function `.expand_return(...)` states a decomposition of its
602        // own, so it gets a row of its own, under a name of its own — the shape
603        // step 2 gave the parameter side (`expand-param@fn#param`), at the
604        // other end. Without it the type's `parts` row and the function's
605        // decomposition sat under one name describing two different things,
606        // which is why the differential could not compare either.
607        for (func, decl) in &self.fn_return_expands {
608            let Some(ty) = decl
609                .key()
610                .ident()
611                .and_then(|ident| model.classify(&syn::parse_quote!(#ident)).ok())
612            else {
613                continue;
614            };
615            match self.decon_of(model, registry, &ty, decl) {
616                Some(deconstruct) => {
617                    recipes.declare(ty, site_row(func), Deconstructing::Product(deconstruct));
618                }
619                None => {
620                    recipes.declare(ty, site_row(func), Deconstructing::Atomic);
621                }
622            }
623        }
624
625        // Every implicit Optional keeps its established whole/default input
626        // row and also offers the allocation-free primitive `pair` row. A
627        // parameter site selects `pair` only after the compiled payload proves
628        // that it is one JNI primitive with no niche, projection, or stage.
629        //
630        // Product-shaped payloads additionally offer `parts`; that row is
631        // selected by the existing flattening bindings below.
632        let mut implicit: BTreeMap<TypeKey, (TypeRef, TypeRef)> = self
633            .implicit_optionals(model)
634            .into_iter()
635            .map(|(key, (outer, inner))| (key, (outer.clone(), inner.clone())))
636            .collect();
637        // A combined constructor expansion represents an inactive arm by
638        // Option-wrapping each required leaf. Those readings are synthesized
639        // after the Flat model was built, so a model walk cannot discover
640        // them. File them in the same table as source-written implicit
641        // Optionals: otherwise a nullable primitive leaf falls back to a boxed
642        // JObject and Rust has to call `intValue()`/`longValue()` across JNI.
643        for leaf in expansion_leaves {
644            let Some(inner) = leaf.optional_inner() else {
645                continue;
646            };
647            if !self.explicitly_declares(leaf) {
648                implicit
649                    .entry(leaf.stripped_key())
650                    .or_insert_with(|| (leaf.clone(), inner.clone()));
651            }
652        }
653        for (outer, _) in implicit.values() {
654            recipes
655                .declare_default(outer.clone(), whole(), Constructing::Optional)
656                .declare(outer.clone(), pair(), Constructing::Optional)
657                .declare_unasked(outer.clone(), parts(), Constructing::Optional);
658        }
659        for (outer, _) in implicit.into_values() {
660            recipes
661                .declare_default(outer.clone(), whole(), Deconstructing::Optional)
662                .declare_unasked(outer.clone(), parts(), Deconstructing::Optional);
663        }
664
665        // A fixed-size array of JNI primitives is one Kotlin `ByteArray` or
666        // `LongArray`, bulk-copied with nothing boxed — one wire value, not a
667        // run this adapter walks. The registry reads `[T; N]` as a run unless
668        // something says otherwise, and this is JniGen saying otherwise for
669        // every array the model holds. Nothing is enumerated twice: a recipe for
670        // a crossing nobody uses is inert.
671        // Keyed by identity in a `BTreeMap`, so each array type is keyed once
672        // and the declaration order is the key order — no sort, no second pass
673        // rebuilding a key it already has, and nothing cloned before the entry
674        // turns out to be new.
675        let mut arrays: BTreeMap<TypeKey, &TypeRef> = BTreeMap::new();
676        for ty in model
677            .elements()
678            .flat_map(element_types)
679            .flat_map(|ty| ty.walk())
680            .filter(|ty| matches!(ty.kind(), prebindgen_registry::flat::TypeKind::Array { .. }))
681        {
682            arrays.entry(ty.key()).or_insert(ty);
683        }
684        for ty in arrays.into_values() {
685            recipes
686                .declare(ty.clone(), whole(), Deconstructing::Atomic)
687                .declare(ty.clone(), whole(), Constructing::Atomic);
688        }
689
690        // The `expand_param!` rows are NOT declared here. They live in the
691        // table `Declarations::expansion_plans` builds for itself, because the
692        // readings they deliver have to be known while the binding declares
693        // itself — before this table can exist. Declaring them here as well
694        // would put a second copy in front of a walk that never reaches them:
695        // no site names one, and both adapters' `choice` refuse the arm shape
696        // they use. They move here when something reads them, in #701's step 3.
697        recipes.build(model)
698    }
699}

701impl Declarations {
702    /// Which recipe each part of a `data_class` takes.
703    ///
704    /// A field that is itself a `data_class` crosses as **its** fields too, so
705    /// the part site asks for [`parts`] rather than letting the field take its
706    /// own default. That is the one thing a site can say that a recipe cannot: the
707    /// same crossing is read one way on its own and another inside a product.
708    ///
709    /// Without it a nested class contributes a single wire and the flattening
710    /// stops one layer down.
711    /// Whether a `data_class` field is itself flattened, or stays one value.
712    ///
713    /// A `data_class` declared `.jobject_input()` crosses Kotlin → Rust as a
714    /// single `JObject` — that is the whole point of the opt-in — so binding
715    /// its parts would flatten a boundary the declaration drew.
716    /// Whether this binding takes a type APART where it appears inside another
717    /// value, rather than crossing it whole.
718    ///
719    /// Three declarations say so: a `data_class`, a `sealed_class`, and a value
720    /// form — `expand_return!(T).fields(..)`, which names its own accessor. The
721    /// proxy for this used to be "declares a `parts` row", which held only
722    /// while an inlined class was the one thing that had one; every
723    /// decomposition states a row now (#701 step 3).
724    fn inlines_parts(
725        &self,
726        model: &Flat,
727        registry: &(impl prebindgen_registry::Conversions + ?Sized),
728        key: &TypeKey,
729    ) -> bool {
730        if self
731            .types
732            .get(key)
733            .is_some_and(|c| c.sum().is_some() || matches!(c.kind, DeclaredKind::Data))
734        {
735            return true;
736        }
737        key.ident()
738            .and_then(|ident| model.classify(&syn::parse_quote!(#ident)).ok())
739            .is_some_and(|ty| self.value_form_of(model, registry, &ty).is_some())
740    }
741
742    fn field_crosses_as_its_fields(&self, ty: &TypeRef) -> bool {
743        // The field's own key, not the stripped one: a `Box<Leaf>` field
744        // crosses as one value through `Box<Leaf>`'s conversion, and only a
745        // parameter spelled that way is peeled back to the class. Binding the
746        // stripped key here would flatten a field the rebuild has no node for.
747        self.types.get(&ty.key()).is_some_and(|c| match c.kind {
748            DeclaredKind::Data => !c.jobject_input,
749            // A `sealed_class` field crosses as a tag plus every alternative's
750            // slots. Whether it *can* is the adapter's answer at compile time,
751            // not a declaration: a payload with no slot form leaves the sum
752            // object-shaped, which is the fragment `Compile::choice` hands
753            // back. So the site asks, and the recipe answers.
754            DeclaredKind::Sealed(_) => true,
755            _ => false,
756        })
757    }
758
759    /// Optional crossings whose payload can use the shared `parts` recipe.
760    ///
761    /// Explicit declarations keep full control of the outer crossing: a
762    /// `convert!(Option<T> => ..)` is an atomic recipe, not an implicit request
763    /// to flatten `T`.
764    fn implicit_optionals<'a>(
765        &self,
766        model: &'a Flat,
767    ) -> BTreeMap<TypeKey, (&'a TypeRef, &'a TypeRef)> {
768        let mut optionals = BTreeMap::new();
769        for ty in model
770            .elements()
771            .flat_map(element_types)
772            .flat_map(|ty| ty.walk())
773        {
774            let Some(inner) = ty.optional_inner() else {
775                continue;
776            };
777            if !self.explicitly_declares(ty) {
778                optionals.entry(ty.stripped_key()).or_insert((ty, inner));
779            }
780        }
781        optionals
782    }
783
784    fn explicitly_declares(&self, ty: &TypeRef) -> bool {
785        self.types.contains_key(&ty.key())
786            || self
787                .convert_decls
788                .iter()
789                .any(|decl| decl.key() == &ty.key() || decl.key() == &ty.stripped_key())
790    }

792    pub(crate) fn bindings(
793        &self,
794        model: &Flat,
795        registry: &(impl prebindgen_registry::Conversions + ?Sized),
796        recipes: &prebindgen_registry::recipe::Recipes,
797    ) -> Result<prebindgen_registry::recipe::Bindings, Vec<RecipeError>> {
798        use prebindgen_registry::recipe::{
799            Ask, Bindings, Crossing, Direction, Origin, RecipeName, Site,
800        };
801
802        let mut bound = Bindings::builder();
803
804        // What crosses at a **convert** return is not the value the signature
805        // names. The binding converts it first, and the converted value's type
806        // is the function's own output expansion answer, which no model can
807        // state. Bound here so the walk plans that crossing rather than the
808        // return type, and so the site needs no adapter-private planning.
809        // What an **expanded** return crosses. A return the binding converts or
810        // takes apart does not cross the value its signature names, and only
811        // the binding's own output expansion says what it does cross. Bound
812        // here so the walk plans that crossing, which is what lets the site be
813        // planned once rather than again at the emission site.
814        //
815        // One binding per function, in the same precedence `build_output`
816        // applies: a function may have BOTH an output expansion and an error
817        // plan, and the output expansion wins. Binding each in its own pass
818        // would answer one site twice and be refused as a rebind.
819        let mut owners: std::collections::BTreeSet<String> = Default::default();
820        let unfolded = self.unfolded();
821        owners.extend(unfolded.unfold_plans.keys().map(|f| f.to_string()));
822        owners.extend(unfolded.error_plans.keys().map(|f| f.to_string()));
823        for owner in owners {
824            let func = quote::format_ident!("{owner}");
825            let (crossed, ask) = match unfolded.unfold_plans.get(&func) {
826                // A convert hands over one converted value and takes whatever
827                // row that value has; a decomposition crosses the value it
828                // takes apart and asks for the row that names its parts.
829                Some(plan) => match plan.delivery {
830                    crate::unfold::Delivery::Return => match plan.convert_out_ty.as_ref() {
831                        Some(out) => (out.clone(), Ask::Default),
832                        None => continue,
833                    },
834                    crate::unfold::Delivery::Callback => {
835                        let crossing = Crossing::new(plan.source.clone(), Direction::Deconstruct);
836                        let ask = if recipes.key_of(&crossing.key(), &parts()).is_some() {
837                            Ask::Recipe(parts())
838                        } else {
839                            Ask::Default
840                        };
841                        (plan.source.clone(), ask)
842                    }
843                },
844                // No output expansion, but the binding peels the error: only
845                // the ok arm crosses at the return, because the error is thrown
846                // rather than crossed there.
847                None => {
848                    let Some((ok, _)) = model
849                        .function(&func)
850                        .and_then(|function| function.ret.fallible_parts())
851                    else {
852                        continue;
853                    };
854                    (ok.clone(), Ask::Default)
855                }
856            };
857            bound.bind(
858                Site {
859                    owner: func,
860                    role: prebindgen_registry::recipe::Role::Return,
861                },
862                Crossing::new(crossed, Direction::Deconstruct),
863                ask,
864                Origin::Function,
865            );
866        }
867
868        let mut declared: Vec<TypeKey> = self
869            .types
870            .iter()
871            .filter(|(_, c)| matches!(c.kind, DeclaredKind::Data))
872            .map(|(k, _)| k.clone())
873            .collect();
874        declared.sort_by(|a, b| a.as_str().cmp(b.as_str()));
875
876        // The named Optional row is the representation suitable inside another
877        // composed shape. Where its child has a Product row it delegates to that
878        // row; a scalar child stays on its default row and lets the JNI compiler
879        // choose a niche or a `(present, value)` intermediate from the compiled
880        // child facts. The whole/default Optional row remains untouched.
881        //
882        // Bind both directions: function inputs construct the Rust optional,
883        // while returns and callback arguments deconstruct it.
884        for (outer, inner) in self.implicit_optionals(model).into_values() {
885            for direction in [Direction::Construct, Direction::Deconstruct] {
886                let outer = Crossing::new(outer.clone(), direction);
887                let row = outer.row(parts());
888                let inner = Crossing::new(inner.clone(), direction);
889                // A DECLARED CLASS's parts are inlined here; anything else
890                // crosses whole. "Has a `parts` row" was the proxy for that
891                // while only a declared class had one — every decomposition
892                // states a row now, so the proxy would inline a field the
893                // decomposition crosses whole. The condition says what it
894                // meant (#701 step 3).
895                let inlines = self.inlines_parts(model, registry, &inner.key().ty)
896                    && recipes.key_of(&inner.key(), &parts()).is_some();
897                bound.bind(
898                    Site::part(&row, 0),
899                    inner.clone(),
900                    if inlines {
901                        Ask::Recipe(parts())
902                    } else {
903                        Ask::Default
904                    },
905                    Origin::Part,
906                );
907            }
908        }
909
910        for key in declared {
911            let Some(ident) = key.ident() else { continue };
912            let Some(prebindgen_registry::flat::Type::Struct(s)) = model.declared_type(&ident)
913            else {
914                continue;
915            };
916            let Ok(ty) = model.classify(&syn::parse_quote!(#ident)) else {
917                continue;
918            };
919            let building = Crossing::new(ty.clone(), Direction::Construct);
920            let handing_out = Crossing::new(ty, Direction::Deconstruct);
921            for (index, field) in s.fields.iter().enumerate() {
922                // A field selects `parts` exactly when its crossing declares
923                // that row AND the declaration says this occurrence decomposes.
924                // Direct classes retain the existing `.jobject_input()` and
925                // erased-wrapper boundaries; every implicit Optional selects its
926                // part representation, which decides whether the child is
927                // another composition or one scalar intermediate.
928                if field.ty.optional_inner().is_none()
929                    && !self.field_crosses_as_its_fields(&field.ty)
930                {
931                    continue;
932                }
933                for (of, direction) in [
934                    (&building, Direction::Construct),
935                    (&handing_out, Direction::Deconstruct),
936                ] {
937                    // Handing a value OUT, an optional field asks the question
938                    // of what is inside the layer: a decomposition looks
939                    // through `Option<T>` only where it inlines T's fields, and
940                    // crosses `Option<Summary>` — an optional handle — whole.
941                    // "Declares a `parts` row" was the proxy for that while only
942                    // an inlined class had one; every decomposition states one
943                    // now, so the proxy would look through a layer the
944                    // decomposition crosses whole (#701 step 3).
945                    //
946                    // Building one is unchanged: a `(present, value)` pair is
947                    // how every optional parameter arrives, whatever is inside.
948                    if direction == Direction::Deconstruct
949                        && field
950                            .ty
951                            .optional_inner()
952                            .is_some_and(|inner| !self.field_crosses_as_its_fields(inner))
953                    {
954                        continue;
955                    }
956                    let row = of.row(parts());
957                    let field_crossing = Crossing::new(field.ty.clone(), direction);
958                    if recipes.key_of(&field_crossing.key(), &parts()).is_none() {
959                        continue;
960                    }
961                    bound.bind(
962                        Site::part(&row, index),
963                        field_crossing,
964                        Ask::Recipe(parts()),
965                        Origin::Part,
966                    );
967                }
968            }
969        }
970        // Every function whose return the binding takes apart, bound to the
971        // `parts` recipe of its return type.
972        //
973        // This is what a site naming several values is: `Ask::Recipe` already
974        // lets a site pick a recipe, and a `parts` fragment already occupies
975        // several wires — the two facts stages 3 and 4 established, meeting.
976        // The registry needs nothing new to express a decomposed return.
977        for f in model.functions() {
978            // The value that crosses, through the layers a decomposition looks
979            // through: a `&T` return decomposes T, and so does an `Option<T>`
980            // or a `Vec<T>` — the shape rides the delivery, not the recipe.
981            let ret = f.ret.borrow_target().unwrap_or(&f.ret);
982            let ret = ret.optional_inner().unwrap_or(ret);
983            let ret = ret.sequence_elem().unwrap_or(ret);
984            let ret = ret.borrow_target().unwrap_or(ret);
985            let crossing = Crossing::new(ret.clone(), Direction::Deconstruct);
986            // Only where the type states one. A `sealed_class` always does; a
987            // `data_class` states one unless a field of it declines, and a
988            // return whose type states none crosses whole.
989            let stated = if self.fn_return_expands.iter().any(|(g, _)| *g == f.name) {
990                site_row(&f.name)
991            } else {
992                parts()
993            };
994            if recipes.key_of(&crossing.key(), &stated).is_none() {
995                continue;
996            }
997            // And only where it is genuinely several. A decomposition that
998            // yields ONE value takes `Delivery::Return`: the wrapper hands that
999            // value back through its own conversion rather than through a
1000            // builder, so the return site's crossing is the value's, not this
1001            // type's. Asked of the composition, which is model and declaration
1002            // only and so answerable here — the same property that let one
1003            // composition serve both sides of `resolve`.
1004            if self.out_wire_count(registry, ret) < 2 {
1005                continue;
1006            }
1007            // A function that wrote its own `.expand_return(...)` names the row
1008            // that states IT, not the type's.
1009            bound.bind(
1010                Site {
1011                    owner: f.name.clone(),
1012                    role: prebindgen_registry::recipe::Role::Return,
1013                },
1014                crossing,
1015                Ask::Recipe(stated),
1016                Origin::Adapter,
1017            );
1018        }
1019
1020        // A callback's `Invoke` recipe owns its argument conversions. Bind a
1021        // model-derived data-class argument to `parts` so the Product fragment
1022        // reaches `Compile::callback` before the trampoline is rendered.
1023        for f in model.functions() {
1024            for param in &f.params {
1025                let prebindgen_registry::flat::TypeKind::Callback { args } =
1026                    param.ty.unwrapped().kind()
1027                else {
1028                    continue;
1029                };
1030                let callback = Crossing::new(param.ty.clone(), Direction::Construct);
1031                let row = callback.row(RecipeName::derived());
1032                for (index, arg) in args.iter().enumerate() {
1033                    let crossed = arg.borrow_target().unwrap_or(arg);
1034                    let core = crossed.optional_inner().unwrap_or(crossed);
1035                    if let Some(element) = core.sequence_elem() {
1036                        let element = element.borrow_target().unwrap_or(element);
1037                        if !self.field_crosses_as_its_fields(element) {
1038                            continue;
1039                        }
1040                        let sequence = Crossing::new(arg.clone(), Direction::Deconstruct);
1041                        let element_crossing =
1042                            Crossing::new(element.borrowed(), Direction::Deconstruct);
1043                        if recipes.key_of(&element_crossing.key(), &parts()).is_none() {
1044                            continue;
1045                        }
1046                        bound.bind(
1047                            Site::part(&sequence.row(RecipeName::derived()), 0),
1048                            element_crossing,
1049                            Ask::Recipe(parts()),
1050                            Origin::Adapter,
1051                        );
1052                        continue;
1053                    }
1054                    // A model-derived data class, or a type whose deconstructor
1055                    // declaration earned the `parts` row above. Both take that
1056                    // row here, so the fragment the trampoline delivers and the
1057                    // one the argument's own `Role::CallbackArg` site names are
1058                    // one fragment rather than two rows over one crossing.
1059                    if !self.field_crosses_as_its_fields(core)
1060                        && !self.unfolded().callback_arg_plans.contains_key(&arg.key())
1061                    {
1062                        continue;
1063                    }
1064                    let crossing = Crossing::new(arg.clone(), Direction::Deconstruct);
1065                    if recipes.key_of(&crossing.key(), &parts()).is_none() {
1066                        continue;
1067                    }
1068                    bound.bind(
1069                        Site::part(&row, index),
1070                        crossing,
1071                        Ask::Recipe(parts()),
1072                        Origin::Adapter,
1073                    );
1074                }
1075            }
1076        }
1077
1078        // Each value a callback delivers is a root site of its own — the
1079        // function-unique `Role::CallbackArg` the registry names, not the
1080        // `Role::Part` above, which is keyed by the callback recipe every
1081        // function with that signature shares.
1082        //
1083        // Bound to `parts` exactly where the argument's crossing states that
1084        // row, which is now every argument the trampoline takes apart: a
1085        // `data_class`, a `sealed_class`, an implicit Optional over one of
1086        // those, or a type whose deconstructor declaration earned the row
1087        // declared above. An argument that crosses whole names nothing here and
1088        // takes its crossing's default, attributed to the adapter.
1089        for f in model.functions() {
1090            for (param, p) in f.params.iter().enumerate() {
1091                let prebindgen_registry::flat::TypeKind::Callback { args } =
1092                    p.ty.unwrapped().kind()
1093                else {
1094                    continue;
1095                };
1096                for (arg, ty) in args.iter().enumerate() {
1097                    let crossing = Crossing::new(ty.clone(), Direction::Deconstruct);
1098                    if recipes.key_of(&crossing.key(), &parts()).is_none() {
1099                        continue;
1100                    }
1101                    bound.bind(
1102                        Site {
1103                            owner: f.name.clone(),
1104                            role: prebindgen_registry::recipe::Role::CallbackArg { param, arg },
1105                        },
1106                        crossing,
1107                        Ask::Recipe(parts()),
1108                        Origin::Adapter,
1109                    );
1110                }
1111            }
1112        }
1113
1114        // Which accessor parts are taken apart further, which is a binding
1115        // rather than a property of the row: a `data_class` states a `parts`
1116        // row and still crosses whole as an accessor's return. The rule is the
1117        // decomposition's own — splice where the part's type has a
1118        // deconstructor DECLARATION — stated here so the leaf view and the
1119        // compiler read one answer (#701 step 3, design decision 3).
1120        for decl in &self.return_expand_decls {
1121            let Some(owner) = registry.reading(decl.key()) else {
1122                continue;
1123            };
1124            let row = Crossing::new(owner, Direction::Deconstruct).row(parts());
1125            for (index, field) in decl.field_list().iter().enumerate() {
1126                let crate::jni::LocalField::Named(func, _) = field else {
1127                    continue;
1128                };
1129                let Some(ret) = model.function(func).map(|f| f.ret.clone()) else {
1130                    continue;
1131                };
1132                // The part crosses as the accessor's return states it, layers
1133                // included: an `Option<&Report>` part is an optional crossing,
1134                // and binding the inner row there would hand the compiler a
1135                // fragment of the wrong type. Only a part that is the value
1136                // itself is bound to its `parts` row; a layered one waits for
1137                // the layer to be a part of its own.
1138                if ret.optional_inner().is_some() {
1139                    continue;
1140                }
1141                // The bounded delivery peel, not `stripped_key`: that reaches
1142                // the leaf type through ANY nesting, so a part typed
1143                // `Vec<Option<T>>` would be read as the declaration's own `T`
1144                // and spliced — making the row reach its own crossing. The
1145                // inner optional is part of the element, and the part crosses
1146                // as the distinct `Vec<Option<T>>`.
1147                let (_, core) = ret.layer_stack();
1148                let core = core.borrow_target().unwrap_or(core);
1149                // Two conditions, and both are needed. A DECLARATION is what
1150                // says this part comes apart — a type may state a `parts` row
1151                // and still cross whole here, which is the difference this
1152                // binding carries. And the row has to exist to be named: a
1153                // declared type whose row this table does not state yet is
1154                // step 3's remaining work, not a binding to write now.
1155                let part = Crossing::new(core.clone(), Direction::Deconstruct);
1156                if !self
1157                    .return_expand_decls
1158                    .iter()
1159                    .any(|d| *d.key() == core.stripped_key())
1160                    || recipes.key_of(&part.key(), &parts()).is_none()
1161                {
1162                    continue;
1163                }
1164                bound.bind(
1165                    Site::arm_part(&row, None, index),
1166                    part,
1167                    Ask::Recipe(parts()),
1168                    Origin::Adapter,
1169                );
1170            }
1171        }
1172
1173        bound.build(recipes)
1174    }
