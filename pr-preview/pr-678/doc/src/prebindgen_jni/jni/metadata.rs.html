5/// How a `Shape::Optional` fold layer represents `None` over the JNI wire —
6/// the per-layer payload `N` of the JNI adapter's [`FoldStrategy`].
7///
8/// The choice is made at the point the `Option<_>` wrapper folds the layer onto
9/// a projection's `FoldStrategy`, and only depends on whether `option_output`
10/// rode the inner converter's niche (wire stayed identical to the inner's wire)
11/// or boxed the primitive into `java.lang.<Box>` (wire widened to `JObject`).
12/// The renderer reads this to pick the matching Kotlin shape — without it, a
13/// primitive-wired `Option<Handle>` would be declared as nullable `Long?` even
14/// though the wire is a non-nullable `jlong` whose `0L` *is* the null.
15#[derive(Clone, Debug, PartialEq, Eq)]
16pub enum NullableKind {
17    /// Wire kept the inner converter's encoding; `None` is the carved niche
18    /// slot's `value` (e.g. `0L` for a handle's `jlong`). On Kotlin the
19    /// declared wire is non-nullable; the wrapper body converts the sentinel
20    /// to `null` explicitly via an `if (it == <sentinel>) null else W(it)`
21    /// pattern.
22    Niche,
23    /// Wire widened to `JObject`; `None` is JVM `null`. On Kotlin the
24    /// declared wire is nullable and `?.let { W(it) }` works directly. This
25    /// is also the rendering object-shaped niches (`JByteArray::null` /
26    /// `JString::null`) collapse onto — Kotlin's `T?` already maps to JVM
27    /// reference-null at no extra cost.
28    Boxed,
29}
30
31/// The JNI adapter's nullability / collection layer stack over a projection
32/// leaf, on the unified [`Shape`](prebindgen_registry::shape::Shape)
33/// with [`NullableKind`] as the per-`Optional`-layer payload:
34///   * `Base` — the receiver *is* the handle;
35///   * `Optional(kind, inner)` — `T?`; `kind` records how null is represented
36///     over the wire (see [`NullableKind`]);
37///   * `Iterable(inner)` — `List<T>`. EXTENSION POINT: no `Vec<Handle>` shape
38///     exists today, so the emitters guard this arm loudly rather than silently
39///     mis-generating.
40pub type FoldStrategy = prebindgen_registry::shape::Shape<NullableKind>;
41
42/// Which flavor of Kotlin newtype a [`Projection`] surfaces. Both share the
43/// same "wire != declared Kotlin type, wrap as `W(wire)`, fold through
44/// `Option`/`Vec`" shape; they differ only in how a struct field stores them
45/// and whether they own a closeable resource.
46#[derive(Clone, Debug, PartialEq, Eq)]
47pub enum ProjectionKind {
48    /// Opaque native handle (`ptr_class`). Wire is `jlong`; a struct field
49    /// stores the **boxed** handle object (`L<fqn>;`); closeable when owned.
50    Handle,
51    /// Rust `u64`: raw JNI `jlong` bit pattern with a typed Kotlin `ULong`
52    /// surface. It owns no resource; wrapping/unwrapping is
53    /// `Long.toULong()` / `ULong.toLong()`.
54    Unsigned64,
55}
56
57/// Folded description of a Kotlin newtype projection (opaque handle or
58/// `ULong`) reached through zero or more wrapper layers. Set at the leaf,
59/// transformed by each wrapper as the type folds (see [`FoldStrategy`]), and
60/// read by every typed-surface emitter (data-class fields, struct
61/// encode/decode, `classify_return`, param classification) so "what Kotlin
62/// class does this surface, how do I wrap/fold it, do I close it" has one
63/// source of truth instead of a parallel ad-hoc decision tree.
64#[derive(Clone, Debug)]
65pub struct Projection {
66    /// Canonical key of the leaf type (e.g. `ZKeyExpr`, `ZenohId`); derive
67    /// the typed Kotlin FQN via `Declarations::kotlin_fqn` — a typed key, so the
68    /// lookup cannot drift from the declaration table's constructor.
69    pub leaf_key: prebindgen_registry::TypeKey,
70    /// `false` for `&T` borrows of a handle — still a projection (param
71    /// classification needs this), but not the holder's to close, so
72    /// `close()` emission skips it.
73    pub owned: bool,
74    /// Nullability / collection layers.
75    pub strategy: FoldStrategy,
76    /// Handle vs `ULong` — see [`ProjectionKind`].
77    pub kind: ProjectionKind,
78    /// Kotlin literals for representation-domain niches, in carve order.
79    /// Empty for ordinary projections; bounded u64 conversions populate it.
80    pub niche_sentinels: Vec<String>,
81}
82
83/// Per-converter language-specific extras carried by every converter this
84/// adapter produces. Filled by the same handler that builds the wire/body,
85/// carried in [`prebindgen_registry::ConverterImpl::metadata`], and read
86/// directly by the Kotlin emitter off the fragment that conversion belongs to
87/// — so cross-language facts flow through the existing wrapper machinery
88/// rather than a parallel side channel.
89#[derive(Clone, Debug, Default)]
90pub struct KotlinMeta {
91    /// Value-context Kotlin type, structured ([`KtType`]). `Long` for
92    /// opaque handles (jlong wire mention), the FQN class
93    /// (`io.zenoh.jni.JNIEncoding`) for user-declared decoder types whose
94    /// wire isn't primitive, a composed `List<ByteArray>` when a wrapper
95    /// wraps an inner. Leaves carry FQNs; the Kotlin renderer's `ImportSet`
96    /// shortens them at render time. `None` only for entries that must not
97    /// appear in any Kotlin signature — the emitter treats that as a hard
98    /// error.
99    pub kotlin_name: Option<KtType>,
100    /// For wrapper converters whose Kotlin projection is the *inner*
101    /// type's projection (e.g. `Result<Publisher, E>` → `Publisher`),
102    /// this carries the inner Flat reading so downstream emitters can find the
103    /// surfaced value's identity without spelling or reconstructing Rust.
104    ///
105    /// This is a classified Flat reading, not Rust syntax. Result planning
106    /// stores its success reading here; return-surface classification asks its
107    /// `TypeKind` and keys directly. `None` means the crossing itself is the
108    /// surfaced value.
109    pub value_reading: Option<prebindgen_registry::flat::TypeRef>,
110    /// Present iff this (possibly wrapped) value is an opaque native handle. Set
111    /// at the opaque-handle leaf and folded outward by the `&_` / `Option<_>`
112    /// wrappers and the `lookup_*` composed branches. The single source of truth
113    /// for typed-handle rendering and `close()` generation — see [`Projection`].
114    pub projection: Option<Projection>,
115    /// Target-language spellings for the currently exposed niche slots, in
116    /// the same carve order as `ConverterImpl::niches`. Optional composition
117    /// removes the spelling for every slot it consumes.
118    pub niche_sentinels: Vec<String>,
119}
120
121impl KotlinMeta {
122    /// True iff this (input-direction) converter decodes a directly-consumable
123    /// owned opaque handle — i.e. its projection is a bare `Handle` leaf with no
124    /// `Option`/`Vec` fold. Replaces the former `converter_returns_owned_object`
125    /// return-type AST sniff; the two are equivalent for every input converter
126    /// this adapter produces.
127    pub(crate) fn is_direct_handle(&self) -> bool {
128        self.projection.as_ref().is_some_and(|p| {
129            p.kind == ProjectionKind::Handle && matches!(p.strategy, FoldStrategy::Base)
130        })
131    }
