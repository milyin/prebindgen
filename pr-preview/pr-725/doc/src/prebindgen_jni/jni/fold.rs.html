12/// Peel the layers that never change whether a value's core is a Kotlin enum,
13/// **off the model**: a borrow and an optional, in any nesting. So `&Priority`,
14/// `Priority`, `Option<Priority>` and `Option<&Priority>` all probe as
15/// `Priority` — letting nullable enum params (`Option<enum>`) wire as the raw
16/// `Int` discriminant plus an allocated niche, instead of leaking the enum
17/// object to the Rust converter.
18///
19/// A **run is not peeled**. `Vec<Priority>` is a `List<Priority>`, not an enum,
20/// so this is deliberately not [`TypeRef::layer_stack`], which strips the
21/// sequence layer too.
22///
23/// Borrowing rather than composing is not a shortcut: every layer of a reading
24/// already holds the next as a `TypeRef` of its own, so there is nothing to
25/// mint — which is also why this needs no flat. What it returns spells
26/// itself (`spell()`) and classifies itself (`kind`), and the two cannot
27/// disagree.
28pub(crate) fn enum_probe(reading: &TypeRef) -> &TypeRef {
29    let mut cur = reading;
30    while let Some(inner) = cur.borrow_target().or_else(|| cur.optional_inner()) {
31        cur = inner;
32    }
33    cur
34}

36// The bottom-up layer fold is the shared `prebindgen_registry::shape::fold_shape`
37// (its `on_optional` receives the layer's `&NullableKind` + the wrapped
38// `&FoldStrategy`, so callers can special-case e.g. a `Niche` layer sitting
39// directly over the `Base` leaf). Used by the **type-name** folds
40// (`handle_kt_type` / `projection_wire_return`). The **expression** folds
41// (`render_handle_close` / `fold_projection_wrap`) are deliberately *not*
42// expressed through it: they fold the other direction (threading a `receiver` /
43// fresh lambda variable top-down rather than combining a bottom-up result), so
44// a shared combinator would obscure rather than simplify them.
45use prebindgen_registry::shape::fold_shape;

47/// The Kotlin type for a closeable handle reached through the folded
48/// [`FoldStrategy`] layers, given the leaf typed-handle type (e.g.
49/// `ZKeyExpr`): `Direct → ZKeyExpr`, `Nullable(inner) → <inner>?`,
50/// `Iterable(inner) → List<<inner>>`.
51pub(crate) fn handle_kt_type(strategy: &FoldStrategy, leaf: &KtType) -> KtType {
52    fold_shape(
53        strategy,
54        &|| leaf.clone(),
55        // The declared Kotlin projection type is `T?` regardless of how null
56        // is represented over the wire — the wrap fold and the wire-return
57        // helper read the kind to handle the wire shape separately.
58        &|inner, _kind, _inner_strategy| inner.nullable(),
59        &|inner| KtType::generic("List", [inner]),
60    )
61}

63/// Typed Kotlin leaf of a projection. Declared handle projections
64/// take their configured class FQN; the built-in `u64` projection is Kotlin's
65/// stable unsigned scalar type.
66pub(crate) fn projection_leaf_kt(ext: &Declarations, proj: &Projection) -> Option<KtType> {
67    match proj.kind {
68        ProjectionKind::Handle => ext.kotlin_fqn(&proj.leaf_key).map(KtType::cls),
69        ProjectionKind::Unsigned64 => Some(KtType::cls("ULong")),
70    }
71}

73/// Wrap one raw projection leaf into its typed Kotlin form.
74pub(crate) fn projection_wrap_expr(kind: &ProjectionKind, short: &str, raw: &str) -> String {

85/// Build the Kotlin `fromParts` factory for a data class, from the struct's
86/// **decomposition** — the one leaf list, which the Rust encode renders from
87/// and a builder interface declares. Not a mirror of a second walk: there is
88/// no second walk (#620).
89///
90/// Returns `(params, reconstruct, mints_handle)`:
91/// * `params` — the flattened `(name, kotlin_type)` list, one per leaf; nested
92///   data-class fields are inlined and `Option<nested>` prepends a
93///   `…__present: Boolean` flag. Order and types match the native call's JVM
94///   descriptor positionally, because both come from
95///   [`plan_leaf_params`](crate::jni::iface::plan_leaf_params).
96/// * `reconstruct` — the Kotlin expression building this struct:
97///   `Class(<part per constructor field>)`, where a nested field reconstructs
98///   via `Child.fromParts(<child param names>)` (`if (present) … else null` when
99///   optional) and a leaf reconstructs with its wrap.
100/// * `mints_handle` — whether any leaf is an opaque handle, i.e. whether this
101///   factory takes a raw native pointer and so needs the raw-pointer guard.
102#[allow(clippy::too_many_arguments)]
103pub(crate) fn flatten_struct_factory(
104    ext: &Declarations,
105    flat: &prebindgen_registry::flat::Flat,
106    s: &prebindgen_registry::flat::Struct,
107    prefix: &str,
108    class_name: &str,
109    imports: &mut BTreeSet<String>,
110    depth: usize,
111) -> Option<StructFactory> {
112    let _ = (prefix, depth);
113    let leaves = ext.struct_out_frozen(flat, s)?;
114    let (params, reconstruct) = factory_from_leaves(ext, flat, &leaves, class_name, imports)?;
115    // Whether this factory takes a raw native pointer, and so needs the
116    // raw-pointer guard: asked of the same leaves, through the wrap that says
117    // a slot is a handle.
118    let mints_handle = crate::jni::iface::plan_leaf_params(ext, &leaves)?
119        .iter()
120        .any(|param| param.wrap.class_fqn().is_some());
121    Some((params, reconstruct, mints_handle))
122}

124/// The Kotlin `fromParts` side, derived from the struct's **decomposition** —
125/// the same leaf list the Rust encode renders from and the same one a builder
126/// interface declares.
127///
128/// One derivation, so the factory's parameters cannot drift from the slots the
129/// encoder fills. What each part of the expression needs, the leaves already
130/// say: `reach` which field a leaf came from, `through` which class a boundary
131/// reassembles through, `groups` and the selectors which arm gates it.
132///
133/// Returns `(params, reconstruct)`, where `reconstruct` is
134/// `Class(<one part per constructor field>)`.
135fn factory_from_leaves(
136    ext: &Declarations,
137    flat: &prebindgen_registry::flat::Flat,
138    leaves: &[crate::jni::compile::OutWire],
139    class_name: &str,
140    imports: &mut BTreeSet<String>,
141) -> Option<(Vec<(String, KtType)>, String)> {
142    let params = crate::jni::iface::plan_leaf_params(ext, leaves)?;
143    let names = crate::jni::render::plan_leaf_names(leaves);
144    let declared: Vec<(String, KtType)> = names
145        .iter()
146        .zip(&params)
147        .map(|(name, param)| (name.clone(), param.raw.clone()))
148        .collect();
149
150    // One part per constructor field, and a field is what a leaf's reach
151    // starts with — a nested class's leaves, a sum's tag and groups, and an
152    // optional's flag all reach through the one field that carries them.
153    let mut parts: Vec<String> = Vec::new();
154    let mut i = 0usize;
155    while i < leaves.len() {
156        let field = |j: usize| {
157            leaves[j]
158                .reach()
159                .first()
160                .map(|step| step.ident().to_string())
161        };
162        let head = field(i);
163        let end = (i..leaves.len())
164            .take_while(|&j| field(j) == head)
165            .last()
166            .map_or(i + 1, |j| j + 1);
167        parts.push(factory_part(
168            ext,
169            flat,
170            &leaves[i..end],
171            &params[i..end],
172            &names[i..end],
173            None,
174            imports,
175        )?);
176        i = end;
177    }
178    Some((declared, format!("{class_name}({})", parts.join(", "))))
179}

181/// The expression rebuilding ONE constructor field from the leaves that carry
182/// it: a plain leaf through its own wrap, a nested class through its
183/// `fromParts`, a sum through the `when` over its tag, and any of those behind
184/// a presence flag when the field is optional.
185#[allow(clippy::too_many_arguments)]
186fn factory_part(
187    ext: &Declarations,
188    flat: &prebindgen_registry::flat::Flat,
189    leaves: &[crate::jni::compile::OutWire],
190    params: &[crate::jni::IfaceParam],
191    names: &[String],
192    // `ungated`: the same leaves as the value's OWN class declares them, when a
193    // presence flag above made the parent's copies nullable. `None` when
194    // nothing gates them, in which case the two are the same.
195    ungated: Option<&[crate::jni::IfaceParam]>,
196    imports: &mut BTreeSet<String>,
197) -> Option<String> {
198    // The class boundary is asked FIRST. A nested child may itself begin with
199    // a selector — an `Option<Mid>` whose `Mid` starts with an optional field,
200    // or with a sum — and that selector belongs to the CHILD's signature, not
201    // to this one. Reading `from` before `through` consumed it here and called
202    // the child factory without it, which is a `fromParts` arity mismatch the
203    // JVM only reports at the call (#620 review).
204    // A nested class: its leaves are the parent's, put back through the class
205    // the decomposition recorded when it flattened them. A gated one receives
206    // wire defaults when absent, so its object slots are nullable in the
207    // parent's signature and re-asserted inside the guard the flag opened.
208    if let Some(fqn) = leaves[0].through.first() {
209        let short = register_fqn(fqn, imports);
210        // Re-asserted only where the NULL came from the gate, never where the
211        // value is legitimately absent: a leaf whose own type is optional is
212        // nullable on its own account, and `!!` there would throw on a value
213        // the class is meant to hold. The same distinction `sum_ctor_arg`
214        // makes for an inert group's payload.
215        let forwarded: Vec<String> = names
216            .iter()
217            .enumerate()
218            .map(|(i, name)| {
219                // Re-asserted only where the null is the GATE's doing: the
220                // parent declares the slot nullable so the encoder can default
221                // it when absent, while the class itself declares it non-null.
222                // A slot the class already declares nullable — its own type is
223                // optional, or it belongs to a sum group that is inert whenever
224                // another alternative is live — is forwarded untouched, because
225                // `!!` there would throw on a value the class is meant to hold.
226                let gate_added = params[i].raw.is_nullable()
227                    && !ungated.is_none_or(|own| own[i].raw.is_nullable());
228                match gate_added {
229                    true => format!("{name}!!"),
230                    false => name.clone(),
231                }
232            })
233            .collect();
234        return Some(format!("{short}.fromParts({})", forwarded.join(", ")));
235    }
236    // A presence flag gates everything after it, and contributes no value of
237    // its own: the field is what its group rebuilds, or null.
238    if matches!(leaves[0].from, crate::jni::compile::OutFrom::Present) {
239        // Inside the guard the gate's own arm is no longer in question, so the
240        // group it opened is stripped: what is left is the value as its own
241        // class declares it, which is what says whether a null here is this
242        // gate's doing or the value's own.
243        let inner: Vec<crate::jni::compile::OutWire> = leaves[1..]
244            .iter()
245            .map(|leaf| {
246                let mut leaf = leaf.clone();
247                leaf.groups = leaf
248                    .groups
249                    .split_first()
250                    .map_or_else(Vec::new, |(_, rest)| rest.to_vec());
251                leaf
252            })
253            .collect();
254        let inner_params = crate::jni::iface::plan_leaf_params(ext, &inner)?;
255        let gated = factory_part(
256            ext,
257            flat,
258            &leaves[1..],
259            &params[1..],
260            &names[1..],
261            Some(&inner_params),
262            imports,
263        )?;
264        return Some(format!("if ({}) {gated} else null", names[0]));
265    }
266    if leaves[0].is_tag() {
267        let (_, when) = ext.sum_reconstruct(
268            flat,
269            &leaves[0].out_ty.unwrapped().key(),
270            leaves,
271            params,
272            names,
273            imports,
274        );
275        return Some(when);
276    }
277    debug_assert_eq!(
278        leaves.len(),
279        1,
280        "an unselected field with no class path is one leaf"
281    );
282    Some(ext.sum_ctor_arg(&leaves[0], &params[0], &names[0], imports))
283}

285/// Render the Kotlin `close()` expression for a handle `receiver` through
286/// the folded [`FoldStrategy`] layers. Fresh lambda variable per nesting
287/// level avoids `it` shadowing; the common single-layer cases are
288/// special-cased for readable output (`x?.close()`, `x.forEach { it.close() }`).
289pub(crate) fn render_handle_close(strategy: &crate::jni::FoldStrategy, receiver: &str) -> String {
290    use prebindgen_registry::shape::Shape::*;
291    fn go(strategy: &crate::jni::FoldStrategy, receiver: &str, depth: usize) -> String {
292        match strategy {
293            Base => format!("{receiver}.close()"),
294            // The Kotlin-side receiver is already nullable (`handle_kt_type`
295            // emits `T?` for both niche and boxed kinds), so `?.close()` covers
296            // both wire representations.
297            Optional(_, inner) => match &**inner {
298                Base => format!("{receiver}?.close()"),
299                _ => {
300                    let v = format!("e{depth}");
301                    format!("{receiver}?.let {{ {v} -> {} }}", go(inner, &v, depth + 1))
302                }
303            },
304            Iterable(inner) => {
305                let v = format!("e{depth}");
306                format!(
307                    "{receiver}.forEach {{ {v} -> {} }}",
308                    go(inner, &v, depth + 1)
309                )
310            }
311        }
312    }
313    go(strategy, receiver, 0)
314}

316/// Fold the projection wrap call `W(receiver)` through the
317/// [`FoldStrategy`] layers:
318/// * `Direct`         → `W(x)`
319/// * `Nullable{Boxed}` → `x?.let { W(it) }` (JVM-null at the wire)
320/// * `Nullable{Niche}` over a primitive wire (e.g. `jlong`) →
321///   `x.let { if (it == <sentinel>) null else W(it) }`
322/// * `Nullable{Niche}` over an object wire (e.g. `JByteArray`) →
323///   `x?.let { W(it) }` (the wire is already a nullable reference)
324/// * `Iterable`       → `x.map { W(it) }`
325///
326/// `niche_sentinel` is the Kotlin literal to compare against for the
327/// `Niche+primitive` arm (e.g. `"0L"` for `jlong`-wired handles). When the
328/// wire is object-shaped the sentinel is unused — `null` is the wire-level
329/// representation and `?.let` is a no-cost null check.
330pub(crate) fn fold_projection_wrap(
331    strategy: &crate::jni::FoldStrategy,
332    receiver: &str,
333    kind: &crate::jni::ProjectionKind,
334    wrap_class: &str,
335    niche_sentinel: Option<&str>,
336) -> String {
337    use prebindgen_registry::shape::Shape::*;
338
339    use crate::jni::NullableKind;
340    fn go(
341        s: &crate::jni::FoldStrategy,
342        r: &str,
343        kind: &crate::jni::ProjectionKind,
344        w: &str,
345        sentinel: Option<&str>,
346        depth: usize,
347    ) -> String {
348        match s {
349            Base => projection_wrap_expr(kind, w, r),
350            Optional(nullable_kind, inner) => match (nullable_kind, &**inner) {
351                // Primitive-wired niche → can't carry null on the wire, so
352                // compare against the sentinel and synthesize null on the
353                // Kotlin side.
354                (NullableKind::Niche, Base) if sentinel.is_some() => {
355                    let s = sentinel.unwrap();
356                    let wrapped = projection_wrap_expr(kind, w, "it");
357                    format!("{r}.let {{ if (it == {s}) null else {wrapped} }}")
358                }
359                // Object-wired niche or fully boxed Nullable → `?.let { W(it) }`.
360                (_, Base) => {
361                    let wrapped = projection_wrap_expr(kind, w, "it");
362                    format!("{r}?.let {{ {wrapped} }}")
363                }
364                // Deeper nesting. The niche/boxed distinction is only
365                // observable at the outermost layer covering a `Direct`
366                // leaf; intermediate layers (nullable-of-iterable etc.)
367                // can keep the simple form because Kotlin's `?.` chain
368                // already represents the layered null.
369                _ => {
370                    let v = format!("e{depth}");
371                    format!(
372                        "{r}?.let {{ {v} -> {} }}",
373                        go(inner, &v, kind, w, sentinel, depth + 1)
374                    )
375                }
376            },
377            Iterable(inner) => match &**inner {
378                Base => {
379                    let wrapped = projection_wrap_expr(kind, w, "it");
380                    format!("{r}.map {{ {wrapped} }}")
381                }
382                _ => {
383                    let v = format!("e{depth}");
384                    format!(
385                        "{r}.map {{ {v} -> {} }}",
386                        go(inner, &v, kind, w, sentinel, depth + 1)
387                    )
388                }
389            },
390        }
391    }
392    go(strategy, receiver, kind, wrap_class, niche_sentinel, 0)
393}

395/// JNI extern's declared Kotlin wire-return for a projection. The leaf wire
396/// is the inner converter's destination Kotlin name — `Long` for both
397/// projection kinds (a handle's pointer, a `ULong`'s raw bit pattern). The
398/// fold honours
399/// [`NullableKind`] so the declared wire matches the runtime ABI:
400/// `Niche+primitive` keeps the layer non-nullable on the wire (the sentinel
401/// represents null); `Niche+object` and `Boxed` add `?`.
402pub(crate) fn projection_wire_return(proj: &crate::jni::Projection) -> KtType {
403    use crate::jni::{FoldStrategy, NullableKind, ProjectionKind};
404    let (inner_wire, inner_is_primitive) = match proj.kind {
405        ProjectionKind::Handle => (KtType::long(), true),
406        ProjectionKind::Unsigned64 => (KtType::long(), true),
407    };
408    fold_shape(
409        &proj.strategy,
410        &|| inner_wire.clone(),
411        &|inner, kind, inner_strategy| {
412            // A niche layer over a primitive wire keeps the wire non-nullable —
413            // the sentinel value is the null representation. Object-wired niches
414            // and full-boxed Nullables both add `?` (JVM null on the reference).
415            match (kind, inner_strategy) {
416                (NullableKind::Niche, FoldStrategy::Base) if inner_is_primitive => inner,
417                _ => inner.nullable(),
418            }
419        },
420        &|inner| KtType::generic("List", [inner]),
421    )
422}

424/// Kotlin null-sentinel literal for the *leaf wire* of a projection. Read
425/// at the wrapper-body call site and forwarded to [`fold_projection_wrap`];
426/// `None` when the leaf wire has no primitive null sentinel, where
427/// `?.let { }` covers the JVM-null case directly.
428pub(crate) fn projection_leaf_sentinel(proj: &crate::jni::Projection) -> Option<String> {
429    if let Some(sentinel) = proj.niche_sentinels.first() {
430        return Some(sentinel.clone());
431    }
432    use crate::jni::ProjectionKind;
433    let leaf_wire: syn::Type = match proj.kind {
434        ProjectionKind::Handle => syn::parse_quote!(jni::sys::jlong),
435        // No niche exists for `u64`; `Option<u64>` uses the boxed path, so a
436        // primitive sentinel must never be synthesized.
437        ProjectionKind::Unsigned64 => return None,
438    };
439    kotlin_null_sentinel(&leaf_wire).map(|s| s.to_string())
440}

442/// The sentinel a **wrap** should test, given the projection and whether an
443/// ancestor makes the leaf nullable — the one rule both derivations of that
444/// wrap read (`unfold_leaf_kt` in `render.rs`, `leaf_iface_param` in
445/// `iface.rs`), so it cannot drift between them again (#142).
446///
447/// A sentinel is the leaf's **own** `None` representation, so it belongs to the
448/// leaf's own type and to nothing above it. Two independent facts meet here and
449/// only the first one answers:
450///
451/// * the leaf's type carries a niche — `Option<Duration>` over a bounded
452///   `convert!`, whose strategy is `Optional(Niche, _)`. Its `None` IS the
453///   sentinel, and the test stays whatever the ancestor does.
454/// * an **ancestor** can be absent (a conditional value form, an
455///   `Option<sum>`/`Option<nested>` field). That widens the wire — the Rust
456///   side boxes any nullable leaf, see
457///   [`leaf_is_prim`](crate::jni::emit::leaf_is_prim) — and `?.` alone carries
458///   it. It grants no sentinel.
459///
460/// [`projection_leaf_sentinel`] answers off `niche_sentinels`, which
461/// `attach_domain_sentinels` puts on the **bare** type's converter as well as
462/// the `Option` one. Asking it without the `Base` check therefore handed a
463/// sentinel to a leaf that has no niche encoding at all, splicing
464/// `?.let { if (it == -1L) … }` into a wrap whose own encoder can never emit
465/// `-1`. Harmless at runtime — the value is outside the declared range — and
466/// wrong on its face.
467pub(crate) fn wrap_sentinel(proj: &crate::jni::Projection, nullable: bool) -> Option<String> {
468    // `Base` means the leaf has no `Option` of its own: whatever absence it can
469    // express is the ancestor's, and `?.` already expresses that.
470    if nullable && matches!(proj.strategy, crate::jni::FoldStrategy::Base) {
471        return None;
472    }
473    projection_leaf_sentinel(proj)
474}

476/// Kotlin literal for the null-sentinel of a primitive wire — used by
477/// [`fold_projection_wrap`] when a `Niche` layer covers a primitive wire and
478/// can't carry JVM null. Mirrors `jni_field_access`'s primitive descriptors.
479/// Returns `None` for object-shaped wires (where JVM null *is* the null
480/// representation and `?.let` is the right pattern).
481pub(crate) fn kotlin_null_sentinel(wire: &syn::Type) -> Option<&'static str> {
482    let (_, _, is_object) = crate::jni::wire_access::jni_field_access(wire)?;
483    if is_object {
484        return None;
485    }
486    let syn::Type::Path(tp) = wire else {
487        return None;
488    };
489    let last = tp.path.segments.last()?;
490    Some(match last.ident.to_string().as_str() {
491        "jlong" => "0L",
492        "jint" | "jshort" | "jbyte" | "jchar" => "0",
493        "jfloat" => "0.0f",
494        "jdouble" => "0.0",
495        "jboolean" => "false",
496        _ => return None,
497    })
498}

500/// Shorten a class FQN to its simple name for use in **raw body text** (a
501/// `fromParts` reconstruct fragment: `Child.fromParts(…)`, `Enum.fromInt(…)`,
502/// `ZenohId(bytes)`), registering the FQN into `used` — the body `Code`'s own
503/// import list. A non-dotted name (a Kotlin builtin like `String`) needs no
504/// import and passes through. Signature types are NOT shortened this way —
505/// those are full-FQN `KtType`s in the AST, shortened + imported by the
506/// render-time `ImportSet`.
507pub(crate) fn register_fqn(fqn: &str, used: &mut BTreeSet<String>) -> String {
508    if fqn.contains('.') {
509        used.insert(fqn.to_string());
510        fqn.rsplit('.').next().unwrap_or(fqn).to_string()
511    } else {
512        fqn.to_string()
513    }
514}
