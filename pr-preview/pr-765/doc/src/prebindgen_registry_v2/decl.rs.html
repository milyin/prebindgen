1//! What a binding declared, in terms neither language owns.
2//!
3//! A frontend turns its own declaration storage into these — one [`Origin`]
4//! per thing the user asked for, inside the
5//! [`BindingRequests`](crate::BindingRequests) it hands the engine. The origin
6//! is what the report accounts for; the request carries the target's
7//! configuration for it, and that policy is also where the foreign name and
8//! the declarator word the report prints come from.
9
10use prebindgen_flat::flat::{Element, Flat, TypeKey};
11use serde::Serialize;
12
13/// What a declaration is named after, and what the engine plans it from.
14///
15/// A name alone says neither: `Sample` may be a captured type, a type key the
16/// binding coined for something the source never exported, or the name of a
17/// Kotlin constant built from an expression. The variant says which, and it
18/// says it once — what the target gets ([`Self::kind`]) and what the name must
19/// find in the captured source both follow from the variant instead of being
20/// stated beside it, so they cannot disagree and a pair that means nothing (a
21/// callback backed by a captured constant, a function the binding both defines
22/// and selects out of the source) cannot be written down.
23///
24/// [`generate`](crate::generate) routes on this: each planner is reached by its
25/// own variants and is handed the captured item they name, rather than a kind
26/// to decode and a name to look up.
27///
28/// It is also the declaration's identity — what an outcome is keyed by, what a
29/// [`Position`](crate::target::Position) is rooted at, what a target's
30/// [`Requirement`](crate::target::Requirement) resolves to. Stable across runs
31/// and across pipelines, so a report, a build script and a capability-selected
32/// test section can all name the same declaration: it is what the *source*
33/// calls the thing, not what the target does, and a rename on the foreign side
34/// must not silently retire a test's requirement. `<kind>:<name>` —
35/// `type:Stamp`, `fn:stamp_sum` — is how it prints and how the report writes
36/// it, and that spelling is a rendering: nothing reads an origin back out of
37/// it. Origins order by kind — types first, then conversions, callbacks,
38/// constants and functions, which is how a report reads — and then by name.
39#[derive(Clone, Debug, PartialEq, Eq, Hash)]
40pub enum Origin {
41    /// A captured `#[prebindgen]` function, exported as a foreign function.
42    Function(syn::Ident),
43    /// A foreign function the binding defines itself — `fun!(crate::x).sig(..)`
44    /// in the JNI frontend. Nothing in the captured source names it.
45    LocalFunction(String),
46    /// A captured `#[prebindgen]` constant, exposed as a foreign constant.
47    Const(syn::Ident),
48    /// A foreign constant read by calling a captured nullary function —
49    /// `constant!(X).fun(fun!(f))`. The target renders a constant; the engine
50    /// plans the function behind it.
51    ConstFromFunction(syn::Ident),
52    /// A foreign constant the binding computes rather than reads —
53    /// `constant!(X).expr(..)`. No captured item holds its value.
54    LocalConst(String),
55    /// A captured `#[prebindgen]` type, given a foreign representation.
56    Type(TypeKey),
57    /// A type the binding gives a representation although the source never
58    /// exported it — `String` crossing as an opaque handle.
59    LocalType(TypeKey),
60    /// A callback signature the binding exports as a foreign callable. No
61    /// captured item names one, so the signature is the name it goes by.
62    Callback(String),
63    /// A declared conversion between a Rust type and its wire form, defined by
64    /// the binding.
65    Conversion(TypeKey),
66}
67
68impl Origin {
69    /// What the target gets: `fn`, `type`, `const`, `callback` or
70    /// `conversion` — the prefix the origin prints with, the report's `kind`
71    /// column, and the middle of an `unsupported.<kind>.…` code.
72    ///
73    /// The coarse, language-neutral category — the five things any binding can
74    /// be made of. Deliberately coarser than an adapter's own vocabulary:
75    /// `prebindgen-c` declares a type with `opaque_ptr` or `data_struct`, and
76    /// `prebindgen-jni` with `ptr_class` or `data_class`, but all four are a
77    /// `type`; the adapter's word is what [`Target::describe`] prints back.
78    /// And it says what the *target* gets, not what the captured Rust source
79    /// held: a Kotlin constant read through a captured function is a `const`.
80    /// It is what keeps the origins' several naming spaces apart — `type:Foo`
81    /// and `conversion:Foo` are two declarations about one Rust type.
82    ///
83    /// [`Target::describe`]: crate::target::Target::describe
84    pub fn kind(&self) -> &'static str {
85        match self {
86            Origin::Function(_) | Origin::LocalFunction(_) => "fn",
87            Origin::Const(_) | Origin::ConstFromFunction(_) | Origin::LocalConst(_) => "const",
88            Origin::Type(_) | Origin::LocalType(_) => "type",
89            Origin::Callback(_) => "callback",
90            Origin::Conversion(_) => "conversion",
91        }
92    }
93
94    /// Whether this declares a type — captured or the binding's own.
95    pub fn is_type(&self) -> bool {
96        matches!(self, Origin::Type(_) | Origin::LocalType(_))
97    }
98
99    /// The captured element this origin names, if the model holds it.
100    ///
101    /// Captured items live in one flat namespace holding functions, types and
102    /// constants, so the name alone finds any element; the variant is what says
103    /// whether the element found is the one the declaration meant. A constant
104    /// read through a function is looked up among the functions, which is what
105    /// separates [`Self::ConstFromFunction`] from [`Self::Const`]. A
106    /// binding-local origin names no captured item and so finds none — which
107    /// is not the same as a missing one, and [`Self::missing_from`] is the
108    /// question to ask about presence.
109    pub(crate) fn captured<'f>(&self, flat: &'f Flat) -> Option<&'f Element> {
110        let element = flat.element(&self.name())?;
111        matches!(
112            (self, element),
113            (
114                Origin::Function(_) | Origin::ConstFromFunction(_),
115                Element::Function(_)
116            ) | (Origin::Type(_), Element::Type(_))
117                | (Origin::Const(_), Element::Constant(_))
118        )
119        .then_some(element)
120    }
121
122    /// Whether the binding defines this itself: a callback signature, a
123    /// binding-local conversion helper, a function or constant of its own, or a
124    /// type the target represents although the source never exported it
125    /// (`String` as an opaque handle). Such an origin requires nothing of the
126    /// model.
127    pub fn is_binding_local(&self) -> bool {
128        matches!(
129            self,
130            Origin::LocalFunction(_)
131                | Origin::LocalConst(_)
132                | Origin::LocalType(_)
133                | Origin::Callback(_)
134                | Origin::Conversion(_)
135        )
136    }
137
138    /// Whether the model lacks what this origin must name.
139    ///
140    /// Naming the wrong kind — `.fun(fun!(x))` where the source captured
141    /// `const x` — counts as missing: it is an error in the binding, and the
142    /// engine fails the run over it instead of reporting a skip. False for a
143    /// binding-local origin.
144    pub(crate) fn missing_from(&self, flat: &Flat) -> bool {
145        !self.is_binding_local() && self.captured(flat).is_none()
146    }
147
148    /// The word a refusal uses for what this origin must name.
149    pub(crate) fn describe_captured(&self) -> &'static str {
150        match self {
151            Origin::Function(_) | Origin::ConstFromFunction(_) => "function",
152            Origin::Const(_) => "constant",
153            Origin::Type(_) => "type",
154            _ => "binding-local item",
155        }
156    }
157
158    /// The name it goes by — the part of the printed form after the kind.
159    pub fn name(&self) -> String {
160        match self {
161            Origin::Function(ident) | Origin::Const(ident) | Origin::ConstFromFunction(ident) => {
162                ident.to_string()
163            }
164            Origin::Type(key) | Origin::LocalType(key) | Origin::Conversion(key) => {
165                key.as_str().to_string()
166            }
167            Origin::LocalFunction(name) | Origin::LocalConst(name) | Origin::Callback(name) => {
168                name.clone()
169            }
170        }
171    }
172}

174impl Ord for Origin {
175    /// By kind in report order, then by name. A captured and a binding-local
176    /// origin of one kind and name print alike and are still distinct, so the
177    /// variant breaks that tie last, and `Ord` agrees with `Eq`.
178    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
179        // Kind first, variant second: the variants of one kind are adjacent.
180        fn rank(origin: &Origin) -> (u8, u8) {
181            match origin {
182                Origin::Type(_) => (0, 0),
183                Origin::LocalType(_) => (0, 1),
184                Origin::Conversion(_) => (1, 0),
185                Origin::Callback(_) => (2, 0),
186                Origin::Const(_) => (3, 0),
187                Origin::ConstFromFunction(_) => (3, 1),
188                Origin::LocalConst(_) => (3, 2),
189                Origin::Function(_) => (4, 0),
190                Origin::LocalFunction(_) => (4, 1),
191            }
192        }
193        let (kind, variant) = rank(self);
194        let (other_kind, other_variant) = rank(other);
195        kind.cmp(&other_kind)
196            .then_with(|| self.name().cmp(&other.name()))
197            .then_with(|| variant.cmp(&other_variant))
198    }
199}
200
201impl PartialOrd for Origin {
202    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
203        Some(self.cmp(other))
204    }
205}
206
207impl std::fmt::Display for Origin {
208    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
209        write!(f, "{}:{}", self.kind(), self.name())
210    }
211}

213impl Serialize for Origin {
214    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
215        serializer.collect_str(self)
216    }
