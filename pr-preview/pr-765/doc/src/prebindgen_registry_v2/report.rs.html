1//! The report: what v2 generated, what it skipped, and why.
2//!
3//! Two renderings of one value — JSON for tooling (the CI smoke check reads it
4//! today; selecting test sections from it is planned) and Markdown for a
5//! person. Both are deterministic: declarations sort by kind
6//! then id, and skip causes are grouped by code so a single missing capability
7//! is stated once with the list of roots it took down, rather than repeated
8//! forty times.
9
10use std::{
11    collections::BTreeMap,
12    path::{Path, PathBuf},
13};

15use serde::Serialize;

17use crate::{
18    decl::Origin,
19    outcome::{EngineError, Outcome},
20    target::Described,
21};

23/// The report's own version. A consumer that reads the JSON checks this
24/// before trusting the shape; it changes whenever a field's meaning does.
25pub const SCHEMA_VERSION: u32 = 2;
26
27/// One accounted-for declaration.
28#[derive(Clone, Debug)]
29pub struct Entry {
30    /// What was declared, and the run's key for it.
31    pub origin: Origin,
32    /// What the target says it is on the foreign side — the adapter's
33    /// declarator word and the placement.
34    pub described: Described,
35    pub outcome: Outcome,
36}
37
38impl Entry {
39    /// Which kind of declaration it is — see [`Origin::kind`].
40    pub fn kind(&self) -> &'static str {
41        self.origin.kind()
42    }
43
44    /// The adapter's declarator word — see [`Described::representation`].
45    pub fn representation(&self) -> &str {
46        &self.described.representation
47    }
48
49    /// The foreign placement — see [`Described::placement`].
50    pub fn placement(&self) -> &str {
51        &self.described.placement
52    }
53}
54
55impl Serialize for Entry {
56    /// Flat, and with the identity spelled out: the report carries `id` (the
57    /// origin as it prints), `kind` and `rust_origin` as separate columns, and
58    /// a reader of the JSON should not have to split the id to get at the last
59    /// two. The outcome's own fields follow at the same level.
60    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
61        #[derive(Serialize)]
62        struct Columns<'a> {
63            id: &'a Origin,
64            kind: &'static str,
65            rust_origin: String,
66            placement: &'a str,
67            representation: &'a str,
68            #[serde(flatten)]
69            outcome: &'a Outcome,
70        }
71        Columns {
72            id: &self.origin,
73            kind: self.origin.kind(),
74            rust_origin: self.origin.name(),
75            placement: self.placement(),
76            representation: self.representation(),
77            outcome: &self.outcome,
78        }
79        .serialize(serializer)
80    }
81}

83/// What a run generated, and what it did not.
84#[derive(Clone, Debug, Serialize)]
85pub struct Report {
86    /// Which engine produced this.
87    pub pipeline: &'static str,
88    /// Which target it generated for — `"c"`, `"jni"`.
89    pub target: &'static str,
90    /// See [`SCHEMA_VERSION`].
91    pub schema_version: u32,
92    /// Enough of the run's input to tell a stale report from a fresh one: the
93    /// declaring crate and the source modules it read.
94    pub source_identity: SourceIdentity,
95    /// Every declaration, sorted by kind then id.
96    pub declarations: Vec<Entry>,
97}
98
99/// What this report was generated from.
100#[derive(Clone, Debug, Default, Serialize)]
101pub struct SourceIdentity {
102    /// The crate whose build script declared the binding.
103    pub declaring_crate: String,
104    /// The source modules the model was built from — crate or module names,
105    /// as the captured items' locations name them, not paths.
106    pub sources: Vec<String>,
107    /// How many `#[prebindgen]` items those modules held.
108    pub captured_items: usize,
109}
110
111impl Report {
112    /// How many declarations ended each way.
113    pub fn counts(&self) -> Counts {
114        let mut counts = Counts::default();
115        for entry in &self.declarations {
116            match entry.outcome {
117                Outcome::Emitted => counts.emitted += 1,
118                Outcome::Skipped(_) => counts.skipped += 1,
119                Outcome::Ignored => counts.ignored += 1,
120            }
121        }
122        counts
123    }
124
125    /// The skipped declarations grouped by capability code, each group's roots
126    /// sorted. One cause, all of its casualties — which is what a reader needs
127    /// to decide what to implement next.
128    pub fn skips_by_capability(&self) -> BTreeMap<&str, Vec<&Entry>> {
129        let mut groups: BTreeMap<&str, Vec<&Entry>> = BTreeMap::new();
130        for entry in &self.declarations {
131            if let Some(skip) = entry.outcome.skip() {
132                groups
133                    .entry(skip.capability.as_str())
134                    .or_default()
135                    .push(entry);
136            }
137        }
138        groups
139    }
140
141    /// The one-line summary a build script prints.
142    pub fn summary(&self) -> String {
143        let counts = self.counts();
144        format!(
145            "{}: {} emitted, {} skipped, {} ignored ({} target)",
146            self.pipeline, counts.emitted, counts.skipped, counts.ignored, self.target
147        )
148    }
149
150    /// The report as JSON.
151    pub fn to_json(&self) -> String {
152        serde_json::to_string_pretty(self).expect("a report is plain data and always serializes")
153    }
154
155    /// The report as Markdown.
156    pub fn to_markdown(&self) -> String {
157        use std::fmt::Write as _;
158        let mut out = String::new();
159        let counts = self.counts();
160        let _ = writeln!(
161            out,
162            "# prebindgen {} report — {}\n",
163            self.pipeline, self.target
164        );
165        let _ = writeln!(
166            out,
167            "Declared by `{}` over {} captured item(s) from: {}\n",
168            self.source_identity.declaring_crate,
169            self.source_identity.captured_items,
170            if self.source_identity.sources.is_empty() {
171                "(no source directory)".to_string()
172            } else {
173                self.source_identity.sources.join(", ")
174            }
175        );
176        let _ = writeln!(
177            out,
178            "| emitted | skipped | ignored |\n| ---: | ---: | ---: |\n| {} | {} | {} |\n",
179            counts.emitted, counts.skipped, counts.ignored
180        );
181
182        let _ = writeln!(out, "## Skipped, by cause\n");
183        let groups = self.skips_by_capability();
184        if groups.is_empty() {
185            let _ = writeln!(out, "Nothing skipped.\n");
186        }
187        for (capability, entries) in &groups {
188            let explanation = entries
189                .first()
190                .and_then(|entry| entry.outcome.skip())
191                .map(|skip| skip.explanation.as_str())
192                .unwrap_or_default();
193            let _ = writeln!(out, "### `{capability}` — {explanation}\n");
194            for entry in entries {
195                let path = entry
196                    .outcome
197                    .skip()
198                    .map(|skip| skip.path())
199                    .unwrap_or_default();
200                let _ = writeln!(out, "- `{}` ({})", entry.origin, path);
201            }
202            let _ = writeln!(out);
203        }
204
205        let _ = writeln!(out, "## Every declaration\n");
206        let _ = writeln!(
207            out,
208            "| declaration | representation | placement | outcome |"
209        );
210        let _ = writeln!(out, "| --- | --- | --- | --- |");
211        for entry in &self.declarations {
212            let outcome = match &entry.outcome {
213                Outcome::Emitted => "emitted".to_string(),
214                Outcome::Ignored => "ignored".to_string(),
215                Outcome::Skipped(skip) => format!("skipped: `{}`", skip.capability),
216            };
217            let _ = writeln!(
218                out,
219                "| `{}` | {} | `{}` | {} |",
220                entry.origin,
221                entry.representation(),
222                entry.placement(),
223                outcome
224            );
225        }
226        out
227    }
228
229    /// Write both renderings into `dir`, which is created if missing.
230    ///
231    /// Returns the paths written, in the order they were written.
232    pub fn write(&self, dir: impl AsRef<Path>) -> Result<Vec<PathBuf>, EngineError> {
233        let dir = dir.as_ref();
234        std::fs::create_dir_all(dir)?;
235        let json = dir.join(format!("{}-report.json", self.target));
236        let markdown = dir.join(format!("{}-report.md", self.target));
237        std::fs::write(&json, self.to_json())?;
238        std::fs::write(&markdown, self.to_markdown())?;
239        Ok(vec![json, markdown])
240    }
241
242    /// Print the summary and one actionable line per skip cause as cargo
243    /// warnings. The full detail stays in the written report — a build log is
244    /// not the report.
245    pub fn warn(&self) {
246        println!("cargo:warning={}", self.summary());
247        for (capability, entries) in self.skips_by_capability() {
248            let roots = entries
249                .iter()
250                .map(|entry| entry.origin.to_string())
251                .collect::<Vec<_>>();
252            let shown = roots.len().min(5);
253            let more = match roots.len() - shown {
254                0 => String::new(),
255                rest => format!(" (+{rest} more)"),
256            };
257            println!(
258                "cargo:warning=SKIP {capability}: {}{more}",
259                roots[..shown].join(", ")
260            );
261        }
262    }
263}

265/// How a run's declarations were accounted for.
266#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
267pub struct Counts {
268    pub emitted: usize,
269    pub skipped: usize,
270    pub ignored: usize,
271}

273/// Sort key: the origin's own order — kind first, so a report reads types,
274/// then functions — then name.
275pub(crate) fn sort_entries(entries: &mut [Entry]) {
276    entries.sort_by(|a, b| a.origin.cmp(&b.origin));
277}
