1//! Planning: requests in, retained plans out.
2//!
3//! One recursive walk plans the value conversions, one
4//! pass per requested output assembles the wrappers, and one fixpoint
5//! decides what survives. The registry owns all three; the target answers the
6//! local questions in [`crate::target`].

30/// A configuration entry the target interprets. Valid inside one request set.
31#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
32pub struct PolicyId(pub usize);

34/// One requested output.
35#[derive(Clone, Debug)]
36pub struct OutputRequest {
37    /// What was declared, and the run's key for it.
38    pub declaration: Origin,
39    /// The target's configuration for this output.
40    pub policy: PolicyId,
41}

43/// What a frontend hands the engine: what to expose, and the choices that
44/// apply.
45///
46/// Users never write this; a frontend builds it from its own recorded calls,
47/// which is what lets two languages share everything after this point.
48pub struct BindingRequests<Policy> {
49    /// The target this binding generates for — `"c"`, `"jni"`.
50    pub target: &'static str,
51    /// The module generated code reaches the source items through.
52    pub source_module: syn::Path,
53    pub outputs: Vec<OutputRequest>,
54    /// Configurations referenced by [`PolicyId`].
55    pub policies: Vec<Policy>,
56    /// The configuration for a value nothing more specific covers.
57    pub default_policy: PolicyId,
58    /// Per-source-type defaults: `Stamp` crosses this way wherever it appears.
59    pub type_policies: BTreeMap<String, PolicyId>,
60    /// Per-site overrides: this parameter, or the result, of this declared
61    /// source function crosses differently from the type's default. Keyed by
62    /// declaration and the site's path, `param 0` or `return`.
63    pub site_policies: BTreeMap<(Origin, String), PolicyId>,
64    /// Declarations the user asked to leave alone.
65    pub ignored: Vec<Origin>,
66}
67
68impl<Policy> BindingRequests<Policy> {
69    /// A request set with one policy, which every value uses unless something
70    /// more specific is recorded.
71    pub fn new(target: &'static str, source_module: syn::Path, default: Policy) -> Self {
72        BindingRequests {
73            target,
74            source_module,
75            outputs: Vec::new(),
76            policies: vec![default],
77            default_policy: PolicyId(0),
78            type_policies: BTreeMap::new(),
79            site_policies: BTreeMap::new(),
80            ignored: Vec::new(),
81        }
82    }
83
84    /// Record a configuration and get its identity.
85    ///
86    /// Two values share a conversion only when they share the entry, never
87    /// because two entries look alike — a policy can hold a closure, and two
88    /// closures cannot be compared.
89    pub fn policy(&mut self, policy: Policy) -> PolicyId {
90        self.policies.push(policy);
91        PolicyId(self.policies.len() - 1)
92    }
93
94    /// Ask for one declaration to be exposed.
95    pub fn output(&mut self, declaration: Origin, policy: PolicyId) -> &mut Self {
96        self.outputs.push(OutputRequest {
97            declaration,
98            policy,
99        });
100        self
101    }
102
103    pub(crate) fn get(&self, id: PolicyId) -> &Policy {
104        &self.policies[id.0]
105    }
106}

127/// A complete wrapper: the exported function, as planned.
128///
129/// One per declaration that exports a source function, plus one per handle

146/// A conversion whose planning has begun, for cycle detection.
147///
148/// Deliberately coarser than [`NodeKey`]: what makes a cycle is a type reaching
149/// its own conversion, which its children cannot be known before.
150#[derive(Clone, PartialEq, Eq, Hash)]
151struct ResolvingKey {
152    ty: prebindgen_flat::TypeKey,
153    direction: Direction,
154    relation: RelationId,
155    policy: PolicyId,
156}
157
158/// What makes two conversions the same conversion.
159///
160/// The children are part of it. A choice recorded for a field is looked up at
161/// that field's position, so two structs with the same policy whose fields were
162/// configured differently resolve to different children — and must not share a
163/// node, or the second use would silently inherit the first one's conversion.
164#[derive(PartialEq, Eq, Hash)]
165struct NodeKey {
166    ty: prebindgen_flat::TypeKey,
167    direction: Direction,
168    relation: RelationId,
169    policy: PolicyId,
170    children: Vec<NodeId>,
171}

174struct Refusal {

179/// A planned value, or the capability that stopped it.
180enum Planned {
181    Ready(NodeId),
182    Unsupported(Refusal),
183}

187        Planned::Unsupported(Refusal::at(reason, at))

200/// The working state of one `generate` call.
201struct Run<'a, T: Target> {
202    flat: &'a Flat,
203    target: &'a T,
204    requests: &'a BindingRequests<T::Policy>,
205    relations: Vec<Relation>,
206    primitives: Vec<PrimitiveSpec<T::Payload>>,
207    nodes: Vec<ValuePlan<T::Payload>>,
208    cache: HashMap<NodeKey, NodeId>,
209    resolving: std::collections::HashSet<ResolvingKey>,
210    /// The relations offered for a type, registered the first time it is
211    /// planned. A relation's identity is part of a conversion's identity, so
212    /// registering a fresh one per visit would give every value its own node
213    /// and share nothing.
214    offered: HashMap<prebindgen_flat::TypeKey, Vec<(RelationId, Relation)>>,
215}
216
217impl<'a, T: Target> Run<'a, T> {
218    fn new(flat: &'a Flat, target: &'a T, requests: &'a BindingRequests<T::Policy>) -> Self {
219        Run {
220            flat,
221            target,
222            requests,
223            relations: Vec::new(),
224            primitives: Vec::new(),
225            nodes: Vec::new(),
226            cache: HashMap::new(),
227            resolving: std::collections::HashSet::new(),
228            offered: HashMap::new(),
229        }
230    }

232    /// The configuration for a value at this position: the site override, then
233    /// the type default, then the request set's default.
234    fn effective_policy(&self, position: &crate::target::Position, ty: &TypeRef) -> PolicyId {
235        let requests = self.requests;
236        let site = position.path.join(".");
237        if let Some(policy) = requests
238            .site_policies
239            .get(&(position.declaration.clone(), site))
240        {
241            return *policy;
242        }
243        if let Some(name) = type_name(ty) {
244            if let Some(policy) = requests.type_policies.get(&name) {
245                return *policy;
246            }
247        }
248        requests.default_policy
249    }
250
251    /// Register an operation, and hand back the identity instructions name it
252    /// by.
253    fn register(&mut self, primitive: PrimitiveSpec<T::Payload>) -> PrimitiveId {
254        self.primitives.push(primitive);
255        PrimitiveId(self.primitives.len() - 1)
256    }

258    /// The relations available for this type: today the implicit one, which is
259    /// the struct's fields for a struct and the atomic conversion for anything
260    /// carried whole.
261    fn candidates(&mut self, ty: &TypeRef) -> Result<Vec<(RelationId, Relation)>, Unsupported> {
262        if let Some(offered) = self.offered.get(&ty.key()) {
263            return Ok(offered.clone());
264        }
265        let mut relations = vec![Relation::Atomic];
266        if let TypeKind::Named { id, .. } = ty.kind() {
267            match self.flat.resolve(id) {
268                Some(Type::Struct(strukt)) => relations.push(Relation::Struct(StructRelation {
269                    name: strukt.name.to_string(),
270                    parts: strukt
271                        .fields
272                        .iter()
273                        .map(|field| Part {
274                            name: field.name.as_ref().map(|name| name.to_string()),
275                            index: field.index,
276                            ty: field.ty.clone(),
277                            conditions: crate::emit::Writer.conditions(Conditioned::Field(field)),
278                        })
279                        .collect(),
280                })),
281                // An opaque declaration has no parts to offer: a target that
282                // carries it does so whole, through the atomic relation.
283                Some(_) => {}
284                None => {
285                    return Err(Unsupported::new(
286                        "unsupported.type.undeclared",
287                        format!("`{}` names a type this model does not declare", ty.key()),
288                    ))
289                }
290            }
291        }
292        let offered: Vec<(RelationId, Relation)> = relations
293            .into_iter()
294            .enumerate()
295            .map(|(index, relation)| (RelationId(self.relations.len() + index), relation))
296            .collect();
297        self.relations
298            .extend(offered.iter().map(|(_, relation)| relation.clone()));
299        self.offered.insert(ty.key(), offered.clone());
300        Ok(offered)
301    }

303    /// Plan one conversion, recursing into the parts of whatever relation the
304    /// target selects.
305    fn plan_value(
306        &mut self,
307        crossing: Crossing,
308        position: &crate::target::Position,
309    ) -> Result<Planned, PlanningError> {
310        let target = self.target;
311        let requests = self.requests;
312        let policy_id = self.effective_policy(position, &crossing.ty);
313
314        let candidates = match self.candidates(&crossing.ty) {
315            Ok(candidates) => candidates,
316            Err(reason) => return Ok(Planned::refused(reason, position)),
317        };
318
319        let query = SelectionQuery {
320            crossing: &crossing,
321            position,
322            candidates: &candidates,
323            policy: requests.get(policy_id),
324        };
325        let relation_id = match target.select(&query)? {
326            TargetAttempt::Ready(relation) => relation,
327            TargetAttempt::Unsupported(reason) => return Ok(Planned::refused(reason, position)),
328        };
329        let relation = match candidates
330            .iter()
331            .find(|(id, _)| *id == relation_id)
332            .map(|(_, relation)| relation.clone())
333        {
334            Some(relation) => relation,
335            None => {
336                return Err(PlanningError::InternalInvariant(format!(
337                    "the target selected a relation that was not offered for `{}`",
338                    crossing.ty.key()
339                )))
340            }
341        };
342
343        // Meeting a conversion that is already being resolved is a cycle: a
344        // type whose conversion needs its own. The mark is bookkeeping, not a
345        // plan, and it is dropped whichever way this call ends.
346        let resolving = ResolvingKey {
347            ty: crossing.ty.key(),
348            direction: crossing.direction,
349            relation: relation_id,
350            policy: policy_id,
351        };
352        if !self.resolving.insert(resolving.clone()) {
353            return Ok(Planned::refused(
354                Unsupported::new(
355                    "unsupported.conversion.recursive",
356                    format!(
357                        "converting `{}` needs its own conversion; v2 has no recursive \
358                         conversion yet",
359                        crossing.ty.key()
360                    ),
361                ),
362                position,
363            ));
364        }
365        let planned = self.plan_relation(&crossing, position, relation_id, &relation, policy_id);
366        self.resolving.remove(&resolving);
367        planned
368    }

370    /// The body of `plan_value`, once the relation is known: children first,
371    /// then the representation, then the instructions that put them together.
372    fn plan_relation(
373        &mut self,
374        crossing: &Crossing,
375        position: &crate::target::Position,
376        relation_id: RelationId,
377        relation: &Relation,
378        policy_id: PolicyId,
379    ) -> Result<Planned, PlanningError> {
380        let target = self.target;
381        let requests = self.requests;
382        let parts = relation.parts().to_vec();
383
384        // Children are planned before this conversion's identity is known,
385        // because a choice recorded for one of them makes this a different
386        // conversion: two values with the same policy whose fields disagree
387        // must not share a node.
388        let mut children = Vec::new();
389        for part in &parts {
390            let child = Crossing {
391                ty: part.ty.clone(),
392                direction: crossing.direction,
393            };
394            let child_position = position.child(format!("field {}", part.label()));
395            match self.plan_value(child, &child_position)? {
396                Planned::Ready(id) => children.push(id),
397                // One unsupported part makes the whole conversion unsupported.
398                // Nothing partial is recorded: a struct missing a field is a
399                // different type, not a reduced one.
400                Planned::Unsupported(refusal) => return Ok(Planned::Unsupported(refusal)),
401            }
402        }
403
404        let key = NodeKey {
405            ty: crossing.ty.key(),
406            direction: crossing.direction,
407            relation: relation_id,
408            policy: policy_id,
409            children: children.clone(),
410        };
411        if let Some(id) = self.cache.get(&key) {
412            return Ok(Planned::Ready(*id));
413        }
414
415        let strukt = match relation {
416            Relation::Struct(strukt) => self.flat.struct_type(strukt.name.as_str()),
417            Relation::Atomic => None,
418        };
419        let shape = ResolvedShape {
420            crossing,
421            relation,
422            strukt,
423        };
424        let child_values: Vec<ChildValue<'_>> = parts
425            .iter()
426            .zip(&children)
427            .map(|(part, id)| ChildValue {
428                part,
429                layout: &self.nodes[id.0].repr.layout,
430            })
431            .collect();
432        let repr = match target.represent(&shape, &child_values, requests.get(policy_id))? {
433            TargetAttempt::Ready(repr) => repr,
434            TargetAttempt::Unsupported(reason) => return Ok(Planned::refused(reason, position)),
435        };
436        drop(child_values);
437
438        // Compose: obtain each part, convert it, construct the Rust value — or,
439        // out of Rust, the target value.
440        let mut failures: Vec<FailureCategory> = children
441            .iter()
442            .flat_map(|id| self.nodes[id.0].failures.clone())
443            .collect();
444        let mut body = BodyBuilder::new();
445        let carrier = body.fresh();
446        let result = match (&repr.protocol, crossing.direction) {
447            (Protocol::Terminal { codec }, _) => {
448                check_operation(codec, repr.layout.wire(), repr.layout.wire())?;
449                failures.extend(codec.failure_category());
450                self.apply(&mut body, (**codec).clone(), &[carrier])?
451            }
452            (Protocol::Product { projections }, Direction::IntoRust) => {
453                if projections.len() != parts.len() {
454                    return Err(PlanningError::InternalInvariant(format!(
455                        "the representation of `{}` describes {} projection(s) for {} part(s)",
456                        crossing.ty.key(),
457                        projections.len(),
458                        parts.len()
459                    )));
460                }
461                if let Layout::Aggregate { members, .. } = &repr.layout {
462                    if members.len() != parts.len() {
463                        return Err(PlanningError::InternalInvariant(format!(
464                            "the representation of `{}` names {} member(s) for {} part(s)",
465                            crossing.ty.key(),
466                            members.len(),
467                            parts.len()
468                        )));
469                    }
470                }
471                let strukt = match relation {
472                    Relation::Struct(strukt) => strukt.name.clone(),
473                    Relation::Atomic => {
474                        return Err(PlanningError::InternalInvariant(
475                            "a product representation needs a relation with parts".to_string(),
476                        ))
477                    }
478                };
479                let mut converted = Vec::new();
480                for (index, (projection, child)) in projections.iter().zip(&children).enumerate() {
481                    // Everything planned for this part, from the read of the
482                    // member to the last step of its conversion, exists exactly
483                    // where the source field does. The range is stamped rather
484                    // than each instruction, because a child conversion's own
485                    // instructions are inlined here and are equally the part's.
486                    let from = body.len();
487                    check_operation(
488                        projection,
489                        repr.layout.wire(),
490                        self.nodes[child.0].repr.layout.wire(),
491                    )?;
492                    // Every projection reads the same carrier, so none of them
493                    // may claim to consume it.
494                    if projections.len() > 1 && projection.consumes_value() {
495                        return Err(PlanningError::InternalInvariant(format!(
496                            "a projection of `{}` consumes the value the other projections \
497                             still read",
498                            crossing.ty.key()
499                        )));
500                    }
501                    // A member read has to name a member the representation
502                    // declared, or the wrapper reads something the foreign
503                    // caller never filled in.
504                    if let (
505                        crate::target::Operation::Standard(crate::target::StandardOp::ReadMember {
506                            member,
507                        }),
508                        Layout::Aggregate { members, .. },
509                    ) = (&projection.implementation, &repr.layout)
510                    {
511                        if !members.iter().any(|declared| same_member(declared, member)) {
512                            return Err(PlanningError::InternalInvariant(format!(
513                                "a projection of `{}` reads a member its representation does \
514                                 not declare",
515                                crossing.ty.key()
516                            )));
517                        }
518                    }
519                    failures.extend(projection.failure_category());
520                    let obtained = self.apply(&mut body, projection.clone(), &[carrier])?;
521                    let obtained = match obtained {
522                        Some(value) => value,
523                        None => {
524                            return Err(PlanningError::InternalInvariant(
525                                "a projection must produce a value".to_string(),
526                            ))
527                        }
528                    };
529                    let child_body = self.nodes[child.0].body.clone();
530                    converted.push(child_body.inline(obtained, &mut body));
531                    body.condition(from, &parts[index].conditions);
532                }
533                let result = body.fresh();
534                body.push(Instr::Construct {
535                    name: strukt,
536                    parts: converted,
537                    result,
538                });
539                Some(result)
540            }
541            (Protocol::Product { .. }, Direction::OutOfRust) => {
542                return Ok(Planned::refused(
543                    Unsupported::new(
544                        "unsupported.struct.out_of_rust",
545                        format!(
546                            "`{}` leaves Rust as a composed value; v2 has no target \
547                             construction operation yet",
548                            crossing.ty.key()
549                        ),
550                    ),
551                    position,
552                ))
553            }
554        };
555        let result = match result {
556            Some(result) => result,
557            None => {
558                return Err(PlanningError::InternalInvariant(
559                    "a value conversion must produce a value".to_string(),
560                ))
561            }
562        };
563
564        failures.sort();
565        failures.dedup();
566        let id = NodeId(self.nodes.len());
567        self.nodes.push(ValuePlan {
568            id,
569            crossing: crossing.clone(),
570            relation: relation.clone(),
571            repr,
572            children,
573            body: NodeBody {
574                carrier,
575                instrs: body.into_instrs(),
576                result,
577            },
578            failures,
579        });
580        self.cache.insert(key, id);
581        Ok(Planned::Ready(id))
582    }

584    /// Apply one operation to already-available values.
585    ///
586    /// An infallible identity produces no instruction at all: the carrier and
587    /// the converted value are one Rust value, so the conversion renders
588    /// nothing and the caller keeps using the value it already had.
589    fn apply(
590        &mut self,
591        body: &mut BodyBuilder,
592        primitive: PrimitiveSpec<T::Payload>,
593        values: &[ValueId],
594    ) -> Result<Option<ValueId>, PlanningError> {
595        if matches!(
596            primitive.implementation,
597            crate::target::Operation::Standard(crate::target::StandardOp::Identity)
598        ) && matches!(
599            primitive.failure,
600            crate::target::PrimitiveFailure::Infallible
601        ) {
602            return Ok(values.first().copied());
603        }
604        let mut supplied = values.iter();
605        let mut operands = Vec::new();
606        for operand in &primitive.operands {
607            operands.push(match &operand.role {
608                OperandRole::Value => match supplied.next() {
609                    Some(value) => Operand::Value(*value),
610                    None => {
611                        return Err(PlanningError::InternalInvariant(
612                            "an operation takes more values than this conversion supplies"
613                                .to_string(),
614                        ))
615                    }
616                },
617                OperandRole::Context(name) => Operand::Context(name.clone()),
618                // An error value exists on a failure route and nowhere else:
619                // inside a conversion there is nothing to bind it to.
620                OperandRole::Error => {
621                    return Err(PlanningError::InternalInvariant(
622                        "a conversion operation asks for an error value, which only a failure \
623                         route has"
624                            .to_string(),
625                    ))
626                }
627            });
628        }
629        let produces = primitive.result.is_some();
630        let id = self.register(primitive);
631        let result = produces.then(|| body.fresh());
632        body.push(Instr::Apply {
633            primitive: id,
634            operands,
635            result,
636        });
637        Ok(result)
638    }
639}

641/// Check one operation against the values it will be applied to.
642///
643/// An operation states the type of every operand and of its result, and the
644/// registry is the only place those statements meet the carriers actually in
645/// hand. `carrier` is what this conversion is given; `produces` is what the
646/// application must yield — the child's carrier for a projection, the
647/// conversion's own for a terminal codec.
648fn check_operation<P>(
649    primitive: &PrimitiveSpec<P>,
650    carrier: &crate::target::WireType,
651    produces: &crate::target::WireType,
652) -> Result<(), PlanningError> {
653    if primitive.value_operands() != 1 {
654        return Err(PlanningError::InternalInvariant(format!(
655            "this operation takes {} value operand(s); a conversion supplies exactly one",
656            primitive.value_operands()
657        )));
658    }
659    let value = primitive
660        .operands
661        .iter()
662        .find(|operand| matches!(operand.role, OperandRole::Value))
663        .expect("the arity check above found one");
664    if let crate::target::OperationType::Carrier(wire) = &value.ty {
665        if !same_type(&wire.ty, &carrier.ty) {
666            return Err(PlanningError::InternalInvariant(format!(
667                "this operation reads a `{}` and is applied to a `{}`",
668                spell(&wire.ty),
669                spell(&carrier.ty)
670            )));
671        }
672    }
673    match &primitive.result {
674        Some(crate::target::OperationType::Carrier(wire)) if !same_type(&wire.ty, &produces.ty) => {
675            Err(PlanningError::InternalInvariant(format!(
676                "this operation produces a `{}` where a `{}` is expected",
677                spell(&wire.ty),
678                spell(&produces.ty)
679            )))
680        }
681        _ => Ok(()),
682    }
683}

685fn spell(ty: &syn::Type) -> String {
686    use quote::ToTokens;
687    ty.to_token_stream().to_string()

707/// Plan `requests` over `flat` with `target`, and render what survives.
708///
709/// Every requested output leaves this with an outcome. A capability the engine
710/// or the target has not implemented is a reported skip and the run continues;
711/// contradictory input and violated invariants fail.
712pub fn generate<T: Target>(
713    flat: Flat,
714    target: &T,
715    requests: BindingRequests<T::Policy>,
716    declaring_crate: impl Into<String>,
717) -> Result<Generation<T::Payload>, EngineError> {
718    let declared: Vec<Origin> = requests
719        .outputs
720        .iter()
721        .map(|output| output.declaration.clone())
722        .collect();
723    check_declarations(&declared, &flat)?;
724
725    let mut run = Run::new(&flat, target, &requests);
726    let mut functions: Vec<FunctionPlan<T::Payload>> = Vec::new();
727    let mut surfaces: Vec<SurfaceSpec<T::Payload>> = Vec::new();
728    let mut outcomes: BTreeMap<Origin, Outcome> = BTreeMap::new();
729
730    for output in &requests.outputs {
731        let declaration = &output.declaration;
732        let policy = requests.get(output.policy);
733        let root = crate::target::Position::root(declaration.clone());
734        // What is planned follows from the origin, which says both what the
735        // target asked for and what the captured source holds for it. Each
736        // planner is reached by its own variants and is given the captured item
737        // they name; nothing below reads a field back to work out what it was
738        // asked for.
739        let planned = match declaration {
740            Origin::Type(_) | Origin::LocalType(_) => {
741                plan_type(&mut run, declaration, policy).map_err(EngineError::Planning)?
742            }
743            // Two surfaces built the same way: a Kotlin `val` read through a
744            // nullary function is planned as that function, and the target
745            // renders the constant.
746            Origin::Function(ident) | Origin::ConstFromFunction(ident) => {
747                let function = flat
748                    .function(&ident.to_string())
749                    .expect("declarations are checked against the model before planning");
750                plan_function(&mut run, declaration, function, policy)
751                    .map_err(EngineError::Planning)?
752            }
753            // A declaration the binding defines itself names no captured item,
754            // so there is nothing to plan from: its signature or its value is
755            // the binding's own, and reading one is a capability this engine
756            // does not have.
757            Origin::LocalFunction(name) => Err(Refusal::at(
758                Unsupported::new(
759                    "unsupported.fn.binding_local",
760                    format!(
761                        "`{name}` is defined by the binding, not captured from the source, so \
762                         there is no captured item to plan from"
763                    ),
764                ),
765                &root,
766            )),
767            Origin::LocalConst(name) => Err(Refusal::at(
768                Unsupported::new(
769                    "unsupported.const.binding_local",
770                    format!(
771                        "`{name}` is defined by the binding, not captured from the source, so \
772                         there is no captured item to plan from"
773                    ),
774                ),
775                &root,
776            )),
777            // One code per kind rather than one for the whole engine: the
778            // report is how the next capability is chosen, and "everything is
779            // unsupported" chooses nothing.
780            Origin::Const(_) => Err(Refusal::at(
781                Unsupported::new(
782                    "unsupported.const.not_implemented",
783                    "the v2 engine has no const lowering yet",
784                ),
785                &root,
786            )),
787            Origin::Callback(_) => Err(Refusal::at(
788                Unsupported::new(
789                    "unsupported.callback.not_implemented",
790                    "the v2 engine has no callback lowering yet",
791                ),
792                &root,
793            )),
794            Origin::Conversion(_) => Err(Refusal::at(
795                Unsupported::new(
796                    "unsupported.conversion.not_implemented",
797                    "the v2 engine has no conversion lowering yet",
798                ),
799                &root,
800            )),
801        };
802        match planned {
803            Ok(Emitted { function, surface }) => {
804                if let Some(function) = function {
805                    functions.push(function);
806                }
807                surfaces.push(surface);
808                outcomes.insert(declaration.clone(), Outcome::Emitted);
809            }
810            Err(refusal) => {
811                // The path is where the walk actually stopped — the exported
812                // function, the parameter, the field — so the report says what
813                // to look at rather than only which declaration vanished.
814                outcomes.insert(
815                    declaration.clone(),
816                    Outcome::Skipped(Skip {
817                        capability: refusal.reason.capability,
818                        explanation: refusal.reason.explanation,
819                        dependency_path: refusal.at.dependency_path(),
820                    }),
821                );
822            }
823        }
824    }
825
826    // Which declaration represents which type. A target names a requirement by
827    // type, because that is what the model told it about a value; matching the
828    // type to the declaration covering it is the engine's side of that.
829    let declared_types: BTreeMap<String, &Origin> = requests
830        .outputs
831        .iter()
832        .map(|output| &output.declaration)
833        .filter(|declaration| declaration.is_type())
834        .map(|declaration| (declaration.name(), declaration))
835        .collect();
836
837    // A public declaration can require another one. Propagate until a pass
838    // changes nothing: one missing capability, several skipped outputs, each
839    // keeping its own path to the cause.
840    loop {
841        let mut changed = false;
842        for surface in &surfaces {
843            if !matches!(outcomes.get(&surface.declaration), Some(Outcome::Emitted)) {
844                continue;
845            }
846            for required in &surface.requires {
847                let declared = declared_types
848                    .get(required.type_name())
849                    .and_then(|id| outcomes.get(*id));
850                let cause = match declared {
851                    Some(Outcome::Skipped(skip)) => skip.clone(),
852                    Some(_) => continue,
853                    None => Skip::direct(
854                        "unsupported.requirement.unrequested",
855                        format!("requires {required}, which this binding declares no type for"),
856                        required.to_string(),
857                    ),
858                };
859                let mut path = vec![surface.declaration.to_string()];
860                path.extend(cause.dependency_path.iter().cloned());
861                outcomes.insert(
862                    surface.declaration.clone(),
863                    Outcome::Skipped(Skip {
864                        capability: cause.capability,
865                        explanation: cause.explanation,
866                        dependency_path: path,
867                    }),
868                );
869                changed = true;
870                break;
871            }
872        }
873        if !changed {
874            break;
875        }
876    }
877
878    // Only what a retained output needs is published.
879    let emitted = |origin: &Origin| matches!(outcomes.get(origin), Some(Outcome::Emitted));
880    functions.retain(|function| emitted(&function.declaration));
881    surfaces.retain(|surface| emitted(&surface.declaration));
882
883    let mut entries: Vec<Entry> = requests
884        .outputs
885        .iter()
886        .map(|output| Entry {
887            origin: output.declaration.clone(),
888            described: target.describe(requests.get(output.policy)),
889            outcome: outcomes
890                .get(&output.declaration)
891                .cloned()
892                .unwrap_or(Outcome::Emitted),
893        })
894        // An ignore is a decision the binding made about a captured item, so
895        // it has no policy and lands nowhere; the kind column already says
896        // whether a function, a type or a constant was left alone.
897        .chain(requests.ignored.iter().map(|origin| Entry {
898            origin: origin.clone(),
899            described: crate::target::Described::new("ignore", ""),
900            outcome: Outcome::Ignored,
901        }))
902        .collect();
903    sort_entries(&mut entries);
904
905    let report = Report {
906        pipeline: PIPELINE,
907        target: requests.target,
908        schema_version: SCHEMA_VERSION,
909        source_identity: SourceIdentity {
910            declaring_crate: declaring_crate.into(),
911            sources: flat.source_modules().to_vec(),
912            captured_items: flat.elements().count(),
913        },
914        declarations: entries,
915    };
916
917    // Planning is over: the working state hands over its tables, and the model
918    // moves into the frozen result.
919    let nodes = std::mem::take(&mut run.nodes);
920    let primitives = std::mem::take(&mut run.primitives);
921    drop(run);
922
923    // Only what a retained output needs is published: a helper is kept when an
924    // emitted wrapper applies an operation that depends on it, and a type
925    // declaration when its own public declaration survived.
926    let mut artifacts: Vec<crate::target::Artifact> = Vec::new();
927    let keep = |artifact: &crate::target::Artifact, artifacts: &mut Vec<_>| {
928        if !artifacts
929            .iter()
930            .any(|kept: &crate::target::Artifact| kept.name == artifact.name)
931        {
932            artifacts.push(artifact.clone());
933        }
934    };
935    for surface in &surfaces {
936        // Rust a target contributes for its own public declaration of a
937        // captured item — the `repr(C)` mirror of a struct — exists only where
938        // that item does, by the rule a wrapper follows. A declaration the
939        // binding defines itself has no captured item and no condition.
940        let conditions = surface
941            .declaration
942            .captured(&flat)
943            .map(|element| crate::emit::Writer.conditions(Conditioned::Item(element)))
944            .unwrap_or_default();
945        for artifact in &surface.rust {
946            let artifact = match conditions.is_empty() {
947                true => artifact.clone(),
948                false => {
949                    let rust = &artifact.rust;
950                    crate::target::Artifact::new(
951                        artifact.name.clone(),
952                        quote::quote!(#(#conditions)* #rust),
953                    )
954                }
955            };
956            keep(&artifact, &mut artifacts);
957        }
958    }
959    for function in &functions {
960        for step in &function.instrs {
961            if let Instr::Apply { primitive, .. } = &step.instr {
962                for artifact in &primitives[primitive.0].dependencies {
963                    keep(artifact, &mut artifacts);
964                }
965            }
966        }
967        for route in &function.failures {
968            for artifact in route.report.iter().flat_map(|report| &report.dependencies) {
969                keep(artifact, &mut artifacts);
970            }
971        }
972    }
973
974    let rust = crate::emit::render(
975        &flat,
976        target,
977        &requests.source_module,
978        &artifacts,
979        &primitives,
980        &functions,
981    );
982
983    Ok(Generation::new(
984        flat, report, nodes, functions, surfaces, primitives, rust,
985    ))
986}

988/// A requested output that survived planning.
989struct Emitted<P> {
990    function: Option<FunctionPlan<P>>,
991    surface: SurfaceSpec<P>,
992}

994/// Plan the wrapper that exports one source function: the conversions of its
995/// parameters, the call, the conversion of its result, and the wrapper's
996/// interface around them.
997fn plan_function<T: Target>(
998    run: &mut Run<'_, T>,
999    declaration: &Origin,
1000    function: &Function,
1001    policy: &T::Policy,
1002) -> Result<Result<Emitted<T::Payload>, Refusal>, PlanningError> {
1003    let root = crate::target::Position::root(declaration.clone());
1004
1005    // The wrapper is a safe function, and the writer renders a plain call.
1006    // Wrapping an `unsafe fn` would need the wrapper to state the caller's
1007    // obligations, which nothing here can do yet; hiding them in an `unsafe`
1008    // block would make a safe public function out of a contract it does not
1009    // uphold.
1010    if function.is_unsafe() {
1011        return Ok(Err(Refusal::at(
1012            Unsupported::new(
1013                "unsupported.fn.unsafe",
1014                format!(
1015                    "`{}` is an `unsafe fn`, and v2 has no way to carry its safety contract \
1016                     through a wrapper yet",
1017                    function.name
1018                ),
1019            ),
1020            &root,
1021        )));
1022    }
1023
1024    let mut inputs = Vec::new();
1025    for (index, param) in function.params.iter().enumerate() {
1026        let position = root.child(format!("param {index}"));
1027        match run.plan_value(
1028            Crossing {
1029                ty: param.ty.clone(),
1030                direction: Direction::IntoRust,
1031            },
1032            &position,
1033        )? {
1034            Planned::Ready(id) => inputs.push(id),
1035            Planned::Unsupported(refusal) => return Ok(Err(refusal)),
1036        }
1037    }
1038    let output = if matches!(function.ret.kind(), TypeKind::Unit) {
1039        None
1040    } else {
1041        match run.plan_value(
1042            Crossing {
1043                ty: function.ret.clone(),
1044                direction: Direction::OutOfRust,
1045            },
1046            &root.child("return"),
1047        )? {
1048            Planned::Ready(id) => Some(id),
1049            Planned::Unsupported(refusal) => return Ok(Err(refusal)),
1050        }
1051    };
1052
1053    let values = ResolvedValues {
1054        inputs: inputs.iter().map(|id| &run.nodes[id.0]).collect(),
1055        output: output.map(|id| &run.nodes[id.0]),
1056    };
1057    let site = SiteDescriptor {
1058        declaration,
1059        function: Some(function),
1060    };
1061    let boundary = match run.target.boundary(&site, &values, policy)? {
1062        TargetAttempt::Ready(boundary) => boundary,
1063        TargetAttempt::Unsupported(reason) => return Ok(Err(Refusal::at(reason, &root))),
1064    };
1065    // Every failure the conversions declare needs a terminal action here.
1066    for category in values.failure_categories() {
1067        if !boundary
1068            .failures
1069            .iter()
1070            .any(|route| route.category == category)
1071        {
1072            return Ok(Err(Refusal::at(
1073                Unsupported::new(
1074                    "unsupported.boundary.unrouted_failure",
1075                    format!(
1076                    "a conversion can fail with a {} error and this boundary declares no route \
1077                     for it",
1078                    category.as_str()
1079                ),
1080                ),
1081                &root,
1082            )));
1083        }
1084    }
1085    let surface = match run.target.surface(
1086        &SurfaceRequest {
1087            declaration,
1088            policy,
1089            item: SourceItem::Function(function),
1090        },
1091        &values,
1092    )? {
1093        TargetAttempt::Ready(surface) => surface,
1094        TargetAttempt::Unsupported(reason) => return Ok(Err(Refusal::at(reason, &root))),
1095    };
1096    drop(values);
1097
1098    let plan = match assemble(
1099        run,
1100        declaration,
1101        Body::Call(function),
1102        &inputs,
1103        output,
1104        boundary,
1105    )? {
1106        Ok(plan) => plan,
1107        Err(unsupported) => return Ok(Err(unsupported)),
1108    };
1109    Ok(Ok(Emitted {
1110        function: Some(plan),
1111        surface,
1112    }))
1113}

1115/// What a wrapper does with its converted inputs.
1116enum Body<'a, P> {
1117    /// Call the source function once with them.
1118    Call(&'a Function),
1119    /// Apply this operation to the one carrier and produce nothing: a handle's
1120    /// release, which is a wrapper with no source function behind it.
1121    Release(Box<PrimitiveSpec<P>>),
1122}
1123
1124/// Put one wrapper together: wrapper parameters in, conversions, one call — or,
1125/// for a release, one drop — the result out.
1126fn assemble<T: Target>(
1127    run: &mut Run<'_, T>,
1128    declaration: &Origin,
1129    action: Body<'_, T::Payload>,
1130    inputs: &[NodeId],
1131    output: Option<NodeId>,
1132    boundary: crate::target::BoundarySpec<T::Payload>,
1133    // The boundary is consumed: its routes and its ABI belong to the plan.
1134) -> Result<Result<FunctionPlan<T::Payload>, Refusal>, PlanningError> {
1135    let refuse = |reason: Unsupported| {
1136        Ok(Err(Refusal::at(
1137            reason,
1138            &crate::target::Position::root(declaration.clone()),
1139        )))
1140    };
1141    // A symbol reaches generated Rust as a function name, so a policy that
1142    // supplies something else is contradictory input rather than a name the
1143    // writer has to escape.
1144    if syn::parse_str::<syn::Ident>(&boundary.abi.symbol).is_err() {
1145        return Err(PlanningError::InvalidInput(format!(
1146            "`{}` exports the symbol `{}`, which is not a Rust identifier",
1147            declaration, boundary.abi.symbol
1148        )));
1149    }
1150    // The writer exports the wrapper under that symbol with `#[no_mangle]`; an
1151    // attribute restating or contradicting the linkage is not a form the
1152    // target may ask for, and rustc would only report the clash later.
1153    for attr in &boundary.abi.attrs {
1154        if attr.path().is_ident("no_mangle") || attr.path().is_ident("export_name") {
1155            return Err(PlanningError::InvalidInput(format!(
1156                "`{}` states `#[{}]` on its wrapper, whose linkage the writer owns: the \
1157                 symbol is `{}`",
1158                declaration,
1159                attr.path()
1160                    .require_ident()
1161                    .map(|i| i.to_string())
1162                    .unwrap_or_default(),
1163                boundary.abi.symbol
1164            )));
1165        }
1166    }
1167    let mut body = BodyBuilder::new();
1168    let mut params = Vec::new();
1169    let mut contexts: BTreeMap<String, ValueId> = BTreeMap::new();
1170    for param in &boundary.abi.params {
1171        // A carrier that never leaves generated Rust cannot be an extern
1172        // parameter, whatever the adapter would like to pass through it.
1173        if !param.ty.abi {
1174            return Err(PlanningError::InternalInvariant(format!(
1175                "`{}` takes `{}` at its boundary, which is not an ABI carrier",
1176                declaration,
1177                spell(&param.ty.ty)
1178            )));
1179        }
1180        let id = body.fresh();
1181        if let ParamRole::Context(name) = &param.role {
1182            contexts.insert(name.clone(), id);
1183        }
1184        params.push((id, param.clone()));
1185    }
1186    if let Some(ret) = &boundary.abi.ret {
1187        if !ret.abi {
1188            return Err(PlanningError::InternalInvariant(format!(
1189                "`{}` returns `{}` at its boundary, which is not an ABI carrier",
1190                declaration,
1191                spell(&ret.ty)
1192            )));
1193        }
1194        let produced = output.map(|node| run.nodes[node.0].repr.layout.wire().ty.clone());
1195        match produced {
1196            Some(produced) if !same_type(&ret.ty, &produced) => {
1197                return Err(PlanningError::InternalInvariant(format!(
1198                    "`{}` returns `{}` at its boundary, and its result conversion produces a `{}`",
1199                    declaration,
1200                    spell(&ret.ty),
1201                    spell(&produced)
1202                )))
1203            }
1204            None => {
1205                return Err(PlanningError::InternalInvariant(format!(
1206                    "`{}` returns `{}` at its boundary and has no result conversion to fill it",
1207                    declaration,
1208                    spell(&ret.ty)
1209                )))
1210            }
1211            _ => {}
1212        }
1213    }
1214
1215    let mut arguments = Vec::new();
1216    for (index, node) in inputs.iter().enumerate() {
1217        let carrier = params
1218            .iter()
1219            .find(|(_, param)| matches!(param.role, ParamRole::Input(i) if i == index))
1220            .map(|(id, param)| (*id, param.ty.clone()));
1221        let (carrier, wrapper_param) = match carrier {
1222            Some(carrier) => carrier,
1223            None => {
1224                return Err(PlanningError::InternalInvariant(format!(
1225                    "`{}` declares no wrapper parameter for source parameter {index}",
1226                    declaration
1227                )))
1228            }
1229        };
1230        // The conversion reads the carrier this parameter passes, so the two
1231        // are the same type or the wrapper reads something else entirely.
1232        let expected = run.nodes[node.0].repr.layout.wire();
1233        if !same_type(&wrapper_param.ty, &expected.ty) {
1234            return Err(PlanningError::InternalInvariant(format!(
1235                "`{}` passes parameter {index} as `{}`, and its conversion reads a `{}`",
1236                declaration,
1237                spell(&wrapper_param.ty),
1238                spell(&expected.ty)
1239            )));
1240        }
1241        match &action {
1242            Body::Call(_) => {
1243                let node_body = run.nodes[node.0].body.clone();
1244                arguments.push(node_body.inline(carrier, &mut body));
1245            }
1246            // A release takes the carrier as the conversion would, and drops
1247            // what it holds instead of converting it.
1248            Body::Release(release) => {
1249                check_operation(release, expected, expected)?;
1250                run.apply(&mut body, (**release).clone(), &[carrier])?;
1251            }
1252        }
1253    }
1254
1255    let result = match &action {
1256        Body::Call(function) => {
1257            let call = output.map(|_| body.fresh());
1258            body.push(Instr::Call {
1259                function: function.name.to_string(),
1260                args: arguments,
1261                result: call,
1262            });
1263            match (output, call) {
1264                (Some(node), Some(call)) => {
1265                    let node_body = run.nodes[node.0].body.clone();
1266                    Some(node_body.inline(call, &mut body))
1267                }
1268                _ => None,
1269            }
1270        }
1271        Body::Release(_) => None,
1272    };
1273    let result = match boundary.output {
1274        OutputPlacement::Return => result,
1275        OutputPlacement::Void => None,
1276    };
1277
1278    let instrs = body.into_instrs();
1279    // An operation that asks for a runtime context the boundary does not supply
1280    // cannot be assembled. That covers the reporting operations as much as the
1281    // conversions: a route whose reporter needs an environment nobody passes
1282    // would otherwise reach the writer and fail there.
1283    let mut needed: Vec<&String> = Vec::new();
1284    for step in &instrs {
1285        if let Instr::Apply { operands, .. } = &step.instr {
1286            for operand in operands {
1287                if let Operand::Context(name) = operand {
1288                    needed.push(name);
1289                }
1290            }
1291        }
1292    }
1293    for route in &boundary.failures {
1294        let Some(report) = &route.report else {
1295            continue;
1296        };
1297        for operand in &report.operands {
1298            match &operand.role {
1299                OperandRole::Context(name) => needed.push(name),
1300                OperandRole::Error => {}
1301                // A reporting operation is handed the error and its contexts,
1302                // and there is no converted value at that point to give it.
1303                OperandRole::Value => {
1304                    return Err(PlanningError::InternalInvariant(format!(
1305                        "the {} failure route of `{}` reports through an operation that reads a \
1306                         value",
1307                        route.category.as_str(),
1308                        declaration
1309                    )))
1310                }
1311            }
1312        }
1313    }
1314    // The reporter is handed the error the operation produced, so the two have
1315    // to be the same type; a route that reports something else would not
1316    // compile.
1317    for step in &instrs {
1318        let Instr::Apply { primitive, .. } = &step.instr else {
1319            continue;
1320        };
1321        let PrimitiveFailure::Fallible { error, category } = &run.primitives[primitive.0].failure
1322        else {
1323            continue;
1324        };
1325        let Some(route) = boundary
1326            .failures
1327            .iter()
1328            .find(|route| route.category == *category)
1329        else {
1330            continue;
1331        };
1332        let Some(report) = &route.report else {
1333            continue;
1334        };
1335        for operand in &report.operands {
1336            if !matches!(operand.role, OperandRole::Error) {
1337                continue;
1338            }
1339            if let (
1340                crate::target::OperationType::Carrier(reported),
1341                crate::target::OperationType::Carrier(raised),
1342            ) = (&operand.ty, &**error)
1343            {
1344                if !same_type(&reported.ty, &raised.ty) {
1345                    return Err(PlanningError::InternalInvariant(format!(
1346                        "the {} failure route of `{}` reports a `{}` where the operation raises \
1347                         a `{}`",
1348                        category.as_str(),
1349                        declaration,
1350                        spell(&reported.ty),
1351                        spell(&raised.ty)
1352                    )));
1353                }
1354            }
1355        }
1356    }
1357    for name in needed {
1358        if !contexts.contains_key(name) {
1359            return refuse(Unsupported::new(
1360                "unsupported.boundary.missing_context",
1361                format!(
1362                    "an operation needs the `{name}` runtime context and this boundary supplies \
1363                     none"
1364                ),
1365            ));
1366        }
1367    }
1368
1369    Ok(Ok(FunctionPlan {
1370        declaration: declaration.clone(),
1371        abi: boundary.abi,
1372        output: boundary.output,
1373        failures: boundary.failures,
1374        params,
1375        instrs,
1376        result,
1377    }))
1378}

1380/// Plan one exported type: the conversion everything taking it needs, the
1381/// public declaration itself — and, for a type the foreign side holds an
1382/// obligation for, the conversion handing one out and the release freeing it.
1383///
1384/// Whether a type is a handle is the target's answer, not the item's kind: a
1385/// struct under an opaque policy is carried whole as an address, exactly as a
1386/// declared alias is, and an alias a target carries by value would be no
1387/// handle. What says "handle" is the into-Rust representation naming a
1388/// release; the registry then requires the out-of-Rust direction too — a
1389/// handle is a promise that what a function returns can be given back — and
1390/// plans the release as a wrapper under the type's own identity.
1391fn plan_type<T: Target>(
1392    run: &mut Run<'_, T>,
1393    declaration: &Origin,
1394    policy: &T::Policy,
1395) -> Result<Result<Emitted<T::Payload>, Refusal>, PlanningError> {
1396    let flat = run.flat;
1397    let root = crate::target::Position::root(declaration.clone());
1398    // A declared type need not be a captured item — a target may represent
1399    // `String` without the source exporting one — so this lookup, unlike a
1400    // function's, can find nothing.
1401    let (ty, item): (TypeRef, SourceItem<'_>) = match flat.declared_type(&declaration.name()) {
1402        Some(Type::Struct(strukt)) => (strukt.type_ref().clone(), SourceItem::Struct(strukt)),
1403        Some(Type::Extern(opaque)) => {
1404            // A declaration answers what reading names it; an extern keeps
1405            // none, so the model is asked to read its own name.
1406            let name = &opaque.name;
1407            match flat.classify(&syn::parse_quote!(#name)) {
1408                Ok(ty) => (ty, SourceItem::Extern(opaque)),
1409                Err(error) => {
1410                    return Err(PlanningError::InternalInvariant(format!(
1411                        "`{name}` is declared and does not classify as a reference to \
1412                             itself: {error}"
1413                    )))
1414                }
1415            }
1416        }
1417        None => {
1418            return Ok(Err(Refusal::at(
1419                Unsupported::new(
1420                    "unsupported.type.undeclared",
1421                    format!(
1422                        "`{}` is exposed as a type the source did not capture; v2 \
1423                             represents captured types only",
1424                        declaration.name()
1425                    ),
1426                ),
1427                &root,
1428            )))
1429        }
1430        Some(Type::Enum(_) | Type::Variant(_)) => {
1431            return Ok(Err(Refusal::at(
1432                Unsupported::new(
1433                    "unsupported.type.enum",
1434                    format!(
1435                        "`{}` is an enum, which v2 has no representation for yet",
1436                        declaration.name()
1437                    ),
1438                ),
1439                &root,
1440            )))
1441        }
1442    };
1443
1444    let taken = match run.plan_value(
1445        Crossing {
1446            ty: ty.clone(),
1447            direction: Direction::IntoRust,
1448        },
1449        &root,
1450    )? {
1451        Planned::Ready(id) => id,
1452        Planned::Unsupported(refusal) => return Ok(Err(refusal)),
1453    };
1454
1455    let mut given = None;
1456    let mut release_plan = None;
1457    if let Some(release) = run.nodes[taken.0].repr.release.clone() {
1458        given = match run.plan_value(
1459            Crossing {
1460                ty,
1461                direction: Direction::OutOfRust,
1462            },
1463            &root.child("out_of_rust"),
1464        )? {
1465            Planned::Ready(id) => Some(id),
1466            Planned::Unsupported(refusal) => return Ok(Err(refusal)),
1467        };
1468        // The release is a wrapper of its own, over the same carrier the
1469        // consuming conversion reads. Its boundary is the target's answer for
1470        // a site with no source function.
1471        let values = ResolvedValues {
1472            inputs: vec![&run.nodes[taken.0]],
1473            output: None,
1474        };
1475        let site = SiteDescriptor {
1476            declaration,
1477            function: None,
1478        };
1479        let boundary = match run.target.boundary(&site, &values, policy)? {
1480            TargetAttempt::Ready(boundary) => boundary,
1481            TargetAttempt::Unsupported(reason) => return Ok(Err(Refusal::at(reason, &root))),
1482        };
1483        drop(values);
1484        if let Some(category) = release.failure_category() {
1485            if !boundary
1486                .failures
1487                .iter()
1488                .any(|route| route.category == category)
1489            {
1490                return Ok(Err(Refusal::at(
1491                    Unsupported::new(
1492                        "unsupported.boundary.unrouted_failure",
1493                        format!(
1494                            "releasing a handle can fail with a {} error and this boundary \
1495                             declares no route for it",
1496                            category.as_str()
1497                        ),
1498                    ),
1499                    &root,
1500                )));
1501            }
1502        }
1503        release_plan = match assemble(
1504            run,
1505            declaration,
1506            Body::Release(Box::new(release)),
1507            &[taken],
1508            None,
1509            boundary,
1510        )? {
1511            Ok(plan) => Some(plan),
1512            Err(refusal) => return Ok(Err(refusal)),
1513        };
1514    }
1515
1516    let values = ResolvedValues {
1517        inputs: vec![&run.nodes[taken.0]],
1518        output: given.map(|id| &run.nodes[id.0]),
1519    };
1520    let surface = match run.target.surface(
1521        &SurfaceRequest {
1522            declaration,
1523            policy,
1524            item,
1525        },
1526        &values,
1527    )? {
1528        TargetAttempt::Ready(surface) => surface,
1529        TargetAttempt::Unsupported(reason) => return Ok(Err(Refusal::at(reason, &root))),
1530    };
1531    drop(values);
1532    Ok(Ok(Emitted {
1533        function: release_plan,
1534        surface,
1535    }))
1536}
