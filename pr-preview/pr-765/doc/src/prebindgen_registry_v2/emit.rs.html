1//! The common Rust writer: one component, both targets.
2//!
3//! It owns the shape of the generated Rust — the locals and their order, the
4//! branches on failure, the construction of the source value, the call, the
5//! return — and allocates every temporary from the plan, so two operations
6//! rendered into one wrapper cannot collide over a name. A target contributes
7//! one expression per operation of its own ([`Target::render_operation`]) and
8//! whole [`Artifact`]s; it contributes no control flow.
9//!
10//! Source types and struct shapes are generated through Flat's emission
11//! capability, never from retained syntax and never spelled by an adapter.
12

28/// This engine's emission key.
29///
30/// Implementing [`RustEmitter`] is a collector's deliberate decision to
31/// establish an emission boundary; v2 establishes its own rather than borrowing
32/// v1's.
33pub(crate) struct Writer;
34
35impl RustEmitter for Writer {}

37/// The whole generated Rust file, as text.
38pub(crate) fn render<T: Target>(
39    flat: &Flat,
40    target: &T,
41    source_module: &syn::Path,
42    artifacts: &[Artifact],
43    primitives: &[PrimitiveSpec<T::Payload>],
44    functions: &[FunctionPlan<T::Payload>],
45) -> String {
46    // The guards come first, and they are emitted whatever else this run
47    // retained. Each one is a compile-time assertion the capture reader
48    // injected — today, that the source crate's features match the set the
49    // capture was filtered by — and a binding that dropped it would compile
50    // happily against a source crate it disagrees with. It belongs to no
51    // declaration, so nothing in retention decides whether to keep it.
52    let guards = flat.guards().map(|guard| Writer.guard(guard));
53    let artifacts = artifacts.iter().map(|artifact| &artifact.rust);
54    let wrappers = functions
55        .iter()
56        .map(|function| wrapper(flat, target, source_module, primitives, function));
57    let tokens = quote! {
58        #(#guards)*
59        #(#artifacts)*
60        #(#wrappers)*
61    };
62    match syn::parse2::<syn::File>(tokens.clone()) {
63        Ok(file) => prettyplease::unparse(&file),
64        // Unparseable output is a defect in this writer, and hiding it behind
65        // pretty printing would only move the compiler error somewhere less
66        // useful.
67        Err(_) => tokens.to_string(),
68    }
69}

71/// One exported function.
72fn wrapper<T: Target>(
73    flat: &Flat,
74    target: &T,
75    source_module: &syn::Path,
76    primitives: &[PrimitiveSpec<T::Payload>],
77    function: &FunctionPlan<T::Payload>,
78) -> TokenStream {
79    let mut names: HashMap<ValueId, syn::Ident> = HashMap::new();
80    // Parameters are named by the boundary, because a target that requires an
81    // environment operand has to say what it is called. Everything else is
82    // named here, and around those names: a temporary that shadowed a live
83    // parameter would compile and read the wrong value.
84    // Reserved by their canonical spelling: `r#v0` and `v0` are one name to
85    // Rust, so reserving the raw form would leave the plain one free to shadow
86    // it.
87    let mut taken: std::collections::HashSet<String> = function
88        .params
89        .iter()
90        .map(|(_, p)| unraw(&p.name))
91        .collect();
92    let params: Vec<TokenStream> = function
93        .params
94        .iter()
95        .map(|(id, param)| {
96            names.insert(*id, param.name.clone());
97            let name = &param.name;
98            let ty = &param.ty.ty;
99            let mutable = param.mutable.then(|| quote!(mut));
100            quote!(#mutable #name: #ty)
101        })
102        .collect();
103    let ok_binding = free_name("value", &mut taken);
104    let error_binding = free_name("error", &mut taken);
105    // Temporaries are numbered by definition order, which is why the same
106    // operation used twice cannot collide with itself.
107    let mut next_local = 0;
108    let mut local = |names: &mut HashMap<ValueId, syn::Ident>,
109                     taken: &mut std::collections::HashSet<String>,
110                     id: ValueId| {
111        let name = loop {
112            let candidate = format!("v{next_local}");
113            next_local += 1;
114            if taken.insert(candidate.clone()) {
115                break format_ident!("{candidate}");
116            }
117        };
118        names.insert(id, name.clone());
119        name
120    };
121
122    // A wrapper names source items — the function it calls, and every struct it
123    // constructs — and compiles only where all of them exist. So it inherits
124    // each one's condition: a `#[cfg]` the capture reader could not answer and
125    // rewrote back onto the item. Rust conjoins repeated `#[cfg]` attributes on
126    // one item, so carrying them side by side settles the wrapper's own
127    // condition without anything here reading what they say. Two items carrying
128    // the same condition state it once, which conjunction makes cosmetic and
129    // review makes worth doing.
130    //
131    // The instructions are where a wrapper spells a source item: its signature
132    // carries target wire types only, and the one operation that spells a
133    // source type — a handle taken back or released, cast to
134    // `*mut source::Ledger` — is a standard one the registry renders, so it is
135    // found here too.
136    let mut conditions = Vec::new();
137    let mut conditioned: std::collections::HashSet<String> = std::collections::HashSet::new();
138    for step in &function.instrs {
139        let named = match &step.instr {
140            Instr::Construct { name, .. } => name.clone(),
141            Instr::Call { function, .. } => function.clone(),
142            Instr::Apply { primitive, .. } => match &primitives[primitive.0].implementation {
143                Operation::Standard(
144                    StandardOp::FromRaw { source } | StandardOp::Release { source },
145                ) => match source.kind() {
146                    prebindgen_flat::flat::TypeKind::Named { id, .. } => id.name.clone(),
147                    _ => continue,
148                },
149                _ => continue,
150            },
151        };
152        // Planning names only items the model declares.
153        let Some(element) = flat.element(named.as_str()) else {
154            continue;
155        };
156        for condition in Writer.conditions(Conditioned::Item(element)) {
157            if conditioned.insert(condition.to_string()) {
158                conditions.push(condition);
159            }
160        }
161    }
162
163    let mut statements = Vec::new();
164    for step in &function.instrs {
165        // What a field's condition holds the statement to: a member declared
166        // only sometimes is read only sometimes, and the initializer that
167        // consumes the read exists only then. The statement is one `let`, so
168        // the attribute goes in front of it.
169        let guarded = &step.conditions;
170        match &step.instr {
171            Instr::Apply {
172                primitive,
173                operands,
174                result,
175            } => {
176                let primitive = &primitives[primitive.0];
177                let operand_names: Vec<syn::Ident> = operands
178                    .iter()
179                    .map(|operand| operand_name(&names, function, operand))
180                    .collect();
181                let expression = operation(target, source_module, primitive, &operand_names);
182                let statement = match (&primitive.failure, result) {
183                    (PrimitiveFailure::Infallible, Some(result)) => {
184                        let name = local(&mut names, &mut taken, *result);
185                        quote!(let #name = #expression;)
186                    }
187                    (PrimitiveFailure::Infallible, None) => quote!(#expression;),
188                    (PrimitiveFailure::Fallible { category, .. }, Some(result)) => {
189                        let name = local(&mut names, &mut taken, *result);
190                        let route =
191                            failure_arm(target, source_module, function, *category, &error_binding);
192                        let ok = &ok_binding;
193                        let error = error_pattern(function, *category, &error_binding);
194                        quote! {
195                            let #name = match #expression {
196                                Ok(#ok) => #ok,
197                                Err(#error) => { #route }
198                            };
199                        }
200                    }
201                    (PrimitiveFailure::Fallible { category, .. }, None) => {
202                        let route =
203                            failure_arm(target, source_module, function, *category, &error_binding);
204                        let error = error_pattern(function, *category, &error_binding);
205                        quote! {
206                            if let Err(#error) = #expression { #route }
207                        }
208                    }
209                };
210                statements.push(quote!(#(#guarded)* #statement));
211            }
212            Instr::Construct {
213                name,
214                parts,
215                result,
216            } => {
217                let item = flat
218                    .struct_type(name.as_str())
219                    .expect("a construction names a struct the model declares");
220                let ident = &item.name;
221                let head = quote!(#source_module::#ident);
222                // An initializer carries its own field's condition: the value
223                // it names was bound by a statement under the same one, so a
224                // field the source does not have is neither read nor filled in.
225                //
226                // Read from the model here, while the statements got theirs
227                // from `Part::conditions` at plan time. The two agree because
228                // both read the same field; should a part's conditions ever
229                // become something a target contributes to, this has to read
230                // the part instead — an initializer written under a condition
231                // its read does not share names a value that is not there.
232                let bound: Vec<TokenStream> = item
233                    .fields
234                    .iter()
235                    .zip(parts)
236                    .map(|(field, value)| {
237                        let conditions = Writer.conditions(Conditioned::Field(field));
238                        let bound = field.bind(&names[value]);
239                        quote!(#(#conditions)* #bound)
240                    })
241                    .collect();
242                let value = Writer.shape_struct(item, head, &bound);
243                let name = local(&mut names, &mut taken, *result);
244                statements.push(quote!(#(#guarded)* let #name = #value;));
245            }
246            Instr::Call {
247                function: callee,
248                args,
249                result,
250            } => {
251                let callee = format_ident!("{callee}");
252                let args: Vec<syn::Ident> = args.iter().map(|value| names[value].clone()).collect();
253                let call = quote!(#source_module::#callee(#(#args),*));
254                // A source function returning nothing produces no value, and a
255                // `let` over it would name one nobody can use.
256                let statement = match result {
257                    Some(result) => {
258                        let name = local(&mut names, &mut taken, *result);
259                        quote!(let #name = #call;)
260                    }
261                    None => quote!(#call;),
262                };
263                statements.push(quote!(#(#guarded)* #statement));
264            }
265        }
266    }
267
268    let tail = match (&function.output, function.result) {
269        (OutputPlacement::Return, Some(result)) => {
270            let name = &names[&result];
271            quote!(#name)
272        }
273        _ => quote!(),
274    };
275    let symbol = format_ident!("{}", function.abi.symbol);
276    let abi = &function.abi.abi;
277    let attrs = &function.abi.attrs;
278    let unsafety = function.abi.unsafety.then(|| quote!(unsafe));
279    let ret = function
280        .abi
281        .ret
282        .as_ref()
283        .map(|wire| {
284            let ty = &wire.ty;
285            quote!(-> #ty)
286        })
287        .unwrap_or_default();
288    quote! {
289        #(#conditions)*
290        #[no_mangle]
291        #(#attrs)*
292        pub #unsafety extern #abi fn #symbol(#(#params),*) #ret {
293            #(#statements)*
294            #tail
295        }
296    }
297}

299/// One operation's expression: the registry renders its own, the target renders
300/// its own.
301fn operation<T: Target>(
302    target: &T,
303    source_module: &syn::Path,
304    primitive: &PrimitiveSpec<T::Payload>,
305    operands: &[syn::Ident],
306) -> TokenStream {
307    // The handle operations spell a source type, which is what makes them the
308    // registry's: an adapter has no way to, and no business doing it.
309    let source_type = |ty| Writer.emit_source_type(ty, &HashMap::new(), source_module);
310    match &primitive.implementation {
311        Operation::Standard(StandardOp::Identity) => {
312            let value = &operands[0];
313            quote!(#value)
314        }
315        Operation::Standard(StandardOp::ReadMember { member }) => {
316            let value = &operands[0];
317            quote!(#value.#member)
318        }
319        Operation::Standard(StandardOp::IntoRaw { carrier }) => {
320            let value = &operands[0];
321            quote!(Box::into_raw(Box::new(#value)) as #carrier)
322        }
323        Operation::Standard(StandardOp::FromRaw { source }) => {
324            let value = &operands[0];
325            let ty = source_type(source);
326            let message = format!("null `{}` handle", source.key());
327            quote! {
328                ::core::ptr::NonNull::new(#value as *mut #ty)
329                    .map(|handle| unsafe { *Box::from_raw(handle.as_ptr()) })
330                    .ok_or_else(|| String::from(#message))
331            }
332        }
333        Operation::Standard(StandardOp::Release { source }) => {
334            let value = &operands[0];
335            let ty = source_type(source);
336            quote! {
337                drop(::core::ptr::NonNull::new(#value as *mut #ty)
338                    .map(|handle| unsafe { Box::from_raw(handle.as_ptr()) }))
339            }
340        }
341        Operation::Target(payload) => target.render_operation(payload, operands),
342    }
343}

345/// The pattern a failure arm binds the error with: the name when a reporter
346/// reads it, `_` when the route only terminates.
347fn error_pattern<P>(
348    function: &FunctionPlan<P>,
349    category: FailureCategory,
350    error: &syn::Ident,
351) -> TokenStream {
352    let reported = function
353        .failures
354        .iter()
355        .any(|route| route.category == category && route.report.is_some());
356    if reported {
357        quote!(#error)
358    } else {
359        quote!(_)
360    }
361}

363/// What one failure category does at this boundary: report if the target has an
364/// operation for it, then terminate.
365fn failure_arm<T: Target>(
366    target: &T,
367    source_module: &syn::Path,
368    function: &FunctionPlan<T::Payload>,
369    category: FailureCategory,
370    error: &syn::Ident,
371) -> TokenStream {
372    let route = function
373        .failures
374        .iter()
375        .find(|route| route.category == category)
376        .expect("an unrouted failure category is refused before assembly");
377    let report = route.report.as_ref().map(|report| {
378        let operands: Vec<syn::Ident> = report
379            .operands
380            .iter()
381            .map(|operand| match &operand.role {
382                crate::target::OperandRole::Context(name) => context_name(function, name),
383                crate::target::OperandRole::Error => error.clone(),
384                // Refused when the boundary is assembled: a reporting operation
385                // is given the error and the contexts, and has no value to read.
386                crate::target::OperandRole::Value => {
387                    unreachable!("a reporting operation with a value operand is refused")
388                }
389            })
390            .collect();
391        let expression = operation(target, source_module, report, &operands);
392        let on_failure = terminal(&route.on_report_failure);
393        match report.failure {
394            PrimitiveFailure::Infallible => quote!(#expression;),
395            PrimitiveFailure::Fallible { .. } => quote! {
396                if #expression.is_err() { #on_failure }
397            },
398        }
399    });
400    let terminate = terminal(&route.terminate);
401    quote! {
402        #report
403        #terminate
404    }
405}

407fn terminal(terminal: &Terminal) -> TokenStream {
408    match terminal {
409        Terminal::Abort => quote!(std::process::abort();),
410        Terminal::Return(expr) => quote!(return #expr;),
411    }
412}

414fn operand_name<P>(
415    names: &HashMap<ValueId, syn::Ident>,
416    function: &FunctionPlan<P>,
417    operand: &Operand,
418) -> syn::Ident {
419    match operand {
420        Operand::Value(id) => names[id].clone(),
421        Operand::Context(name) => context_name(function, name),
422    }
423}

425/// An identifier's canonical spelling: `r#type` and `type` name one binding.
426fn unraw(ident: &syn::Ident) -> String {
427    ident.to_string().trim_start_matches("r#").to_string()
428}

430/// A name nothing in this wrapper already uses.
431fn free_name(preferred: &str, taken: &mut std::collections::HashSet<String>) -> syn::Ident {
432    if taken.insert(preferred.to_string()) {
433        return format_ident!("{preferred}");
434    }
435    for suffix in 1.. {
436        let candidate = format!("{preferred}_{suffix}");
437        if taken.insert(candidate.clone()) {
438            return format_ident!("{candidate}");
439        }
440    }
441    unreachable!("the search above only ends by returning")
442}

444/// The wrapper parameter supplying a runtime context.
445fn context_name<P>(function: &FunctionPlan<P>, context: &str) -> syn::Ident {
446    function
447        .params
448        .iter()
449        .find(|(_, param)| matches!(&param.role, ParamRole::Context(name) if name == context))
450        .map(|(_, param)| param.name.clone())
451        .expect("a missing runtime context is refused before assembly")
452}
