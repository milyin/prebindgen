9/// Adapter-supplied array input transport. Element conversion remains shared.
10pub struct SequenceInput<W: prebindgen_flat::WireKind, G> {
11    /// Adapter policy for interpreting `controls`; must not own wire descriptors.
12    pub metadata: G,
13    /// Supporting boundary values, stored on the resulting sequence node.
14    pub controls: Vec<Wire<W>>,
15    /// Validate the outer input values and make the supplied length identifier
16    /// available, either as a boundary parameter or a local binding.
17    pub open: TokenStream,
18    /// Per-row setup, including any locals used by `validation_values`.
19    pub bind: TokenStream,
20    /// Raw element expressions keyed by binding name, used during validation.
21    pub validation_values: Vec<(syn::Ident, TokenStream)>,
22    /// Wire expressions safe to capture after releasing the row. `None` marks
23    /// JNI local references, which must be decoded during validation. Any locals
24    /// retained by construction must also remain valid after `release`.
25    pub prepared_values: Vec<(syn::Ident, Option<TokenStream>)>,
26    /// Permit two traversals instead of retaining prepared row closures for a
27    /// grouped resource check. The transport must guarantee stable wire values
28    /// for the whole call, including source constructor invocations. Binding and
29    /// validation must be repeatable without ownership transfer, user code,
30    /// owned temporaries or other observable effects; `release` must also be
31    /// repeatable. These guarantees apply to every child, including resources
32    /// whose alias registration is already handled by their codec. Registration
33    /// alone does not imply acquisition. JNI object rows should retain the
34    /// single-decode path by leaving this false.
35    pub reread_rows: bool,
36    /// Per-row cleanup after construction. Generated code runs this before
37    /// propagating an ordinary conversion error from the row.
38    pub release: TokenStream,
39}
40/// Adapter-supplied array output transport. Owners are committed only after delivery.
41pub struct SequenceOutput<W: prebindgen_flat::WireKind, G> {
42    /// Transfer each element inside `store`, after capturing its prepared wire
43    /// values, into a rollback-capable carrier or to a foreign visitor. A carrier
44    /// must reclaim transferred owners if outer output delivery fails.
45    /// Every fallible row store must precede the marker. Element commit code is
46    /// evaluated in an infallible, unit-returning closure.
47    /// `store` must contain exactly one
48    /// `template_parameter("commit_element")` marker when true. When false, all
49    /// element owners remain in Rust until the outer slot commits.
50    pub commit_each: bool,
51    /// Whether preparing or delivering the array transport can return a binding
52    /// error, in addition to failures from converting its elements.
53    pub fallible: bool,
54    /// Adapter delivery policy; must not own wire descriptors or child trees.
55    pub metadata: G,
56    /// Supporting boundary values, stored on the resulting sequence node.
57    pub controls: Vec<Wire<W>>,
58    /// Initialize delivery state, with the supplied element count available.
59    pub begin: TokenStream,
60    /// Per-row setup before element conversion.
61    pub enter: TokenStream,
62    /// Store or deliver one prepared element's wire values.
63    pub store: TokenStream,
64    /// Finish preparing the result after every element has been delivered.
65    pub finish: TokenStream,
66    /// Named expressions for every outer value. The adapter determines ABI order.
67    pub values: Vec<(syn::Ident, TokenStream)>,
68    /// Complete ownership transfer of the outer carrier, such as a row allocation.
69    pub commit: TokenStream,
70}

72fn replace(code: &TokenStream, values: &[(syn::Ident, TokenStream)]) -> Result<TokenStream> {
73    prebindgen_conversions::validate_value_bindings(values)?;
74    let replacements = values
75        .iter()
76        .map(|(name, value)| {
77            Ok((
78                name.to_string(),
79                syn::parse2::<syn::Expr>(quote!((#value))).map_err(|e| e.to_string())?,
80            ))
81        })
82        .collect::<Result<std::collections::BTreeMap<_, _>>>()?;
83    struct Replace<'a> {
84        values: &'a std::collections::BTreeMap<String, syn::Expr>,
85        error: Option<String>,
86    }
87    impl syn::visit_mut::VisitMut for Replace<'_> {
88        fn visit_macro_mut(&mut self, mac: &mut syn::Macro) {
89            // Alias scopes remain unresolved while outer sequence fields are
90            // bound. Visit both statement branches so nested JNI row expressions
91            // receive the same substitutions as ordinary Rust statements.
92            if mac.path.segments.len() == 2
93                && mac.path.segments[0].ident == "prebindgen_runtime"
94                && mac.path.segments[1].ident == "__prebindgen_alias_checks"
95            {
96                let mut branches = TokenStream::new();
97                for token in mac.tokens.clone() {
98                    let mut block: syn::Block = match syn::parse2(token.into_token_stream()) {
99                        Ok(block) => block,
100                        Err(error) => {
101                            self.error = Some(format!("invalid input alias branch: {error}"));
102                            return;
103                        }
104                    };
105                    self.visit_block_mut(&mut block);
106                    branches.extend(block.into_token_stream());
107                }
108                mac.tokens = branches;
109            }
110        }
111        fn visit_expr_mut(&mut self, expr: &mut syn::Expr) {
112            if let syn::Expr::Path(p) = expr {
113                if p.qself.is_none() && p.path.segments.len() == 1 {
114                    if let Some(value) = self.values.get(&p.path.segments[0].ident.to_string()) {
115                        *expr = value.clone();
116                        return;
117                    }
118                }
119            }
120            syn::visit_mut::visit_expr_mut(self, expr);
121            // An argument already has its own expression boundary. Keep grouping
122            // elsewhere (notably when replacing the operand of `&wire`).
123            if let syn::Expr::Match(value) = expr {
124                if let syn::Expr::Paren(paren) = &*value.expr {
125                    value.expr = paren.expr.clone();
126                }
127            }
128            let args = match expr {
129                syn::Expr::Call(call) => Some(&mut call.args),
130                syn::Expr::MethodCall(call) => Some(&mut call.args),
131                _ => None,
132            };
133            if let Some(args) = args {
134                for arg in args {
135                    if let syn::Expr::Paren(paren) = arg {
136                        *arg = *paren.expr.clone();
137                    }
138                }
139            }
140        }
141    }
142    let mut expr: syn::Expr = syn::parse2(quote!({ #code })).map_err(|e| e.to_string())?;
143    let mut visitor = Replace {
144        values: &replacements,
145        error: None,
146    };
147    syn::visit_mut::VisitMut::visit_expr_mut(&mut visitor, &mut expr);
148    if let Some(error) = visitor.error {
149        return Err(error);
150    }
151    let syn::Expr::Block(block) = expr else {
152        unreachable!()
153    };
154    Ok(block
155        .block
156        .stmts
157        .iter()
158        .map(ToTokens::to_token_stream)
159        .collect())
160}

162fn prepared(
163    code: &TokenStream,
164    values: &[(syn::Ident, Option<TokenStream>)],
165) -> Result<TokenStream> {
166    fn mentions(code: TokenStream, name: &syn::Ident) -> bool {
167        code.into_iter().any(|token| match token {
168            proc_macro2::TokenTree::Ident(id) => id == *name,
169            proc_macro2::TokenTree::Group(group) => mentions(group.stream(), name),
170            _ => false,
171        })
172    }
173    let mut reads = Vec::new();
174    for (name, value) in values {
175        if value.is_none() && mentions(code.clone(), name) {
176            return Err(format!(
177                "collection wire `{}` cannot be read after row validation",
178                name
179            ));
180        }
181        reads.push((name.clone(), value.clone().unwrap_or_else(|| quote!(#name))));
182    }
183    replace(code, &reads)
184}

187pub(crate) fn input<C: Codec>(
188    codec: &C,
189    ty: &Type,
190    extra: &C::Extra,
191    wire: syn::Ident,
192    var: syn::Ident,
193    length: syn::Ident,
194    index: syn::Ident,
195    active: &Condition,
196    child: CodecInput<C>,
197    bulk: bool,
198    deferred_construction: bool,
199) -> Result<CodecInput<C>> {
200    let (element, view) = prebindgen_flat::sequence_parts(ty)
201        .ok_or("sequence recipe requires Vec<T>, &[T] or &Vec<T>")?;
202    if view == SequenceView::Mutable {
203        return Err("mutable collections require explicit write-back support".into());
204    }
205    if matches!(element, Type::Reference(_)) {
206        return Err("collection reference elements are unsupported; use owned elements".into());
207    }
208    codec.validate_input(&child)?;
209    if !view.is_owned()
210        && child
211            .resources
212            .iter()
213            .any(|r| r.access == prebindgen_conversions::Access::Move)
214    {
215        return Err("borrowed collection inputs cannot consume elements; declare an element constructor using borrowed inputs, or use Vec<T>".into());
216    }
217    if child.acknowledgement.is_some() {
218        return Err("collection element acknowledgement belongs to the whole call".into());
219    }
220    let transport = if bulk {
221        codec.sequence_input_bulk(
222            extra,
223            wire.clone(),
224            length.clone(),
225            index.clone(),
226            &child.node,
227            active,
228        )
229    } else {
230        codec.sequence_input(
231            extra,
232            wire.clone(),
233            length.clone(),
234            index.clone(),
235            &child.node,
236            active,
237        )
238    }?;
239    if transport.validation_values.is_empty() && !bulk {
240        return Err("collection elements must have at least one input binding".into());
241    }
242    prebindgen_conversions::validate_value_bindings(&transport.validation_values)?;
243    for (name, _) in &transport.validation_values {
244        if !prebindgen_flat::wire_definitions(std::iter::once(&child.node))
245            .any(|wire| wire.name == *name)
246        {
247            return Err(format!("unknown collection validation binding `{name}`"));
248        }
249    }
250    let expected_names = transport
251        .validation_values
252        .iter()
253        .map(|(name, _)| name.to_string())
254        .collect::<std::collections::BTreeSet<_>>();
255    let prepared_names = transport
256        .prepared_values
257        .iter()
258        .map(|(name, _)| name.to_string())
259        .collect::<std::collections::BTreeSet<_>>();
260    if expected_names != prepared_names || prepared_names.len() != transport.prepared_values.len() {
261        return Err("collection prepared input names differ from validation bindings".into());
262    }
263    let grouped = child.resources.iter().any(|r| !r.registered);
264    let reread = transport.reread_rows && deferred_construction;
265    let shared_only = child
266        .resources
267        .iter()
268        .all(|r| r.access == prebindgen_conversions::Access::Shared);
269    let validation = replace(&child.validation, &transport.validation_values)?;
270    let construction = |code: &TokenStream| {
271        if grouped {
272            prepared(code, &transport.prepared_values)
273        } else {
274            replace(code, &transport.validation_values)
275        }
276    };
277    let acquire = construction(&child.acquire)?;
278    let statements = construction(&child.statements)?;
279    let child_argument = &child.argument;
280    let argument = construction(&quote!(let #var = #child_argument;))?;
281    let group = format_ident!("{wire}_accesses");
282    let staged = format_ident!("{wire}_prepared");
283    let mut collect = TokenStream::new();
284    let mut count = TokenStream::new();
285    let resource_count = child.resources.iter().filter(|r| !r.registered).count();
286    for resource in child.resources.iter().filter(|r| !r.registered) {
287        let condition = &resource.active;
288        let addresses = match &resource.identity {
289            ResourceIdentity::Single(value) => quote!(::core::iter::once((#value) as usize)),
290            ResourceIdentity::Many(value) => quote!(#value),
291        };
292        let exclusive = resource.access != prebindgen_conversions::Access::Shared;
293        collect.extend(prepared(
294            &quote!(if #condition {for address in #addresses {#group.push((address,#exclusive));}}),
295            &transport.prepared_values,
296        )?);
297        if resource.access == prebindgen_conversions::Access::Move && !resource.records_consumption
298        {
299            count.extend(prepared(&quote!(if #condition {__prebindgen_input_consumed.set(__prebindgen_input_consumed.get()+(#addresses).count());}), &transport.prepared_values)?);
300        }
301    }
302    let resources = child
303        .resources
304        .into_iter()
305        .map(|resource| Resource {
306            registered: true,
307            records_consumption: true,
308            // Runtime addresses are checked while each row is alive. This descriptor
309            // retains dynamic cardinality for choosing the enclosing call's history.
310            identity: ResourceIdentity::Many(quote!(::core::iter::empty::<usize>())),
311            active: active.rust(),
312            access: resource.access,
313        })
314        .collect();
315    let open = transport.open;
316    let bind = transport.bind;
317    let release = transport.release;
318    let condition = active.rust();
319    let array = match ty {
320        syn::Type::Reference(r) => r.elem.as_ref(),
321        _ => ty,
322    };
323    let length_check = if let syn::Type::Array(array) = array {
324        let expected = &array.len;
325        quote!(if #length != #expected { return Err("wrong fixed array length".into()); })
326    } else {
327        TokenStream::new()
328    };
329    let to_array = if view == SequenceView::Array {
330        quote!(let #var: #ty = #var.try_into().map_err(|_|"wrong fixed array length".to_string())?;)
331    } else {
332        TokenStream::new()
333    };
334    let value = match view {
335        SequenceView::Owned | SequenceView::Array => quote!(#var),
336        SequenceView::ArrayReference => {
337            quote!(<#ty>::try_from(#var.as_slice()).map_err(|_|"wrong fixed array length".to_string())?)
338        }
339        SequenceView::Vector => quote!(&#var),
340        SequenceView::Slice => quote!(#var.as_slice()),
341        SequenceView::Mutable => unreachable!(),
342    };
343    let element_value = format_ident!("{wire}_element_built");
344    let direct = quote! {
345            let mut #var:Vec<#element>=Vec::new();
346            if #condition { #open #length_check
347                #var.reserve_exact(#length);
348                for #index in 0..#length {
349                    #bind
350                    #[allow(clippy::redundant_closure_call)]
351                    let #element_value=(|| -> ::core::result::Result<#element,String> {
352                        #validation #count #acquire #statements #argument Ok(#var)
353                    })();
354                    #release
355                    let element = #element_value?;
356                    // reserve_exact(length) covers the whole loop; update len after
357                    // each initialized element so ordinary Rust cleanup is sufficient.
358                    // Pointer addition is also valid (and a no-op) for ZSTs.
359                    #[allow(clippy::zst_offset)]
360                    unsafe {
361                        let initialized = #var.len();
362                        #var.as_mut_ptr().add(initialized).write(element);
363                        #var.set_len(initialized + 1);
364                    }
365                }
366            }
367    };
368    let decode = if grouped {
369        let batch = if reread {
370            quote! {
371                let mut #var: Vec<#element> = Vec::new();
372                if #condition { #open #length_check
373                    let mut #group = __prebindgen_input_aliases.begin_group(#length.saturating_mul(#resource_count));
374                    for #index in 0..#length {
375                        #bind
376                        #[allow(unused_variables,clippy::redundant_closure_call)]
377                        let checked = (|| { #validation #collect Ok::<_,String>(()) })();
378                        #release
379                        checked?;
380                    }
381                    __prebindgen_input_aliases.accept_group(#group)?;
382                    #var.reserve_exact(#length);
383                    for #index in 0..#length {
384                        #bind
385                        #[allow(clippy::redundant_closure_call)]
386                        let #element_value=(|| -> ::core::result::Result<#element,String> {
387                            #validation #count #acquire #statements #argument Ok(#var)
388                        })();
389                        #release
390                        let element = #element_value?;
391                        #[allow(clippy::zst_offset)]
392                        unsafe {
393                            let initialized = #var.len();
394                            #var.as_mut_ptr().add(initialized).write(element);
395                            #var.set_len(initialized + 1);
396                        }
397                    }
398                }
399            }
400        } else {
401            quote! {
402                let mut #var: Vec<#element> = Vec::new();
403                if #condition { #open #length_check
404                    let mut #group = __prebindgen_input_aliases.begin_group(#length.saturating_mul(#resource_count));
405                    let mut #staged = Vec::with_capacity(#length);
406                    for #index in 0..#length {
407                        #bind
408                        #[allow(clippy::redundant_closure_call)]
409                        let prepared = (|| {
410                            #validation #collect
411                            Ok::<_,String>(move || -> ::core::result::Result<#element,String> {
412                                #count #acquire #statements #argument Ok(#var)
413                            })
414                        })();
415                        #release
416                        #staged.push(prepared?);
417                    }
418                    __prebindgen_input_aliases.accept_group(#group)?;
419                    #var.reserve_exact(#length);
420                    for build in #staged {
421                        let element = build()?;
422                        // Pointer addition is also valid (and a no-op) for ZSTs.
423                        #[allow(clippy::zst_offset)]
424                        unsafe {
425                            let initialized = #var.len();
426                            #var.as_mut_ptr().add(initialized).write(element);
427                            #var.set_len(initialized + 1);
428                        }
429                    }
430                }
431
432            }
433        };
434        let checked = if shared_only {
435            quote! {
436                let #var = if __prebindgen_input_aliases.needs_group() {
437                    #batch #var
438                } else { #direct #var };
439            }
440        } else {
441            batch
442        };
443        prebindgen_conversions::staged::input_alias_checks(checked, direct)
444    } else {
445        direct
446    };
447    Ok(InputPlan {
448        fallible: true,
449        context: Default::default(),
450        acknowledgement: None,
451        node: WireNode::Chain {
452            controls: transport.controls,
453            child: Box::new(child.node),
454            meta: transport.metadata,
455        },
456        validation: decode,
457        acquire: quote!(),
458        statements: to_array,
459        argument: value,
460        resources,
461    })
462}

464#[allow(clippy::too_many_arguments)]
465pub(crate) fn output<C: Codec>(
466    codec: &C,
467    ty: &Type,
468    extra: &C::Extra,
469    wire: syn::Ident,
470    element: syn::Ident,
471    length: syn::Ident,
472    index: syn::Ident,
473    value: TokenStream,
474    borrowed: bool,
475    mode: ProjectionMode,
476    child: CodecOutput<C>,
477    delivery: crate::SequenceDelivery,
478    bulk: bool,
479) -> Result<CodecOutput<C>> {
480    let (element_type, view) =
481        prebindgen_flat::sequence_parts(ty).ok_or("sequence recipe requires a collection")?;
482    if matches!(element_type, Type::Reference(_)) {
483        return Err("collection reference elements are unsupported; use owned elements".into());
484    }
485    if view == SequenceView::Mutable {
486        return Err("mutable collection outputs are unsupported".into());
487    }
488    if mode == ProjectionMode::Owned && (borrowed || !view.is_owned()) {
489        return Err("owned collection projection cannot consume a borrowed collection".into());
490    }
491    validate_nested_output(&child)?;
492    codec.validate_output(&child)?;
493    if child.values.is_empty() && !bulk {
494        return Err("collection elements must have at least one wire".into());
495    }
496    if !child.delivery.statements.is_empty() {
497        return Err("collection element delivery belongs to the outer slot".into());
498    }
499    let transport = if bulk {
500        codec.sequence_output_bulk(
501            extra,
502            wire.clone(),
503            length.clone(),
504            index.clone(),
505            &child.outcomes.success,
506            &child.values,
507            delivery,
508        )
509    } else {
510        codec.sequence_output(
511            extra,
512            wire.clone(),
513            length.clone(),
514            index.clone(),
515            &child.outcomes.success,
516            &child.values,
517            delivery,
518        )
519    }?;
520    let owner = format_ident!("{wire}_source");
521    let commits = format_ident!("{wire}_commits");
522    let vector = matches!(view, SequenceView::Owned | SequenceView::Vector);
523    // Use standard operations explicitly: extension traits in the including
524    // module must not replace length/iteration and violate native row extents.
525    let (binding, iterator, extent) = if mode == ProjectionMode::Owned {
526        let extent = if vector {
527            quote!(::std::vec::Vec::len(&#owner))
528        } else {
529            quote!(<[#element_type]>::len(&#owner))
530        };
531        (
532            value,
533            quote!(::core::iter::IntoIterator::into_iter(#owner)),
534            extent,
535        )
536    } else {
537        let slice = quote!(::core::convert::AsRef::<[#element_type]>::as_ref(#owner));
538        (
539            quote!(&(#value)),
540            quote!(<[#element_type]>::iter(#slice)),
541            quote!(<[#element_type]>::len(#slice)),
542        )
543    };
544    // Fix the binding's type, rather than relying on a custom Vec name or
545    // Deref target to give len and iteration the standard collection semantics.
546    let binding = if matches!(view, SequenceView::Owned | SequenceView::Vector) {
547        let vector = quote!(::std::vec::Vec<#element_type>);
548        let vector = if mode == ProjectionMode::Owned {
549            vector
550        } else {
551            quote!(&#vector)
552        };
553        quote!(let #owner: #vector = #binding;)
554    } else {
555        quote!(let #owner = #binding;)
556    };
557    let prepare = child.statements;
558    let child_commit = child.commit;
559    let SequenceOutput {
560        commit_each,
561        fallible,
562        metadata,
563        controls,
564        begin,
565        enter,
566        store,
567        finish,
568        values,
569        commit,
570    } = transport;
571    let store = if commit_each {
572        let count = std::cell::Cell::new(0);
573        let store = prebindgen_conversions::render_template(&store, &|name| {
574            if name == "commit_element" {
575                count.set(count.get() + 1);
576                Ok(quote!(
577                    #[allow(clippy::redundant_closure_call, clippy::let_unit_value, clippy::unused_unit)]
578                    let _: () = (|| -> () { #child_commit })();
579                ))
580            } else {
581                Ok(prebindgen_conversions::template_parameter(name))
582            }
583        })?;
584        if count.get() != 1 {
585            return Err("per-element delivery requires exactly one commit_element marker".into());
586        }
587        store
588    } else {
589        quote!(#store #commits.push(move || {#child_commit});)
590    };
591    let declarations = if commit_each {
592        quote!()
593    } else {
594        quote!(let mut #commits=Vec::with_capacity(#length);)
595    };
596    let commit_elements = if commit_each {
597        quote!()
598    } else {
599        quote!(for commit in #commits {commit();})
600    };
601    let out: CodecOutput<C> = OutputPlan {
602        fallible: child.fallible || fallible,
603        context: Default::default(),
604        delivery: Default::default(),
605        outcomes: prebindgen_flat::Outcomes {
606            success: WireNode::Chain {
607                controls,
608                child: Box::new(child.outcomes.success),
609                meta: metadata,
610            },
611            source_error: None,
612            binding_failure: None,
613        },
614        statements: quote! {
615            #binding let #length=#extent; #declarations
616            #begin
617            for (#index,#element) in ::core::iter::Iterator::enumerate(#iterator){ #enter #prepare #store }
618            #finish
619        },
620        values,
621        commit: quote!(#commit #commit_elements),
622    };
623    codec.validate_output(&out)?;
624    Ok(out)
625}

645    #[test]
646    fn wire_substitution_visits_both_policy_branches_and_reports_bad_syntax() {
647        let choice = prebindgen_conversions::staged::input_alias_checks(
648            quote!(let v=wire;),
649            quote!(let v=wire+1;),
650        );
651        let values = vec![(syn::parse_quote!(wire), quote!(42))];

653        for checked in [true, false] {
654            let selected = prebindgen_conversions::render_input_alias_checks(&replaced, checked)
655                .unwrap()
656                .to_string();
657            assert!(selected.contains("42") && !selected.contains("wire"));
658        }
659        let malformed = quote!(::prebindgen_runtime::__prebindgen_alias_checks! {not_a_block});
660        assert!(replace(&malformed, &values)
661            .unwrap_err()
662            .contains("invalid input alias branch"));
