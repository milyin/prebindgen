5/// Initialize the prebindgen output directory for the current crate
6///
7/// This function must be called in the `build.rs` file of any crate that uses
8/// the `#[prebindgen]` attribute macro.
9///
10/// # Panics
11///
12/// Panics if:
13/// - `CARGO_PKG_NAME` environment variable is not set
14/// - `OUT_DIR` environment variable is not set  
15/// - Directory creation or file operations fail
16///
17/// # Example
18///
19/// ```rust,no_run
20/// // build.rs
21/// fn main() {
22///     prebindgen::init_prebindgen_out_dir();
23/// }
24/// ```
25// The `fn main` above is the subject of the example, not boilerplate rustdoc
26// would supply: this shows a whole `build.rs`, and where the call sits in it.
27#[allow(clippy::needless_doctest_main)]
28pub fn init_prebindgen_out_dir() {
29    env::var("OUT_DIR").expect(
30        "OUT_DIR environment variable not set. This function should be called from build.rs.",
31    );
32    // Get the crate name from CARGO_PKG_NAME or use fallback
33    // For doctests, use "source_ffi" even if CARGO_PKG_NAME is set to "prebindgen"
34    let crate_name = env::var("CARGO_PKG_NAME").expect("CARGO_PKG_NAME environment variable not set. This function should be called from build.rs.");
35
36    // delete all files in the prebindgen directory
37    let prebindgen_dir = get_prebindgen_out_dir();
38    if prebindgen_dir.exists() {
39        for entry in fs::read_dir(&prebindgen_dir).unwrap() {
40            let entry = entry.unwrap();
41            let path = entry.path();
42            if path.is_file() {
43                fs::remove_file(&path).unwrap_or_else(|e| {
44                    panic!("Failed to delete {}: {}", path.display(), e);
45                });
46            }
47        }
48    } else {
49        fs::create_dir_all(&prebindgen_dir).unwrap_or_else(|e| {
50            panic!(
51                "Failed to create prebindgen directory {}: {}",
52                prebindgen_dir.display(),
53                e
54            );
55        });
56    }
57
58    // Store the crate name in a separate file
59    let crate_name_path = prebindgen_dir.join(CRATE_NAME_FILE);
60    fs::write(&crate_name_path, &crate_name).unwrap_or_else(|e| {
61        panic!(
62            "Failed to write crate name to {}: {}",
63            crate_name_path.display(),
64            e
65        );
66    });
67
68    let features = get_enabled_features();
69
70    // Save features list to features.txt (one per line)
71    let features_path = prebindgen_dir.join(FEATURES_FILE);
72    let mut feature_contents = String::new();
73    for feature in &features {
74        feature_contents += feature;
75        feature_contents.push('\n');
76    }
77    fs::write(&features_path, feature_contents).unwrap_or_else(|e| {
78        panic!(
79            "Failed to write features to {}: {}",
80            features_path.display(),
81            e
82        );
83    });
84
85    // Export features list to the main crate as an env variable
86    // Accessible via env!("PREBINDGEN_FEATURES") or std::env::var at compile time/runtime
87    // Make the list of format "crate_name/f1 crate_name/f2"
88    println!(
89        "cargo:rustc-env=PREBINDGEN_FEATURES={}",
90        features
91            .into_iter()
92            .map(|f| format!("{}/{}", crate_name, f))
93            .collect::<Vec<_>>()
94            .join(" ")
95    );
96}

98/// Read all feature names declared in the current crate's `Cargo.toml`.
99///
100/// Notes:
101/// - Excludes the special `default` feature from the returned set.
102/// - Panics if called outside of build.rs (requires OUT_DIR and CARGO_MANIFEST_DIR).
103pub fn get_all_features() -> BTreeSet<String> {
104    // Ensure we are in build.rs context
105    env::var("OUT_DIR").expect(
106        "OUT_DIR environment variable not set. This function should be called from build.rs.",
107    );
108    let manifest_dir = env::var("CARGO_MANIFEST_DIR").expect(
109        "CARGO_MANIFEST_DIR environment variable not set. This function should be called from build.rs.",
110    );
111    let manifest_path = Path::new(&manifest_dir).join("Cargo.toml");
112
113    let manifest_content = fs::read_to_string(&manifest_path).unwrap_or_else(|e| {
114        panic!(
115            "Failed to read Cargo.toml at {}: {}",
116            manifest_path.display(),
117            e
118        )
119    });
120
121    let doc: toml::Value = toml::from_str(&manifest_content).unwrap_or_else(|e| {
122        panic!(
123            "Failed to parse Cargo.toml at {} as TOML: {}",
124            manifest_path.display(),
125            e
126        )
127    });
128
129    let mut set = BTreeSet::new();
130    if let Some(features_tbl) = doc.get("features").and_then(|v| v.as_table()) {
131        for key in features_tbl.keys() {
132            if key != "default" {
133                set.insert(key.to_string());
134            }
135        }
136    }
137    set
138}

140/// Check whether a feature is enabled by looking at the corresponding
141/// `CARGO_FEATURE_<NAME>` environment variable provided to build scripts by Cargo.
142/// Hyphens in feature names are converted to underscores to match Cargo's env var format.
143pub fn is_feature_enabled(feature: &str) -> bool {
144    // Ensure we are in build.rs context
145    env::var("OUT_DIR").expect(
146        "OUT_DIR environment variable not set. This function should be called from build.rs.",
147    );
148    let env_key = format!(
149        "CARGO_FEATURE_{}",
150        feature
151            .to_ascii_uppercase()
152            .chars()
153            .map(|c| if c == '-' { '_' } else { c })
154            .collect::<String>()
155    );
156    env::var_os(env_key).is_some()
157}

159/// Filter the full features list to only those that are currently enabled.
160pub fn get_enabled_features() -> BTreeSet<String> {
161    get_all_features()
162        .into_iter()
163        .filter(|f| is_feature_enabled(f))
164        .collect()
165}

167/// Name of the prebindgen output directory
168const PREBINDGEN_DIR: &str = "prebindgen";
169
170/// Get the full path to the prebindgen output directory in OUT_DIR.
171pub fn get_prebindgen_out_dir() -> std::path::PathBuf {
172    let out_dir = std::env::var("OUT_DIR")
173        .expect("OUT_DIR environment variable not set. Check if build.rs for the crate exitsts");
174    std::path::Path::new(&out_dir).join(PREBINDGEN_DIR)
175}

177/// Macro for debug tracing in build.rs. Used by prebindgen-proc-macro to display paths to
178/// generated files, but can be also used in other contexts.
179#[macro_export]
180macro_rules! trace {
181    ($($arg:tt)*) => {
182        println!("cargo:warning=[{}:{}] {}",
183            file!(),
184            line!(),
185            format!($($arg)*)
186        );
187    };
