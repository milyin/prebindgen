1//! Build-time selection of native or externally measured opaque storage.
2use super::*;
3use std::collections::BTreeSet;

5/// How to obtain an inline owner's size and alignment.
6/// Native builds can measure a Rust type available to the build script; cross
7/// builds always require an external artifact, never a host-size fallback.
8#[derive(Clone, Debug, PartialEq, Eq)]
9pub struct OpaqueLayout {
10    native: Option<StorageLayout>,
11    key: Option<String>,
12}
13impl OpaqueLayout {
14    /// Measure `T` only for a native binding build. During cross compilation,
15    /// use the external entry named by the C declaration (or a built-in key).
16    pub const fn auto<T>() -> Self {
17        Self {
18            native: Some(StorageLayout {
19                size: std::mem::size_of::<T>(),
20                alignment: std::mem::align_of::<T>(),
21            }),
22            key: None,
23        }
24    }
25    /// Require external data even on native builds. No host type is needed.
26    pub const fn external() -> Self {
27        Self {
28            native: None,
29            key: None,
30        }
31    }
32    /// Select a shared artifact key instead of the declaration's default key.
33    pub fn key(mut self, key: impl Into<String>) -> Self {
34        self.key = Some(key.into());
35        self
36    }
37}
38pub(crate) type StorageLayout = prebindgen_reprc::SizeAlign;
39/// External target measurements shared with other language generators.
40pub type OpaqueLayouts = prebindgen_reprc::LayoutArtifact;
41pub(crate) fn builtin(ty: &syn::Type) -> OpaqueLayout {
42    if prebindgen_flat::type_key(ty) == "String" {
43        OpaqueLayout::auto::<String>().key("std.string")
44    } else {
45        OpaqueLayout::auto::<Vec<u8>>().key("std.vec")
46    }
47}
48fn nodes(
49    plan: &OutputPlan<CWire>,
50) -> impl Iterator<Item = &prebindgen_flat::WireNode<CWire, CChain, CBranch>> {
51    std::iter::once(&plan.outcomes.success)
52        .chain(plan.outcomes.source_error.iter())
53        .chain(plan.outcomes.binding_failure.iter())
54}
55fn used_node(
56    node: &prebindgen_flat::WireNode<CWire, CChain, CBranch>,
57    names: &mut BTreeSet<String>,
58) {
59    for wire in prebindgen_flat::wire_definitions(std::iter::once(node)) {
60        match &wire.kind {
61            CWire::Owned { owner, .. } => {
62                names.insert(owner.name.clone());
63            }
64            CWire::Callback { callback } => callback.layout_names(names),
65            _ => {}
66        }
67    }
68}
69pub(crate) fn output_names(plan: &OutputPlan<CWire>, names: &mut BTreeSet<String>) {
70    for node in nodes(plan) {
71        used_node(node, names);
72    }
73}
74pub(crate) fn input_names(plan: &InputPlan<CWire>, names: &mut BTreeSet<String>) {
75    used_node(&plan.node, names);
76}
77pub(crate) fn prepare(
78    flat: &mut Bindings,
79    target: &str,
80    artifact: Option<&OpaqueLayouts>,
81    sources: &[prebindgen::Source],
82) -> Result<()> {
83    let mut used = BTreeSet::new();
84    for f in flat.functions.values().filter(|f| f.ret.slot.is_some()) {
85        for p in &f.params {
86            if let Some(plan) = &p.ty.slot {
87                input_names(plan, &mut used);
88            }
89        }
90        output_names(f.ret.slot.as_ref().unwrap(), &mut used);
91    }
92    for info in flat.types.values() {
93        if let Some(error) = &info.metadata.error {
94            output_names(error.plan()?, &mut used);
95        }
96    }
97    // Keep nested standard owners so Vec<String>/Vec<Vec<T>> can expose typed
98    // borrowed element pointers and their ordinary accessors.
99    loop {
100        let before = used.len();
101        for info in flat.types.values() {
102            if let Some(owner) = &info.metadata.owned {
103                if used.contains(&owner.name) || owner.explicit {
104                    if let Some((element, _)) = prebindgen_flat::sequence_parts(&owner.ty) {
105                        if let Some(child) = owned::declaration(flat, element) {
106                            used.insert(child.name.clone());
107                        }
108                    }
109                }
110            }
111        }
112        if before == used.len() {
113            break;
114        }
115    }
116    for info in flat.types.values_mut() {
117        if info
118            .metadata
119            .owned
120            .as_ref()
121            .is_some_and(|o| !o.explicit && !used.contains(&o.name))
122        {
123            info.metadata.owned = None;
124        }
125    }
126    if let Some(data) = artifact {
127        if data.target != target {
128            return Err(format!(
129                "opaque layout target mismatch: artifact `{}`, binding `{target}`",
130                data.target
131            ));
132        }
133        let mut verified = BTreeSet::new();
134        for source in sources {
135            let name = source.crate_name();
136            verified.insert(name.to_owned());
137            let features = data
138                .sources
139                .get(name)
140                .ok_or_else(|| format!("missing opaque layout source features for `{name}`"))?;
141            let guard = source.require_features(&features.features)?;
142            if !flat
143                .guards
144                .iter()
145                .any(|g| quote!(#g).to_string() == quote!(#guard).to_string())
146            {
147                flat.guards.push(guard);
148            }
149        }
150        for function in flat.functions.values() {
151            if let Some(name) = &function.location.crate_name {
152                if !verified.contains(name) {
153                    return Err(format!(
154                        "register layout source `{name}` to verify captured features"
155                    ));
156                }
157            }
158        }
159    }
160    flat.metadata.opaque_layouts.clear();
161    for info in flat.types.values() {
162        if let Some(owner) = &info.metadata.owned {
163            let key = owner.layout.key.clone().unwrap_or_else(|| {
164                if owned::std_owner(&owner.ty) {
165                    if prebindgen_flat::type_key(&owner.ty) == "String" {
166                        "std.string".into()
167                    } else {
168                        "std.vec".into()
169                    }
170                } else {
171                    owner.name.clone()
172                }
173            });
174            if key.trim().is_empty() {
175                return Err(format!(
176                    "opaque owner `{}` has an empty layout key",
177                    owner.name
178                ));
179            }
180            let layout = if let Some(data) = artifact {
181                data.storage(&key)?
182            } else if target != env!("PREBINDGEN_C_GENERATOR_TARGET") {
183                return Err(format!("cross-compilation to `{target}` requires an external opaque layout artifact (missing `{key}`); generator platform is `{}`",env!("PREBINDGEN_C_GENERATOR_TARGET")));
184            } else {
185                owner.layout.native.ok_or_else(|| {
186                    format!(
187                        "opaque owner `{}` requires external layout `{key}`",
188                        owner.name
189                    )
190                })?
191            };
192            layout.validate()?;
193            flat.metadata
194                .opaque_layouts
195                .insert(owner.name.clone(), layout);
196        }
197    }
198    Ok(())
199}

215    fn artifact(
216        target: &str,
217        layouts: serde_json::Value,
218        sources: serde_json::Value,
219    ) -> OpaqueLayouts {
220        OpaqueLayouts::from_json(&serde_json::json!({"format_version":1,"target":target,"sources":sources,"layouts":layouts}).to_string()).unwrap()
221    }

225    #[test]
226    fn native_measurement_and_strict_cross_lookup_are_order_independent() {
227        let make = || Cbindgen::new(model("pub type Item=i64;pub fn make()->Item{}"));
228        let native = make()
229            .opaque_owned_struct("item", syn::parse_quote!(Item), OpaqueLayout::auto::<i64>())
230            .function(fun!(make))
231            .prepare()
232            .unwrap();
233        assert_eq!(
234            native.metadata.opaque_layouts["item"].size,
235            std::mem::size_of::<i64>()
236        );
237        let cross = "i686-unknown-linux-gnu";
238        let error = make()
239            .target(cross)
240            .opaque_owned_struct("item", syn::parse_quote!(Item), OpaqueLayout::auto::<i64>())
241            .function(fun!(make))
242            .prepare()
243            .err()
244            .unwrap();
245        assert!(
246            error.contains("requires an external opaque layout artifact"),
247            "{error}"
248        );
249        let data = artifact(
250            cross,
251            serde_json::json!({"item":{"size":8,"alignment":4}}),
252            serde_json::json!({}),
253        );
254        // A zero-sized host measurement must be ignored for a cross target.
255        let first = make()
256            .target(cross)
257            .opaque_layouts(data.clone())
258            .opaque_owned_struct("item", syn::parse_quote!(Item), OpaqueLayout::auto::<()>())
259            .function(fun!(make))
260            .prepare()
261            .unwrap();
262        let second = make()
263            .opaque_owned_struct("item", syn::parse_quote!(Item), OpaqueLayout::auto::<()>())
264            .function(fun!(make))
265            .opaque_layouts(data)
266            .target(cross)
267            .prepare()
268            .unwrap();
269        assert_eq!(
270            first.metadata.opaque_layouts,
271            second.metadata.opaque_layouts
272        );
273        assert_eq!(
274            first.metadata.opaque_layouts["item"],
275            StorageLayout {
276                size: 8,
277                alignment: 4
278            }
279        );
280    }

406    fn source_dir(features: &str) -> (tempfile::TempDir, prebindgen::Source) {
407        let dir = tempfile::tempdir().unwrap();
408        std::fs::write(dir.path().join("crate_name.txt"), "flat").unwrap();
409        std::fs::write(dir.path().join("features.txt"), features).unwrap();
410        let source = prebindgen::Source::builder(dir.path())
411            .enable_target_filtering(None::<String>)
412            .build();
413        (dir, source)
414    }
415    #[test]
416    fn probe_features_are_checked_and_guarded_against_the_target_library() {
417        let (_dir, source) = source_dir("b\na\n");
418        let make = || {
419            Cbindgen::new(model("pub type Item=i64;")).opaque_owned_struct(
420                "item",
421                syn::parse_quote!(Item),
422                OpaqueLayout::external(),
423            )
424        };
425        let data = artifact(
426            native(),
427            serde_json::json!({"item":{"size":8,"alignment":8}}),
428            serde_json::json!({"flat":{"features":["a","b"]}}),
429        );
430        let prepared = make()
431            .opaque_layouts(data.clone())
432            .layout_source(source.clone())
433            .layout_source(source.clone())
434            .prepare()
435            .unwrap();
436        let guards = prepared
437            .guards
438            .iter()
439            .map(|g| quote!(#g).to_string())
440            .collect::<String>();
441        assert!(guards.contains("flat :: FEATURES"));
442        assert!(guards.contains("flat/a flat/b"));
443        assert_eq!(
444            prepared
445                .guards
446                .iter()
