1//! Layout queries for existing native records; no replacement struct is emitted.
2use crate::{MeasuredLayout, Record, Result};
3use proc_macro2::TokenStream;
4use quote::quote;
5use syn::{Ident, Type};
6/// A named-field native type whose actual target layout is measured in generated code.
7/// The adapter verifies representation and field-carrier compatibility.
8#[derive(Clone, Debug)]
9pub struct NativeRecord {
10    ty: Type,
11    fields: Vec<(Ident, Type)>,
12}
13impl NativeRecord {
14    /// Describe an existing type and its real fields, preserving declaration order.
15    /// No ownership, bit-validity or representation guarantee is inferred.
16    pub fn new(ty: Type, fields: impl IntoIterator<Item = (Ident, Type)>) -> Result<Self> {
17        let fields = fields.into_iter().collect::<Vec<_>>();
18        Record::new(syn::parse_quote!(__NativeRecordValidation), fields.clone())?;
19        Ok(Self { ty, fields })
20    }
21    /// Actual Rust source type, qualified for the generated crate.
22    pub fn rust_type(&self) -> &Type {
23        &self.ty
24    }
25    /// Real source field names and types, in schema order.
26    pub fn fields(&self) -> impl ExactSizeIterator<Item = (&Ident, &Type)> {
27        self.fields.iter().map(|(n, t)| (n, t))
28    }
29    /// Generate target-evaluated size, alignment and field offsets.
30    pub fn layout(&self) -> TokenStream {
31        let ty = &self.ty;
32        let offsets = self
33            .fields
34            .iter()
35            .map(|(f, _)| quote!(::core::mem::offset_of!(#ty,#f)));
36        quote!([::core::mem::size_of::<#ty>(),::core::mem::align_of::<#ty>(),#(#offsets),*])
37    }
38    /// Require a complete external record measurement and generate target assertions.
39    pub fn assertions(&self, measured: &MeasuredLayout) -> Result<TokenStream> {
40        measured.validate()?;
41        if measured.fields.len() != self.fields.len() {
42            return Err("external record layout must include every field exactly once".into());
43        }
44        let ty = &self.ty;
45        let size = measured.size;
46        let alignment = measured.alignment;
47        let mut checks = TokenStream::new();
48        for (field, _) in &self.fields {
49            let key = field.to_string();
50            let key = key.trim_start_matches("r#");
51            let offset = measured
52                .fields
53                .get(key)
54                .ok_or_else(|| format!("missing external record field `{key}`"))?;
55            checks.extend(quote!(assert!(::core::mem::offset_of!(#ty,#field)==#offset);));
56        }
57        Ok(
58            quote!(const _:()={assert!(::core::mem::size_of::<#ty>()==#size);assert!(::core::mem::align_of::<#ty>()==#alignment);#checks};),
59        )
60    }
61}

63#[cfg(test)]
64mod tests {
65    use super::*;
66    #[test]
67    fn external_fields_must_match_the_source_exactly() {
68        let native = NativeRecord::new(
69            syn::parse_quote!(source::Row),
70            [
71                (syn::parse_quote!(r#type), syn::parse_quote!(u64)),
72                (syn::parse_quote!(tag), syn::parse_quote!(u8)),
73            ],
74        )
75        .unwrap();
76        let mut layout = MeasuredLayout {
77            size: 16,
78            alignment: 8,
79            fields: [("type".into(), 0), ("tag".into(), 8)].into(),
80        };
81        let checks = native.assertions(&layout).unwrap();
82        assert!(checks.to_string().contains("offset_of"));
83        let measured = native.layout().to_string();
84        assert!(measured.contains("source :: Row"));
85        layout.fields.remove("tag");
86        assert!(native.assertions(&layout).is_err());
87        layout.fields.insert("typo".into(), 8);
88        assert!(native.assertions(&layout).unwrap_err().contains("tag"));
89    }
