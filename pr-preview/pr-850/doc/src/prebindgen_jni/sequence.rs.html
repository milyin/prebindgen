8/// Selected delivery of a logical sequence.
9#[derive(Clone, Copy, Debug, PartialEq, Eq)]
10pub enum JniSequence {
11    /// All rows travel in one collection carrier.
12    Buffered,
13    /// Native row ranges inside an enclosing bulk frame.
14    Bulk,
15    /// Each row is delivered through the function's sequence callback.
16    Callback,
17}

19pub(crate) fn input(
20    wire: syn::Ident,
21    length: syn::Ident,
22    index: syn::Ident,
23    element: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
24    active: &Condition,
25) -> Result<SequenceInput<JniWire, crate::JniChain>> {
26    let env = prebindgen_conversions::template_parameter("environment");
27    let row = format_ident!("{wire}_row");
28    let count = crate::group::row_fields(element).len();
29    let capacity = i32::try_from(count + 8).map_err(|_| "too many collection element fields")?;
30    let frame = format_ident!("{wire}_frame");
31    let mut bind = quote! {let #frame=unsafe {::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;let #row=::prebindgen_jni_runtime::row(&mut #env,&#wire,#index,#count)?;};
32    let mut validation_values = Vec::new();
33    let mut prepared_values = Vec::new();
34    for (position, field) in crate::group::row_fields(element).enumerate() {
35        let name = &field.name;
36        let read = field.kind.read_array_element(&quote!(#row), position);
37        bind.extend(quote! {
38            let #name=#read;
39            if #name.is_err() && #env.exception_check().map_err(|e|e.to_string())? {return Err("JNI collection field decoding failed".into());}
40        });
41        let borrow = quote!(*);
42        validation_values.push((
43            field.name.clone(),
44            quote!(#borrow #name.as_ref().map_err(|e|e.clone())?),
45        ));
46        prepared_values.push((
47            field.name.clone(),
48            if field.kind.is_reference() {
49                None
50            } else {
51                Some(quote!(#borrow #name.as_ref().expect("validated collection field")))
52            },
53        ));
54    }
55    Ok(SequenceInput {
56        controls: vec![prebindgen_flat::Wire {
57            name: wire.clone(),
58            kind: JniWire::Rows {
59                condition: kotlin_condition(active),
60                null_niche_available: true,
61            },
62        }],
63        metadata: JniChain::Sequence(JniSequence::Buffered),
64        open: quote! {if #wire.is_null(){return Err("null collection".into());}let #length=#env.get_array_length(&#wire).map_err(|e|e.to_string())? as usize;},
65        bind,
66        validation_values,
67        prepared_values,
68        release: quote!(drop(#frame);),
69    })
70}
71pub(crate) fn output(
72    wire: syn::Ident,
73    length: syn::Ident,
74    index: syn::Ident,
75    element: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
76    values: &[(syn::Ident, TokenStream)],
77) -> Result<SequenceOutput<JniWire, crate::JniChain>> {
78    let env = prebindgen_conversions::template_parameter("environment");
79    let row = format_ident!("{wire}_row");
80    let count = crate::group::row_fields(element).len();
81    let capacity = i32::try_from(count + 8).map_err(|_| "too many collection fields")?;
82    let encode = encode_array(crate::group::row_values(element, values)?, &row, 0);
83    let frame = format_ident!("{wire}_frame");
84    Ok(SequenceOutput {
85        controls: vec![prebindgen_flat::Wire {
86            name: wire.clone(),
87            kind: JniWire::Rows {
88                condition: ("true").into(),
89                null_niche_available: true,
90            },
91        }],
92        commit_each: false,
93        fallible: true,
94        metadata: JniChain::Sequence(JniSequence::Buffered),
95        begin: quote!(let #wire=::prebindgen_jni_runtime::array(&mut #env,#length)?;),
96        enter: quote!(let #frame=unsafe {::prebindgen_jni_runtime::LocalFrame::new(&#env,#capacity)}?;),
97        store: quote! {
98            let #row=::prebindgen_jni_runtime::array(&mut #env,#count)?;#encode
99            #env.set_object_array_element(&#wire,#index as i32,&#row).map_err(|e|e.to_string())?;
100            drop(#frame);
101        },
102        finish: quote!(let #wire=Some(#wire);),
103        values: vec![(wire.clone(), quote!(#wire))],
104        commit: quote!(),
105    })
106}

108pub(crate) fn input_list(
109    carrier: &prebindgen_flat::Wire<JniWire>,
110    sequence: &JniSequence,
111    element: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
112    context: &KotlinContext<'_>,
113) -> Result<KotlinInput> {
114    if *sequence == JniSequence::Bulk {
115        return crate::bulk::input_list(carrier, element, context);
116    }
117    if *sequence != JniSequence::Buffered {
118        return Err(
119            "callback sequence is only supported as the function's successful return".into(),
120        );
121    }
122    let child = input_kotlin(element, context)?;
123    if child.parameters.len() != 1 {
124        return Err("Kotlin collection elements must expose exactly one value; declare a data class for expanded fields".into());
125    }
126    let (parameter, ty) = child.parameters[0]
127        .split_once(": ")
128        .ok_or("invalid element parameter")?;
129    let name = carrier.name.to_string();
130    let state = format!("{name}_elements");
131    let JniWire::Rows { condition, .. } = &carrier.kind else {
132        return Err("sequence requires an object-array control".into());
133    };
134    let mut code = String::new();
135    for binding in &child.setup {
136        code.push_str(&format!("val {} = {}\n", binding.name, binding.expression));
137    }
138    let mut locks = if child.locks.is_empty() {
139        "kotlin.collections.emptyList<Nothing>()".into()
140    } else {
141        format!(
142            "kotlin.collections.listOfNotNull({})",
143            child.locks.join(", ")
144        )
145    };
146    for group in &child.lock_groups {
147        locks = format!("({locks} + ({group}))");
148    }
149    let expression=format!("if ({condition}) {name}?.toList()?.map {{ {parameter} ->\n{code}kotlin.Triple({{ kotlin.arrayOf<kotlin.Any?>({}) }}, {{ {locks} }}, {{ {}; kotlin.Unit }})\n}} else null",child.arguments.join(", "),child.consumed.join("; "));
150    Ok(KotlinInput {
151        parameters: vec![format!(
152            "{name}: kotlin.collections.List<{ty}>{}",
153            if condition == "true" { "" } else { "?" }
154        )],
155        setup: vec![KotlinBinding {
156            name: state.clone(),
157            expression,
158            absent: "null".into(),
159        }],
160        arguments: vec![format!(
161            "{state}?.map {{ it.first() }}?.toTypedArray<kotlin.Any?>()"
162        )],
163        locks: Vec::new(),
164        lock_groups: if child.locks.is_empty() && child.lock_groups.is_empty() {
165            vec![]
166        } else {
167            vec![format!("{state}.orEmpty().flatMap {{ it.second() }}")]
168        },
169        consumed: if child.consumed.is_empty() {
170            vec![]
171        } else {
172            vec![format!("{state}.orEmpty().forEach {{ it.third() }}")]
173        },
174    })
175}

177pub(crate) fn output_list(
178    element: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
179    context: &KotlinContext<'_>,
180) -> Result<KotlinOutput> {
181    let row = format!("{}_row", context.output_prefix);
182    let prefix = format!("{}item_", context.output_prefix);
183    let failure = format!("{}failure", context.output_prefix);
184    let ctx = KotlinContext {
185        failure: &failure,
186        cleanup_failure: Some(&failure),
187        result: &row,
188        output_prefix: &prefix,
189        ..*context
190    };
191    let child = output_kotlin(element, &ctx)?;
192    if child.delivered.len() != 1 {
193        return Err("Kotlin collection elements must deliver exactly one value; declare a data class for expanded outputs".into());
194    }
195    let ty = &child.result_types[0];
196    let value = &child.delivered[0];
197    let raw = format!("({}[0] as kotlin.Array<*>)", context.result);
198    let output = format!("{}list", context.output_prefix);
199    let actions = format!("{}cleanups", context.output_prefix);
200    let initial = format!("{}initialFailure", context.output_prefix);
201    let action = format!("{}cleanupAction", context.output_prefix);
202    let raw_row = format!("{}rawRow", context.output_prefix);
203    let index = format!("{}index", context.output_prefix);
204    let cleanup = format!("{}cleanupError", context.output_prefix);
205    let record_row_failure = kotlin::record_cleanup_failure(&ctx, &cleanup);
206    let record_outer_failure = kotlin::record_cleanup_failure(context, &cleanup);
207    let current_failure = context.cleanup_failure.unwrap_or(context.failure);
208    let run_action = if let Some(target) = context.cleanup_failure {
209        format!("{target} = {action}({target})")
210    } else {
211        format!("{action}({current_failure})")
212    };
213    let finish_row = context
214        .cleanup_failure
215        .map(|target| format!("{target} = {failure}"))
216        .unwrap_or_default();
217    Ok(KotlinOutput {
218        result_names: vec![None],
219        delivered: vec![format!("requireNotNull({output})")],
220        result_types: vec![format!("kotlin.collections.List<{ty}>")],
221        output_locals: format!("var {output}: kotlin.collections.MutableList<{ty}>? = null\nvar {actions}: kotlin.collections.MutableList<(kotlin.Throwable?) -> kotlin.Throwable?>? = null\n"),
222        materialize: format!(
223            "{actions} = kotlin.collections.mutableListOf()\n{output} = kotlin.collections.mutableListOf()\nfor ({raw_row} in {raw}) {{\nval {row} = {raw_row} as kotlin.Array<*>\n{}\n{actions}.add {{ {initial} -> var {failure} = {initial}; try {{ {} }} catch ({cleanup}: Throwable) {{ {record_row_failure} }}; {failure} }}\n{}\n{output}.add({value})\n}}\n",
224            child.output_locals, child.cleanup, child.materialize
225        ),
226        cleanup: format!(
227            "try {{\nfor ({action} in {actions}.orEmpty()) {{ try {{ {run_action} }} catch ({cleanup}: Throwable) {{ {record_outer_failure} }} }}\nfor ({index} in ({actions}?.size ?: 0) until {raw}.size) {{\nvar {failure}: kotlin.Throwable? = {current_failure}\ntry {{ val {row} = {raw}[{index}] as kotlin.Array<*>\n{}\n{}\n}} catch ({cleanup}: Throwable) {{ {record_row_failure} }}\n{finish_row}\n}}\n}} catch ({cleanup}: Throwable) {{ {record_outer_failure} }}\n",
228            child.output_locals, child.cleanup
229        ),
230    })
231}

233/// The adapter selects its native scope before looking for a sequence.
234#[allow(clippy::type_complexity)]
235pub(crate) fn return_sequence(
236    layout: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
237) -> Option<(
238    &prebindgen_flat::Wire<JniWire>,
239    &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
240    &JniSequence,
241)> {
242    outer_sequences(layout)
243        .into_iter()
244        .next()
245        .and_then(|(element, controls, metadata)| match controls {
246            [carrier] => Some((carrier, element, metadata)),
247            _ => None,
248        })
249}
250pub(crate) fn callback_target(
251    layout: &prebindgen_flat::WireNode<JniWire, crate::JniChain, crate::JniBranch>,
252) -> Result<syn::Ident> {
253    if outer_sequences(layout).len() != 1 || crate::group::result_packet_fields(layout).len() > 2 {
254        return Err(
255            "callback_return permits only one sequence and its optional presence flag; cannot discard additional return values".into(),
256        );
257    }
258    return_sequence(layout)
259        .map(|(carrier, _, _)| carrier.name.clone())
260        .ok_or_else(|| "callback_return requires a sequence boundary".into())
261}

263/// Whether the successful return uses the function's sequence callback.
264pub(crate) fn callback_delivery(
265    node: &prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
266) -> Result<bool> {
267    let sequences = outer_sequences(node);
268    if !sequences
269        .iter()
270        .any(|(_, _, metadata)| **metadata == JniSequence::Callback)
271    {
272        return Ok(false);
273    }
274    callback_target(node)?;
275    if crate::group::result_packet_fields(node).len() != 1 {
276        return Err("callback_return cannot discard additional return values".into());
277    }
278    Ok(true)
279}

281/// Function sequence callbacks are not available in parameters, error channels or rows.
282pub(crate) fn validate_buffered(
283    node: &prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
284) -> Result<()> {
285    match node {
286        prebindgen_flat::WireNode::Leaf(_) => {}
287        prebindgen_flat::WireNode::Chain {
288            child: element,
289            meta: JniChain::Sequence(metadata),
290            ..
291        } => {
292            if *metadata == JniSequence::Callback {
293                return Err(
294                    "callback sequence is only supported as the function's successful return"
295                        .into(),
296                );
297            }
298            validate_buffered(element)?;
299        }
300        prebindgen_flat::WireNode::Chain {
301            meta: JniChain::Optional(_) | JniChain::Bulk(_),
302            child: element,
303            ..
304        } => validate_buffered(element)?,
305        prebindgen_flat::WireNode::Branch {
306            meta: JniBranch::Group(_),
307            children: elements,
308            ..
309        }
310        | prebindgen_flat::WireNode::Branch {
311            meta: JniBranch::Alternatives,
312            children: elements,
313            ..
314        } => {
315            for element in elements {
316                validate_buffered(element)?;
317            }
318        }
319    }
320    Ok(())
321}

323// A sequence owns a separate row scope: find the outer occurrences only.
324#[allow(clippy::type_complexity)] // Borrow the element schema, controls and delivery policy together.
325fn outer_sequences(
326    node: &prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
327) -> Vec<(
328    &prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
329    &[prebindgen_flat::Wire<JniWire>],
330    &JniSequence,
331)> {
332    fn visit<'a>(
333        node: &'a prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
334        out: &mut Vec<(
335            &'a prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
336            &'a [prebindgen_flat::Wire<JniWire>],
337            &'a JniSequence,
338        )>,
339    ) {
340        use prebindgen_flat::WireNode;
341        match node {
342            WireNode::Chain {
343                meta: JniChain::Sequence(meta),
344                child,
345                controls,
346            } => out.push((child, controls, meta)),
347            WireNode::Chain { child, .. } => visit(child, out),
348            WireNode::Branch { children, .. } => {
349                for child in children {
350                    visit(child, out)
351                }
352            }
353            WireNode::Leaf(_) => {}
354        }
355    }
356    let mut out = vec![];
357    visit(node, &mut out);
358    out
359}

361#[cfg(test)]
362mod delivery_tests {
363    use super::*;
364    use prebindgen_flat::WireNode;
365    fn leaf() -> prebindgen_flat::WireNode<JniWire, JniChain, JniBranch> {
366        WireNode::Leaf(vec![prebindgen_flat::Wire {
367            name: syn::parse_quote!(element),
368            kind: JniWire::scalar(Primitive::I32),
369        }])
370    }
371    fn sequence(
372        delivery: JniSequence,
373        name: &str,
374        element: prebindgen_flat::WireNode<JniWire, JniChain, JniBranch>,
375    ) -> prebindgen_flat::WireNode<JniWire, JniChain, JniBranch> {
376        let kind = match delivery {
377            JniSequence::Bulk => JniWire::NativeRows {
378                condition: "true".into(),
379            },
380            JniSequence::Buffered | JniSequence::Callback => JniWire::Rows {
381                condition: "true".into(),
382                null_niche_available: true,
383            },
384        };
385        WireNode::Chain {
386            controls: vec![prebindgen_flat::Wire {
387                name: syn::parse_str(name).unwrap(),
388                kind,
389            }],
390            child: Box::new(element),
391            meta: JniChain::Sequence(delivery),
392        }
393    }
394    #[test]
395    fn delivery_is_explicit_and_callback_scope_is_checked() {
396        let visitor = sequence(JniSequence::Callback, "visitor", leaf());
397        assert!(callback_delivery(&visitor).unwrap());
398        assert!(validate_buffered(&visitor)
399            .unwrap_err()
400            .contains("successful return"));
401        let packed = sequence(JniSequence::Buffered, "rows", leaf());
402        assert!(!callback_delivery(&packed).unwrap());
403        validate_buffered(&packed).unwrap();
404        let nested = sequence(JniSequence::Buffered, "rows", visitor.clone());
405        assert!(crate::group::validate_layout(&nested)
406            .unwrap_err()
407            .contains("successful return"));
408        let extra = WireNode::Branch {
409            controls: vec![],
410            children: vec![visitor, leaf()],
411            meta: JniBranch::Group(JniGroup::Values),
412        };
413        assert!(callback_delivery(&extra)
414            .unwrap_err()
415            .contains("additional return"));
416    }
