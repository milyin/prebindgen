1//! #52: idiomatic Kotlin overloads for expanded params.
2//!
3//! A multi-variant `expand_param!(T)` crosses the wire as a selector tuple
4//! (`expectedSel: Int, expected00: Long?, …`); the raw call site passes magic
5//! ints and null-padding. Two mechanisms turn that into idiomatic Kotlin:
6//!
7//! * **Proactive splittability check** ([`Declarations::validate_split_declarations`]):
8//!   every multi-variant `expand_param!` declaration (type-level or per-fn) is
9//!   verified up front to be *splittable* — its arms surface as pairwise-distinct
10//!   JVM signatures — so a function can safely request overloads. A collision is
11//!   a hard build error attributed to the declaration; `.no_split()` opts out.
12//! * **Per-function emission** ([`render_param_overloads`], driven by
13//!   [`FunctionDecl::split_on_param`](prebindgen_registry::fun)): for the named split params the
14//!   generator emits, alongside the selector wrapper, the **cartesian product** of
15//!   their arms as typed overloads, each delegating to the selector form. The
16//!   concrete product must have no two combinations sharing a JVM signature.
17//!
18//! Only flat arms are splittable; an explicit `.split_on_param` on a
19//! recursively-built / single-variant / unknown parameter is a hard error.
20//!
21//! **Optional params (nullable-arm rule)** — an `Option<…>` parameter is
22//! splittable through its **single-leaf** arms only (the identity arm, or a
23//! build arm with exactly one constructor input): the overload keeps the arm's
24//! nullable type (`T?`) and `null` selects absence (selector `-1`), mirroring
25//! the "one nullable leaf decides presence" convention of optional single-ctor
26//! expansion. Multi-leaf arms have no nullable single type and stay
27//! selector-only; an optional param with no single-leaf arm is a hard error.
28//! With two qualifying nullable arms a bare `null` argument is ambiguous at
29//! Kotlin call sites (both mean absent) and needs a cast — signatures are
30//! still pairwise-distinct per the checks above.

42impl Declarations {
43    /// Proactively verify every multi-variant `expand_param!` declaration is
44    /// splittable (its arms have pairwise-distinct JVM-erased signatures), so
45    /// [`FunctionDecl::split_on_param`](prebindgen_registry::fun) can emit unambiguous
46    /// overloads. Runs regardless of whether any function actually splits the
47    /// type, so authorship errors surface early. `.no_split()` opts a decl out.
48    /// A collision errors with a message attributed to the declaration
49    /// (surfaced through the [`validate_resolved`] boundary before any
50    /// artifact is written).
51    ///
52    /// [`validate_resolved`]: prebindgen_registry::Prebindgen::validate_resolved
53    pub(crate) fn validate_split_declarations(&self, registry: &Registry) -> Result<(), String> {
54        let type_level = self
55            .param_expand_decls
56            .iter()
57            .map(|d| (d.key().as_str().to_string(), d));
58        let per_fn = self
59            .fn_param_expands
60            .iter()
61            .map(|(func, param, d)| (format!("fun `{func}` param `{param}`"), d));
62        for (site, decl) in type_level.chain(per_fn) {
63            if decl.is_no_split() || decl.variants().len() < 2 {
64                continue;
65            }
66            let target = decl.rust_type().key();
67            let sigs: Vec<(String, Vec<ErasedJvmType>)> = decl
68                .variants()
69                .iter()
70                .map(|v| {
71                    let ctor = match v {
72                        LocalVariant::Ctor(c) => Some(c),
73                        LocalVariant::SelfIdentity => None,
74                    };
75                    (
76                        ctor.map(|c| c.to_string())
77                            .unwrap_or_else(|| "variant_self()".to_string()),
78                        {
79                            // Validation sees every target and holds the model;
80                            // freezing the reading here is what lets the render
81                            // path answer without one (#613 step 7).
82                            if let Some(reading) = registry.reading(&target) {
83                                self.freeze_reading(&reading);
84                            }
85                            arm_erased_sig(self, registry.flat(), &target, ctor)
86                        },
87                    )
88                })
89                .collect();
90            for i in 0..sigs.len() {
91                for j in (i + 1)..sigs.len() {
92                    if sigs[i].1 == sigs[j].1 {
93                        return Err(format!(
94                            "expand_param!({t}) [{site}]: variants {a} and {b} both surface as \
95                             `({sig})` — a split would emit two overloads with the same JVM \
96                             signature; disambiguate the constructors or add .no_split()",
97                            t = decl.key().as_str(),
98                            a = sigs[i].0,
99                            b = sigs[j].0,
100                            sig = sigs[i]
101                                .1
102                                .iter()
103                                .map(|e| e.to_string())
104                                .collect::<Vec<_>>()
105                                .join(", "),
106                        ));
107                    }
108                }
109            }
110        }
111        Ok(())
112    }
113}

115/// The JVM-erased type list of one arm: the constructor's parameter types
116/// (build arm), or the single target type (`variant_self`, `ctor == None`).
117/// Uses the shared [`erase_kt_type`] model (issue #89 stage 2) so the split
118/// ambiguity check and the whole-artifact overload table agree on erasure.
119fn arm_erased_sig(
120    ext: &Declarations,
121    flat: &prebindgen_registry::flat::Flat,
122    target: &TypeKey,
123    ctor: Option<&syn::Ident>,
124) -> Vec<ErasedJvmType> {
125    match ctor {
126        Some(cf) => match flat.function(&cf) {
127            Some(f) => f
128                .params
129                .iter()
130                .map(|p| rust_type_erased(ext, &p.ty))
131                .collect(),
132            None => Vec::new(),
133        },
134        // The identity, because the two callers reach it differently: one holds
135        // a plan's reading, the other a `.expand_param(...)` DECLARATION, which
136        // the build script wrote and the model may never have interned. Where
137        // it did, the reading answers as before; where it did not, the erased
138        // form is the identity's own canonical spelling — which is all a
139        // declaration has to be told apart by.
140        // The reading, frozen where validation saw it — the render path has no
141        // model to ask (#613 step 7). A target the freeze never reached falls
142        // back to the identity's own canonical spelling, which is the same
143        // answer `reading` gave for an un-interned declaration.
144        None => vec![match ext.frozen_reading(target.as_str()) {
145            Some(reading) => rust_type_erased(ext, &reading),
146            None => ErasedJvmType::raw(target.as_str().to_string()),
147        }],
148    }
149}

151/// The [`ErasedJvmType`] a Rust arm type surfaces as: map it to its Kotlin
152/// surface type (a declared class's FQN, else the resolved converter's Kotlin
153/// name) and run the shared [`erase_kt_type`]; a plain class folds to its FQN
154/// there. Falls back to the token string for a
155/// type with no resolved surface. References are peeled first (`&T` erases
156/// like `T`).
157fn rust_type_erased(ext: &Declarations, ty: &prebindgen_registry::flat::TypeRef) -> ErasedJvmType {
158    // The spelling's own outermost borrow, as `TypeKind::Ref` — not
159    // `borrow_target`, which reaches through the erased wrappers (#S31).
160    let peeled = match ty.kind() {
161        prebindgen_registry::flat::TypeKind::Ref { inner, .. } => inner,
162        _ => ty,
163    };
164    let key = peeled.key();
165    if ext.types.get(&key).is_some_and(|c| c.name_spec.is_some()) {
166        if let Some(fqn) = ext.kotlin_fqn(&key) {
167            return erase_kt_type(&[], &KtType::cls(fqn));
168        }
169    }
170    if let Some(kt) = ext
171        .in_frag(peeled)
172        .and_then(|e| e.metadata.kotlin_name.clone())
173    {
174        return erase_kt_type(&[], &kt);
175    }
176    // The type's IDENTITY, not its spelling. `ErasedJvmType` is a string
177    // compared with `==` — "two parameters collide as overloads iff their
178    // tokens are equal" — so keying it on what the source wrote meant two
179    // spellings of one type (`Box<Payload>` and `Payload`, `crate::Foo` and
180    // `Foo`) erased to different tokens and were judged NOT to collide. The
181    // JVM would then reject the emitted class for a clash this check exists
182    // to catch first. A `TypeKey` is canonical, so equal types compare equal.
183    ErasedJvmType::raw(peeled.key().as_str())
184}

186/// Whether `plan` is a multi-variant expansion that can be turned into
187/// overloads: a selector (≥2 arms) and no recursively-built arm. An `Option`
188/// outer shape is in scope — its arms are filtered to single-leaf ones by
189/// [`resolve_split`] (nullable-arm rule, see module docs).
190fn plan_in_scope(plan: &FoldPlan) -> bool {
191    plan.selector.is_some()
192        && !plan
193            .variants
194            .iter()
195            .any(|v| v.inputs.iter().any(|a| matches!(a, FoldArg::Build(_))))
196}

198/// One resolved split parameter of a function, positioned against the rendered
199/// selector wrapper.
200struct Split<'a> {
201    /// The original Rust parameter ident (e.g. `expected`).
202    param: syn::Ident,
203    plan: &'a FoldPlan,
204    /// Start of this param's contiguous leaf block in `sel_fun.params`.
205    start: usize,
206    /// Block length (`plan.leaves.len()`).
207    len: usize,
208    /// Selector-leaf index within the block.
209    sel_idx: usize,
210    /// `Option<…>` parameter — overload arms keep their nullable type and the
211    /// delegated selector is conditional (`null` = absent = `-1`).
212    optional: bool,
213    /// Overloadable arms: original variant index (the selector value) plus the
214    /// arm's typed params as `(param, leaf-index-within-block)`. For an
215    /// optional param this is the single-leaf subset of the variants.
216    arms: Vec<(usize, Vec<(KtParam, usize)>)>,
217}

220/// The Kotlin names of a constructor's parameters.

225/// already required.
226fn ctor_param_names(f: &prebindgen_registry::flat::Function) -> Vec<String> {
227    f.params
228        .iter()
229        .map(|p| kt_param_name(&p.name.to_string()))
230        .collect()

233/// `origin` + Capitalized `name` (`primary` + `count` → `primaryCount`).
234fn prefixed(origin: &str, name: &str) -> String {
235    let mut c = name.chars();
236    match c.next() {
237        Some(first) => format!("{origin}{}{}", first.to_uppercase(), c.as_str()),
238        None => origin.to_string(),
239    }

242/// Clear selector-dispatch nullability (`T?` → `T`). Genuinely optional
243/// constructor parameters bypass this helper and retain their rendered `T?`.
244fn non_null(mut ty: KtType) -> KtType {
245    match &mut ty {
246        KtType::Named { nullable, .. } | KtType::Function { nullable, .. } => *nullable = false,
247    }
248    ty
249}

251/// The typed overload params of one variant arm, paired with the leaf index
252/// each fills. `origin`/`multi` drive name disambiguation (build-arm params are
253/// prefixed with the origin parameter name when the function splits more than
254/// one parameter). `optional_plan` keeps every slot's rendered nullability —
255/// for an `Option<…>` parameter `null` encodes absence (nullable-arm rule).
256/// Returns `None` if any input is not a flat leaf.
257fn variant_typed_params(
258    registry: &impl Conversions,
259    variant: &prebindgen_registry::expand::FoldVariant,
260    origin: &syn::Ident,
261    block: &[KtParam],
262    multi: bool,
263    optional_plan: bool,
264) -> Option<Vec<(KtParam, usize)>> {
265    let origin_kt = kt_param_name(&origin.to_string());
266    let (names, optional): (Vec<String>, Vec<bool>) = match &variant.ctor {
267        Some(cf) => {
268            let f = registry.flat().function(&cf)?;
269            // `Optional` off the kind, not `is_option` off a path: the same
270            // question, asked of the grammar the source wrote.
271            let optional = f
272                .params
273                .iter()
274                .map(|p| {
275                    matches!(
276                        p.ty.kind(),
277                        prebindgen_registry::flat::TypeKind::Optional(_)
278                    )
279                })
280                .collect();
281            (ctor_param_names(f), optional)
282        }
283        // Identity arm: one parameter, the value itself, named after the origin
284        // parameter (already unique across split params — never prefixed).
285        None => (vec![origin_kt.clone()], vec![false]),
286    };
287    let mut out = Vec::new();
288    for (m, arg) in variant.inputs.iter().enumerate() {
289        let FoldArg::Leaf(idx, _) = arg else {
290            return None;
291        };
292        let slot = block.get(*idx)?;
293        let base = names.get(m).cloned().unwrap_or_else(|| slot.name.clone());
294        let name = if multi && variant.ctor.is_some() {
295            prefixed(&origin_kt, &base)
296        } else {
297            base
298        };
299        let ty = if optional_plan || optional.get(m).copied().unwrap_or(false) {
300            slot.ty.clone()
301        } else {
302            non_null(slot.ty.clone())
303        };
304        out.push((KtParam::new(&name, ty), *idx));
305    }
306    Some(out)
307}

309/// Locate a param's contiguous leaf block in the selector wrapper's parameter
310/// list, matching by leaf name in order.
311fn find_block(params: &[KtParam], leaf_names: &[String]) -> Option<usize> {
312    if leaf_names.is_empty() || params.len() < leaf_names.len() {
313        return None;
314    }
315    (0..=params.len() - leaf_names.len()).find(|&s| {
316        params[s..s + leaf_names.len()]
317            .iter()
318            .zip(leaf_names)
319            .all(|(p, n)| &p.name == n)
320    })
321}

323/// Resolve one `.split_on_param` request against the rendered selector wrapper,
324/// validating the parameter is expandable, multi-variant, and in-scope (all
325/// hard errors — the user explicitly asked to split it).
326fn resolve_split<'a>(
327    registry: &'a Registry,
328    fplan: &'a JniFunctionPlan,
329    f: &prebindgen_registry::flat::Function,
330    sel_fun: &KtFun,
331    param_name: &str,
332    multi: bool,
333) -> Split<'a> {
334    let param = syn::Ident::new(param_name, Span::call_site());
335    let plan = fplan
336        .params
337        .iter()
338        .find(|planned| planned.ident == param)
339        .and_then(|planned| match &planned.form {
340            ParamForm::Expanded { plan, .. } => Some(plan.fold()),
341            ParamForm::Single(_) => None,
342        })
343        .unwrap_or_else(|| {
344            panic!(
345                "fun!({}).split_on_param(\"{param_name}\"): `{param_name}` is not an expandable \
346                 parameter (it has no `expand_param!` variants)",
347                f.name
348            )
349        });
350    assert!(
351        plan.selector.is_some(),
352        "fun!({}).split_on_param(\"{param_name}\"): `{param_name}` has a single variant — there \
353         is nothing to split (it already flattens to one signature)",
354        f.name
355    );
356    assert!(
357        plan_in_scope(plan),
358        "fun!({}).split_on_param(\"{param_name}\"): `{param_name}` has a recursively-built arm — \
359         it cannot be overloaded; keep the selector form",
360        f.name
361    );
362    let leaf_names: Vec<String> = plan
363        .leaves
364        .iter()
365        .map(|l| kt_param_name(&l.name.to_string()))
366        .collect();
367    let len = leaf_names.len();
368    let start = find_block(&sel_fun.params, &leaf_names).unwrap_or_else(|| {
369        panic!(
370            "fun!({}).split_on_param(\"{param_name}\"): could not locate the parameter's leaf \
371             block in the generated wrapper",
372            f.name
373        )
374    });
375    let block = &sel_fun.params[start..start + len];
376    let sel_idx = plan.selector.expect("selector present");
377    // Nullable-arm rule: an `Option<…>` param is overloadable through its
378    // single-leaf arms only — the arm's one nullable param doubles as the
379    // presence flag (`null` = absent). Multi-leaf arms stay selector-only.
380    let optional = plan.produces_option();
381    let arms: Vec<(usize, Vec<(KtParam, usize)>)> = plan
382        .variants
383        .iter()
384        .enumerate()
385        .filter(|(_, v)| !optional || v.inputs.len() == 1)
386        .map(|(vi, v)| {
387            let typed = variant_typed_params(registry, v, &param, block, multi, optional)
388                .unwrap_or_else(|| {
389                    panic!(
390                        "fun!({}).split_on_param(\"{param_name}\"): an arm has a non-flat input; \
391                         it cannot be overloaded",
392                        f.name
393                    )
394                });
395            (vi, typed)
396        })
397        .collect();
398    assert!(
399        !arms.is_empty(),
400        "fun!({}).split_on_param(\"{param_name}\"): `{param_name}` is an `Option<_>` parameter \
401         and none of its arms is a single leaf — its overload has no clean nullable type; keep \
402         the selector form",
403        f.name
404    );
405    Split {
406        param,
407        plan,
408        start,
409        len,
410        sel_idx,
411        optional,
412        arms,
413    }
414}

416/// The overloads for one function, delegating to its already-rendered selector
417/// wrapper `sel_fun`. Empty unless the function has `.split_on_param` requests.
418/// Emits the cartesian product of the named params' arms; panics (a build
419/// error) if the product has two combinations with the same JVM signature.
420pub(crate) fn render_param_overloads(
421    ext: &Declarations,
422    f: &prebindgen_registry::flat::Function,
423    registry: &Registry,
424    sel_fun: &KtFun,
425) -> Vec<KtFun> {
426    // Requested split params for this function, in signature order.
427    let requested: Vec<String> = {
428        let want: std::collections::HashSet<&str> = ext
429            .fn_split_params
430            .iter()
431            .filter(|(func, _)| func == &f.name)
432            .map(|(_, p)| p.as_str())
433            .collect();
434        if want.is_empty() {
435            return Vec::new();
436        }
437        f.params
438            .iter()
439            .filter(|p| want.contains(p.name.to_string().as_str()))
440            .map(|p| p.name.to_string())
441            .collect()
442    };
443    // Any requested name that didn't match a real parameter is a typo — surface
444    // it rather than silently dropping.
445    for (func, p) in &ext.fn_split_params {
446        if func == &f.name && !requested.iter().any(|r| r == p) {
447            panic!(
448                "fun!({}).split_on_param(\"{p}\"): no parameter named `{p}` on this function",
449                f.name
450            );
451        }
452    }
453
454    let fplan = ext.fn_plan(registry, f).unwrap_or_else(|_| {
455        panic!(
456            "fun!({}): its frozen JNI generation plan is unavailable",
457            f.name
458        )
459    });
460    let multi = requested.len() > 1;
461    let splits: Vec<Split> = requested
462        .iter()
463        .map(|name| resolve_split(registry, &fplan, f, sel_fun, name, multi))
464        .collect();
465
466    // Cartesian product of arm indices across all split params.
467    let combos = cartesian(&splits.iter().map(|s| s.arms.len()).collect::<Vec<_>>());
468
469    // Product-global JVM-signature collision check (fixed params are identical
470    // across every overload, so only the split-arm lists can collide).
471    let sigs: Vec<Vec<ErasedJvmType>> = combos
472        .iter()
473        .map(|combo| {
474            splits
475                .iter()
476                .zip(combo)
477                .flat_map(|(s, &ai)| {
478                    let ctor = s.plan.variants[s.arms[ai].0].ctor.as_ref();
479                    arm_erased_sig(ext, registry.flat(), &s.plan.target.key(), ctor)
480                })
481                .collect()
482        })
483        .collect();
484    for i in 0..sigs.len() {
485        for j in (i + 1)..sigs.len() {
486            if sigs[i] == sigs[j] {
487                panic!(
488                    "fun!({}): split_on_param product is ambiguous — combinations {} and {} both \
489                     surface as `({})`; add .no_split() intent is not enough here, disambiguate \
490                     the constructors or drop one .split_on_param",
491                    f.name,
492                    combo_label(&splits, &combos[i]),
493                    combo_label(&splits, &combos[j]),
494                    sigs[i]
495                        .iter()
496                        .map(|e| e.to_string())
497                        .collect::<Vec<_>>()
498                        .join(", "),
499                );
500            }
501        }
502    }
503
504    // Emit one overload per product combination.
505    let n = sel_fun.params.len();
506    let mut out = Vec::with_capacity(combos.len());
507    for combo in &combos {
508        // Per-split delegation slots, keyed by block start.
509        let mut params: Vec<KtParam> = Vec::new();
510        let mut call_args: Vec<String> = Vec::new();
511        let mut pos = 0usize;
512        while pos < n {
513            if let Some((si, s)) = splits.iter().enumerate().find(|(_, s)| s.start == pos) {
514                // Replace this param's whole leaf block with the chosen arm's
515                // typed params; fill the block's delegation slots.
516                let (vi, typed) = &s.arms[combo[si]];
517                let mut leaf_arg: Vec<String> = vec!["null".to_string(); s.len];
518                leaf_arg[s.sel_idx] = if s.optional {
519                    // Nullable-arm rule: the arm's single param doubles as the
520                    // presence flag — `null` delegates absence (selector -1).
521                    format!("if ({} != null) {vi} else -1", typed[0].0.name)
522                } else {
523                    vi.to_string()
524                };
525                for (p, lidx) in typed {
526                    params.push(p.clone());
527                    leaf_arg[*lidx] = p.name.clone();
528                }
529                call_args.extend(leaf_arg);
530                pos += s.len;
531            } else {
532                // A fixed (or non-split expanded) param — passes through.
533                let p = &sel_fun.params[pos];
534                params.push(p.clone());
535                call_args.push(p.name.clone());
536                pos += 1;
537            }
538        }
539
540        // Guard against a param-name clash (e.g. an arm name colliding with a
541        // fixed param) rather than emitting uncompilable Kotlin.
542        let mut seen = std::collections::HashSet::new();
543        for p in &params {
544            assert!(
545                seen.insert(p.name.clone()),
546                "fun!({}): split overload has a duplicate parameter name `{}` — rename the \
547                 constructor parameter",
548                f.name,
549                p.name
550            );
551        }
552
553        out.push(overload_shell(
554            sel_fun,
555            params,
556            KtCode::new().line(format!("{}({})", sel_fun.name, call_args.join(", "))),
557        ));
558    }
559    out
560}

562/// An overload derived from the selector wrapper by a signature-preserving
563/// transform: generics, annotations, modifiers, visibility, and return type
564/// carry over unchanged (a wrapper generic over `R`/`A` — builder/fold
565/// delivery — keeps its `fun <R> …` declaration, #87); only the parameter
566/// list and the delegating body are replaced. Kdoc stays on the selector
567/// form — overloads are bare.
568fn overload_shell(sel_fun: &KtFun, params: Vec<KtParam>, body: KtCode) -> KtFun {
569    let mut ov = sel_fun.clone();
570    ov.params = params;
571    ov.kdoc = None;
572    ov.body = KtBody::Expr(body);
573    ov
574}

576/// Cartesian product of index ranges `0..counts[k]`, as a list of index
577/// tuples. `[2, 2]` → `[[0,0],[0,1],[1,0],[1,1]]`.
578fn cartesian(counts: &[usize]) -> Vec<Vec<usize>> {
579    let mut acc = vec![Vec::new()];
580    for &c in counts {
581        acc = acc
582            .into_iter()
583            .flat_map(|prefix| {
584                (0..c).map(move |i| {
585                    let mut next = prefix.clone();
586                    next.push(i);
587                    next
588                })
589            })
590            .collect();
591    }
592    acc
593}

595/// A `param=variant` label for one product combination, for error messages.
596fn combo_label(splits: &[Split], combo: &[usize]) -> String {
597    let parts: Vec<String> = splits
598        .iter()
599        .zip(combo)
600        .map(|(s, &ai)| {
601            let v = match &s.plan.variants[s.arms[ai].0].ctor {
602                Some(c) => c.to_string(),
603                None => "variant_self()".to_string(),
604            };
605            format!("{}={v}", s.param)
606        })
607        .collect();
608    format!("({})", parts.join(", "))
609}
