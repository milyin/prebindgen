1//! The walk over a decomposed value.
2//!
3//! A decomposition — an [`UnfoldPlan`](crate::unfold::UnfoldPlan) — says which
4//! **leaves** a value is delivered as and which **hoists** (intermediate value
5//! forms) they are reached through. Performing that walk is the registry's:
6//! binding each hoist once, reaching each leaf through its own path, and
7//! deciding at every step whether what it holds is owned or borrowed.
8//!
9//! What an adapter supplies is the target-language half — how a leaf is
10//! encoded, and how the delivered values are handed over — plus the two facts
11//! this module cannot know: where a source item is qualified from
12//! (`qualify`), and what each leaf is ([`DecomposedLeaf`]).
13
14use prebindgen_flat::flat::TypeRef;
15use proc_macro2::TokenStream;
16use quote::{format_ident, quote};
17
18use crate::unfold::{steps_are_movable, PathStep};

20/// What the walk needs to know about one delivered leaf.
21///
22/// The adapter's own leaf carries far more — its wire type, its JNI or C
23/// encoding, its projection — and none of that reaches here.
24pub trait DecomposedLeaf {
25    /// Steps from the decomposed value to this leaf.
26    fn reach(&self) -> &[PathStep];
27
28    /// The leaf's own name, used in diagnostics and in local names.
29    fn name(&self) -> &str;
30
31    /// Whether this leaf *is* the decomposed value rather than a part of it.
32    fn identity(&self) -> bool;
33
34    /// The reading the leaf delivers.
35    fn source(&self) -> &TypeRef;
36
37    /// Whether this leaf is a synthesized **selector** — a value naming which
38    /// group of the leaves after it is live. It chooses between groups rather
39    /// than joining one, so its own [`Self::group`] is `None`.
40    ///
41    /// Two kinds select: a sum's tag, which names the live alternative, and an
42    /// optional value's presence, which says whether its one group carries
43    /// anything.
44    fn selects(&self) -> bool;
45
46    /// The group this leaf belongs to — an alternative's tag, or `0` for the
47    /// one group a presence flag gates. `None` for a leaf that is always live,
48    /// and for a selector.
49    ///
50    /// A group's leaves may not include a selector: see
51    /// [`UnfoldLeaf::group`](crate::unfold::UnfoldLeaf::group) for why one
52    /// `Option<i32>` cannot express nesting, and what a decomposition does
53    /// instead.
54    fn group(&self) -> Option<i32>;
55
56    /// Whether the last step reaching it is a field read rather than a call.
57    ///
58    /// Derived from [`Self::reach`], so no implementation can disagree with
59    /// its own path — and the answer decides whether the reached place is
60    /// cloned, which is an ownership decision rather than a spelling.
61    fn is_field_read(&self) -> bool {
62        matches!(self.reach().last(), Some(PathStep::Field { .. }))
63    }
64}

66/// One delivered value's slot, as the walk needs to see it: a slot exists for
67/// every leaf whether or not the value behind it does, so a gate that skips an
68/// encode still has to declare the slot and fill it.
69pub struct Slot {
70    /// The slot's type.
71    pub ty: TokenStream,
72    /// What an unfilled slot carries.
73    pub default: TokenStream,
74}

76/// A leaf's reach, rendered **around** an encoding body: the walk hands the
77/// body the reached Rust expression and gets back the encoded value, so an
78/// absent value short-circuits to the adapter's own absence rather than to one
79/// the walk chose. `dyn` on both halves because each side is written where it
80/// is known and neither can name the other's closure.
81pub type Reach<'a> = dyn Fn(&dyn Fn(TokenStream) -> TokenStream) -> TokenStream + 'a;

83/// The adapter's half of a delivery.
84///
85/// The walk in this module owns how a decomposed value is reached — which
86/// hoist a leaf sits under, what is owned, what is borrowed. What a leaf
87/// becomes in the target language, and how the delivered values are handed to
88/// the call, is the adapter's, and this is where it says so. Stated over
89/// [`DecomposedLeaf`] rather than over an adapter's own leaf type, so the walk
90/// can call it.
91pub trait DeliveryBridge {
92    /// The adapter's leaf.
93    type Leaf: DecomposedLeaf;
94
95    /// Where the source item declaring `ident` is qualified from — the one
96    /// fact about a captured path this crate cannot answer for itself.
97    fn qualify(&self, ident: &syn::Ident) -> syn::Path;
98
99    /// The sum a selector leaf names: the qualified source path of the type,
100    /// and its Flat shape.
101    fn sum(&self, leaf: &Self::Leaf) -> (syn::Path, &prebindgen_flat::flat::Variant);
102
103    /// Encode one leaf into the value the delivery call receives, and bind
104    /// that value to `slot`.
105    ///
106    /// The adapter emits the binding rather than the walk, because the slot's
107    /// type is the adapter's: one leaf may cross as a scalar and the next as a
108    /// reference. `index` is the leaf's position in the delivery, which is
109    /// what any temporary the encode needs is named from. An encoding that can
110    /// fail routes its failure through `fail`.
111    fn encode(
112        &self,
113        leaf: &Self::Leaf,
114        index: usize,
115        slot: &syn::Ident,
116        reach: &Reach<'_>,
117        fail: &dyn Fn(TokenStream) -> TokenStream,
118        emit: &crate::RustWriter,
119    ) -> TokenStream;
120
121    /// The expression that passes `slot` as one argument of the delivery call.
122    fn argument(&self, leaf: &Self::Leaf, slot: &syn::Ident) -> TokenStream;
123
124    /// What one leaf's slot holds when the walk fills it without running the
125    /// leaf's encode — the `None` arm of a gate, where a slot exists but the
126    /// value behind it does not.
127    fn slot(&self, leaf: &Self::Leaf) -> Slot;
128
129    /// What a leaf delivers when an optional step on the way to it found
130    /// nothing. [`reach_leaf`] gates on this rather than on a value of its
131    /// own: absence is a target-language representation, and this crate has
132    /// none.
133    fn absent(&self) -> TokenStream;
134}

136/// Compose one [`PathStep`] onto the reference expression reached so far.
137/// A `Call` applies its accessor (origin-qualified); a `Field` reads the field
138/// and re-borrows, so the result is a reference either way and steps chain
139/// uniformly.
140pub fn compose_step(
141    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
142    step: &PathStep,
143    e: TokenStream,
144) -> TokenStream {
145    match step {
146        PathStep::Call { ident, .. } => {
147            let m = qualify(ident);
148            quote!(#m::#ident(#e))
149        }
150        PathStep::Field { ident, .. } => quote!(&(#e).#ident),
151    }
152}
153/// Fold a run of steps onto `e`, borrowing wherever ownership demands it.
154///
155/// [`compose_step`] hands a `Call` its receiver as written, and an accessor
156/// takes that receiver **by reference** — so an owned value in hand has to be
157/// borrowed before the next call composes onto it. A value is in hand whenever
158/// the previous step returned one (`f(..) -> T` rather than `-> &T`), which is
159/// what [`PathStep::yields_owned`] records; `owned` says whether `e` itself
160/// started that way.
161///
162/// A `Field` step needs no borrow either way — it composes as `&(e).f`, which
163/// reads through a value and a reference alike.
164///
165/// This is the ONE place the rule lives, so every fold — a leaf's reach, a
166/// conditional hoist's prefix, a sum's matched value — answers it the same way.
167/// Splitting it produced exactly the bug it exists to prevent: ownership was
168/// handled at the optional binding and at the value form, and lost at every
169/// ordinary call in between.
170pub fn fold_steps(
171    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
172    steps: &[PathStep],
173    mut e: TokenStream,
174    mut owned: bool,
175) -> TokenStream {
176    for step in steps {
177        if owned && matches!(step, PathStep::Call { .. }) {
178            e = quote!(&#e);
179        }
180        e = compose_step(qualify, step, e);
181        owned = step.yields_owned();
182    }
183    e
184}
185/// Compose a value form's OWN call — the one step in the whole system whose
186/// receiver may be by value.
187///
188/// Four cases, from what the fold ended up holding crossed with what the
189/// accessor takes. A CONSUMING form takes its receiver by value: hand it the
190/// value when that is ours, clone when it is not — the same cost the borrowing
191/// form of the accessor would have paid, which keeps one declaration usable by
192/// both owned and `&T` roots. A borrowing form takes a reference, so an owned
193/// value is borrowed for it.
194///
195/// The decision is made from the fold's RESULT, never from where the fold
196/// began: that is what lets a consuming form sit behind ordinary accessors,
197/// where the chain in front borrows and the form itself still moves.
198///
199/// One function because both hoist paths — the conditional binding and the
200/// ordinary one — need exactly this rule, and stating it twice is what turned
201/// each new shape into a new defect.
202fn compose_value_form_call(
203    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
204    call: &PathStep,
205    e: TokenStream,
206    e_owned: bool,
207    consuming: bool,
208) -> TokenStream {
209    match (consuming, e_owned) {
210        (true, owned) => {
211            let (m, f) = (qualify(call.ident()), call.ident());
212            // Parenthesized: the clone applies to whatever the fold holds, and
213            // `&x.clone()` would parse as `&(x.clone())`.
214            let arg = if owned { e } else { quote!((#e).clone()) };
215            quote!(#m::#f(#arg))
216        }
217        (false, true) => compose_step(qualify, call, quote!(&#e)),
218        (false, false) => compose_step(qualify, call, e),
219    }
220}
221/// Start a reach from `base`, projecting the **leading run of plain field
222/// steps** directly (`&base.a.b`) instead of through a borrow of the base
223/// (`&(&base).a.b`). Returns the expression and how many steps it consumed.
224///
225/// The two forms name the same value, but the second borrows the base **as a
226/// whole**, which the borrow checker rejects once a sibling leaf has moved a
227/// different field out of it. Projecting directly makes each leaf's borrow
228/// disjoint, so a consuming value form's field moves are order-independent
229/// rather than compiling only while the borrowing leaves happen to be declared
230/// first.
231pub fn project_leading_fields(
232    base: &TokenStream,
233    base_is_ref: bool,
234    path: &[PathStep],
235) -> (TokenStream, usize) {
236    if base_is_ref {
237        return (base.clone(), 0);
238    }
239    let n = path.iter().take_while(|s| s.is_plain_field()).count();
240    if n == 0 {
241        return (quote!(&#base), 0);
242    }
243    let segs: Vec<&syn::Ident> = path[..n].iter().map(PathStep::ident).collect();
244    (quote!(&#base #(.#segs)*), n)
245}

247/// Where one leaf's reach starts, once the hoists are bound.
248///
249/// Produced by [`Hoisted::place`], which is the one place that decides it: the
250/// innermost value form the leaf sits under, the name a conditional form's
251/// `Some` arm binds, or the delivered value itself.
252pub struct LeafPlace {
253    /// What to reach from.
254    pub base: TokenStream,
255    /// Whether `base` is already a reference.
256    pub base_is_ref: bool,
257    /// The steps left from `base` — the leaf's own path with the prefix that
258    /// bound the hoist already consumed.
259    pub path: Vec<PathStep>,
260    /// Whether the form that produced `base` gave its value away.
261    pub consuming: bool,
262    /// The conditional hoist whose `Some` arm this leaf belongs in. Its
263    /// statements go there rather than beside the others, because the arm is
264    /// where the binding they reach off exists.
265    pub conditional: Option<usize>,
266}
267
268impl LeafPlace {
269    /// The place this leaf may MOVE out of, or `None` when what it reaches is
270    /// not the delivery's to give away, or is not a place a move can name.
271    ///
272    /// The same question [`reach_leaf`] answers on its way to the terminal
273    /// treatment, asked separately by a leaf whose encode needs the place
274    /// itself rather than an expression reaching it — a handle that is boxed
275    /// rather than converted.
276    pub fn owned<L: DecomposedLeaf>(&self, leaf: &L) -> Option<TokenStream> {
277        (reached_is_ours(leaf, self.consuming) && steps_are_movable(&self.path)).then(|| {
278            let base = &self.base;
279            let segs: Vec<&syn::Ident> = self.path.iter().map(PathStep::ident).collect();
280            quote!(#base #(.#segs)*)
281        })
282    }
283}

285/// Whether what a leaf reaches is OURS, and so is moved rather than borrowed
286/// or cloned. The two leaf kinds say it differently:
287///
288/// * an IDENTITY leaf carries the answer in its own reading — the plan
289///   resolved that to the owned type exactly when the value is the plan's to
290///   give away (an owned root, or a field of a CONSUMING value form), and that
291///   is also what selected the owning converter, which boxes the move rather
292///   than cloning a borrow;
293/// * every other leaf reads a field, whose reading is the field type as
294///   written and owned either way, so ownership is the enclosing form's: only
295///   a consuming one gives its fields away.
296pub fn reached_is_ours<L: DecomposedLeaf>(leaf: &L, consuming: bool) -> bool {
297    if leaf.identity() {
298        !matches!(
299            leaf.source().kind(),
300            prebindgen_flat::flat::TypeKind::Ref { .. }
301        )
302    } else {
303        consuming
304    }
305}

307/// One leaf, and the place a reach of it starts from.
308///
309/// A caller that has already bound a hoist hands over the local it bound and
310/// the steps that are left, which is why `path` is a suffix of the leaf's own
311/// reach rather than always equal to it.
312pub struct LeafAt<'a, L> {
313    /// The leaf being reached.
314    pub leaf: &'a L,
315    /// The steps still to walk from `base`.
316    pub path: &'a [PathStep],
317    /// What the walk starts from — the delivered value, or a local a hoist or
318    /// a gate bound.
319    pub base: TokenStream,
320    /// Whether `base` is already a reference.
321    pub base_is_ref: bool,
322    /// Whether the form that produced `base` gave its value away, which is
323    /// what lets a field of it be moved rather than cloned.
324    pub consuming: bool,
325    /// Whether a FINAL optional step is gated too. An identity leaf delivers
326    /// the reached value itself, so it is; every other leaf hands the final
327    /// step's full type to its own converter, `Option` and all.
328    pub unwrap_last: bool,
329}

331/// Reach one leaf from `base` and hand what it reached to `body`.
332///
333/// Three things happen on the way, and all three are facts about Rust values
334/// rather than about any target language:
335///
336/// * **Gating.** An optional step before the leaf becomes a `match` whose
337///   `None` arm yields `absent()` — the adapter's own absence, since this
338///   crate has none to offer. `unwrap_last` says whether a FINAL optional step
339///   is gated too: an identity leaf delivers the reached value itself, so it
340///   is, while every other leaf hands the final step's full type to its own
341///   converter, `Option` and all.
342/// * **Refusal.** `absent` is `None` at a site that cannot express absence at
343///   all — a single delivered return value, which has no arm to put one in.
344///   An optional step before the end is then refused rather than composed into
345///   code the consumer's crate cannot type-check.
346/// * **Ownership**: the reached place is moved when it is ours, cloned when
347///   it is a field read of a place that is not, and borrowed otherwise.
348///
349/// One derivation, used by every delivery an adapter renders and by the
350/// single-leaf shortcut a decomposed return takes. They drifted once while
351/// they were two — the shortcut was missing the field clone and handed `&F` to
352/// an `F` converter — and a second pair, ungated here and gated in JniGen,
353/// drifted the same way until #607 folded them together.
354pub fn reach_leaf<L: DecomposedLeaf>(
355    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
356    at: LeafAt<'_, L>,
357    absent: Option<&dyn Fn() -> TokenStream>,
358    body: &dyn Fn(TokenStream) -> TokenStream,
359) -> TokenStream {
360    reach_leaf_at(qualify, at, absent, 0, body)
361}

363fn reach_leaf_at<L: DecomposedLeaf>(
364    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
365    at: LeafAt<'_, L>,
366    absent: Option<&dyn Fn() -> TokenStream>,
367    depth: usize,
368    body: &dyn Fn(TokenStream) -> TokenStream,
369) -> TokenStream {
370    let LeafAt {
371        path,
372        ref base,
373        base_is_ref,
374        unwrap_last,
375        ..
376    } = at;
377    // Every optional step on the way to the leaf becomes a `match` whose
378    // `None` arm is the adapter's absent value. A site with no absent value to
379    // give — a single delivered return, which has no arm to put one in —
380    // passes `None`, and an optional step before the end is then refused
381    // rather than composed into code the consumer's crate cannot type-check.
382    if let Some(absent) = absent {
383        let limit = if unwrap_last {
384            path.len()
385        } else {
386            // A non-identity leaf's converter takes the final step's FULL type,
387            // `Option` included, so only the steps before it are nesting.
388            path.len().saturating_sub(1)
389        };
390        let (projected, lead) = project_leading_fields(base, base_is_ref, path);
391        if let Some(k) = (lead..limit).find(|&i| path[i].is_optional()) {
392            // Through the optional step INCLUSIVE: the same fold, so the borrow
393            // in front of it is the ordinary rule rather than a second
394            // statement of it.
395            let opt_e = fold_steps(qualify, &path[lead..=k], projected, false);
396            let nested = format_ident!("__n{}", depth);
397            // What the arm binds is the step's OWN value: an owned payload is
398            // a bare `T`, so composing the next step onto it directly would
399            // hand `T` to an accessor typed for `&T`. Say it is not a
400            // reference and let `project_leading_fields` borrow it; a borrowed
401            // payload is already one and passes through. With no steps left
402            // the binding goes to `body` untouched, which is what lets an
403            // owned payload be moved rather than borrowed straight back.
404            //
405            // The same rule [`reach_optional`] states for a conditional
406            // hoist's prefix. This used to be a hardcoded `true` in the gated
407            // reach JniGen owned, where an `Option<T>` accessor followed by
408            // another accessor emitted `next(__n0)` against a `&T` receiver.
409            let rest = &path[k + 1..];
410            let inner = reach_leaf_at(
411                qualify,
412                LeafAt {
413                    path: rest,
414                    base: quote!(#nested),
415                    base_is_ref: rest.is_empty() || !path[k].yields_owned(),
416                    ..at
417                },
418                Some(absent),
419                depth + 1,
420                body,
421            );
422            let gone = absent();
423            // A FIELD read composes to a borrow (`&(e).f`), so it goes through
424            // a coercion site and the destructuring stops caring which
425            // representation the source spelled the optional as (#268).
426            //
427            // A CALL composes to the accessor's own returned value, which is
428            // owned and whose payload downstream may move. Borrowing it to
429            // coerce would change that ownership, so it keeps its direct match.
430            if path[k].is_field() {
431                let opt_bind = format_ident!("__o{}", depth);
432                return quote! {
433                    {
434                        let #opt_bind: &::core::option::Option<_> = #opt_e;
435                        match #opt_bind {
436                            ::core::option::Option::Some(#nested) => { #inner }
437                            ::core::option::Option::None => #gone,
438                        }
439                    }
440                };
441            }
442            return quote! {
443                match #opt_e {
444                    ::core::option::Option::Some(#nested) => { #inner }
445                    ::core::option::Option::None => #gone,
446                }
447            };
448        }
449        return body(reached_place(qualify, &at));
450    }
451    // An optional step BEFORE the last one needs a `match` whose `None` arm has
452    // somewhere to go. This derivation has none — it yields a plain Rust value,
453    // not a representation that can carry absence — so the shape is refused
454    // here rather than composed into code that cannot type-check in the
455    // consumer's crate.
456    //
457    // Asked of the leaf's OWN path, not of `path`. The caller may hand a
458    // suffix: `wrapper.rs` rebases onto a hoisted local, and `Hoisted::innermost`
459    // strips the prefix that bound it — including any optional step inside it.
460    // Checking the parameter would therefore pass exactly when the hoist is the
461    // conditional one, which is the case that cannot compose (an `Option<T>`
462    // local with a field read hung off it). The full path is what the shape
463    // question is about.
464    let own_path = at.leaf.reach();
465    assert!(
466        !own_path.iter().rev().skip(1).any(PathStep::is_optional),
467        "unfold: leaf `{}` reaches through an optional step but is \
468         delivered as a single return value, which has no `None` arm — this \
469         shape needs callback delivery",
470        at.leaf.name(),
471    );
472    // Whether what this leaf reaches is OURS, and so is moved rather than
473    // borrowed or cloned. The two leaf kinds say it differently:
474    //
475    // * an IDENTITY leaf carries the answer in its `out_ty` — the plan resolved
476    //   it to the owned type exactly when the value is the plan's to give away
477    //   (`place_is_owned`: an owned root, or a field of a CONSUMING value form),
478    //   and that is also what selected the owning converter, which boxes the
479    //   move rather than cloning a borrow;
480    // * a FIELD leaf's `out_ty` is the field type as written, owned either way,
481    //   so ownership is the enclosing form's: only a consuming one gives its
482    //   fields away.
483    //
484    // How to project that place is `steps_are_movable`'s question, and it is
485    // asked there rather than restated here. This used to spell it
486    // `all(is_plain_field)`, defending the restatement on the grounds that a
487    // trailing `Option` cannot reach return delivery anyway — true, and enforced
488    // in `single_return` (`core/unfold.rs`), which is precisely why a local
489    // restatement could disagree with the rule for as long as the invariant held
490    // somewhere else. `plan.rs` says two readings would drift and the
491    // disagreement would be a borrow handed to an owning converter; this is the
492    // second reading, removed.
493    body(reached_place(qualify, &at))
494}

496/// The place a leaf reaches, with the terminal treatment its ownership calls
497/// for: moved out when it is ours and the path projects a place, cloned out
498/// when it is a field read of a place that is not, and borrowed otherwise.
499fn reached_place<L: DecomposedLeaf>(
500    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
501    at: &LeafAt<'_, L>,
502) -> TokenStream {
503    let LeafAt {
504        leaf,
505        path,
506        base,
507        base_is_ref,
508        consuming,
509        ..
510    } = at;
511    let (base_is_ref, consuming) = (*base_is_ref, *consuming);
512    // How to project a movable place is `steps_are_movable`'s question, asked
513    // there rather than restated here. This used to spell it
514    // `all(is_plain_field)`, defending the restatement on the grounds that a
515    // trailing `Option` cannot reach return delivery anyway — true, and
516    // enforced in `single_return`, which is precisely why a local restatement
517    // could disagree with the rule for as long as the invariant held somewhere
518    // else. `plan.rs` says two readings would drift and the disagreement would
519    // be a borrow handed to an owning converter; this is the second reading,
520    // removed.
521    if reached_is_ours(*leaf, consuming) && steps_are_movable(path) {
522        let segs: Vec<&syn::Ident> = path.iter().map(PathStep::ident).collect();
523        return quote!(#base #(.#segs)*);
524    }
525    let (e, lead) = project_leading_fields(base, base_is_ref, path);
526    let e = fold_steps(qualify, &path[lead..], e, false);
527    // Cloned out of the BORROW the reach yields, not out of the place behind
528    // it. The two agree for a field of an owned type and disagree for a field
529    // that is itself a reference: `place.clone()` there resolves through the
530    // reference and deep-clones the pointee, where the converter takes the
531    // field type as written. JniGen's own reach spelled it the short way and
532    // no declared field exercised the difference — the drift this step exists
533    // to close.
534    if leaf.is_field_read() {
535        quote!((#e).clone())
536    } else {
537        e
538    }
539}
540/// The **segments** of a decomposition: each selector leaf, together with the
541/// group leaves that follow it.
542///
543/// A segment's leaves are not independent — one alternative of a sum is live
544/// per value, and a gated group is live only when its value is there — so a
545/// segment is emitted as one `match` rather than as a leaf at a time. The plan
546/// already says which leaves those are: a selector says so of itself, and each
547/// of the leaves after it carries the group it belongs to until one does not.
548///
549/// Segments do not nest: a group's leaves may not include a selector, or these
550/// ranges would overlap. [`UnfoldLeaf::group`](crate::unfold::UnfoldLeaf::group)
551/// states that invariant and what a decomposition does with a shape that would
552/// need it. A decomposition may carry several segments, or none: a sum
553/// that IS the delivered value is one segment covering everything, and a value
554/// form contributes one per sum-typed field.
555///
556/// Not recognising a segment is silent rather than a compile error — the
557/// leaves are all there, and a delivery that treats them as independent reads
558/// a dead alternative's fields — which is why this is the registry's answer
559/// and not an adapter's.
560pub fn segments<L: DecomposedLeaf>(leaves: &[L]) -> Vec<std::ops::Range<usize>> {
561    let n = leaves.len();
562    (0..n)
563        .filter(|&i| leaves[i].selects())
564        .map(|start| {
565            let end = (start + 1..n)
566                .take_while(|&i| leaves[i].group().is_some())
567                .last()
568                .map_or(start + 1, |i| i + 1);
569            start..end
570        })
571        .collect()
572}

574/// Render one segment off `place`, gating the whole of it when the selector
575/// reaches the value it selects on through an optional step.
576///
577/// A segment's **selected value** is what its selector chooses over: the sum a
578/// tag names an alternative of, or the optional value a presence flag says is
579/// there. `group` renders the segment's own encode from the expression naming
580/// it — that half is the adapter's, since what a group of slots is filled with
581/// is a target-language question.
582///
583/// **An optional value gates the segment, not each slot.** A segment's leaves
584/// are not independent, so absence cannot be the per-leaf absent value
585/// [`reach_leaf`] gives an ordinary optional field: it is one tuple bind whose
586/// `None` arm carries every slot's default. That is the same shape a
587/// conditional value form's hoist emits, applied to an optional step inside
588/// the segment's own path (#220).
589///
590/// `index` names the segment's locals, so two segments of one delivery cannot
591/// collide.
592pub fn segment<B: DeliveryBridge>(
593    bridge: &B,
594    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
595    place: &LeafPlace,
596    index: usize,
597    leaves: &[B::Leaf],
598    slots: &[syn::Ident],
599    group: &dyn Fn(TokenStream) -> TokenStream,
600) -> TokenStream {
601    let path = &place.path;
602    // A plain field chain is borrowed DIRECTLY (`&base.a.b`) rather than
603    // through the base (`&(&base).a.b`). The two are the same value, but the
604    // second borrows the base as a whole, which the borrow checker rejects
605    // once a sibling leaf has moved another field out of it — and borrowing
606    // this field while sibling fields move is exactly what a consuming value
607    // form does.
608    let (projected, lead) = project_leading_fields(&place.base, place.base_is_ref, path);
609    // The selector's own path reaches the selected value (empty when that
610    // value IS the delivered one), and a step on it MAY be optional — which is
611    // always so for a presence flag, whose whole subject is that step.
612    let Some(k) = (lead..path.len()).find(|&i| path[i].is_optional()) else {
613        return group(fold_steps(qualify, &path[lead..], projected, false));
614    };
615    // Through the optional step INCLUSIVE, then the rest off the binding — the
616    // same split [`reach_leaf`] makes, so the borrow in front of it stays the
617    // ordinary rule rather than a second statement of it.
618    let opt_e = fold_steps(qualify, &path[lead..=k], projected, false);
619    let bind = format_ident!("__sg{}", index);
620    // ONE optional step is what this gate handles. A second one in the tail
621    // would compose `match &Option<..>` against the patterns the group's own
622    // encode writes — an E0308 in the consumer's crate — so the named
623    // diagnostic is raised here instead.
624    //
625    // `assert!`, not `debug_assert!`: a build script inherits the consumer's
626    // profile, so a debug-only check is absent from exactly the release build
627    // where a mis-emission costs the most to diagnose. Same rule, same
628    // phrasing, as the single-return optional-step assert in [`reach_leaf`].
629    //
630    // The condition is what actually breaks, not the stronger fact that
631    // happens to hold: a selector's path stops AT the value it selects on, so
632    // the tail is empty today, but a NON-optional tail composes correctly
633    // through [`fold_steps`] — refusing it would refuse a shape that works.
634    assert!(
635        !path[k + 1..].iter().any(PathStep::is_optional),
636        "unfold: leaf `{}` reaches the value it selects on through TWO optional \
637         steps — the segment gate has one `None` arm, so the second would be \
638         matched as if it were that value itself",
639        leaves
640            .first()
641            .expect("a segment has at least its selector")
642            .name(),
643    );
644    // What the `match` binds, asked of the step rather than assumed: the FIELD
645    // branch scrutinizes `&Option<_>`, so ergonomics binds a borrow of the
646    // selected value. Only an owned-yielding CALL binds an owned one.
647    let inner = group(fold_steps(
648        qualify,
649        &path[k + 1..],
650        quote!(#bind),
651        path[k].yields_owned(),
652    ));
653    let filled: Vec<Slot> = leaves.iter().map(|leaf| bridge.slot(leaf)).collect();
654    let tys = filled.iter().map(|slot| &slot.ty);
655    let defaults = filled.iter().map(|slot| &slot.default);
656    // A FIELD step composes to a borrow, so it goes through a coercion site
657    // and the destructure stops caring which representation the source spelled
658    // the optional as (#268). A CALL yields its own owned value, whose payload
659    // downstream may move, so it keeps the direct match — the same division
660    // [`reach_leaf`] makes.
661    let (prelude, scrutinee) = if path[k].is_field() {
662        let opt_bind = format_ident!("__so{}", index);
663        (bind_as_option(&opt_e, &opt_bind), quote!(#opt_bind))
664    } else {
665        (TokenStream::new(), opt_e)
666    };
667    quote! {
668        let (#(#slots,)*): (#(#tys,)*) = {
669            #prelude
670            match #scrutinee {
671                ::core::option::Option::Some(#bind) => {
672                    #inner
673                    (#(#slots,)*)
674                }
675                ::core::option::Option::None => (#(#defaults,)*),
676            }
677        };
678    }
679}

681/// The `match` a **conditional** value form's leaves share.
682///
683/// A hoist under an optional step ran only where the value was present, so its
684/// leaves cannot be independent statements: their slots exist either way, and
685/// only one arm computes them. The `Some` arm runs `body` — the statements
686/// those leaves contributed — and yields their slots as a tuple; the `None`
687/// arm yields each slot's default, the same shape [`segment`] gives an absent
688/// sum. Binding the tuple outside the `match` keeps the slots in scope for the
689/// call's argument list, which is indifferent to how a slot was filled.
690///
691/// Matched BY VALUE: the local is this arm's alone, every leaf under the hoist
692/// being in it, so a consuming value form's fields move out here exactly as
693/// they do at an unconditional one.
694pub fn conditional_arm<B: DeliveryBridge>(
695    bridge: &B,
696    hoisted: &Hoisted,
697    index: usize,
698    leaves: &[B::Leaf],
699    slots: &[syn::Ident],
700    body: TokenStream,
701) -> TokenStream {
702    let local = hoisted.local(index);
703    let bind = format_ident!("__u{}", index);
704    let under: Vec<usize> = (0..leaves.len())
705        .filter(|&k| {
706            hoisted
707                .conditional(leaves[k].reach())
708                .is_some_and(|(j, ..)| j == index)
709        })
710        .collect();
711    let ids: Vec<&syn::Ident> = under.iter().map(|&k| &slots[k]).collect();
712    let filled: Vec<Slot> = under.iter().map(|&k| bridge.slot(&leaves[k])).collect();
713    let tys = filled.iter().map(|slot| &slot.ty);
714    let defaults = filled.iter().map(|slot| &slot.default);
715    quote! {
716        let (#(#ids,)*): (#(#tys,)*) = match #local {
717            ::core::option::Option::Some(#bind) => { #body (#(#ids,)*) }
718            ::core::option::Option::None => (#(#defaults,)*),
719        };
720    }
721}

723/// Bind `e` so it can be destructured as an `Option` **whatever Rust
724/// representation the source used for it**.
725///
726/// The model says a position is optional; it deliberately does not say whether
727/// Rust spells that `Option<T>`, `Box<Option<T>>`, or something else — the
728/// side interpreting the classification is the side that must accept any
729/// representation. Matching the reached place directly assumed one, which is
730/// "classify off the model, spell from the model" broken in the direction
731/// nothing was watching: `Box<Option<T>>` then produced
732/// `match &place { Some(..) => .. }` and an E0308 (#268).
733///
734/// A type-ascribed `let` is a coercion site, and deref coercion is transitive
735/// **and** a no-op when the types already match — so this one shape serves
736/// every representation, and the plain `Option<T>` case is unchanged.
737///
738/// `e` is expected to be a **reference** already — [`compose_step`] composes a
739/// field read as `&(e).f` — so nothing is borrowed here. Borrowing only: an
740/// owned position cannot be made representation-agnostic this way, because
741/// deref coercion applies to references and moving out of a wrapper is
742/// something only some of them permit (`Box` does, `Rc` cannot). A site that
743/// must MOVE the payload keeps its direct match.
744pub fn bind_as_option(e: &TokenStream, bind: &syn::Ident) -> TokenStream {
745    quote! { let #bind: &::core::option::Option<_> = #e; }
746}

748/// Every value form on a plan, evaluated **once** and bound to a local
749/// (`__vf0`, `__vf1`, …), so a struct is built once per delivery rather than
750/// once per field. The bound prefixes come back with the statements, since
751/// reaching a leaf means starting from the innermost local it sits under.
752///
753/// Shared by both delivery paths — the multi-leaf encoder below and the
754/// single-leaf `Delivery::Return` shortcut in `emit/wrapper.rs`. The shortcut
755/// used to compose its reach straight off the raw value, which for a consuming
756/// value form emitted `f(&v)` against a by-value receiver: ill-typed Rust in
757/// the consumer's crate. One binder, so the two cannot disagree about what a
758/// hoist is or who owns it.
759pub struct Hoisted {
760    /// The `let __vfN = …;` bindings, outermost-first.
761    pub stmts: TokenStream,
762    /// Each hoist's path prefix and the local it was bound to.
763    bound: Vec<(Vec<PathStep>, syn::Ident)>,
764    /// Whether each bound hoist consumed the value it decomposed.
765    consuming: Vec<bool>,
766    /// Whether each bound local is `Option<TStruct>` rather than `TStruct` —
767    /// the hoist sits under an optional step, so the value form ran only where
768    /// the value was present. Its leaves cannot be emitted as independent
769    /// statements: they share ONE `match` on the local (see
770    /// [`encode_plan_leaves`]), taken by value, so a consuming form's fields
771    /// still move out inside the arm.
772    optional: Vec<bool>,
773}
774impl Hoisted {
775    /// Index of the innermost bound hoist `path` sits under, with that prefix
776    /// already consumed. `None` for a leaf under no value form at all — a
777    /// sibling `.field()` / `.field_self()`, which still reaches from the value
778    /// itself.
779    fn innermost(&self, path: &[PathStep]) -> Option<(usize, Vec<PathStep>)> {
780        self.bound
781            .iter()
782            .enumerate()
783            .filter(|(_, (p, _))| p.len() < path.len() && path.starts_with(p))
784            .max_by_key(|(_, (p, _))| p.len())
785            .map(|(i, (p, _))| (i, path[p.len()..].to_vec()))
786    }
787
788    /// The innermost bound local `path` sits under, with that prefix already
789    /// consumed, and whether that hoist gave its value away.
790    pub fn rebase(&self, path: &[PathStep]) -> Option<(syn::Ident, Vec<PathStep>, bool)> {
791        self.innermost(path)
792            .map(|(i, rest)| (self.bound[i].1.clone(), rest, self.consuming[i]))
793    }
794
795    /// The innermost **conditional** hoist `path` sits under: its index, the
796    /// local holding the `Option`, the name its `Some` arm binds, and the steps
797    /// left to reach the leaf from there. `None` when the leaf's innermost
798    /// hoist is unconditional (or there is none) — then [`Self::rebase`]
799    /// applies and the leaf is an ordinary standalone statement.
800    pub fn conditional(
801        &self,
802        path: &[PathStep],
803    ) -> Option<(usize, syn::Ident, syn::Ident, Vec<PathStep>)> {
804        let (i, rest) = self.innermost(path)?;
805        self.optional[i].then(|| (i, self.bound[i].1.clone(), format_ident!("__u{}", i), rest))
806    }
807
808    /// Where a leaf's reach starts, once these hoists are bound.
809    ///
810    /// Three cases, and the leaf's own path decides which.
811    ///
812    /// A leaf under a **conditional** value form reaches off the name that
813    /// form's `Some` arm binds. That arm matches the hoist's `Option` by
814    /// value — every leaf under the hoist is inside it, so the local is the
815    /// arm's alone — which makes the binding an owned payload: a consuming
816    /// form's fields move out of it exactly as they do at an unconditional
817    /// one, and [`LeafPlace::owned`] says so. The leaf's statements belong in
818    /// that arm, because that is where the binding exists.
819    ///
820    /// A leaf under an **ordinary** value form reaches off the local that form
821    /// was bound to, with the prefix already consumed and the form's own
822    /// ownership carried along.
823    ///
824    /// A leaf under **no** value form — a sibling accessor of the delivered
825    /// value — reaches from that value.
826    pub fn place<L: DecomposedLeaf>(
827        &self,
828        leaf: &L,
829        value: &TokenStream,
830        by_ref: bool,
831    ) -> LeafPlace {
832        let path = leaf.reach();
833        if let Some((i, _, bind, rest)) = self.conditional(path) {
834            return LeafPlace {
835                base: quote!(#bind),
836                base_is_ref: false,
837                path: rest,
838                consuming: self.consumed(i),
839                conditional: Some(i),
840            };
841        }
842        match self.rebase(path) {
843            Some((local, rest, consuming)) => LeafPlace {
844                base: quote!(#local),
845                base_is_ref: false,
846                path: rest,
847                consuming,
848                conditional: None,
849            },
850            None => LeafPlace {
851                base: value.clone(),
852                base_is_ref: by_ref,
853                path: path.to_vec(),
854                consuming: false,
855                conditional: None,
856            },
857        }
858    }
859
860    /// The local a hoist was bound to.
861    pub fn local(&self, i: usize) -> syn::Ident {
862        self.bound[i].1.clone()
863    }
864
865    /// Whether a hoist consumed the value it decomposed.
866    pub fn consumed(&self, i: usize) -> bool {
867        self.consuming[i]
868    }
869}
870/// Fold `path` over `base` the way an adapter's own gated reach does, but
871/// yielding an `Option<…>` rather than the adapter's absent value: the optional
872/// steps become a `map`/`and_then` chain, so an absent value short-circuits to
873/// `None` instead of to whatever that adapter uses for absence. `body` renders
874/// the innermost reached expression as a BARE value — the chain's last link
875/// wraps it.
876///
877/// The gated reach this mirrors is [`reach_leaf`], which takes the absent
878/// value from the adapter rather than choosing one. This one yields an
879/// `Option` instead, because what it binds is a hoist rather than a delivered
880/// leaf: the value form runs where the value is present, and the `None` the
881/// chain produces is the local's own.
882///
883/// This is how a CONDITIONAL value form is bound — the accessor runs only where
884/// the value it decomposes is actually present.
885fn reach_optional(
886    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
887    path: &[PathStep],
888    base: TokenStream,
889    base_is_ref: bool,
890    depth: usize,
891    body: &dyn Fn(TokenStream) -> TokenStream,
892) -> TokenStream {
893    let (e, lead) = project_leading_fields(&base, base_is_ref, path);
894    match (lead..path.len()).find(|&i| path[i].is_optional()) {
895        None => body(fold_steps(qualify, &path[lead..], e, false)),
896        Some(k) => {
897            // Through the optional step INCLUSIVE: the same fold, so the
898            // borrow in front of it is the ordinary rule rather than a second
899            // statement of it.
900            let opt_e = fold_steps(qualify, &path[lead..=k], e, false);
901            let bind = format_ident!("__hb{}", depth);
902            // What the arm binds is the step's own value: an OWNED payload is a
903            // bare `T`, so composing the next step onto it directly would hand
904            // `T` to an accessor typed for `&T`. Say it is not a reference and
905            // let `project_leading_fields` borrow it; a borrowed payload is
906            // already one and passes through.
907            //
908            // With NO steps left the binding goes to `body` untouched — that is
909            // what lets a consuming value form MOVE an owned payload rather than
910            // borrow it straight back, so the terminal case stays "already a
911            // reference" whatever the payload is.
912            let rest = &path[k + 1..];
913            let inner = reach_optional(
914                qualify,
915                rest,
916                quote!(#bind),
917                rest.is_empty() || !path[k].yields_owned(),
918                depth + 1,
919                body,
920            );
921            // `map` when this is the LAST optional step (the body yields a bare
922            // value) and `and_then` when another follows (the recursion yields
923            // an `Option` that must not nest). The equivalent `match` reads the
924            // same but generated code runs through the consumer's lints, where
925            // `clippy::manual_map` is a denial.
926            let combinator = if rest.iter().any(PathStep::is_optional) {
927                format_ident!("and_then")
928            } else {
929                format_ident!("map")
930            };
931            quote! {
932                #opt_e.#combinator(|#bind| #inner)
933            }
934        }
935    }
936}
937pub fn bind_hoists(
938    qualify: &dyn Fn(&syn::Ident) -> syn::Path,
939    hoists: &[crate::unfold::Hoist],
940    value: &TokenStream,
941    by_ref: bool,
942) -> Hoisted {
943    let mut out = Hoisted {
944        stmts: TokenStream::new(),
945        bound: Vec::new(),
946        consuming: Vec::new(),
947        optional: Vec::new(),
948    };
949    // Value forms COMPOSE, so each hoist is built from the longest hoist that
950    // is already a proper prefix of it (they arrive outermost-first), and from
951    // `value` otherwise.
952    for (i, h) in hoists.iter().enumerate() {
953        let local = format_ident!("__vf{}", i);
954        // A hoist under an optional step binds `Option<TStruct>`: the value
955        // form runs in the `Some` arm only. Core refuses to nest these, so the
956        // enclosing value is always the plan's own — no rebase to consider.
957        if h.prefix.iter().any(PathStep::is_optional) {
958            let (last, lead) = h
959                .prefix
960                .split_last()
961                .expect("a hoist prefix ends in its value-form call");
962            let consuming = h.consuming;
963            // The value form is handed the payload only when the optional step
964            // is the LAST thing before it; any step in between composes as a
965            // borrow, so what arrives is a reference either way.
966            let owned = lead.last().is_some_and(PathStep::yields_owned);
967            let expr = reach_optional(qualify, lead, value.clone(), by_ref, 0, &|reached| {
968                compose_value_form_call(qualify, last, reached, owned, consuming)
969            });
970            out.stmts.extend(quote! { let #local = #expr; });
971            out.bound.push((h.prefix.clone(), local));
972            out.consuming.push(h.consuming);
973            out.optional.push(true);
974            continue;
975        }
976        // Where the fold starts, and whether what it starts from is OWNED. The
977        // value form's own boundary is decided below, from what the fold ends
978        // up holding — never from where it began.
979        let (from, start, start_owned) = match out.rebase(&h.prefix) {
980            // A NESTED consuming form is handed the parent's field by MOVE: a
981            // hoisted value form is an owned struct and its fields are
982            // disjoint, so moving one out leaves every sibling leaf readable.
983            // `compose_step` borrows (`&(e).f`), so a plain field run to that
984            // field is projected here instead of going through it.
985            Some((outer, rest, _))
986                if h.consuming && rest[..rest.len() - 1].iter().all(PathStep::is_plain_field) =>
987            {
988                let lead = &rest[..rest.len() - 1];
989                let segs: Vec<&syn::Ident> = lead.iter().map(PathStep::ident).collect();
990                (h.prefix.len() - 1, quote!(#outer #(.#segs)*), true)
991            }
992            // Any other rebased hoist: project its own leading field run
993            // DIRECTLY off the parent local rather than reaching it through a
994            // borrow of the parent. A sibling hoist may already have moved a
995            // different field out — that is what a consuming value form does —
996            // and `&(&__vf0).wrapper` borrows the partially moved parent as a
997            // whole where `&__vf0.wrapper` is a disjoint borrow that survives.
998            // Same invariant `project_leading_fields` states for leaf reaches,
999            // and the same reason.
1000            Some((outer, rest, _)) => {
1001                let (e, lead) = project_leading_fields(&quote!(#outer), false, &rest);
1002                (h.prefix.len() - rest.len() + lead, e, false)
1003            }
1004            None if by_ref => (0, value.clone(), false),
1005            None => (0, value.clone(), true),
1006        };
1007        // Everything before the value form is an ordinary accessor chain.
1008        let last = h.prefix.len() - 1;
1009        let head = &h.prefix[from..last];
1010        let e = fold_steps(qualify, head, start, start_owned);
1011        let e_owned = head.last().map_or(start_owned, PathStep::yields_owned);
1012        // The value form itself. A CONSUMING one takes its receiver BY VALUE —
1013        // that is the move the whole declaration exists for — so it is handed
1014        // what the fold holds when that is ours, and a clone when it is not:
1015        // the same cost the borrowing form of the accessor would have paid,
1016        // which keeps one declaration usable by both owned and `&T` returns.
1017        // A borrowing one takes a reference, so an owned value is borrowed.
1018        //
1019        // Deciding this from the fold's RESULT rather than from its start is
1020        // what lets a consuming form sit behind ordinary accessors: the chain
1021        // in front borrows, the form itself still moves.
1022        let expr = compose_value_form_call(qualify, &h.prefix[last], e, e_owned, h.consuming);
1023        out.stmts.extend(quote! { let #local = #expr; });
1024        out.bound.push((h.prefix.clone(), local));
1025        out.consuming.push(h.consuming);
1026        out.optional.push(false);
1027    }
1028    out
1029}
