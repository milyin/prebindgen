1//! Explicit Result error outputs. Message buffers and structured errors are caller-owned.

6#[derive(Clone, Debug)]
7pub(crate) struct ErrorBinding {
8    pub name: String,
9    pub ty: syn::Type,
10    pub plan: Option<OutputPlan<CWire>>,
11    pub fields: Vec<String>,
12}
13pub(crate) fn helper(name: &str) -> syn::Ident {
14    format_ident!("__pb_report_{name}")
15}
16impl ErrorBinding {
17    pub(crate) fn declared(name: String, ty: syn::Type) -> Self {
18        Self {
19            name,
20            ty,
21            plan: None,
22            fields: Vec::new(),
23        }
24    }
25    pub(crate) fn plan(&self) -> Result<&OutputPlan<CWire>> {
26        self.plan
27            .as_ref()
28            .ok_or_else(|| format!("C error `{}` has no prepared conversion", self.name))
29    }
30}
31/// Fill the conversions of error declarations already attached to their source types.
32pub(crate) fn prepare(
33    flat: &mut Bindings,
34    codec: &enums::EnumCodec,
35    declarations: &prebindgen_declarations::Declarations,
36) -> Result<()> {
37    let mut types = flat
38        .types
39        .values()
40        .filter_map(|t| {
41            t.metadata
42                .error
43                .as_ref()
44                .filter(|e| e.plan.is_none())
45                .map(|e| (t.metadata.error_order, e.ty.clone()))
46        })
47        .collect::<Vec<_>>();
48    types.sort_by_key(|(order, _)| *order);
49    for (_, ty) in types {
50        let recipe = declarations.owned_record_output(flat, codec, &ty, &())?;
51        let plan = prebindgen_declarations::compile_output_value(flat, codec, &ty, &recipe, &())?;
52        let fields = field_names(&plan.outcomes.success);
53        let error = flat.type_metadata_mut(&ty)?.error.as_mut().unwrap();
54        let binding = std::rc::Rc::make_mut(error);
55        binding.plan = Some(plan);
56        binding.fields = fields;
57    }
58    Ok(())
59}

61fn field_names(
62    layout: &prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
63) -> Vec<String> {
64    use prebindgen_flat::WireNode;
65    let mut names = Vec::new();
66    if let WireNode::Branch {
67        meta: CBranch::Group(CGroup::Fields),
68        children,
69        ..
70    } = layout
71    {
72        for node in children.as_slice() {
73            if let WireNode::Branch {
74                meta: CBranch::Group(CGroup::Field(name)),
75                ..
76            } = node
77            {
78                for _ in crate::group::record_fields(node) {
79                    names.push(if crate::group::record_fields(node).len() == 1 {
80                        name.to_string().trim_start_matches("r#").to_owned()
81                    } else {
82                        format!("v{}", names.len())
83                    });
84                }
85            }
86        }
87    }
88    let keywords="auto break case char const continue default do double else enum extern float for goto if inline int long register restrict return short signed sizeof static struct switch typedef union unsigned void volatile while _Alignas _Alignof _Atomic _Bool _Complex _Generic _Imaginary _Noreturn _Static_assert _Thread_local bool true false NULL";
89    let mut unique = std::collections::BTreeSet::new();
90    if names.len() != crate::group::result_packet_fields(layout).len()
91        || names.iter().any(|name| {
92            !name.is_ascii()
93                || keywords.split_whitespace().any(|word| word == name)
94                || !unique.insert(name.clone())
95        })
96    {
97        return (0..crate::group::result_packet_fields(layout).len())
98            .map(|i| format!("v{i}"))
99            .collect();
100    }
101    names
102}

104pub(crate) fn emit(
105    errors: &[&ErrorBinding],
106    prefix: &str,
107    pointers: &std::collections::BTreeMap<String, prebindgen_flat::TypeInfo<CType>>,
108    symbols: &mut std::collections::BTreeSet<String>,
109    tokens: &mut TokenStream,
110    header: &mut String,
111) -> Result<()> {
112    for error in errors {
113        let plan = error.plan()?;
114        let name = &error.name;
115        let c_name = format!("{prefix}_{name}");
116        let drop_name = format_ident!("{prefix}_drop_{name}");
117        for symbol in [c_name.clone(), drop_name.to_string()] {
118            if !symbols.insert(symbol) {
119                return Err("C error type/destructor symbol collision".into());
120            }
121        }
122        let rust_name = format_ident!("__PbError_{name}");
123        let source_ty = &error.ty;
124        let helper = helper(name);
125        let input = format_ident!("source_error");
126        let context = CContext {
127            row_types: {
128                let mut schemas = Vec::new();
129                sequence::schemas(&plan.outcomes.success, &mut schemas);
130                schemas
131                    .into_iter()
132                    .map(|(wire, _, _)| {
133                        (
134                            wire.name.to_string(),
135                            format_ident!("__PbRow_error_{}_{}", name, wire.name),
136                        )
137                    })
138                    .collect()
139            },
140            acknowledgement: format_ident!("unused_acknowledgement"),
141            output_pointer: format_ident!("error_output"),
142            output_type: syn::parse_quote!(#rust_name),
143            error_pointer: format_ident!("unused_error"),
144            error_reported: format_ident!("unused_reported"),
145            output_committed: TokenStream::new(),
146            buffer_type: syn::parse_quote!(__PbBuffer),
147            wide_integer_type: syn::parse_quote!(__Pb128),
148            read_buffer: syn::parse_quote!(__pb_read),
149        };
150        let mut schemas = Vec::new();
151        sequence::schemas(&plan.outcomes.success, &mut schemas);
152        let carriers = schemas
153            .iter()
154            .map(|(wire, _, _)| wire.name.to_string())
155            .collect();
156        sequence::emit_schemas(
157            schemas, &carriers, &context, prefix, pointers, symbols, tokens, header,
158        )?;
159        let commit = plan.code_fragment_slot(
160            OutputFragment::Commit {
161                context: &context,
162                source: &input,
163            },
164            source_ty,
165        )?;
166        let commit: syn::ExprBlock = syn::parse2(commit).map_err(|e| e.to_string())?;
167        let commit = commit.block.stmts;
168        let deliver = plan.code_fragment_slot(
169            OutputFragment::Deliver {
170                context: &context,
171                source: &input,
172                next: quote!(#(#commit)* *reported=true; Ok(())),
173            },
174            source_ty,
175        )?;
176        let prepare = plan.code_fragment_slot(
177            OutputFragment::Prepare {
178                context: &context,
179                source: &input,
180                next: deliver,
181            },
182            source_ty,
183        )?;
184        let mut fields = Vec::new();
185        let mut drops = TokenStream::new();
186        header.push_str("typedef struct {\n");
187        for (i, wire) in crate::group::result_packet_fields(&plan.outcomes.success).enumerate() {
188            let field = format_ident!("v{i}");
189            let ty = wire.rust_type();
190            fields.push(quote!(pub #field:#ty));
191            let c_field = if error.fields.len()
192                == crate::group::result_packet_fields(&plan.outcomes.success).len()
193            {
194                error.fields[i].trim_start_matches("r#").to_owned()
195            } else {
196                format!("v{i}")
197            };
198            prebindgen_declarations::ident(&c_field)?;
199            header.push_str(&format!(
200                "    {} {c_field};\n",
201                c_ty(wire, prefix, pointers, &context)
202            ));
203        }
204        let values = crate::group::result_packet_fields(&plan.outcomes.success)
205            .enumerate()
206            .map(|(i, wire)| {
207                let field = format_ident!("v{i}");
208                if !matches!(
209                    wire.kind,
210                    CWire::Buffer | CWire::Pointer { .. } | CWire::Rows
211                ) {
212                    return (wire, quote!((*error).#field));
213                }
214                let empty = match wire.kind {
215                    CWire::Buffer => quote!(__PbBuffer {
216                        data: std::ptr::null(),
217                        len: 0
218                    }),
219                    CWire::Rows => quote!(__PbRows {
220                        data: std::ptr::null(),
221                        len: 0
222                    }),
223                    CWire::Pointer { .. } => quote!(std::ptr::null_mut()),
224                    _ => unreachable!(),
225                };
226                let local = format_ident!("owned_{i}");
227                drops.extend(quote!(let #local=std::mem::replace(&mut (*error).#field,#empty);));
228                (wire, quote!(#local))
229            })
230            .collect::<Vec<_>>();
231        drops.extend(sequence::release(
232            &plan.outcomes.success,
233            &values,
234            &context,
235            false,
236        ));
237        if fields.is_empty() {
238            fields.push(quote!(pub unused:u8));
239            header.push_str("    uint8_t unused;\n");
240        }
241        header.push_str(&format!("}} {c_name};\n/* Frees owned error fields and clears them; repeated calls are harmless. */\nvoid {drop_name}({c_name} *error);\n"));
242        let wires = crate::group::result_packet_fields(&plan.outcomes.success).collect::<Vec<_>>();
243        let mut comments = describe_groups(&plan.outcomes.success, &wires, true, &mut 0, "error");
244        for i in (0..wires.len()).rev() {
245            let field = if error.fields.len() == wires.len() {
246                error.fields[i].trim_start_matches("r#").to_owned()
247            } else {
248                format!("v{i}")
249            };
250            comments = comments.replace(&format!("output->v{i}"), &format!("error->{field}"));
251        }
252        header.push_str(&comments);
253        tokens.extend(quote! {
254            #[repr(C)] pub struct #rust_name {#(#fields),*}
255            #[allow(clippy::needless_borrow,clippy::let_unit_value,clippy::unnecessary_cast,clippy::useless_conversion)]
256            unsafe fn #helper(#input:#source_ty,error_output:*mut #rust_name,reported:&mut bool)->Result<(),String>{
257                if error_output.is_null(){return Ok(());}
258                #prepare
259            }
260            #[doc="# Safety\nThe pointer must be null or writable storage containing a live error produced by this library."]
261            #[no_mangle]pub unsafe extern "C" fn #drop_name(error:*mut #rust_name) {
262                if let Err(message)=::prebindgen_runtime::boundary(|| {if !error.is_null(){#drops}Ok(())}) {panic!("C error destructor failed: {}",message);}
263            }
264        });
265    }
266    Ok(())
267}

269#[allow(clippy::too_many_arguments)]
270pub(crate) fn result_function(
271    source_name: &syn::Ident,
272    error_ty: &syn::Type,
273    outputs: &[&Wire],
274    function_output_layout: &prebindgen_flat::WireNode<CWire, crate::CChain, crate::CBranch>,
275    mut params: Vec<TokenStream>,
276    mut cparams: Vec<String>,
277    body: &TokenStream,
278    context: &CContext,
279    prefix: &str,
280    pointers: &std::collections::BTreeMap<String, prebindgen_flat::TypeInfo<CType>>,
281    errors: &[&ErrorBinding],
282    consumes: bool,
283    comments: &str,
284    tokens: &mut TokenStream,
285    header: &mut String,
286) -> Result<()> {
287    let name = format_ident!("{prefix}_{source_name}");
288    let internal = &context.output_type;
289    let out = &context.output_pointer;
290    let accepted = &context.acknowledgement;
291    let error = &context.error_pointer;
292    let reported = &context.error_reported;
293    let typed = errors
294        .iter()
295        .find(|e| prebindgen_flat::type_key(&e.ty) == prebindgen_flat::type_key(error_ty));
296    let (rust_error, c_error, report_binding) = if let Some(typed) = typed {
297        let rust = format_ident!("__PbError_{}", typed.name);
298        let report = helper(&typed.name);
299        (
300            quote!(#rust),
301            format!("{prefix}_{}", typed.name),
302            quote! {
303                let mut delivered=false;
304                if ::prebindgen_runtime::boundary(|| unsafe {#report(<#error_ty as From<String>>::from(message),#error,&mut delivered)}).is_err() && !delivered {
305                // All CWire carriers have a valid zero representation with no
306                // owned allocations. Never expose uninitialized error storage.
307                std::ptr::write(#error,std::mem::zeroed());
308            }
309            },
310        )
311    } else {
312        (
313            quote!(*mut std::ffi::c_char),
314            "char *".into(),
315            quote!(unsafe {__pb_report_message(message,#error)};),
316        )
317    };
318    let niche = outputs
319        .iter()
320        .enumerate()
321        .find_map(|(i, w)| w.kind.available_niche().map(|n| (i, n)));
322    let return_position = niche.as_ref().map(|(i, _)| *i);
323    let (rust_return, c_return, failure, success) = if let Some((i, niche)) = &niche {
324        let wire = outputs[*i];
325        let ty = wire.rust_type();
326        let field = format_ident!("v{i}");
327        let failure = CCodec.absent_value(&wire.kind, Some(niche), false)?;
328        (
329            quote!(#ty),
330            c_ty(wire, prefix, pointers, context),
331            failure,
332            quote!(value.#field),
333        )
334    } else {
335        (quote!(bool), "bool".into(), quote!(false), quote!(true))
336    };
337    let mut checks = TokenStream::new();
338    let mut writes = TokenStream::new();
339    let mut display = Vec::new();
340    for (i, wire) in outputs.iter().enumerate() {
341        if return_position == Some(i) {
342            display.push("the return value".to_owned());
343            continue;
344        }
345        let id = format_ident!("out{i}");
346        let field = format_ident!("v{i}");
347        let ty = wire.rust_type();
348        params.push(quote!(#id:*mut #ty));
349        cparams.push(format!("{} *{id}", c_ty(wire, prefix, pointers, context)));
350        checks.extend(quote!(if #id.is_null(){return Err("null success-output pointer".into());}));
351        writes.extend(quote!(std::ptr::write(#id,value.#field);));
352        display.push(format!("*{id}"));
353    }
354    let acknowledgement = if consumes {
355        params.push(quote!(ownership:*mut bool));
356        cparams.push("bool *ownership".into());
357        quote!(if !ownership.is_null(){std::ptr::write(ownership,#accepted);})
358    } else {
359        quote!()
360    };
361    let preflight = if consumes {
362        quote!(if ownership.is_null() {
363            return Err("null ownership-output pointer".into());
364        })
365    } else {
366        quote!()
367    };
368    params.push(quote!(#error:*mut #rust_error));
369    cparams.push(format!("{c_error} *{error}"));
370    let mut comments = comments.to_owned();
371    for (i, path) in display.iter().enumerate().rev() {
372        comments = comments.replace(&format!("output->v{i}"), path);
373    }
374    header.push_str(&comments);
375    if let Some((i, niche)) = &niche {
376        let absent = match niche {
377            CNiche::Null => "NULL".to_owned(),
378            CNiche::Integer(value) => value.to_string(),
379        };
380        header.push_str(&format!(
381            "/* Result: success field v{i} is returned directly; {absent} means failure. */\n"
382        ));
383    } else {
384        header.push_str("/* Result: true = success; false = failure. Success outputs are unchanged on failure. */\n");
385    }
386    if consumes {
387        header.push_str("/* ownership is required: false before acceptance; true after active owned inputs transfer, including on failure. */\n");
388    }
389    header.push_str("/* error may be NULL to discard details. Failure initializes nonnull error with caller-owned data (or an empty record if error conversion fails). Success leaves error unchanged. Free a prior error before reusing its storage. All output/ownership/error storage must be disjoint. */\n");
390    header.push_str(&format!("{c_return} {name}({});\n", cparams.join(", ")));
391    let rollback = sequence::release(
392        function_output_layout,
393        &outputs
394            .iter()
395            .enumerate()
396            .map(|(i, wire)| {
397                let field = format_ident!("v{i}");
398                (*wire, quote!(value.#field))
399            })
400            .collect::<Vec<_>>(),
401        context,
402        true,
403    );
404    tokens.extend(quote! {
405        #[doc="# Safety\nSuccess, ownership and nonnull error outputs must be writable and pairwise disjoint, with no unreleased prior error. Input allocations must satisfy their declared provenance, aliasing and ownership contract."]
406        #[allow(clippy::needless_borrow,clippy::let_unit_value,clippy::unnecessary_cast,clippy::useless_conversion)]
407        #[no_mangle]pub unsafe extern "C" fn #name(#(#params),*)->#rust_return {
408            #[allow(unused_mut)]let mut #accepted=false;
409            #[allow(unused_mut)]let mut #reported=false;
410            let mut success_committed=false;
411            let mut storage=std::mem::MaybeUninit::<#internal>::uninit();
412            let #out=storage.as_mut_ptr();
413            let outcome=::prebindgen_runtime::boundary(|| {#preflight #checks #body});
414            #acknowledgement
415            match outcome {
416                Ok(())=>{let value=storage.assume_init();#writes #success}
417                Err(message)=>{
418                    if success_committed {let value=storage.assume_init();#rollback}
419                    if !#reported && !#error.is_null(){#report_binding}#failure
420                }
421            }
422        }
423    });
424    Ok(())
