5/// C conversions read representation and error declarations from Flat.
6pub(crate) struct EnumCodec;
7impl EnumCodec {
8    fn mapping<'a>(
9        &self,
10        flat: &'a Bindings,
11        ty: &syn::Type,
12    ) -> Option<(&'a str, &'a std::rc::Rc<EnumMapping>)> {
13        let ty = if let syn::Type::Reference(r) = ty {
14            &r.elem
15        } else {
16            ty
17        };
18        flat.type_info(ty)
19            .ok()?
20            .metadata
21            .enumeration
22            .as_ref()
23            .map(|e| (e.name.as_str(), &e.mapping))
24    }
25    fn wire(name: &str, mapping: &std::rc::Rc<EnumMapping>) -> CWire {
26        CWire::enumeration(name, mapping.clone())
27    }
28}

114    fn result_error_value(
115        &self,
116        flat: &prebindgen_flat::Flat<Self::Schema>,
117        ty: &syn::Type,
118        value: TokenStream,
119        message: Option<TokenStream>,
120        extra: &(),
121    ) -> Result<TokenStream> {
122        if let Some(name) = flat
123            .type_info(ty)
124            .ok()
125            .and_then(|t| t.metadata.error.as_ref())
126            .map(|e| e.name.as_str())
127        {
128            if message.is_some() {
129                return Err("structured C errors cannot also select a message mapper".into());
130            }
131            let helper = errors::helper(name);
132            let pointer = prebindgen_conversions::template_parameter("error_pointer");
133            let reported = prebindgen_conversions::template_parameter("error_reported");
134            Ok(
135                quote!({unsafe {#helper(#value,#pointer,&mut #reported)}?;#reported=true;String::new()}),
136            )
137        } else {
138            CCodec.result_error_value(flat, ty, value, message, extra)
139        }
140    }

168    fn consumes(&self, flat: &prebindgen_flat::Flat<Self::Schema>, ty: &syn::Type, e: &()) -> bool {
169        if let Some(inner) = prebindgen_declarations::option_inner(ty) {
170            return self.consumes(flat, inner, e);
171        }
172        self.mapping(flat, ty).is_none() && CCodec.consumes(flat, ty, e)
173    }
174    fn input_leaf(
175        &self,
176        flat: &prebindgen_flat::Flat<Self::Schema>,
177        ty: &syn::Type,
178        e: &(),
179        w: syn::Ident,
180        v: syn::Ident,
181        active: &Condition,
182    ) -> Result<InputPlan<CWire>> {
183        if prebindgen_declarations::option_inner(ty).is_some() {
184            return Err("resolve Optional operations before leaf compilation".into());
185        }
186        if let Some((name, m)) = self.mapping(flat, ty) {
187            return prebindgen_declarations::enum_input(m, ty, w, v, active, Self::wire(name, m));
188        }
189        CCodec.input_leaf(flat, ty, e, w, v, active)
190    }
191    fn output_leaf(
192        &self,
193        flat: &prebindgen_flat::Flat<Self::Schema>,
194        ty: &syn::Type,
195        e: &(),
196        w: syn::Ident,
197        v: TokenStream,
198        borrowed: bool,
199    ) -> Result<OutputPlan<CWire>> {
200        if prebindgen_declarations::option_inner(ty).is_some() {
201            return Err("resolve Optional operations before leaf compilation".into());
202        }
203        if let Some((name, m)) = self.mapping(flat, ty) {
204            if matches!(ty,syn::Type::Reference(r) if r.mutability.is_some()) {
205                return Err("mutable-reference outputs are unsupported".into());
206            }
207            return Ok(prebindgen_declarations::enum_output(
208                m,
209                w,
210                v,
211                Self::wire(name, m),
212            ));
213        }
214        CCodec.output_leaf(flat, ty, e, w, v, borrowed)
215    }

246    }

270        let mapping =
271            std::rc::Rc::new(EnumMapping::new(model.enums.values().next().unwrap()).unwrap());
272        for name in ["first", "updated"] {
