63/// A recipe for converting boundary inputs into one source argument.
64///
65/// Binding expands the recipe into wires and Rust conversion code.
66#[derive(Clone, Debug)]
67pub enum Input {
68    /// Resolve the registered default for this source occurrence during preparation.
69    Auto,
70    /// Apply a conversion to this source type, including inside Option/sequence wrappers.
71    ForType {
72        /// Source type targeted by this default or nested override.
73        ty: Type,
74        /// Conversion applied at the matching source occurrence.
75        input: Box<Input>,
76    },
77    /// Infer constructor argument conversions from its source signature.
78    Call(String),
79    /// Construct a foreign record value using its configured language representation.
80    Record(Vec<Input>),
81    /// Construct a collection by applying one recipe to every element.
82    Sequence(Box<Input>),
83    /// Convert an optional payload, gating every conversion phase.
84    Optional(Box<Input>),
85    /// Convert the occurrence type itself with [`Codec::input_leaf`].
86    Leaf,
87    /// Construct a named record from children in field order; an empty list builds unit.
88    Fields(Vec<Input>),
89    /// Call a source function after preparing its arguments with the child recipes.
90    Constructor {
91        /// Unqualified source function name; its result must match the occurrence type.
92        function: String,
93        /// Recipes in the constructor's parameter order.
94        args: Vec<Input>,
95    },
96    /// Choose an owned input using an `i32` arm index followed by all arms' wires.
97    /// Inactive arms are neither validated nor acquired; indices are zero-based.
98    Selector(Vec<Input>),
99}
100impl Input {
101    /// Associate a conversion with a source type for a default or nested override.
102    pub fn for_type(ty: Type, input: Self) -> Self {
103        Self::ForType {
104            ty,
105            input: Box::new(input),
106        }
107    }
108    /// Infer a source constructor's argument conversions from registered defaults.
109    pub fn call(name: impl Into<String>) -> Self {
110        Self::Call(name.into())
111    }
112    /// Add a constructor alternative to this conversion tree.
113    pub fn variant(self, name: impl Into<String>) -> Self {
114        self.add_alternative(Self::call(name))
115    }
116    /// Add a direct codec alternative, bypassing this type's expansion default.
117    pub fn variant_self(self) -> Self {
118        self.add_alternative(Self::Leaf)
119    }
120    fn add_alternative(self, arm: Self) -> Self {
121        match self {
122            Self::ForType { ty, input } => Self::for_type(ty, input.add_alternative(arm)),
123            Self::Selector(mut arms) => {
124                arms.push(arm);
125                Self::Selector(arms)
126            }
127            other => Self::Selector(vec![other, arm]),
128        }
129    }
130}
131impl Input {
132    /// Construct a record from field recipes, or unit from an empty list.
133    ///
134    /// `fields` holds one recipe per field of the source record, in declaration
135    /// order. Binding verifies the count against the record.
136    pub fn fields(fields: impl IntoIterator<Item = Input>) -> Self {
137        Self::Fields(fields.into_iter().collect())
138    }
139    /// Call a named source constructor with the supplied argument recipes.
140    ///
141    /// Binding checks its result type and argument count. Argument validation and
142    /// ownership acquisition precede the source call.
143    ///
144    /// # Arguments
145    ///
146    /// * `name` — unqualified name of a source function whose result type is the
147    ///   input's type.
148    /// * `args` — one recipe per parameter of that function, in order.
149    pub fn constructor(name: impl Into<String>, args: impl IntoIterator<Item = Input>) -> Self {
150        Self::Constructor {
151            function: name.into(),
152            args: args.into_iter().collect(),
153        }
154    }
155    /// Choose between owned representations of the same input type.
156    ///
157    /// `arms` holds one recipe per alternative; each must produce the input's
158    /// type. The caller passes the zero-based index of the chosen alternative in an
159    /// extra wire ahead of all arms' wires.
160    ///
161    /// Requires at least one arm. Each arm retains its own wires; this does not
162    /// merge equivalent wire parameters or generate destination-language overloads.
163    pub fn selector(arms: impl IntoIterator<Item = Input>) -> Self {
164        Self::Selector(arms.into_iter().collect())
165    }

167/// A result conversion recipe, interpreted together with its source type and codec input data.
168///
169/// `Fields` observes fields; `OwnedFields` moves them. Explicit `Projected`
170/// leaves select borrowed or owned conversion independently of type contents.
171#[derive(Clone, Debug)]
172pub enum Output {
173    /// Resolve the registered default for this source occurrence during preparation.
174    Auto,
175    /// Apply a conversion to this source type, including inside Option/sequence wrappers.
176    ForType {
177        /// Source type targeted by this default or nested override.
178        ty: Type,
179        /// Conversion applied at the matching source occurrence.
180        output: Box<Output>,
181    },
182    /// Resolve source fields by name, using defaults for unspecified fields.
183    NamedFields {
184        /// Per-field conversions; unspecified fields use registered defaults.
185        overrides: std::collections::BTreeMap<String, Output>,
186        /// Ownership of source-field decomposition.
187        mode: ProjectionMode,
188    },
189    /// Deliver a configured foreign record value with explicit field conversions.
190    Record {
191        /// Explicit source-field conversions in declaration order.
192        fields: Vec<Output>,
193        /// Ownership of the foreign-record projection.
194        mode: ProjectionMode,
195    },
196    /// Convert collection elements using ordinary or explicitly selected ownership.
197    Sequence {
198        /// Delivery selected before conversion compilation.
199        delivery: SequenceDelivery,
200        /// Element conversion, interpreted independently for each item.
201        element: Box<Output>,
202        /// `None` follows the occurrence's ownership: move owned values and observe
203        /// borrowed values. `Some` explicitly selects and checks a projection policy.
204        mode: Option<ProjectionMode>,
205    },
206    /// Deliver `Ok` through its recipe and route an owned `Err` through the codec.
207    /// An optional mapper takes exactly `E` or `&E` and returns `String`.
208    /// Without one, the codec must explicitly support preserving the error value.
209    /// The codec selects how that message reaches the destination's error channel.
210    Result {
211        /// Conversion of the owned success value.
212        ok: Box<Output>,
213        /// Optional unqualified source message mapper. Without one, the codec
214        /// must explicitly support the source error value itself.
215        error: Option<String>,
216    },
217    /// Convert an optional payload without preparing absent values.
218    Optional(Box<Output>),
219    /// Convert the occurrence type itself with [`Codec::output_leaf`].
220    Leaf,
221    /// Observe named fields in declaration order; an empty list represents unit.
222    Fields(Vec<Output>),
223    /// Move named fields into their output conversions.
224    OwnedFields(Vec<Output>),
225    /// Convert a leaf with an explicit ownership policy.
226    Projected(Projection),
227    /// Call a consuming accessor, then convert its owned result.
228    Into {
229        /// Source accessor taking exactly the owned occurrence type.
230        function: String,
231        /// Conversion of its owned result.
232        output: Box<Output>,
233    },
234    /// A shared-borrow accessor, followed by the output conversion of its result.
235    Getter {
236        /// Source getter taking exactly one shared reference to the current type.
237        function: String,
238        /// Conversion of the getter's owned result; borrowed results are unsupported.
239        output: Box<Output>,
240    },
241    /// Produce several views of one value. At most one child may consume it.
242    /// Observations run before consumption, while wire order follows this list.
243    Combine(Vec<Output>),
244}
245impl Output {
246    /// Associate a conversion with a source type for a default or nested override.
247    pub fn for_type(ty: Type, output: Self) -> Self {
248        Self::ForType {
249            ty,
250            output: Box::new(output),
251        }
252    }
253    /// Append another simultaneous output view.
254    pub fn view(self, view: Self) -> Self {
255        match self {
256            Self::ForType { ty, output } => Self::for_type(ty, output.view(view)),
257            Self::Combine(mut parts) => {
258                parts.push(view);
259                Self::Combine(parts)
260            }
261            other => Self::Combine(vec![other, view]),
262        }
263    }
264    /// Append a source getter whose result uses its registered conversion default.
265    pub fn field(self, name: impl Into<String>) -> Self {
266        self.view(Self::getter(name, Self::Auto))
267    }
268    /// Append the direct codec representation, bypassing this type's expansion default.
269    pub fn field_self(self) -> Self {
270        self.view(Self::Leaf)
271    }
272    /// Append an explicit borrowed projection.
273    pub fn field_self_ref(self) -> Self {
274        self.project_self(Projection::borrowed())
275    }
276    /// Append an explicit owned projection.
277    pub fn field_self_into(self) -> Self {
278        self.project_self(Projection::owned())
279    }
280    /// Append a projection with explicit ownership and nested field policies.
281    pub fn project_self(self, policy: Projection) -> Self {
282        self.view(Self::Projected(policy))
283    }
284    /// Append a consuming accessor followed by owned field decomposition.
285    pub fn fields_into(self, view: Self) -> Self {
286        let Self::Getter { function, output } = view else {
287            panic!("fields_into requires a fields! accessor")
288        };
289        let Self::NamedFields { overrides, .. } = *output else {
290            panic!("fields_into requires named fields")
291        };
292        self.view(Self::Into {
293            function,
294            output: Box::new(Self::NamedFields {
295                overrides,
296                mode: ProjectionMode::Owned,
297            }),
298        })
299    }
300    /// Deprecated spelling of [`Self::fields_into`].
301    #[deprecated(note = "use fields_into instead")]
302    pub fn fields_self_into(self, view: Self) -> Self {
303        self.fields_into(view)
304    }
305    /// Override one named source field, preserving defaults for its siblings.
306    pub fn field_override(mut self, name: impl Into<String>, output: Self) -> Self {
307        match &mut self {
308            Self::NamedFields { overrides, .. } => {
309                assert!(
310                    overrides.insert(name.into(), output).is_none(),
311                    "duplicate field override"
312                );
313            }
314            Self::Getter { output: inner, .. }
315            | Self::Into { output: inner, .. }
316            | Self::ForType { output: inner, .. } => {
317                let old = std::mem::replace(inner, Box::new(Self::Auto));
318                **inner = old.field_override(name, output);
319            }
320            _ => panic!("field override requires named field decomposition"),
321        }
322        self
323    }
324}
325impl Output {
326    /// Observe record fields in declaration order; use an empty list for unit.
327    ///
328    /// `fields` holds one recipe per field of the source record, in declaration order.
329    pub fn fields(fields: impl IntoIterator<Item = Output>) -> Self {
330        Self::Fields(fields.into_iter().collect())
331    }
332    /// Observe a value through a source getter and convert the getter's result.
333    ///
334    /// The getter must take exactly `&T` for the current type `T`, and return an
335    /// owned value. Binding rejects mismatched signatures.
336    ///
337    /// # Arguments
338    ///
339    /// * `name` — unqualified name of the source getter.
340    /// * `output` — recipe for the getter's result.
341    pub fn getter(name: impl Into<String>, output: Output) -> Self {
342        Self::Getter {
343            function: name.into(),
344            output: Box::new(output),
345        }
346    }
347    /// Produce multiple outputs from one value.
348    ///
349    /// `parts` holds one recipe per view of the value; their wires appear in this
350    /// order. Binding rejects an empty list or more than one consuming part. The
351    /// framework uses [`Codec::consumes`] to order observations before a move.
352    pub fn combine(parts: impl IntoIterator<Item = Output>) -> Self {
353        Self::Combine(parts.into_iter().collect())
354    }
355    fn consumes<C: Codec>(&self, ty: &Type, extra: &C::Extra, codec: &C, borrowed: bool) -> bool {
356        match self {
357            Self::Sequence { mode, .. } => match mode {
358                Some(mode) => *mode == ProjectionMode::Owned,
359                None => {
360                    !borrowed
361                        && sequence_parts(ty).is_some_and(|(_, view)| view == SequenceView::Owned)
362                }
363            },
364            Self::Optional(child) => {
365                child.consumes(option_inner(ty).unwrap_or(ty), extra, codec, borrowed)
366            }
367            Self::Leaf => codec.consumes(ty, extra),
368            Self::Projected(p) => p.mode == ProjectionMode::Owned,
369            Self::OwnedFields(_) | Self::Into { .. } | Self::Result { .. } => true,
370            Self::Record { mode, .. } | Self::NamedFields { mode, .. } => {
371                *mode == ProjectionMode::Owned
372            }
373            // Child compilation reports unresolved operations as errors, including
374            // when Combine first checks whether any child consumes its input.
375            Self::Auto | Self::ForType { .. } => false,
376            Self::Combine(p) => p.iter().any(|p| p.consumes(ty, extra, codec, borrowed)),
377            _ => false,
378        }
379    }
380}

382/// Language-specific leaf conversions. Recursive composition stays in this declaration compiler.
383///
384/// A leaf is a source type the adapter converts itself: a value copied across
385/// the boundary or an object kept in Rust. `prebindgen-conversions` provides helpers
386/// for both representations. The
387/// adapter chooses by type; a recipe only says where the leaves are.
388///
389/// Input leaves must validate without consuming in `validation`, then
390/// acquire ownership without failure or source calls in `acquire`. Each acquired
391/// value needs an ordinary Rust owner so later failures drop it. `statements`
392/// can construct arguments from those owners. Every input wire must retain
393/// the supplied activation condition. Output leaves keep ownership in Rust
394/// locals until their infallible `commit` runs at the selected protocol's
395/// transfer point. Buffered delivery commits after all foreign delivery succeeds;
396/// an explicit visitor protocol can transfer each prepared element at callback entry.
397///
398/// # Implementing a codec
399///
400/// This minimal codec uses Rust values as its carriers. It illustrates converter
401/// construction without a foreign ABI or object storage. A language adapter
402/// additionally writes wrappers and chooses native and destination-language
403/// carrier types for its wire kinds.
404///
405/// ```
406/// use prebindgen_flat::{Result, Wire, validate_bindings};
407/// use prebindgen_conversions::{value};
408/// use prebindgen_declarations::{
409///     Codec, CodecInput, CodecOutput, Declarations,
410///     Bindings, fun, expand_param, expand_return, bind_input, Input,
411/// };
412/// use proc_macro2::TokenStream;
413/// use quote::quote;
414/// use syn::{Ident, Type};
415///
416/// use prebindgen_declarations::Condition;
417/// fn carrier(kind: value::Value) -> Type {
418///     match kind {
419///         value::Value::Scalar(p) => p.rust_type(),
420///         value::Value::Text => syn::parse_quote!(String),
421///         value::Value::Bytes => syn::parse_quote!(Vec<u8>),
422///     }
423/// }
424/// struct RustCarriers;
425/// impl Codec for RustCarriers {
426///     type Metadata = ();
427///     type Niche = ();
428///     type Chain = ();
429///     type Branch = ();
430///     fn group_metadata(&self) {}
431///     type Extra = ();
432///     type Context = ();
433///     type Input = CodecInput<Self>;
434///     type Output = CodecOutput<Self>;
435///     type Wire = Type; // The Rust carrier type is the complete wire description.
436///
437///     fn optional_metadata(&self, _: Ident, _: Ident, _: Option<()>, _: &[Wire<Type>]) -> Result<()> {Ok(())}
438///     fn alternatives_metadata(&self, _: &Wire<Type>) -> Result<()> {Ok(())}
439///     fn field_group(&self, _: Ident) {}
440///     fn fields_group(&self) {}
441///     fn optional_niche(&self, _: &prebindgen_flat::WireNode<Self::Wire>) -> Option<prebindgen_declarations::NicheReservation<()>> {None}
442///     fn retire_niches(&self, _: &mut prebindgen_flat::WireNode<Self::Wire>, _: Option<&prebindgen_declarations::NicheReservation<()>>) -> Result<()> {Ok(())}
443///     fn niche(&self, _: &Self::Wire) -> Option<()> { None }
444///     fn set_niche(&self, _: &mut Self::Wire, _: Option<()>) {}
445///     fn selector(&self, _: &()) -> Self::Wire { syn::parse_quote!(i32) }
446///     fn consumes(&self, _: &Type, _: &()) -> bool { false }
447///     fn input_leaf(
448///         &self, ty: &Type, _: &(), wire: Ident, var: Ident, active: &Condition,
449///     ) -> Result<CodecInput<Self>> {
450///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
451///         let decode = quote!(#wire.clone());
452///         value::input(ty, wire, var, &active.rust(), decode, carrier(kind))
453///     }
454///     fn output_leaf(
455///         &self, ty: &Type, _: &(), wire: Ident, v: TokenStream, borrowed: bool,
456///     ) -> Result<CodecOutput<Self>> {
457///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
458///         value::output(ty, wire, v, borrowed, |v| v, carrier(kind))
459///     }
460/// }
461///
462/// let source = syn::parse_file(r#"
463///     pub fn from_i32(value: i32) -> i64 { value as i64 }
464///     pub fn to_i32(value: &i64) -> i32 { *value as i32 }
465///     pub fn echo(value: i64) -> i64 { value }
466/// "#)?;
467/// let mut flat = Bindings::<Type>::parse(
468///     source.items.into_iter().map(|item| (item, Default::default())),
469/// )?;
470/// let defaults = Declarations::new(&flat,
471///     vec![expand_param!(i64).variant("from_i32")],
472///     vec![expand_return!(i64).field("to_i32").field_self()],
473///     vec![],
474/// )?;
475/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
476/// let function = flat.function("echo")?;
477/// // Input: one constructor argument. Output: two projected values.
478/// assert_eq!(prebindgen_flat::wire_definitions(std::iter::once(&function.params[0].ty.slot.as_ref().unwrap().node)).count(), 1);
479/// assert_eq!(prebindgen_flat::wire_definitions(std::iter::once(&function.ret.slot.as_ref().unwrap().outcomes.success)).count(), 2);
480/// // Replace just this parameter's converter; later default filling preserves it.
481/// bind_input(&mut flat, &RustCarriers, "echo", "value", &Input::Leaf, &())?;
482/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
483/// validate_bindings(&flat)?;
484/// let input = flat.function("echo")?.params[0].ty.slot.as_ref().unwrap();
485/// assert_eq!(prebindgen_flat::wire_definitions(std::iter::once(&input.node)).count(), 1);
486/// assert_eq!(prebindgen_flat::type_key(&prebindgen_flat::wire_definitions(std::iter::once(&input.node)).next().unwrap().kind), "i64");
487/// assert!(input.resources.is_empty()); // No foreign ownership in this codec.
488/// # Ok::<(), Box<dyn std::error::Error>>(())
489/// ```
490#[allow(clippy::type_complexity)] // Explicit, independently typed metadata for each parent variant.
491pub trait Codec {
492    /// Metadata schema carried by the source and prepared binding model.
493    type Metadata: prebindgen_flat::Metadata;
494    /// Source-aware input normalization, called only during tree preparation.
495    /// Return an explicit operation tree for a language-specific representation.
496    fn expand_input<I, O>(
497        &self,
498        _flat: &Flat<I, O, Self::Metadata>,
499        _ty: &Type,
500        _tree: &Input,
501        _extra: &Self::Extra,
502    ) -> Result<Option<Input>> {
503        Ok(None)
504    }
505    /// Source-aware output normalization, called before conversion compilation.
506    fn expand_output<I, O>(
507        &self,
508        _flat: &Flat<I, O, Self::Metadata>,
509        _ty: &Type,
510        _tree: &Output,
511        _extra: &Self::Extra,
512    ) -> Result<Option<Output>> {
513        Ok(None)
514    }
515    /// Annotate an already compiled foreign-record input; structural expansion is complete.
516    fn input_record_metadata(
517        &self,
518        _ty: &Type,
519        _wire: &syn::Ident,
520        _active: &Condition,
521        _plan: &mut CodecInput<Self>,
522        _extra: &Self::Extra,
523    ) -> Result<()> {
524        Err("foreign record inputs are unsupported by this codec".into())
525    }
526    /// Annotate an already compiled foreign-record output with its selected representation.
527    fn output_record_metadata(
528        &self,
529        _ty: &Type,
530        _wire: &syn::Ident,
531        _plan: &mut CodecOutput<Self>,
532        _extra: &Self::Extra,
533    ) -> Result<()> {
534        Err("foreign record outputs are unsupported by this codec".into())
535    }
536    /// Adapter-owned description of an absent wire representation. Use `()` if unused.
537    /// The shared compiler never interprets this metadata.
538    type Niche: Clone;
539    /// Adapter metadata for parents with exactly one child description.
540    type Chain: Clone;
541    /// Adapter metadata for parents with an ordered list of child descriptions.
542    type Branch: Clone;
543    /// Describe an ordered set of conversion components. Constructors, output
544    /// accessors and combinations retain this group with one component. Constructors
545    /// retain an empty group when there are no arguments. Source
546    /// record fields instead use [`Codec::fields_group`].
547    fn group_metadata(&self) -> Self::Branch;
548    /// Describe an optional value in the adapter's own metadata format.
549    /// `presence` is the normalized 0/1 binding used by payload conditions.
550    /// `discriminator` names a descendant wire; `niche` is its reserved absent
551    /// representation, or `None` when a separate flag is an inline control wire.
552    /// With a flag, `presence` and `discriminator` name that same wire. The
553    /// supplied controls stay in the tree. Borrow them to validate the encoding;
554    /// metadata must not retain wire descriptors. Return an error when they cannot form the
555    /// adapter's flag or niche representation; failed preparation preserves existing slots.
556    fn optional_metadata(
557        &self,
558        presence: syn::Ident,
559        discriminator: syn::Ident,
560        niche: Option<Self::Niche>,
561        controls: &[Wire<Self::Wire>],
562    ) -> Result<Self::Chain>;
563    /// Record the supplied selector for mutually exclusive input alternatives,
564    /// preserving its niche state in the tree. Metadata must not retain the wire.
565    /// Reject an unsupported tag representation.
566    fn alternatives_metadata(&self, selector: &Wire<Self::Wire>) -> Result<Self::Branch>;
567    /// Describe one named source field, whose subtree may contain any number of wires.
568    fn field_group(&self, name: syn::Ident) -> Self::Branch;
569    /// Describe a record or unit value's ordered collection of field subtrees.
570    /// The shared helpers check field-name uniqueness before calling this hook.
571    fn fields_group(&self) -> Self::Branch;
572
573    /// Input data borrowed while building one occurrence's conversion; use `()` if unused.
574    /// The same value is passed through nested conversions. It is not stored in Flat;
575    /// hooks retain only the data needed for emission in their code and wires.
576    type Extra;
577    /// Context supplied by the language generator when emitting stored templates.
578    type Context: prebindgen_conversions::TemplateContext + 'static;
579    /// Final self-contained input converter installed after preparation.
580    type Input: prebindgen_flat::InputConversion + From<CodecInput<Self>>;

894/// A flat model whose slots contain only conversion data for wrapper emission.
895///
896/// Unbound source helpers have `None` slots. Bound functions need `Some` for their
897/// result and every parameter; [`prebindgen_flat::validate_bindings`] checks this before emission.
898/// Record fields retain their canonical source information and metadata. Descriptions and [`Codec::Extra`]
899/// are construction inputs, not stored in this model.
900///
901/// Output conversion data cannot be attached to an input position:
902/// ```compile_fail
903/// use prebindgen_declarations::Bindings;
904/// fn swap(flat: &mut Bindings<syn::Type>) {
905///     let function = flat.function_mut("foo").unwrap();
906///     function.params[0].ty.slot = function.ret.slot.clone();
907/// }
908/// ```
909/// Nor can a recipe be stored in a slot:
910/// ```compile_fail
911/// use prebindgen_declarations::{Bindings, Input};
912/// fn configure(flat: &mut Bindings<syn::Type>) {
913///     flat.function_mut("foo").unwrap().params[0].ty.slot = Some(Input::Leaf);
914/// }
915/// ```
916pub type Bindings<W, C = (), N = (), B = (), M = ()> =
917    Flat<InputPlan<W, C, N, B>, OutputPlan<W, C, N, B>, M>;

919/// Build and attach a converter to a named function parameter.
920///
921/// Resolves the occurrence's source type from `flat`, checks recipe compatibility,
922/// and recursively composes wires and code. Nested conversions use the supplied
923/// `extra`, not data from referenced constructors or getters. Recipes and `extra`
924/// are borrowed only for this call. A failure leaves the existing slot unchanged.
925/// Rebinding an occurrence replaces its converter without changing other slots.
926/// Generated names are scoped by parameter position, so separate calls cannot
927/// collide within the wrapper. Keep the function signature fixed after binding.
928/// Check completeness with [`prebindgen_flat::validate_bindings`].
929///
930/// # Arguments
931///
932/// * `flat` — the model holding the function; the parameter's slot is replaced.
933/// * `codec` — the adapter's leaf conversions.
934/// * `function` — unqualified name of the source function.
935/// * `parameter` — name of the parameter, as written in the source.
936/// * `recipe` — how to build the argument from boundary inputs.
937/// * `extra` — occurrence data passed to every codec call for this conversion.
938///
939/// Input binding rejects output recipes:
940/// ```compile_fail
941/// use prebindgen_declarations::{bind_input, Codec, Output};
942/// use prebindgen_flat::Flat;
943/// fn wrong<C: Codec>(flat: &mut Flat<C::Input, C::Output, C::Metadata>, codec: &C, extra: &C::Extra) {
944///     bind_input(flat, codec, "foo", "x", &Output::Leaf, extra).unwrap();
945/// }
946/// ```
947pub fn bind_input<C: Codec>(
948    flat: &mut Flat<C::Input, C::Output, C::Metadata>,
949    codec: &C,
950    function: &str,
951    parameter: &str,
952    recipe: &Input,
953    extra: &C::Extra,
954) -> Result<()> {
955    let f = flat.function(function)?;
956    let index = f
957        .params
958        .iter()
959        .position(|p| p.name == parameter)
960        .ok_or_else(|| format!("`{function}`: unknown parameter `{parameter}`"))?;
961    let defaults = Declarations::new(flat, vec![], vec![], vec![])?;
962    let recipe = defaults.resolve_input(flat, codec, &f.params[index].ty.ty, recipe, extra)?;
963    let mut conversion = Compiler::new(flat, codec, format!("p{index}")).input(
964        &f.params[index].ty.ty,
965        &recipe,
966        extra,
967        &Condition::default(),
968    )?;
969
970    codec.finish_input(&mut conversion, extra)?;
971    codec.validate_input(&conversion)?;
972    flat.function_mut(function)?.params[index].ty.slot = Some(conversion.into());
973    Ok(())
974}

976/// Build and attach a converter to a function result, replacing any existing one.
977///
978/// Uses the same construction and failure rules as [`bind_input`]. Output code
979/// uses [`prebindgen_conversions::source_result_expr`] for the source call's result local.
980/// The recipe is checked for compatible types and signatures, and for repeated
981/// consumption of the same value. Unit results use `Output::fields([])`.
982///
983/// # Arguments
984///
985/// * `flat` — the model holding the function; the result's slot is replaced.
986/// * `codec` — the adapter's leaf conversions.
987/// * `function` — unqualified name of the source function.
988/// * `recipe` — how to deliver the result as boundary outputs.
989/// * `extra` — occurrence data passed to every codec call for this conversion.
990///
991/// Output binding rejects input recipes:
992/// ```compile_fail
993/// use prebindgen_declarations::{bind_output, Codec, Input};
994/// use prebindgen_flat::Flat;
995/// fn wrong<C: Codec>(flat: &mut Flat<C::Input, C::Output, C::Metadata>, codec: &C, extra: &C::Extra) {
996///     bind_output(flat, codec, "foo", &Input::Leaf, extra).unwrap();
997/// }
998/// ```
999pub fn bind_output<C: Codec>(
1000    flat: &mut Flat<C::Input, C::Output, C::Metadata>,
1001    codec: &C,
1002    function: &str,
1003    recipe: &Output,
1004    extra: &C::Extra,
1005) -> Result<()> {
1006    let defaults = Declarations::new(flat, vec![], vec![], vec![])?;
1007    let recipe =
1008        defaults.resolve_output(flat, codec, &flat.function(function)?.ret.ty, recipe, extra)?;
1009    let mut conversion = Compiler::new(flat, codec, "ret".into()).output_root(
1010        &flat.function(function)?.ret.ty,
1011        &recipe,
1012        extra,
1013        prebindgen_conversions::source_result_expr(),
1014        false,
1015    )?;
1016    finalize_output_schema(codec, &mut conversion, extra)?;
1017    codec.finish_output(&mut conversion, extra)?;
1018    codec.validate_output(&conversion)?;
1019    flat.function_mut(function)?.ret.slot = Some(conversion.into());
1020    Ok(())
1021}

1034/// Compile resolved Input/Output slots without consulting expansion defaults.
1035/// Canonical source declarations and metadata stay in the same Flat as slots
1036/// change type. Builders and default environments are not needed after resolution.
1037/// Raw Auto/ForType/Call operations are rejected: resolve them before this stage.
1038pub fn compile<C: Codec>(
1039    flat: Flat<Input, Output, C::Metadata>,
1040    codec: &C,
1041    extra: &C::Extra,
1042) -> Result<Flat<C::Input, C::Output, C::Metadata>> {
1043    let mut plans = std::collections::BTreeMap::new();
1044    for (name, function) in &flat.functions {
1045        let Some(output) = &function.ret.slot else {
1046            continue;
1047        };
1048        let (inputs, output) = (|| -> Result<_> {
1049            let mut inputs = Vec::new();
1050            for (index, param) in function.params.iter().enumerate() {
1051                let tree = param
1052                    .ty
1053                    .slot
1054                    .as_ref()
1055                    .ok_or_else(|| format!("missing resolved input `{}`", param.name))?;
1056                let mut plan = Compiler::new(&flat, codec, format!("p{index}")).input(
1057                    &param.ty.ty,
1058                    tree,
1059                    extra,
1060                    &Condition::default(),
1061                )?;
1062                codec.finish_input(&mut plan, extra)?;
1063                codec.validate_input(&plan)?;
1064                inputs.push(plan.into());
1065            }
1066            let mut output = Compiler::new(&flat, codec, "ret".into()).output_root(
1067                &function.ret.ty,
1068                output,
1069                extra,
1070                prebindgen_conversions::source_result_expr(),
1071                false,
1072            )?;
1073            finalize_output_schema(codec, &mut output, extra)?;
1074            codec.finish_output(&mut output, extra)?;
1075            codec.validate_output(&output)?;
1076            Ok((inputs, output.into()))
1077        })()
1078        .map_err(|error| format!("{name}: {error}"))?;
1079        plans.insert(name.clone(), (inputs, output));
1080    }
1081    let mut flat = flat.map_slots(|_, _| Ok(None), |_, _| Ok(None))?;
1082    for (name, (inputs, output)) in plans {
1083        let function = flat.function_mut(&name)?;
1084        for (param, input) in function.params.iter_mut().zip(inputs) {
1085            param.ty.slot = Some(input);
1086        }
1087        function.ret.slot = Some(output);
1088    }
1089    prebindgen_flat::validate_bindings(&flat)?;
1090    Ok(flat)
1091}

1098}
1099impl<'a, C: Codec, I, O> Compiler<'a, C, I, O> {
1100    fn new(flat: &'a Flat<I, O, C::Metadata>, codec: &'a C, prefix: String) -> Self {
1101        Self {
1102            flat,
1103            codec,
1104            prefix,
1105            next: std::cell::Cell::new(0),
1106        }
1107    }
1108    fn name(&self) -> syn::Ident {
1109        let next = self.next.get();
1110        let n = format_ident!("{}_w{next}", self.prefix);
1111        self.next.set(next + 1);
1112        n
1113    }
1114    #[allow(clippy::type_complexity)] // Preserve the codec metadata types while walking its tree.
1115    fn prefix(
1116        &self,
1117        wires: &mut prebindgen_flat::WireNode<C::Wire, C::Chain, C::Branch>,
1118        segment: ComponentSegment,
1119    ) {
1120        self.codec.prefix_layout(wires, segment);
1121    }
1122    fn input(
1123        &self,
1124        ty: &Type,
1125        recipe: &Input,
1126        extra: &C::Extra,
1127        active: &Condition,
1128    ) -> Result<CodecInput<C>> {
1129        self.input_named(ty, recipe, extra, active, None)
1130    }
1131    fn input_named(
1132        &self,
1133        ty: &Type,
1134        recipe: &Input,
1135        extra: &C::Extra,
1136        active: &Condition,
1137        names: Option<(&syn::Ident, &syn::Ident)>,
1138    ) -> Result<CodecInput<C>> {
1139        if contains_result(ty) {
1140            return Err("Result inputs require an explicit data representation; error routing is output-only".into());
1141        }
1142        let var = names
1143            .map(|(_, var)| var.clone())
1144            .unwrap_or_else(|| self.name());
1145        let mut wires = Vec::new();
1146        let mut fallible = false;
1147        let mut validation = TokenStream::new();
1148        let mut acquire = TokenStream::new();
1149        let mut resources = Vec::new();
1150        let mut acknowledgement = None;
1151        let mut statements = TokenStream::new();
1152        let argument = quote!(#var);
1153        match recipe {
1154            Input::Sequence(element_recipe) => {
1155                let (element, _) =
1156                    sequence_parts(ty).ok_or("sequence input requires Vec<T>, &[T] or &Vec<T>")?;
1157                let wire = self.name();
1158                let length = self.name();
1159                let index = self.name();
1160                let child = self.input(element, element_recipe, extra, &Condition::default())?;
1161                return sequence::input(
1162                    self.codec, ty, extra, wire, var, length, index, active, child,
1163                );
1164            }
1165            Input::Optional(inner) => {
1166                let ty = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
1167                let tag = names
1168                    .map(|(wire, _)| format_ident!("{wire}_present"))
1169                    .unwrap_or_else(|| self.name());
1170                let child = if let Some((wire, var)) = names {
1171                    self.input_named(
1172                        ty,
1173                        inner,
1174                        extra,
1175                        &active.arm(tag.clone(), 1),
1176                        Some((&format_ident!("{wire}_some"), &format_ident!("{var}_some"))),
1177                    )?
1178                } else {
1179                    self.input(ty, inner, extra, &active.arm(tag.clone(), 1))?
1180                };
1181                return optional_input(self.codec, extra, tag, var, active, child);
1182            }
1183            Input::Leaf => {
1184                let w = names
1185                    .map(|(wire, _)| wire.clone())
1186                    .unwrap_or_else(|| self.name());
1187                return self.codec.input_leaf(ty, extra, w, var, active);
1188            }
1189            Input::Auto | Input::ForType { .. } | Input::Call(_) => {
1190                return Err("resolve Input operations before compilation".into())
1191            }
1192            Input::Fields(children) | Input::Record(children) => {
1193                if type_key(ty) == "()" && children.is_empty() {
1194                    return Ok(InputPlan {
1195                        fallible: false,
1196                        context: Default::default(),
1197                        acknowledgement,
1198                        node: WireNode::Branch {
1199                            controls: vec![],
1200                            children: vec![],
1201                            meta: self.codec.fields_group(),
1202                        },
1203                        validation,
1204                        acquire,
1205                        resources,
1206                        statements,
1207                        argument: quote!(()),
1208                    });
1209                }
1210                let r = self.flat.record(ty)?;
1211                if r.fields.len() != children.len() {
1212                    return Err(format!(
1213                        "`{}`: expected {} field converters, got {}",
1214                        type_key(ty),
1215                        r.fields.len(),
1216                        children.len()
1217                    ));
1218                }
1219                let grouped = matches!(recipe, Input::Record(_));
1220                let wire =
1221                    grouped.then(|| names.map(|(w, _)| w.clone()).unwrap_or_else(|| self.name()));
1222                let mut plan = record_input(
1223                    self.codec,
1224                    ty,
1225                    var.clone(),
1226                    r.fields
1227                        .iter()
1228                        .zip(children)
1229                        .enumerate()
1230                        .map(|(index, (field, child))| {
1231                            let plan = if let Some(wire) = &wire {
1232                                self.input_named(
1233                                    &field.ty,
1234                                    child,
1235                                    extra,
1236                                    active,
1237                                    Some((
1238                                        &format_ident!("{wire}_f{index}"),
1239                                        &format_ident!("{var}_f{index}"),
1240                                    )),
1241                                )?
1242                            } else {
1243                                self.input(&field.ty, child, extra, active)?
1244                            };
1245                            Ok((field.name.clone(), plan))
1246                        }),
1247                )?;
1248                if let Some(wire) = wire {
1249                    self.codec
1250                        .input_record_metadata(ty, &wire, active, &mut plan, extra)?;
1251                }
1252                return Ok(plan);
1253            }
1254            Input::Constructor { function, args } => {
1255                let f = self.flat.function(function)?;
1256                if !same_type(&f.ret.ty, ty) {
1257                    return Err(format!(
1258                        "constructor `{function}` returns `{}`, expected `{}`",
1259                        type_key(&f.ret.ty),
1260                        type_key(ty)
1261                    ));
1262                }
1263                if f.params.len() != args.len() {
1264                    return Err(format!("constructor `{function}`: wrong argument count"));
1265                }
1266                let mut call_args = Vec::new();
1267                for (param, child) in f.params.iter().zip(args) {
1268                    let mut p = self.input(&param.ty.ty, child, extra, active)?;
1269                    self.prefix(
1270                        &mut p.node,
1271                        ComponentSegment::Argument(param.name.to_string()),
1272                    );
1273                    fallible |= p.fallible;
1274                    validation.extend(p.validation);
1275                    acquire.extend(p.acquire);
1276                    resources.extend(p.resources);
1277                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
1278                    statements.extend(p.statements);
1279                    wires.push(p.node);
1280                    call_args.push(p.argument);
1281                }
1282                let path = &f.path;
1283                statements.extend(quote!(let #var:#ty=#path(#(#call_args),*);));
1284                wires = vec![WireNode::Branch {
1285                    controls: vec![],
1286                    children: wires,
1287                    meta: self.codec.group_metadata(),
1288                }];
1289            }
1290            Input::Selector(arms) => {
1291                fallible = true;
1292                if arms.is_empty() {
1293                    return Err("an input selector needs at least one alternative".into());
1294                }
1295                if matches!(ty, Type::Reference(_)) {
1296                    return Err("selector results must be owned in this subset".into());
1297                }
1298                let tag = self.name();
1299                let mut selector_layout = prebindgen_flat::WireNode::Leaf(vec![Wire {
1300                    name: tag.clone(),
1301                    kind: self.codec.selector(extra),
1302                }]);
1303                self.prefix(&mut selector_layout, ComponentSegment::Selector);
1304                let WireNode::Leaf(mut selector_values) = selector_layout else {
1305                    return Err("selector prefixing must retain its value".into());
1306                };
1307                if selector_values.len() != 1 {
1308                    return Err("selector prefixing must retain its value".into());
1309                }
1310                let selector = selector_values.remove(0);
1311                let mut alternatives = Vec::new();
1312                let count = arms.len() as i32;
1313                let invalid = active.and(quote!(!(0..#count).contains(&#tag)));
1314                validation.extend(quote!(if #invalid {return Err("invalid selector".into());}));
1315                let mut cases = Vec::new();
1316                for (i, arm) in arms.iter().enumerate() {
1317                    let mut p = self.input(ty, arm, extra, &active.arm(tag.clone(), i as i32))?;
1318                    self.prefix(&mut p.node, ComponentSegment::Alternative(i));
1319                    self.codec.retire_niches(&mut p.node, None)?;
1320                    fallible |= p.fallible;
1321                    validation.extend(p.validation);
1322                    acquire.extend(p.acquire);
1323                    resources.extend(p.resources);
1324                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
1325                    let body = p.statements;
1326                    let value = p.argument;
1327                    alternatives.push(p.node);
1328                    let i = i as i32;
1329                    cases.push(quote!(#i=>{#body #value}));
1330                }
1331                let metadata = self.codec.alternatives_metadata(&selector)?;
1332                let node = WireNode::Branch {
1333                    controls: vec![selector],
1334                    meta: metadata,
1335                    children: alternatives,
1336                };
1337                wires = vec![node];
1338                statements = quote!(let #var:#ty=match #tag{#(#cases,)* _=>unreachable!("validated selector")};);
1339            }
1340        }
1341        Ok(InputPlan {
1342            fallible,
1343            context: Default::default(),
1344            acknowledgement,
1345            node: wires.remove(0),
1346            validation,
1347            acquire,
1348            resources,
1349            statements,
1350            argument,
1351        })
1352    }
1353    fn output_root(
1354        &self,
1355        ty: &Type,
1356        recipe: &Output,
1357        extra: &C::Extra,
1358        value: TokenStream,
1359        borrowed: bool,
1360    ) -> Result<CodecOutput<C>> {
1361        if let Output::Result { ok, error } = recipe {
1362            if borrowed {
1363                return Err("Result error routing requires an owned Result occurrence".into());
1364            }
1365            let (success_ty, error_ty) =
1366                result_parts(ty).ok_or("Result recipe requires Result<T, E>")?;
1367            let payload = self.name();
1368            let failure = self.name();
1369            let mapped = if let Some(error) = error {
1370                let mapper = self.flat.function(error)?;
1371                let mapper_arg = mapper.params.first().map(|p| &p.ty.ty);
1372                let shared = matches!(mapper_arg, Some(Type::Reference(r)) if r.mutability.is_none() && same_type(&r.elem, error_ty));
1373                if mapper.params.len() != 1
1374                    || !(mapper_arg.is_some_and(|ty| same_type(ty, error_ty)) || shared)
1375                    || type_key(&mapper.ret.ty) != "String"
1376                {
1377                    return Err(
1378                        "Result error mapper must take exactly E or &E and return String".into(),
1379                    );
1380                }
1381                let path = &mapper.path;
1382                let argument = if shared {
1383                    quote!(&#failure)
1384                } else {
1385                    quote!(#failure)
1386                };
1387                Some(quote!(#path(#argument)))
1388            } else {
1389                None
1390            };
1391            let mapped_error = mapped.is_some();
1392            let message =
1393                self.codec
1394                    .result_error_value(error_ty, quote!(#failure), mapped, extra)?;
1395            let mut child = self.output(success_ty, ok, extra, quote!(#payload), false)?;
1396            let prepare = child.statements;
1397            child.statements = quote! {
1398                let #payload = match #value {
1399                    ::core::result::Result::Ok(value) => value,
1400                    ::core::result::Result::Err(#failure) => return Err(#message),
1401                };
1402                #prepare
1403            };
1404
1405            let error = self
1406                .codec
1407                .source_error_layout(error_ty, mapped_error, extra)?;
1408            child.outcomes.source_error = Some(error);
1409            return Ok(child);
1410        }
1411        self.output(ty, recipe, extra, value, borrowed)
1412    }
1413    fn output(
1414        &self,
1415        ty: &Type,
1416        recipe: &Output,
1417        extra: &C::Extra,
1418        value: TokenStream,
1419        borrowed: bool,
1420    ) -> Result<CodecOutput<C>> {
1421        self.output_named(ty, recipe, extra, value, borrowed, None)
1422    }
1423    fn output_named(
1424        &self,
1425        ty: &Type,
1426        recipe: &Output,
1427        extra: &C::Extra,
1428        value: TokenStream,
1429        borrowed: bool,
1430        wire_name: Option<&syn::Ident>,
1431    ) -> Result<CodecOutput<C>> {
1432        if contains_result(ty) {
1433            return Err("Result error routing is supported only at the outermost return with an explicit mapper".into());
1434        }
1435        let mut out = OutputPlan {
1436            fallible: false,
1437            context: Default::default(),
1438            delivery: Default::default(),
1439            commit: TokenStream::new(),
1440            outcomes: prebindgen_flat::Outcomes {
1441                success: WireNode::Branch {
1442                    controls: vec![],
1443                    children: vec![],
1444                    meta: self.codec.group_metadata(),
1445                },
1446                source_error: None,
1447                binding_failure: None,
1448            },
1449            statements: TokenStream::new(),
1450            values: Vec::new(),
1451        };
1452        match recipe {
1453            Output::Sequence {
1454                element,
1455                mode,
1456                delivery,
1457            } => {
1458                let (element_type, _) =
1459                    sequence_parts(ty).ok_or("sequence output requires a collection")?;
1460                let mode = mode.unwrap_or_else(|| {
1461                    if borrowed || sequence_parts(ty).unwrap().1 != SequenceView::Owned {
1462                        ProjectionMode::Borrowed
1463                    } else {
1464                        ProjectionMode::Owned
1465                    }
1466                });
1467                let wire = self.name();
1468                let payload = self.name();
1469                let length = self.name();
1470                let index = self.name();
1471                let child = self.output(
1472                    element_type,
1473                    element,
1474                    extra,
1475                    quote!(#payload),
1476                    mode == ProjectionMode::Borrowed,
1477                )?;
1478                return sequence::output(
1479                    self.codec, ty, extra, wire, payload, length, index, value, borrowed, mode,
1480                    child, *delivery,
1481                );
1482            }
1483            Output::Result { .. } => {
1484                return Err("Result error routing is supported only at the outermost return".into())
1485            }
1486            Output::Optional(inner) => {
1487                let ty = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
1488                let tag = wire_name
1489                    .map(|w| format_ident!("{w}_present"))
1490                    .unwrap_or_else(|| self.name());
1491                let payload = wire_name
1492                    .map(|w| format_ident!("{w}_payload"))
1493                    .unwrap_or_else(|| self.name());
1494                let child_wire = wire_name.map(|w| format_ident!("{w}_some"));
1495                let child = self.output_named(
1496                    ty,
1497                    inner,
1498                    extra,
1499                    quote!(#payload),
1500                    borrowed,
1501                    child_wire.as_ref(),
1502                )?;
1503                return optional_output(self.codec, extra, tag, payload, value, borrowed, child);
1504            }
1505            Output::Auto | Output::ForType { .. } | Output::NamedFields { .. } => {
1506                return Err("resolve Output operations before compilation".into())
1507            }
1508            Output::Leaf => {
1509                let w = wire_name.cloned().unwrap_or_else(|| self.name());
1510                return self.codec.output_leaf(ty, extra, w, value, borrowed);
1511            }
1512            Output::Projected(projection) => {
1513                if borrowed && projection.mode == ProjectionMode::Owned {
1514                    return Err(
1515                        "owned projection cannot consume a shared-reference occurrence".into(),
1516                    );
1517                }
1518                let value = if !borrowed && projection.mode == ProjectionMode::Borrowed {
1519                    quote!(&(#value))
1520                } else {
1521                    value
1522                };
1523                return self.codec.output_projected(
1524                    ty,
1525                    extra,
1526                    wire_name.cloned().unwrap_or_else(|| self.name()),
1527                    value,
1528                    projection,
1529                );
1530            }
1531            Output::Fields(children)
1532            | Output::OwnedFields(children)
1533            | Output::Record {
1534                fields: children, ..
1535            } => {
1536                let grouped = matches!(recipe, Output::Record { .. });
1537                let wire = grouped.then(|| wire_name.cloned().unwrap_or_else(|| self.name()));
1538                let owned = matches!(
1539                    recipe,
1540                    Output::OwnedFields(_)
1541                        | Output::Record {
1542                            mode: ProjectionMode::Owned,
1543                            ..
1544                        }
1545                );
1546                if owned && borrowed {
1547                    return Err("owned fields cannot consume a shared-reference occurrence".into());
1548                }
1549                if type_key(ty) == "()" && children.is_empty() {
1550                    out.outcomes.success = WireNode::Branch {
1551                        controls: vec![],
1552                        meta: self.codec.fields_group(),
1553                        children: vec![],
1554                    };
1555
1556                    return Ok(out);
1557                }
1558                let r = self.flat.record(ty)?;
1559                if r.fields.len() != children.len() {
1560                    return Err("wrong output field count".into());
1561                }
1562                let convert = |index: usize, value| {
1563                    let child_wire = wire.as_ref().map(|w| format_ident!("{w}_f{index}"));
1564                    self.output_named(
1565                        &r.fields[index].ty,
1566                        &children[index],
1567                        extra,
1568                        value,
1569                        !owned,
1570                        child_wire.as_ref(),
1571                    )
1572                };
1573                let var = wire
1574                    .as_ref()
1575                    .map(|w| format_ident!("{w}_record"))
1576                    .unwrap_or_else(|| self.name());
1577                let mut plan = if owned {
1578                    record_output_owned(
1579                        self.codec,
1580                        var,
1581                        value,
1582                        r.fields.iter().map(|f| f.name.clone()),
1583                        convert,
1584                    )?
1585                } else {
1586                    record_output(
1587                        self.codec,
1588                        var,
1589                        value,
1590                        r.fields.iter().map(|f| f.name.clone()),
1591                        convert,
1592                    )?
1593                };
1594                if let Some(wire) = wire {
1595                    self.codec
1596                        .output_record_metadata(ty, &wire, &mut plan, extra)?;
1597                }
1598                return Ok(plan);
1599            }
1600            Output::Getter { function, output } | Output::Into { function, output } => {
1601                let consuming = matches!(recipe, Output::Into { .. });
1602                if consuming && borrowed {
1603                    return Err(
1604                        "consuming accessor cannot consume a shared-reference occurrence".into(),
1605                    );
1606                }
1607                let f = self.flat.function(function)?;
1608                let expected: Type = if consuming {
1609                    ty.clone()
1610                } else {
1611                    syn::parse_quote!(&#ty)
1612                };
1613                if f.params.len() != 1 || !same_type(&f.params[0].ty.ty, &expected) {
1614                    return Err(format!(
1615                        "getter `{function}` must take exactly `{}`",
1616                        type_key(&expected)
1617                    ));
1618                }
1619                if matches!(f.ret.ty, Type::Reference(_)) {
1620                    return Err("borrowed getter results are outside this subset".into());
1621                }
1622                let v = self.name();
1623                let path = &f.path;
1624                let arg = if borrowed || consuming {
1625                    value
1626                } else {
1627                    quote!(&#value)
1628                };
1629                out.statements = quote!(let #v=#path(#arg););
1630                let mut child = self.output(&f.ret.ty, output, extra, quote!(#v), false)?;
1631                self.prefix(
1632                    &mut child.outcomes.success,
1633                    ComponentSegment::Getter(function.clone()),
1634                );
1635                append(&mut out, child)?;
1636            }
1637            Output::Combine(parts) => {
1638                if parts.is_empty() {
1639                    return Err("an output combination needs at least one part".into());
1640                }
1641                if parts
1642                    .iter()
1643                    .filter(|p| p.consumes(ty, extra, self.codec, borrowed))
1644                    .count()
1645                    > 1
1646                {
1647                    return Err("a combination cannot consume the same value twice".into());
1648                }
1649                let mut bodies = Vec::new();
1650                for (index, child) in parts.iter().enumerate() {
1651                    let consuming = child.consumes(ty, extra, self.codec, borrowed);
1652                    let arg = if borrowed || consuming {
1653                        value.clone()
1654                    } else {
1655                        quote!(&#value)
1656                    };
1657                    let mut p = self.output(ty, child, extra, arg, borrowed || !consuming)?;
1658                    self.prefix(&mut p.outcomes.success, ComponentSegment::View(index));
1659                    bodies.push((consuming, std::mem::take(&mut p.statements)));
1660                    append(&mut out, p)?;
1661                }
1662                // Wire order follows the declaration. Observations execute before the move.
1663                bodies.sort_by_key(|(consumes, _)| *consumes);
1664                for (_, body) in bodies {
1665                    out.statements.extend(body);
1666                }
1667            }
1668        }
1669
1670        Ok(out)
1671    }
1672}
1673fn validate_nested_output<W: prebindgen_flat::WireKind, C, N, B>(
1674    plan: &OutputPlan<W, C, N, B>,
1675) -> Result<()> {
1676    if plan.outcomes.source_error.is_some() || plan.outcomes.binding_failure.is_some() {
1677        return Err("completion channels belong to the outermost output conversion".into());
1678    }
1679    Ok(())
1680}
1681fn append<W: prebindgen_flat::WireKind, C, N, B>(
1682    to: &mut OutputPlan<W, C, N, B>,
1683    from: OutputPlan<W, C, N, B>,
1684) -> Result<()> {
1685    if !from.delivery.statements.is_empty()
1686        || from.delivery.result.to_string() != quote!(()).to_string()
1687    {
1688        return Err("composed output delivery must be installed by Codec::finish_output".into());
1689    }
1690    validate_nested_output(&from)?;
1691    let WireNode::Branch {
1692        children: elements, ..
1693    } = &mut to.outcomes.success
1694    else {
1695        return Err("output components require an ordered group".into());
1696    };
1697    elements.push(from.outcomes.success);
1698    to.fallible |= from.fallible;
1699    to.commit.extend(from.commit);
1700    to.statements.extend(from.statements);
1701    to.values.extend(from.values);
1702    prebindgen_conversions::validate_value_bindings(&to.values)
1703}
