9#[macro_export]
10/// Start a typed input conversion tree; add direct or constructor alternatives.
11macro_rules! expand_param {($($ty:tt)+)=>{$crate::Input::for_type($crate::declarations::parse_type(stringify!($($ty)+)),$crate::Input::Selector(Vec::new()))};}
12#[macro_export]
13/// Start a typed output conversion tree; append the requested views.
14macro_rules! expand_return {($($ty:tt)+)=>{$crate::Output::for_type($crate::declarations::parse_type(stringify!($($ty)+)),$crate::Output::Combine(Vec::new()))};}
15#[macro_export]
16/// Decompose the named fields returned by a source getter, using field defaults.
17macro_rules! fields {
18    ($name:ident) => {
19        $crate::Output::getter(
20            stringify!($name),
21            $crate::Output::NamedFields {
22                overrides: std::collections::BTreeMap::new(),
23                mode: $crate::ProjectionMode::Borrowed,
24            },
25        )
26    };
27}
28#[macro_export]
29/// Reference a source function to export and configure its conversions/placement.
30macro_rules! fun {
31    ($name:ident) => {
32        $crate::FunctionDecl::new($crate::ident(stringify!($name)).expect("function identifier"))
33    };
34}

36/// Conversion defaults indexed by canonical Rust type during preparation.
37/// Values are the same Input/Output trees accepted by per-function overrides.
38/// This environment is discarded after prepared conversions have been installed.
39pub struct Declarations {
40    inputs: BTreeMap<String, Input>,
41    outputs: BTreeMap<String, Output>,
42    errors: BTreeMap<String, String>,
43}
44fn input_body(input: Input) -> Input {
45    match input {
46        Input::Selector(mut arms) if arms.len() == 1 => arms.remove(0),
47        other => other,
48    }
49}
50fn output_body(output: Output) -> Output {
51    match output {
52        Output::Combine(mut parts) if parts.len() == 1 => parts.remove(0),
53        other => other,
54    }
55}
56impl Declarations {
57    /// Register typed Input/Output defaults and explicit error-message functions.
58    /// Typed trees are created by Input::for_type / Output::for_type, or the
59    /// expand_param! / expand_return! convenience macros; no parallel declaration AST exists.
60    pub fn new<I, O, M: prebindgen_flat::Metadata>(
61        flat: &Flat<I, O, M>,
62        inputs: Vec<Input>,
63        outputs: Vec<Output>,
64        errors: Vec<(syn::Type, String)>,
65    ) -> Result<Self> {
66        let mut out = Self {
67            inputs: BTreeMap::new(),
68            outputs: BTreeMap::new(),
69            errors: BTreeMap::new(),
70        };
71        for input in inputs {
72            let Input::ForType { ty, input } = input else {
73                return Err("input defaults require a source type".into());
74            };
75            if crate::contains_result(&ty) {
76                return Err("Result expansions are unsupported; configure success and error mapping separately".into());
77            }
78            let key = type_key(&flat.resolve_type(&ty)?);
79            if out.inputs.insert(key, input_body(*input)).is_some() {
80                return Err("duplicate input expansion".into());
81            }
82        }
83        for output in outputs {
84            let Output::ForType { ty, output } = output else {
85                return Err("output defaults require a source type".into());
86            };
87            if crate::contains_result(&ty) {
88                return Err("Result expansions are unsupported; configure success and error mapping separately".into());
89            }
90            let key = type_key(&flat.resolve_type(&ty)?);
91            if out.outputs.insert(key, output_body(*output)).is_some() {
92                return Err("duplicate output expansion".into());
93            }
94        }
95        for (ty, function) in errors {
96            let key = type_key(&flat.resolve_type(&ty)?);
97            if out.errors.insert(key, function).is_some() {
98                return Err("duplicate error mapper for source type".into());
99            }
100        }
101        Ok(out)
102    }
103    /// Resolve a source-only Flat into explicit Input/Output occurrence slots.
104    /// Source declarations and all metadata are moved into the returned model.
105    /// The descriptions contain resolved operations and refer to this model's
106    /// canonical declarations; no plans or wire encodings are generated here.
107    pub fn resolve<C: Codec>(
108        &self,
109        flat: Flat<(), (), C::Metadata>,
110        codec: &C,
111        functions: &[FunctionDecl],
112        extra: &C::Extra,
113    ) -> Result<Flat<Input, Output, C::Metadata>> {
114        let mut flat = flat.map_slots(|_, _| Ok(None), |_, _| Ok(None))?;
115        for decl in functions {
116            let (inputs, output) = (|| -> Result<_> {
117                let source = flat.function(&decl.name)?;
118                for name in decl.params.keys() {
119                    if !source.params.iter().any(|p| p.name == name.as_str()) {
120                        return Err(format!("`{}`: unknown parameter `{name}`", decl.name));
121                    }
122                }
123                if decl.error.is_some() && result_parts(&source.ret.ty).is_none() {
124                    return Err("on_error requires a top-level Result<T, E> return".into());
125                }
126                let inputs = source
127                    .params
128                    .iter()
129                    .map(|p| {
130                        self.resolve_input(
131                            &flat,
132                            codec,
133                            &p.ty.ty,
134                            decl.params.get(&p.name.to_string()).unwrap_or(&Input::Auto),
135                            extra,
136                        )
137                    })
138                    .collect::<Result<Vec<_>>>()?;
139                let output = self.function_output(&flat, codec, &source.ret.ty, decl, extra)?;
140                Ok((inputs, output))
141            })()
142            .map_err(|error| format!("{}: {error}", decl.name))?;
143            let target = flat.function_mut(&decl.name)?;
144            for (param, input) in target.params.iter_mut().zip(inputs) {
145                param.ty.slot = Some(input);
146            }
147            target.ret.slot = Some(output);
148        }
149        Ok(flat)
150    }
151    /// Clone a source binding for a second explicit sequence delivery policy.
152    pub fn callback_variant<C: Codec>(
153        &self,
154        flat: &mut Flat<C::Input, C::Output, C::Metadata>,
155        codec: &C,
156        decl: &FunctionDecl,
157        name: &str,
158        extra: &C::Extra,
159    ) -> Result<FunctionDecl>
160    where
161        C::Input: Clone,
162        C::Output: Clone,
163        <C::Metadata as prebindgen_flat::Metadata>::Function: Clone,
164    {
165        let source = flat.function(&decl.name)?;
166        let ty = result_parts(&source.ret.ty).map_or(&source.ret.ty, |(ok, _)| ok);
167        let mut output = self.resolve_output(
168            flat,
169            codec,
170            ty,
171            decl.ret.as_ref().unwrap_or(&Output::Auto),
172            extra,
173        )?;
174        fn select(output: &mut Output) -> usize {
175            match output {
176                Output::Sequence { delivery, .. } => {
177                    *delivery = SequenceDelivery::Callback;
178                    1
179                }
180                Output::Optional(child)
181                | Output::Getter { output: child, .. }
182                | Output::Into { output: child, .. } => select(child),
183                Output::Combine(children)
184                | Output::Fields(children)
185                | Output::OwnedFields(children)
186                | Output::Record {
187                    fields: children, ..
188                } => children.iter_mut().map(select).sum(),
189                _ => 0,
190            }
191        }
192        if select(&mut output) != 1 {
193            return Err("callback_return requires exactly one outer sequence".into());
194        }
195        let mut function = flat.function(&decl.name)?.clone();
196        function.name = crate::ident(name)?;
197        function.ret.slot = None;
198        if flat.functions.contains_key(name) {
199            return Err("callback return function name collision".into());
200        }
201        flat.functions.insert(name.into(), function);
202        let mut variant = decl.clone();
203        variant.name = name.into();
204        variant.foreign = None;
205        variant.callback_return = None;
206        variant.ret = Some(output);
207        Ok(variant)
208    }
209    /// Resolve source-dependent operations, compile them, and atomically install one binding.
210    /// Existing converters are preserved unless explicitly replaced.
211    pub fn fill<C: Codec>(
212        &self,
213        flat: &mut Flat<C::Input, C::Output, C::Metadata>,
214        codec: &C,
215        decl: &FunctionDecl,
216        extra: &C::Extra,
217    ) -> Result<()> {
218        let source = flat.function(&decl.name)?;
219        if decl.callback_return.is_some()
220            && source.ret.slot.is_some()
221            && decl.ret.is_none()
222            && decl.error.is_none()
223        {
224            return Err("callback_return cannot reconstruct a prebound output slot; supply an explicit output override".into());
225        }
226        for name in decl.params.keys() {
227            if !source.params.iter().any(|p| p.name == name.as_str()) {
228                return Err(format!("`{}`: unknown parameter `{name}`", decl.name));
229            }
230        }
231        let mut inputs = vec![];
232        for (index, p) in source.params.iter().enumerate() {
233            let explicit = decl.params.get(&p.name.to_string());
234            inputs.push(if explicit.is_some() || p.ty.slot.is_none() {
235                let tree = self.resolve_input(
236                    flat,
237                    codec,
238                    &p.ty.ty,
239                    explicit.unwrap_or(&Input::Auto),
240                    extra,
241                )?;
242                Some(Compiler::new(flat, codec, format!("p{index}")).input(
243                    &p.ty.ty,
244                    &tree,
245                    extra,
246                    &Condition::default(),
247                )?)
248            } else {
249                None
250            });
251        }
252        if decl.error.is_some() && result_parts(&source.ret.ty).is_none() {
253            return Err("on_error requires a top-level Result<T, E> return".into());
254        }
255        let mut output = if decl.ret.is_some() || decl.error.is_some() || source.ret.slot.is_none()
256        {
257            let tree = self.function_output(flat, codec, &source.ret.ty, decl, extra)?;
258            Some(Compiler::new(flat, codec, "ret".into()).output_root(
259                &source.ret.ty,
260                &tree,
261                extra,
262                prebindgen_conversions::source_result_expr(),
263                false,
264            )?)
265        } else {
266            None
267        };
268        for input in inputs.iter_mut().flatten() {
269            codec.finish_input(input, extra)?;
270            codec.validate_input(input)?;
271        }
272        if let Some(output) = &mut output {
273            crate::finalize_output_schema(codec, output, extra)?;
274            codec.finish_output(output, extra)?;
275            codec.validate_output(output)?;
276        }
277        let dest = flat.function_mut(&decl.name)?;
278        for (p, input) in dest.params.iter_mut().zip(inputs) {
279            if let Some(input) = input {
280                p.ty.slot = Some(input.into());
281            }
282        }
283        if let Some(output) = output {
284            dest.ret.slot = Some(output.into());
285        }
286        Ok(())
287    }
288    fn function_output<C: Codec, I, O>(
289        &self,
290        flat: &Flat<I, O, C::Metadata>,
291        codec: &C,
292        ty: &syn::Type,
293        decl: &FunctionDecl,
294        extra: &C::Extra,
295    ) -> Result<Output> {
296        let tree = if let Some((ok, error)) = result_parts(ty) {
297            Output::Result {
298                ok: Box::new(self.resolve_output(
299                    flat,
300                    codec,
301                    ok,
302                    decl.ret.as_ref().unwrap_or(&Output::Auto),
303                    extra,
304                )?),
305                error: self.error_mapper_for(decl, error).map(str::to_owned),
306            }
307        } else {
308            decl.ret.clone().unwrap_or(Output::Auto)
309        };
310        self.resolve_output(flat, codec, ty, &tree, extra)
311    }

353        &self,
354        flat: &Flat<I, O, C::Metadata>,
355        codec: &C,
356        ty: &syn::Type,
357        tree: &Input,

362    fn input<C: Codec, I, O>(
363        &self,
364        flat: &Flat<I, O, C::Metadata>,
365        codec: &C,
366        ty: &syn::Type,
367        tree: &Input,
368        extra: &C::Extra,
369        stack: &mut Vec<String>,
370    ) -> Result<Input> {
371        let resolve = |ty: &syn::Type, tree: &Input, stack: &mut Vec<String>| {
372            self.input(flat, codec, ty, tree, extra, stack)
373        };
374        match tree {
375            Input::Auto => {
376                let key = type_key(ty);
377                if let Some(tree) = self.inputs.get(&key) {
378                    if stack.contains(&key) {
379                        return Err(format!("recursive input expansion `{key}`"));
380                    }
381                    stack.push(key);
382                    let result = resolve(ty, tree, stack);
383                    stack.pop();
384                    return result;
385                }
386                if let Some(inner) = option_inner(ty) {
387                    return Ok(Input::Optional(Box::new(resolve(
388                        inner,
389                        &Input::Auto,
390                        stack,
391                    )?)));
392                }
393                if let Some((element, _)) = sequence_parts(ty) {
394                    return Ok(Input::Sequence(Box::new(resolve(
395                        element,
396                        &Input::Auto,
397                        stack,
398                    )?)));
399                }
400                if type_key(ty) == "()" {
401                    return Ok(Input::Fields(vec![]));
402                }
403                resolve(ty, &Input::Leaf, stack)
404            }
405            Input::ForType { ty: target, input } => {
406                let target = flat.resolve_type(target)?;
407                if same_type(ty, &target) {
408                    return resolve(ty, &input_body((**input).clone()), stack);
409                }
410                if let Some(inner) = option_inner(ty) {
411                    return Ok(Input::Optional(Box::new(resolve(inner, tree, stack)?)));
412                }
413                if let Some((element, _)) = sequence_parts(ty) {
414                    return Ok(Input::Sequence(Box::new(resolve(element, tree, stack)?)));
415                }
416                Err("input expansion type does not match occurrence".into())
417            }
418            Input::Leaf => {
419                if let Some(inner) = option_inner(ty) {
420                    return Ok(Input::Optional(Box::new(resolve(
421                        inner,
422                        &Input::Leaf,
423                        stack,
424                    )?)));
425                }
426                if let Some((element, _)) = sequence_parts(ty) {
427                    return Ok(Input::Sequence(Box::new(resolve(
428                        element,
429                        &Input::Auto,
430                        stack,
431                    )?)));
432                }
433                if let Some(replacement) = codec.expand_input(flat, ty, tree, extra)? {
434                    return resolve(ty, &replacement, stack);
435                }
436                Ok(Input::Leaf)
437            }
438            Input::Call(function) => {
439                let f = flat.function(function)?;
440                resolve(
441                    ty,
442                    &Input::Constructor {
443                        function: function.clone(),
444                        args: vec![Input::Auto; f.params.len()],
445                    },
446                    stack,
447                )
448            }
449            Input::Constructor { function, args } => {
450                let f = flat.function(function)?;
451                if !same_type(&f.ret.ty, ty) {
452                    return Err(format!(
453                        "constructor `{function}` returns `{}`, expected `{}`",
454                        type_key(&f.ret.ty),
455                        type_key(ty)
456                    ));
457                }
458                if args.len() != f.params.len() {
459                    return Err(format!("constructor `{function}`: wrong argument count"));
460                }
461                Ok(Input::Constructor {
462                    function: function.clone(),
463                    args: f
464                        .params
465                        .iter()
466                        .zip(args)
467                        .map(|(p, child)| resolve(&p.ty.ty, child, stack))
468                        .collect::<Result<_>>()?,
469                })
470            }
471            Input::Optional(child) => {
472                let inner = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
473                Ok(Input::Optional(Box::new(resolve(inner, child, stack)?)))
474            }
475            Input::Sequence(child) => {
476                let (element, _) =
477                    sequence_parts(ty).ok_or("sequence input requires Vec<T>, &[T] or &Vec<T>")?;
478                Ok(Input::Sequence(Box::new(resolve(element, child, stack)?)))
479            }
480            Input::Fields(children) | Input::Record(children) => {
481                if type_key(ty) == "()" && children.is_empty() {
482                    return Ok(Input::Fields(vec![]));
483                }
484                let fields = &flat.record(ty)?.fields;
485                if fields.len() != children.len() {
486                    return Err(format!(
487                        "`{}`: expected {} field converters, got {}",
488                        type_key(ty),
489                        fields.len(),
490                        children.len()
491                    ));
492                }
493                let children = fields
494                    .iter()
495                    .zip(children)
496                    .map(|(f, c)| resolve(&f.ty, c, stack))
497                    .collect::<Result<_>>()?;
498                Ok(if matches!(tree, Input::Record(_)) {
499                    Input::Record(children)
500                } else {
501                    Input::Fields(children)
502                })
503            }
504            Input::Selector(arms) => {
505                if arms.is_empty() {
506                    return Err("an input selector needs at least one alternative".into());
507                }
508                Ok(Input::Selector(
509                    arms.iter()
510                        .map(|child| resolve(ty, child, stack))
511                        .collect::<Result<_>>()?,
512                ))
513            }
514        }
515    }
516    /// Resolve every output operation, source reference and ownership choice before compilation.
517    pub fn resolve_output<C: Codec, I, O>(
518        &self,
519        flat: &Flat<I, O, C::Metadata>,
520        codec: &C,
521        ty: &syn::Type,
522        tree: &Output,
523        extra: &C::Extra,
524    ) -> Result<Output> {
525        self.output(flat, codec, ty, tree, extra, &mut vec![])
526    }
527    fn output<C: Codec, I, O>(
528        &self,
529        flat: &Flat<I, O, C::Metadata>,
530        codec: &C,
531        ty: &syn::Type,
532        tree: &Output,
533        extra: &C::Extra,
534        stack: &mut Vec<String>,
535    ) -> Result<Output> {
536        let resolve = |ty: &syn::Type, tree: &Output, stack: &mut Vec<String>| {
537            self.output(flat, codec, ty, tree, extra, stack)
538        };
539        match tree {
540            Output::Auto => {
541                let key = type_key(ty);
542                if let Some(tree) = self.outputs.get(&key) {
543                    if stack.contains(&key) {
544                        return Err(format!("recursive output expansion `{key}`"));
545                    }
546                    stack.push(key);
547                    let result = resolve(ty, tree, stack);
548                    stack.pop();
549                    return result;
550                }
551                if let Some(inner) = option_inner(ty) {
552                    return Ok(Output::Optional(Box::new(resolve(
553                        inner,
554                        &Output::Auto,
555                        stack,
556                    )?)));
557                }
558                if let Some((element, _)) = sequence_parts(ty) {
559                    return Ok(Output::Sequence {
560                        element: Box::new(resolve(element, &Output::Auto, stack)?),
561                        mode: None,
562                        delivery: SequenceDelivery::Value,
563                    });
564                }
565                if type_key(ty) == "()" {
566                    return Ok(Output::Fields(vec![]));
567                }
568                resolve(ty, &Output::Leaf, stack)
569            }
570            Output::ForType { ty: target, output } => {
571                let target = flat.resolve_type(target)?;
572                if same_type(ty, &target) {
573                    return resolve(ty, &output_body((**output).clone()), stack);
574                }
575                if let Some(inner) = option_inner(ty) {
576                    return Ok(Output::Optional(Box::new(resolve(inner, tree, stack)?)));
577                }
578                if let Some((element, _)) = sequence_parts(ty) {
579                    return Ok(Output::Sequence {
580                        element: Box::new(resolve(element, tree, stack)?),
581                        mode: None,
582                        delivery: SequenceDelivery::Value,
583                    });
584                }
585                Err("output expansion type does not match occurrence".into())
586            }
587            Output::Leaf | Output::Projected(_) => {
588                if let Some(inner) = option_inner(ty) {
589                    return Ok(Output::Optional(Box::new(resolve(inner, tree, stack)?)));
590                }
591                if let Some((element, _)) = sequence_parts(ty) {
592                    let mode = if let Output::Projected(p) = tree {
593                        if !p.fields.is_empty() {
594                            return Err(
595                                "collection field projections belong on the element type".into()
596                            );
597                        }
598                        Some(p.mode)
599                    } else {
600                        None
601                    };
602                    return Ok(Output::Sequence {
603                        element: Box::new(resolve(element, &Output::Auto, stack)?),
604                        mode,
605                        delivery: SequenceDelivery::Value,
606                    });
607                }
608                if let Some(replacement) = codec.expand_output(flat, ty, tree, extra)? {
609                    return resolve(ty, &replacement, stack);
610                }
611                Ok(tree.clone())
612            }
613            Output::Optional(child) => {
614                let inner = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
615                Ok(Output::Optional(Box::new(resolve(inner, child, stack)?)))
616            }
617            Output::Sequence {
618                element,
619                mode,
620                delivery,
621            } => {
622                let (ty, _) =
623                    sequence_parts(ty).ok_or("sequence output requires a collection source")?;
624                Ok(Output::Sequence {
625                    element: Box::new(resolve(ty, element, stack)?),
626                    mode: *mode,
627                    delivery: *delivery,
628                })
629            }
630            Output::NamedFields { overrides, mode } => {
631                let fields = &flat.record(ty)?.fields;
632                for name in overrides.keys() {
633                    if !fields.iter().any(|f| f.name == name.as_str()) {
634                        return Err(format!("unknown output field `{name}`"));
635                    }
636                }
637                let children = fields
638                    .iter()
639                    .map(|f| {
640                        resolve(
641                            &f.ty,
642                            overrides.get(&f.name.to_string()).unwrap_or(&Output::Auto),
643                            stack,
644                        )
645                    })
646                    .collect::<Result<_>>()?;
647                Ok(if *mode == ProjectionMode::Owned {
648                    Output::OwnedFields(children)
649                } else {
650                    Output::Fields(children)
651                })
652            }
653            Output::Fields(children)
654            | Output::OwnedFields(children)
655            | Output::Record {
656                fields: children, ..
657            } => {
658                if type_key(ty) == "()" && children.is_empty() {
659                    return Ok(tree.clone());
660                }
661                let fields = &flat.record(ty)?.fields;
662                if fields.len() != children.len() {
663                    return Err("wrong output field count".into());
664                }
665                let children = fields
666                    .iter()
667                    .zip(children)
668                    .map(|(f, c)| resolve(&f.ty, c, stack))
669                    .collect::<Result<_>>()?;
670                Ok(match tree {
671                    Output::Fields(_) => Output::Fields(children),
672                    Output::OwnedFields(_) => Output::OwnedFields(children),
673                    Output::Record { mode, .. } => Output::Record {
674                        fields: children,
675                        mode: *mode,
676                    },
677                    _ => unreachable!(),
678                })
679            }
680            Output::Getter { function, output } | Output::Into { function, output } => {
681                let f = flat.function(function)?;
682                let expected = if matches!(tree, Output::Into { .. }) {
683                    ty.clone()
684                } else {
685                    syn::parse_quote!(&#ty)
686                };
687                if f.params.len() != 1 || !same_type(&f.params[0].ty.ty, &expected) {
688                    return Err(format!(
689                        "getter `{function}` must take exactly `{}`",
690                        type_key(&expected)
691                    ));
692                }
693                if matches!(f.ret.ty, syn::Type::Reference(_)) {
694                    return Err("borrowed getter results are outside this subset".into());
695                }
696                let output = Box::new(resolve(&f.ret.ty, output, stack)?);
697                Ok(if matches!(tree, Output::Into { .. }) {
698                    Output::Into {
699                        function: function.clone(),
700                        output,
701                    }
702                } else {
703                    Output::Getter {
704                        function: function.clone(),
705                        output,
706                    }
707                })
708            }
709            Output::Combine(parts) => {
710                if parts.is_empty() {
711                    return Err("an output combination needs at least one part".into());
712                }
713                Ok(Output::Combine(
714                    parts
715                        .iter()
716                        .map(|child| resolve(ty, child, stack))
717                        .collect::<Result<_>>()?,
718                ))
719            }
720            Output::Result { ok, error } => {
721                let (success, failure) =
722                    result_parts(ty).ok_or("Result recipe requires Result<T, E>")?;
723                if let Some(name) = error {
724                    let f = flat.function(name)?;
725                    let valid = f.params.len() == 1
726                        && (same_type(&f.params[0].ty.ty, failure)
727                            || matches!(&f.params[0].ty.ty,syn::Type::Reference(r) if r.mutability.is_none()&&same_type(&r.elem,failure)));
728                    if !valid || type_key(&f.ret.ty) != "String" {
729                        return Err(
730                            "Result error mapper must take exactly E or &E and return String"
731                                .into(),
732                        );
733                    }
734                }
735                Ok(Output::Result {
736                    ok: Box::new(resolve(success, ok, stack)?),
737                    error: error.clone(),
738                })
739            }
740        }
741    }
742}

744/// Which sequence return variants an adapter should expose.
745#[derive(Clone, Copy, Debug, PartialEq, Eq)]
746pub enum CallbackReturn {
747    /// Keep the ordinary return and add a callback variant.
748    Additional,
749    /// Expose only the callback variant, including multi-value Kotlin elements
750    /// that have no ordinary List representation.
751    Only,
752}
753
754/// A source callable to export, with optional destination name and occurrence overrides.
755#[derive(Clone)]
756pub struct FunctionDecl {
757    /// Also emit an explicit callback-delivery variant for a sequence return.
758    pub callback_return: Option<CallbackReturn>,
759    /// Unqualified source function name in the flat namespace.
760    pub name: String,
761    /// Explicit destination name; supported by JNI, currently rejected by C.
762    pub foreign: Option<String>,
763    /// Input expansion overrides keyed by source parameter name.
764    pub params: BTreeMap<String, Input>,
765    /// Result expansion override, replacing the type default.
766    pub ret: Option<Output>,
767    /// Source mapper for a top-level Result error; it takes E or &E and returns String.
768    pub error: Option<String>,
769}
770impl FunctionDecl {
771    /// Reference a source function without overriding names or conversions.
772    ///
773    /// `name` is the function's unqualified source name.
774    pub fn new(name: syn::Ident) -> Self {
775        Self {
776            callback_return: None,
777            name: name.to_string(),
778            foreign: None,
779            params: BTreeMap::new(),
780            ret: None,
781            error: None,
782        }
783    }
784    /// Add callback delivery alongside the ordinary sequence return.
785    ///
786    /// C emits `<function>_each`, whose visitor receives each element's converted
787    /// fields. Kotlin emits `<function>Fold(initial, ..., fold)`, whose folder
788    /// receives an accumulator and each element's converted Kotlin values.
789    /// Supports one projected sequence, optionally inside Option and an outer Result.
790    /// Existing buffer/list returns are preserved. Adapter builders validate the
791    /// return shape and generated-name collisions.
792    pub fn callback_return(mut self) -> Self {
793        self.callback_return = Some(CallbackReturn::Additional);
794        self
795    }
796    /// Expose only callback delivery, without generating an ordinary return.
797    ///
798    /// This explicitly permits decomposed multi-value Kotlin elements without
799    /// requiring a Kotlin data class. Generated names are the same as for
800    /// `callback_return`: C `_each` and Kotlin `Fold`.
801    pub fn callback_return_only(mut self) -> Self {
802        self.callback_return = Some(CallbackReturn::Only);
803        self
804    }
805    /// Map the error of a top-level `Result<T, E>` return with a source function.
806    ///
807    /// The helper must take exactly `E` or `&E` and return `String`. Only its
808    /// source name is accepted; it is selected for capture but is not exported.
809    /// The success value uses normal return defaults and `expand_return` overrides.
810    /// Replaces any previously selected mapper. Non-Result returns reject this option.
811    pub fn on_error(mut self, mapper: impl Into<String>) -> Self {
812        self.error = Some(mapper.into());
813        self
814    }
815    /// Set a literal destination name, subject to adapter validation.
816    ///
817    /// The JNI adapter uses it instead of its default camel-case name.
818    pub fn name(mut self, name: impl Into<String>) -> Self {
819        self.foreign = Some(name.into());
820        self
821    }
822    /// Replace the expansion for one source parameter.
823    ///
824    /// # Arguments
825    ///
826    /// * `name` — the parameter's name, as written in the source.
827    /// * `decl` — how to build that argument instead of its type's default.
828    ///
829    /// # Panics
830    /// Panics if that parameter already has an override. Resolution checks that the
831    /// parameter exists and the expansion type matches.
832    pub fn input(mut self, name: impl AsRef<str>, decl: Input) -> Self {
833        assert!(
834            self.params.insert(name.as_ref().into(), decl).is_none(),
835            "duplicate parameter override"
836        );
837        self
838    }
839    /// Replace the result expansion, including any previously assigned override.
840    pub fn output(mut self, decl: Output) -> Self {
841        self.ret = Some(decl);
842        self
843    }
844}

846/// Collect referenced source helper functions directly from the conversion trees.
847pub fn source_functions<'a>(
848    inputs: &[Input],
849    outputs: &[Output],
850    errors: &[(syn::Type, String)],
851    functions: impl IntoIterator<Item = &'a FunctionDecl>,
852) -> Vec<String> {
853    fn input(tree: &Input, names: &mut Vec<String>) {
854        match tree {
855            Input::Call(name) => names.push(name.clone()),
856            Input::Constructor { function, args } => {
857                names.push(function.clone());
858                for child in args {
859                    input(child, names);
860                }
861            }
862            Input::ForType { input: child, .. }
863            | Input::Optional(child)
864            | Input::Sequence(child) => input(child, names),
865            Input::Fields(children) | Input::Record(children) | Input::Selector(children) => {
866                for child in children {
867                    input(child, names)
868                }
869            }
870            Input::Leaf | Input::Auto => {}
871        }
872    }
873    fn output(tree: &Output, names: &mut Vec<String>) {
874        match tree {
875            Output::Getter {
876                function,
877                output: child,
878            }
879            | Output::Into {
880                function,
881                output: child,
882            } => {
883                names.push(function.clone());
884                output(child, names);
885            }
886            Output::ForType { output: child, .. }
887            | Output::Optional(child)
888            | Output::Sequence { element: child, .. } => output(child, names),
889            Output::Result { ok, error } => {
890                output(ok, names);
891                if let Some(name) = error {
892                    names.push(name.clone());
893                }
894            }
895            Output::Fields(children)
896            | Output::OwnedFields(children)
897            | Output::Record {
898                fields: children, ..
899            }
900            | Output::Combine(children) => {
901                for child in children {
902                    output(child, names)
903                }
904            }
905            Output::NamedFields { overrides, .. } => {
906                for child in overrides.values() {
907                    output(child, names)
908                }
909            }
910            Output::Auto | Output::Leaf | Output::Projected(_) => {}
911        }
912    }
913    let mut names = vec![];
914    for tree in inputs {
915        input(tree, &mut names)
916    }
917    for tree in outputs {
918        output(tree, &mut names)
919    }
920    names.extend(errors.iter().map(|(_, name)| name.clone()));
921    for f in functions {
922        names.push(f.name.clone());
923        for tree in f.params.values() {
924            input(tree, &mut names)
925        }
926        if let Some(tree) = &f.ret {
927            output(tree, &mut names)
928        }
929        if let Some(name) = &f.error {
930            names.push(name.clone())
931        }
932    }
933    names
934}

937    syn::parse_str(text).expect("source type syntax")
