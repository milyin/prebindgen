12pub(crate) type NameHook = Box<dyn Fn(&str, &str) -> String>;
13pub(crate) type MethodHook = Box<dyn Fn(&str, &str, &str) -> String>;
14pub(crate) type HarnessHook = Box<dyn Fn(&str) -> String>;

16/// Generator-wide settings (`set_*`), plus the collected declarations.
17pub struct JniGenBuilder {
18    pub(crate) items: Vec<(syn::Item, SourceLocation)>,
19    pub(crate) source_module: Option<syn::Path>,
20    pub(crate) package_prefix: String,
21    pub(crate) native_init: Option<String>,
22    pub(crate) harness_hook: Option<HarnessHook>,
23    pub(crate) fun_hook: Option<NameHook>,
24    pub(crate) ptr_hook: Option<NameHook>,
25    pub(crate) data_hook: Option<NameHook>,
26    pub(crate) enum_hook: Option<NameHook>,
27    pub(crate) method_hook: Option<MethodHook>,
28    pub(crate) handle_locks: bool,
29    pub(crate) packages: Vec<PackageDecl>,
30    pub(crate) converts: Vec<ConvertDecl>,
31    pub(crate) expands: Vec<ExpandDecl>,
32    pub(crate) ignores: Vec<IgnoreDecl>,
33}

50impl JniGenBuilder {
51    pub fn new() -> Self {
52        Self {
53            items: Vec::new(),
54            source_module: None,
55            package_prefix: String::new(),
56            native_init: None,
57            harness_hook: None,
58            fun_hook: None,
59            ptr_hook: None,
60            data_hook: None,
61            enum_hook: None,
62            method_hook: None,
63            handle_locks: true,
64            packages: Vec::new(),
65            converts: Vec::new(),
66            expands: Vec::new(),
67            ignores: Vec::new(),
68        }
69    }

71    /// Every item captured in a source crate's `PREBINDGEN_OUT_DIR`.
72    pub fn source<P: AsRef<std::path::Path>>(mut self, dir: P) -> Self {
73        self.items.extend(prebindgen::Source::new(dir).items_all());
74        self
75    }
76
77    /// The items of a dependency renamed in `Cargo.toml` to `crate_name`.
78    pub fn source_named<P: AsRef<std::path::Path>>(mut self, dir: P, crate_name: &str) -> Self {
79        let source = prebindgen::Source::builder(dir)
80            .crate_name(crate_name)
81            .build();
82        self.items.extend(source.items_all());
83        self
84    }
85
86    /// Items from any stream.
87    pub fn items<I: IntoIterator<Item = (syn::Item, SourceLocation)>>(mut self, items: I) -> Self {
88        self.items.extend(items);
89        self
90    }
91
92    /// The module an item is named by when its capture carries no crate.
93    pub fn set_source_module(mut self, module: syn::Path) -> Self {
94        self.source_module = Some(module);
95        self
96    }
97
98    /// The base Kotlin package: `io.example.binding`.
99    pub fn set_package_prefix(mut self, prefix: impl Into<String>) -> Self {
100        self.package_prefix = prefix.into();
101        self
102    }
103
104    /// A Kotlin statement the native-method holder runs before its first
105    /// call — typically loading the library.
106    pub fn set_jni_native_init(mut self, stmt: impl Into<String>) -> Self {
107        self.native_init = Some(stmt.into());
108        self
109    }
110
111    /// The name of the object holding the `external` declarations, from its
112    /// default `JNINative`.
113    pub fn set_harness_name_mangle(mut self, f: impl Fn(&str) -> String + 'static) -> Self {
114        self.harness_hook = Some(Box::new(f));
115        self
116    }
117
118    /// A package function's Kotlin name: `(package, camelCaseName)`.
119    pub fn set_fun_name_mangle(mut self, f: impl Fn(&str, &str) -> String + 'static) -> Self {
120        self.fun_hook = Some(Box::new(f));
121        self
122    }
123
124    /// A handle class's Kotlin name: `(package, RustName)`.
125    pub fn set_ptr_class_name_mangle(mut self, f: impl Fn(&str, &str) -> String + 'static) -> Self {
126        self.ptr_hook = Some(Box::new(f));
127        self
128    }
129
130    /// A data or sealed class's Kotlin name: `(package, RustName)`.
131    pub fn set_data_class_name_mangle(
132        mut self,
133        f: impl Fn(&str, &str) -> String + 'static,
134    ) -> Self {
135        self.data_hook = Some(Box::new(f));
136        self
137    }
138
139    /// An enum class's Kotlin name: `(package, RustName)`.
140    pub fn set_enum_name_mangle(mut self, f: impl Fn(&str, &str) -> String + 'static) -> Self {
141        self.enum_hook = Some(Box::new(f));
142        self
143    }
144
145    /// A member's Kotlin name: `(package, ClassName, camelCaseName)`. Also
146    /// names the native-method holder's `external` declarations, with the
147    /// holder as the class.
148    pub fn set_method_name_mangle(
149        mut self,
150        f: impl Fn(&str, &str, &str) -> String + 'static,
151    ) -> Self {
152        self.method_hook = Some(Box::new(f));
153        self
154    }
155
156    /// Whether wrappers lock the handles they pass (default on): a handle
157    /// cannot be closed by another thread while a call uses it.
158    pub fn set_emit_handle_locks(mut self, on: bool) -> Self {
159        self.handle_locks = on;
160        self
161    }
162
163    /// Bind classes, functions and constants under one Kotlin package.
164    ///
165    /// Build the [`PackageDecl`] with [`package!`](crate::package) and add
166    /// classes ([`ptr_class!`](crate::ptr_class), [`data_class!`](crate::data_class),
167    /// [`enum_class!`](crate::enum_class), [`sealed_class!`](crate::sealed_class)),
168    /// functions ([`fun!`](crate::fun)) and constants
169    /// ([`constant!`](crate::constant)). A source-defined type a bound function
170    /// uses must be declared as a class or a conversion; primitives, strings,
171    /// `Option`, `Vec`, arrays, borrows and callbacks need no declaration (see
172    /// [How values cross](crate#how-values-cross)).
173    /// `examples/covertest-kotlin/build.rs` declares every kind.
174    pub fn package(mut self, p: PackageDecl) -> Self {
175        self.packages.push(p);
176        self
177    }
178
179    /// Let a type cross as another type, through declared functions or
180    /// `From`/`TryFrom` impls: `convert!(Millis).input(fun!(from_raw)).output(fun!(to_raw))`.
181    /// Kotlin sees the representation type; see [`ConvertDecl`].
182    pub fn convert(mut self, c: ConvertDecl) -> Self {
183        self.converts.push(c);
184        self
185    }
186
187    /// Set a type's default boundary shape:
188    ///
189    /// * [`ExpandReturnDecl`](crate::ExpandReturnDecl) (`expand_return!`) — a result of the type is
190    ///   delivered as its fields to a caller-supplied builder (a folder for a
191    ///   `Vec`, the handler for an `Err`, the arguments of a callback);
192    /// * [`ExpandParamDecl`](crate::ExpandParamDecl) (`expand_param!`) — a parameter of the type is
193    ///   built by a constructor or passed as a handle, chosen by a selector.
194    ///
195    /// A function overrides either for itself with
196    /// [`FunctionDecl::expand_return`](crate::FunctionDecl::expand_return) / [`FunctionDecl::expand_param`](crate::FunctionDecl::expand_param).
197    pub fn expand(mut self, e: impl Into<ExpandDecl>) -> Self {
198        self.expands.push(e.into());
199        self
200    }
201
202    /// Acknowledge items that are deliberately not bound.
203    pub fn ignore(mut self, i: impl Into<IgnoreDecl>) -> Self {
204        self.ignores.push(i.into());
205        self
206    }
207
208    /// Parse the sources and generate the binding.
209    pub fn build(self) -> Result<Generation, Error> {
210        let flat = Flat::builder()
211            .items(self.items.clone())
212            .build()
213            .map_err(|e| Error(e.to_string()))?;
214        let plan = Plan::new(&self, &flat)?;
215        write::write(&plan)
216    }
