33/// Write one binding: the extern into `out`, the wrapper returned.
34pub(crate) fn function(plan: &Plan, out: &mut Out, b: &Binding) -> Res<Wrapper> {
35    out.rust.push(extern_rust(plan, b)?);
36    out.externs.push(extern_kotlin(plan, b)?);
37    wrapper(plan, b)
38}

42fn extern_rust(plan: &Plan, b: &Binding) -> Res<TokenStream> {
43    let symbol = jni_symbol(&plan.harness_fqn(), &b.ext_name);
44    let jobject = quote!(::prebindgen_jni_runtime::jni::objects::JObject<'a>);
45    let callee = match &b.callee {
46        Callee::Call(c) | Callee::Value(c) => c.clone(),
47    };
48    let mut w = FunctionWriter::new(&b.func, callee)
49        .name(format_ident!("{}", symbol))
50        .abi(Some("system"))
51        .generics(quote!(<'a>))
52        .attr(quote!(#[no_mangle]))
53        .attr(quote!(#[allow(non_snake_case, unused_mut, unused_variables, unused_braces, unused_parens, unused_unsafe, dead_code, clippy::all)]))
54        .leading(Wire::new(
55            format_ident!("__env"),
56            quote!(::prebindgen_jni_runtime::jni::JNIEnv<'a>),
57        ))
58        .leading(Wire::new(
59            format_ident!("_class"),
60            quote!(::prebindgen_jni_runtime::jni::objects::JClass<'a>),
61        ))
62        .prologue(quote!(let mut __env = __env; let env = &mut __env;))
63        .trailing(Wire::new(format_ident!("__error_sink"), jobject.clone()));
64    if let Callee::Value(_) = b.callee {
65        w = w.call_with(|value, _| value.clone());
66    }
67    if let EPlan::Domain { .. } = b.err {
68        w = w.trailing(Wire::new(format_ident!("__domain_sink"), jobject));
69    }
70    w.write(&mut RustCb { plan, b })
71}

79impl FunctionCallbacks for RustCb<'_, '_> {
80    type Error = crate::Error;

82    fn ret(&mut self, _ret: &TypeRef) -> Res<Return> {
83        let (plan, b) = (self.plan, self.b);
84        let r = result_ident();
85        let whole = match &b.err {
86            EPlan::None => rust_ret_value(plan, &b.ret, r.to_token_stream())?,
87            e => {
88                let ok = rust_ret_value(plan, &b.ret, quote!(__ok))?;
89                let on_err = rust_error(plan, e)?;
90                quote!(match #r {
91                    ::core::result::Result::Ok(__ok) => #ok,
92                    ::core::result::Result::Err(__e) => #on_err,
93                })
94            }
95        };
96        let (ty, default) = ret_wire(&b.ret);
97        let wires = if uses_sink(&b.ret) {
98            vec![Wire::new(
99                format_ident!("__sink"),
100                quote!(::prebindgen_jni_runtime::jni::objects::JObject<'a>),
101            )]
102        } else {
103            Vec::new()
104        };
105        let rty = ty.clone().unwrap_or(quote!(()));
106        let e = error_ident();
107        Ok(Return {
108            ty,
109            wires,
110            body: quote! {
111                let __r: ::core::result::Result<#rty, ::std::string::String> = (|| #whole)();
112                match __r {
113                    ::core::result::Result::Ok(__v) => __v,
114                    ::core::result::Result::Err(#e) => {
115                        if !#e.is_empty() {
116                            __jni_signal(env, &__error_sink, &#e);
117                        }
118                        #default
119                    }
120                }
121            },
122        })
123    }
124
125    fn param(&mut self, name: &syn::Ident, _ty: &TypeRef) -> Res<Input> {
126        let plan = self.plan;
127        let p = self
128            .b
129            .params
130            .iter()
131            .find(|p| match p {
132                PPlan::Receiver(fp) | PPlan::Value(fp, _) => fp.name == *name,
133                PPlan::Selector(s, _) => s.param.name == *name,
134            })
135            .expect("a planned parameter");
136        let (fp, packed) = match p {
137            PPlan::Selector(s, _) => return plan.selector_input(s),
138            PPlan::Receiver(fp) => (fp, false),
139            PPlan::Value(fp, packed) => (fp, *packed),
140        };
141        let root = names::bare(&fp.name);
142        let mut input = plan.rs_decode(&fp.ty, &root, 0)?;
143        if packed {
144            let leaves: Vec<Leaf> = plan
145                .leaves(&fp.ty, Dir::In)?
146                .into_iter()
147                .map(|l| l.under(&root))
148                .collect();
149            input = pack::pack_input(&root, &leaves, input);
150        }
151        // A borrowed parameter lends the decoded value.
152        Ok(match plan.shape(&fp.ty)? {
153            Shape::Str {
154                access: Access::Shared,
155                ..
156            }
157            | Shape::Seq {
158                access: Access::Shared,
159                ..
160            }
161            | Shape::Ref {
162                access: prebindgen_tools::Access::Shared,
163                ..
164            } => input.with_pass(quote!(&#name)),
165            Shape::Ref { .. }
166            | Shape::Str {
167                access: Access::Exclusive,
168                ..
169            }
170            | Shape::Seq {
171                access: Access::Exclusive,
172                ..
173            } => {
174                return err(format!(
175                    "`{name}`: a `&mut` value parameter cannot cross from Kotlin"
176                ))
177            }
178            _ => input,
179        })
180    }
181
182    fn fail(&mut self, _ret: &Return) -> TokenStream {
183        let e = error_ident();
184        let (_, default) = ret_wire(&self.b.ret);
185        quote!({
186            __jni_signal(env, &__error_sink, &#e);
187            return #default;
188        })
189    }
190}

192fn uses_sink(ret: &RPlan) -> bool {
193    matches!(
194        ret,
195        RPlan::Sink { .. } | RPlan::Builder { .. } | RPlan::Fold { .. }
196    )

199/// The extern's return type, and the value it returns on failure.
200fn ret_wire(ret: &RPlan) -> (Option<TokenStream>, TokenStream) {
201    let object = (
202        Some(quote!(::prebindgen_jni_runtime::jni::sys::jobject)),
203        quote!(::core::ptr::null_mut()),
204    );
205    match ret {
206        RPlan::Unit => (None, quote!(())),
207        RPlan::Direct { leaf, .. } => match leaf.prim() {
208            Some(p) => (Some(p.rs()), p.rs_default()),
209            None => object,
210        },
211        _ => object,
212    }
213}

215/// The encoding of a successful value per the return plan, as statements
216/// ending in `Ok(<return wire>)`.
217fn rust_ret_value(plan: &Plan, ret: &RPlan, value: TokenStream) -> Res<TokenStream> {
218    Ok(match ret {
219        RPlan::Unit => quote!({ let _ = #value; ::core::result::Result::Ok(()) }),
220        RPlan::Direct { ty, leaf } => {
221            let out = plan.rs_encode(ty, value, "r", 1)?;
222            let bind = out.bind();
223            let n = &out.wires[0].name;
224            let v = if leaf.is_obj() {
225                quote!(#n.into_raw())
226            } else {
227                quote!(#n)
228            };
229            quote!({ #bind ::core::result::Result::Ok(#v) })
230        }
231        RPlan::Sink {
232            ty, leaves, iface, ..
233        } => call_sink(&plan.rs_encode(ty, value, "r", 1)?, leaves, iface),
234        RPlan::Builder {
235            whole,
236            exp,
237            iface,
238            leaves,
239            ..
240        } => {
241            let iface = format!("{iface}Raw");
242            layered(whole, value, &mut |t, v| {
243                let x = format_ident!("__x");
244                let d = plan.deliver(t, x.to_token_stream(), "r", "", Some(exp), &[], false, 1)?;
245                let call = call_sink(&d.output, leaves, &iface);
246                Ok(quote!({ let #x = #v; #call }))
247            })?
248        }
249        RPlan::Fold {
250            whole,
251            exp,
252            columns_iface,
253            leaves,
254            ..
255        } => layered(whole, value, &mut |t, v| {
256            let TypeKind::Vec(elem) = t.kind() else {
257                return err(format!("`{t}`: a folded result must be a `Vec`"));
258            };
259            let e = format_ident!("__x");
260            let d = plan.deliver(elem, e.to_token_stream(), "r", "", Some(exp), &[], false, 1)?;
261            let cols = columns(&d.output, leaves, &e, quote!(__items));
262            let mut col_leaves = vec![Leaf::new(LeafTy::Prim(Prim::I))];
263            col_leaves.extend(leaves.iter().map(Leaf::column));
264            let call = call_sink(&cols, &col_leaves, columns_iface);
265            Ok(
266                quote!({ let __items: ::std::vec::Vec<_> = ::core::iter::IntoIterator::into_iter(#v).collect(); #call }),
267            )
268        })?,
269    })
270}

272/// Walk the layers a builder or folder absorbs — `Box`, the first `Option`
273/// (absent: a null result) — down to the value `core` delivers (the `Vec`
274/// itself, for a fold).
275fn layered(
276    ty: &TypeRef,
277    value: TokenStream,
278    core: &mut dyn FnMut(&TypeRef, TokenStream) -> Res<TokenStream>,
279) -> Res<TokenStream> {
280    match ty.kind() {
281        TypeKind::Boxed(t) => layered(t, quote!((*#value)), core),
282        TypeKind::Optional(t) => {
283            let inner = unbox(t, quote!(__some), core)?;
284            Ok(quote!(match #value {
285                ::core::option::Option::Some(__some) => #inner,
286                ::core::option::Option::None => ::core::result::Result::Ok(::core::ptr::null_mut()),
287            }))
288        }
289        _ => core(ty, value),
290    }
291}

293/// Below the `Option`: only `Box` layers remain.
294fn unbox(
295    ty: &TypeRef,
296    value: TokenStream,
297    core: &mut dyn FnMut(&TypeRef, TokenStream) -> Res<TokenStream>,
298) -> Res<TokenStream> {
299    match ty.kind() {
300        TypeKind::Boxed(t) => unbox(t, quote!((*#value)), core),
301        _ => core(ty, value),
302    }
303}

305/// Columns of a sequence of deliveries: a count and one array per leaf.
306fn columns(elem: &Output, leaves: &[Leaf], x: &syn::Ident, items: TokenStream) -> Output {
307    let rt = quote!(::prebindgen_jni_runtime);
308    let mut setup = Vec::new();
309    let mut pushes = Vec::new();
310    let mut finals = vec![quote!(__n as i32)];
311    let mut wires = vec![Wire::new(format_ident!("__cn"), Prim::I.rs())];
312    for (k, (l, w)) in leaves.iter().zip(&elem.wires).enumerate() {
313        let col = format_ident!("__col{}", k);
314        let local = &w.name;
315        wires.push(Wire::new(format_ident!("__colw{}", k), l.column().rs()));
316        match l.prim() {
317            Some(p) => {
318                let write = format_ident!("{}", p.array_helpers().1);
319                setup.push(quote!(let mut #col = ::std::vec::Vec::with_capacity(__n);));
320                pushes.push(quote!(#col.push(#local);));
321                finals.push(quote!(#rt::#write(env, &#col)?));
322            }
323            None => {
324                setup.push(quote!(let #col = #rt::new_object_array(env, __n)?;));
325                pushes.push(quote!(#rt::object_array_set(env, &#col, __i, #local)?;));
326                finals.push(quote!(#col));
327            }
328        }
329    }
330    let bind = elem.bind();
331    Output::fallible(
332        wires,
333        quote!({
334            let __n = #items.len();
335            #(#setup)*
336            for (__i, #x) in #items.into_iter().enumerate() {
337                #bind
338                #(#pushes)*
339            }
340            (#(#finals),*)
341        }),
342    )
343}

345/// What an `Err` does: a message for the binding-error channel, or a
346/// delivery to the typed handler (the empty message then says it was
347/// delivered).
348fn rust_error(plan: &Plan, e: &EPlan) -> Res<TokenStream> {
349    Ok(match e {
350        EPlan::None => quote!(unreachable!()),
351        EPlan::Binding => quote!(::core::result::Result::Err(
352            ::std::string::ToString::to_string(&__e)
353        )),
354        EPlan::Domain {
355            ty,
356            raw_iface,
357            leaves,
358            ..
359        } => {
360            let d = plan.deliver(ty, quote!(__e), "e", "", None, &[], false, 1)?;
361            let bind = d.output.bind();
362            let values = d
363                .output
364                .wires
365                .iter()
366                .zip(leaves)
367                .map(|(w, l)| l.jvalue(&w.name.to_token_stream()));
368            let desc = method_desc(leaves, "V");
369            let fqn = raw_iface.replace('.', "/");
370            quote!({
371                #bind
372                static __D: ::prebindgen_jni_runtime::CachedIfaceMethod = ::prebindgen_jni_runtime::CachedIfaceMethod::new();
373                __D.call_void(env, #fqn, "run", #desc, &__domain_sink, &[#(#values),*])?;
374                ::core::result::Result::Err(::std::string::String::new())
375            })
376        }
377    })
378}

380/// Encode `out`'s wires as `jvalue`s and call the sink interface's `run`,
381/// returning its result object.
382fn call_sink(out: &Output, leaves: &[Leaf], iface: &str) -> TokenStream {
383    let bind = out.bind();
384    let values = out
385        .wires
386        .iter()
387        .zip(leaves)
388        .map(|(w, l)| l.jvalue(&w.name.to_token_stream()));
389    let desc = method_desc(leaves, "Ljava/lang/Object;");
390    let fqn = iface.replace('.', "/");
391    quote!({
392        #bind
393        static __S: ::prebindgen_jni_runtime::CachedIfaceMethod = ::prebindgen_jni_runtime::CachedIfaceMethod::new();
394        let __o = __S.call_object(env, #fqn, "run", #desc, &__sink, &[#(#values),*])?;
395        ::core::result::Result::Ok(__o.into_raw())
396    })
397}

401/// The extern's trailing object parameters: sink, error sink, domain sink.
402fn trailing(b: &Binding) -> Vec<&'static str> {
403    let mut v = Vec::new();
404    if uses_sink(&b.ret) {
405        v.push("sink");
406    }
407    v.push("errorSink");
408    if let EPlan::Domain { .. } = b.err {
409        v.push("domainSink");
410    }
411    v
412}

414fn extern_kotlin(plan: &Plan, b: &Binding) -> Res<String> {
415    let mut params = Vec::new();
416    for p in &b.params {
417        for l in plan.param_leaves(p)? {
418            params.push(format!("{}: {}", raw_name(&l), l.kt_raw()));
419        }
420    }
421    params.extend(trailing(b).iter().map(|n| format!("{n}: Any")));
422    let ret = match &b.ret {
423        RPlan::Unit => String::new(),
424        RPlan::Direct { leaf, .. } => format!(": {}", leaf.kt_raw()),
425        _ => ": Any?".to_string(),
426    };
427    Ok(format!(
428        "    external fun {}({}){ret}",
429        b.ext_name,
430        params.join(", ")
431    ))
432}

436/// The Kotlin type the wrapper returns, and its type parameters.
437fn kt_ret(plan: &Plan, ret: &RPlan) -> Res<(String, String)> {
438    let generic = |optional: bool, t: &str| {
439        let r = if optional {
440            format!("{t}?")
441        } else {
442            t.to_string()
443        };
444        (r, format!("<{t}> "))
445    };
446    Ok(match ret {
447        RPlan::Unit => ("Unit".to_string(), String::new()),
448        RPlan::Direct { ty, .. } | RPlan::Sink { ty, .. } => (plan.kt_type(ty)?, String::new()),
449        RPlan::Builder { optional, .. } => generic(*optional, "R"),
450        RPlan::Fold { optional, .. } => generic(*optional, "A"),
451    })
452}

454fn wrapper(plan: &Plan, b: &Binding) -> Res<Wrapper> {
455    let base = &plan.base_pkg;
456    let (ret, generics) = kt_ret(plan, &b.ret)?;
457    let mut cx = KtEnc::default();
458    let mut sig: Vec<SigParam> = Vec::new();
459    let mut args: Vec<String> = Vec::new();
460    for p in &b.params {
461        match p {
462            PPlan::Receiver(fp) => {
463                args.extend(plan.kt_encode(&fp.ty, "this", false, &mut cx, true)?);
464            }
465            PPlan::Value(fp, packed) => {
466                let n = kt_ident(&names::camel(&names::bare(&fp.name)));
467                sig.push(SigParam::Plain(n.clone(), plan.kt_type(&fp.ty)?));
468                let exprs = plan.kt_encode(&fp.ty, &n, false, &mut cx, true)?;
469                if *packed {
470                    args.extend(pack::pack_kotlin(&plan.leaves(&fp.ty, Dir::In)?, &exprs));
471                } else {
472                    args.extend(exprs);
473                }
474            }
475            PPlan::Selector(s, split) => {
476                args.extend(plan.selector_encode(s, &mut cx)?);
477                sig.push(if *split {
478                    SigParam::Split(s.clone())
479                } else {
480                    SigParam::Selector(s.clone())
481                });
482            }
483        }
484    }
485    // Tail parameters: accumulator, error handlers, builder / folder.
486    let mut tail: Vec<(String, String)> = Vec::new();
487    if let RPlan::Fold { .. } = b.ret {
488        tail.push(("acc".to_string(), "A".to_string()));
489    }
490    let je = format!("{base}.JniErrorHandler");
491    match &b.err {
492        EPlan::Domain { handler, .. } => {
493            tail.push(("onBindingError".to_string(), format!("{je}<{ret}>")));
494            tail.push(("onError".to_string(), format!("{handler}<{ret}>")));
495        }
496        _ => tail.push(("onError".to_string(), format!("{je}<{ret}>"))),
497    }
498    match &b.ret {
499        RPlan::Builder { iface, .. } => tail.push(("build".to_string(), format!("{iface}<R>"))),
500        RPlan::Fold { iface, .. } => tail.push(("fold".to_string(), format!("{iface}<A>"))),
501        _ => {}
502    }
503    let binding_handler = match b.err {
504        EPlan::Domain { .. } => "onBindingError",
505        _ => "onError",
506    };
507
508    // The call.
509    let mut call_args = args;
510    match &b.ret {
511        RPlan::Sink { kt_sink, .. } => call_args.push(kt_sink.clone()),
512        RPlan::Builder { .. } => call_args.push("build.asRaw()".to_string()),
513        RPlan::Fold {
514            columns_iface,
515            params,
516            leaves,
517            ..
518        } => {
519            let raws: Vec<String> = leaves.iter().map(raw_name).collect();
520            // The parameters decode over the raw leaf names: bind each to the
521            // element of its column.
522            let rebinds: Vec<String> = leaves
523                .iter()
524                .zip(&raws)
525                .map(|(l, r)| match l.prim() {
526                    Some(_) => format!("val {r} = __c_{r}[__i]"),
527                    None => format!("val {r} = __c_{r}[__i] as {}", l.kt_raw()),
528                })
529                .collect();
530            let decodes: Vec<String> = params.iter().map(|p| p.decode.clone()).collect();
531            let mut col_params = vec!["n".to_string()];
532            col_params.extend(raws.iter().map(|r| format!("__c_{r}")));
533            call_args.push(format!(
534                "{columns_iface} {{ {} ->\n            var __a = acc\n            for (__i in 0 until n) {{\n                {}\n                __a = fold.run(__a{}{})\n            }}\n            __a\n        }}",
535                col_params.join(", "),
536                rebinds.join("\n                "),
537                if decodes.is_empty() { "" } else { ", " },
538                decodes.join(", ")
539            ));
540        }
541        _ => {}
542    }
543    call_args.push("__bcap".to_string());
544    if let EPlan::Domain { .. } = b.err {
545        call_args.push("__dcap".to_string());
546    }
547    let call = format!(
548        "{}.{}({})",
549        plan.harness_fqn(),
550        b.ext_name,
551        call_args.join(", ")
552    );
553    let (pre, call) = guard_handles(plan, &cx.handles, call, binding_handler);
554
555    // Body.
556    let mut body = pre;
557    body.extend(cx.prelude);
558    body.push(format!(
559        "val __bcap = {base}.JniErrorHandlerCapture.acquire()"
560    ));
561    if let EPlan::Domain { capture, .. } = &b.err {
562        body.push(format!("val __dcap = {capture}.acquire()"));
563    }
564    body.push(format!("val __ret = {call}"));
565    body.push(format!(
566        "if (__bcap.failed) return {binding_handler}.run(__bcap.ze0)"
567    ));
568    if let EPlan::Domain { params, leaves, .. } = &b.err {
569        // The handler's parameters decode over the raw leaf names: read them
570        // off the capture.
571        let rebinds: Vec<String> = leaves
572            .iter()
573            .map(|l| {
574                let r = raw_name(l);
575                if l.is_obj() && l.nullable {
576                    format!("val {r} = __dcap.{r}")
577                } else {
578                    format!("val {r} = __dcap.{r}!!")
579                }
580            })
581            .collect();
582        let decodes: Vec<String> = params.iter().map(|p| p.decode.clone()).collect();
583        body.push(format!(
584            "if (__dcap.failed) {{\n        {}\n        return onError.run({})\n    }}",
585            rebinds.join("\n        "),
586            decodes.join(", ")
587        ));
588    }
589    let ret_expr = match &b.ret {
590        RPlan::Unit => "Unit".to_string(),
591        RPlan::Direct { ty, .. } => plan.kt_decode(ty, &["__ret".to_string()], false, 0)?,
592        _ => format!("__ret as {ret}"),
593    };
594    body.push(format!("return {ret_expr}"));
595
596    // Signature.
597    let vis = match &b.placement {
598        Placement::Method(c) if c.iface.is_some() => "public override",
599        _ => "public",
600    };
601    let suppress = if uses_sink(&b.ret) {
602        "@Suppress(\"UNCHECKED_CAST\")\n"
603    } else {
604        ""
605    };
606    let mut sig_strs: Vec<String> = Vec::new();
607    for p in &sig {
608        match p {
609            SigParam::Plain(n, t) => sig_strs.push(format!("{n}: {t}")),
610            SigParam::Selector(s) | SigParam::Split(s) => {
611                for (n, t) in plan.selector_params(s)? {
612                    sig_strs.push(format!("{n}: {t}"));
613                }
614            }
615        }
616    }
617    sig_strs.extend(tail.iter().map(|(n, t)| format!("{n}: {t}")));
618    let mut text = format!(
619        "{suppress}{vis} fun {generics}{}({}): {ret} {{\n    {}\n}}\n",
620        b.kt_name,
621        sig_strs.join(", "),
622        body.join("\n    ")
623    );
624    // Split overloads.
625    let splits = sig
626        .iter()
627        .filter(|p| matches!(p, SigParam::Split(_)))
628        .count();
629    if splits > 0 {
630        let head = format!("public fun {generics}");
631        for o in plan.split_overloads(&head, &b.kt_name, &sig, &tail, &ret, splits > 1)? {
632            text.push('\n');
633            text.push_str(&o);
634        }
635    }
636    let header = format!(
637        "fun {generics}{}({}): {ret}",
638        b.kt_name,
639        sig_strs.join(", ")
640    );
641    Ok(Wrapper { text, header })
642}

644/// The checks and locks around a call that passes handles: each must be
645/// open, a consumed one may not appear twice, every one is locked (in
646/// address order) for the call, and a consumed one is marked afterwards.
647/// Returns the statements to run first and the guarded call.
648fn guard_handles(
649    plan: &Plan,
650    handles: &[HandleSite],
651    mut call: String,
652    binding_handler: &str,
653) -> (Vec<String>, String) {
654    let base = &plan.base_pkg;
655    let fail = |msg: &str| format!("return {binding_handler}.run(\"{msg}\")");
656    let ptr = |h: &HandleSite| {
657        if h.nullable {
658            format!("({}?.ptr ?: 0L)", h.expr)
659        } else {
660            format!("{}.ptr", h.expr)
661        }
662    };
663    let mut pre = Vec::new();
664    let consumed: Vec<&HandleSite> = handles.iter().filter(|h| h.consumed).collect();
665    for (i, a) in consumed.iter().enumerate() {
666        for c in &consumed[i + 1..] {
667            if a.class.rust != c.class.rust {
668                continue;
669            }
670            let (pa, pc) = (ptr(a), ptr(c));
671            pre.push(format!(
672                "if (({pa} and -2L) != 0L && ({pa} and -2L) == ({pc} and -2L)) return {binding_handler}.run(\"Aliasing arguments: '{}' and '{}' are the same native resource; a consumed handle may not be passed twice in one call.\")",
673                a.expr, c.expr
674            ));
675        }
676    }
677    for h in handles {
678        let closed = if h.nullable {
679            format!("{}?.isClosed() == true", h.expr)
680        } else {
681            format!("{}.isClosed()", h.expr)
682        };
683        pre.push(format!(
684            "if ({closed}) {}",
685            fail("Operation on a closed native handle.")
686        ));
687    }
688    if !consumed.is_empty() {
689        let marks = consumed
690            .iter()
691            .map(|h| {
692                if h.nullable {
693                    format!("{}?.markConsumed()", h.expr)
694                } else {
695                    format!("{}.markConsumed()", h.expr)
696                }
697            })
698            .collect::<Vec<_>>()
699            .join("; ");
700        call = format!(
701            "try {{\n            {call}\n        }} finally {{\n            {marks}\n        }}"
702        );
703    }
704    if plan.handle_locks && !handles.is_empty() {
705        if handles.iter().all(|h| !h.nullable) && handles.len() <= 3 {
706            let hs = handles
707                .iter()
708                .map(|h| h.expr.clone())
709                .collect::<Vec<_>>()
710                .join(", ");
711            call = format!("{base}.withSortedHandleLocks({hs}) {{\n        {call}\n    }}");
712        } else {
713            let adds = handles
714                .iter()
715                .map(|h| {
716                    if h.nullable {
717                        format!("{}?.let {{ __locks.add(it) }}", h.expr)
718                    } else {
719                        format!("__locks.add({})", h.expr)
720                    }
721                })
722                .collect::<Vec<_>>()
723                .join("; ");
724            call = format!(
725                "run {{\n        val __locks = ArrayList<{base}.NativeHandle>(); {adds}\n        {base}.withSortedHandleLocks(__locks) {{\n            {call}\n        }}\n    }}"
726            );
727        }
728    }
729    (pre, call)
730}
