6/// Prepared boundary roots, including typed error converters and declarations
7/// whose callback interfaces are emitted without a direct exported occurrence.
8pub(crate) fn nodes<I: KotlinInputConversion, O: KotlinOutputConversion>(
9    flat: &Flat<JniSchema<I, O>>,
10) -> Result<Vec<(bool, &kotlin::JniNode)>> {
11    let mut nodes = Vec::new();
12    for source in flat.functions.values() {
13        for binding in &source.metadata {
14            let call = binding.call()?;
15            for plan in &call.inputs {
16                nodes.push((true, plan.input_node()));
17            }
18            let outcomes = call.output.output_outcomes();
19            nodes.extend(
20                std::iter::once(&outcomes.success)
21                    .chain(outcomes.source_error.iter())
22                    .chain(outcomes.binding_failure.iter())
23                    .map(|node| (false, node)),
24            );
25            if let Some(output) = binding
26                .metadata
27                .error
28                .as_deref()
29                .and_then(|e| e.output.as_ref())
30            {
31                nodes.push((false, &output.output_outcomes().success));
32            }
33        }
34    }
35    for input in crate::callback::declarations(flat)? {
36        nodes.push((true, input.input_node()));
37    }
38    Ok(nodes)
39}

41/// Metadata schema for a Kotlin/JNI binding model.
42/// `InConverter` and `OutConverter` select the converters stored in each export.
43/// Source parameter/return attachments are unit; type metadata keeps the JNI
44/// representation. Default containers also accept custom converters through their
45/// owned containers. No schema value is needed to construct Flat.
46pub struct JniSchema<InConverter = JniInputConverter, OutConverter = JniOutputConverter>(
47    std::marker::PhantomData<fn() -> (InConverter, OutConverter)>,
48);
49impl<InConverter, OutConverter> prebindgen_flat::Schema for JniSchema<InConverter, OutConverter> {
50    type Parameter = ();
51    type Return = ();
52    type Module = JniModule;
53    type Function = Vec<JniBinding<InConverter, OutConverter>>;
54    type Type = JniType;
55    type Field = data::JniField;
56    type Variant = ();
57    type Record = ();
58    type Enum = ();
59    type PayloadEnum = ();
60}

84/// Kotlin/JNI declaration information shared by occurrences of a Rust type.
85#[derive(Clone, Debug, Default)]
86pub struct JniType {
87    /// Standalone callback interface attached to its canonical callback type.
88    /// Configured selects this type's input default; later stages retain its recipe/converter.
89    pub callback:
90        Option<prebindgen_conversions::BindingState<(), crate::JniInput, crate::JniInputConverter>>,
91    /// Input/output defaults stored on this type, including callback signatures.
92    pub defaults:
93        prebindgen_declarations::DeclTypeDefaults<crate::JniInputConfig, crate::JniOutputConfig>,
94    /// Canonical value conversions selected for this source type.
95    pub conversion: prebindgen_conversions::ValueConversions,
96    pub(crate) class: Option<Rc<Class>>,
97    pub(crate) errors: BTreeMap<bool, Rc<errors::ErrorBinding>>,
98    pub(crate) error_class: Option<String>,
99}

122// Borrow the actual Flat type table, independent of its export converter types.
123pub(crate) type JniTypes = BTreeMap<String, prebindgen_flat::TypeInfo<JniType, data::JniField>>;

125pub(crate) fn handle_classes(types: &JniTypes) -> impl Iterator<Item = (&syn::Type, &Class)> {
126    let mut classes = types
127        .values()
128        .filter_map(|source| {
129            source
130                .metadata
131                .class
132                .as_deref()
133                .filter(|class| matches!(class.representation, ClassRepresentation::Handle { .. }))
134                .map(|class| (&source.ty, class))
135        })
136        .collect::<Vec<_>>();
137    // Preserve emitted declaration order across captured and external types.
138    classes.sort_by_key(|(ty, _)| type_key(ty));
139    classes.into_iter()
140}

142pub(crate) fn source_handle<'a>(
143    types: &'a JniTypes,
144    ty: &syn::Type,
145) -> Result<(&'a syn::Type, &'a Class)> {
146    let source = source_type(types, ty)?;
147    let class = source
148        .metadata
149        .class
150        .as_deref()
151        .filter(|class| matches!(class.representation, ClassRepresentation::Handle { .. }))
152        .ok_or_else(|| format!("no handle class for `{}`", type_key(ty)))?;
153    Ok((&source.ty, class))
154}

156pub(crate) fn source_type<'a>(
157    types: &'a JniTypes,
158    ty: &syn::Type,
159) -> Result<&'a prebindgen_flat::TypeInfo<JniType, data::JniField>> {
160    types
161        .values()
162        .find(|entry| prebindgen_flat::same_type(&entry.ty, ty))
163        .ok_or_else(|| format!("unregistered JNI source type `{}`", type_key(ty)))
164}
165pub(crate) fn source_data_class<'a>(
166    types: &'a JniTypes,
167    ty: &syn::Type,
168) -> Result<(&'a prebindgen_flat::Record<data::JniField>, &'a Class)> {
169    let source = source_type(types, ty)?;
170    let class = source
171        .metadata
172        .class
173        .as_deref()
174        .filter(|class| matches!(class.representation, ClassRepresentation::Data { .. }))
175        .ok_or("record conversion requires a declared data class")?;
176    match &source.definition {
177        prebindgen_flat::TypeDefinition::Record(record) => Ok((record, class)),
178        _ => Err(format!("`{}` is not a source record", type_key(ty))),
179    }
180}
181pub(crate) fn source_enum<'a>(
182    types: &'a JniTypes,
183    ty: &syn::Type,
184) -> Result<&'a prebindgen_flat::PayloadEnum<(), data::JniField>> {
185    match &source_type(types, ty)?.definition {
186        prebindgen_flat::TypeDefinition::PayloadEnum(enumeration) => Ok(enumeration),
187        _ => Err(format!("`{}` is not a source payload enum", type_key(ty))),
188    }
