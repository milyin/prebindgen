1//! Declaration language and composition for preparing conversion slots.
2//!
3//! [`Declarations::fill`], [`bind_input`] and [`bind_output`] build conversions
4//! directly in a [`Bindings`] model. Recipes and codec input data are used only
5//! during construction; slots retain wires and code needed by wrapper writers.
6//! Generation is owned by the language generator;
7//! this crate owns only declaration resolution and slot preparation.
8//! Each [`Wire`] carries the adapter's own description in [`Wire::kind`].
9//! Reusable value and boxed-allocation leaves live in `prebindgen-conversions`.

41/// A recipe for converting boundary inputs into one source argument.
42///
43/// Binding expands the recipe into wires and Rust conversion code.
44#[derive(Clone, Debug)]
45pub enum Input {
46    /// Convert an optional payload, gating every conversion phase.
47    Optional(Box<Input>),
48    /// Convert the occurrence type itself with [`Codec::input_leaf`].
49    Leaf,
50    /// Construct a named record from children in field order; an empty list builds unit.
51    Fields(Vec<Input>),
52    /// Call a source function after preparing its arguments with the child recipes.
53    Constructor {
54        /// Unqualified source function name; its result must match the occurrence type.
55        function: String,
56        /// Recipes in the constructor's parameter order.
57        args: Vec<Input>,
58    },
59    /// Choose an owned input using an `i32` arm index followed by all arms' wires.
60    /// Inactive arms are neither validated nor acquired; indices are zero-based.
61    Selector(Vec<Input>),
62}
63impl Input {
64    /// Construct a record from field recipes, or unit from an empty list.
65    ///
66    /// `fields` holds one recipe per field of the source record, in declaration
67    /// order. Binding verifies the count against the record.
68    pub fn fields(fields: impl IntoIterator<Item = Input>) -> Self {
69        Self::Fields(fields.into_iter().collect())
70    }
71    /// Call a named source constructor with the supplied argument recipes.
72    ///
73    /// Binding checks its result type and argument count. Argument validation and
74    /// ownership acquisition precede the source call.
75    ///
76    /// # Arguments
77    ///
78    /// * `name` — unqualified name of a source function whose result type is the
79    ///   input's type.
80    /// * `args` — one recipe per parameter of that function, in order.
81    pub fn constructor(name: impl Into<String>, args: impl IntoIterator<Item = Input>) -> Self {
82        Self::Constructor {
83            function: name.into(),
84            args: args.into_iter().collect(),
85        }
86    }
87    /// Choose between owned representations of the same input type.
88    ///
89    /// `arms` holds one recipe per alternative; each must produce the input's
90    /// type. The caller passes the zero-based index of the chosen alternative in an
91    /// extra wire ahead of all arms' wires.
92    ///
93    /// Requires at least one arm. Each arm retains its own wires; this does not
94    /// merge equivalent wire parameters or generate destination-language overloads.
95    pub fn selector(arms: impl IntoIterator<Item = Input>) -> Self {
96        Self::Selector(arms.into_iter().collect())
97    }
98}
99/// A result conversion recipe, interpreted together with its source type and codec input data.
100///
101/// `Fields` observes fields; `OwnedFields` moves them. Explicit `Projected`
102/// leaves select borrowed or owned conversion independently of type contents.
103#[derive(Clone, Debug)]
104pub enum Output {
105    /// Deliver `Ok` through its recipe and map an owned `Err` to a message.
106    /// The mapper must be a source function taking exactly `E` or `&E` and returning `String`.
107    /// The codec selects how that message reaches the destination's error channel.
108    Result {
109        /// Conversion of the owned success value.
110        ok: Box<Output>,
111        /// Unqualified source error-message function name.
112        error: String,
113    },
114    /// Convert an optional payload without preparing absent values.
115    Optional(Box<Output>),
116    /// Convert the occurrence type itself with [`Codec::output_leaf`].
117    Leaf,
118    /// Observe named fields in declaration order; an empty list represents unit.
119    Fields(Vec<Output>),
120    /// Move named fields into their output conversions.
121    OwnedFields(Vec<Output>),
122    /// Convert a leaf with an explicit ownership policy.
123    Projected(Projection),
124    /// Call a consuming accessor, then convert its owned result.
125    Into {
126        /// Source accessor taking exactly the owned occurrence type.
127        function: String,
128        /// Conversion of its owned result.
129        output: Box<Output>,
130    },
131    /// A shared-borrow accessor, followed by the output conversion of its result.
132    Getter {
133        /// Source getter taking exactly one shared reference to the current type.
134        function: String,
135        /// Conversion of the getter's owned result; borrowed results are unsupported.
136        output: Box<Output>,
137    },
138    /// Produce several views of one value. At most one child may consume it.
139    /// Observations run before consumption, while wire order follows this list.
140    Combine(Vec<Output>),
141}
142impl Output {
143    /// Observe record fields in declaration order; use an empty list for unit.
144    ///
145    /// `fields` holds one recipe per field of the source record, in declaration order.
146    pub fn fields(fields: impl IntoIterator<Item = Output>) -> Self {
147        Self::Fields(fields.into_iter().collect())
148    }
149    /// Observe a value through a source getter and convert the getter's result.
150    ///
151    /// The getter must take exactly `&T` for the current type `T`, and return an
152    /// owned value. Binding rejects mismatched signatures.
153    ///
154    /// # Arguments
155    ///
156    /// * `name` — unqualified name of the source getter.
157    /// * `output` — recipe for the getter's result.
158    pub fn getter(name: impl Into<String>, output: Output) -> Self {
159        Self::Getter {
160            function: name.into(),
161            output: Box::new(output),
162        }
163    }
164    /// Produce multiple outputs from one value.
165    ///
166    /// `parts` holds one recipe per view of the value; their wires appear in this
167    /// order. Binding rejects an empty list or more than one consuming part. The
168    /// framework uses [`Codec::consumes`] to order observations before a move.
169    pub fn combine(parts: impl IntoIterator<Item = Output>) -> Self {
170        Self::Combine(parts.into_iter().collect())
171    }
172    fn consumes<C: Codec>(&self, ty: &Type, extra: &C::Extra, codec: &C) -> bool {
173        match self {
174            Self::Optional(child) => child.consumes(option_inner(ty).unwrap_or(ty), extra, codec),
175            Self::Leaf => codec.consumes(ty, extra),
176            Self::Projected(p) => p.mode == ProjectionMode::Owned,
177            Self::OwnedFields(_) | Self::Into { .. } | Self::Result { .. } => true,
178            Self::Combine(p) => p.iter().any(|p| p.consumes(ty, extra, codec)),
179            _ => false,
180        }
181    }
182}

184/// Language-specific leaf conversions. Recursive composition stays in this declaration compiler.
185///
186/// A leaf is a source type the adapter converts itself: a value copied across
187/// the boundary or an object kept in Rust. `prebindgen-conversions` provides helpers
188/// for both representations. The
189/// adapter chooses by type; a recipe only says where the leaves are.
190///
191/// Input leaves must validate without consuming in `validation`, then
192/// acquire ownership without failure or source calls in `acquire`. Each acquired
193/// value needs an ordinary Rust owner so later failures drop it. `statements`
194/// can construct arguments from those owners. Every input wire must retain
195/// the supplied activation condition. Output leaves keep ownership in Rust
196/// locals until their infallible `commit` runs after successful delivery.
197///
198/// # Implementing a codec
199///
200/// This minimal codec uses Rust values as its carriers. It illustrates converter
201/// construction without a foreign ABI or object storage. A language adapter
202/// additionally writes wrappers and chooses native and destination-language
203/// carrier types for its wire kinds.
204///
205/// ```
206/// use prebindgen_flat::{Result, validate_bindings};
207/// use prebindgen_conversions::value;
208/// use prebindgen_declarations::{
209///     Codec, CodecInput, CodecOutput, Declarations,
210///     Bindings, fun, expand_param, expand_return, bind_input, Input,
211/// };
212/// use proc_macro2::TokenStream;
213/// use quote::quote;
214/// use syn::{Ident, Type};
215///
216/// use prebindgen_declarations::Condition;
217/// fn carrier(kind: value::Value) -> Type {
218///     match kind {
219///         value::Value::Scalar(p) => p.rust_type(),
220///         value::Value::Text => syn::parse_quote!(String),
221///         value::Value::Bytes => syn::parse_quote!(Vec<u8>),
222///     }
223/// }
224/// struct RustCarriers;
225/// impl Codec for RustCarriers {
226///     type Niche = ();
227///     type Group = ();
228///     type Extra = ();
229///     type Context = ();
230///     type Input = CodecInput<Self>;
231///     type Output = CodecOutput<Self>;
232///     type Wire = Type; // The Rust carrier type is the complete wire description.
233///
234///     fn optional_group(&self, _: Ident, _: Ident, _: Option<()>) {}
235///     fn field_group(&self, _: Ident) {}
236///     fn fields_group(&self) {}
237///     fn niche(&self, _: &Self::Wire) -> Option<()> { None }
238///     fn set_niche(&self, _: &mut Self::Wire, _: Option<()>) {}
239///     fn selector(&self, _: &()) -> Self::Wire { syn::parse_quote!(i32) }
240///     fn consumes(&self, _: &Type, _: &()) -> bool { false }
241///     fn input_leaf(
242///         &self, ty: &Type, _: &(), wire: Ident, var: Ident, active: &Condition,
243///     ) -> Result<CodecInput<Self>> {
244///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
245///         let decode = quote!(#wire.clone());
246///         value::input(ty, wire, var, &active.rust(), decode, carrier(kind))
247///     }
248///     fn output_leaf(
249///         &self, ty: &Type, _: &(), wire: Ident, v: TokenStream, borrowed: bool,
250///     ) -> Result<CodecOutput<Self>> {
251///         let kind = value::Value::of(ty).ok_or("only values cross this codec")?;
252///         value::output(ty, wire, v, borrowed, |v| v, carrier(kind))
253///     }
254/// }
255///
256/// let source = syn::parse_file(r#"
257///     pub fn from_i32(value: i32) -> i64 { value as i64 }
258///     pub fn to_i32(value: &i64) -> i32 { *value as i32 }
259///     pub fn echo(value: i64) -> i64 { value }
260/// "#)?;
261/// let mut flat = Bindings::<Type>::parse(
262///     source.items.into_iter().map(|item| (item, Default::default())),
263/// )?;
264/// let defaults = Declarations::new(&flat, vec![
265///     expand_param!(i64).variant(fun!(from_i32)).variant_self().into(),
266///     expand_return!(i64).field(fun!(to_i32)).field_self().into(),
267/// ])?;
268/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
269/// let function = flat.function("echo")?;
270/// // Input: selector, constructor argument, direct value. Output: two values.
271/// assert_eq!(function.params[0].ty.slot.as_ref().unwrap().layout.len(), 3);
272/// assert_eq!(function.ret.slot.as_ref().unwrap().layout.len(), 2);
273/// // Replace just this parameter's converter; later default filling preserves it.
274/// bind_input(&mut flat, &RustCarriers, "echo", "value", &Input::Leaf, &())?;
275/// defaults.fill(&mut flat, &RustCarriers, &fun!(echo), &())?;
276/// validate_bindings(&flat)?;
277/// let input = flat.function("echo")?.params[0].ty.slot.as_ref().unwrap();
278/// assert_eq!(input.layout.len(), 1);
279/// assert_eq!(prebindgen_flat::type_key(&input.layout[0].kind), "i64");
280/// assert!(input.resources.is_empty()); // No foreign ownership in this codec.
281/// # Ok::<(), Box<dyn std::error::Error>>(())
282/// ```
283pub trait Codec {
284    /// Adapter-owned description of an absent wire representation. Use `()` if unused.
285    /// The shared compiler never interprets this metadata.
286    type Niche: Clone;
287    /// Complete adapter-owned metadata for a parent node in the wire tree.
288    /// The shared compiler creates it through the group hooks below and never
289    /// interprets it. Use `()` when no group information is needed.
290    type Group: Clone;
291    /// Describe an optional value in the adapter's own metadata format.
292    /// `presence` is the normalized 0/1 binding used by payload conditions.
293    /// `discriminator` names a descendant wire; `niche` is its reserved absent
294    /// representation, or `None` when a separate flag is the first child.
295    /// With a flag, `presence` and `discriminator` name that same wire.
296    fn optional_group(
297        &self,
298        presence: syn::Ident,
299        discriminator: syn::Ident,
300        niche: Option<Self::Niche>,
301    ) -> Self::Group;
302    /// Describe one named source field, whose subtree may contain any number of wires.
303    fn field_group(&self, name: syn::Ident) -> Self::Group;
304    /// Describe a record or unit value's ordered collection of field subtrees.
305    /// The shared helpers check field-name uniqueness before calling this hook.
306    fn fields_group(&self) -> Self::Group;
307
308    /// Input data borrowed while building one occurrence's conversion; use `()` if unused.
309    /// The same value is passed through nested conversions. It is not stored in Flat;
310    /// hooks retain only the data needed for emission in their code and wires.
311    type Extra;
312    /// Context supplied by the language generator when emitting stored templates.
313    type Context: prebindgen_conversions::TemplateContext + 'static;
314    /// Final self-contained input converter installed after preparation.
315    type Input: prebindgen_flat::Conversion + From<CodecInput<Self>>;
316    /// Final self-contained output converter installed after preparation.
317    type Output: prebindgen_flat::Conversion + From<CodecOutput<Self>>;
318    /// Preserve a logical path in private preparation metadata when the codec needs it.
319    /// The final converter decides how to use this mapping; Flat never interprets it.
320    /// Called from the innermost component outward; prepend each segment to its path.
321    fn prefix_component(&self, _wire: &mut Self::Wire, _segment: ComponentSegment) {}
322
323    /// Complete description of one wire, including its type and adapter metadata.
324    /// Grouping belongs to parent nodes; no wire kind contains child wires.
325    type Wire: prebindgen_flat::WireKind + Clone + std::fmt::Debug;
326    /// Describe an `i32` selector tag or optional presence flag.
327    ///
328    /// Selector arm indices start at zero; optional flags use 0 and 1. A codec
329    /// may advertise excluded negative values as niches for enclosing options.
330    /// A newly added flag retains its own niches; inactive payload wires do not.
331    /// `extra` is the occurrence data of the conversion that contains the selector.
332    fn selector(&self, extra: &Self::Extra) -> Self::Wire;
333    /// Whether a leaf output consumes the current source value.
334    ///
335    /// This must agree with [`Self::output_leaf`]. It governs move ordering and
336    /// double-consumption rejection in [`Output::Combine`], not input ownership.
337    /// A non-consuming leaf receives a shared reference with `borrowed` set.
338    ///
339    /// # Arguments
340    ///
341    /// * `ty` — owned source type of the value being delivered.
342    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`].
343    fn consumes(&self, ty: &Type, extra: &Self::Extra) -> bool;
344    /// Prepare a leaf input of source type `ty` using occurrence data `extra`.
345    ///
346    /// All fallible checks belong in [`InputPlan::validation`]; do not
347    /// consume values there. Return a generation error when this type or access
348    /// mode is unsupported.
349    ///
350    /// # Arguments
351    ///
352    /// * `ty` — the source parameter type at this position, as written: `T`,
353    ///   `&T` or `&mut T`.
354    /// * `extra` — occurrence data of the conversion, as passed to [`bind_input`]
355    ///   or [`Declarations::fill`].
356    /// * `wire` — fresh identifier for the boundary parameter this leaf adds.
357    ///   A leaf that needs more wires or locals derives their names from it.
358    /// * `var` — fresh identifier for the Rust local that holds the converted value.
359    ///   [`InputPlan::argument`] normally refers to it.
360    /// * `active` — the selector arms under which this leaf is used. Copy it to every
361    ///   returned wire and resource, and guard validation and acquisition with it.
362    fn input_leaf(
363        &self,
364        ty: &Type,
365        extra: &Self::Extra,
366        wire: syn::Ident,
367        var: syn::Ident,
368        active: &Condition,
369    ) -> Result<CodecInput<Self>>;
370    /// Convert a leaf with explicit ownership, independently of its usual policy.
371    ///
372    /// The compiler supplies either an owned value or a shared reference according
373    /// to `projection.mode`. The default supports leaves without field overrides.
374    fn output_projected(
375        &self,
376        ty: &Type,
377        extra: &Self::Extra,
378        wire: syn::Ident,
379        value: TokenStream,
380        projection: &Projection,
381    ) -> Result<CodecOutput<Self>> {
382        if !projection.fields.is_empty() {
383            return Err("this output converter does not support field projection overrides".into());
384        }
385        let mut payload = ty;
386        while let Some(inner) = option_inner(payload) {
387            payload = inner;
388        }
389        if projection.mode == ProjectionMode::Owned && matches!(payload, Type::Reference(_)) {
390            return Err("owned projection cannot consume a reference payload".into());
391        }
392        self.output_leaf(
393            ty,
394            extra,
395            wire,
396            value,
397            projection.mode == ProjectionMode::Borrowed,
398        )
399    }
400    /// Adapt a source-error message to this language's error channel.
401    ///
402    /// `message` evaluates to an owned `String`; the returned expression must also
403    /// produce a `String`. It may first report an adapter-specific error, such as
404    /// throwing a JVM exception. The compiler then returns `Err` from the active
405    /// fallible call before success-output preparation or ownership commit.
406    /// The default rejects Result error routing rather than guessing a policy.
407    fn result_error(
408        &self,
409        _ty: &Type,
410        _message: TokenStream,
411        _extra: &Self::Extra,
412    ) -> Result<TokenStream> {
413        Err("Result error routing is unsupported by this codec".into())
414    }
415    /// The unused representation currently available on this wire.
416    /// Initialize it when creating the wire and return the state last written by
417    /// `set_niche`. Never offer a representation already used by an inner option
418    /// or accepted as a present leaf payload. It may describe one value or a set.
419    fn niche(&self, wire: &Self::Wire) -> Option<Self::Niche>;
420    /// Update the unused representation stored in a wire's own description.
421    /// The compiler calls this after reserving a niche or disabling a conditional wire.
422    /// Codecs must preserve this update in their wire description. A codec with
423    /// no niches can implement this as a no-op and always return `None` from `niche`.
424    fn set_niche(&self, wire: &mut Self::Wire, niche: Option<Self::Niche>);
425    /// Recognize presence without decoding the payload. Called on original inner metadata.
426    /// The expression must be total and side-effect-free for every initialized wire
427    /// value, including inactive placeholders; it must not dereference raw pointers.
428    /// It returns false for every representation reserved by this niche.
429    fn is_present(
430        &self,
431        _wire: &Self::Wire,
432        _niche: &Self::Niche,
433        _value: TokenStream,
434    ) -> Result<TokenStream> {
435        Err("niche presence testing is unsupported by this codec".into())
436    }
437    /// Remaining excluded representation after reserving `niche` for one optional layer.
438    /// The compiler stores the result in the wire through `set_niche`. Returning
439    /// `None` exhausts this domain; enclosing options then use another wire or a flag.
440    fn remaining_niche(&self, _wire: &Self::Wire, _niche: &Self::Niche) -> Option<Self::Niche> {
441        None
442    }
443    /// Inert wire expression used for an absent payload, optionally using a niche.
444    fn absent_value(
445        &self,
446        _wire: &Self::Wire,
447        _niche: Option<&Self::Niche>,
448        _input: bool,
449    ) -> Result<TokenStream> {
450        Err("optional values are unsupported by this codec".into())
451    }
452    /// Complete a composed parameter plan before it is stored in a slot.
453    /// Called once per bound parameter, including records and selectors.
454    fn finish_input(&self, _plan: &mut CodecInput<Self>, _extra: &Self::Extra) -> Result<()> {
455        Ok(())
456    }
457    /// Complete a composed result plan, including unit and multi-wire results.
458    /// Install call-level delivery here, after all output expressions are known.
459    fn finish_output(&self, _plan: &mut CodecOutput<Self>, _extra: &Self::Extra) -> Result<()> {
460        Ok(())
461    }
462    /// Prepare delivery of the Rust expression `value` as one or more wires.
463    ///
464    /// Keep unpublished allocations under Rust ownership and transfer them only in
465    /// [`OutputPlan::commit`]. Return a generation error for unsupported
466    /// conversions. Install final delivery in [`Codec::finish_output`], rather
467    /// than on nested leaf plans; recursive composition rejects leaf delivery.
468    ///
469    /// # Arguments
470    ///
471    /// * `ty` — the owned source type of the value.
472    /// * `extra` — occurrence data of the conversion, as passed to [`bind_output`]
473    ///   or [`Declarations::fill`].
474    /// * `wire` — fresh identifier for the delivered value. A leaf that needs more
475    ///   wires or locals derives their names from it.
476    /// * `value` — Rust expression producing the value to deliver.
477    /// * `borrowed` — `true` when `value` is a shared reference to `ty`, such as a
478    ///   record field observed in place; do not move out of it. `false` when
479    ///   `value` is owned.
480    fn output_leaf(
481        &self,
482        ty: &Type,
483        extra: &Self::Extra,
484        wire: syn::Ident,
485        value: TokenStream,
486        borrowed: bool,
487    ) -> Result<CodecOutput<Self>>;
488}
489/// A flat model whose slots contain only conversion data for wrapper emission.
490///
491/// Unbound source helpers have `None` slots. Bound functions need `Some` for their
492/// result and every parameter; [`prebindgen_flat::validate_bindings`] checks this before emission.
493/// Record fields retain only names and source types. Recipes and [`Codec::Extra`]
494/// are construction inputs, not stored in this model.
495///
496/// Output conversion data cannot be attached to an input position:
497/// ```compile_fail
498/// use prebindgen_declarations::Bindings;
499/// fn swap(flat: &mut Bindings<syn::Type>) {
500///     let function = flat.function_mut("foo").unwrap();
501///     function.params[0].ty.slot = function.ret.slot.clone();
502/// }
503/// ```
504/// Nor can a recipe be stored in a slot:
505/// ```compile_fail
506/// use prebindgen_declarations::{Bindings, Input};
507/// fn configure(flat: &mut Bindings<syn::Type>) {
508///     flat.function_mut("foo").unwrap().params[0].ty.slot = Some(Input::Leaf);
509/// }
510/// ```
511pub type Bindings<W, C = (), G = ()> = Flat<InputPlan<W, C, G>, OutputPlan<W, C, G>>;

513/// Build and attach a converter to a named function parameter.
514///
515/// Resolves the occurrence's source type from `flat`, checks recipe compatibility,
516/// and recursively composes wires and code. Nested conversions use the supplied
517/// `extra`, not data from referenced constructors or getters. Recipes and `extra`
518/// are borrowed only for this call. A failure leaves the existing slot unchanged.
519/// Rebinding an occurrence replaces its converter without changing other slots.
520/// Generated names are scoped by parameter position, so separate calls cannot
521/// collide within the wrapper. Keep the function signature fixed after binding.
522/// Check completeness with [`prebindgen_flat::validate_bindings`].
523///
524/// # Arguments
525///
526/// * `flat` — the model holding the function; the parameter's slot is replaced.
527/// * `codec` — the adapter's leaf conversions.
528/// * `function` — unqualified name of the source function.
529/// * `parameter` — name of the parameter, as written in the source.
530/// * `recipe` — how to build the argument from boundary inputs.
531/// * `extra` — occurrence data passed to every codec call for this conversion.
532///
533/// Input binding rejects output recipes:
534/// ```compile_fail
535/// use prebindgen_declarations::{bind_input, Codec, Output};
536/// use prebindgen_flat::Flat;
537/// fn wrong<C: Codec>(flat: &mut Flat<C::Input, C::Output>, codec: &C, extra: &C::Extra) {
538///     bind_input(flat, codec, "foo", "x", &Output::Leaf, extra).unwrap();
539/// }
540/// ```
541pub fn bind_input<C: Codec>(
542    flat: &mut Flat<C::Input, C::Output>,
543    codec: &C,
544    function: &str,
545    parameter: &str,
546    recipe: &Input,
547    extra: &C::Extra,
548) -> Result<()> {
549    let f = flat.function(function)?;
550    let index = f
551        .params
552        .iter()
553        .position(|p| p.name == parameter)
554        .ok_or_else(|| format!("`{function}`: unknown parameter `{parameter}`"))?;
555    let mut conversion = Compiler::new(flat, codec, format!("p{index}")).input(
556        &f.params[index].ty.ty,
557        recipe,
558        extra,
559        &Condition::default(),
560    )?;
561    codec.finish_input(&mut conversion, extra)?;
562    conversion.layout.validate()?;
563    flat.function_mut(function)?.params[index].ty.slot = Some(conversion.into());
564    Ok(())
565}

567/// Build and attach a converter to a function result, replacing any existing one.
568///
569/// Uses the same construction and failure rules as [`bind_input`]. Output code
570/// uses [`prebindgen_conversions::source_result_expr`] for the source call's result local.
571/// The recipe is checked for compatible types and signatures, and for repeated
572/// consumption of the same value. Unit results use `Output::fields([])`.
573///
574/// # Arguments
575///
576/// * `flat` — the model holding the function; the result's slot is replaced.
577/// * `codec` — the adapter's leaf conversions.
578/// * `function` — unqualified name of the source function.
579/// * `recipe` — how to deliver the result as boundary outputs.
580/// * `extra` — occurrence data passed to every codec call for this conversion.
581///
582/// Output binding rejects input recipes:
583/// ```compile_fail
584/// use prebindgen_declarations::{bind_output, Codec, Input};
585/// use prebindgen_flat::Flat;
586/// fn wrong<C: Codec>(flat: &mut Flat<C::Input, C::Output>, codec: &C, extra: &C::Extra) {
587///     bind_output(flat, codec, "foo", &Input::Leaf, extra).unwrap();
588/// }
589/// ```
590pub fn bind_output<C: Codec>(
591    flat: &mut Flat<C::Input, C::Output>,
592    codec: &C,
593    function: &str,
594    recipe: &Output,
595    extra: &C::Extra,
596) -> Result<()> {
597    let mut conversion = Compiler::new(flat, codec, "ret".into()).output_root(
598        &flat.function(function)?.ret.ty,
599        recipe,
600        extra,
601        prebindgen_conversions::source_result_expr(),
602        false,
603    )?;
604    codec.finish_output(&mut conversion, extra)?;
605    conversion.validate_value_count()?;
606    conversion.layout.validate()?;
607    flat.function_mut(function)?.ret.slot = Some(conversion.into());
608    Ok(())
609}

611struct Compiler<'a, C: Codec> {
612    flat: &'a Flat<C::Input, C::Output>,
613    codec: &'a C,
614    prefix: String,
615    next: std::cell::Cell<usize>,
616}
617impl<'a, C: Codec> Compiler<'a, C> {
618    fn new(flat: &'a Flat<C::Input, C::Output>, codec: &'a C, prefix: String) -> Self {
619        Self {
620            flat,
621            codec,
622            prefix,
623            next: std::cell::Cell::new(0),
624        }
625    }
626    fn name(&self) -> syn::Ident {
627        let next = self.next.get();
628        let n = format_ident!("{}_w{next}", self.prefix);
629        self.next.set(next + 1);
630        n
631    }
632    fn prefix(&self, wires: &mut WireLayout<C::Wire, C::Group>, segment: ComponentSegment) {
633        for wire in wires {
634            self.codec.prefix_component(&mut wire.kind, segment.clone());
635        }
636    }
637    fn input(
638        &self,
639        ty: &Type,
640        recipe: &Input,
641        extra: &C::Extra,
642        active: &Condition,
643    ) -> Result<CodecInput<C>> {
644        if contains_result(ty) {
645            return Err("Result inputs require an explicit data representation; error routing is output-only".into());
646        }
647        let var = self.name();
648        let mut wires = WireLayout::default();
649        let mut validation = TokenStream::new();
650        let mut acquire = TokenStream::new();
651        let mut resources = Vec::new();
652        let mut acknowledgement = None;
653        let mut statements = TokenStream::new();
654        let argument = quote!(#var);
655        match recipe {
656            Input::Optional(inner) => {
657                let ty = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
658                let tag = self.name();
659                let child = self.input(ty, inner, extra, &active.arm(tag.clone(), 1))?;
660                return optional_input(self.codec, extra, tag, var, active, child);
661            }
662            Input::Leaf => {
663                let w = self.name();
664                return self.codec.input_leaf(ty, extra, w, var, active);
665            }
666            Input::Fields(children) => {
667                if type_key(ty) == "()" && children.is_empty() {
668                    wires.enclose(self.codec.fields_group());
669                    return Ok(InputPlan {
670                        context: Default::default(),
671                        acknowledgement,
672                        layout: wires,
673                        validation,
674                        acquire,
675                        resources,
676                        statements,
677                        argument: quote!(()),
678                    });
679                }
680                let r = self.flat.record(ty)?;
681                if r.fields.len() != children.len() {
682                    return Err(format!(
683                        "`{}`: expected {} field converters, got {}",
684                        type_key(ty),
685                        r.fields.len(),
686                        children.len()
687                    ));
688                }
689                return record_input(
690                    self.codec,
691                    ty,
692                    var,
693                    r.fields.iter().zip(children).map(|(field, child)| {
694                        Ok((
695                            field.name.clone(),
696                            self.input(&field.ty, child, extra, active)?,
697                        ))
698                    }),
699                );
700            }
701            Input::Constructor { function, args } => {
702                let f = self.flat.function(function)?;
703                if !same_type(&f.ret.ty, ty) {
704                    return Err(format!(
705                        "constructor `{function}` returns `{}`, expected `{}`",
706                        type_key(&f.ret.ty),
707                        type_key(ty)
708                    ));
709                }
710                if f.params.len() != args.len() {
711                    return Err(format!("constructor `{function}`: wrong argument count"));
712                }
713                let mut call_args = Vec::new();
714                for (param, child) in f.params.iter().zip(args) {
715                    let mut p = self.input(&param.ty.ty, child, extra, active)?;
716                    self.prefix(
717                        &mut p.layout,
718                        ComponentSegment::Argument(param.name.to_string()),
719                    );
720                    validation.extend(p.validation);
721                    acquire.extend(p.acquire);
722                    resources.extend(p.resources);
723                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
724                    statements.extend(p.statements);
725                    wires.extend(p.layout);
726                    call_args.push(p.argument);
727                }
728                let path = &f.path;
729                statements.extend(quote!(let #var:#ty=#path(#(#call_args),*);));
730            }
731            Input::Selector(arms) => {
732                if arms.is_empty() {
733                    return Err("an input selector needs at least one alternative".into());
734                }
735                if matches!(ty, Type::Reference(_)) {
736                    return Err("selector results must be owned in this subset".into());
737                }
738                let tag = self.name();
739                wires.push(Wire {
740                    name: tag.clone(),
741                    kind: {
742                        let mut kind = self.codec.selector(extra);
743                        self.codec
744                            .prefix_component(&mut kind, ComponentSegment::Selector);
745                        kind
746                    },
747                });
748                let count = arms.len() as i32;
749                let invalid = active.and(quote!(!(0..#count).contains(&#tag)));
750                validation.extend(quote!(if #invalid {return Err("invalid selector".into());}));
751                let mut cases = Vec::new();
752                for (i, arm) in arms.iter().enumerate() {
753                    let mut p = self.input(ty, arm, extra, &active.arm(tag.clone(), i as i32))?;
754                    self.prefix(&mut p.layout, ComponentSegment::Alternative(i));
755                    for wire in p.layout.iter_mut() {
756                        self.codec.set_niche(&mut wire.kind, None);
757                    }
758                    validation.extend(p.validation);
759                    acquire.extend(p.acquire);
760                    resources.extend(p.resources);
761                    merge_acknowledgement(&mut acknowledgement, p.acknowledgement)?;
762                    let body = p.statements;
763                    let value = p.argument;
764                    wires.extend(p.layout);
765                    let i = i as i32;
766                    cases.push(quote!(#i=>{#body #value}));
767                }
768                statements = quote!(let #var:#ty=match #tag{#(#cases,)* _=>unreachable!("validated selector")};);
769            }
770        }
771        Ok(InputPlan {
772            context: Default::default(),
773            acknowledgement,
774            layout: wires,
775            validation,
776            acquire,
777            resources,
778            statements,
779            argument,
780        })
781    }
782    fn output_root(
783        &self,
784        ty: &Type,
785        recipe: &Output,
786        extra: &C::Extra,
787        value: TokenStream,
788        borrowed: bool,
789    ) -> Result<CodecOutput<C>> {
790        if let Output::Result { ok, error } = recipe {
791            if borrowed {
792                return Err("Result error routing requires an owned Result occurrence".into());
793            }
794            let (success_ty, error_ty) =
795                result_parts(ty).ok_or("Result recipe requires Result<T, E>")?;
796            let mapper = self.flat.function(error)?;
797            let mapper_arg = mapper.params.first().map(|p| &p.ty.ty);
798            let shared = matches!(mapper_arg, Some(Type::Reference(r)) if r.mutability.is_none() && same_type(&r.elem, error_ty));
799            if mapper.params.len() != 1
800                || !(mapper_arg.is_some_and(|ty| same_type(ty, error_ty)) || shared)
801                || type_key(&mapper.ret.ty) != "String"
802            {
803                return Err(
804                    "Result error mapper must take exactly E or &E and return String".into(),
805                );
806            }
807            let payload = self.name();
808            let failure = self.name();
809            let path = &mapper.path;
810            let argument = if shared {
811                quote!(&#failure)
812            } else {
813                quote!(#failure)
814            };
815            let message = self
816                .codec
817                .result_error(error_ty, quote!(#path(#argument)), extra)?;
818            let mut child = self.output(success_ty, ok, extra, quote!(#payload), false)?;
819            let prepare = child.statements;
820            child.statements = quote! {
821                let #payload = match #value {
822                    ::core::result::Result::Ok(value) => value,
823                    ::core::result::Result::Err(#failure) => return Err(#message),
824                };
825                #prepare
826            };
827            return Ok(child);
828        }
829        self.output(ty, recipe, extra, value, borrowed)
830    }
831    fn output(
832        &self,
833        ty: &Type,
834        recipe: &Output,
835        extra: &C::Extra,
836        value: TokenStream,
837        borrowed: bool,
838    ) -> Result<CodecOutput<C>> {
839        if contains_result(ty) {
840            return Err("Result error routing is supported only at the outermost return with an explicit mapper".into());
841        }
842        let mut out = OutputPlan {
843            context: Default::default(),
844            delivery: Default::default(),
845            commit: TokenStream::new(),
846            layout: Default::default(),
847            statements: TokenStream::new(),
848            values: Vec::new(),
849        };
850        match recipe {
851            Output::Result { .. } => {
852                return Err("Result error routing is supported only at the outermost return".into())
853            }
854            Output::Optional(inner) => {
855                let ty = option_inner(ty).ok_or("optional recipe requires Option<T>")?;
856                let tag = self.name();
857                let payload = self.name();
858                let child = self.output(ty, inner, extra, quote!(#payload), borrowed)?;
859                return optional_output(self.codec, extra, tag, payload, value, borrowed, child);
860            }
861            Output::Leaf => {
862                let w = self.name();
863                return self.codec.output_leaf(ty, extra, w, value, borrowed);
864            }
865            Output::Projected(projection) => {
866                if borrowed && projection.mode == ProjectionMode::Owned {
867                    return Err(
868                        "owned projection cannot consume a shared-reference occurrence".into(),
869                    );
870                }
871                let value = if !borrowed && projection.mode == ProjectionMode::Borrowed {
872                    quote!(&(#value))
873                } else {
874                    value
875                };
876                return self
877                    .codec
878                    .output_projected(ty, extra, self.name(), value, projection);
879            }
880            Output::Fields(children) | Output::OwnedFields(children) => {
881                let owned = matches!(recipe, Output::OwnedFields(_));
882                if owned && borrowed {
883                    return Err("owned fields cannot consume a shared-reference occurrence".into());
884                }
885                if type_key(ty) == "()" && children.is_empty() {
886                    out.layout.enclose(self.codec.fields_group());
887                    return Ok(out);
888                }
889                let r = self.flat.record(ty)?;
890                if r.fields.len() != children.len() {
891                    return Err("wrong output field count".into());
892                }
893                let convert = |index: usize, value| {
894                    self.output(&r.fields[index].ty, &children[index], extra, value, !owned)
895                };
896                return if owned {
897                    record_output_owned(
898                        self.codec,
899                        self.name(),
900                        value,
901                        r.fields.iter().map(|f| f.name.clone()),
902                        convert,
903                    )
904                } else {
905                    record_output(
906                        self.codec,
907                        self.name(),
908                        value,
909                        r.fields.iter().map(|f| f.name.clone()),
910                        convert,
911                    )
912                };
913            }
914            Output::Getter { function, output } | Output::Into { function, output } => {
915                let consuming = matches!(recipe, Output::Into { .. });
916                if consuming && borrowed {
917                    return Err(
918                        "consuming accessor cannot consume a shared-reference occurrence".into(),
919                    );
920                }
921                let f = self.flat.function(function)?;
922                let expected: Type = if consuming {
923                    ty.clone()
924                } else {
925                    syn::parse_quote!(&#ty)
926                };
927                if f.params.len() != 1 || !same_type(&f.params[0].ty.ty, &expected) {
928                    return Err(format!(
929                        "getter `{function}` must take exactly `{}`",
930                        type_key(&expected)
931                    ));
932                }
933                if matches!(f.ret.ty, Type::Reference(_)) {
934                    return Err("borrowed getter results are outside this subset".into());
935                }
936                let v = self.name();
937                let path = &f.path;
938                let arg = if borrowed || consuming {
939                    value
940                } else {
941                    quote!(&#value)
942                };
943                out.statements = quote!(let #v=#path(#arg););
944                let mut child = self.output(&f.ret.ty, output, extra, quote!(#v), false)?;
945                self.prefix(
946                    &mut child.layout,
947                    ComponentSegment::Getter(function.clone()),
948                );
949                append(&mut out, child)?;
950            }
951            Output::Combine(parts) => {
952                if parts.is_empty() {
953                    return Err("an output combination needs at least one part".into());
954                }
955                if parts
956                    .iter()
957                    .filter(|p| p.consumes(ty, extra, self.codec))
958                    .count()
959                    > 1
960                {
961                    return Err("a combination cannot consume the same value twice".into());
962                }
963                let mut bodies = Vec::new();
964                for (index, child) in parts.iter().enumerate() {
965                    let consuming = child.consumes(ty, extra, self.codec);
966                    let arg = if borrowed || consuming {
967                        value.clone()
968                    } else {
969                        quote!(&#value)
970                    };
971                    let mut p = self.output(ty, child, extra, arg, borrowed || !consuming)?;
972                    self.prefix(&mut p.layout, ComponentSegment::View(index));
973                    bodies.push((consuming, std::mem::take(&mut p.statements)));
974                    append(&mut out, p)?;
975                }
976                // Wire order follows the declaration. Observations execute before the move.
977                bodies.sort_by_key(|(consumes, _)| *consumes);
978                for (_, body) in bodies {
979                    out.statements.extend(body);
980                }
981            }
982        }
983        Ok(out)
984    }
985}
986fn append<W: prebindgen_flat::WireKind, C, M>(
987    to: &mut OutputPlan<W, C, M>,
988    from: OutputPlan<W, C, M>,
989) -> Result<()> {
990    if !from.delivery.statements.is_empty()
991        || from.delivery.result.to_string() != quote!(()).to_string()
992    {
993        return Err("composed output delivery must be installed by Codec::finish_output".into());
994    }
995    to.commit.extend(from.commit);
996    to.layout.extend(from.layout);
997    to.statements.extend(from.statements);
998    to.values.extend(from.values);
999    Ok(())
1000}

1021fn merge_acknowledgement(
1022    target: &mut Option<TokenStream>,
1023    incoming: Option<TokenStream>,
1024) -> Result<()> {
1025    if let Some(code) = incoming {
1026        if let Some(previous) = target {
1027            if previous.to_string() != code.to_string() {
1028                return Err("conflicting input acknowledgements".into());
1029            }
1030        } else {
1031            *target = Some(code);
1032        }
1033    }
1034    Ok(())
1035}
