1//! Output (data) expansion — the dual of constructor expansion
2//! ([`crate::expand`]). A function returning a rich type is *decomposed* by a
3//! **deconstructor** into a set of leaf values.
4//!
5//! A **deconstructor** (a type-level `expand_return!` `.field*` list,
6//! or the per-fn `.expand_return` override) is a
7//! **deterministic product**: every record always runs and contributes its leaf
8//! — there is no selector (unlike a *constructor*, whose selector picks one
9//! variant). A record's accessor is a `#[prebindgen]` function `f(&T) -> &F` (a
10//! reference return where possible, for zero-copy); an accessor whose return
11//! type has its own deconstructor splices the child's records with prefixed
12//! leaf names.
13//!
14//! Two **deliveries** (see [`Delivery`]), derived from the resolved leaf count:
15//! * `Callback` — replaces the return with a foreign **callback** receiving
16//!   all the leaves (any leaf count).
17//! * `Return` — **returns** the single leaf value directly (no callback);
18//!   requires a single-leaf decomposition.
19//!
20//! Resolution is language-agnostic: it turns the declarations into
21//! [`UnfoldPlan`]s (stored on the registry, keyed by function ident) and
22//! registers every leaf's `out_ty` as a required **output** so the resolver
23//! produces its converter (and projection). The jnigen adapter reads the
24//! plan at the return-emission site.
25//!
26//! [`Iterable`]: UnfoldShape::Iterable

54/// One output-side registration a decomposition asks the registry for.
55///
56/// Applying the declarations reads the model and produces plans; the only thing
57/// it needs the registry itself for is to say which readings must cross on the
58/// output side. Those asks are recorded here, in the order they were made, and
59/// replayed onto the registry's type tables in one place — so the plans can be
60/// built before a registry exists, and nothing but the replay mutates it.
61///
62/// The order is part of the meaning: [`Self::Unrequire`] drops a demand an
63/// earlier [`Self::Output`] made.
64#[derive(Clone, Debug)]
65pub enum Requirement {
66    /// This reading must cross on the output side, and a converter for it must
67    /// resolve.
68    Output(prebindgen_flat::flat::TypeRef),
69    /// This reading enters the output table without demanding a converter — a
70    /// type a plan *names* rather than one that crosses.
71    Reference(prebindgen_flat::flat::TypeRef),
72    /// Drop an earlier demand that this reading's converter must resolve. The
73    /// table entry stays, so a converter is still produced if one resolves.
74    Unrequire(prebindgen_flat::flat::TypeRef),
75}

78/// plans they produce, and the registrations they ask for.

80/// The adapter that declared the decompositions owns this. It holds no

92/// The decomposition plans applying a set of declarations produced.
93///
94/// The adapter keeps this and reads it back at every emission site; nothing in
95/// the registry holds one.
96#[derive(Default, Clone)]
97pub struct Unfolded {
98    /// Resolved output-expansion plans, keyed by function ident. Read at the
99    /// return-emission site.
100    pub unfold_plans: std::collections::HashMap<syn::Ident, UnfoldPlan>,
101
102    /// Resolved **error**-position expansion plans, keyed by function ident: the
103    /// decomposition of a fallible fn's `Result<_, E>` domain error `E` (from
104    /// `.convert_error` / `.deconstruct_error`). Separate from
105    /// [`Self::unfold_plans`] — a fn may have both an output and an error plan.
106    pub error_plans: std::collections::HashMap<syn::Ident, UnfoldPlan>,
107
108    /// Default decomposition of a **callback argument** type — the `T` of a
109    /// declared fn's `impl Fn(T, …)` parameter — keyed by the bare arg type
110    /// (type-level, fn-independent), from the type's default deconstructor
111    /// (`by_ref = false`: the trampoline owns the value). A type without a
112    /// default deconstructor has no entry and is delivered whole.
113    pub callback_arg_plans: std::collections::HashMap<TypeKey, UnfoldPlan>,
114
115    /// The declaration-default decomposition per deconstructor declaration
116    /// ([`DeconId`]) — resolved once with normalized inputs, independent of
117    /// using functions and processing order. The single source an adapter
118    /// derives declaration-keyed signature artifacts (generated callback
119    /// interfaces, say) from, so every function selecting the same declaration
120    /// sees one signature by construction.
121    pub decon_plans: std::collections::HashMap<DeconId, DeconSpec>,
122}

124impl<'f> Unfolding<'f> {
125    /// Start applying declarations against `flat`.
126    pub fn new(flat: &'f prebindgen_flat::flat::Flat) -> Self {
127        Self {
128            flat,
129            requirements: Vec::new(),
130            plans: Unfolded::default(),
131        }
132    }
133
134    /// The model the declarations are read against.
135    pub fn flat(&self) -> &'f prebindgen_flat::flat::Flat {
136        self.flat
137    }
138
139    /// The registrations asked for, in the order they were asked. Hand these to
140    /// [`Decompositions::requirements`](crate::Decompositions::requirements).
141    pub fn requirements(&self) -> &[Requirement] {
142        &self.requirements
143    }
144
145    /// Per callback-argument type, the readings its decomposition delivers.
146    ///
147    /// The ordering fact the registry needs and cannot see: a callback argument
148    /// delivered as leaves needs each leaf's own conversion before the
149    /// callback's can be built, and a leaf is named by a plan rather than by the
150    /// argument's syntax. Hand this to
151    /// [`Decompositions::callback_arg_leaves`](crate::Decompositions::callback_arg_leaves).
152    pub fn callback_arg_leaves(
153        &self,
154    ) -> std::collections::HashMap<TypeKey, Vec<prebindgen_flat::flat::TypeRef>> {
155        self.plans
156            .callback_arg_plans
157            .iter()
158            .map(|(key, plan)| {
159                (
160                    key.clone(),
161                    plan.leaves.iter().map(|leaf| leaf.out_ty.clone()).collect(),
162                )
163            })
164            .collect()
165    }
166
167    /// Take the plans, leaving the requirements behind.
168    pub fn into_plans(self) -> Unfolded {
169        self.plans
170    }
171
172    /// Register `reading` (and its nested positions) as a required **output** so
173    /// the resolver produces a converter for it.
174    fn require_output(&mut self, reading: &prebindgen_flat::flat::TypeRef) {
175        self.requirements.push(Requirement::Output(reading.clone()));
176    }
177
178    /// Register `reading` as an output cell **without** demanding a converter —
179    /// a type some plan names rather than one that crosses. What a
180    /// [`SumTag`](LeafSource::SumTag) selector needs: it names *which* sum it
181    /// chooses between, and that sum has no whole-value output converter at all,
182    /// so requiring one would fail resolution (#282).
183    fn reference_output(&mut self, reading: &prebindgen_flat::flat::TypeRef) {
184        self.requirements
185            .push(Requirement::Reference(reading.clone()));
186    }
187
188    /// Drop `reading` from the required-output set. Used by
189    /// [`apply_leaf_vec_folds`]: when a `Vec<T>` / `Option<Vec<T>>` return is
190    /// delivered element-by-element through a fold, the whole-collection
191    /// converter is genuinely not needed — and for a `Vec` of opaque handles it
192    /// cannot resolve at all, so requiring it would wrongly fail resolution.
193    fn unrequire_output(&mut self, reading: &prebindgen_flat::flat::TypeRef) {
194        self.requirements
195            .push(Requirement::Unrequire(reading.clone()));
196    }
197}

203/// One record (field) of a deconstructor. A deconstructor is a product: every
204/// record contributes a leaf.
205// large_enum_variant: a handful of records exist per binding — boxing the
206// syn payloads would only complicate the arms (same trade-off as
207// `ConvertSourceKind`).
208#[allow(clippy::large_enum_variant)]
209#[derive(Clone)]
210pub enum DeconRecord {
211    /// Read this field by calling the accessor function `f(&T) -> &F`. `name`
212    /// is the author-supplied leaf name, used **literally** (no casing /
213    /// stripping); it may not contain the reserved `"__"` chain separator.
214    /// An accessor whose return type has its own deconstructor splices that
215    /// child's records with the leaf names prefixed `name__<child>`.
216    Acc { func: syn::Ident, name: String },
217    /// Read this field by calling a **custom, locally-defined** accessor: any
218    /// callable in the binding crate (`path`) with a STATED return type
219    /// (`ty`) — there is no `#[prebindgen]` item behind it, so the signature
220    /// cannot be looked up. The adapter's `local_functions()` pre-pass
221    /// synthesizes a registry entry from the stated signature (so call
222    /// qualification and `Option`-nesting checks work unchanged); splicing
223    /// follows [`Self::Acc`] rules except that a self-referential field (its
224    /// type already being decomposed) degrades to a plain converter leaf
225    /// instead of a cycle error — that is what lets such a field re-deliver
226    /// (part of) the value itself, e.g. under a binding-defined condition.
227    LocalAcc { path: syn::Path, name: String },
228    /// The value itself — the handle/identity leaf (cloned for a `&T` return,
229    /// moved for an owned `T`). At most one per
230    /// deconstructor.
231    Identity,
232    /// Read the fields of the type's **value form**: call `func` once
233    /// (`f(&T) -> TStruct`) and contribute one record per [`FieldRecord`],
234    /// reached by field access on the returned struct. The language adapter
235    /// builds the field list (it knows which structs are declared classes and
236    /// therefore inline); this record only says how to get there.
237    ///
238    /// Each field then decomposes exactly like an [`Acc`](Self::Acc) record's
239    /// return does — its own `records` if the
240    /// declaration overrode it, else its type's own deconstructor if it has
241    /// one, else one leaf — so a value form and a hand-written field list
242    /// produce the same leaves.
243    Fields {
244        func: syn::Ident,
245        /// The accessor **consumes** its receiver (`f(T) -> TStruct`): the
246        /// value is moved in and each field moved *out* into its leaf, instead
247        /// of being cloned out of a borrow. Declared by the adapter rather than
248        /// read off the signature — giving the value away is a boundary
249        /// decision — and cross-checked against the signature when the records
250        /// are flattened, so the two cannot drift.
251        consuming: bool,
252        fields: Vec<FieldRecord>,
253    },
254}

256/// One field of a value form (see [`DeconRecord::Fields`]).
257#[derive(Clone)]
258pub struct FieldRecord {
259    /// Field-access chain from the value form's returned struct. More than one
260    /// element when the adapter inlined a nested declared class.
261    pub members: Vec<syn::Ident>,
262    /// The leaf name (already `__`-joined across inlined nesting).
263    pub name: String,
264    /// The field's **reading**, `Option` / `Vec` layers included — and its
265    /// syntax with it, which is what a leaf's `out_ty` spells.
266    ///
267    /// The declaration carries this rather than naming a `syn::Type` for the
268    /// walk to look up, because there was nothing to look up: a field record is
269    /// built from an element whose every field already has a reading, and the
270    /// types it names are the ones the caller registers *after* the walk
271    /// returns. Asking the registry here was asking before registration
272    /// (#266) — a lookup that could only miss, answered by a second source of
273    /// readings that hid the ordering.
274    pub ty: prebindgen_flat::flat::TypeRef,
275    /// How this field decomposes.
276    pub decon: FieldDecon,
277}

279/// How one [`FieldRecord`] decomposes.
280#[derive(Clone)]
281pub enum FieldDecon {
282    /// By the field type's own deconstructor if it has one, else one leaf —
283    /// the same default a [`DeconRecord::Acc`] record's return follows.
284    Default,
285    /// Explicit records, replacing the type default wholesale (the declaration
286    /// stated this field's complete leaf set).
287    Records(Vec<DeconRecord>),
288    /// Leaves the **adapter** built, appended with this field's path and name
289    /// prefixed onto each. For shapes whose leaf structure only the adapter
290    /// knows — a decomposed sum, which is a selector plus one group per
291    /// alternative rather than a product of records.
292    Leaves(Vec<UnfoldLeaf>),
293}

391/// Structural validation of the declaration records — duplicate targets,
392/// collected (EVERY offender before failing) so a build surfaces all
393/// declaration problems at once. Empty record lists are NOT diagnosed:
394/// an empty inline list is the valid whole-element delivery form.
395fn validate_declarations(acc: &Deconstructors) -> Result<(), UnfoldError> {
396    let mut entries: Vec<UnfoldDeclError> = Vec::new();
397    let mut decon_targets: std::collections::HashSet<String> = std::collections::HashSet::new();
398    for d in &acc.deconstructors {
399        let target = d.target.as_str().to_string();
400        if !decon_targets.insert(target.clone()) {
401            entries.push(UnfoldDeclError::DuplicateDeconstructor { target });
402        }
403    }
404    let mut output_keys: std::collections::HashSet<(String, DeconTarget)> =
405        std::collections::HashSet::new();
406    for od in &acc.outputs {
407        if !output_keys.insert((od.func.to_string(), od.target)) {
408            entries.push(UnfoldDeclError::DuplicateOutput {
409                func: od.func.clone(),
410                target: od.target,
411            });
412        }
413    }
414    if entries.is_empty() {
415        Ok(())
416    } else {
417        Err(UnfoldError::InvalidDeclarations { entries })
418    }
419}

421/// Resolve every output-expansion declaration (explicit + `.default()`
422/// auto-applied) into an [`UnfoldPlan`], register each leaf's `out_ty` as a
423/// required output, and store the plans on the registry (`unfold_plans` for
424/// `Output`, `error_plans` for `Error`).
425///
426/// `declared_fns` is the adapter's claimed `#[prebindgen]` fn set — the domain
427/// over which `.default()` deconstructors are auto-applied. `accessor_fns` is
428/// the `.fun_accessor` subset — the only functions a decomposer record may
429/// reference.
430///
431/// Runs inside `write_rust` after `expand::apply` and before `resolve`, so leaf
432/// converters resolve through the normal rank machinery.
433pub fn apply(
434    registry: &mut Unfolding<'_>,
435    acc: &Deconstructors,
436    declared_fns: &std::collections::HashSet<syn::Ident>,
437    accessor_fns: &std::collections::HashSet<syn::Ident>,
438) -> Result<(), UnfoldError> {
439    validate_declarations(acc)?;
440    // Binding-local accessors (`LocalAcc` records) resolve through registry
441    // entries synthesized by the adapter's `local_functions()` pre-pass in
442    // the builder's scan — by this point they read exactly like
443    // `#[prebindgen]` accessors.
444
445    // Gate: every accessor-function record of every declared deconstructor must
446    // be a `.fun_accessor` (the single source of truth for "accessor").
447    // Binding-local records skip the gate — there is no `#[prebindgen]` item
448    // behind them — but keep the reserved-separator name check.
449    for d in &acc.deconstructors {
450        check_records(&d.records, accessor_fns)?;
451    }
452
453    // Explicit decls first; they take precedence over (and suppress) a default
454    // for the same `(fn, target)`.
455    let mut done: std::collections::HashSet<(syn::Ident, DeconTarget)> = Default::default();
456    for ed in &acc.outputs {
457        // Per-fn decl cross-check: the decl's declared type must match the
458        // fn's peeled (`Option`/`Vec`/`&`) return type — the typo guard for
459        // `.expand_return(expand_return!(T)…)`.
460        if let Some(declared) = &ed.declared_source {
461            let ret = registry
462                .flat()
463                .function(&ed.func)
464                .map(|f| f.ret.clone())
465                .ok_or_else(|| UnfoldError::UnknownFunction(ed.func.clone()))?;
466            if !returns_type(&ret, declared) {
467                return Err(UnfoldError::ReturnTypeMismatch {
468                    func: ed.func.clone(),
469                    declared: declared.as_str().to_string(),
470                    actual: ret.to_string(),
471                });
472            }
473        }
474        // Identity-only field set = the raw whole-value return (the
475        // complete-set rule: "the set is {self}"): no plan — mark done so the
476        // type default doesn't re-apply, and let the return cross through the
477        // type's ordinary output converter (borrowed-`&T`-capable).
478        if let DeconSel::Inline(records) = &ed.sel {
479            if matches!(records.as_slice(), [DeconRecord::Identity]) {
480                done.insert((ed.func.clone(), ed.target));
481                continue;
482            }
483        }
484        process_decl(registry, acc, ed)?;
485        done.insert((ed.func.clone(), ed.target));
486    }
487
488    // Default auto-apply: a type's deconstructor (`expand_return!`) is
489    // applied to every declared fn that returns it (Output) or has it as a
490    // `Result<_, E>` error (Error), unless the fn is `fun_accessor` or has a
491    // per-fn override. `Delivery` is recomputed from leaf count inside
492    // `process_decl` for Output (1 ⇒ Return, N ⇒ Callback).
493    for d in &acc.deconstructors {
494        if d.default.is_none() {
495            continue;
496        }
497        let dkey = d.target.clone();
498        let sel = DeconSel::TopLevel;
499        for func in declared_fns {
500            // Read accessors are never output-decomposed (they ARE the records).
501            if accessor_fns.contains(func) {
502                continue;
503            }
504            let Some(ret) = registry.flat().function(&func).map(|f| f.ret.clone()) else {
505                continue;
506            };
507            // Error position: fn returns `Result<_, E>` and `E == d.target`.
508            if let Some(err_ty) = ret.fallible_parts().map(|(_, e)| e) {
509                if err_ty.key() == dkey && done.insert((func.clone(), DeconTarget::Error)) {
510                    process_decl(
511                        registry,
512                        acc,
513                        &OutputDecl {
514                            func: func.clone(),
515                            sel: sel.clone(),
516                            target: DeconTarget::Error,
517                            delivery: Delivery::Callback,
518                            declared_source: None,
519                        },
520                    )?;
521                }
522            }
523            // Output position: fn returns `T` / `&T` / `Option<T|&T>` / `Vec<T>`
524            // with `T == d.target` (Result returns keep a handle — factories).
525            if returns_type(&ret, &dkey)
526                && !acc.skip_output.contains(func)
527                && done.insert((func.clone(), DeconTarget::Output))
528            {
529                process_decl(
530                    registry,
531                    acc,
532                    &OutputDecl {
533                        func: func.clone(),
534                        sel: sel.clone(),
535                        target: DeconTarget::Output,
536                        delivery: Delivery::Callback,
537                        declared_source: None,
538                    },
539                )?;
540            }
541        }
542    }
543
544    // Callback-argument decomposition: each `T` of a declared fn's
545    // `impl Fn(T, …)` parameter is delivered per `T`'s default deconstructor —
546    // the same default output a *return* of `T` would use — so the foreign
547    // callback receives the flattened leaves in one crossing instead of a
548    // whole value. Plans are type-level (keyed by `T`, fn-independent) with
549    // `by_ref = false` (the trampoline owns the value, so a root identity
550    // record moves it). Delivery is always `Callback` regardless of leaf count
551    // (there is no return-value lane in a callback invocation). A type without
552    // a default deconstructor gets no plan and is delivered whole.
553    for func in declared_fns {
554        let Some(params) = registry.flat().function(&func).map(|f| f.params.clone()) else {
555            continue;
556        };
557        for param in &params {
558            // The callback's argument types, read off the parameter's
559            // classification. `TypeKind::Callback` carries them as `TypeRef`s, so
560            // there is nothing to re-extract from the signature's syntax.
561            let prebindgen_flat::flat::TypeKind::Callback { args } = param.ty.kind() else {
562                continue;
563            };
564            for arg_ty in args {
565                // A borrowed arg (`impl Fn(&T)`) decomposes through the same
566                // machinery as a `&T` return: strip the leading `&` to reach the
567                // deconstructor target and set `by_ref` so the leaves are read
568                // (cloned) through the reference instead of by move. The plan is
569                // keyed under the ACTUAL arg type (`&T`) — that is what
570                // `callback_input`/`callback_iface_spec` look up.
571                let (by_ref, core_ty) = peel_borrow(arg_ty);
572                // Only a NAMED core can match a deconstructor target: an
573                // `Option<T>` / `Vec<T>` / tuple arg is delivered whole. The model
574                // says which, and `unwrapped` is where a wrapper the destination
575                // cannot see — `Box<T>` — stops reading as un-nameable.
576                if !matches!(
577                    core_ty.unwrapped().kind(),
578                    prebindgen_flat::flat::TypeKind::Named { .. }
579                ) {
580                    continue;
581                }
582                let key = arg_ty.key();
583                if registry.plans.callback_arg_plans.contains_key(&key) {
584                    continue;
585                }
586                let core_key = core_ty.key();
587                let Some(d) = acc
588                    .deconstructors
589                    .iter()
590                    .find(|d| d.default.is_some() && d.target == core_key)
591                else {
592                    continue;
593                };
594                let ed = OutputDecl {
595                    func: func.clone(),
596                    sel: DeconSel::TopLevel,
597                    target: DeconTarget::Output,
598                    delivery: Delivery::Callback,
599                    declared_source: None,
600                };
601                let decon = decl_id(&core_key, d);
602                let records = d.records.clone();
603                register_decon_spec(registry, acc, &decon, &records, core_ty)?;
604                let plan = build_plan(
605                    acc,
606                    registry,
607                    &ed,
608                    by_ref,
609                    core_ty,
610                    UnfoldShape::Base,
611                    &records,
612                    decon,
613                )?;
614                if plan.leaves.is_empty() {
615                    continue;
616                }
617                for leaf in &plan.leaves {
618                    registry.require_output(&leaf.out_ty);
619                }
620                registry.plans.callback_arg_plans.insert(key, plan);
621            }
622        }
623    }
624    Ok(())
625}

627/// A synthesized by-value `data_class` decomposition, produced by the language
628/// adapter (which knows the per-field encoding — projections, enums, nested
629/// classes) and handed to [`apply_value_structs`].
630/// Its [`leaves`](Self::leaves)
631/// are [`LeafSource::Reach`] leaves: each crosses the boundary as its own field
632/// value and the foreign side reassembles the object (no Java object is built
633/// on the Rust side).
634pub struct ValueDecon {
635    /// Canonical key of the value struct (the `DeconId::Default` key).
636    pub key: TypeKey,
637    /// The struct type (owned) the leaves decompose.
638    pub source: prebindgen_flat::flat::TypeRef,
639    /// Field-access leaves in foreign-signature / `fromParts` order.
640    pub leaves: Vec<UnfoldLeaf>,
641}

643/// Wire the synthesized by-value `data_class` decompositions into the registry:
644/// register each as a `DeconId::Default` [`DeconSpec`], then build a
645/// **fixed-builder** [`UnfoldPlan`] for every declared function that returns the
646/// struct (`T` / `&T` / `Option<T>` / `Vec<T>`) and a callback-arg plan for
647/// every `impl Fn(&T)` / `impl Fn(T)` parameter. Each leaf's `out_ty` is
648/// registered as a required output. Mirrors the per-function matching of
649/// [`apply`], but the builder/folder is a fixed foreign singleton
650/// (`fixed_builder = true`) reconstructing the concrete class, so delivery is
651/// always `Callback` (never the single-leaf `Return` shortcut) and the wrapper
652/// stays non-generic.
653///
654/// Runs in `write_rust` right after [`apply`] and before `resolve`.
655pub fn apply_value_structs(
656    registry: &mut Unfolding<'_>,
657    decons: Vec<ValueDecon>,
658    declared_fns: &std::collections::HashSet<syn::Ident>,
659) -> Result<(), UnfoldError> {
660    for vd in &decons {
661        let decon = wire_fixed_decon(registry, &vd.key, &vd.source, &vd.leaves)?;
662
663        // Output position: a declared fn returning the struct
664        // (`T` / `&T` / `Option<T|&T>` / `Vec<T|&T>`) decomposes into a
665        // fixed-builder plan. (`Result<T, E>` is left to the whole-value
666        // converter — the synthesizer covers the infallible returns.)
667        wire_fixed_returns(registry, vd, &decon, declared_fns, false);
668
669        // Callback-argument position: an `impl Fn(&T)` / `impl Fn(T)` parameter
670        // of a declared fn delivers the flattened leaves to the foreign
671        // callback, which reassembles the whole value via the data class's
672        // `fromParts` before invoking the user's typed callback (the group
673        // reassembly lives in the JNI adapter's `asRaw` proxy).
674        wire_fixed_callbacks(registry, vd, &decon, declared_fns)?;
675    }
676    Ok(())
677}

679/// A synthesized **sum** decomposition, produced by the language adapter (which
680/// knows how each payload encodes) and handed to [`apply_sum_returns`] — the
681/// selector-carrying sibling of [`ValueDecon`].
682///
683/// Its [`leaves`](Self::leaves) are a [`LeafSource::SumTag`] selector followed
684/// by one **group** per alternative ([`LeafSource::VariantField`] leaves
685/// carrying [`UnfoldLeaf::groups`]). Exactly one group is live per value; the
686/// emitter reads the whole list as ONE `match` over the value, filling every
687/// inert slot with its wire default. The foreign side picks the live group by
688/// the tag and rebuilds the alternative, so no object is built on the Rust
689/// side.
690pub struct SumDecon {
691    /// Canonical key of the sum type (the `DeconId::Default` key).
692    pub key: TypeKey,
693    /// The enum type (owned) the leaves decompose.
694    pub source: prebindgen_flat::flat::TypeRef,
695    /// The tag leaf followed by every variant's group, in tag order.
696    pub leaves: Vec<UnfoldLeaf>,
697}

699/// Wire the synthesized **sum** decompositions into the registry — the
700/// [`apply_value_structs`] analog for a value whose alternatives are chosen at
701/// runtime instead of being a fixed product.
702///
703/// For every declared function returning the sum (`E` / `&E` / `Option<E>` /
704/// `Vec<E>`) and every `impl Fn(E)` / `impl Fn(&E)` callback parameter, builds a
705/// **fixed-builder** [`UnfoldPlan`] over the tag + group leaves, registering
706/// each leaf's `out_ty` as a required output.
707///
708/// A sum has no converter of its own (it is boundary-only: a tag plus groups is
709/// not a single wire), so the declared return's scan-time output requirement —
710/// including the `Option<E>` / `Vec<E>` layers, which the boundary-only pass
711/// does not reach — is dropped here as the plan takes over.
712///
713/// Put every leaf's `out_ty` in the table, and demand a converter for the ones
714/// that need one.
715///
716/// **Every leaf is registered; only a converter-bearing leaf is a root** (#282).
717/// The two are separate facts and this is the one place a sum plan states both:
718/// a cell says the type entered the pipeline, a root says the binding needs its
719/// conversion to resolve. The `SumTag` selector is registered and not required —
720/// it names *which* sum it chooses between, and a sum has no whole-value output
721/// converter, so requiring one would fail resolution over a type that never
722/// crosses whole.
723///
724/// This used to `filter` the selector out entirely, which left its `out_ty`
725/// with a cell only when the adapter happened to declare the sum separately —
726/// true for jnigen via `export_type`, and not true at all for a registry
727/// assembled without declarations. The invariant holds by construction now
728/// rather than by declaration order.
729fn register_leaves(registry: &mut Unfolding<'_>, leaves: &[UnfoldLeaf]) {
730    for leaf in leaves {
731        if leaf.has_converter() {
732            registry.require_output(&leaf.out_ty);
733        } else {
734            registry.reference_output(&leaf.out_ty);
735        }
736    }
737}

758/// Register the declaration-canonical [`DeconSpec`] of a synthesized
759/// decomposition (first writer wins) and return its identity.
760fn wire_fixed_decon(
761    registry: &mut Unfolding<'_>,
762    key: &TypeKey,
763    source: &prebindgen_flat::flat::TypeRef,
764    leaves: &[UnfoldLeaf],
765) -> Result<DeconId, UnfoldError> {
766    let decon = DeconId::Default(key.to_string());

768    registry
769        .plans
770        .decon_plans
771        .entry(decon.clone())
772        .or_insert_with(|| DeconSpec {
773            source: source.clone(),
774            leaves: leaves.to_vec(),
775        });
776    Ok(decon)

779/// Build the fixed-builder output plan for every declared fn returning the
780/// decomposed type (`T` / `&T` / `Option<T|&T>` / `Vec<T|&T>`). `no_converter`
781/// marks a type that has no whole-value converter at all (a sum), so the
782/// declared return's scan-time output requirement is dropped as the plan
783/// replaces it.
784fn wire_fixed_returns(
785    registry: &mut Unfolding<'_>,
786    vd: &ValueDecon,
787    decon: &DeconId,
788    declared_fns: &std::collections::HashSet<syn::Ident>,
789    no_converter: bool,
790) {
791    for func in declared_fns {
792        let Some(ret) = registry.flat().function(&func).map(|f| f.ret.clone()) else {
793            continue;
794        };
795        if !returns_type(&ret, &vd.key) || registry.plans.unfold_plans.contains_key(func) {
796            continue;
797        }
798        // Shape over the leaf decomposition: peel an outer `Option`, then a
799        // `Vec`, then a leading `&`. `Vec<T|&T>` ⇒ Iterable (a **fixed
800        // folder**: each element's leaves cross raw and the foreign folder
801        // rebuilds it + appends, so no Java object is built on the Rust
802        // side); `Option<…>` wraps the inner shape in Optional (`None` ⇒ a
803        // null result). `element: None` keeps the decomposed-leaf path. The
804        // element/inner borrow-ness sets `by_ref` (the reach clones either
805        // way).
806        let layers = peel(&ret);
807        let by_ref = layers.by_ref;
808        // The model's layer stack is the plan's shape — `UnfoldShape` is `Shape`
809        // — so there is nothing to rebuild here.
810        let shape = layers.shape.clone();
811        if no_converter {
812            // The plan delivers the return leaf-by-leaf, so no converter is
813            // needed for the declared return — and for a sum none can exist.
814            // Drop the scan-time registrations of every layer (the boundary-only
815            // pass only reaches the bare type), so the missing converters are not
816            // flagged as unresolved-required.
817            // EVERY layer, the `Vec` element included. The shape fold peels here,
818            // so the matching unrequire belongs here; leaving the element out made
819            // the invariant depend on the adapter's `boundary_only_types` covering
820            // it — true for JniGenBuilder today, and the only reason a
821            // `Vec<sum>`-only declaration resolves.
822            for layer in &layers.layer_types {
823                registry.unrequire_output(layer);
824            }
825        }
826        register_leaves(registry, &vd.leaves);
827        let plan = UnfoldPlan {
828            source: vd.source.clone(),
829            decon: Some(decon.clone()),
830            by_ref,
831            shape,
832            leaves: vd.leaves.clone(),
833            element: None,
834            delivery: Delivery::Callback,
835            convert_out_ty: None,
836            fixed_builder: true,
837            hoists: Vec::new(),
838        };
839        registry.plans.unfold_plans.insert(func.clone(), plan);
840    }
841}

843/// Build a fixed-builder callback-arg plan for every `impl Fn(&T)`,
844/// `impl Fn(T)`, or `impl Fn(Option<T>)` parameter (of a declared fn) whose
845/// value is the decomposed type `vd`. The foreign callback receives the
846/// flattened leaves (reassembled there) instead of a whole value built on the
847/// Rust side. Separate from the
848/// output-position wiring so the callback path (which needs the foreign-side
849/// group-reassembly adapter) can be enabled on its own.
850fn wire_fixed_callbacks(
851    registry: &mut Unfolding<'_>,
852    vd: &ValueDecon,
853    decon: &DeconId,
854    declared_fns: &std::collections::HashSet<syn::Ident>,
855) -> Result<(), UnfoldError> {
856    for func in declared_fns {
857        let Some(params) = registry.flat().function(&func).map(|f| f.params.clone()) else {
858            continue;
859        };
860        for param in &params {
861            // The callback's argument types, read off the parameter's
862            // classification. `TypeKind::Callback` carries them as `TypeRef`s, so
863            // there is nothing to re-extract from the signature's syntax.
864            let prebindgen_flat::flat::TypeKind::Callback { args } = param.ty.kind() else {
865                continue;
866            };
867            for arg_ty in args {
868                // Peel a leading `&`, then one by-value `Option`, then detect a
869                // slice element. A scalar value-struct arg decomposes into a
870                // `Base` fixed builder; `Option<T>` wraps that in `Optional` and
871                // adds a presence slot. A run becomes an `Iterable` fixed FOLDER.
872                // Optional runs are deliberately left to their own shape step.
873                let (by_ref, after_ref) = peel_borrow(arg_ty);
874                let (optional, after_optional) = match after_ref.optional_inner() {
875                    Some(inner) if !by_ref => (true, inner),
876                    _ => (false, after_ref),
877                };
878                // A run of `T` is an Iterable fold over the element; anything else
879                // is a Base fold over the value itself. `Sequence` is the one
880                // question, and it covers `[T]` and `Vec<T>` alike.
881                let (shape, matches_key) = match after_optional.sequence_elem() {
882                    Some(elem) if !optional => (
883                        UnfoldShape::Iterable(Box::new(UnfoldShape::Base)),
884                        elem.key() == vd.key,
885                    ),
886                    Some(_) => continue,
887                    None => (UnfoldShape::Base, after_optional.key() == vd.key),
888                };
889                let shape = if optional {
890                    UnfoldShape::Optional((), Box::new(shape))
891                } else {
892                    shape
893                };
894                if !matches_key {
895                    continue;
896                }
897                let key = arg_ty.key();
898                if registry.plans.callback_arg_plans.contains_key(&key) {
899                    continue;
900                }
901                register_leaves(registry, &vd.leaves);
902                let plan = UnfoldPlan {
903                    source: vd.source.clone(),
904                    decon: Some(decon.clone()),
905                    by_ref,
906                    shape,
907                    leaves: vd.leaves.clone(),
908                    element: None,
909                    delivery: Delivery::Callback,
910                    convert_out_ty: None,
911                    fixed_builder: true,
912                    hoists: Vec::new(),
913                };
914                registry.plans.callback_arg_plans.insert(key, plan);
915            }
916        }
917    }
918    Ok(())
919}

921/// Wire **whole-element** `Iterable` fold plans for bare `Vec<T>` /
922/// `Option<Vec<T>>` returns and `impl Fn(&[T])` callback args whose element `T`
923/// is a single leaf (String, scalar, opaque handle) nominated by the adapter
924/// via `Prebindgen::leaf_vec_fold_elements`. Each
925/// such position crosses as decoupled raw leaves folded into a **foreign-built**
926/// list — the single-leaf dual of [`apply_value_structs`] (which handles
927/// multi-field `data_class` elements). The fold is a **fixed** foreign singleton
928/// (`fixed_builder = true`): the wrapper allocates the list, passes the hoisted
929/// appender, and returns the concrete `List<T>` (never a caller `fold` param), so
930/// no `java.util.ArrayList` is built on the Rust side.
931///
932/// Runs right after [`apply_value_structs`]; skips any function/arg that already
933/// carries a plan (an explicit `.deconstruct_output`, a `data_class` fold, …) so
934/// declared decompositions and value-struct folds win.
935pub fn apply_leaf_vec_folds(
936    registry: &mut Unfolding<'_>,
937    elements: Vec<TypeKey>,
938    declared_fns: &std::collections::HashSet<syn::Ident>,
939) -> Result<(), UnfoldError> {
940    if elements.is_empty() {
941        return Ok(());
942    }
943    let elem_keys = elements;
944    // Is the leading-`&`-peeled `bare` one of the nominated single-leaf elements?
945    let is_nominated = |bare: &prebindgen_flat::flat::TypeRef| elem_keys.contains(&bare.key());
946    for func in declared_fns {
947        let Some(params) = registry.flat().function(&func).map(|f| f.params.clone()) else {
948            continue;
949        };
950        // Output position: `Vec<T>` / `Option<Vec<T>>` return. Skip if a plan
951        // already exists (declared deconstructor / value-struct fold).
952        if !registry.plans.unfold_plans.contains_key(func) {
953            let Some(ret) = registry.flat().function(&func).map(|f| f.ret.clone()) else {
954                continue;
955            };
956            let (optional, after_opt) = match ret.optional_inner() {
957                Some(inner) => (true, inner),
958                None => (false, &ret),
959            };
960            if let Some(vec_elem) = after_opt.sequence_elem() {
961                let bare = peel_borrow(vec_elem).1;
962                if is_nominated(bare) {
963                    let inner_shape = UnfoldShape::Iterable(Box::new(UnfoldShape::Base));
964                    let shape = if optional {
965                        UnfoldShape::Optional((), Box::new(inner_shape))
966                    } else {
967                        inner_shape
968                    };
969                    registry.require_output(vec_elem);
970                    // The fold delivers the return element-by-element, so the
971                    // whole `Vec<T>` / `Option<Vec<T>>` converter is not needed.
972                    // De-require it: for String / scalar elements it still
973                    // resolves (and is emitted as harmless dead code); for an
974                    // opaque-handle element it cannot resolve (`jlong` wire isn't
975                    // JObject-shaped), and de-requiring keeps that `None` from
976                    // being flagged as an unresolved-required error.
977                    registry.unrequire_output(&ret);
978                    registry
979                        .plans
980                        .unfold_plans
981                        .insert(func.clone(), whole_leaf_fold_plan(vec_elem, shape));
982                }
983            }
984        }
985        // Callback-arg position: `impl Fn(&[T])` / `impl Fn([T])`.
986        for param in &params {
987            // The callback's argument types, read off the parameter's
988            // classification. `TypeKind::Callback` carries them as `TypeRef`s, so
989            // there is nothing to re-extract from the signature's syntax.
990            let prebindgen_flat::flat::TypeKind::Callback { args } = param.ty.kind() else {
991                continue;
992            };
993            for arg_ty in args {
994                let (_, after_ref) = peel_borrow(arg_ty);
995                let Some(elem) = after_ref.sequence_elem() else {
996                    continue;
997                };
998                if !is_nominated(peel_borrow(elem).1) {
999                    continue;
1000                }
1001                let key = arg_ty.key();
1002                if registry.plans.callback_arg_plans.contains_key(&key) {
1003                    continue;
1004                }
1005                registry.require_output(elem);
1006                let plan =
1007                    whole_leaf_fold_plan(elem, UnfoldShape::Iterable(Box::new(UnfoldShape::Base)));
1008                registry.plans.callback_arg_plans.insert(key, plan);
1009            }
1010        }
1011    }
1012    Ok(())
1013}

1036/// The deconstructor gate: every accessor-function record must be a declared
1037/// `.fun_accessor` (the single source of truth for "accessor"), and no author
1038/// leaf name may contain the reserved `"__"` chain separator. Binding-local
1039/// records skip the accessor check — there is no `#[prebindgen]` item behind
1040/// them — but keep the name check.
1041///
1042/// Recurses into a value form's per-field override records, so an override is
1043/// held to the same rules as the declaration it replaces.
1044fn check_records(
1045    records: &[DeconRecord],
1046    accessor_fns: &HashSet<syn::Ident>,
1047) -> Result<(), UnfoldError> {
1048    for rec in records {
1049        let (func, name) = match rec {
1050            DeconRecord::Acc { func, name } => (Some(func), name),
1051            DeconRecord::LocalAcc { name, .. } => (None, name),
1052            DeconRecord::Identity => continue,
1053            // A value form's field names come from struct idents, not from the
1054            // author, so the `"__"` in an inlined nested name is the separator
1055            // doing its job. An author-supplied rename is checked where it is
1056            // declared.
1057            DeconRecord::Fields { func, fields, .. } => {
1058                if !accessor_fns.contains(func) {
1059                    return Err(UnfoldError::RecordNotAccessor { func: func.clone() });
1060                }
1061                for fr in fields {
1062                    if let FieldDecon::Records(recs) = &fr.decon {
1063                        check_records(recs, accessor_fns)?;
1064                    }
1065                }
1066                continue;
1067            }
1068        };
1069        if name.contains("__") {
1070            return Err(UnfoldError::ReservedSeparator { name: name.clone() });
1071        }
1072        if let Some(func) = func {
1073            if !accessor_fns.contains(func) {
1074                return Err(UnfoldError::RecordNotAccessor { func: func.clone() });
1075            }
1076        }
1077    }
1078    Ok(())
1079}

1081/// The arity layers over `ty`, the types they wrap, and the value underneath.
1082///
1083/// A thin owned view over
1084/// [`TypeRef::layer_stack`](prebindgen_flat::flat::TypeRef::layer_stack) and
1085/// [`layer_types`](prebindgen_flat::flat::TypeRef::layer_types): the borrows are
1086/// resolved to clones because these feed plan fields and registry calls that own
1087/// their types. The classification is the model's; only the copying is local, and
1088/// it happens **once**, where a value is stored — not on every question asked.
1089///
1090/// The stack **is** the plan's shape — `UnfoldShape` is `Shape` — so a caller
1091/// stores it rather than rebuilding one from flags.
1092struct Layered {
1093    /// The arity layers, outermost first.
1094    shape: UnfoldShape,
1095    /// Every type on the way down, outermost first — what a registration walks.
1096    layer_types: Vec<prebindgen_flat::flat::TypeRef>,
1097    /// Past the borrow too: what actually crosses — as an **identity**, which
1098    /// is all its one consumer ever asked of it.
1099    core: TypeKey,
1100    /// Whether the core is reached through a borrow.
1101    by_ref: bool,
1102}
1103
1104/// The layers of a type the model has **already read**.
1105///
1106/// Takes a `&TypeRef`, not a `&syn::Type`, and that is the whole point: a caller
1107/// must hold a reading, and the ways to hold one are to take it off an element or
1108/// to be the scan admitting a type with no element. Re-deriving a reading from
1109/// `spell()` — the round trip this signature makes impossible — is reasoning
1110/// from the spelling, which is what `origin` is not for.
1111fn peel(ty: &prebindgen_flat::flat::TypeRef) -> Layered {
1112    let (shape, layered) = ty.layer_stack();
1113    let borrowed = layered.borrow_target();
1114    Layered {
1115        shape,
1116        layer_types: ty.layer_types().into_iter().cloned().collect(),
1117        core: borrowed.unwrap_or(layered).key(),
1118        by_ref: borrowed.is_some(),
1119    }
1120}

1122/// Just the borrow: whether `ty` is one, and what it borrows.
1123///
1124/// Deliberately **not** [`peel`]: a site that peels only the borrow means it,
1125/// because the layer underneath is the thing it is about to classify. `Vec<T>`
1126/// answers `(false, Vec<T>)` here and `(iterable, T)` there, and confusing the two
1127/// turns "this arg is a collection" into "this arg is a T".
1128fn peel_borrow(ty: &prebindgen_flat::flat::TypeRef) -> (bool, &prebindgen_flat::flat::TypeRef) {
1129    match ty.borrow_target() {
1130        Some(inner) => (true, inner),
1131        None => (false, ty),
1132    }
1133}

1135/// True when `ret` is `T` / `&T` / `Option<T|&T>` / `Vec<T|&T>` with
1136/// `T == key` — the default-output match. `Result<_, _>` is NOT peeled, so a
1137/// fallible factory (`-> Result<T, E>`) keeps its handle return; the error
1138/// position is matched separately on `E`.
1139fn returns_type(ret: &prebindgen_flat::flat::TypeRef, key: &TypeKey) -> bool {
1140    peel(ret).core == *key

1143/// Build one output/error plan for `ed` and store it in the right registry map.
1144fn process_decl(
1145    registry: &mut Unfolding<'_>,
1146    acc: &Deconstructors,
1147    ed: &OutputDecl,
1148) -> Result<(), UnfoldError> {
1149    {
1150        // The value to decompose: the success return (`Output`) or the
1151        // `Result<_, E>` domain error `E` (`Error`).
1152        let fn_ret = registry
1153            .flat()
1154            .function(&ed.func)
1155            .map(|f| f.ret.clone())
1156            .ok_or_else(|| UnfoldError::UnknownFunction(ed.func.clone()))?;
1157        let ret_ty = match ed.target {
1158            DeconTarget::Output => fn_ret,
1159            DeconTarget::Error => {
1160                fn_ret
1161                    .fallible_parts()
1162                    .map(|(_, e)| e.clone())
1163                    .ok_or_else(|| UnfoldError::Unsupported {
1164                        func: ed.func.clone(),
1165                        reason: "convert_error/deconstruct_error on a non-Result return",
1166                    })?
1167            }
1168        };
1169
1170        // Peel an outer `Option` off the success return BEFORE probing for a
1171        // `Vec`, so `Option<Vec<T>>` composes as `Optional(Iterable)` — the
1172        // fold is skipped and a null result delivered for `None` (issue
1173        // #105). The scalar arm below reuses this peel. Error targets keep
1174        // the historical probe order (the `Vec` probe runs on `E` itself), so
1175        // an `Option<Vec<E>>` error stays whole.
1176        let (optional, after_opt) = match ed.target {
1177            DeconTarget::Output => match ret_ty.optional_inner() {
1178                Some(inner) => (true, inner),
1179                None => (false, &ret_ty),
1180            },
1181            DeconTarget::Error => (false, &ret_ty),
1182        };
1183        // `Vec<T>` / `Option<Vec<T>>` return → `Iterable` (± an `Optional`
1184        // layer). Two element-delivery modes:
1185        //   * **decomposed** (M5): the element type has an accessor →
1186        //     flatten it into leaves, fold `(acc, leaf0, …) -> acc`.
1187        //   * **whole** (M4): no accessor → deliver each element whole
1188        //     via its own output converter + projection, fold `(acc, T) -> acc`.
1189        // The other shapes (`Option`/scalar) decompose via an accessor
1190        // (M1–M3). `Vec<Option<…>>` is not supported.
1191        let plan = if let Some(inner) = after_opt.sequence_elem() {
1192            if inner.optional_inner().is_some() {
1193                return Err(UnfoldError::Unsupported {
1194                    func: ed.func.clone(),
1195                    reason: "Vec<Option<…>> returns",
1196                });
1197            }
1198            let iterable = UnfoldShape::Iterable(Box::new(UnfoldShape::Base));
1199            let shape = if optional {
1200                UnfoldShape::Optional((), Box::new(iterable))
1201            } else {
1202                iterable
1203            };
1204            // The fold delivers the return element-by-element, so the
1205            // whole-collection converter is not needed — and for an
1206            // opaque-handle element it cannot resolve at all (a `jlong` wire
1207            // isn't JObject-shaped). De-require the scan-time registrations
1208            // (the declared return and, under `Option`, the inner `Vec` its
1209            // recursive registration also required) — same reasoning as
1210            // [`apply_leaf_vec_folds`] for the fixed folds.
1211            if ed.target == DeconTarget::Output {
1212                registry.unrequire_output(&ret_ty);
1213                if optional {
1214                    registry.unrequire_output(after_opt);
1215                }
1216            }
1217            // Element type peeled of a leading `&` (accessors take `&Element`).
1218            let (by_ref, element) = peel_borrow(inner);
1219            let ekey = element.key();
1220            if let Some(d) = find_deconstructor_by_type(acc, &ekey) {
1221                // Decomposed: reuse the shared flatten (M3 nesting composes).
1222                let records = d.records.clone();
1223                let decon = decl_id(&ekey, d);
1224                register_decon_spec(registry, acc, &decon, &records, element)?;
1225                let plan = build_plan(acc, registry, ed, by_ref, element, shape, &records, decon)?;
1226                for leaf in &plan.leaves {
1227                    registry.require_output(&leaf.out_ty);
1228                }
1229                plan
1230            } else {
1231                // Whole element: keep the type exactly as written so the
1232                // element's own output converter matches `into_iter()`'s yield.
1233                // No declaration is involved (`decon: None`) — the element
1234                // crosses whole through its own converter.
1235                let by_ref = peel_borrow(inner).0;
1236                registry.require_output(inner);
1237                UnfoldPlan {
1238                    source: inner.clone(),
1239                    decon: None,
1240                    by_ref,
1241                    shape,
1242                    leaves: vec![],
1243                    element: Some(inner.clone()),
1244                    delivery: ed.delivery,
1245                    convert_out_ty: None,
1246                    fixed_builder: false,
1247                    hoists: Vec::new(),
1248                }
1249            }
1250        } else {
1251            // Scalar/decomposed arm. The `Option` peel already happened above
1252            // for `Output` (exactly one layer — `Option<Option<…>>` is NOT
1253            // re-peeled and fails as "no deconstructor" for the inner
1254            // `Option`); for `Error` it happens here, unchanged.
1255            let (optional, core_ty) = match ed.target {
1256                DeconTarget::Output => (optional, after_opt),
1257                DeconTarget::Error => match after_opt.optional_inner() {
1258                    Some(inner) => (true, inner),
1259                    None => (false, after_opt),
1260                },
1261            };
1262            let (by_ref, source) = peel_borrow(core_ty);
1263            let source_key = source.key();
1264            let shape = if optional {
1265                UnfoldShape::Optional((), Box::new(UnfoldShape::Base))
1266            } else {
1267                UnfoldShape::Base
1268            };
1269            let (records, decon) = resolve_deconstructor(acc, &source_key, ed)?;
1270            register_decon_spec(registry, acc, &decon, &records, source)?;
1271            let plan = build_plan(acc, registry, ed, by_ref, source, shape, &records, decon)?;
1272            for leaf in &plan.leaves {
1273                registry.require_output(&leaf.out_ty);
1274            }
1275            plan
1276        };
1277        // Delivery is by **leaf count**, not a per-decl flag:
1278        //   * Output, single non-nullable leaf, non-Iterable ⇒ Return (wrapper
1279        //     returns the value via its ordinary output converter —
1280        //     `convert_out_ty`).
1281        //   * Output, multiple leaves or Iterable (at any layer — an
1282        //     `Optional(Iterable)` fold has no single value to return) ⇒
1283        //     Callback (builder / fold).
1284        //   * Error ⇒ always Callback-shaped: every leaf is a `ze` arg after the
1285        //     fixed `je` (no return-value path; `convert_out_ty` stays None).
1286        //
1287        // A NULLABLE leaf is one whose path passes through an `Option` that
1288        // something is decomposed below (`Option<Handle>` reached by
1289        // `.field_self()`, a nested value form behind an `Option`). Returning it
1290        // has nowhere to put the absent case: a return value is one expression,
1291        // so there is no `None` arm, and `convert_out_ty` names the leaf's own
1292        // type rather than an optional of it. Callback delivery has that arm
1293        // already — the leaf crosses as a boxed `Long` / JVM null — so the
1294        // shape goes there instead of being composed into Rust that hands
1295        // `&Option<T>` to a converter typed for `T`.
1296        let single_return = ed.target == DeconTarget::Output
1297            && !plan.shape.has_iterable_layer()
1298            && plan.leaves.len() == 1
1299            && !plan.leaves[0].nullable;
1300        let plan = if single_return {
1301            // Composed with the model's own layering rather than by spelling
1302            // `Option<#leaf_ty>` and handing the tokens over: `optional()` pairs
1303            // the `kind` with its spelling in one place, so the reading that
1304            // reaches the table is the one this plan carries (#281).
1305            let cv = if matches!(plan.shape, UnfoldShape::Optional((), _)) {
1306                plan.leaves[0].out_ty.optional()
1307            } else {
1308                plan.leaves[0].out_ty.clone()
1309            };
1310            registry.require_output(&cv);
1311            UnfoldPlan {
1312                delivery: Delivery::Return,
1313                convert_out_ty: Some(cv.clone()),
1314                ..plan
1315            }
1316        } else {
1317            UnfoldPlan {
1318                delivery: Delivery::Callback,
1319                ..plan
1320            }
1321        };
1322        match ed.target {
1323            DeconTarget::Output => registry.plans.unfold_plans.insert(ed.func.clone(), plan),
1324            DeconTarget::Error => registry.plans.error_plans.insert(ed.func.clone(), plan),
1325        };
1326    }
1327    Ok(())
1328}

1335/// Register the declaration-default [`DeconSpec`] for `decon` (no-op when
1336/// already present): re-flatten the records with normalized inputs —
1337/// borrowed identity, no outer shape — so the stored spec is independent of
1338/// the using function's return shape and of processing order.
1339fn register_decon_spec(
1340    registry: &mut Unfolding<'_>,
1341    acc: &Deconstructors,
1342    decon: &DeconId,
1343    records: &[DeconRecord],
1344    source: &prebindgen_flat::flat::TypeRef,
1345) -> Result<(), UnfoldError> {
1346    if registry.plans.decon_plans.contains_key(decon) {
1347        return Ok(());
1348    }
1349    let mut leaves: Vec<UnfoldLeaf> = Vec::new();
1350    let mut visited: HashSet<TypeKey> = HashSet::new();
1351    visited.insert(source.key());
1352    flatten(
1353        acc,
1354        registry,
1355        records,
1356        source,
1357        &[],
1358        &[],
1359        true,
1360        false,
1361        &mut visited,
1362        &mut leaves,
1363        // A `DeconSpec` describes the leaf list only — signature artifacts are
1364        // derived from it, never emitted code — so its hoists are discarded.
1365        &mut Vec::new(),
1366    )?;
1367    require_unique_leaf_names(source, &leaves)?;
1368    registry.plans.decon_plans.insert(
1369        decon.clone(),
1370        DeconSpec {
1371            source: source.clone(),
1372            leaves,
1373        },
1374    );
1375    Ok(())
1376}

1409/// Build the [`UnfoldPlan`] for a chosen accessor. `shape` is the outer
1410/// shape over the core decomposition (`Decompose` for `T`/`&T`,
1411/// `Optional(Decompose)` for `Option<T>`/`Option<&T>`). The records are
1412/// recursively flattened ([`flatten`]) — nested accessors contribute
1413/// their leaves with the access path prefixed.
1414#[allow(clippy::too_many_arguments)]
1415fn build_plan(
1416    acc: &Deconstructors,
1417    registry: &Unfolding<'_>,
1418    ed: &OutputDecl,
1419    by_ref: bool,
1420    source: &prebindgen_flat::flat::TypeRef,
1421    shape: UnfoldShape,
1422    records: &[DeconRecord],
1423    decon: DeconId,
1424) -> Result<UnfoldPlan, UnfoldError> {
1425    let mut leaves: Vec<UnfoldLeaf> = Vec::new();
1426    let mut visited: HashSet<TypeKey> = HashSet::new();
1427    visited.insert(source.key());
1428    let mut hoists: Vec<Hoist> = Vec::new();
1429    flatten(
1430        acc,
1431        registry,
1432        records,
1433        source,
1434        &[],
1435        &[],
1436        by_ref,
1437        false,
1438        &mut visited,
1439        &mut leaves,
1440        &mut hoists,
1441    )?;
1442    require_unique_leaf_names(source, &leaves)?;
1443    require_root_identity_last(by_ref, source, &leaves)?;
1444
1445    Ok(UnfoldPlan {
1446        source: source.clone(),
1447        decon: Some(decon),
1448        by_ref,
1449        shape,
1450        leaves,
1451        element: None,
1452        delivery: ed.delivery,
1453        convert_out_ty: None,
1454        fixed_builder: false,
1455        hoists,
1456    })
1457}

1459/// Error when an **owned** decomposition emits the root identity leaf before a
1460/// nested identity leaf. Leaves are emitted in declaration order, and the root
1461/// identity MOVES the owned value while a nested identity clones from a borrow
1462/// of it — the wrong order generates non-compiling Rust ("use of moved value")
1463/// with a cryptic rustc message. Caught here instead, with the fix in the
1464/// error: declare `.field_self()` after the nested-identity fields. (Borrowed
1465/// decompositions clone the root identity, so any order is fine.)
1466fn require_root_identity_last(
1467    by_ref: bool,
1468    source: &prebindgen_flat::flat::TypeRef,
1469    leaves: &[UnfoldLeaf],
1470) -> Result<(), UnfoldError> {
1471    if by_ref {
1472        return Ok(());
1473    }
1474    let root_at = leaves.iter().position(|l| l.identity && l.path.is_empty());
1475    let last_nested_at = leaves
1476        .iter()
1477        .rposition(|l| l.identity && !l.path.is_empty());
1478    if let (Some(root), Some(nested)) = (root_at, last_nested_at) {
1479        if root < nested {
1480            return Err(UnfoldError::RootIdentityBeforeNested {
1481                target: source.key().to_string(),
1482            });
1483        }
1484    }
1485    Ok(())
1486}

1488/// Recursively flatten an accessor's records into [`UnfoldLeaf`]s.
1489///
1490/// * `source` — the type whose accessor `records` belong to (the root
1491///   on the first call, a nested child type on recursion).
1492/// * `path_prefix` — accessor chain from the root value to `source` (empty at
1493///   the root; `[…, nesting_accessor]` when recursing into a nested child).
1494/// * `by_ref` — the top-level return/element borrow-ness. The identity leaf is
1495///   **owned** (`source`) only at the root of an owned value (`path_prefix`
1496///   empty && `!by_ref`) — a `Copy` value delivers itself by copy and an
1497///   opaque handle moves; everywhere else it is **borrowed** (`&source`,
1498///   cloned).
1499/// * `nullable` — `true` once any nesting accessor on the path returned
1500///   `Option` (the reached value may be absent ⇒ the leaf is `null`).
1501/// * `visited` — type keys on the current nesting chain (cycle guard; entries
1502///   are removed after each nested recursion so sibling records may reuse a type).
1503#[allow(clippy::too_many_arguments)]
1504fn flatten(
1505    acc: &Deconstructors,
1506    registry: &Unfolding<'_>,
1507    records: &[DeconRecord],
1508    source: &prebindgen_flat::flat::TypeRef,
1509    path_prefix: &[PathStep],
1510    name_prefix: &[String],
1511    by_ref: bool,
1512    nullable: bool,
1513    visited: &mut HashSet<TypeKey>,
1514    leaves: &mut Vec<UnfoldLeaf>,
1515    hoists: &mut Vec<Hoist>,
1516) -> Result<(), UnfoldError> {
1517    let source_key = source.key();
1518    // The author-supplied (literal) leaf-name segment at this level, appended
1519    // to the inherited chain prefix. Segments are joined with `"__"`.
1520    let seg_name = |name: &str| -> Vec<String> {
1521        let mut v = name_prefix.to_vec();
1522        v.push(name.to_string());
1523        v
1524    };
1525    // Identity uniqueness is per accessor (one move/clone of the value
1526    // at this level); nested levels each get their own identity budget.
1527    let mut seen_identity = false;
1528
1529    for rec in records {
1530        match rec {
1531            DeconRecord::Identity => {
1532                if seen_identity {
1533                    return Err(UnfoldError::MultipleIdentity {
1534                        target: source_key.to_string(),
1535                    });
1536                }
1537                seen_identity = true;
1538                // Owned where the value is OURS to give: the root of an owned
1539                // plan (a `Copy` blob copies / an opaque handle moves), or a
1540                // field of a value form that CONSUMED its value — that form was
1541                // handed the value, so its fields move out like every other
1542                // field of it. Borrowed (clone) otherwise. The adapter-side type
1543                // + projection come from this `out_ty`'s output converter, so
1544                // this is what decides whether the leaf is boxed by move or
1545                // cloned through the borrowed-opaque one.
1546                // A plan field: the drop to spelling happens here, where the value
1547                // is stored for emission, and the borrowed form is composed rather
1548                // than looked up because no source wrote it.
1549                // The borrowed form is COMPOSED — no source wrote it — and the
1550                // composition pairs the kind with its own spelling, so nothing
1551                // downstream has to look either up.
1552                let out_ty = if place_is_owned(hoists, path_prefix, by_ref) {
1553                    source.clone()
1554                } else {
1555                    source.borrowed()
1556                };
1557                leaves.push(UnfoldLeaf {
1558                    name: if path_prefix.is_empty() {
1559                        "handle".to_string()
1560                    } else {
1561                        name_prefix.join("__")
1562                    },
1563                    path: path_prefix.to_vec(),
1564                    out_ty,
1565                    identity: true,
1566                    nullable,
1567                    source: LeafSource::Reach,
1568                    groups: Vec::new(),
1569                });
1570            }
1571            DeconRecord::Fields {
1572                func,
1573                consuming,
1574                fields,
1575            } => {
1576                let consuming = *consuming;
1577                // The value form is called once; every field hangs off that one
1578                // call, so the whole record shares a single `Call` step and the
1579                // emitter can hoist it.
1580                accessor_signature(registry, func, &source.key())?;
1581                // The declarator states whether the value is given away; the
1582                // signature has to agree, or the emitted call would not compile
1583                // in the consumer's crate. Checked rather than inferred so that
1584                // declaring `.fields_self_into(..)` on a borrowing accessor is a
1585                // named error instead of a silently downgraded boundary.
1586                if consuming != accessor_consumes(registry, func) {
1587                    return Err(UnfoldError::Unsupported {
1588                        func: func.clone(),
1589                        reason: if consuming {
1590                            "declared as a CONSUMING value form (`.fields_self_into(..)`) but the \
1591                             accessor borrows its receiver — declare it with `.fields(..)`, or \
1592                             name the by-value accessor"
1593                        } else {
1594                            "declared as a BORROWING value form (`.fields(..)`) but the accessor \
1595                             takes its receiver by value — declare it with `.fields_self_into(..)`, or \
1596                             name the `&Self` accessor"
1597                        },
1598                    });
1599                }
1600                let mut root_path = path_prefix.to_vec();
1601                root_path.push(PathStep::call(func.clone(), false, false));
1602                // A hoist below an optional step is CONDITIONAL: it binds an
1603                // `Option<TStruct>` local (built only in the `Some` arm) and
1604                // every leaf under it is null when the value is absent — which
1605                // is the nullability `flatten` already propagates down here.
1606                // What it cannot do is nest: composing a second hoist off a
1607                // conditional one would have to reach through the outer
1608                // `Option`, and the binder has no arm to put that in. One level
1609                // is the shape real bindings need (`Option<&Sample>` delivering
1610                // a sample's value form), so implement that and name the rest.
1611                //
1612                // A top-level `Option<T>` is represented by
1613                // `UnfoldShape::Optional`, not by a path step, and is unaffected.
1614                if root_path.iter().any(PathStep::is_optional)
1615                    && hoists.iter().any(|h| {
1616                        h.prefix.len() < root_path.len() && root_path.starts_with(&h.prefix)
1617                    })
1618                {
1619                    return Err(UnfoldError::Unsupported {
1620                        func: func.clone(),
1621                        reason: "a value form nested under another one that is reached through \
1622                                 `Option` — conditional hoists do not nest",
1623                    });
1624                }
1625                // A consuming value form DESTROYS the value into its parts, so
1626                // a sibling record — `.field_self()` or another `.field()` —
1627                // would read what it just gave away. jnigen refuses this in the
1628                // declarator, where the author can see it; this is the backstop
1629                // for records built directly against core.
1630                //
1631                // Being reached through ANOTHER value form is fine: a hoisted
1632                // value form is an owned struct and its fields are disjoint, so
1633                // the parent's field is handed over by move.
1634                if consuming && records.len() > 1 {
1635                    return Err(UnfoldError::Unsupported {
1636                        func: func.clone(),
1637                        reason: "a consuming value form must be the only record of its \
1638                                 declaration — it moves the value, so `.field_self()` or \
1639                                 a sibling `.field()` would read a moved value",
1640                    });
1641                }
1642                // Evaluate this value form ONCE. Recorded at the prefix it sits
1643                // at rather than as a lone accessor, so a nested value form
1644                // (this record reached through another one's field) gets its own
1645                // hoist instead of being rebuilt per child leaf. `path_prefix`
1646                // grows as `flatten` descends, so the list comes out
1647                // outermost-first.
1648                hoists.push(Hoist {
1649                    prefix: root_path.clone(),
1650                    consuming,
1651                });
1652
1653                for fr in fields {
1654                    // The declaration carries the field's reading, so nothing is
1655                    // looked up and nothing is re-classified.
1656                    //
1657                    // A field's own `Option` makes everything under it nullable,
1658                    // exactly as an `Option`-returning accessor step does.
1659                    let (opt, core) = match fr.ty.optional_inner() {
1660                        Some(inner) => (true, inner),
1661                        None => (false, &fr.ty),
1662                    };
1663                    let child_ty = core.borrow_target().unwrap_or(core);
1664                    let child_key = child_ty.key();
1665
1666                    // Same three-way choice a `.field()` record makes: declared
1667                    // override, else the field type's own deconstructor, else
1668                    // one leaf — with the adapter able to pre-build the leaves
1669                    // for a shape only it can describe.
1670                    let child_records = match &fr.decon {
1671                        FieldDecon::Records(recs) => Some(recs.clone()),
1672                        FieldDecon::Leaves(_) => None,
1673                        FieldDecon::Default => match find_deconstructor_by_type(acc, &child_key) {
1674                            Some(child_decl) if !visited.contains(&child_key) => {
1675                                Some(child_decl.records.clone())
1676                            }
1677                            Some(_) => {
1678                                return Err(UnfoldError::Cycle {
1679                                    target: child_key.to_string(),
1680                                });
1681                            }
1682                            None => None,
1683                        },
1684                    };
1685                    let decomposed =
1686                        child_records.is_some() || matches!(fr.decon, FieldDecon::Leaves(_));
1687
1688                    // The field's own `Option` is a nullable NESTING step only
1689                    // when something is decomposed below it. For a plain leaf
1690                    // the whole `Option<F>` is what the converter takes — the
1691                    // same rule that makes a terminal accessor's `Option` ride
1692                    // its converter instead of being unwrapped.
1693                    let mut field_path = root_path.clone();
1694                    let (last, lead) = fr
1695                        .members
1696                        .split_last()
1697                        .expect("a field record addresses at least one member");
1698                    // Only the LAST member can be optional — an inlined nested
1699                    // class is reached directly, never through an `Option`.
1700                    field_path.extend(lead.iter().map(|m| PathStep::field(m.clone(), false)));
1701                    field_path.push(PathStep::field(last.clone(), opt && decomposed));
1702
1703                    // Adapter-built leaves: rebase each onto this field's path
1704                    // and name. Their internal structure (a selector plus its
1705                    // groups) is opaque here and passes through untouched.
1706                    if let FieldDecon::Leaves(built) = &fr.decon {
1707                        for l in built {
1708                            let mut path = field_path.clone();
1709                            path.extend(l.path.iter().cloned());
1710                            let mut name = seg_name(&fr.name);
1711                            name.push(l.name.clone());
1712                            leaves.push(UnfoldLeaf {
1713                                name: name.join("__"),
1714                                path,
1715                                nullable: l.nullable || nullable || opt,
1716                                ..l.clone()
1717                            });
1718                        }
1719                        continue;
1720                    }
1721
1722                    if let Some(child_records) = child_records {
1723                        visited.insert(child_key.clone());
1724                        flatten(
1725                            acc,
1726                            registry,
1727                            &child_records,
1728                            // The declaration's reading, peeled — not a new one.
1729                            child_ty,
1730                            &field_path,
1731                            &seg_name(&fr.name),
1732                            by_ref,
1733                            nullable || opt,
1734                            visited,
1735                            leaves,
1736                            hoists,
1737                        )?;
1738                        visited.remove(&child_key);
1739                    } else {
1740                        // A plain field leaf: the value is CLONED out of the
1741                        // struct, so its converter takes the owned field type as
1742                        // written — `Option` and all, which is why a terminal
1743                        // `Option` step is not a nesting step for it.
1744                        leaves.push(UnfoldLeaf {
1745                            name: seg_name(&fr.name).join("__"),
1746                            path: field_path,
1747                            out_ty: fr.ty.clone(),
1748                            identity: false,
1749                            nullable,
1750                            source: LeafSource::Reach,
1751                            groups: Vec::new(),
1752                        });
1753                    }
1754                }
1755            }
1756            DeconRecord::Acc { name, .. } | DeconRecord::LocalAcc { name, .. } => {
1757                // A binding-local record resolves through its synthesized
1758                // registry entry (see `synthesize_local_accessors`), so both
1759                // kinds read one signature source; only the cycle rule below
1760                // differs.
1761                let (func, local) = match rec {
1762                    DeconRecord::Acc { func, .. } => (func.clone(), false),
1763                    DeconRecord::LocalAcc { path, .. } => (DeconRecord::local_ident(path), true),
1764                    DeconRecord::Identity | DeconRecord::Fields { .. } => unreachable!(),
1765                };
1766                let ret = accessor_signature(registry, &func, &source.key())?;
1767                // Default unwrap: if the return type has its own deconstructor,
1768                // splice it (recurse); otherwise the return is one leaf. Peel an
1769                // `Option` (value may be absent) + leading `&` to reach the child.
1770                // This site peels an `Option` only — an accessor returning a run
1771                // of values is not spliced — so it asks the model for that one
1772                // layer rather than the whole stack.
1773                let after_opt = ret.optional_inner();
1774                let opt = after_opt.is_some();
1775                let core = after_opt.unwrap_or(&ret);
1776                let (core_by_ref, child_ty) = peel_borrow(core);
1777                let child_key = child_ty.key();
1778                // A child already on the nesting chain: for a `#[prebindgen]`
1779                // accessor that is an authoring cycle (hard error); a
1780                // binding-local field re-delivering (part of) its own type
1781                // under a binding-defined condition is the POINT — degrade to
1782                // a plain converter leaf instead of splicing.
1783                let splice = match find_deconstructor_by_type(acc, &child_key) {
1784                    Some(child_decl) if !visited.contains(&child_key) => Some(child_decl),
1785                    Some(_) if local => None,
1786                    Some(_) => {
1787                        return Err(UnfoldError::Cycle {
1788                            target: child_key.to_string(),
1789                        });
1790                    }
1791                    None => None,
1792                };
1793                if let Some(child_decl) = splice {
1794                    visited.insert(child_key.clone());
1795                    let child_records = child_decl.records.clone();
1796                    let mut child_path = path_prefix.to_vec();
1797                    child_path.push(PathStep::call(func.clone(), opt, !core_by_ref));
1798                    flatten(
1799                        acc,
1800                        registry,
1801                        &child_records,
1802                        child_ty,
1803                        &child_path,
1804                        &seg_name(name),
1805                        by_ref,
1806                        nullable || opt,
1807                        visited,
1808                        leaves,
1809                        hoists,
1810                    )?;
1811                    visited.remove(&child_key);
1812                } else {
1813                    // Leaf: the return value as written (`&str`, enum, `i64`, …).
1814                    // One exception: a binding-local field returning an
1815                    // OPTIONAL BORROW (`Option<&T>`) is the conditional
1816                    // HANDLE-delivery idiom — structurally a spliced identity
1817                    // behind an `Option`-returning step (cf. an
1818                    // `Option`-returning nesting accessor + the child's
1819                    // `field_self`), so it contributes a nullable IDENTITY
1820                    // leaf of the borrowed type: the reach path unwraps the
1821                    // final `Option` (the synthesized signature keeps the
1822                    // full return) and the value clones through its handle
1823                    // projection, `None` delivering null. It shares the
1824                    // one-identity-per-deconstructor budget with
1825                    // `.field_self()` — two handle deliveries of one value
1826                    // make no sense.
1827                    let cond_handle = local && opt && core_by_ref;
1828                    if cond_handle {
1829                        if seen_identity {
1830                            return Err(UnfoldError::MultipleIdentity {
1831                                target: source_key.to_string(),
1832                            });
1833                        }
1834                        seen_identity = true;
1835                    }
1836                    // A plan field: the spelling is taken here, once, where the
1837                    // leaf is stored for emission.
1838                    let (out_ty, nullable, identity) = if cond_handle {
1839                        (core.clone(), true, true)
1840                    } else {
1841                        (ret.clone(), nullable, false)
1842                    };
1843                    let mut path = path_prefix.to_vec();
1844                    path.push(PathStep::call(func.clone(), opt, !core_by_ref));
1845                    leaves.push(UnfoldLeaf {
1846                        name: seg_name(name).join("__"),
1847                        path,
1848                        out_ty,
1849                        identity,
1850                        nullable,
1851                        source: LeafSource::Reach,
1852                        groups: Vec::new(),
1853                    });
1854                }
1855            }
1856        }
1857    }
1858
1859    Ok(())
1860}

1862/// Error if two leaves of one flattened deconstructor share a name. Author leaf
1863/// names are explicit and emitted literally, so a collision is a declaration
1864/// bug — never auto-resolved.
1865fn require_unique_leaf_names(
1866    source: &prebindgen_flat::flat::TypeRef,
1867    leaves: &[UnfoldLeaf],
1868) -> Result<(), UnfoldError> {
1869    let mut seen: HashSet<&str> = HashSet::new();
1870    for l in leaves {
1871        if !seen.insert(l.name.as_str()) {
1872            return Err(UnfoldError::DuplicateLeafName {
1873                target: source.key().to_string(),
1874                name: l.name.clone(),
1875            });
1876        }
1877    }
1878    Ok(())
1879}

1881/// Make a signature's name list unique: a duplicate gets a numeric suffix
1882/// (`name2`, `name3`, …). Adapters run this over the final per-signature
1883/// list (after their own casing), since one signature may concatenate the
1884/// leaves of several plans.
1885pub fn dedup_names(names: &mut [String]) {
1886    let mut seen: HashSet<String> = HashSet::new();
1887    for n in names.iter_mut() {
1888        if !seen.insert(n.clone()) {
1889            let mut k = 2;
1890            while !seen.insert(format!("{n}{k}")) {
1891                k += 1;
1892            }
1893            *n = format!("{n}{k}");
1894        }
1895    }
1896}

1898/// An accessor `f(&T) -> R`: returns its return type `R` as written (a
1899/// reference where possible).
1900///
1901/// `expected` is the type the deconstructor decomposes, and what comes back is
1902/// an accessor already proven to take it. Taking it as a parameter rather than
1903/// leaving the caller to check afterwards is the point: a declarator cannot
1904/// reach an accessor's signature without saying what that accessor is supposed
1905/// to be about, so the check cannot be the thing a new declarator forgets
1906/// (#223). The comparison is [`check_declared_target`], shared with the input
1907/// side's constructor lookup.
1908fn accessor_signature(
1909    registry: &Unfolding<'_>,
1910    func: &syn::Ident,
1911    expected: &TypeKey,
1912) -> Result<prebindgen_flat::flat::TypeRef, UnfoldError> {
1913    let f = registry
1914        .flat()
1915        .function(&func)
1916        .ok_or_else(|| UnfoldError::UnknownAccessor(func.clone()))?;
1917
1918    // First parameter is the receiver `&T`; peel the borrow to get `T`.
1919    // `borrow_target` is the model's own answer, so the peel reads a
1920    // classification instead of re-deciding it from `syn::Type::Reference`.
1921    let first = f
1922        .params
1923        .first()
1924        .ok_or_else(|| UnfoldError::UnknownAccessor(func.clone()))?;
1925    // The receiver's identity, keyed so the comparison below cannot fail on a
1926    // spelling difference that does not change which type this is about.
1927    let takes = match first.ty.borrow_target() {
1928        Some(inner) => inner.key(),
1929        None => first.ty.key(),
1930    };
1931    check_declared_target(func, &takes, expected)?;
1932    Ok(f.ret.clone())
1933}

1935/// Whether the value sitting at `path_prefix` is the plan's **to give away**:
1936/// the root of an owned plan, or a field of a value form that consumed its
1937/// value and is reached by a movable run of field steps.
1938///
1939/// Consulted where a leaf's `out_ty` is chosen, so the ownership decision is
1940/// made ONCE, in the plan, rather than re-derived by each emitter — a leaf
1941/// whose `out_ty` is the owned type is boxed by move, one whose `out_ty` is a
1942/// borrow is cloned through the borrowed-opaque converter.
1943fn place_is_owned(hoists: &[Hoist], path_prefix: &[PathStep], by_ref: bool) -> bool {
1944    if path_prefix.is_empty() {
1945        return !by_ref;
1946    }
1947    hoists
1948        .iter()
1949        .filter(|h| h.prefix.len() <= path_prefix.len() && path_prefix.starts_with(&h.prefix))
1950        .max_by_key(|h| h.prefix.len())
1951        .is_some_and(|h| h.consuming && steps_are_movable(&path_prefix[h.prefix.len()..]))
1952}

1954/// Whether an accessor takes its receiver **by value** — a *consuming* value
1955/// form, which destroys the object into its parts instead of cloning them out
1956/// of a borrow.
1957///
1958/// Asked separately because [`accessor_signature`] peels the `&` in order to
1959/// compare target types, so `f(v: T)` and `f(v: &T)` are indistinguishable
1960/// there by design.
1961fn accessor_consumes(registry: &Unfolding<'_>, func: &syn::Ident) -> bool {
1962    registry
1963        .flat()
1964        .function(&func)
1965        .and_then(|f| f.params.first())
1966        .is_some_and(|p| p.ty.borrow_target().is_none())
1967}
