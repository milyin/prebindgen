9    prebindgen_reprc::schema_id(&format!("jni-allocation:{}", names::type_identity(ty))) as i64
10}
11pub(crate) fn access_key(access: Access) -> u8 {
12    match access {
13        Access::Shared => 0,
14        Access::Exclusive => 1,
15        Access::Move => 2,
16    }
17}
18pub(crate) fn key(handle: &BoxedWire) -> (String, u8) {
19    (type_key(&handle.ty), access_key(handle.access))
20}
21/// Local groups in one prepared carrier. The same order is used by its writer,
22/// patch walk, and the enclosing Kotlin call; no runtime type lookup is needed.
23pub(crate) fn kinds(node: &JniNode) -> Vec<BoxedWire> {
24    let mut kinds = BTreeMap::new();
25    for wire in prebindgen_flat::wire_definitions(std::iter::once(node)) {
26        if let JniWire::Handle { handle, .. } = &wire.kind {
27            kinds
28                .entry(key(handle))
29                .or_insert_with(|| (**handle).clone());
30        }
31    }
32    kinds.into_values().collect()
33}
34pub(crate) fn carrier_inputs(node: &JniNode, state: &str) -> Vec<KotlinHandleInput> {
35    let kinds = kinds(node);
36    let count = kinds.len();
37    kinds
38        .into_iter()
39        .enumerate()
40        .map(|(group, handle)| KotlinHandleInput {
41            ty: handle.ty,
42            access: handle.access,
43            many: true,
44            argument: None,
45            consumed: None,
46            value: if count == 1 {
47                format!("{state}?.handles.orEmpty()")
48            } else {
49                format!("{state}?.handleGroup({group}).orEmpty()")
50            },
51        })
52        .collect()
53}
54fn controlled_child<'a>(node: &'a JniNode, name: &syn::Ident) -> Option<&'a JniNode> {
55    match node {
56        WireNode::Chain {
57            controls, child, ..
58        } => {
59            if controls.iter().any(|w| w.name == *name) {
60                Some(child)
61            } else if matches!(
62                node,
63                WireNode::Chain {
64                    meta: JniChain::Sequence(_) | JniChain::Bulk(_),
65                    ..
66                }
67            ) {
68                None // Child rows are a different name scope.
69            } else {
70                controlled_child(child, name)
71            }
72        }
73        WireNode::Branch { children, .. } => {
74            children.iter().find_map(|n| controlled_child(n, name))
75        }
76        _ => None,
77    }
78}
79pub(crate) fn column_groups(node: &JniNode) -> Result<String> {
80    let groups = kinds(node);
81    if groups.len() <= 1 {
82        return Ok("null".into());
83    }
84    let columns = crate::group::row_fields(node)
85        .map(|wire| {
86            let local = if let JniWire::Handle { handle, .. } = &wire.kind {
87                vec![(**handle).clone()]
88            } else if matches!(wire.kind, JniWire::Rows { .. } | JniWire::BulkBuffer { .. }) {
89                let child = controlled_child(node, &wire.name).ok_or_else(|| {
90                    format!(
91                        "missing prepared child for handle-group column `{}`",
92                        wire.name
93                    )
94                })?;
95                kinds(child)
96            } else {
97                vec![]
98            };
99            let indices = local
100                .iter()
101                .map(|kind| {
102                    groups
103                        .iter()
104                        .position(|g| key(g) == key(kind))
105                        .map(|index| index.to_string())
106                        .ok_or_else(|| {
107                            format!("missing parent handle group for `{}`", type_key(&kind.ty))
108                        })
109                })
110                .collect::<Result<Vec<_>>>()?
111                .join(", ");
112            Ok(format!("intArrayOf({indices})"))
113        })
114        .collect::<Result<Vec<_>>>()?
115        .join(", ");
116    Ok(format!("arrayOf({columns})"))
117}
118/// Every native resource must be represented by prepared Kotlin handle wires.
119/// Custom conversions without this information cannot use the generated JNI API.
120pub(crate) fn require_preflight<I: KotlinInputConversion>(input: &I) -> Result<()> {
121    fn prepared(node: &JniNode) -> bool {
122        match node {
123            WireNode::Chain {
124                meta: JniChain::Bulk(_),
125                ..
126            } => true,
127            WireNode::Chain {
128                meta: JniChain::Sequence(JniSequence::Bulk),
129                ..
130            } => !crate::kotlin::input_handles(node, false),
131            WireNode::Chain { child, .. } => prepared(child),
132            WireNode::Branch { children, .. } => children.iter().all(prepared),
133            WireNode::Leaf(_) => true,
134        }
135    }
136    if !prepared(input.input_node()) {
137        return Err("Kotlin handle preflight requires an enclosing prepared bulk frame".into());
138    }
139    Ok(())
140}

142/// Check the Kotlin emitter's retained sources against the wire shape. Carriers
143/// combine repeated fields into one list per allocation type and access mode.
144fn complete_handle_inputs(node: &JniNode, input: &mut KotlinInput) -> Result<()> {
145    fn sources(node: &JniNode, expected: &mut BTreeMap<(String, u8, bool), usize>) {
146        match node {
147            WireNode::Chain {
148                meta: JniChain::Bulk(_) | JniChain::Sequence(_),
149                child,
150                ..
151            } => {
152                for kind in kinds(child) {
153                    let (ty, access) = key(&kind);
154                    *expected.entry((ty, access, true)).or_default() += 1;
155                }
156            }
157            WireNode::Chain { child, .. } => sources(child, expected),
158            WireNode::Branch { children, .. } => {
159                for child in children {
160                    sources(child, expected);
161                }
162            }
163            WireNode::Leaf(wires) => {
164                for wire in wires {
165                    if let JniWire::Handle { handle, .. } = &wire.kind {
166                        let (ty, access) = key(handle);
167                        *expected.entry((ty, access, false)).or_default() += 1;
168                    }
169                }
170            }
171        }
172    }
173    let mut expected = BTreeMap::new();
174    sources(node, &mut expected);
175    if input.handle_inputs.is_none()
176        && expected.is_empty()
177        && input.locks.is_empty()
178        && input.lock_groups.is_empty()
179    {
180        input.handle_inputs = Some(vec![]);
181    }
182    let handles = input.handle_inputs.as_ref().ok_or(
183        "Kotlin handle preflight requires complete prepared handle inputs from custom conversions",
184    )?;
185    let mut actual = BTreeMap::new();
186    for handle in handles {
187        *actual
188            .entry((type_key(&handle.ty), access_key(handle.access), handle.many))
189            .or_default() += 1usize;
190    }
191    if actual != expected {
192        return Err("Kotlin handle preflight requires retained inputs matching every handle wire's allocation type, access mode and carrier grouping".into());
193    }
194    if handles.is_empty() && (!input.locks.is_empty() || !input.lock_groups.is_empty()) {
195        return Err("prepared handle inputs omit existing Kotlin locks".into());
196    }
197    Ok(())
198}

206/// Private emission state for a complete ordinary call or callback result.
207/// Ownership receipts still walk their original wire order, independently of
208/// these groups, which exist only to validate inputs and acquire locks.
209pub(crate) struct HandleLocks {
210    sources: Vec<Source>,
211    pub(crate) setup: String,
212    gc: bool,
213    prefix: String,
214}
215impl HandleLocks {
216    pub(crate) fn new<'a, I: KotlinInputConversion + 'a>(
217        inputs: impl IntoIterator<Item = (&'a I, &'a mut KotlinInput)>,
218        context: &KotlinContext<'_>,
219    ) -> Result<Self> {
220        let mut out = Self {
221            sources: vec![],
222            setup: String::new(),
223            gc: false,
224            prefix: context.package.into(),
225        };
226        for (native, input) in inputs {
227            complete_handle_inputs(native.input_node(), input)?;
228            for handle in input.handle_inputs.clone().unwrap() {
229                let class = context
230                    .classes
231                    .get(&type_key(&handle.ty))
232                    .ok_or("missing handle input class")?;
233                let local = format!("__pb_handle{}", out.sources.len());
234                let value = &handle.value;
235                let ty = if handle.many {
236                    format!("kotlin.collections.List<{}.NativeHandle>", context.package)
237                } else {
238                    format!("{}.NativeHandle?", context.package)
239                };
240                out.setup.push_str(&format!("val {local}: {ty} = try {{ {value} }} catch (failure: IllegalArgumentException) {{ throw {}.HandleValidationFailure(failure.message ?: \"invalid handle\") }}; ", context.package));
241                if handle.many == handle.argument.is_some() {
242                    return Err(
243                        "retained handles require individual argument indices or carrier lists"
244                            .into(),
245                    );
246                }
247                if let Some(index) = handle.argument {
248                    let argument = input
249                        .arguments
250                        .get_mut(index)
251                        .ok_or("invalid retained-handle argument index")?;
252                    *argument = format!("{local}?.raw ?: 0L");
253                }
254                if let Some(original) = &handle.consumed {
255                    let consume = input
256                        .consumed
257                        .iter_mut()
258                        .find(|code| *code == original)
259                        .ok_or("retained handle ownership update does not match its input")?;
260                    *consume = format!("{local}?.let {{ __pb_consume(it) }}");
261                }
262                out.sources.push(Source {
263                    local,
264                    many: handle.many,
265                    access: handle.access,
266                    ty: class.handle_type,
267                });
268                out.gc |= class.gc_managed;
269            }
270        }
271        Ok(out)
272    }
273    pub(crate) fn wrap(&self, body: &str) -> String {
274        let p = &self.prefix;
275        let gc = if self.gc { "Gc" } else { "" };
276        let mut body = body.to_owned();
277        if self.sources.len() <= 3 && self.sources.iter().all(|s| !s.many) {
278            let mut checks = String::new();
279            for (i, a) in self.sources.iter().enumerate() {
280                for b in self.sources.iter().skip(i + 1) {
281                    if a.ty == b.ty && (a.access != Access::Shared || b.access != Access::Shared) {
282                        let (a, b) = (&a.local, &b.local);
283                        checks.push_str(&format!("if ({a} != null && {b} != null && {a}.address == {b}.address) throw {p}.HandleValidationFailure(\"conflicting input aliases\"); "));
284                    }
285                }
286            }
287            if !self.sources.is_empty() {
288                body = format!(
289                    "{checks}{p}.with{gc}HandleLocks({}) {{ {body} }}",
290                    self.sources
291                        .iter()
292                        .map(|s| s.local.as_str())
293                        .collect::<Vec<_>>()
294                        .join(", ")
295                );
296            }
297        } else {
298            let types = self.sources.iter().map(|s| s.ty).collect::<BTreeSet<_>>();
299            let mut prepare = String::new();
300            let mut ordered = vec![];
301            for (group_index, ty) in types.into_iter().enumerate() {
302                let shared = self.list(ty, true);
303                let exclusive = self.list(ty, false);
304                let shared_present = self
305                    .sources
306                    .iter()
307                    .any(|s| s.ty == ty && s.access == Access::Shared);
308                let exclusive_present = self
309                    .sources
310                    .iter()
311                    .any(|s| s.ty == ty && s.access != Access::Shared);
312                let name = format!("__pb_ordered{group_index}");
313                if ordered.is_empty() && self.sources.iter().all(|s| s.ty == ty) {
314                    // Single-type calls retain their short-list fast paths and
315                    // need neither a group array nor a second-dimensional walk.
316                    let helper = if shared_present && exclusive_present {
317                        format!("with{gc}MixedInputLocks({shared}, {exclusive})")
318                    } else {
319                        format!(
320                            "with{gc}InputHandleLocks({}, {})",
321                            if shared_present { shared } else { exclusive },
322                            exclusive_present
323                        )
324                    };
325                    body = format!("{p}.{helper} {{ {body} }}");
326                    ordered.clear();
327                    break;
328                }
329                let expr = if shared_present && exclusive_present {
330                    format!("{p}.prepareMixedInputHandles({shared}, {exclusive})")
331                } else {
332                    format!(
333                        "{p}.prepareInputHandles({}, {})",
334                        if shared_present { shared } else { exclusive },
335                        exclusive_present
336                    )
337                };
338                prepare.push_str(&format!("val {name} = {expr}; "));
339                ordered.push(name);
340            }
341            for name in ordered.iter().rev() {
342                body = format!("{p}.with{gc}OrderedHandleLocks({name}) {{ {body} }}");
343            }
344            body = format!("{prepare}{body}");
345        }
346        format!("run {{ {}{body} }}", self.setup)
347    }
348    fn list(&self, ty: i64, shared: bool) -> String {
349        let sources = self
350            .sources
351            .iter()
352            .filter(|s| s.ty == ty && (s.access == Access::Shared) == shared)
353            .collect::<Vec<_>>();
354        crate::kotlin::lock_list(
355            &sources
356                .iter()
357                .filter(|s| !s.many)
358                .map(|s| s.local.clone())
359                .collect::<Vec<_>>(),
360            &sources
361                .iter()
362                .filter(|s| s.many)
363                .map(|s| s.local.clone())
364                .collect::<Vec<_>>(),
365        )
366    }
367}
