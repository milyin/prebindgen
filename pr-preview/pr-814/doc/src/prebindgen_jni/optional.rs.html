3pub(crate) fn base_niche(w: &JniKind) -> Option<JniNiche> {
4    match w {
5        JniKind::Handle(..) => Some(JniNiche::Null),
6        JniKind::Scalar(Primitive::Char | Primitive::U8 | Primitive::U16 | Primitive::U32) => {
7            Some(JniNiche::Integer(-1))
8        }
9        _ => None,
10    }
11}
12pub(crate) fn absent(w: &JniWire, niche: Option<&JniNiche>, input: bool) -> Result<TokenStream> {
13    if !input && matches!(&w.carrier, JniKind::Handle(..)) {
14        return Ok(quote!(std::ptr::null_mut()));
15    }
16    if let Some(n) = niche {
17        return Ok(n.integer_value());
18    }
19    Ok(match &w.carrier {
20        JniKind::Scalar(Primitive::F32 | Primitive::F64) => quote!(0.0),
21        JniKind::Text if !input => quote!(String::new()),
22        JniKind::Bytes if !input => quote!(Vec::<u8>::new()),
23        JniKind::Scalar(Primitive::I128 | Primitive::U128) if !input => quote!([0u8; 16]),
24        JniKind::Text | JniKind::Bytes | JniKind::Scalar(Primitive::I128 | Primitive::U128) => {
25            quote!(Default::default())
26        }
27        _ => quote!(0),
28    })
29}
30pub(crate) fn default_kt(w: &JniWire, niche: Option<&JniNiche>) -> String {
31    if let Some(n) = niche {
32        let v = match n {
33            JniNiche::Null => 0,
34            JniNiche::Integer(n) => *n,
35        };
36        return if matches!(
37            &w.carrier,
38            JniKind::Handle(..)
39                | JniKind::Scalar(
40                    Primitive::U32
41                        | Primitive::I64
42                        | Primitive::U64
43                        | Primitive::Isize
44                        | Primitive::Usize
45                )
46        ) {
47            format!("{v}L")
48        } else {
49            v.to_string()
50        };
51    }
52    match &w.carrier {
53        JniKind::Scalar(Primitive::Bool) => "false",
54        JniKind::Scalar(Primitive::I8) => "0.toByte()",
55        JniKind::Scalar(Primitive::I16) => "0.toShort()",
56        JniKind::Handle(..)
57        | JniKind::Scalar(
58            Primitive::U32 | Primitive::I64 | Primitive::U64 | Primitive::Isize | Primitive::Usize,
59        ) => "0L",
60        JniKind::Scalar(Primitive::F32) => "0.0f",
61        JniKind::Scalar(Primitive::F64) => "0.0",
62        JniKind::Text | JniKind::Bytes | JniKind::Scalar(Primitive::I128 | Primitive::U128) => {
63            "null"
64        }
65        _ => "0",
66    }
67    .into()
68}
69// Replace generated identifiers only, not substrings of other generated names.
70pub(crate) fn replace(s: &str, from: &str, to: &str) -> String {
71    let mut out = String::new();
72    let mut word = String::new();
73    for c in s.chars().chain(std::iter::once(' ')) {
74        if c.is_ascii_alphanumeric() || c == '_' {
75            word.push(c)
76        } else {
77            if word == from {
78                out.push_str(to)
79            } else {
80                out.push_str(&word)
81            }
82            word.clear();
83            out.push(c);
84        }
85    }
86    out.pop();
87    out
88}
89pub(crate) fn input_kotlin(
90    children: &[kotlin::JniNode],
91    group: &JniGroup,
92    o: &prebindgen_declarations::OptionalGroup<JniNiche>,
93    context: &KotlinContext<'_>,
94) -> Result<KotlinInput> {
95    let root = &group.name;
96    let nodes = &children[usize::from(o.niche.is_none())..];
97    let inner = kotlin::leaf_wires(nodes);
98    let mut out = KotlinInput::default();
99    let mut c = kotlin::input_kotlin(nodes, context)?;
100    // A single converted value is idiomatic T?. Multiple views retain an
101    // explicit presence argument so Some with nullable components is distinct.
102    let presence = if c.parameters.len() == 1 && !kotlin::has_optional(nodes) {
103        let (name, ty) = c.parameters[0].split_once(": ").unwrap();
104        let name = name.to_string();
105        let ty = ty.trim_end_matches('?');
106        out.parameters.push(format!("{name}: {ty}?"));
107        for a in c
108            .arguments
109            .iter_mut()
110            .chain(c.locks.iter_mut())
111            .chain(c.consumed.iter_mut())
112        {
113            *a = a
114                .replace(&format!("{name}?.raw ?: 0L"), &format!("{name}.raw"))
115                .replace(&format!("{name}?."), &format!("{name}."))
116                .replace(&format!("{name}!!"), &name);
117            *a = replace(a, &name, &format!("requireNotNull({name})"));
118        }
119        format!("{name} != null")
120    } else {
121        out.parameters.push(format!("{}: Boolean", root));
122        out.parameters.extend(c.parameters.clone());
123        root.to_string()
124    };
125    let tag = format!("(if ({presence}) 1 else 0)");
126    for a in c
127        .arguments
128        .iter_mut()
129        .chain(c.locks.iter_mut())
130        .chain(c.consumed.iter_mut())
131    {
132        // Every payload expression below runs only in the present branch.
133        *a = a.replace(&format!("{} == 1", root), "true");
134        *a = replace(a, root, "1");
135    }
136    if o.niche.is_none() {
137        out.arguments.push(tag);
138    }
139    for (j, a) in c.arguments.into_iter().enumerate() {
140        let niche = o
141            .niche
142            .as_ref()
143            .filter(|_| inner[j].name == o.discriminator);
144        let default = default_kt(&inner[j].kind, niche);
145        let a = if matches!(niche, Some(JniNiche::Integer(_))) {
146            format!("run {{ val value = {a}; require(value != {default}) {{ \"invalid option payload\" }}; value }}")
147        } else {
148            a
149        };
150        out.arguments
151            .push(format!("if ({presence}) {a} else {default}"));
152    }
153    out.locks.extend(
154        c.locks
155            .into_iter()
156            .map(|s| format!("if ({presence}) {s} else null")),
157    );
158    out.consumed.extend(
159        c.consumed
160            .into_iter()
161            .map(|s| format!("if ({presence}) {{ {s} }}")),
162    );
163    Ok(out)
164}
165pub(crate) fn output_kotlin(
166    children: &[kotlin::JniNode],
167    o: &prebindgen_declarations::OptionalGroup<JniNiche>,
168    context: &KotlinContext<'_>,
169) -> Result<KotlinOutput> {
170    let wires = kotlin::leaf_wires(children);
171    let flag = usize::from(o.niche.is_none());
172    let nodes = &children[flag..];
173    let p = if let Some(n) = &o.niche {
174        let i = wires
175            .iter()
176            .position(|w| w.name == o.discriminator)
177            .ok_or("missing optional discriminator")?;
178        let absent = default_kt(&wires[i].kind, Some(n));
179        format!("{}[{i}] != {absent}", context.result)
180    } else {
181        format!("{}[0] == 1", context.result)
182    };
183    let local = format!("{}_payload", context.result);
184    let ctx = KotlinContext {
185        result: &local,
186        ..*context
187    };
188    let mut c = kotlin::output_kotlin(nodes, &ctx)?;
189    kotlin::remap_output(&mut c, &local, context.result, flag, wires.len() - flag);
190    let mut out = KotlinOutput {
191        output_locals: c.output_locals,
192        ..Default::default()
193    };
194    if c.delivered.is_empty() || kotlin::has_optional(nodes) {
195        out.delivered.push(p.clone());
196        out.result_types.push("Boolean".into());
197    }
198    out.materialize = format!("            if ({p}) {{\n{}            }}\n", c.materialize);
199    out.cleanup = format!("            if ({p}) {{\n{}            }}\n", c.cleanup);
200    for (v, t) in c.delivered.into_iter().zip(c.result_types) {
201        out.delivered.push(format!("if ({p}) {v} else null"));
202        out.result_types
203            .push(format!("{}?", t.trim_end_matches('?')));
204    }
205    Ok(out)
206}

208/// A wire value excluded from the representation of a present value.
209#[derive(Clone, Debug)]
210pub enum JniNiche {
211    /// Null address (pointer or integer-encoded handle).
212    Null,
213    /// An invalid integer representation, within the wire's integer range.
214    Integer(i64),
215}
216impl JniNiche {
217    /// Expression recognizing the present case without decoding the payload.
218    pub(crate) fn present(&self, value: TokenStream) -> TokenStream {
219        match self {
220            Self::Null => quote!((#value as usize) != 0),
221            Self::Integer(n) => {
222                let n = syn::parse_str::<syn::Expr>(&n.to_string()).unwrap();
223                quote!(#value != #n)
224            }
225        }
226    }
227    /// Absence expression for a numeric wire. Pointer codecs supply a null pointer.
228    pub(crate) fn integer_value(&self) -> TokenStream {
229        match self {
230            Self::Null => quote!(0),
231            Self::Integer(n) => syn::parse_str::<TokenStream>(&n.to_string()).unwrap(),
232        }
233    }
234}

236pub(crate) fn remaining(wire: &JniWire, niche: &JniNiche) -> Option<JniNiche> {
237    let min = if matches!(&wire.carrier, JniKind::Scalar(Primitive::U32)) {
238        i64::MIN
239    } else {
240        i32::MIN as i64
241    };
242    match niche {
243        JniNiche::Integer(n) if *n > min => Some(JniNiche::Integer(n - 1)),
244        _ => None,
245    }
