1//! JNI adapter-owned descriptions of the wire tree.
2use super::*;
3use prebindgen_flat::{WireLayout, WireNode};
4use syn::ext::IdentExt;

6/// Metadata interpreted by the JNI binding writer.
7#[derive(Clone, Debug)]
8pub enum JniGroup {
9    /// Presence rule for the optional payload subtree.
10    Optional(JniOptional),
11    /// An ordered collection of field subtrees.
12    /// `Some` constructs a Kotlin data class; `None` exposes expanded values.
13    Fields(Option<DataGroup>),
14    /// A named source field, including empty and multi-wire fields.
15    Field(syn::Ident),
16}
17/// Optional-value information retained by the JNI adapter.
18#[derive(Clone, Debug)]
19pub struct JniOptional {
20    /// Normalized presence binding used by payload conditions.
21    pub presence: syn::Ident,
22    /// Descendant wire inspected for presence.
23    pub discriminator: syn::Ident,
24    /// Reserved absent representation; `None` means a separate 0/1 flag.
25    pub niche: Option<JniNiche>,
26}

28pub(crate) fn validate_layout(layout: &WireLayout<JniWire, JniGroup>) -> Result<()> {
29    layout.validate()?;
30    fn visit(
31        layout: &WireLayout<JniWire, JniGroup>,
32        bindings: &mut std::collections::BTreeSet<String>,
33        in_fields: bool,
34    ) -> Result<()> {
35        for node in layout.nodes() {
36            if let WireNode::Group { metadata, children } = node {
37                if let JniGroup::Optional(o) = metadata {
38                    if !bindings.insert(o.presence.to_string()) {
39                        return Err(format!("duplicate Kotlin group binding `{}`", o.presence));
40                    }
41                    if !children.iter().any(|w| w.name == o.discriminator) {
42                        return Err("optional discriminator outside group".into());
43                    }
44                    if o.niche.is_none()
45                        && !matches!(children.nodes().first(),Some(WireNode::Wire(w)) if w.name==o.discriminator)
46                    {
47                        return Err("optional flag must be the first child wire".into());
48                    }
49                }
50                if let JniGroup::Fields(Some(record)) = metadata {
51                    if !bindings.insert(record.root.clone()) {
52                        return Err(format!("duplicate Kotlin group binding `{}`", record.root));
53                    }
54                }
55                match metadata {
56                    JniGroup::Field(_) if !in_fields => {
57                        return Err(
58                            "named field must be a direct child of a field collection".into()
59                        );
60                    }
61                    JniGroup::Fields(_) => {
62                        let mut fields = std::collections::BTreeSet::new();
63                        for child in children.nodes() {
64                            let WireNode::Group { metadata, .. } = child else {
65                                return Err("field collection requires named field children".into());
66                            };
67                            let JniGroup::Field(name) = metadata else {
68                                return Err("field collection requires named field children".into());
69                            };
70                            if !fields.insert(name.unraw().to_string()) {
71                                return Err(format!("duplicate field `{name}`"));
72                            }
73                        }
74                    }
75                    _ => {}
76                }
77                visit(children, bindings, matches!(metadata, JniGroup::Fields(_)))?;
78            }
79        }
80        Ok(())
81    }
82    visit(layout, &mut std::collections::BTreeSet::new(), false)
83}

85#[cfg(test)]
86mod tests {
87    use super::*;
88    fn leaf(name: &str) -> prebindgen_flat::Wire<JniWire> {
89        prebindgen_flat::Wire {
90            name: syn::parse_str(name).unwrap(),
91            kind: JniWire::from(JniKind::Scalar(Primitive::Char)),
92        }
93    }
94    fn optional(presence: &str, discriminator: &str, niche: bool) -> JniGroup {
95        JniGroup::Optional(JniOptional {
96            presence: syn::parse_str(presence).unwrap(),
97            discriminator: syn::parse_str(discriminator).unwrap(),
98            niche: niche.then_some(JniNiche::Integer(-1)),
99        })
100    }
101    #[test]
102    fn nested_optional_references_and_presence_bindings_are_validated() {
103        let mut tree = WireLayout::from(vec![leaf("value")]);
104        tree.enclose(optional("inner", "value", true));
105        tree.enclose(optional("outer", "value", true));
106        validate_layout(&tree).unwrap();
107        assert_eq!(tree.len(), 1);
108        let mut bad = tree.clone();
109        bad.enclose(optional("missing", "elsewhere", true));
110        assert!(validate_layout(&bad).unwrap_err().contains("outside group"));
111        tree.enclose(optional("inner", "value", true));
112        assert!(validate_layout(&tree).unwrap_err().contains("duplicate"));
113        let mut flag = WireLayout::from(vec![leaf("flag")]);
114        flag.enclose(optional("flag", "flag", false));
115        validate_layout(&flag).unwrap();
116        flag.enclose(optional("outer", "flag", false));
117        assert!(validate_layout(&flag).unwrap_err().contains("first child"));
118    }
119    #[test]
120    fn field_names_placement_and_empty_records_are_validated() {
121        let mut field: WireLayout<JniWire, JniGroup> = WireLayout::default();
122        field.enclose(JniGroup::Field(syn::parse_quote!(x)));
123        assert!(validate_layout(&field)
124            .unwrap_err()
125            .contains("direct child"));
126        let mut duplicates = field.clone();
127        duplicates.extend(field.clone());
128        duplicates.enclose(JniGroup::Fields(None));
129        assert!(validate_layout(&duplicates)
130            .unwrap_err()
131            .contains("duplicate field"));
132        field.enclose(JniGroup::Fields(None));
133        validate_layout(&field).unwrap();
134        let mut malformed = WireLayout::from(vec![leaf("value")]);
135        malformed.enclose(JniGroup::Fields(None));
136        assert!(validate_layout(&malformed)
137            .unwrap_err()
138            .contains("named field children"));
139        let mut empty: WireLayout<JniWire, JniGroup> = WireLayout::default();
140        empty.enclose(JniGroup::Fields(None));
141        validate_layout(&empty).unwrap();
142    }
143    #[test]
144    fn malformed_metadata_is_rejected_before_final_slots_are_installed() {
145        let mut input = JniCodec
146            .input_leaf(
147                &syn::parse_quote!(i32),
148                &(),
149                syn::parse_quote!(wire),
150                syn::parse_quote!(value),
151                &prebindgen_declarations::Condition::default(),
152            )
153            .unwrap();
154        input
155            .layout
156            .enclose(JniGroup::Field(syn::parse_quote!(orphan)));
157        assert!(JniCodec
158            .finish_input(&mut input, &())
159            .unwrap_err()
160            .contains("direct child"));
161        let mut output = JniCodec
162            .output_leaf(
163                &syn::parse_quote!(i32),
164                &(),
165                syn::parse_quote!(wire),
166                quote!(value),
167                false,
168            )
169            .unwrap();
170        output
171            .layout
172            .enclose(JniGroup::Field(syn::parse_quote!(orphan)));
173        assert!(JniCodec
174            .finish_output(&mut output, &())
175            .unwrap_err()
176            .contains("direct child"));
177    }
