1//! Shared descriptions attached to parent nodes in a slot's wire tree.
2use super::*;
3use prebindgen_flat::WireNode;
4
5/// How the children of a tree node form one value.
6///
7/// The node owns its children in `WireLayout`; this description contains only
8/// the information needed to interpret them. For example, an optional integer
9/// has a flag leaf and an integer leaf, and its group names the flag to inspect.
10#[derive(Clone, Debug)]
11pub enum CompositionGroup<N, G> {
12    /// Describes how to distinguish `None` from `Some`.
13    Optional(OptionalGroup<N>),
14    /// Adapter information, such as the Kotlin class constructed from fields.
15    Adapter(G),
16}
17/// How an optional node decides whether its child value is present.
18#[derive(Clone, Debug)]
19pub struct OptionalGroup<N> {
20    /// Normalized 0/1 presence binding used by the prepared payload conditions.
21    /// Unique within the slot. With a separate flag it equals `discriminator`;
22    /// with a niche it may be a computed local rather than a physical wire.
23    pub presence: syn::Ident,
24    /// Name of a descendant wire to inspect: a flag or a payload wire with a niche.
25    pub discriminator: syn::Ident,
26    /// Reserved representation for `None`, or `None` when a separate 0/1 flag is used.
27    /// Remaining available niches belong to the leaf's wire kind instead.
28    pub niche: Option<N>,
29}
30/// Check optional discriminator references as well as the tree's unique physical wire names.
31/// Empty adapter groups are allowed; an optional group needs a discriminator wire.
32pub fn validate_layout<W: prebindgen_flat::WireKind, N, G>(
33    layout: &WireLayout<W, CompositionGroup<N, G>>,
34) -> Result<()> {
35    layout.validate()?;
36    fn visit<W: prebindgen_flat::WireKind, N, G>(
37        layout: &WireLayout<W, CompositionGroup<N, G>>,
38        bindings: &mut std::collections::BTreeSet<String>,
39    ) -> Result<()> {
40        for node in layout.nodes() {
41            if let WireNode::Group { metadata, children } = node {
42                if let CompositionGroup::Optional(o) = metadata {
43                    if !bindings.insert(o.presence.to_string()) {
44                        return Err(format!(
45                            "duplicate optional presence binding `{}`",
46                            o.presence
47                        ));
48                    }
49                    if !children.iter().any(|w| w.name == o.discriminator) {
50                        return Err("optional discriminator outside group".into());
51                    }
52                    if o.niche.is_none()
53                        && !matches!(children.nodes().first(),Some(WireNode::Wire(w)) if w.name==o.discriminator)
54                    {
55                        return Err("optional flag must be the first child wire".into());
56                    }
57                }
58                visit(children, bindings)?;
59            }
60        }
61        Ok(())
62    }
63    visit(layout, &mut std::collections::BTreeSet::new())
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69    fn wire(name: &str) -> Wire<syn::Type> {
70        Wire {
71            name: syn::parse_str(name).unwrap(),
72            kind: syn::parse_quote!(u32),
73        }
74    }
75    fn optional(name: &str, discriminator: &str, niche: Option<()>) -> CompositionGroup<(), ()> {
76        CompositionGroup::Optional(OptionalGroup {
77            presence: syn::parse_str(name).unwrap(),
78            discriminator: syn::parse_str(discriminator).unwrap(),
79            niche,
80        })
81    }
82    #[test]
83    fn nested_options_share_one_leaf_but_keep_distinct_group_rules() {
84        let mut layout: WireLayout<_, _> = vec![wire("value")].into();
85        layout.enclose(optional("inner", "value", Some(())));
86        layout.enclose(optional("outer", "value", Some(())));
87        validate_layout(&layout).unwrap();
88        assert_eq!(layout.len(), 1);
89        assert!(layout
90            .iter()
91            .all(|w| w.name != "inner" && w.name != "outer"));
92        layout.enclose(optional("wrong", "missing", Some(())));
93        assert!(validate_layout(&layout)
94            .unwrap_err()
95            .contains("outside group"));
96    }
97    #[test]
98    fn flags_are_direct_children_and_adapter_groups_need_no_identity() {
99        let mut layout: WireLayout<_, _> = vec![wire("flag")].into();
100        layout.enclose(CompositionGroup::Adapter(()));
101        layout.enclose(optional("option", "flag", None));
102        assert!(validate_layout(&layout)
103            .unwrap_err()
104            .contains("first child"));
105        let mut layout: WireLayout<_, _> = vec![wire("flag")].into();
106        layout.enclose(optional("option", "flag", None));
107        validate_layout(&layout).unwrap(); // Option<()> has only its flag.
108        layout.enclose(optional("option", "flag", Some(())));
109        assert!(validate_layout(&layout)
110            .unwrap_err()
111            .contains("duplicate optional presence"));
112        let mut empty: WireLayout<syn::Type, CompositionGroup<(), ()>> = WireLayout::default();
113        empty.enclose(CompositionGroup::Adapter(()));
114        empty.enclose(CompositionGroup::Adapter(()));
115        validate_layout(&empty).unwrap();
116        empty.enclose(optional("bad", "missing", Some(())));
117        assert!(validate_layout(&empty).is_err());
118    }
