1//! Shared descriptions attached to parent nodes in a slot's wire tree.
2use super::*;
3use prebindgen_flat::WireNode;
4
5/// How the children of a tree node form one value.
6///
7/// The node owns its children in `WireLayout`; this description contains only
8/// the information needed to interpret them. For example, an optional integer
9/// has a flag leaf and an integer leaf, and its group names the flag to inspect.
10#[derive(Clone, Debug)]
11pub struct CompositionGroup<N, G> {
12    /// Generated name for this value, unique within the slot.
13    pub name: String,
14    /// An optional value or a value described by the language adapter.
15    pub kind: GroupKind<N, G>,
16}
17/// Interpretation of a group's child values.
18#[derive(Clone, Debug)]
19pub enum GroupKind<N, G> {
20    /// Describes how to distinguish `None` from `Some`.
21    Optional(OptionalGroup<N>),
22    /// Adapter information, such as the Kotlin class constructed from fields.
23    Adapter(G),
24}
25/// How an optional node decides whether its child value is present.
26#[derive(Clone, Debug)]
27pub struct OptionalGroup<N> {
28    /// Name of a descendant wire to inspect: a flag or a payload wire with a niche.
29    pub discriminator: syn::Ident,
30    /// Reserved representation for `None`, or `None` when a separate 0/1 flag is used.
31    /// Remaining available niches belong to the leaf's wire kind instead.
32    pub niche: Option<N>,
33}
34/// Check shared group references as well as the tree's unique physical wire names.
35/// Empty adapter groups are allowed; an optional group needs a discriminator wire.
36pub fn validate_layout<W: prebindgen_flat::WireKind, N, G>(
37    layout: &WireLayout<W, CompositionGroup<N, G>>,
38) -> Result<()> {
39    layout.validate()?;
40    fn visit<W: prebindgen_flat::WireKind, N, G>(
41        layout: &WireLayout<W, CompositionGroup<N, G>>,
42        names: &mut std::collections::BTreeSet<String>,
43    ) -> Result<()> {
44        for node in layout.nodes() {
45            if let WireNode::Group { metadata, children } = node {
46                if !names.insert(metadata.name.clone()) {
47                    return Err(format!("duplicate group name `{}`", metadata.name));
48                }
49                if let GroupKind::Optional(o) = &metadata.kind {
50                    if !children.iter().any(|w| w.name == o.discriminator) {
51                        return Err("optional discriminator outside group".into());
52                    }
53                    if o.niche.is_none()
54                        && !matches!(children.nodes().first(),Some(WireNode::Wire(w)) if w.name==o.discriminator)
55                    {
56                        return Err("optional flag must be the first child wire".into());
57                    }
58                }
59                visit(children, names)?;
60            }
61        }
62        Ok(())
63    }
64    visit(layout, &mut std::collections::BTreeSet::new())
65}

67#[cfg(test)]
68mod tests {
69    use super::*;
70    fn wire(name: &str) -> Wire<syn::Type> {
71        Wire {
72            name: syn::parse_str(name).unwrap(),
73            kind: syn::parse_quote!(u32),
74        }
75    }
76    fn optional(name: &str, discriminator: &str, niche: Option<()>) -> CompositionGroup<(), ()> {
77        CompositionGroup {
78            name: name.into(),
79            kind: GroupKind::Optional(OptionalGroup {
80                discriminator: syn::parse_str(discriminator).unwrap(),
81                niche,
82            }),
83        }
84    }
85    #[test]
86    fn nested_options_share_one_leaf_but_keep_distinct_group_rules() {
87        let mut layout: WireLayout<_, _> = vec![wire("value")].into();
88        layout.enclose(optional("inner", "value", Some(())));
89        layout.enclose(optional("outer", "value", Some(())));
90        validate_layout(&layout).unwrap();
91        assert_eq!(layout.len(), 1);
92        layout.enclose(optional("wrong", "missing", Some(())));
93        assert!(validate_layout(&layout)
94            .unwrap_err()
95            .contains("outside group"));
96    }
97    #[test]
98    fn flags_are_direct_children_and_group_names_are_unique() {
99        let mut layout: WireLayout<_, _> = vec![wire("flag")].into();
100        layout.enclose(CompositionGroup {
101            name: "record".into(),
102            kind: GroupKind::Adapter(()),
103        });
104        layout.enclose(optional("option", "flag", None));
105        assert!(validate_layout(&layout)
106            .unwrap_err()
107            .contains("first child"));
108        let mut layout: WireLayout<_, _> = vec![wire("flag")].into();
109        layout.enclose(optional("option", "flag", None));
110        validate_layout(&layout).unwrap(); // Option<()> has only its flag.
111        layout.enclose(optional("option", "flag", Some(())));
112        assert!(validate_layout(&layout)
113            .unwrap_err()
114            .contains("duplicate group"));
115        let mut empty: WireLayout<syn::Type, CompositionGroup<(), ()>> = WireLayout::default();
116        empty.enclose(CompositionGroup {
117            name: "unit".into(),
118            kind: GroupKind::Adapter(()),
119        });
120        validate_layout(&empty).unwrap();
121        empty.enclose(optional("bad", "missing", Some(())));
122        assert!(validate_layout(&empty).is_err());
123    }
