1//! Shared descriptions attached to parent nodes in a slot's wire tree.
2use super::*;
3use prebindgen_flat::WireNode;
4
5/// Shared structure and optional destination-specific information for a parent node.
6///
7/// The structure describes how the children form a value. An annotation can add
8/// destination information, such as the Kotlin class constructed from fields,
9/// without replacing that structure. Physical wires still belong only to leaves.
10#[derive(Clone, Debug)]
11pub struct CompositionGroup<N, A> {
12    /// Meaning of the ordered children, independent of the destination language.
13    pub structure: GroupStructure<N>,
14    /// Destination-specific information attached to this same structural node.
15    pub annotation: Option<A>,
16}
17impl<N, A> CompositionGroup<N, A> {
18    /// Describe a shared structure without destination-specific information.
19    pub fn new(structure: GroupStructure<N>) -> Self {
20        Self {
21            structure,
22            annotation: None,
23        }
24    }
25    /// Attach destination-specific information while preserving the structure.
26    pub fn annotated(mut self, annotation: A) -> Self {
27        self.annotation = Some(annotation);
28        self
29    }
30}
31/// Meaning of a parent's children, produced by the shared declaration compiler.
32#[derive(Clone, Debug)]
33pub enum GroupStructure<N> {
34    /// Optional payload and its presence rule.
35    Optional(OptionalGroup<N>),
36    /// Ordered named fields. Each immediate child must be a `Field` group.
37    Fields,
38    /// One source field; its children may contain zero, one or several wires.
39    Field(syn::Ident),
40    /// Adapter-defined structure. Requires an annotation describing its meaning.
41    Custom,
42}
43/// How an optional node decides whether its child value is present.
44#[derive(Clone, Debug)]
45pub struct OptionalGroup<N> {
46    /// Normalized 0/1 presence binding used by the prepared payload conditions.
47    /// Unique within the slot. With a separate flag it equals `discriminator`;
48    /// with a niche it may be a computed local rather than a physical wire.
49    pub presence: syn::Ident,
50    /// Name of a descendant wire to inspect: a flag or a payload wire with a niche.
51    pub discriminator: syn::Ident,
52    /// Reserved representation for `None`, or `None` when a separate 0/1 flag is used.
53    /// Remaining available niches belong to the leaf's wire kind instead.
54    pub niche: Option<N>,
55}
56/// Check physical wire names, optional bindings and references, and field structure.
57/// Field collections contain uniquely named field children, including empty fields.
58/// Custom structures require an adapter annotation describing their meaning.
59pub fn validate_layout<W: prebindgen_flat::WireKind, N, G>(
60    layout: &WireLayout<W, CompositionGroup<N, G>>,
61) -> Result<()> {
62    layout.validate()?;
63    fn visit<W: prebindgen_flat::WireKind, N, G>(
64        layout: &WireLayout<W, CompositionGroup<N, G>>,
65        bindings: &mut std::collections::BTreeSet<String>,
66        in_fields: bool,
67    ) -> Result<()> {
68        for node in layout.nodes() {
69            if let WireNode::Group { metadata, children } = node {
70                if let GroupStructure::Optional(o) = &metadata.structure {
71                    if !bindings.insert(o.presence.to_string()) {
72                        return Err(format!(
73                            "duplicate optional presence binding `{}`",
74                            o.presence
75                        ));
76                    }
77                    if !children.iter().any(|w| w.name == o.discriminator) {
78                        return Err("optional discriminator outside group".into());
79                    }
80                    if o.niche.is_none()
81                        && !matches!(children.nodes().first(),Some(WireNode::Wire(w)) if w.name==o.discriminator)
82                    {
83                        return Err("optional flag must be the first child wire".into());
84                    }
85                }
86                match &metadata.structure {
87                    GroupStructure::Field(_) if !in_fields => {
88                        return Err(
89                            "named field must be a direct child of a field collection".into()
90                        );
91                    }
92                    GroupStructure::Fields => {
93                        let mut fields = std::collections::BTreeSet::new();
94                        for child in children.nodes() {
95                            let WireNode::Group { metadata, .. } = child else {
96                                return Err("field collection requires named field children".into());
97                            };
98                            let GroupStructure::Field(name) = &metadata.structure else {
99                                return Err("field collection requires named field children".into());
100                            };
101                            if !fields.insert(name.to_string()) {
102                                return Err(format!("duplicate field `{name}`"));
103                            }
104                        }
105                    }
106                    GroupStructure::Custom if metadata.annotation.is_none() => {
107                        return Err("custom structure requires an adapter annotation".into());
108                    }
109                    _ => {}
110                }
111                visit(
112                    children,
113                    bindings,
114                    matches!(metadata.structure, GroupStructure::Fields),
115                )?;
116            }
117        }
118        Ok(())
119    }
120    visit(layout, &mut std::collections::BTreeSet::new(), false)
121}

123#[cfg(test)]
124mod tests {
125    use super::*;
126    fn wire(name: &str) -> Wire<syn::Type> {
127        Wire {
128            name: syn::parse_str(name).unwrap(),
129            kind: syn::parse_quote!(u32),
130        }
131    }
132    fn optional(name: &str, discriminator: &str, niche: Option<()>) -> CompositionGroup<(), ()> {
133        CompositionGroup::new(GroupStructure::Optional(OptionalGroup {
134            presence: syn::parse_str(name).unwrap(),
135            discriminator: syn::parse_str(discriminator).unwrap(),
136            niche,
137        }))
138    }
139    #[test]
140    fn nested_options_share_one_leaf_but_keep_distinct_group_rules() {
141        let mut layout: WireLayout<_, _> = vec![wire("value")].into();
142        layout.enclose(optional("inner", "value", Some(())));
143        layout.enclose(optional("outer", "value", Some(())));
144        validate_layout(&layout).unwrap();
145        assert_eq!(layout.len(), 1);
146        assert!(layout
147            .iter()
148            .all(|w| w.name != "inner" && w.name != "outer"));
149        layout.enclose(optional("wrong", "missing", Some(())));
150        assert!(validate_layout(&layout)
151            .unwrap_err()
152            .contains("outside group"));
153    }
154    #[test]
155    fn flags_are_direct_children_and_adapter_groups_need_no_identity() {
156        let mut layout: WireLayout<_, _> = vec![wire("flag")].into();
157        layout.enclose(CompositionGroup::new(GroupStructure::Custom).annotated(()));
158        layout.enclose(optional("option", "flag", None));
159        assert!(validate_layout(&layout)
160            .unwrap_err()
161            .contains("first child"));
162        let mut layout: WireLayout<_, _> = vec![wire("flag")].into();
163        layout.enclose(optional("option", "flag", None));
164        validate_layout(&layout).unwrap(); // Option<()> has only its flag.
165        layout.enclose(optional("option", "flag", Some(())));
166        assert!(validate_layout(&layout)
167            .unwrap_err()
168            .contains("duplicate optional presence"));
169        let mut empty: WireLayout<syn::Type, CompositionGroup<(), ()>> = WireLayout::default();
170        empty.enclose(CompositionGroup::new(GroupStructure::Custom).annotated(()));
171        empty.enclose(CompositionGroup::new(GroupStructure::Custom).annotated(()));
172        validate_layout(&empty).unwrap();
173        empty.enclose(optional("bad", "missing", Some(())));
174        assert!(validate_layout(&empty).is_err());
175    }
176    #[test]
177    fn shared_structures_accept_annotations_and_validate_field_children() {
178        let mut field: WireLayout<_, CompositionGroup<(), &str>> = vec![wire("value")].into();
179        field.enclose(CompositionGroup::new(GroupStructure::Field(
180            syn::parse_quote!(x),
181        )));
182        field.enclose(CompositionGroup::new(GroupStructure::Fields).annotated("KotlinPoint"));
183        validate_layout(&field).unwrap();
184        field.enclose(
185            CompositionGroup::new(GroupStructure::Optional(OptionalGroup {
186                presence: syn::parse_quote!(present),
187                discriminator: syn::parse_quote!(value),
188                niche: Some(()),
189            }))
190            .annotated("nullable point"),
191        );
192        validate_layout(&field).unwrap();
193        let mut invalid: WireLayout<_, CompositionGroup<(), ()>> = vec![wire("value")].into();
194        invalid.enclose(CompositionGroup::new(GroupStructure::Fields));
195        assert!(validate_layout(&invalid)
196            .unwrap_err()
197            .contains("named field children"));
198        let mut a: WireLayout<syn::Type, CompositionGroup<(), ()>> = WireLayout::default();
199        a.enclose(CompositionGroup::new(GroupStructure::Field(
200            syn::parse_quote!(x),
201        )));
202        a.extend(a.clone());
203        a.enclose(CompositionGroup::new(GroupStructure::Fields));
204        assert!(validate_layout(&a).unwrap_err().contains("duplicate field"));
205        let mut orphan: WireLayout<syn::Type, CompositionGroup<(), ()>> = WireLayout::default();
206        orphan.enclose(CompositionGroup::new(GroupStructure::Field(
207            syn::parse_quote!(orphan),
208        )));
209        assert!(validate_layout(&orphan)
210            .unwrap_err()
211            .contains("direct child"));
212        let mut custom: WireLayout<syn::Type, CompositionGroup<(), ()>> = WireLayout::default();
213        custom.enclose(CompositionGroup::new(GroupStructure::Custom));
214        assert!(validate_layout(&custom)
215            .unwrap_err()
216            .contains("requires an adapter annotation"));
217    }
