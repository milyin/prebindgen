1//! The parameter side, stated as rows and read back as leaves.
2//!
3//! An `expand_param!` declaration says a value may be built by calling one of
4//! several constructors, or handed over already built. That is a row: a product
5//! for one constructor, a choice for several. This module writes those rows,
6//! and answers the questions a row deliberately does not — what the leaves are
7//! called, what a selector and a presence flag are — so
8//! [`Folding::fold`](prebindgen_registry::fold::Folding::fold) can read a
9//! `FoldPlan` back off them.
10//!
11//! This is where the parameter side is decided. The plans the emitters read
12//! come from here, built before there is a registry, and the readings they
13//! deliver are handed over beside them (#701 step 2).
14

25/// The row every `expand_param!` for a type declares.
26///
27/// One name for all of them: a nested build looks up the argument's own type
28/// under this name, and a type is built one way wherever it is reached.
29pub(crate) fn type_row() -> RecipeName {
30    RecipeName::new("expand-param")
31}
32
33/// The row one function's `.expand_param(...)` declares for one parameter.
34///
35/// A name of its own, because it sits on the same crossing as the type-level
36/// row and says something different. A site takes this one; a nested build
37/// never does.
38pub(crate) fn site_row(func: &syn::Ident, param: &str) -> RecipeName {
39    RecipeName::new(format!("expand-param@{func}#{param}"))
40}

42/// What the JVM calls the values a parameter comes apart into.
43///
44/// A row states none of this: what a selector and a presence flag are called
45/// and typed, how a leaf is named under a product, an arm or a nested build,
46/// and what a part's type becomes inside an arm are all the target's answers.
47pub(crate) struct JniFold;

49fn ident(name: &str) -> syn::Ident {
50    syn::Ident::new(name, proc_macro2::Span::call_site())
51}
52
53impl FoldPolicy for JniFold {
54    fn selector(&self, prefix: &str) -> FoldLeaf {
55        FoldLeaf {
56            name: ident(&format!("{prefix}_sel")),
57            ty: TypeRef::scalar(ScalarKind::I32),
58        }
59    }
60
61    fn presence(&self, prefix: &str) -> FoldLeaf {
62        FoldLeaf {
63            name: ident(&format!("{prefix}_present")),
64            ty: TypeRef::scalar(ScalarKind::Bool),
65        }
66    }
67
68    fn sole(&self, prefix: &str) -> syn::Ident {
69        ident(prefix)
70    }
71
72    fn part(&self, prefix: &str, name: &str) -> syn::Ident {
73        ident(&format!("{prefix}_{name}"))
74    }
75
76    fn arm_sole(&self, prefix: &str, arm: usize) -> syn::Ident {
77        ident(&format!("{prefix}_{arm}"))
78    }
79
80    fn arm_part(&self, prefix: &str, arm: usize, index: usize) -> syn::Ident {
81        ident(&format!("{prefix}_{arm}_{index}"))
82    }
83
84    fn presence_leaf(&self, parts: usize) -> bool {
85        // EXACTLY one part carries absence itself, by being nullable. Past one
86        // it cannot, and a flag in front is cheaper than boxing a nullable
87        // primitive per part — an `Option<i32>` argument would arrive as an
88        // `Integer?`. Below one there is nothing to carry it at all: a value
89        // built by a nullary constructor has no leaf but the flag.
90        parts != 1
91    }
92
93    fn identity_leaf_ty(&self, ty: &TypeRef, borrowed: bool) -> TypeRef {
94        // Optional because the selector decides whether this arm is live; a
95        // borrowed crossing lends the value, so the leaf carries the borrow and
96        // the arm clones out of it.
97        if borrowed {
98            ty.borrowed().optional()
99        } else {
100            ty.optional()
101        }
102    }
103
104    fn arm_leaf_ty(&self, ty: &TypeRef) -> (TypeRef, bool) {
105        // An argument already optional keeps its own type. The wire has no
106        // second absence to spend, and `None` is a legitimate value for the arm
107        // the selector picked.
108        if ty.optional_inner().is_some() {
109            (ty.clone(), true)
110        } else {
111            (ty.optional(), false)
112        }
113    }
114}

116/// One `expand_param!` variant list, as a row.
117///
118/// A single constructor is a product — there is nothing to choose between. Two
119/// or more is a choice whose arms name no alternative: they are ways to obtain
120/// the value, not alternatives of a sum.
121fn row_of(variants: &[LocalVariant]) -> Constructing {
122    if let [LocalVariant::Ctor(func)] = variants {
123        return Shape::Product(Construct::Call(func.clone()));
124    }
125    Shape::Choice {
126        arms: variants
127            .iter()
128            .map(|variant| Arm {
129                alternative: None,
130                op: match variant {
131                    LocalVariant::Ctor(func) => Construct::Call(func.clone()),
132                    LocalVariant::SelfIdentity => Construct::Identity,
133                },
134            })
135            .collect(),
136    }
137}

139/// An identity-only list declares the plain form — the value crosses as itself,
140/// which is what happens with no declaration at all. It states no row.
141fn states_a_row(variants: &[LocalVariant]) -> bool {
142    !matches!(variants, [] | [LocalVariant::SelfIdentity])
143}

145impl Declarations {
146    /// Every row an `expand_param!` declaration states, type-level and per
147    /// function.
148    ///
149    /// Read off the declarations and the model, so this can be stated before
150    /// there is a registry to ask.
151    pub(crate) fn expansion_rows(&self, model: &Flat) -> Vec<(TypeRef, RecipeName, Constructing)> {
152        let reading = |key: &TypeKey| {
153            let ty: syn::Type = syn::parse_str(key.as_str()).ok()?;
154            model.classify(&ty).ok()
155        };
156        let mut rows = Vec::new();
157        for decl in &self.param_expand_decls {
158            assert!(
159                self.is_class_declared(decl.key())
160                    || !decl
161                        .variants()
162                        .iter()
163                        .any(|v| matches!(v, LocalVariant::SelfIdentity)),
164                "expand_param!({k}).variant_self(): `{k}` has no class declaration, so there is \
165                 no Kotlin object to pass — drop .variant_self() (the type is rust-side-only) \
166                 or declare the type in a package",
167                k = decl.key().as_str()
168            );
169            if !states_a_row(decl.variants()) {
170                continue;
171            }
172            if let Some(ty) = reading(decl.key()) {
173                rows.push((ty, type_row(), row_of(decl.variants())));
174            }
175        }
176        for (func, param, decl) in &self.fn_param_expands {
177            assert!(
178                self.is_class_declared(decl.key())
179                    || !decl
180                        .variants()
181                        .iter()
182                        .any(|v| matches!(v, LocalVariant::SelfIdentity)),
183                "fun!({func}).expand_param(\"{param}\", expand_param!({k}).variant_self()): `{k}` \
184                 has no class declaration, so there is no Kotlin object to pass — drop \
185                 .variant_self() (the type is rust-side-only) or declare the type in a package",
186                k = decl.key().as_str()
187            );
188            if !states_a_row(decl.variants()) {
189                continue;
190            }
191            if let Some(ty) = reading(decl.key()) {
192                rows.push((ty, site_row(func, param), row_of(decl.variants())));
193            }
194        }
195        rows
196    }

198    /// The row one parameter's construction is stated under.
199    ///
200    /// A function that overrode this parameter states its own; everything else
201    /// takes the type's.
202    fn row_for(&self, func: &syn::Ident, param: &syn::Ident) -> RecipeName {
203        let overridden = self.fn_param_expands.iter().any(|(f, p, decl)| {
204            f == func && p == &param.to_string() && states_a_row(decl.variants())
205        });
206        if overridden {
207            site_row(func, &param.to_string())
208        } else {
209            type_row()
210        }
211    }

213    /// Which parameters are built from leaves, derived from the declarations
214    /// alone.
215    ///
216    /// This is the half of the parameter side a row cannot state: a row says
217    /// how a type is built, and this says where that happens. A per-function `.expand_param(...)`
218    /// names its own position. A type-level `expand_param!` applies to every
219    /// parameter of that type in every exported function — except an accessor,
220    /// which is not a composer; except the receiver a method is called on,
221    /// which binds to `this`; and except a position whose own declaration
222    /// asked for the plain value instead.
223    pub(crate) fn expanded_positions(
224        &self,
225        model: &Flat,
226        exports: &std::collections::HashSet<syn::Ident>,
227        accessors: &std::collections::HashSet<syn::Ident>,
228    ) -> std::collections::BTreeSet<(String, String)> {
229        // The value a declaration is about, under the layers a parameter may
230        // wear: `Option<&T>` is a way of passing a `T`.
231        let core = |ty: &TypeRef| {
232            let after_opt = ty.optional_inner().unwrap_or(ty);
233            after_opt.borrow_target().unwrap_or(after_opt).key()
234        };
235        let mut positions = std::collections::BTreeSet::new();
236        // A per-function declaration that asked for the plain value is not an
237        // expansion, and it also stops the type-level one from applying there.
238        let mut plain = std::collections::BTreeSet::new();
239        for (func, param, decl) in &self.fn_param_expands {
240            let position = (func.to_string(), param.clone());
241            if states_a_row(decl.variants()) {
242                positions.insert(position);
243            } else {
244                plain.insert(position);
245            }
246        }
247        let receivers = self.method_receivers();
248        for decl in &self.param_expand_decls {
249            if !states_a_row(decl.variants()) {
250                continue;
251            }
252            let target = decl.key().clone();
253            let mut names: Vec<&syn::Ident> = exports.iter().collect();
254            names.sort_by_key(|name| name.to_string());
255            for func in names {
256                if accessors.contains(func) {
257                    continue;
258                }
259                let Some(function) = model.function(func) else {
260                    continue;
261                };
262                let receiver = receivers.get(func);
263                let mut took_receiver = false;
264                for param in &function.params {
265                    let bare = core(&param.ty);
266                    if !took_receiver && receiver == Some(&bare) {
267                        took_receiver = true;
268                        continue;
269                    }
270                    if bare != target {
271                        continue;
272                    }
273                    let position = (func.to_string(), param.name.to_string());
274                    if plain.contains(&position) {
275                        continue;
276                    }
277                    positions.insert(position);
278                }
279            }
280        }
281        positions
282    }

284    /// What a per-function `.expand_param(...)` says about a position, checked
285    /// against the position itself.
286    ///
287    /// Neither check belongs to a row: a row says how a TYPE is built, and
288    /// these are about the parameter the declaration was written for. Naming a
289    /// parameter the function does not have, or a type it does not take, is a
290    /// declaration that would otherwise do nothing at all — the row would sit
291    /// on a crossing nothing reaches.
292    fn check_param_expands(
293        &self,
294        model: &Flat,
295        accessors: &std::collections::HashSet<syn::Ident>,
296    ) -> Result<(), String> {
297        for (func, param, decl) in &self.fn_param_expands {
298            // An accessor reads a value out; it never builds one. Refused for
299            // every such declaration, including one asking for the plain
300            // value: the answer is the same either way, and a declaration that
301            // cannot mean anything is worth saying so about.
302            if accessors.contains(func) {
303                return Err(format!(
304                    "`{func}` is an accessor, so its `{param}` is not built from leaves — an \
305                     accessor reads a value out rather than composing one"
306                ));
307            }
308            let Some(function) = model.function(func) else {
309                return Err(format!(
310                    "`{func}` has a parameter expansion, and no `#[prebindgen]` function of \
311                     that name"
312                ));
313            };
314            let Some(found) = function.params.iter().find(|p| p.name == ident(param)) else {
315                let names = function
316                    .params
317                    .iter()
318                    .map(|p| format!("`{}`", p.name))
319                    .collect::<Vec<_>>()
320                    .join(", ");
321                return Err(format!(
322                    "`{func}` has no parameter `{param}` to expand; it takes {names}"
323                ));
324            };
325            let after_opt = found.ty.optional_inner().unwrap_or(&found.ty);
326            let bare = after_opt.borrow_target().unwrap_or(after_opt).key();
327            if bare != *decl.key() {
328                return Err(format!(
329                    "`{func}`'s `{param}` is a `{bare}`, and its expansion is declared for \
330                     `{declared}`",
331                    declared = decl.key().as_str()
332                ));
333            }
334        }
335        Ok(())
336    }

338    /// Every expanded parameter's plan, read off its row, and the readings
339    /// those plans deliver.
340    ///
341    /// Read from the declarations and the model, so this can be answered before
342    /// there is a registry — which is what lets the leaves be handed over as a
343    /// fact rather than asked for mid-resolve. The rows go in a table of their
344    /// own for the same reason: the binding's own table is built later, from a
345    /// registry this runs before.
346    pub(crate) fn expansion_plans(
347        &self,
348        model: &Flat,
349        exports: &std::collections::HashSet<syn::Ident>,
350        accessors: &std::collections::HashSet<syn::Ident>,
351    ) -> Result<(ExpansionPlans, Vec<TypeRef>), String> {
352        self.check_param_expands(model, accessors)?;
353        let mut builder = Recipes::builder();
354        let mut seen = std::collections::HashSet::new();
355        for (ty, name, row) in self.expansion_rows(model) {
356            // Nothing else declares a row here, so the value's own crossing is
357            // the derived one — see `Declarations::recipes` for the same
358            // arrangement beside the rows a class declares.
359            if seen.insert(ty.key()) {
360                builder.declare_derived_default(ty.clone(), Direction::Construct);
361            }
362            builder.declare(ty, name, row);
363        }
364        let recipes = builder.build(model).map_err(|errors| {
365            errors
366                .iter()
367                .map(|e| e.to_string())
368                .collect::<Vec<_>>()
369                .join("; ")
370        })?;
371        let folding = Folding::new(&recipes, model);
372
373        let mut plans = ExpansionPlans::new();
374        let mut leaves = Vec::new();
375        for (func, param) in self.expanded_positions(model, exports, accessors) {
376            let (func, param) = (ident(&func), ident(&param));
377            let reading = model
378                .function(&func)
379                .and_then(|f| f.params.iter().find(|p| p.name == param))
380                .map(|p| p.ty.clone())
381                .ok_or_else(|| format!("`{func}` has no parameter `{param}` to expand"))?;
382            let row = self.row_for(&func, &param);
383            let plan = folding
384                .fold(&JniFold, &param.to_string(), &reading, &row, &type_row())
385                .map_err(|e| format!("`{func}`'s `{param}` does not fold from its row: {e}"))?;
386            // One callback per expanded parameter. A delivered value's site is
387            // named by the parameter it arrived on rather than by the leaf, so
388            // two callbacks here would give two positions one identity.
389            if plan
390                .leaves
391                .iter()
392                .filter(|leaf| leaf.ty.callback_args().is_some())
393                .count()
394                > 1
395            {
396                return Err(format!(
397                    "`{func}`'s `{param}` is built from more than one callback, and one \
398                     parameter has no way to name them apart"
399                ));
400            }
401            leaves.extend(plan.leaves.iter().map(|leaf| leaf.ty.clone()));
402            plans.insert((func, param), plan);
403        }
404        Ok((plans, leaves))
405    }
406}
