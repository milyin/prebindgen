1//! The common Rust writer: one component, both targets.
2//!
3//! It owns the shape of the generated Rust — the locals and their order, the
4//! branches on failure, the construction of the source value, the call, the
5//! return — and allocates every temporary from the plan, so two operations
6//! rendered into one wrapper cannot collide over a name. A target contributes
7//! one expression per operation of its own ([`Target::render_operation`]) and
8//! whole [`Artifact`]s; it contributes no control flow.
9//!
10//! Source types and struct shapes are generated through Flat's emission
11//! capability, never from retained syntax and never spelled by an adapter.
12

28/// This engine's emission key.
29///
30/// Implementing [`RustEmitter`] is a collector's deliberate decision to
31/// establish an emission boundary; v2 establishes its own rather than borrowing
32/// v1's.
33pub(crate) struct Writer;

35impl RustEmitter for Writer {}

37/// How generated code names each entity: the module it is reached through.
38///
39/// A captured item is reached through the source module the frontend
40/// configured for the run. An item the binding defines itself is reached
41/// where the binding said it is — `crate::helpers::x` — which its origin
42/// records as its crate stamp, since for a binding-local item that is what
43/// the stamp means. One value carries both so the writer asks one question.
44pub(crate) struct Reach {
45    /// Name → module, for every entity that is not reached through the
46    /// default. The shape [`RustEmitter::emit_source_type`] takes.
47    modules: HashMap<String, syn::Path>,
48    default: syn::Path,
49}
50
51impl Reach {
52    pub(crate) fn new(flat: &Flat, default: syn::Path) -> Self {
53        let modules = flat
54            .elements()
55            .filter_map(|element| Some((element.name()?, element)))
56            .filter(|(name, _)| flat.is_binding_local(*name))
57            .filter_map(|(name, element)| {
58                let module = element.location().crate_name.as_deref()?;
59                Some((name.to_string(), syn::parse_str(module).ok()?))
60            })
61            .collect();
62        Reach { modules, default }
63    }
64
65    /// The module the entity with this name is reached through.
66    fn of(&self, name: &str) -> &syn::Path {
67        self.modules.get(name).unwrap_or(&self.default)
68    }
69}

71/// The whole generated Rust file, as text.
72pub(crate) fn render<T: Target>(
73    flat: &Flat,
74    target: &T,
75    reach: &Reach,
76    artifacts: &[Artifact],
77    primitives: &[PrimitiveSpec<T::Payload>],
78    functions: &[FunctionPlan<T::Payload>],
79) -> String {
80    // The guards come first, and they are emitted whatever else this run
81    // retained. Each one is a compile-time assertion the capture reader
82    // injected — today, that the source crate's features match the set the
83    // capture was filtered by — and a binding that dropped it would compile
84    // happily against a source crate it disagrees with. It belongs to no
85    // declaration, so nothing in retention decides whether to keep it.
86    let guards = flat.guards().map(|guard| Writer.guard(guard));
87    let artifacts = artifacts.iter().map(|artifact| &artifact.rust);
88    let wrappers = functions
89        .iter()
90        .map(|function| wrapper(flat, target, reach, primitives, function));
91    let tokens = quote! {
92        #(#guards)*
93        #(#artifacts)*
94        #(#wrappers)*
95    };
96    match syn::parse2::<syn::File>(tokens.clone()) {
97        Ok(file) => prettyplease::unparse(&file),
98        // Unparseable output is a defect in this writer, and hiding it behind
99        // pretty printing would only move the compiler error somewhere less
100        // useful.
101        Err(_) => tokens.to_string(),
102    }
103}

105/// One exported function.
106fn wrapper<T: Target>(
107    flat: &Flat,
108    target: &T,
109    reach: &Reach,
110    primitives: &[PrimitiveSpec<T::Payload>],
111    function: &FunctionPlan<T::Payload>,
112) -> TokenStream {
113    let mut names: HashMap<ValueId, syn::Ident> = HashMap::new();
114    // Parameters are named by the boundary, because a target that requires an
115    // environment operand has to say what it is called. Everything else is
116    // named here, and around those names: a temporary that shadowed a live
117    // parameter would compile and read the wrong value.
118    // Reserved by their canonical spelling: `r#v0` and `v0` are one name to
119    // Rust, so reserving the raw form would leave the plain one free to shadow
120    // it.
121    let mut taken: std::collections::HashSet<String> = function
122        .params
123        .iter()
124        .map(|(_, p)| unraw(&p.name))
125        .collect();
126    let params: Vec<TokenStream> = function
127        .params
128        .iter()
129        .map(|(id, param)| {
130            names.insert(*id, param.name.clone());
131            let name = &param.name;
132            let ty = &param.ty.ty;
133            let mutable = param.mutable.then(|| quote!(mut));
134            quote!(#mutable #name: #ty)
135        })
136        .collect();
137    let ok_binding = free_name("value", &mut taken);
138    let error_binding = free_name("error", &mut taken);
139    // Temporaries are numbered by definition order, which is why the same
140    // operation used twice cannot collide with itself.
141    let mut next_local = 0;
142    let mut local = |names: &mut HashMap<ValueId, syn::Ident>,
143                     taken: &mut std::collections::HashSet<String>,
144                     id: ValueId| {
145        let name = loop {
146            let candidate = format!("v{next_local}");
147            next_local += 1;
148            if taken.insert(candidate.clone()) {
149                break format_ident!("{candidate}");
150            }
151        };
152        names.insert(id, name.clone());
153        name
154    };
155
156    // A wrapper names source items — the function it calls, and every struct it
157    // constructs — and compiles only where all of them exist. So it inherits
158    // each one's condition: a `#[cfg]` the capture reader could not answer and
159    // rewrote back onto the item. Rust conjoins repeated `#[cfg]` attributes on
160    // one item, so carrying them side by side settles the wrapper's own
161    // condition without anything here reading what they say. Two items carrying
162    // the same condition state it once, which conjunction makes cosmetic and
163    // review makes worth doing.
164    //
165    // The instructions are where a wrapper spells a source item: its signature
166    // carries target wire types only, and the one operation that spells a
167    // source type — a handle taken back or released, cast to
168    // `*mut source::Ledger` — is a standard one the registry renders, so it is
169    // found here too.
170    let mut conditions = Vec::new();
171    let mut conditioned: std::collections::HashSet<String> = std::collections::HashSet::new();
172    for step in &function.instrs {
173        let named = match &step.instr {
174            Instr::Construct { name, .. } => name.clone(),
175            Instr::Call { function, .. } => function.clone(),
176            Instr::Apply { primitive, .. } => match &primitives[primitive.0].implementation {
177                Operation::Standard(
178                    StandardOp::FromRaw { source } | StandardOp::Release { source },
179                ) => match source.kind() {
180                    prebindgen_flat::flat::TypeKind::Named { id, .. } => id.name.clone(),
181                    _ => continue,
182                },
183                _ => continue,
184            },
185        };
186        // Planning names only items the model declares.
187        let Some(element) = flat.element(named.as_str()) else {
188            continue;
189        };
190        for condition in Writer.conditions(Conditioned::Item(element)) {
191            if conditioned.insert(condition.to_string()) {
192                conditions.push(condition);
193            }
194        }
195    }
196
197    let mut statements = Vec::new();
198    for step in &function.instrs {
199        // What a field's condition holds the statement to: a member declared
200        // only sometimes is read only sometimes, and the initializer that
201        // consumes the read exists only then. The statement is one `let`, so
202        // the attribute goes in front of it.
203        let guarded = &step.conditions;
204        match &step.instr {
205            Instr::Apply {
206                primitive,
207                operands,
208                result,
209            } => {
210                let primitive = &primitives[primitive.0];
211                let operand_names: Vec<syn::Ident> = operands
212                    .iter()
213                    .map(|operand| operand_name(&names, function, operand))
214                    .collect();
215                let expression = operation(target, reach, primitive, &operand_names);
216                let statement = match (&primitive.failure, result) {
217                    (PrimitiveFailure::Infallible, Some(result)) => {
218                        let name = local(&mut names, &mut taken, *result);
219                        quote!(let #name = #expression;)
220                    }
221                    (PrimitiveFailure::Infallible, None) => quote!(#expression;),
222                    (PrimitiveFailure::Fallible { category, .. }, Some(result)) => {
223                        let name = local(&mut names, &mut taken, *result);
224                        let route = failure_arm(target, reach, function, *category, &error_binding);
225                        let ok = &ok_binding;
226                        let error = error_pattern(function, *category, &error_binding);
227                        quote! {
228                            let #name = match #expression {
229                                Ok(#ok) => #ok,
230                                Err(#error) => { #route }
231                            };
232                        }
233                    }
234                    (PrimitiveFailure::Fallible { category, .. }, None) => {
235                        let route = failure_arm(target, reach, function, *category, &error_binding);
236                        let error = error_pattern(function, *category, &error_binding);
237                        quote! {
238                            if let Err(#error) = #expression { #route }
239                        }
240                    }
241                };
242                statements.push(quote!(#(#guarded)* #statement));
243            }
244            Instr::Construct {
245                name,
246                parts,
247                result,
248            } => {
249                let item = flat
250                    .struct_type(name.as_str())
251                    .expect("a construction names a struct the model declares");
252                let ident = &item.name;
253                let module = reach.of(&ident.to_string());
254                let head = quote!(#module::#ident);
255                // An initializer carries its own field's condition: the value
256                // it names was bound by a statement under the same one, so a
257                // field the source does not have is neither read nor filled in.
258                //
259                // Read from the model here, while the statements got theirs
260                // from `Part::conditions` at plan time. The two agree because
261                // both read the same field; should a part's conditions ever
262                // become something a target contributes to, this has to read
263                // the part instead — an initializer written under a condition
264                // its read does not share names a value that is not there.
265                let bound: Vec<TokenStream> = item
266                    .fields
267                    .iter()
268                    .zip(parts)
269                    .map(|(field, value)| {
270                        let conditions = Writer.conditions(Conditioned::Field(field));
271                        let bound = field.bind(&names[value]);
272                        quote!(#(#conditions)* #bound)
273                    })
274                    .collect();
275                let value = Writer.shape_struct(item, head, &bound);
276                let name = local(&mut names, &mut taken, *result);
277                statements.push(quote!(#(#guarded)* let #name = #value;));
278            }
279            Instr::Call {
280                function: callee,
281                args,
282                result,
283            } => {
284                let module = reach.of(callee);
285                let callee = format_ident!("{callee}");
286                let args: Vec<syn::Ident> = args.iter().map(|value| names[value].clone()).collect();
287                let call = quote!(#module::#callee(#(#args),*));
288                // A source function returning nothing produces no value, and a
289                // `let` over it would name one nobody can use.
290                let statement = match result {
291                    Some(result) => {
292                        let name = local(&mut names, &mut taken, *result);
293                        quote!(let #name = #call;)
294                    }
295                    None => quote!(#call;),
296                };
297                statements.push(quote!(#(#guarded)* #statement));
298            }
299        }
300    }
301
302    let tail = match (&function.output, function.result) {
303        (OutputPlacement::Return, Some(result)) => {
304            let name = &names[&result];
305            quote!(#name)
306        }
307        _ => quote!(),
308    };
309    let symbol = format_ident!("{}", function.abi.symbol);
310    let abi = &function.abi.abi;
311    let attrs = &function.abi.attrs;
312    let unsafety = function.abi.unsafety.then(|| quote!(unsafe));
313    let ret = function
314        .abi
315        .ret
316        .as_ref()
317        .map(|wire| {
318            let ty = &wire.ty;
319            quote!(-> #ty)
320        })
321        .unwrap_or_default();
322    quote! {
323        #(#conditions)*
324        #[no_mangle]
325        #(#attrs)*
326        pub #unsafety extern #abi fn #symbol(#(#params),*) #ret {
327            #(#statements)*
328            #tail
329        }
330    }
331}

333/// One operation's expression: the registry renders its own, the target renders
334/// its own.
335fn operation<T: Target>(
336    target: &T,
337    reach: &Reach,
338    primitive: &PrimitiveSpec<T::Payload>,
339    operands: &[syn::Ident],
340) -> TokenStream {
341    // The handle operations spell a source type, which is what makes them the
342    // registry's: an adapter has no way to, and no business doing it.
343    let source_type = |ty| Writer.emit_source_type(ty, &reach.modules, &reach.default);
344    match &primitive.implementation {
345        Operation::Standard(StandardOp::Identity) => {
346            let value = &operands[0];
347            quote!(#value)
348        }
349        Operation::Standard(StandardOp::ReadMember { member }) => {
350            let value = &operands[0];
351            quote!(#value.#member)
352        }
353        Operation::Standard(StandardOp::IntoRaw { carrier }) => {
354            let value = &operands[0];
355            quote!(Box::into_raw(Box::new(#value)) as #carrier)
356        }
357        Operation::Standard(StandardOp::FromRaw { source }) => {
358            let value = &operands[0];
359            let ty = source_type(source);
360            let message = format!("null `{}` handle", source.key());
361            quote! {
362                ::core::ptr::NonNull::new(#value as *mut #ty)
363                    .map(|handle| unsafe { *Box::from_raw(handle.as_ptr()) })
364                    .ok_or_else(|| String::from(#message))
365            }
366        }
367        Operation::Standard(StandardOp::EnumOut { source, values }) => {
368            let value = &operands[0];
369            let ty = source_type(source);
370            let arms = values.iter().map(|arm| {
371                let pattern = enum_value(&ty, arm);
372                let carried = &arm.carried;
373                quote!(#pattern => #carried)
374            });
375            quote!(match #value { #(#arms),* })
376        }
377        Operation::Standard(StandardOp::EnumIn {
378            source,
379            values,
380            invalid,
381        }) => {
382            let value = &operands[0];
383            let ty = source_type(source);
384            let arms = values.iter().map(|arm| {
385                let carried = &arm.carried;
386                let constructed = enum_value(&ty, arm);
387                match invalid {
388                    Some(_) => quote!(#carried => ::core::result::Result::Ok(#constructed)),
389                    None => quote!(#carried => #constructed),
390                }
391            });
392            match invalid {
393                Some(message) => quote! {
394                    match #value {
395                        #(#arms,)*
396                        other => ::core::result::Result::Err(::std::format!(#message, other)),
397                    }
398                },
399                None => quote!(match #value { #(#arms),* }),
400            }
401        }
402        Operation::Standard(StandardOp::Release { source }) => {
403            let value = &operands[0];
404            let ty = source_type(source);
405            quote! {
406                drop(::core::ptr::NonNull::new(#value as *mut #ty)
407                    .map(|handle| unsafe { Box::from_raw(handle.as_ptr()) }))
408            }
409        }
410        Operation::Target(payload) => target.render_operation(payload, operands),
411    }
412}

414/// One value of a fieldless enum, spelled as the source declared it.
415///
416/// A pattern and a constructor are the same text for a fieldless value, so
417/// this serves both sides of the two enum operations. `Add` stays `Add`, and
418/// `Add()` and `Mul {}` keep their delimiters — the model's speller decides,
419/// from the shape it captured.
420fn enum_value(ty: &TokenStream, arm: &crate::target::EnumArm) -> TokenStream {
421    let name = &arm.name;
422    arm.shape.spell_fieldless(quote!(#ty::#name))
423}

425/// The pattern a failure arm binds the error with: the name when a reporter
426/// reads it, `_` when the route only terminates.
427fn error_pattern<P>(
428    function: &FunctionPlan<P>,
429    category: FailureCategory,
430    error: &syn::Ident,
431) -> TokenStream {
432    let reported = function
433        .failures
434        .iter()
435        .any(|route| route.category == category && route.report.is_some());
436    if reported {
437        quote!(#error)
438    } else {
439        quote!(_)
440    }
441}

443/// What one failure category does at this boundary: report if the target has an
444/// operation for it, then terminate.
445fn failure_arm<T: Target>(
446    target: &T,
447    reach: &Reach,
448    function: &FunctionPlan<T::Payload>,
449    category: FailureCategory,
450    error: &syn::Ident,
451) -> TokenStream {
452    let route = function
453        .failures
454        .iter()
455        .find(|route| route.category == category)
456        .expect("an unrouted failure category is refused before assembly");
457    let report = route.report.as_ref().map(|report| {
458        let operands: Vec<syn::Ident> = report
459            .operands
460            .iter()
461            .map(|operand| match &operand.role {
462                crate::target::OperandRole::Context(name) => context_name(function, name),
463                crate::target::OperandRole::Error => error.clone(),
464                // Refused when the boundary is assembled: a reporting operation
465                // is given the error and the contexts, and has no value to read.
466                crate::target::OperandRole::Value => {
467                    unreachable!("a reporting operation with a value operand is refused")
468                }
469            })
470            .collect();
471        let expression = operation(target, reach, report, &operands);
472        let on_failure = terminal(&route.on_report_failure);
473        match report.failure {
474            PrimitiveFailure::Infallible => quote!(#expression;),
475            PrimitiveFailure::Fallible { .. } => quote! {
476                if #expression.is_err() { #on_failure }
477            },
478        }
479    });
480    let terminate = terminal(&route.terminate);
481    quote! {
482        #report
483        #terminate
484    }
485}

487fn terminal(terminal: &Terminal) -> TokenStream {
488    match terminal {
489        Terminal::Abort => quote!(std::process::abort();),
490        Terminal::Return(expr) => quote!(return #expr;),
491    }
492}

494fn operand_name<P>(
495    names: &HashMap<ValueId, syn::Ident>,
496    function: &FunctionPlan<P>,
497    operand: &Operand,
498) -> syn::Ident {
499    match operand {
500        Operand::Value(id) => names[id].clone(),
501        Operand::Context(name) => context_name(function, name),
502    }
503}

505/// An identifier's canonical spelling: `r#type` and `type` name one binding.
506fn unraw(ident: &syn::Ident) -> String {
507    ident.to_string().trim_start_matches("r#").to_string()
508}

510/// A name nothing in this wrapper already uses.
511fn free_name(preferred: &str, taken: &mut std::collections::HashSet<String>) -> syn::Ident {
512    if taken.insert(preferred.to_string()) {
513        return format_ident!("{preferred}");
514    }
515    for suffix in 1.. {
516        let candidate = format!("{preferred}_{suffix}");
517        if taken.insert(candidate.clone()) {
518            return format_ident!("{candidate}");
519        }
520    }
521    unreachable!("the search above only ends by returning")
522}

524/// The wrapper parameter supplying a runtime context.
525fn context_name<P>(function: &FunctionPlan<P>, context: &str) -> syn::Ident {
526    function
527        .params
528        .iter()
529        .find(|(_, param)| matches!(&param.role, ParamRole::Context(name) if name == context))
530        .map(|(_, param)| param.name.clone())
531        .expect("a missing runtime context is refused before assembly")
532}
