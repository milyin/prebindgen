22/// The `Ok`/`Err` of a `Result`, spelled.
23///
24/// The model classifies a `Result` as [`TypeKind::Fallible`], so a reading
25/// answers this directly — no lookup, and no syntactic fallback.
26///
27/// The fallback there used to be (`types_util::result_parts` over the node)
28/// existed because the caller held only a **spelling**: a `Result` the adapter
29/// composed itself would have no entry in `flat`, so the reading lookup could
30/// miss. A caller holding a `TypeRef` cannot be in that position — #280 sealed
31/// minting to the model, so every reading reaching here was classified, and
32/// `kind` is what says whether it is a `Result`. The fallback was measured
33/// never to fire in-tree; it is now unreachable by construction.
34fn fallible_parts(
35    ty: &prebindgen_registry::flat::TypeRef,
36    emit: &prebindgen_registry::Emit,
37) -> Option<(syn::Type, syn::Type)> {
38    let (ok, err) = ty.fallible_parts()?;
39    let (ok, err) = (emit.spell(ok), emit.spell(err));
40    Some((syn::parse_quote!(#ok), syn::parse_quote!(#err)))
41}
