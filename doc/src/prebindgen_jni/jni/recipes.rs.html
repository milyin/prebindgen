1//! JniGen's Kotlin class declarations, lowered into the registry's recipe
2//! vocabulary.
3//!
4//! A build script writes `ptr_class!`, `data_class!`, `enum_class!` or
5//! `sealed_class!`, each naming the Kotlin type one Rust type becomes, and
6//! `convert!` for a conversion the binding supplies itself. This turns those
7//! into recipes in a [`Recipes`] table: the shared statement of **which parts** a
8//! value gets across in, with nothing about the JNI wire in it.
9//!
10//! One recipe per declared type and direction, and every crossing nobody declared
11//! takes
12//! the recipe the registry derives from its kind. The chain of guesses this
13//! replaced — try a terminal, then a `Result`, then an optional, then a run,
14//! then a borrow, then a transparent bridge — is a lookup for the declared
15//! half, and the derived half is the registry's.
16

29/// Every type one captured item names, so a reading nested inside it can be
30/// found without the caller knowing which kind of item it came from.
31fn element_types(element: &prebindgen_registry::Element) -> Vec<&TypeRef> {
32    use prebindgen_registry::flat::Type;
33    match element {
34        prebindgen_registry::Element::Function(f) => f
35            .params
36            .iter()
37            .map(|p| &p.ty)
38            .chain(std::iter::once(&f.ret))
39            .collect(),
40        prebindgen_registry::Element::Type(Type::Struct(s)) => {
41            s.fields.iter().map(|f| &f.ty).collect()
42        }
43        prebindgen_registry::Element::Type(Type::Variant(v)) => v
44            .alternatives
45            .iter()
46            .flat_map(|a| a.fields.iter().map(|f| &f.ty))
47            .collect(),
48        prebindgen_registry::Element::Constant(c) => vec![&c.ty],
49        _ => Vec::new(),
50    }
51}

53/// One arm per alternative of a declared sum, each assembled from its own
54/// payload fields.
55///
56/// Empty for anything the model does not hold as a data-carrying enum, which
57/// includes a `sealed_class!` declared over a type the scan dropped — the scan
58/// reports that, and a recipe over no arms would report it a second time.
59fn alternatives(model: &Flat, ty: &TypeRef) -> Vec<Arm<Construct>> {
60    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
61        return Vec::new();
62    };
63    let Some(prebindgen_registry::flat::Type::Variant(v)) =
64        id.ident().and_then(|ident| model.declared_type(&ident))
65    else {
66        return Vec::new();
67    };
68    v.alternatives
69        .iter()
70        .map(|alt| Arm {
71            alternative: alt.index,
72            op: Construct::Fields,
73        })
74        .collect()
75}
76
77/// One arm per alternative of a declared sum, each taken apart into its own
78/// payload fields.
79///
80/// The deconstructing twin of [`alternatives`]. Separate because the two directions
81/// name different operations — building an alternative is `Construct::Fields`,
82/// reading one is `Deconstruct::Fields` over the reaches that get there — and
83/// a shared helper would have to be generic over an operation neither side
84/// chooses.
85fn out_alternatives(model: &Flat, ty: &TypeRef) -> Vec<Arm<Deconstruct>> {
86    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
87        return Vec::new();
88    };
89    let Some(prebindgen_registry::flat::Type::Variant(v)) =
90        id.ident().and_then(|ident| model.declared_type(&ident))
91    else {
92        return Vec::new();
93    };
94    v.alternatives
95        .iter()
96        .map(|alt| Arm {
97            alternative: alt.index,
98            op: Deconstruct::Fields((0..alt.fields.len()).map(Reach::Field).collect()),
99        })
100        .collect()
101}

104    /// How many values this type hands out, or 0 if it hands out none.
105    ///
106    /// Model and declaration only, like the compositions it asks — so a
107    /// declaration-time caller may ask it, which is what binding a return site
108    /// needs.
109    fn out_wire_count(
110        &self,
111        registry: &impl prebindgen_registry::Conversions,
112        ty: &TypeRef,
113    ) -> usize {
114        let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
115            return 0;
116        };
117        let Some(ident) = id.ident() else { return 0 };
118        match registry.flat().declared_type(&ident) {
119            Some(prebindgen_registry::flat::Type::Variant(_)) => self
120                .sum_out_wires(registry.flat(), &ident, ty)
121                .map_or(0, |w| w.len()),
122            Some(prebindgen_registry::flat::Type::Struct(_)) => {
123                self.struct_out_wires(registry, ty).map_or(0, |w| w.len())
124            }
125            _ => 0,
126        }
127    }

129    /// The accessor a type's value form names, and one reach per field of the
130    /// struct it returns.
131    ///
132    /// `None` unless the declaration is exactly one `.fields(fields!(f))` whose
133    /// every field is a plain leaf. A record that states its own decomposition
134    /// — a per-field `expand_return!` override, an inlined nested class, a sum
135    /// laid out as a selector and its groups — is a shape this recipe does not
136    /// state yet, and stating half of one would describe a boundary nothing
137    /// emits.
138    fn value_form_of(
139        &self,
140        model: &Flat,
141        registry: &impl prebindgen_registry::Conversions,
142        ty: &TypeRef,
143    ) -> Option<(syn::Ident, Vec<Reach>)> {
144        use crate::unfold::{FieldDecon, FieldRecord};
145        let decl = self
146            .return_expand_decls
147            .iter()
148            .find(|d| *d.key() == ty.stripped_key())?;
149        let [crate::jni::LocalField::Fields(fields)] = decl.field_list() else {
150            return None;
151        };
152        let records: Vec<FieldRecord> = self.lower_value_form(registry, decl.key(), fields);
153        let ret = model.function(fields.func())?.ret.clone();
154        let ret = ret.borrow_target().unwrap_or(&ret);
155        let prebindgen_registry::flat::TypeKind::Named { id, .. } = ret.unwrapped().kind() else {
156            return None;
157        };
158        let prebindgen_registry::flat::Type::Struct(st) =
159            id.ident().and_then(|i| model.declared_type(&i))?
160        else {
161            return None;
162        };
163        let mut reaches = Vec::new();
164        for record in &records {
165            if !matches!(record.decon, FieldDecon::Default) {
166                return None;
167            }
168            // A record with several members is an INLINED nested class: each
169            // member is one hop of a field-access chain. `Reach::Path` states
170            // that; before it existed this row declined and fell back to a
171            // placeholder (#613 step 10).
172            let mut here = st;
173            let mut indices = Vec::with_capacity(record.members.len());
174            for member in &record.members {
175                let hop = here
176                    .fields
177                    .iter()
178                    .position(|f| f.name.as_ref() == Some(member))?;
179                indices.push(hop);
180                if indices.len() < record.members.len() {
181                    let next = here.fields[hop].ty.stripped_key().ident()?;
182                    let prebindgen_registry::flat::Type::Struct(next) =
183                        model.declared_type(&next)?
184                    else {
185                        return None;
186                    };
187                    here = next;
188                }
189            }
190            let [index] = indices.as_slice() else {
191                reaches.push(Reach::Path(indices));
192                continue;
193            };
194            let index = *index;
195            // A field that reaches the type being taken apart. The leaf
196            // synthesis treats `Box<T>` as a spelling of its own and stops
197            // there; a recipe keys a crossing by the value that crosses, so
198            // `Box<ZSample>` inside `ZSample`'s own value form is that recipe
199            // reaching itself. Refused here rather than by `Recipes::build`,
200            // which would refuse the whole binding over a shape the leaf
201            // synthesis handles.
202            if st.fields[index].ty.stripped_key() == ty.stripped_key() {
203                return None;
204            }
205            reaches.push(Reach::Field(index));
206        }
207        Some((fields.func().clone(), reaches))
208    }
209}

212    /// What a value form calls each of its fields, by the Rust field ident.
213    ///
214    /// The declaration's answer, so a `.name(..)` rename carries through to the
215    /// builder parameter. `None` for a type with no value form, or one whose
216    /// records this recipe does not state — the same test [`Self::value_form_of`]
217    /// makes when it declares the recipe.
218    pub(crate) fn value_form_names(
219        &self,
220        registry: &impl prebindgen_registry::Conversions,
221        ty: &TypeRef,
222    ) -> Option<std::collections::HashMap<String, String>> {
223        use crate::unfold::FieldDecon;
224        let decl = self
225            .return_expand_decls
226            .iter()
227            .find(|d| *d.key() == ty.stripped_key())?;
228        let [crate::jni::LocalField::Fields(fields)] = decl.field_list() else {
229            return None;
230        };
231        self.lower_value_form(registry, decl.key(), fields)
232            .into_iter()
233            .map(|r| match (r.members.as_slice(), &r.decon) {
234                ([member], FieldDecon::Default) => Some((member.to_string(), r.name)),
235                _ => None,
236            })
237            .collect()
238    }
239}

241/// One reach per field of a declared struct, in the model's order.
242///
243/// Empty for anything the model does not hold as a struct, which includes a
244/// `data_class!` over a type the scan dropped — the scan reports that, and a
245/// recipe over no fields would report it a second time.
246fn out_fields(model: &Flat, ty: &TypeRef) -> Vec<Reach> {
247    (0..fields_of(model, ty).len()).map(Reach::Field).collect()
248}

250/// The fields of a declared struct, or none for anything the model does not
251/// hold as one.
252fn fields_of<'a>(model: &'a Flat, ty: &TypeRef) -> &'a [prebindgen_registry::flat::Field] {
253    let prebindgen_registry::flat::TypeKind::Named { id, .. } = ty.unwrapped().kind() else {
254        return &[];
255    };
256    match id.ident().and_then(|ident| model.declared_type(&ident)) {
257        Some(prebindgen_registry::flat::Type::Struct(s)) => &s.fields,
258        _ => &[],
259    }
260}

283impl Declarations {
284    /// Every recipe this binding's declarations state.
285    ///
286    /// A type declared but absent from the model is skipped rather than
287    /// refused: the scan already reports it, and reporting it twice in
288    /// different words helps nobody.
289    /// This type's `parts` decomposition, read off its `expand_return!`
290    /// declaration in the recipe table's own vocabulary.
291    ///
292    /// `None` when the declaration states a shape a `Deconstruct` cannot hold —
293    /// a value-form record mixed with others, since `ValueForm` is the whole
294    /// row while `LocalField::Fields` is one record among many. No declaration
295    /// in this workspace has that shape (measured: every `Fields` record stands
296    /// alone), so the caller's `Atomic` fallback is a guard rather than a path
297    /// anything takes.
298    fn parts_deconstruct(
299        &self,
300        model: &Flat,
301        registry: &impl prebindgen_registry::Conversions,
302        ty: &TypeRef,
303    ) -> Option<Deconstruct> {
304        let decl = self
305            .return_expand_decls
306            .iter()
307            .find(|d| *d.key() == ty.stripped_key())?;
308        let fields = decl.field_list();
309        if matches!(fields, [crate::jni::LocalField::Fields(_)]) {
310            // `ValueForm` means "call this accessor, then read these parts off
311            // its result" — the parts are how the compiler reaches the returned
312            // struct's fields, so an empty list is a decomposition with nothing
313            // in it rather than one of everything (#638 review).
314            //
315            // `value_form_of` already computes exactly that mapping, and it is
316            // not a count: `lower_value_form` returns FLATTENED records, one of
317            // which can splice a child's fields, so `Report` yields six records
318            // over a five-field struct. It maps each record back to the struct
319            // field it names and declines the shapes a row cannot hold.
320            let (func, parts) = self.value_form_of(model, registry, ty)?;
321            return Some(Deconstruct::ValueForm { func, parts });
322        }
323        let mut reaches = Vec::with_capacity(fields.len());
324        for field in fields {
325            reaches.push(match field {
326                crate::jni::LocalField::Named(func, _) => Reach::Accessor(func.clone()),
327                // The handle itself — the leaf `Reach::Identity` was added for.
328                crate::jni::LocalField::SelfField => Reach::Identity,
329                // A binding-local accessor resolves by its path's last segment:
330                // `local_functions()` registers it under exactly that ident.
331                crate::jni::LocalField::Local { path, .. } => {
332                    Reach::Accessor(path.segments.last()?.ident.clone())
333                }
334                crate::jni::LocalField::Fields(_) => return None,
335            });
336        }
337        Some(Deconstruct::Fields(reaches))
338    }

340    pub(crate) fn recipes(
341        &self,
342        model: &Flat,
343        expansion_leaves: &[TypeRef],
344        registry: &impl prebindgen_registry::Conversions,
345    ) -> Result<Recipes, Vec<RecipeError>> {
346        let mut recipes = Recipes::builder();
347        // Which crossings already state a deconstructing `parts` row, so the
348        // decomposition block below adds one only where none was declared.
349        let mut parts_out: std::collections::HashSet<TypeKey> = std::collections::HashSet::new();
350        // Every Kotlin class declaration, and every `convert!`-declared
351        // conversion. The second matters as much as the first: a conversion may
352        // be declared on a type the registry would otherwise read as an arity
353        // layer, and `convert!(Option<Duration> => ..)` means the adapter emits
354        // that optional's conversion itself rather than wrapping `Duration`'s.
355        let mut declared: Vec<(TypeKey, Origin<syn::Type>)> = self
356            .declared_types()
357            .into_iter()
358            .chain(
359                self.convert_decls
360                    .iter()
361                    .map(|d| (d.key().clone(), d.rust_type().clone())),
362            )
363            .collect();
364        declared.sort_by(|a, b| a.0.as_str().cmp(b.0.as_str()));
365        declared.dedup_by(|a, b| a.0 == b.0);
366
367        for (_key, origin) in declared {
368            // The declarator's own tokens, re-parsed: a declared type may be an
369            // alias the model holds as an `Extern`, which carries no reading of
370            // its own to borrow.
371            let Ok(spelled) = syn::parse2::<syn::Type>(origin.declared_spelling()) else {
372                continue;
373            };
374            let Ok(ty) = model.classify(&spelled) else {
375                continue;
376            };
377            // Every declared Kotlin shape is one recipe with no parts today,
378            // including the two that plainly have some: a `data_class` arrives
379            // as separate JNI parameters and is rebuilt inside one generated
380            // function, and a `sealed_class` is a selector plus the live arm's
381            // slots inside another. `Atomic` is what a recipe says about that —
382            // the adapter emits the conversion itself, and how many wire values
383            // that costs is its own business — so it describes the adapter as it
384            // stands rather than as it will be.
385            // A `sealed_class` hands its value out as a tag plus every
386            // alternative's payloads, laid side by side — the deconstructing
387            // twin of the constructing recipe below, and the direction that
388            // actually has no alternative: a sum has no whole-value output
389            // conversion, because a tag plus groups is not one wire.
390            match self.types.get(&ty.key()).map(|c| &c.kind) {
391                Some(DeclaredKind::Sealed(_)) => {
392                    let arms = out_alternatives(model, &ty);
393                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
394                    if !arms.is_empty() {
395                        recipes.declare(ty.clone(), parts(), Deconstructing::Choice { arms });
396                        parts_out.insert(ty.stripped_key());
397                    }
398                }
399                // A `data_class` hands its value out as its fields too, so the
400                // foreign side reassembles the object and nothing builds one on
401                // the Rust side.
402                Some(DeclaredKind::Data) => {
403                    let reaches = out_fields(model, &ty);
404                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
405                    if !reaches.is_empty() {
406                        recipes.declare(
407                            ty.clone(),
408                            parts(),
409                            Deconstructing::Product(Deconstruct::Fields(reaches)),
410                        );
411                        parts_out.insert(ty.stripped_key());
412                    }
413                }
414                _ => {
415                    recipes.declare_default(ty.clone(), whole(), Deconstructing::Atomic);
416                }
417            }
418            // A value form: `expand_return!(T).fields(fields!(f))` calls `f`
419            // once and hands out the fields of the struct it returns. The one
420            // declaration shape that names its own accessor, and the reason
421            // `Deconstruct::ValueForm` exists.
422            if let Some((func, reaches)) = self.value_form_of(model, registry, &ty) {
423                recipes.declare(
424                    ty.clone(),
425                    parts(),
426                    Deconstructing::Product(Deconstruct::ValueForm {
427                        func,
428                        parts: reaches,
429                    }),
430                );
431                parts_out.insert(ty.stripped_key());
432            }
433            // A `data_class` also has a recipe that says what it is made of, so
434            // its constructing side names which of the two a site takes by
435            // default. See [`parts`] for why that is still `whole`.
436            match self.types.get(&ty.key()).map(|c| &c.kind) {
437                // A struct with no fields is nothing to be made of, so it
438                // states no `parts` recipe — the same condition the deconstructing
439                // side applies, and the reason `recipe_of` may be asked for the
440                // recipe by name.
441                Some(DeclaredKind::Data) if !fields_of(model, &ty).is_empty() => {
442                    recipes
443                        .declare_default(ty.clone(), whole(), Constructing::Atomic)
444                        .declare(ty, parts(), Constructing::Product(Construct::Fields));
445                }
446                // A `sealed_class` has one too, and it is a choice rather than
447                // a product: exactly one alternative is live, every one of them
448                // still crosses, and the tag says which. The arms are the
449                // model's — a recipe states which parts, never what they are.
450                Some(DeclaredKind::Sealed(_)) => {
451                    let arms = alternatives(model, &ty);
452                    recipes.declare_default(ty.clone(), whole(), Constructing::Atomic);
453                    if !arms.is_empty() {
454                        recipes.declare(ty, parts(), Constructing::Choice { arms });
455                    }
456                }
457                _ => {
458                    recipes.declare(ty, whole(), Constructing::Atomic);
459                }
460            }
461        }
462
463        // A type a callback delivers by taking it apart states that as a row,
464        // so the argument's site has a recipe to name. Without one the site
465        // could only be fabricated — a `Bound` no binding answered — which is
466        // what #622's first two attempts produced and why they were withdrawn.
467        //
468        // `Atomic` for the reason the declared-type loop above gives for the
469        // two shapes that plainly have parts: the adapter emits this
470        // conversion itself, and how many wire values that costs is its own
471        // business. The parts are stated by the deconstructor declaration the
472        // registry already resolved into an `UnfoldPlan`, which this row does
473        // not restate — it names the crossing that plan decomposes. Deleting
474        // that second statement is #613 step 5b's, not this row's.
475        //
476        // The source is the plan's own owned core, so a plan keyed under `&T`
477        // and one keyed under `T` state one row; a type that already declared
478        // `parts` — a `data_class`, a `sealed_class`, a value form — keeps it.
479        //
480        // A **whole-element fold** is not one of these and must not earn a row.
481        // `apply_leaf_vec_folds` files a plan for `impl Fn(&[T])` under `&[T]`
482        // whose `source` is the ELEMENT and whose `decon` is `None`: nothing is
483        // taken apart, each element crosses whole through its own converter.
484        // Declaring `parts` off it would state a decomposition of `T` that does
485        // not exist — and, worse, a scalar `T` argument elsewhere in the same
486        // model would then bind to that row and name a fragment compiled under
487        // a different recipe (#623 review). `decon` is the gate the model
488        // already carries for this: `None` only for the whole-element arm.
489        let mut decomposed: BTreeMap<TypeKey, TypeRef> = BTreeMap::new();
490        for plan in self.unfolded().callback_arg_plans.values() {
491            if plan.decon.is_none() {
492                continue;
493            }
494            let key = plan.source.stripped_key();
495            if parts_out.contains(&key) {
496                continue;
497            }
498            decomposed.entry(key).or_insert_with(|| plan.source.clone());
499        }
500        for ty in decomposed.into_values() {
501            // A real row where the declaration can state one. #622 wrote
502            // `Atomic` here because `Reach` could not spell an identity leaf;
503            // it can now (#635), so the row says how the value comes apart
504            // instead of only existing to be selected (#613 step 10).
505            match self.parts_deconstruct(model, registry, &ty) {
506                Some(deconstruct) => {
507                    recipes.declare(ty, parts(), Deconstructing::Product(deconstruct));
508                }
509                None => {
510                    recipes.declare(ty, parts(), Deconstructing::Atomic);
511                }
512            }
513        }
514
515        // Every implicit Optional keeps its established whole/default input
516        // row and also offers the allocation-free primitive `pair` row. A
517        // parameter site selects `pair` only after the compiled payload proves
518        // that it is one JNI primitive with no niche, projection, or stage.
519        //
520        // Product-shaped payloads additionally offer `parts`; that row is
521        // selected by the existing flattening bindings below.
522        let mut implicit: BTreeMap<TypeKey, (TypeRef, TypeRef)> = self
523            .implicit_optionals(model)
524            .into_iter()
525            .map(|(key, (outer, inner))| (key, (outer.clone(), inner.clone())))
526            .collect();
527        // A combined constructor expansion represents an inactive arm by
528        // Option-wrapping each required leaf. Those readings are synthesized
529        // after the Flat model was built, so a model walk cannot discover
530        // them. File them in the same table as source-written implicit
531        // Optionals: otherwise a nullable primitive leaf falls back to a boxed
532        // JObject and Rust has to call `intValue()`/`longValue()` across JNI.
533        for leaf in expansion_leaves {
534            let Some(inner) = leaf.optional_inner() else {
535                continue;
536            };
537            if !self.explicitly_declares(leaf) {
538                implicit
539                    .entry(leaf.stripped_key())
540                    .or_insert_with(|| (leaf.clone(), inner.clone()));
541            }
542        }
543        for (outer, _) in implicit.values() {
544            recipes
545                .declare_default(outer.clone(), whole(), Constructing::Optional)
546                .declare(outer.clone(), pair(), Constructing::Optional)
547                .declare(outer.clone(), parts(), Constructing::Optional);
548        }
549        for (outer, _) in implicit.into_values() {
550            recipes
551                .declare_default(outer.clone(), whole(), Deconstructing::Optional)
552                .declare(outer.clone(), parts(), Deconstructing::Optional);
553        }
554
555        // A fixed-size array of JNI primitives is one Kotlin `ByteArray` or
556        // `LongArray`, bulk-copied with nothing boxed — one wire value, not a
557        // run this adapter walks. The registry reads `[T; N]` as a run unless
558        // something says otherwise, and this is JniGen saying otherwise for
559        // every array the model holds. Nothing is enumerated twice: a recipe for
560        // a crossing nobody uses is inert.
561        // Keyed by identity in a `BTreeMap`, so each array type is keyed once
562        // and the declaration order is the key order — no sort, no second pass
563        // rebuilding a key it already has, and nothing cloned before the entry
564        // turns out to be new.
565        let mut arrays: BTreeMap<TypeKey, &TypeRef> = BTreeMap::new();
566        for ty in model
567            .elements()
568            .flat_map(element_types)
569            .flat_map(|ty| ty.walk())
570            .filter(|ty| matches!(ty.kind(), prebindgen_registry::flat::TypeKind::Array { .. }))
571        {
572            arrays.entry(ty.key()).or_insert(ty);
573        }
574        for ty in arrays.into_values() {
575            recipes
576                .declare(ty.clone(), whole(), Deconstructing::Atomic)
577                .declare(ty.clone(), whole(), Constructing::Atomic);
578        }
579
580        recipes.build(model)
581    }
582}

584impl Declarations {
585    /// Which recipe each part of a `data_class` takes.
586    ///
587    /// A field that is itself a `data_class` crosses as **its** fields too, so
588    /// the part site asks for [`parts`] rather than letting the field take its
589    /// own default. That is the one thing a site can say that a recipe cannot: the
590    /// same crossing is read one way on its own and another inside a product.
591    ///
592    /// Without it a nested class contributes a single wire and the flattening
593    /// stops one layer down.
594    /// Whether a `data_class` field is itself flattened, or stays one value.
595    ///
596    /// A `data_class` declared `.jobject_input()` crosses Kotlin → Rust as a
597    /// single `JObject` — that is the whole point of the opt-in — so binding
598    /// its parts would flatten a boundary the declaration drew.
599    fn field_crosses_as_its_fields(&self, ty: &TypeRef) -> bool {
600        // The field's own key, not the stripped one: a `Box<Leaf>` field
601        // crosses as one value through `Box<Leaf>`'s conversion, and only a
602        // parameter spelled that way is peeled back to the class. Binding the
603        // stripped key here would flatten a field the rebuild has no node for.
604        self.types.get(&ty.key()).is_some_and(|c| match c.kind {
605            DeclaredKind::Data => !c.jobject_input,
606            // A `sealed_class` field crosses as a tag plus every alternative's
607            // slots. Whether it *can* is the adapter's answer at compile time,
608            // not a declaration: a payload with no slot form leaves the sum
609            // object-shaped, which is the fragment `Compile::choice` hands
610            // back. So the site asks, and the recipe answers.
611            DeclaredKind::Sealed(_) => true,
612            _ => false,
613        })
614    }
615
616    /// Optional crossings whose payload can use the shared `parts` recipe.
617    ///
618    /// Explicit declarations keep full control of the outer crossing: a
619    /// `convert!(Option<T> => ..)` is an atomic recipe, not an implicit request
620    /// to flatten `T`.
621    fn implicit_optionals<'a>(
622        &self,
623        model: &'a Flat,
624    ) -> BTreeMap<TypeKey, (&'a TypeRef, &'a TypeRef)> {
625        let mut optionals = BTreeMap::new();
626        for ty in model
627            .elements()
628            .flat_map(element_types)
629            .flat_map(|ty| ty.walk())
630        {
631            let Some(inner) = ty.optional_inner() else {
632                continue;
633            };
634            if !self.explicitly_declares(ty) {
635                optionals.entry(ty.stripped_key()).or_insert((ty, inner));
636            }
637        }
638        optionals
639    }
640
641    fn explicitly_declares(&self, ty: &TypeRef) -> bool {
642        self.types.contains_key(&ty.key())
643            || self
644                .convert_decls
645                .iter()
646                .any(|decl| decl.key() == &ty.key() || decl.key() == &ty.stripped_key())
647    }
648
649    pub(crate) fn bindings(
650        &self,
651        model: &Flat,
652        registry: &impl prebindgen_registry::Conversions,
653        recipes: &prebindgen_registry::recipe::Recipes,
654    ) -> Result<prebindgen_registry::recipe::Bindings, Vec<RecipeError>> {
655        use prebindgen_registry::recipe::{
656            Ask, Bindings, Crossing, Direction, Origin, RecipeName, Site,
657        };
658
659        let mut bound = Bindings::builder();
660        let mut declared: Vec<TypeKey> = self
661            .types
662            .iter()
663            .filter(|(_, c)| matches!(c.kind, DeclaredKind::Data))
664            .map(|(k, _)| k.clone())
665            .collect();
666        declared.sort_by(|a, b| a.as_str().cmp(b.as_str()));
667
668        // The named Optional row is the representation suitable inside another
669        // composed shape. Where its child has a Product row it delegates to that
670        // row; a scalar child stays on its default row and lets the JNI compiler
671        // choose a niche or a `(present, value)` intermediate from the compiled
672        // child facts. The whole/default Optional row remains untouched.
673        //
674        // Bind both directions: function inputs construct the Rust optional,
675        // while returns and callback arguments deconstruct it.
676        for (outer, inner) in self.implicit_optionals(model).into_values() {
677            for direction in [Direction::Construct, Direction::Deconstruct] {
678                let outer = Crossing::new(outer.clone(), direction);
679                let row = outer.row(parts());
680                let inner = Crossing::new(inner.clone(), direction);
681                bound.bind(
682                    Site::part(&row, 0),
683                    inner.clone(),
684                    if recipes.key_of(&inner.key(), &parts()).is_some() {
685                        Ask::Recipe(parts())
686                    } else {
687                        Ask::Default
688                    },
689                    Origin::Part,
690                );
691            }
692        }
693
694        for key in declared {
695            let Some(ident) = key.ident() else { continue };
696            let Some(prebindgen_registry::flat::Type::Struct(s)) = model.declared_type(&ident)
697            else {
698                continue;
699            };
700            let Ok(ty) = model.classify(&syn::parse_quote!(#ident)) else {
701                continue;
702            };
703            let building = Crossing::new(ty.clone(), Direction::Construct);
704            let handing_out = Crossing::new(ty, Direction::Deconstruct);
705            for (index, field) in s.fields.iter().enumerate() {
706                // A field selects `parts` exactly when its crossing declares
707                // that row AND the declaration says this occurrence decomposes.
708                // Direct classes retain the existing `.jobject_input()` and
709                // erased-wrapper boundaries; every implicit Optional selects its
710                // part representation, which decides whether the child is
711                // another composition or one scalar intermediate.
712                if field.ty.optional_inner().is_none()
713                    && !self.field_crosses_as_its_fields(&field.ty)
714                {
715                    continue;
716                }
717                for (of, direction) in [
718                    (&building, Direction::Construct),
719                    (&handing_out, Direction::Deconstruct),
720                ] {
721                    let row = of.row(parts());
722                    let field_crossing = Crossing::new(field.ty.clone(), direction);
723                    if recipes.key_of(&field_crossing.key(), &parts()).is_none() {
724                        continue;
725                    }
726                    bound.bind(
727                        Site::part(&row, index),
728                        field_crossing,
729                        Ask::Recipe(parts()),
730                        Origin::Part,
731                    );
732                }
733            }
734        }
735        // Every function whose return the binding takes apart, bound to the
736        // `parts` recipe of its return type.
737        //
738        // This is what a site naming several values is: `Ask::Recipe` already
739        // lets a site pick a recipe, and a `parts` fragment already occupies
740        // several wires — the two facts stages 3 and 4 established, meeting.
741        // The registry needs nothing new to express a decomposed return.
742        for f in model.functions() {
743            // The value that crosses, through the layers a decomposition looks
744            // through: a `&T` return decomposes T, and so does an `Option<T>`
745            // or a `Vec<T>` — the shape rides the delivery, not the recipe.
746            let ret = f.ret.borrow_target().unwrap_or(&f.ret);
747            let ret = ret.optional_inner().unwrap_or(ret);
748            let ret = ret.sequence_elem().unwrap_or(ret);
749            let ret = ret.borrow_target().unwrap_or(ret);
750            let crossing = Crossing::new(ret.clone(), Direction::Deconstruct);
751            // Only where the type states one. A `sealed_class` always does; a
752            // `data_class` states one unless a field of it declines, and a
753            // return whose type states none crosses whole.
754            if recipes.key_of(&crossing.key(), &parts()).is_none() {
755                continue;
756            }
757            // And only where it is genuinely several. A decomposition that
758            // yields ONE value takes `Delivery::Return`: the wrapper hands that
759            // value back through its own conversion rather than through a
760            // builder, so the return site's crossing is the value's, not this
761            // type's. Asked of the composition, which is model and declaration
762            // only and so answerable here — the same property that let one
763            // composition serve both sides of `resolve`.
764            if self.out_wire_count(registry, ret) < 2 {
765                continue;
766            }
767            bound.bind(
768                Site {
769                    owner: f.name.clone(),
770                    role: prebindgen_registry::recipe::Role::Return,
771                },
772                crossing,
773                Ask::Recipe(parts()),
774                Origin::Adapter,
775            );
776        }
777
778        // A callback's `Invoke` recipe owns its argument conversions. Bind a
779        // model-derived data-class argument to `parts` so the Product fragment
780        // reaches `Compile::callback` before the trampoline is rendered.
781        for f in model.functions() {
782            for param in &f.params {
783                let prebindgen_registry::flat::TypeKind::Callback { args } =
784                    param.ty.unwrapped().kind()
785                else {
786                    continue;
787                };
788                let callback = Crossing::new(param.ty.clone(), Direction::Construct);
789                let row = callback.row(RecipeName::derived());
790                for (index, arg) in args.iter().enumerate() {
791                    let crossed = arg.borrow_target().unwrap_or(arg);
792                    let core = crossed.optional_inner().unwrap_or(crossed);
793                    if let Some(element) = core.sequence_elem() {
794                        let element = element.borrow_target().unwrap_or(element);
795                        if !self.field_crosses_as_its_fields(element) {
796                            continue;
797                        }
798                        let sequence = Crossing::new(arg.clone(), Direction::Deconstruct);
799                        let element_crossing =
800                            Crossing::new(element.borrowed(), Direction::Deconstruct);
801                        if recipes.key_of(&element_crossing.key(), &parts()).is_none() {
802                            continue;
803                        }
804                        bound.bind(
805                            Site::part(&sequence.row(RecipeName::derived()), 0),
806                            element_crossing,
807                            Ask::Recipe(parts()),
808                            Origin::Adapter,
809                        );
810                        continue;
811                    }
812                    // A model-derived data class, or a type whose deconstructor
813                    // declaration earned the `parts` row above. Both take that
814                    // row here, so the fragment the trampoline delivers and the
815                    // one the argument's own `Role::CallbackArg` site names are
816                    // one fragment rather than two rows over one crossing.
817                    if !self.field_crosses_as_its_fields(core)
818                        && !self.unfolded().callback_arg_plans.contains_key(&arg.key())
819                    {
820                        continue;
821                    }
822                    let crossing = Crossing::new(arg.clone(), Direction::Deconstruct);
823                    if recipes.key_of(&crossing.key(), &parts()).is_none() {
824                        continue;
825                    }
826                    bound.bind(
827                        Site::part(&row, index),
828                        crossing,
829                        Ask::Recipe(parts()),
830                        Origin::Adapter,
831                    );
832                }
833            }
834        }
835
836        // Each value a callback delivers is a root site of its own — the
837        // function-unique `Role::CallbackArg` the registry names, not the
838        // `Role::Part` above, which is keyed by the callback recipe every
839        // function with that signature shares.
840        //
841        // Bound to `parts` exactly where the argument's crossing states that
842        // row, which is now every argument the trampoline takes apart: a
843        // `data_class`, a `sealed_class`, an implicit Optional over one of
844        // those, or a type whose deconstructor declaration earned the row
845        // declared above. An argument that crosses whole names nothing here and
846        // takes its crossing's default, attributed to the adapter.
847        for f in model.functions() {
848            for (param, p) in f.params.iter().enumerate() {
849                let prebindgen_registry::flat::TypeKind::Callback { args } =
850                    p.ty.unwrapped().kind()
851                else {
852                    continue;
853                };
854                for (arg, ty) in args.iter().enumerate() {
855                    let crossing = Crossing::new(ty.clone(), Direction::Deconstruct);
856                    if recipes.key_of(&crossing.key(), &parts()).is_none() {
857                        continue;
858                    }
859                    bound.bind(
860                        Site {
861                            owner: f.name.clone(),
862                            role: prebindgen_registry::recipe::Role::CallbackArg { param, arg },
863                        },
864                        crossing,
865                        Ask::Recipe(parts()),
866                        Origin::Adapter,
867                    );
868                }
869            }
870        }
871
872        bound.build(recipes)
873    }
