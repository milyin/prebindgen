1//! Fixed-size arrays (`[T; N]`) of JNI-primitive elements, crossing as the
2//! matching Kotlin primitive array.
3//!
4//! `[u8; 16]` ⇄ `ByteArray`, `[i64; 4]` ⇄ `LongArray`, and so on for every
5//! [`JniPrim`](super::prim::JniPrim) scalar. Primitive arrays bulk-copy through
6//! `set_*_array_region` / `get_*_array_region` and box nothing, which is why a
7//! fixed-size array does NOT go through the `Vec<T>` → `List<T>` path.
8//!
9//! Wider unsigned elements (`[u16; N]`, `[u32; N]`, `[u64; N]`) carry the **raw
10//! bit pattern** in the signed array. That matches the existing scalar rule — a
11//! `u64` already crosses as a raw `jlong` — and `Vec<u8>` → `ByteArray`, where
12//! Kotlin's `Byte` is signed. Kotlin's own `UByteArray`/`ULongArray` are
13//! `@ExperimentalUnsignedTypes`, so using them would push an opt-in onto every
14//! consumer of a shared binding tier.
15//!
16//! **`N` is never needed at generation time.** It is often a const *path* rather
17//! than a literal (`ZenohId` is `[u8; ZENOH_ID_MAX_SIZE]`), so the decode leans
18//! on `TryFrom<&[T]> for [T; N]` and lets `rustc` infer the length; a JVM array
19//! of the wrong length becomes a binding error rather than a panic.
20//!
21//! Element conversion is by `as`-cast (or `!= 0` for `bool`), never a transmute:
22//! `jboolean` is a `u8`, and reinterpreting a byte of `2` as a Rust `bool` would
23//! be UB — the very hazard that retired the raw-memory value blob this module
24//! replaces.

30/// The JNI/Kotlin array pair for one primitive element type.
31#[derive(Clone)]
32pub(crate) struct PrimArray {
33    /// Wire type: `jni::objects::JLongArray`.
34    pub wire: syn::Type,
35    /// JNI element type: `jni::sys::jlong`.
36    pub elem_wire: syn::Type,
37    /// Kotlin surface: `LongArray`.
38    pub kotlin: KtType,
39    /// `JNIEnv::new_long_array`.
40    pub new_fn: syn::Ident,
41    /// `JNIEnv::set_long_array_region`.
42    pub set_region: syn::Ident,
43    /// `JNIEnv::get_long_array_region`.
44    pub get_region: syn::Ident,
45    /// True for `[bool; N]` — its element needs normalizing rather than casting.
46    pub is_bool: bool,
47    /// True for `[u8; N]` — the one case with a dedicated bulk helper.
48    pub is_u8: bool,
49}

51/// Classify `ty` as a fixed-size array of JNI-primitive elements.
52///
53/// `None` for everything else, including `[T; N]` of a declared class or enum —
54/// those keep resolving as unsupported, so an unhandled shape is a clear
55/// resolve error rather than silently wrong code.
56pub(crate) fn prim_array_of(ty: &prebindgen_registry::flat::TypeRef) -> Option<PrimArray> {
57    use prebindgen_registry::flat::{ScalarKind, TypeKind};
58    let TypeKind::Array { elem, .. } = ty.kind() else {
59        return None;
60    };
61    let &TypeKind::Scalar(elem) = elem.kind() else {
62        return None;
63    };
64    // `usize`/`isize` are deliberately absent: their width is platform
65    // dependent, so there is no stable JNI element type to pick. Reaching them
66    // by name is the ScalarKind's own spelling, so the set below is still the
67    // set Rust writes — a kind that is not on it falls through as before.
68    let (letter, jni_elem) = match elem.as_str() {
69        "u8" | "i8" => ("byte", "jbyte"),
70        "u16" | "i16" => ("short", "jshort"),
71        "u32" | "i32" => ("int", "jint"),
72        "u64" | "i64" => ("long", "jlong"),
73        "f32" => ("float", "jfloat"),
74        "f64" => ("double", "jdouble"),
75        "bool" => ("boolean", "jboolean"),
76        _ => return None,
77    };
78    let cap = format!("{}{}", letter[..1].to_uppercase(), &letter[1..]);
79    let wire_ident = format_ident!("J{}Array", cap);
80    let elem_wire_ident = format_ident!("{}", jni_elem);
81    Some(PrimArray {
82        wire: syn::parse_quote!(jni::objects::#wire_ident),
83        elem_wire: syn::parse_quote!(jni::sys::#elem_wire_ident),
84        kotlin: KtType::cls(format!("{cap}Array")),
85        new_fn: format_ident!("new_{}_array", letter),
86        set_region: format_ident!("set_{}_array_region", letter),
87        get_region: format_ident!("get_{}_array_region", letter),
88        is_bool: elem == ScalarKind::Bool,
89        is_u8: elem == ScalarKind::U8,
90    })
91}

93/// `[T; N]` → the Kotlin primitive array (Rust → wire).
94pub(crate) fn output_body(spec: &PrimArray) -> syn::Expr {
95    if spec.is_u8 {
96        // `&[u8; N]` derefs to `&[u8]`, so the dedicated helper applies with no
97        // intermediate buffer — the common case (`ZenohId`).
98        return syn::parse_quote!({
99            env.byte_array_from_slice(&v).map_err(|e| {
100                <__JniErr as ::core::convert::From<String>>::from(format!(
101                    "fixed-size array encode: {}",
102                    e
103                ))
104            })?
105        });
106    }
107    let elem_wire = &spec.elem_wire;
108    let new_fn = &spec.new_fn;
109    let set_region = &spec.set_region;
110    // One form for every element: `bool as u8` yields 0/1, and the rest are
111    // same-width numeric casts. Only the DECODE needs a special case, because
112    // `u8 as bool` is not a cast at all.
113    let to_wire: syn::Expr = syn::parse_quote!(*__x as #elem_wire);
114    syn::parse_quote!({
115        let __buf: ::std::vec::Vec<#elem_wire> = v.iter().map(|__x| #to_wire).collect();
116        let __arr = env.#new_fn(__buf.len() as jni::sys::jsize).map_err(|e| {
117            <__JniErr as ::core::convert::From<String>>::from(format!(
118                "fixed-size array encode: {}",
119                e
120            ))
121        })?;
122        env.#set_region(&__arr, 0, &__buf).map_err(|e| {
123            <__JniErr as ::core::convert::From<String>>::from(format!(
124                "fixed-size array encode: {}",
125                e
126            ))
127        })?;
128        __arr
129    })
130}

132/// The Kotlin primitive array → `[T; N]` (wire → Rust).
133///
134/// The length check is the `try_into`: a JVM array of the wrong size becomes a
135/// binding error naming the type, never a panic or a partially-filled array.
136pub(crate) fn input_body(
137    ty: &prebindgen_registry::flat::TypeRef,
138    spec: &PrimArray,
139    emit: &prebindgen_registry::RustWriter,
140) -> syn::Expr {
141    let key = ty.key();
142    // The element spelled from the model's own `Array`, so the local's type
143    // ascription cannot disagree with what `prim_array_of` matched.
144    let elem_ty = match ty.kind() {
145        prebindgen_registry::flat::TypeKind::Array { elem, .. } => emit.emit_source_type(elem),
146        _ => unreachable!("prim_array_of matched a non-array"),
147    };
148    // The ascription the decoded array is checked against — spelled from the
149    // reading, as generated Rust always spells.
150    let ty = emit.emit_source_type(ty);
151    let len_err = format!("fixed-size array decode: `{key}` expects a different length");
152    if spec.is_u8 {
153        return syn::parse_quote!({
154            let __buf = env.convert_byte_array(v).map_err(|e| {
155                <__JniErr as ::core::convert::From<String>>::from(format!(
156                    "fixed-size array decode: {}",
157                    e
158                ))
159            })?;
160            let __arr: #ty = __buf.as_slice().try_into().map_err(|_| {
161                <__JniErr as ::core::convert::From<String>>::from(#len_err.to_string())
162            })?;
163            __arr
164        });
165    }
166    let elem_wire = &spec.elem_wire;
167    let get_region = &spec.get_region;
168    // A `jboolean` is a `u8`: normalize it, never reinterpret it — an out-of-
169    // range byte read back as a Rust `bool` would be undefined behavior.
170    let from_wire: syn::Expr = if spec.is_bool {
171        syn::parse_quote!(*__x != 0)
172    } else {
173        syn::parse_quote!(*__x as #elem_ty)
174    };
175    syn::parse_quote!({
176        let __len = env.get_array_length(v).map_err(|e| {
177            <__JniErr as ::core::convert::From<String>>::from(format!(
178                "fixed-size array decode: {}",
179                e
180            ))
181        })? as usize;
182        let mut __buf: ::std::vec::Vec<#elem_wire> = ::std::vec![0 as #elem_wire; __len];
183        env.#get_region(v, 0, &mut __buf).map_err(|e| {
184            <__JniErr as ::core::convert::From<String>>::from(format!(
185                "fixed-size array decode: {}",
186                e
187            ))
188        })?;
189        let __vals: ::std::vec::Vec<#elem_ty> = __buf.iter().map(|__x| #from_wire).collect();
190        let __arr: #ty = __vals.as_slice().try_into().map_err(|_| {
191            <__JniErr as ::core::convert::From<String>>::from(#len_err.to_string())
192        })?;
193        __arr
194    })
195}
