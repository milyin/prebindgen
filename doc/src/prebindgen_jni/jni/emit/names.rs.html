6/// Last-segment ident of a `TypeKey` — e.g. `"Publisher<'static>"` →
7/// `"Publisher"`, `"AdvancedSubscriber<()>"` → `"AdvancedSubscriber"`. Used by
8/// the structured builders ([`Declarations::ptr_class`],
9/// [`Declarations::data_class`]) to derive a default Kotlin class name from
10/// the Rust type-key. Panics for non-path types (e.g. closures, references) —
11/// the per-kind `*_name_mangle` closures see only path-shaped
12/// shorts. For verbatim Kotlin expressions on non-path types, use a
13/// scalar / generic type wrapper.
14pub(crate) fn rust_short_name(key: &TypeKey) -> String {
15    rust_short_name_opt(key).unwrap_or_else(|| {
16        panic!(
17            "rust_short_name: cannot derive Kotlin name from type-key `{}` — \
18             only path-shaped types are supported here",
19            key.as_str()
20        )
21    })
22}

24/// Fallible variant of [`rust_short_name`] — returns `None` for
25/// non-path types instead of panicking. Used by
26/// [`Declarations::note_wrapper_registration`] which is called for rank-0
27/// wrapper patterns including non-path shapes like `()` where there
28/// is no Kotlin short name to derive.
29pub(crate) fn rust_short_name_opt(key: &TypeKey) -> Option<String> {
30    key.short_name()
31}

33/// `VisitMut` that prefixes every bare single-segment `Type::Path` whose
34/// ident is a key of `source_names` with that name's origin module. Walks
35/// the full AST — function signatures, generic args, type ascriptions,
36/// casts, turbofish — so any emitted item passes through one universal pass
37/// instead of each emit site having to remember to qualify.
38pub(crate) struct QualifyEmittedTypes<'a> {
39    /// Bare type name → the module it is reachable under (the item's origin
40    /// crate, or the registry's default module).
41    pub(crate) source_names: &'a std::collections::HashMap<String, syn::Path>,
42    /// Every indexed source name (consts, structs, enums) → the same. Applied
43    /// ONLY inside an array length — see [`QualifyLengthPaths`].
44    pub(crate) length_names: &'a std::collections::HashMap<String, syn::Path>,
45}
46
47impl syn::visit_mut::VisitMut for QualifyEmittedTypes<'_> {
48    fn visit_type_path_mut(&mut self, tp: &mut syn::TypePath) {
49        if tp.qself.is_none() && tp.path.leading_colon.is_none() && tp.path.segments.len() == 1 {
50            let ident = tp.path.segments[0].ident.to_string();
51            if let Some(module) = self.source_names.get(&ident) {
52                let mut qualified = module.clone();
53                qualified.segments.push(tp.path.segments[0].clone());
54                tp.path = qualified;
55            }
56        }
57        syn::visit_mut::visit_type_path_mut(self, tp);
58    }
59
60    /// Qualify a `#[prebindgen]` const used as an ARRAY LENGTH (`[u8; MAX]`).
61    ///
62    /// `syn` models the length as an expression, so `visit_type_path_mut` never
63    /// sees it and the generated file would reference a const that is not in
64    /// scope. The rewrite is confined to `arr.len` on purpose: a generated
65    /// converter body is full of expression paths that are LOCALS (`v`, `env`),
66    /// and a source crate may legally declare `pub const env: usize` — so a
67    /// whole-item expression pass would rewrite those locals to
68    /// `mycrate::env` even when restricted to registered const idents. An array
69    /// length cannot contain a local, which is what makes this scope safe.
70    fn visit_type_array_mut(&mut self, arr: &mut syn::TypeArray) {
71        syn::visit_mut::visit_type_mut(self, &mut arr.elem);
72        reject_unsupported_array_length(arr);
73        let mut lengths = QualifyLengthPaths {
74            length_names: self.length_names,
75        };
76        syn::visit_mut::visit_expr_mut(&mut lengths, &mut arr.len);
77    }
78}

80/// Refuse an array length built from anything but a small, closed set of
81/// expression forms.
82///
83/// [`QualifyLengthPaths`] rewrites a bare path to its origin module, which is
84/// sound only while every path in the length names a source ITEM. Anything that
85/// can bind a name breaks that premise — a local shadowing a source item gets
86/// rewritten into it:
87///
88/// ```ignore
89/// [u8; const { let array_len = 3; array_len }]   // `array_len` is a LOCAL
90/// [u8; match 3 { array_len => array_len }]       // ...so is this one
91/// ```
92///
93/// Qualifying either yields `myflat::array_len`, a function item where a
94/// `usize` was meant; a same-typed collision would compile and silently change
95/// the length.
96///
97/// This is a WHITELIST on purpose. Listing the binding forms instead means
98/// every omitted or newly added `syn::Expr` variant silently reopens the hole —
99/// which is exactly how `match` and `if let` slipped past the first attempt.
100/// Inverting it makes the failure mode "a legitimate length is refused", which
101/// is loud and trivially worked around by hoisting the value into a named
102/// `const`.
103///
104/// # The whitelist is the model's
105///
106/// It listed eight forms — adding `Binary`, `Unary`, `Paren`, `Group`, `Cast`
107/// and `Call` "const arithmetic over those" — and six of them could never
108/// arrive. [`lower_array_len`](prebindgen_registry::flat) accepts an integer
109/// literal or a **bare single-segment name of a marked const**, and nothing
110/// else: `[u8; A + 1]`, `[u8; A as usize]` and `[u8; array_len()]` are all
111/// `ArrayLenReason::NotLiteralOrName`, so the type never becomes a
112/// `TypeKind::Array` and never reaches an emitter at all
113/// (`an_extent_is_a_literal_or_a_marked_const_and_nothing_else` pins it).
114///
115/// So the two forms below are the two the language accepts, restated where the
116/// qualifier needs them. Narrowing a whitelist to what upstream already
117/// enforces cannot open a hole — it closes the gap between the two lists, which
118/// is the only way they could have disagreed.
119fn reject_unsupported_array_length(arr: &mut syn::TypeArray) {
120    // Rendered before the mutable walk below borrows the length.
121    let rendered = quote::ToTokens::to_token_stream(&*arr).to_string();
122    struct Check(Option<&'static str>);
123    // `VisitMut` rather than `Visit`: syn's immutable visitor is behind a
124    // feature this crate does not enable, and the walk mutates nothing.
125    impl syn::visit_mut::VisitMut for Check {
126        fn visit_expr_mut(&mut self, e: &mut syn::Expr) {
127            // The two forms the language accepts — see the note above.
128            let ok = matches!(
129                e,
130                // A literal length, `[u8; 4]`.
131                syn::Expr::Lit(_)
132                    // The name this pass exists to qualify: a marked const.
133                    | syn::Expr::Path(_)
134            );
135            if !ok && self.0.is_none() {
136                self.0 = Some("an unsupported expression form");
137            }
138            syn::visit_mut::visit_expr_mut(self, e);
139        }
140    }
141    let mut check = Check(None);
142    syn::visit_mut::VisitMut::visit_expr_mut(&mut check, &mut arr.len);
143    if let Some(what) = check.0 {
144        panic!(
145            "fixed-size array `{rendered}`: the length uses {what}. Only a literal and the name \
146             of a `#[prebindgen]` const are supported — anything that can bind a name \
147             (`const {{ … }}`, `match`, `if let`, a closure, a loop) would let a LOCAL be \
148             mistaken for a source item, because this generator qualifies the length's paths \
149             against their source module. Hoist the value into a named `const` and use that as \
150             the length."
151        );
152    }
153}

155/// Qualifies the source-crate paths in an array's LENGTH expression, run only
156/// by [`QualifyEmittedTypes::visit_type_array_mut`]. Separate from the type
157/// visitor so it can never reach a converter body's locals.
158///
159/// A length is an ordinary Rust const expression, so it reaches the source
160/// crate two ways and both need the origin module prefixed:
161///
162/// * a **free const**, `[u8; MAX]` — the whole path is the name;
163/// * an **associated const**, `[u8; Holder::N]` — the LEADING segment is the
164///   owning type. Only that segment is rewritten; the rest (`::N`, and any
165///   further associated item) is relative to it and must be left alone.
166///
167/// Both look up the same registry-wide map, so an owner that exists only as a
168/// compile-time namespace does not have to be declared to the binding: forcing
169/// that would emit a dead Kotlin class purely to make the generated Rust
170/// compile.
171struct QualifyLengthPaths<'a> {
172    length_names: &'a std::collections::HashMap<String, syn::Path>,
173}
174
175impl syn::visit_mut::VisitMut for QualifyLengthPaths<'_> {
176    fn visit_expr_path_mut(&mut self, ep: &mut syn::ExprPath) {
177        if ep.qself.is_none() && ep.path.leading_colon.is_none() {
178            // One segment names the const itself; more than one means the
179            // leading segment is the type that owns it. Either way it is the
180            // leading segment that carries the origin module.
181            let ident = ep.path.segments[0].ident.to_string();
182            if let Some(module) = self.length_names.get(&ident) {
183                let mut qualified = module.clone();
184                qualified.segments.extend(ep.path.segments.iter().cloned());
185                ep.path = qualified;
186            }
187        }
188        syn::visit_mut::visit_expr_path_mut(self, ep);
189    }
190}

192/// Add the generated function lifetime to every JNI reference in an
193/// intermediate type, including references nested in registry-owned tuples.
194pub(crate) fn annotate_jobject_with_lifetime(ty: &syn::Type, life: &str) -> syn::Type {
195    if let syn::Type::Tuple(tuple) = ty {
196        let mut tuple = tuple.clone();
197        for elem in &mut tuple.elems {
198            *elem = annotate_jobject_with_lifetime(elem, life);
199        }
200        return syn::Type::Tuple(tuple);
201    }
202    if let syn::Type::Path(tp) = ty {
203        if let Some(last) = tp.path.segments.last() {
204            if crate::jni::wire_access::is_jni_reference_wire(ty)
205                && matches!(last.arguments, syn::PathArguments::None)
206            {
207                let mut new = tp.clone();
208                if let Some(last) = new.path.segments.last_mut() {
209                    let lt =
210                        syn::Lifetime::new(&format!("'{}", life), proc_macro2::Span::call_site());
211                    last.arguments =
212                        syn::PathArguments::AngleBracketed(syn::AngleBracketedGenericArguments {
213                            colon2_token: None,
214                            lt_token: syn::token::Lt::default(),
215                            args: syn::punctuated::Punctuated::from_iter(std::iter::once(
216                                syn::GenericArgument::Lifetime(lt),
217                            )),
218                            gt_token: syn::token::Gt::default(),
219                        });
220                }
221                return syn::Type::Path(new);
222            }
223        }
224    }
225    ty.clone()
226}

232// `pat_match` lived here — `ty.to_token_stream().to_string() == pat` — and was
233// how the converter selector decided what a type WAS: rebuild a wildcard
234// pattern from the spelling, render it to a string, compare. That made the
235// answer depend on how Rust happened to spell the type, so `Box<Option<T>>`
236// reconstructed as `Box<_>`, matched nothing, and got no converter at all
237// (#270). Dispatch reads `TypeKind` now; nothing needs it.

239/// INPUT: wire → rust. Format `<wire_id>_to_<rust_id>_<hash>` (including
240/// `impl Fn(...)` lambda converters — the legacy
241/// `process_kotlin_<Name>_callback` naming is gone with the fun-interface
242/// subsystem).
243pub(crate) fn input_name(rust: &TokenStream, wire: &syn::Type) -> syn::Ident {
244    let rust_id = sanitize_for_ident(&rust.to_string());
245    let wire_id = wire_short(wire);
246    let h = hash_pair(rust, wire);
247    let s = format!("{}_to_{}_{:08x}", wire_id, rust_id, h & 0xffff_ffff);
248    syn::Ident::new(&s, Span::call_site())
249}

251/// OUTPUT: rust → wire. Format `<rust_id>_to_<wire_id>_<hash>`.
252pub(crate) fn output_name(rust: &TokenStream, wire: &syn::Type) -> syn::Ident {
253    let rust_id = sanitize_for_ident(&rust.to_string());
254    let wire_id = wire_short(wire);
255    let h = hash_pair(rust, wire);
256    let s = format!("{}_to_{}_{:08x}", rust_id, wire_id, h & 0xffff_ffff);
257    syn::Ident::new(&s, Span::call_site())
258}

260pub(crate) fn sanitize_for_ident(s: &str) -> String {
261    // Special-case the empty tuple — the all-punctuation token stream
262    // would sanitize to a meaningless fallback. `unit` is recognisable.
263    if s.trim() == "()" {
264        return "unit".to_string();
265    }
266    let mut out = String::with_capacity(s.len());
267    let mut prev_underscore = false;
268    for c in s.chars() {
269        if c.is_ascii_alphanumeric() {
270            out.push(c);
271            prev_underscore = false;
272        } else if !prev_underscore {
273            out.push('_');
274            prev_underscore = true;
275        }
276    }
277    while out.starts_with('_') {
278        out.remove(0);
279    }
280    while out.ends_with('_') {
281        out.pop();
282    }
283    if out.is_empty() {
284        out.push_str("ty");
285    }
286    if out.chars().next().is_some_and(|c| c.is_ascii_digit()) {
287        out.insert(0, '_');
288    }
289    out
290}

292pub(crate) fn wire_short(wire: &syn::Type) -> String {
293    if let syn::Type::Path(tp) = wire {
294        if let Some(last) = tp.path.segments.last() {
295            return sanitize_for_ident(&last.ident.to_string());
296        }
297    }
298    sanitize_for_ident(&wire.to_token_stream().to_string())
299}

301pub(crate) fn hash_pair(rust: &TokenStream, wire: &syn::Type) -> u64 {
302    hash_name_pair(&rust.to_string(), wire)
303}

305/// Hash the canonical source name and intermediate type used in a converter
306/// identifier. Late plans pass a `TypeKey` string so naming never requires
307/// access to Rust syntax; legacy rendered converters pass their token spelling.
308pub(crate) fn hash_name_pair(source: &str, wire: &syn::Type) -> u64 {
309    use std::{
310        collections::hash_map::DefaultHasher,
311        hash::{Hash, Hasher},
312    };
313    let mut h = DefaultHasher::new();
314    source.hash(&mut h);
315    "::".hash(&mut h);
316    wire.to_token_stream().to_string().hash(&mut h);
317    h.finish()
318}
