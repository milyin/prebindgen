8/// Resolve the typed-handle Kotlin FQN for a handle-bearing struct field
9/// and assert its folded strategy is one the struct encode/decode bridge
10/// supports. Today only scalar handle slots (`Direct`, optionally wrapped
11/// in `Nullable`) are encodable as a single `L<FQN>;` ctor arg; a
12/// collection layer (`Iterable`, i.e. `Vec<Handle>`) would need array
13/// codegen and is a loud build-time error until implemented.
14pub(crate) fn handle_field_fqn(ext: &Declarations, h: &Projection) -> String {
15    fn assert_scalar(s: &FoldStrategy) {
16        match s {
17            FoldStrategy::Base => {}
18            FoldStrategy::Optional(_, inner) => assert_scalar(inner),
19            FoldStrategy::Iterable(_) => panic!(
20                "struct handle field: collection (Vec<Handle>) layers are not yet \
21                 supported by the struct encode/decode bridge — add array codegen \
22                 to the late struct output/input plans to lift this guard"
23            ),
24        }
25    }
26    assert_scalar(&h.strategy);
27    ext.kotlin_fqn(&h.leaf_key)
28        .map(|v| v.to_string())
29        .unwrap_or_else(|| {
30            panic!(
31                "struct handle field: leaf `{}` has no Kotlin FQN registered \
32                 (ptr_class)",
33                h.leaf_key
34            )
35        })
36}

38/// Zero/null wire value for a JVM descriptor — used to fill an absent
39/// `Option<nested>`'s leaf slots (the Kotlin `present` flag tells the factory
40/// to ignore them).
41pub(crate) fn primitive_default_for_descriptor(sig: &str) -> TokenStream {
42    match sig {
43        "Z" => quote!(0u8),
44        "B" => quote!(0i8),
45        "C" => quote!(0u16),
46        "S" => quote!(0i16),
47        "I" => quote!(0i32),
48        "J" => quote!(0i64),
49        "F" => quote!(0.0f32),
50        "D" => quote!(0.0f64),
51        _ => quote!(jni::objects::JObject::null()),
52    }
53}

55/// The [`LeafSource::Reach`](crate::unfold::LeafSource) leaves of
56/// a by-value `data_class`, as the fixed-builder output and callback paths want
57/// them.
58///
59/// The list is [`Declarations::struct_out_wires`]', mapped. Runs BEFORE
60/// `resolve`, which is exactly why it shares that composition rather than
61/// walking the struct a second time: the leaf names reach the foreign
62/// `fromParts` parameters and the recipe identically, and two walks agreeing was a
63/// property nothing checked.
64///
65/// `None` — the type keeps the whole-value `fromParts` path — when a field is
66/// one the decomposition cannot state at all. Those are structural: a repeated
67/// nested class, a field the model does not hold as a named one, nesting past
68/// 16. Not a missing transform — a handle, an `enum_class`, a sum, an optional
69/// nested class and a selector inside a gated group all state fine, each
70/// carrying its own conversion (#602). See [`Declarations::struct_out_wires`]
71/// for the list and why one such field declines the whole value.
72pub(crate) fn synth_value_struct_leaves(
73    ext: &Declarations,
74    registry: &(impl Conversions + ?Sized),
75    s: &prebindgen_registry::flat::Struct,
76) -> Option<Vec<crate::unfold::UnfoldLeaf>> {
77    use crate::unfold::{LeafSource, UnfoldLeaf};
78    Some(
79        ext.struct_out_wires_of(registry.flat(), &s.name)?
80            .into_iter()
81            .map(|w| UnfoldLeaf {
82                name: w.name,
83                path: w.reach,
84                out_ty: w.out_ty,
85                identity: false,
86                nullable: w.nullable,
87                // A selector is not read off a place: the emitter assigns the
88                // live alternative's number in each arm of its `match`, which
89                // is what `LeafSource::SumTag` says and what keeps
90                // `OutWire::from_leaf` round-tripping it back to a tag rather
91                // than trying to compile the sum type as a crossing.
92                source: match w.from {
93                    crate::jni::compile::OutFrom::Tag => LeafSource::SumTag,
94                    crate::jni::compile::OutFrom::Present => LeafSource::Presence,
95                    _ => LeafSource::Reach,
96                },
97                groups: w.groups,
98            })
99            .collect(),
100    )
101}

103/// The prefix an arm-local binding carries, and the outer slot it feeds.
104///
105/// A leaf's name is the same on both sides — it is one leaf — so the two would
106/// collide in one function. The arm-local name is the outer name with a
107/// **prefix** this module adds, which is why recovering the outer one is
108/// stripping a known prefix at a known position rather than searching for a
109/// marker: `base` and every slot fragment come from source identifiers, and
110/// any sentinel spelled inside one of them can appear in the middle of a name
111/// too (#616 review).
112/// The one "arm" an optional value has: present. Absence contributes no
113/// bindings of its own, only the defaults its group carries.
114pub(crate) const PRESENT_ARM: i32 = 0;

116/// Assemble the Rust→JVM half of a frozen whole-struct codec.
117///
118/// Classification, child selection, JVM descriptors, and the target Kotlin
119/// class are all frozen before this point. This function only turns that plan
120/// into the final converter body; it performs no registry or declaration
121/// lookup and cannot rediscover source-type facts.
122pub(crate) fn render_struct_output_body(
123    delivery: &crate::jni::emit::FrozenDelivery,
124    java_class_name: &str,
125    emit: &prebindgen_registry::RustWriter,
126) -> syn::Expr {
127    // The whole object graph flattened into leaf wires, then built with ONE
128    // `call_static_method("fromParts", …)` — no per-nested-struct JNI crossing.
129    // The Kotlin `fromParts` factory reassembles the graph in bytecode.
130    //
131    // The flattening is the registry's walk, the same one a fixed-builder site
132    // delivers through: `v` is borrowed and each leaf cloned out of it, which
133    // is what `is_field_read` decides, and a gated group is one `match` the
134    // walk emits rather than an arm this module builds.
135    let n = delivery.wire_count();
136    let obj_idents: Vec<syn::Ident> = (0..n).map(|i| format_ident!("__obj{}", i)).collect();
137    // A converter returns its failure; there is no error sink at this site.
138    let fail = |msg: TokenStream| -> TokenStream {
139        quote! {
140            return ::core::result::Result::Err(
141                <__JniErr as ::core::convert::From<String>>::from(#msg),
142            );
143        }
144    };
145    let (preludes, args, _) = crate::jni::emit::encode_plan_leaves(
146        delivery,
147        delivery.delivered(),
148        &obj_idents,
149        &quote!(&v),
150        &fail,
151        emit,
152    );
153    let factory_sig_lit = syn::LitStr::new(
154        &delivery.factory_signature(java_class_name),
155        Span::call_site(),
156    );
157
158    syn::parse_quote!({
159        #preludes
160        let __obj = env.call_static_method(
161            #java_class_name,
162            "fromParts",
163            #factory_sig_lit,
164            &[#(#args),*],
165        )
166        .and_then(|__v| __v.l())
167        .map_err(|e| <__JniErr as ::core::convert::From<String>>::from(format!("encode struct via fromParts: {}", e)))?;
168        __obj
169    })
170}

172pub(crate) fn struct_module_path(
173    ext: &Declarations,
174    registry: &(impl Conversions + ?Sized),
175    name: &syn::Ident,
176) -> syn::Path {
177    // The module the struct is reachable under from the generated file: its
178    // origin crate (multi-source registries) or the default module. Takes the
179    // NAME, which is all it needs — so it serves a caller holding the element
180    // and one still holding the item.
181    ext.fn_module(registry, name)
182}
