1//! Feature expression evaluation for `#[cfg(...)]` attributes.
2//!
3//! This module provides functionality to parse and evaluate complex cfg expressions
4//! that include logical operators like `any`, `all`, and `not`, as well as simple
5//! feature checks.
6
7use crate::{codegen::CfgExprRules, SourceLocation};
8
9/// Represents a cfg expression that can be evaluated against a set of enabled/disabled features
10#[derive(Debug, Clone, PartialEq)]
11pub enum CfgExpr {
12    // Simple predicates first
13    /// A simple feature check: `feature = "name"`
14    Feature(String),
15    /// Target architecture check: `target_arch = "arch"`
16    TargetArch(String),
17    /// Target vendor check: `target_vendor = "vendor"`
18    TargetVendor(String),
19    /// Target OS check: `target_os = "os"`
20    TargetOs(String),
21    /// Target environment check: `target_env = "env"`
22    TargetEnv(String),
23
24    // Logical operators next
25    /// Logical NOT: `not(expr)`
26    Not(Box<CfgExpr>),
27    /// Logical AND: `all(expr1, expr2, ...)`
28    All(Vec<CfgExpr>),
29    /// Logical OR: `any(expr1, expr2, ...)`
30    Any(Vec<CfgExpr>),
31
32    // Fallbacks
33    /// Any other cfg expression we don't specifically handle
34    Other(String),
35    /// Explicit false value (for feature processing)
36    False,
37}
38
39impl CfgExpr {
40    /// Parse a cfg expression from syn tokens
41    pub fn parse_from_tokens(tokens: &proc_macro2::TokenStream) -> Result<Self, String> {
42        let token_string = tokens.to_string();
43        Self::parse_from_string(&token_string)
44    }
45
46    /// Parse a cfg expression from a string representation
47    pub fn parse_from_string(input: &str) -> Result<Self, String> {
48        let input = input.trim();
49
50        // Handle logical expressions first (before simple features)
51        if let Some(inner) = strip_function_call(input, "not") {
52            let inner_expr = Self::parse_from_string(&inner)?;
53            return Ok(CfgExpr::Not(Box::new(inner_expr)));
54        }
55
56        if let Some(inner) = strip_function_call(input, "all") {
57            let exprs = parse_comma_separated(&inner)?;
58            return Ok(CfgExpr::All(exprs));
59        }
60
61        if let Some(inner) = strip_function_call(input, "any") {
62            let exprs = parse_comma_separated(&inner)?;
63            return Ok(CfgExpr::Any(exprs));
64        }
65
66        // Handle simple feature expressions
67        if let Some(feature_name) = extract_simple_feature(input) {
68            return Ok(CfgExpr::Feature(feature_name));
69        }
70
71        // Handle target_arch expressions
72        if let Some(arch) = extract_target_arch(input) {
73            return Ok(CfgExpr::TargetArch(arch));
74        }
75        // Handle target_vendor expressions
76        if let Some(vendor) = extract_target_vendor(input) {
77            return Ok(CfgExpr::TargetVendor(vendor));
78        }
79
80        // Handle target_os expressions
81        if let Some(os) = extract_target_os(input) {
82            return Ok(CfgExpr::TargetOs(os));
83        }
84
85        // Handle target_env expressions
86        if let Some(env) = extract_target_env(input) {
87            return Ok(CfgExpr::TargetEnv(env));
88        }
89
90        // If we can't parse it, store it as "Other"
91        Ok(CfgExpr::Other(input.to_string()))
92    }
93
94    /// Process features according to the rules:
95    /// - Features in enabled list: replaced with true and removed from expression
96    /// - Features in disabled list: replaced with false and removed from expression  
97    /// - Features in mapping list: renamed
98    /// - Any unmapped feature remaining: panic with "unmapped feature"
99    /// - If architecture, os, target, env condition is specified, replace
100    ///   matching condition with false and unmatching to true
101    /// - If architecture, os, target, env condition is not specified, leave as is
102    pub fn apply_rules(
103        &self,
104        rules: &CfgExprRules,
105        source_location: &SourceLocation,
106    ) -> Option<Self> {
107        match self {
108            CfgExpr::Feature(name) => {
109                if rules.enabled_features.contains(name) {
110                    // Feature is enabled - replace with true (remove from expression)
111                    None // This means "always true", caller should handle removal
112                } else if rules.disabled_features.contains(name) {
113                    // Feature is disabled - replace with false (remove from expression)
114                    Some(CfgExpr::False) // Explicit false value
115                } else if let Some(new_name) = rules.feature_mappings.get(name) {
116                    // Feature should be mapped
117                    Some(CfgExpr::Feature(new_name.clone()))
118                } else if rules.disable_unknown_features {
119                    // Treat unknown feature as disabled
120                    Some(CfgExpr::False)
121                } else {
122                    // Unmapped feature - panic with source location information
123                    panic!("unmapped feature: {name} (at {source_location})");
124                }
125            }
126            CfgExpr::TargetVendor(val) => {
127                if let Some(sel) = rules.enabled_target_vendor.as_ref() {
128                    if val == sel {
129                        None
130                    } else {
131                        Some(CfgExpr::False)
132                    }
133                } else {
134                    Some(self.clone())
135                }
136            }
137            CfgExpr::TargetArch(val) => {
138                if let Some(sel) = rules.enabled_target_arch.as_ref() {
139                    if val == sel {
140                        None
141                    } else {
142                        Some(CfgExpr::False)
143                    }
144                } else {
145                    Some(self.clone())
146                }
147            }
148            CfgExpr::All(exprs) => {
149                let mut processed_exprs = Vec::new();
150                for expr in exprs {
151                    match expr.apply_rules(rules, source_location) {
152                        Some(CfgExpr::False) => {
153                            // If any expression in All is false, the whole All is false
154                            return Some(CfgExpr::False);
155                        }
156                        Some(processed) => processed_exprs.push(processed),
157                        None => {
158                            // This expression evaluated to true, can be omitted from All
159                        }
160                    }
161                }
162                match processed_exprs.len() {
163                    0 => None, // All expressions were true, so All is true
164                    1 => Some(processed_exprs.into_iter().next().unwrap()), // Simplify single expression
165                    _ => Some(CfgExpr::All(processed_exprs)),
166                }
167            }
168            CfgExpr::Any(exprs) => {
169                let mut processed_exprs = Vec::new();
170                let mut has_true = false;
171                for expr in exprs {
172                    match expr.apply_rules(rules, source_location) {
173                        Some(CfgExpr::False) => {
174                            // False expressions in Any can be omitted
175                        }
176                        Some(processed) => processed_exprs.push(processed),
177                        None => {
178                            // This expression evaluates to true
179                            has_true = true;
180                        }
181                    }
182                }
183                if has_true {
184                    None // If any expression in Any is true, the whole Any is true
185                } else {
186                    match processed_exprs.len() {
187                        0 => Some(CfgExpr::False), // All expressions were false, so Any is false
188                        1 => Some(processed_exprs.into_iter().next().unwrap()), // Simplify single expression
189                        _ => Some(CfgExpr::Any(processed_exprs)),
190                    }
191                }
192            }
193            CfgExpr::Not(expr) => {
194                match expr.apply_rules(rules, source_location) {
195                    Some(CfgExpr::False) => None, // not(false) = true
196                    Some(processed) => Some(CfgExpr::Not(Box::new(processed))),
197                    None => Some(CfgExpr::False), // not(true) = false
198                }
199            }
200            CfgExpr::TargetOs(val) => {
201                if let Some(sel) = rules.enabled_target_os.as_ref() {
202                    if val == sel {
203                        None
204                    } else {
205                        Some(CfgExpr::False)
206                    }
207                } else {
208                    Some(self.clone())
209                }
210            }
211            CfgExpr::TargetEnv(val) => {
212                if let Some(sel) = rules.enabled_target_env.as_ref() {
213                    if val == sel {
214                        None
215                    } else {
216                        Some(CfgExpr::False)
217                    }
218                } else {
219                    Some(self.clone())
220                }
221            }
222            CfgExpr::Other(_) => Some(self.clone()),
223            CfgExpr::False => Some(CfgExpr::False),
224        }
225    }

227    /// Convert back to a token stream for syn attributes
228    pub fn to_tokens(&self) -> proc_macro2::TokenStream {
229        match self {
230            CfgExpr::Feature(name) => {
231                quote::quote! { feature = #name }
232            }
233            CfgExpr::TargetArch(arch) => {
234                quote::quote! { target_arch = #arch }
235            }
236            CfgExpr::TargetVendor(vendor) => {
237                quote::quote! { target_vendor = #vendor }
238            }
239            CfgExpr::TargetOs(os) => {
240                quote::quote! { target_os = #os }
241            }
242            CfgExpr::TargetEnv(env) => {
243                quote::quote! { target_env = #env }
244            }
245            CfgExpr::All(exprs) => {
246                let tokens: Vec<_> = exprs.iter().map(|e| e.to_tokens()).collect();
247                quote::quote! { all(#(#tokens),*) }
248            }
249            CfgExpr::Any(exprs) => {
250                let tokens: Vec<_> = exprs.iter().map(|e| e.to_tokens()).collect();
251                quote::quote! { any(#(#tokens),*) }
252            }
253            CfgExpr::Not(expr) => {
254                let inner = expr.to_tokens();
255                quote::quote! { not(#inner) }
256            }
257            CfgExpr::Other(content) => {
258                // Try to parse as a token stream
259                if let Ok(tokens) = content.parse::<proc_macro2::TokenStream>() {
260                    tokens
261                } else {
262                    // Fallback: treat as identifier
263                    let ident = proc_macro2::Ident::new(content, proc_macro2::Span::call_site());
264                    quote::quote! { #ident }
265                }
266            }
267            CfgExpr::False => {
268                // This should not normally be converted to tokens, but if needed, use a false literal
269                quote::quote! { any() } // any() with no arguments is always false
270            }
271        }
272    }
273}

275/// Extract a simple feature name from expressions like `feature = "name"`
276fn extract_simple_feature(input: &str) -> Option<String> {
277    use regex::Regex;
278    let feature_regex = Regex::new(r#"feature\s*=\s*"([^"]+)""#).unwrap();
279
280    feature_regex
281        .captures(input)
282        .map(|captures| captures[1].to_string())
283}

316/// Strip a function call wrapper, returning the inner content
317/// For example: `not(feature = "test")` -> `feature = "test"`
318fn strip_function_call(input: &str, function_name: &str) -> Option<String> {
319    let input = input.trim();
320
321    // Handle function calls with or without spaces: "not(" and "not ("
322    let pattern1 = format!("{function_name}(");
323    let pattern2 = format!("{function_name} (");
324
325    if (input.starts_with(&pattern1) || input.starts_with(&pattern2)) && input.ends_with(')') {
326        let start = if input.starts_with(&pattern1) {
327            pattern1.len()
328        } else {
329            pattern2.len()
330        };
331        let end = input.len() - 1;
332
333        if start < end {
334            Some(input[start..end].to_string())
335        } else {
336            Some(String::new())
337        }
338    } else {
339        None
340    }
341}

343/// Parse comma-separated expressions
344fn parse_comma_separated(input: &str) -> Result<Vec<CfgExpr>, String> {
345    let mut exprs = Vec::new();
346    let mut current = String::new();
347    let mut paren_depth = 0;
348    let mut in_quotes = false;
349    let mut escape_next = false;
350
351    for ch in input.chars() {
352        if escape_next {
353            current.push(ch);
354            escape_next = false;
355            continue;
356        }
357
358        match ch {
359            '\\' => {
360                escape_next = true;
361                current.push(ch);
362            }
363            '"' => {
364                in_quotes = !in_quotes;
365                current.push(ch);
366            }
367            '(' if !in_quotes => {
368                paren_depth += 1;
369                current.push(ch);
370            }
371            ')' if !in_quotes => {
372                paren_depth -= 1;
373                current.push(ch);
374            }
375            ',' if !in_quotes && paren_depth == 0 => {
376                if !current.trim().is_empty() {
377                    exprs.push(CfgExpr::parse_from_string(current.trim())?);
378                }
379                current.clear();
380            }
381            _ => {
382                current.push(ch);
383            }
384        }
385    }
386
387    if !current.trim().is_empty() {
388        exprs.push(CfgExpr::parse_from_string(current.trim())?);
389    }
390
391    Ok(exprs)
392}
