1//! Language-neutral declaration vocabulary: the decl objects and constructor
2//! macros a build script uses to describe boundary expansion (`expand_param!`
3//! / `expand_return!`), free functions (`fun!`), and canonical single-value
4//! conversions (`convert!`). Moved down from the JNI adapter (formerly
5//! `api/lang/jnigen`, now the separate `prebindgen-jni` crate) because none of
6//! it is Kotlin/JNI-specific — it only references [`TypeKey`],
7//! [`Origin<syn::Type>`], and plain `syn` types. `prebindgen-jni`'s own
8//! `jni/decl.rs` keeps the genuinely Kotlin-specific declarations
9//! (`ptr_class!`, `enum_class!`, `sealed_class!`, `data_class!`, `constant!`,
10//! `package!`) and re-exports these types from here, so the public
11//! `prebindgen_flat::*` surface is unaffected by the split.
12

18/// The origin of a type a **build script** wrote.
19///
20/// Real tokens, and deliberately no source position: `SourceLocation::default()`
21/// is the sanctioned placeless location for exactly this — a signature or type a
22/// build script authored was never in a captured file, and `has_position` already
23/// gates what a diagnostic prints for one.
24pub fn declared_origin(ty: syn::Type) -> Origin<syn::Type> {
25    Origin::new(ty, std::rc::Rc::new(prebindgen::SourceLocation::default()))
26}

33/// One arm of an `expand_param!` `.variant*` list (type-level or per-fn).
34#[derive(Clone)]
35pub enum LocalVariant {
36    /// Build via this declared constructor member / constructor fn.
37    Ctor(syn::Ident),
38    /// Accept an already-built value directly.
39    SelfIdentity,
40}

42/// One arm of an `expand_return!` `.field*` list (type-level or per-fn). The name is stored raw (`None` = derive at replay time: for a
43/// class-level field, the class member's target-language name if the accessor is a
44/// declared member, else `snake_to_camel`; for a per-fn field,
45/// `snake_to_camel`).
46// large_enum_variant: a handful of fields exist per binding, held while
47// declarations replay — boxing the syn payloads would only complicate the
48// arms (same trade-off as `ConvertSourceKind`).
49#[allow(clippy::large_enum_variant)]
50#[derive(Clone)]
51pub enum LocalField {
52    /// Include the named accessor's value as a leaf/field, with an optional
53    /// explicit name override.
54    Named(syn::Ident, Option<String>),
55    /// Include the handle itself as a field.
56    SelfField,
57    /// Include a **custom, locally-defined** accessor's value: any fn the
58    /// binding crate defines, declared with the one binding-local vocabulary
59    /// (`fun!(crate::f).sig(sig!((v: &Self) -> Ret))`) — no `#[prebindgen]`
60    /// item behind it, so the full signature (receiver explicit) is stated.
61    /// `name_override` follows the uniform field-name precedence.
62    Local {
63        path: syn::Path,
64        sig: syn::Signature,
65        name_override: Option<String>,
66    },
67    /// Include **every field of the type's value form** — the struct returned
68    /// by the named accessor — each as its own field. Expands to the same
69    /// records the fields would produce if named one by one; see
70    /// [`ExpandReturnDecl::fields`].
71    Fields(FieldsDecl),
72}

74/// Build a [`FunctionDecl`] from a bare function ident or a path:
75///
76/// * `fun!(foo)` — a `#[prebindgen]` fn; its signature is read from the
77///   registry.
78/// * `fun!(crate::foo)` — a **binding-local** fn: any fn the binding crate
79///   defines, exported through the same machinery as a `#[prebindgen]` one.
80///   A path carries no signature to read, so chain
81///   [`.sig(sig!(…))`](crate::FunctionDecl::sig). The generated file
82///   calls it by the declared path (it compiles inside the binding crate,
83///   so `crate::`-rooted paths resolve).
84#[macro_export]
85macro_rules! fun {
86    ($name:ident) => {
87        $crate::FunctionDecl::new($crate::ident!($name))
88    };
89    ($path:path) => {
90        $crate::FunctionDecl::new_local($crate::__macro_support::parse_path(stringify!($path)))
91    };
92}

231/// Declares a type's **default input boundary**: how a parameter of this type
232/// may be supplied, as a list of *variants* — "built from this constructor's
233/// ingredients, OR that one's, OR passed as an existing handle". Applies to
234/// every function with a parameter of the type; a single function opts out or
235/// narrows via [`FunctionDecl::expand_param`].
236///
237/// Build one with [`expand_param!`](crate::expand_param), add arms with
238/// [`variant`](Self::variant) / [`variant_self`](Self::variant_self), and hand
239/// it to the adapter's `expand` declaration.
240///
241/// **Generated shape** — at the wire tier this is a selector dispatch: with
242/// more than one arm the parameter crosses as a selector `Int` plus one
243/// nullable slot per arm (`keyExprSel: Int, keyExpr0: String?,
244/// keyExpr1: KeyExpr?`), and the raw call site passes `(0, "key", null)`-style
245/// tuples. That selector form is always emitted; the wrapper's generated KDoc
246/// shape-notes document the exact slots per function.
247///
248/// **Splittability (checked)** — a multi-variant declaration must be
249/// *splittable*: its arms must surface as **distinct JVM signatures**, so a
250/// function can request idiomatic typed **overloads** on top of the selector
251/// form (`f(key: String, …)` / `f(key: KeyExpr, …)`) via
252/// [`FunctionDecl::split_on_param`](crate::fun). This is verified up front —
253/// two arms with the same erased parameter types are a hard build error.
254/// [`.no_split()`](Self::no_split) suppresses that check for a variant set that
255/// will only ever be used as the selector form. The type-level declaration
256/// itself emits no overloads; emission is per-function via `.split_on_param`.
257/// The selector form always stays public, so consumers can also add their own
258/// same-named overloads by hand.
259///
260/// The type does **not** have to be declared in any package. A boundary decl
261/// on an undeclared type makes it **rust-side-only**: the value is always
262/// built from its ingredients at the boundary and never materializes in
263/// the target language — no class, no handle, nothing to close. The one
264/// restriction is structural: [`variant_self`](Self::variant_self) hard-errors
265/// for such a type, since there is no target-language object to pass.
266///
267/// ```
268/// // A KeyExpr param accepts EITHER a String (built via keyexpr_new_try_from)
269/// // OR an existing KeyExpr handle:
270/// let _ = prebindgen_registry::expand_param!(KeyExpr)
271///     .variant(prebindgen_registry::fun!(keyexpr_new_try_from))
272///     .variant_self();
273/// ```
274#[derive(Clone)]
275pub struct ExpandParamDecl {
276    key: TypeKey,
277    /// The type this declaration was **written with** — the `X` the macro
278    /// received. Kept because the declaration is where it came from: recovering
279    /// it later *from* the key was reasoning backwards from an identity (#291).
280    rust_type: Origin<syn::Type>,
281    variants: Vec<LocalVariant>,
282    /// `.no_split()` — suppress the proactive splittability check for this
283    /// variant set (it will only ever be used as the selector form). See
284    /// [`Self::no_split`].
285    no_split: bool,
286}
287
288impl ExpandParamDecl {
289    pub fn new(rust_type: syn::Type) -> Self {
290        Self {
291            key: TypeKey::from_type(&rust_type),
292            rust_type: declared_origin(rust_type),
293            variants: Vec::new(),
294            no_split: false,
295        }
296    }
297
298    /// The type identity this declaration is registered under.
299    pub fn key(&self) -> &TypeKey {
300        &self.key
301    }
302
303    /// The type this declaration was written with, as originally parsed.
304    pub fn rust_type(&self) -> &Origin<syn::Type> {
305        &self.rust_type
306    }
307
308    /// The declared build-from / existing-handle arms, in declaration order.
309    /// `pub(crate)`, not `pub` — [`LocalVariant`] itself is `pub(crate)`
310    /// (a public fn cannot return a private type).
311    pub fn variants(&self) -> &[LocalVariant] {
312        &self.variants
313    }
314
315    /// Whether `.no_split()` was declared (suppresses the splittability
316    /// check). Named `is_no_split` rather than `no_split` — that name is
317    /// already the builder method that sets the flag ([`Self::no_split`]).
318    pub fn is_no_split(&self) -> bool {
319        self.no_split
320    }
321
322    /// Add a **build-from** arm: parameters of this type also carry the
323    /// named `#[prebindgen]` constructor's inputs on the wire, and Rust
324    /// builds the value in the same call. E.g. `keyexpr_new_try_from(&str)`
325    /// gives every function taking a `KeyExpr` a String-carrying arm —
326    /// as a selector + nullable slot at the wire tier (see the type-level
327    /// docs for the exact generated shape), not as a target-language overload.
328    ///
329    /// A variant arm only *names* the constructor: no target-language surface of its
330    /// own, so a decorated `fun!` (`.name()` / expand overrides) is a hard
331    /// error rather than a silent discard.
332    pub fn variant(mut self, ctor: FunctionDecl) -> Self {
333        assert!(
334            ctor.name_override.is_none()
335                && ctor.param_expands.is_empty()
336                && ctor.return_expand.is_none(),
337            "expand_param!({}).variant(fun!({})): a variant arm only names the \
338             `#[prebindgen]` constructor — .name()/expand overrides don't apply",
339            self.key.as_str(),
340            ctor.rust_ident
341        );
342        assert!(
343            ctor.local.is_none(),
344            "expand_param!({}).variant(fun!(…::{f})): a variant arm only NAMES a fn — \
345             declare the binding-local fn via .fun/.method/.constructor/convert! first, \
346             then reference it here by ident: fun!({f})",
347            self.key.as_str(),
348            f = ctor.rust_ident
349        );
350        self.variants.push(LocalVariant::Ctor(ctor.rust_ident));
351        self
352    }
353
354    /// Add the **existing-handle** arm: also accept an already-built value.
355    /// On its own this is simply the default (a bare handle), so declaring it
356    /// alone changes nothing; it earns its place next to build variants.
357    pub fn variant_self(mut self) -> Self {
358        self.variants.push(LocalVariant::SelfIdentity);
359        self
360    }
361
362    /// **Suppress the splittability check.** A multi-variant expansion is
363    /// verified up front to be *splittable* (its arms surface as distinct JVM
364    /// signatures) so that [`FunctionDecl::split_on_param`](crate::fun) can emit
365    /// idiomatic typed overloads. `.no_split()` opts this variant set out of
366    /// that check — declare it when two arms genuinely share a JVM signature and
367    /// you only ever want the selector form (a function that then tries to
368    /// `.split_on_param` such a parameter gets the concrete ambiguity error).
369    ///
370    /// A no-op on a single-variant declaration (nothing to check).
371    pub fn no_split(mut self) -> Self {
372        self.no_split = true;
373        self
374    }
375}

377/// Declares a type's **default output boundary**: wherever the type is
378/// returned or handed to a callback, it is decomposed into this set of
379/// *fields*, all delivered in one FFI crossing — instead of an opaque handle
380/// the caller must then query field by field with more JNI calls. Applies to
381/// every function returning the type; a single function opts out or replaces
382/// the set via [`FunctionDecl::expand_return`].
383///
384/// Build one with [`expand_return!`](crate::expand_return), add fields with
385/// [`field`](Self::field) / [`field_self`](Self::field_self), and hand it to
386/// the adapter's `expand` declaration.
387///
388/// The type does **not** have to be declared in any package. A boundary decl
389/// on an undeclared type makes it **rust-side-only**: every returned /
390/// callback-delivered / `Result`-error value of it is decomposed into these
391/// fields and the value itself never reaches the target language. This is the
392/// natural shape for an error type consumed by an error channel — no dead
393/// class is emitted. Restrictions for such a type:
394/// [`field_self`](Self::field_self) hard-errors (there is no target-language object to
395/// deliver), and field names cannot inherit from class members (there are
396/// none) — use `.name(...)` on each field or accept the camel-cased default.
397///
398/// ```
399/// // A returned Sample crosses as { payload, kind } in one call:
400/// let _ = prebindgen_registry::expand_return!(Sample)
401///     .field(prebindgen_registry::fun!(sample_get_payload))
402///     .field(prebindgen_registry::fun!(sample_get_kind));
403/// ```
404
405#[derive(Clone)]
406pub struct ExpandReturnDecl {
407    key: TypeKey,
408    /// The type this declaration was **written with** — the `X` the macro
409    /// received. Kept because the declaration is where it came from: recovering
410    /// it later *from* the key was reasoning backwards from an identity (#291).
411    rust_type: Origin<syn::Type>,
412    fields: Vec<LocalField>,
413}
414
415impl ExpandReturnDecl {
416    pub fn new(rust_type: syn::Type) -> Self {
417        Self {
418            key: TypeKey::from_type(&rust_type),
419            rust_type: declared_origin(rust_type),
420            fields: Vec::new(),
421        }
422    }
423
424    /// The type identity this declaration is registered under.
425    pub fn key(&self) -> &TypeKey {
426        &self.key
427    }
428
429    /// The type this declaration was written with, as originally parsed.
430    pub fn rust_type(&self) -> &Origin<syn::Type> {
431        &self.rust_type
432    }
433
434    /// The declared field records, in declaration order. Named `field_list`
435    /// rather than `fields` — that name is already the builder method that
436    /// appends a value-form ([`Self::fields`]). `pub(crate)`, not `pub` —
437    /// [`LocalField`] itself is `pub(crate)` (a public fn cannot return a
438    /// private type).
439    pub fn field_list(&self) -> &[LocalField] {
440        &self.fields
441    }
442
443    /// Add one field — a reader whose value crosses as this leaf:
444    ///
445    /// * `fun!(f)` — a `#[prebindgen]` reader (`f(&Self) -> Field`), its
446    ///   signature read from the registry.
447    /// * `fun!(crate::f).sig(sig!((v: &Self) -> Field))` — a **custom,
448    ///   locally-defined** reader: any fn the binding crate defines, its
449    ///   signature stated (the receiver explicit — it must take `&Self`).
450    ///   One use among many: conditional delivery, an `Option<&Self>` return
451    ///   becoming a nullable handle leaf that is null when the binding-side
452    ///   predicate declines.
453    ///
454    /// The target-language field name is uniform for both: an explicit
455    /// `.name(...)` on the `fun!`; else the name of the class member if the same
456    /// fn is also declared as a method on this type's class (so a getter
457    /// that is both a method and a field is named once); else the
458    /// camel-cased fn ident (a path's LAST segment).
459    ///
460    /// Only the accessor's name is used here: expand overrides on the `fun!`
461    /// are a hard error rather than a silent discard (the field's own
462    /// decomposition comes from ITS type's boundary decl, not from the
463    /// accessor).
464    pub fn field(mut self, accessor: FunctionDecl) -> Self {
465        self.reject_beside_consuming("field(..)");
466        assert!(
467            accessor.param_expands.is_empty() && accessor.return_expand.is_none(),
468            "expand_return!({}).field(fun!({})): expand overrides don't apply to a \
469             field accessor — only .name() is honored",
470            self.key.as_str(),
471            accessor.rust_ident
472        );
473        self.fields.push(match accessor.local {
474            None => LocalField::Named(accessor.rust_ident, accessor.name_override),
475            Some((path, sig)) => {
476                let Some(sig) = sig else {
477                    panic!(
478                        "expand_return!({}).field(fun!({p})): a binding-local field states \
479                         its accessor's signature — chain .sig(sig!((v: &{k}) -> Ret))",
480                        self.key.as_str(),
481                        p = quote::quote!(#path),
482                        k = self.key.as_str()
483                    );
484                };
485                LocalField::Local {
486                    path,
487                    sig,
488                    name_override: accessor.name_override,
489                }
490            }
491        });
492        self
493    }
494
495    /// Include the **handle itself** among the fields, so the consumer gets a
496    /// live, closeable object in addition to the read-out values (e.g. a
497    /// `Query` delivered with its fields *and* the handle it needs to reply).
498    /// Declare it **last**, after any field that decomposes a nested handle,
499    /// so the generated Rust moves the value only after those borrows.
500    pub fn field_self(mut self) -> Self {
501        self.reject_beside_consuming("field_self()");
502        self.fields.push(LocalField::SelfField);
503        self
504    }
505
506    /// The one rule [`Self::fields_self_into`] adds: it hands the value **itself**
507    /// over, so nothing else in the decl can still read it.
508    fn reject_beside_consuming(&self, what: &str) {
509        if let Some(f) = self.fields.iter().find_map(|f| match f {
510            LocalField::Fields(d) if d.consuming => Some(&d.func),
511            _ => None,
512        }) {
513            panic!(
514                "expand_return!({k}).fields_self_into(fields!({f})).{what}: `.fields_self_into(..)` hands \
515                 the value ITSELF over as its fields, so nothing else can read it afterwards — \
516                 it must be the decl's only record. Use `.fields(fields!(..))` with the \
517                 borrowing form of the accessor if you need both.",
518                k = self.key.as_str(),
519                f = f,
520            );
521        }
522    }
523
524    /// Take the fields from the type's **value form** — a `#[prebindgen]`
525    /// accessor returning "this type's own accessors gathered into one struct"
526    /// — instead of restating them.
527    ///
528    /// `.fields(fields!(f))` is exactly `.field(...)` applied to each field of
529    /// that struct, so it has the same configurability (per-field overrides and
530    /// renames live on the [`FieldsDecl`]) and, crucially, the same
531    /// decomposition rule: **each field crosses by its own type's default
532    /// output boundary**. A field whose type has its own `expand_return!` is
533    /// decomposed by it (a `KeyExpr` field still crosses as its string, not as
534    /// a handle); a declared `data_class!` field expands into its fields; a
535    /// field behind `Option` / `Vec` stays one leaf. So swapping a hand-written
536    /// field list for `.fields(...)` keeps the boundary shape it already had —
537    /// what changes is that the list can no longer drift from the struct.
538    ///
539    /// ```
540    /// // Instead of restating SampleStruct's fields one by one:
541    /// let _ = prebindgen_registry::expand_return!(Sample)
542    ///     .fields(prebindgen_registry::fields!(sample_to_struct));
543    /// ```
544    ///
545    /// The accessor **borrows** its receiver (`f(v: &Self) -> SelfStruct`):
546    /// the struct is built from a borrow, so each field is cloned into it and
547    /// the leaves clone again out of it, and the value survives. It therefore
548    /// mixes freely — `.fields(...).field_self()` delivers the value form's
549    /// fields *and* the live handle. At most one value form per decl.
550    ///
551    /// Where the value is delivered **owned** — a callback argument
552    /// (`impl Fn(Sample)`), an owned return — and nothing else needs it, use
553    /// [`fields_self_into`](Self::fields_self_into) instead: those clones are being paid
554    /// on a value that is about to be dropped.
555    pub fn fields(mut self, decl: FieldsDecl) -> Self {
556        self.reject_beside_consuming("fields(..)");
557        self.reject_second_value_form(&decl);
558        self.fields.push(LocalField::Fields(decl));
559        self
560    }
561
562    /// Like [`fields`](Self::fields), but the accessor **consumes** its
563    /// receiver (`f(v: Self) -> SelfStruct`): the value is moved in and each
564    /// field is moved *out* into its leaf. No clones at all.
565    ///
566    /// This is the same decision [`field_self`](Self::field_self) makes, one
567    /// step further: `.field_self()` hands the value over whole,
568    /// `.fields_self_into(...)` hands *the value itself* over as its parts, and
569    /// `.fields(...)` hands over a copy of its parts. Use it wherever the value
570    /// arrives owned and is not needed afterwards — the hot receive path this
571    /// whole declarator exists to make cheap.
572    ///
573    /// ```
574    /// let _ = prebindgen_registry::expand_return!(Sample)
575    ///     .fields_self_into(prebindgen_registry::fields!(sample_into_struct));
576    /// ```
577    ///
578    /// Because it gives the value away it must be the decl's **only** record —
579    /// a `.field_self()` or a sibling `.field(...)` would read a value that is
580    /// gone — which is a declaration-time panic either way round. It may still
581    /// be reached through *another* value form: the parent's field is handed to
582    /// it by move, since a hoisted value form is an owned struct and its fields
583    /// are disjoint.
584    ///
585    /// The declarator and the accessor's signature must agree; naming a
586    /// `&Self` accessor here (or a by-value one on [`fields`](Self::fields)) is
587    /// an error, so the declared intent cannot drift from the function it
588    /// names. At a **borrowed** delivery position there is no value to give up,
589    /// so the emitter clones once up front and consumes the clone — the same
590    /// cost the borrowing form would have paid, which keeps one declaration
591    /// usable by both owned and `&T` returns of the type.
592    pub fn fields_self_into(mut self, decl: FieldsDecl) -> Self {
593        self.reject_second_value_form(&decl);
594        assert!(
595            self.fields.is_empty(),
596            "expand_return!({k}).fields_self_into(fields!({f})): `.fields_self_into(..)` hands the value \
597             ITSELF over as its fields, so it must be the decl's only record — the records \
598             already declared would read a value that is gone. Use `.fields(fields!(..))` with \
599             the borrowing form of the accessor if you need both.",
600            k = self.key.as_str(),
601            f = decl.func,
602        );
603        self.fields.push(LocalField::Fields(decl.consuming()));
604        self
605    }
606
607    fn reject_second_value_form(&self, decl: &FieldsDecl) {
608        assert!(
609            !self
610                .fields
611                .iter()
612                .any(|f| matches!(f, LocalField::Fields(_))),
613            "expand_return!({}): the decl already expands a value form (fields!({})) — \
614             one value form states the whole field set",
615            self.key.as_str(),
616            decl.func
617        );
618    }
619}

621/// A **value-form expansion**: the accessor whose returned struct supplies the
622/// fields, plus the per-field adjustments. Built with
623/// [`fields!`](crate::fields) and handed to
624/// [`ExpandReturnDecl::fields`].
625///
626/// Both adjusters key on the **Rust struct field name**, mirroring
627/// [`FunctionDecl::expand_param`]'s Rust-parameter-name key: an unknown field
628/// name or a repeated one is a hard error, so a field renamed upstream is
629/// caught rather than silently ignored.
630#[derive(Clone)]
631pub struct FieldsDecl {
632    func: syn::Ident,
633    overrides: Vec<(String, ExpandReturnDecl)>,
634    names: Vec<(String, String)>,
635    /// Set by [`ExpandReturnDecl::fields_self_into`] — the accessor consumes its
636    /// receiver. Declared rather than read off the signature, because giving
637    /// the value away is a boundary decision; the two are cross-checked when
638    /// the records are resolved.
639    consuming: bool,
640}
641
642impl FieldsDecl {
643    pub fn new(func: syn::Ident) -> Self {
644        Self {
645            func,
646            overrides: Vec::new(),
647            names: Vec::new(),
648            consuming: false,
649        }
650    }
651
652    pub(crate) fn consuming(mut self) -> Self {
653        self.consuming = true;
654        self
655    }
656
657    /// The value-form accessor's ident, as declared with [`fields!`](crate::fields).
658    pub fn func(&self) -> &syn::Ident {
659        &self.func
660    }
661
662    /// The per-field decomposition overrides, in declaration order.
663    pub fn overrides(&self) -> &[(String, ExpandReturnDecl)] {
664        &self.overrides
665    }
666
667    /// The per-field name overrides, in declaration order.
668    pub fn names(&self) -> &[(String, String)] {
669        &self.names
670    }
671
672    /// Whether the accessor consumes its receiver (set by
673    /// [`ExpandReturnDecl::fields_self_into`]). Named `is_consuming` rather
674    /// than `consuming` — that name is already the crate-internal builder
675    /// method that sets the flag.
676    pub fn is_consuming(&self) -> bool {
677        self.consuming
678    }
679
680    /// Replace **one** field's decomposition, with the same
681    /// [`ExpandReturnDecl`] a type-level default uses — so the complete-set
682    /// rule applies here too: the decl states that field's entire leaf set.
683    /// Use it where the field's type default is not what this boundary wants
684    /// (a lone `.field_self()` keeps the raw handle instead of decomposing it).
685    ///
686    /// An override declaring **no** records states an empty leaf set: the field
687    /// **does not cross**. That is the one way to drop a field a value form
688    /// carries, and it follows from the complete-set rule rather than adding a
689    /// rule — a boundary that wants none of a field's leaves says so the same
690    /// way it says it wants some of them. (A *generator-level*
691    /// `expand_return!(T)` with no records is still an error: a type has to
692    /// cross somehow.) Use it for a field the binding has no surface for —
693    /// diagnostics a consumer never reads, a type whose accessors would drag in
694    /// a subtree nothing asks for:
695    ///
696    /// ```ignore
697    /// prebindgen_registry::expand_return!(Sample).fields_self_into(
698    ///     prebindgen_registry::fields!(sample_into_struct)
699    ///         .field("timestamp_stack", prebindgen_registry::expand_return!(TimestampStack)),
700    /// );
701    /// ```
702    pub fn field(mut self, field: impl AsRef<str>, decl: ExpandReturnDecl) -> Self {
703        let field = field.as_ref().to_string();
704        assert!(
705            !self.overrides.iter().any(|(f, _)| *f == field),
706            "fields!({}).field(\"{}\", ...): field already has an override — declare its \
707             complete field set in ONE decl",
708            self.func,
709            field
710        );
711        self.overrides.push((field, decl));
712        self
713    }
714
715    /// Rename **one** field's leaf, overriding the name derived from the struct
716    /// field ident. The literal target-language name, like `fun!(f).name(...)`.
717    pub fn name(mut self, field: impl AsRef<str>, target_name: impl Into<String>) -> Self {
718        let field = field.as_ref().to_string();
719        let target_name = target_name.into();
720        assert!(
721            !self.names.iter().any(|(f, _)| *f == field),
722            "fields!({}).name(\"{}\", ...): field is already renamed",
723            self.func,
724            field
725        );
726        // The derived names of inlined nested fields are joined with `"__"`, so
727        // an author name carrying one would forge a nesting that isn't there.
728        // (Core rejects it for a `.field()` name; here the name never reaches
729        // that check, so it is made at the point of declaration.)
730        assert!(
731            !target_name.contains("__"),
732            "fields!({}).name(\"{}\", \"{}\"): `__` is the reserved chain separator \
733             and cannot appear in a leaf name",
734            self.func,
735            field,
736            target_name,
737        );
738        self.names.push((field, target_name));
739        self
740    }
741}

745/// declarator.
746/// Deliberately **no** `impl From<syn::Type> for ExpandDecl` — a bare

771/// Declares one `#[prebindgen]` function to export. The adapter either adds it
772/// to a package or attaches it to a class as a method or a factory.
773///
774/// Build it from a bare Rust name with [`fun!`](crate::fun) and chain
775/// [`name`](Self::name) to set its target-language name.
776/// [`expand_param`](Self::expand_param) / [`expand_return`](Self::expand_return)
777/// **override, for this one function**, the boundary defaults its
778/// parameter/return types declare at the generator level
779/// — using the very same decl objects, so the complete-set rule is identical
780/// at both scopes.
781pub struct FunctionDecl {
782    rust_ident: syn::Ident,
783    name_override: Option<String>,
784    param_expands: Vec<(String, ExpandParamDecl)>,
785    return_expand: Option<ExpandReturnDecl>,
786    split_on_params: Vec<String>,
787    /// `fun!(crate::f)` — a **binding-local** fn: the declared path plus the
788    /// stated signature ([`sig`](Self::sig), required by acceptance time).
789    /// `None` = an ordinary `#[prebindgen]` registry fn.
790    local: Option<(syn::Path, Option<syn::Signature>)>,
791}
792
793impl FunctionDecl {
794    pub fn new(rust_ident: syn::Ident) -> Self {
795        Self {
796            rust_ident,
797            name_override: None,
798            param_expands: Vec::new(),
799            return_expand: None,
800            split_on_params: Vec::new(),
801            local: None,
802        }
803    }
804
805    /// The Rust-side fn ident (the path's last segment for a binding-local fn).
806    pub fn rust_ident(&self) -> &syn::Ident {
807        &self.rust_ident
808    }
809
810    /// The explicit target-language name, if `.name(...)` was declared.
811    pub fn name_override(&self) -> &Option<String> {
812        &self.name_override
813    }
814
815    /// The per-parameter expand overrides declared with `.expand_param(...)`.
816    pub fn param_expands(&self) -> &[(String, ExpandParamDecl)] {
817        &self.param_expands
818    }
819
820    /// The return expand override declared with `.expand_return(...)`, if any.
821    pub fn return_expand(&self) -> &Option<ExpandReturnDecl> {
822        &self.return_expand
823    }
824
825    /// The parameters named with `.split_on_param(...)`, in declaration order.
826    pub fn split_on_params(&self) -> &[String] {
827        &self.split_on_params
828    }
829
830    /// The binding-local fn's declared path and stated signature, if this is
831    /// a `fun!(crate::f)` rather than an ordinary `#[prebindgen]` registry fn.
832    pub fn local(&self) -> &Option<(syn::Path, Option<syn::Signature>)> {
833        &self.local
834    }
835
836    /// Consume `self` into its raw fields, by value. Not a plain accessor —
837    /// the one call site (`accept_fn_expands` in the JNI builder) destructures
838    /// a whole `FunctionDecl` it owns to move each field (`Vec`s, the `local`
839    /// signature) onward without cloning, so a reference-returning accessor
840    /// won't do.
841    #[allow(clippy::type_complexity)]
842    pub fn into_parts(
843        self,
844    ) -> (
845        syn::Ident,
846        Option<String>,
847        Vec<(String, ExpandParamDecl)>,
848        Option<ExpandReturnDecl>,
849        Vec<String>,
850        Option<(syn::Path, Option<syn::Signature>)>,
851    ) {
852        (
853            self.rust_ident,
854            self.name_override,
855            self.param_expands,
856            self.return_expand,
857            self.split_on_params,
858            self.local,
859        )
860    }
861
862    /// `fun!(crate::f)` — declare a **binding-local** fn by path. The fn
863    /// ident (the path's last segment) names it everywhere a registry fn's
864    /// ident would; chain [`sig`](Self::sig) to state its signature.
865    pub fn new_local(path: syn::Path) -> Self {
866        assert!(
867            path.segments.len() >= 2,
868            "fun!({}): a binding-local fn is called QUALIFIED from the generated file — \
869             give at least a `crate::`-rooted path (a bare ident declares a `#[prebindgen]` fn)",
870            quote::quote!(#path)
871        );
872        let ident = path.segments.last().expect("non-empty path").ident.clone();
873        Self {
874            local: Some((path, None)),
875            ..Self::new(ident)
876        }
877    }
878
879    /// State a binding-local fn's exact Rust signature (build it with
880    /// [`sig!`](crate::sig)) — a path carries no signature to read. The
881    /// parameter names become the foreign-side parameter names. Required for
882    /// a path-built [`fun!`](crate::fun); a hard error on a registry fn
883    /// (its signature is read from the registry).
884    pub fn sig(mut self, signature: syn::Signature) -> Self {
885        let Some((_, slot)) = &mut self.local else {
886            panic!(
887                "fun!({}).sig(...): a `#[prebindgen]` fn's signature is read from the \
888                 registry — .sig() applies to path-built binding-local fns (fun!(crate::f))",
889                self.rust_ident
890            );
891        };
892        assert!(
893            slot.is_none(),
894            "fun!({}).sig(...): the signature is already stated",
895            self.rust_ident
896        );
897        *slot = Some(signature);
898        self
899    }
900
901    /// Set the name this function takes in the target language. Left unset,
902    /// the adapter derives one from the Rust name — the JNI adapter, for
903    /// example, camel-cases it (`session_declare_publisher` →
904    /// `sessionDeclarePublisher`).
905    pub fn name(mut self, target_name: impl Into<String>) -> Self {
906        self.name_override = Some(target_name.into());
907        self
908    }
909
910    /// Override, for the named parameter of this function only, how that
911    /// parameter is supplied — with the same [`ExpandParamDecl`] a type-level
912    /// default uses, so the **complete-set rule** applies here too: the decl
913    /// states the entire variant set for this param (a lone `.variant_self()`
914    /// = "only a ready-made handle", replacing the type's build variants —
915    /// e.g. *un*-declaring a key expression needs the handle, not a string).
916    ///
917    /// `param` is the Rust parameter name; the decl's type is cross-checked
918    /// against that parameter's (peeled) type at generation time — an unknown
919    /// parameter or a type mismatch is a hard error. Call again with a
920    /// different `param` to override several parameters independently;
921    /// declaring the same parameter twice is a hard error.
922    pub fn expand_param(mut self, param: impl AsRef<str>, decl: ExpandParamDecl) -> Self {
923        let param = param.as_ref().to_string();
924        assert!(
925            !self.param_expands.iter().any(|(p, _)| *p == param),
926            "fun!({}).expand_param(\"{}\", ...): parameter already has an expand override — \
927             declare each parameter's complete variant set in ONE decl",
928            self.rust_ident,
929            param
930        );
931        self.param_expands.push((param, decl));
932        self
933    }
934
935    /// **Emit idiomatic typed Kotlin overloads for this parameter.** By default
936    /// a multi-variant expanded parameter crosses only as the selector tuple
937    /// (`expectedSel: Int, expected0: …, expected1: …`). `.split_on_param("p")`
938    /// additionally emits, alongside the selector wrapper, one typed overload
939    /// per variant of `p` — `f(count: Long, total: Double, …)` for a
940    /// `summary_new(count, total)` arm, `f(expected: Summary, …)` for
941    /// `variant_self()` — each delegating to the selector form.
942    ///
943    /// The parameter's variant set must be *splittable* (its arms surface as
944    /// distinct JVM signatures) — enforced up front on the
945    /// [`expand_param!`](crate::expand_param) declaration unless it opted out
946    /// with [`.no_split()`](ExpandParamDecl::no_split).
947    ///
948    /// Call again for **several** parameters: the generated overloads are then
949    /// the **cartesian product** of the named parameters' arms. That concrete
950    /// product must have no two combinations sharing a JVM signature — a hard
951    /// build error if it does.
952    ///
953    /// An `Option<…>` parameter splits through its **single-leaf** arms only
954    /// (nullable-arm rule): the overload keeps the arm's nullable type and
955    /// `null` selects absence — `f(encoding: Encoding?, …)` for a
956    /// `variant_self()` arm of an `Option<&Encoding>` parameter. Multi-leaf
957    /// arms stay selector-only; an optional parameter with no single-leaf arm
958    /// is a hard error.
959    ///
960    /// `param` is the Rust parameter name; it must be an expanded,
961    /// multi-variant parameter of this function (unknown / single-variant /
962    /// recursively-built ⇒ a hard error). Declaring the same parameter twice
963    /// is a hard error.
964    pub fn split_on_param(mut self, param: impl AsRef<str>) -> Self {
965        let param = param.as_ref().to_string();
966        assert!(
967            !self.split_on_params.contains(&param),
968            "fun!({}).split_on_param(\"{}\"): parameter is already split",
969            self.rust_ident,
970            param
971        );
972        self.split_on_params.push(param);
973        self
974    }
975
976    /// Override this function's return decomposition — with the same
977    /// [`ExpandReturnDecl`] a type-level default uses, stating the complete
978    /// field set (a lone `.field_self()` = the raw whole value, which for a
979    /// borrowed `&T` / `Option<&T>` return crosses by cloning into a fresh
980    /// owned handle). The decl's type is cross-checked against the function's
981    /// (peeled) return type at generation time — a mismatch is a hard error.
982    /// At most one per function.
983    pub fn expand_return(mut self, decl: ExpandReturnDecl) -> Self {
984        assert!(
985            self.return_expand.is_none(),
986            "fun!({}).expand_return(...): the function already has a return expand override — \
987             declare the complete field set in ONE decl",
988            self.rust_ident
989        );
990        self.return_expand = Some(decl);
991        self
992    }
993}

996// Convert decl — the canonical single-value conversion for a type
997// ──────────────────────────────────────────────────────────────────────

999/// Declares a type's **canonical single-value conversion**: how one value of
1000/// the type crosses the boundary wherever a single value is needed — as a
1001/// parameter or return, inside `Option<_>` / `Vec<_>` / the `Result<T, E>`
1002/// success position, as a `data_class` field. Each direction takes one
1003/// [`ConvertSourceDecl`]:
1004///
1005/// ```rust,ignore
1006/// .convert(convert!(Millis)
1007///     .input(fun!(millis_from_long))   // fn(u64) -> Millis    (wire → rust)
1008///     .output(fun!(millis_value))      // fn(&Millis) -> u64   (rust → wire)
1009///     .valid_range(0u64..=86_400_000)) // rejects invalid values; Option uses a niche
1010/// .convert(convert!(Celsius).input(from!(i32)).output(into!(i32)))
1011/// .convert(convert!(Label)
1012///     .input(try_from!(String).with(path!(crate::label_in)).error(ty!(String)))
1013///     .output(into!(String).with(path!(crate::label_out))))
1014/// ```
1015///
1016/// The foreign surface derives from the conversion's other-side type
1017/// (`u64` ⇒ Kotlin `ULong` / C `uint64_t`) — nothing is stated verbatim.
1018/// [`valid_range`](ConvertDecl::valid_range) and
1019/// [`valid_values`](ConvertDecl::valid_values) declare the legal subset of a
1020/// scalar representation. Generated converters validate the subset, and
1021/// adapters may reuse values outside it as allocation-free `Option`/`Result`
1022/// markers; [`exclude_values`](ConvertDecl::exclude_values) reserves holes in
1023/// an otherwise legal domain. A `try_` source's `Err`
1024/// routes to the caller's error handler. Conversion fns may live in the flat
1025/// crate or in a **helper crate** whose item stream is chained into the same
1026/// [`prebindgen_flat::Flat::builder`] parse; generated calls qualify each
1027/// function with its origin crate.
1028///
1029/// Distinct from the [`expand_param!`](crate::expand_param) /
1030/// [`expand_return!`](crate::expand_return) boundary decls: those reshape a
1031/// **function boundary** into multiple leaves (variants in / fields out),
1032/// while `convert!` defines the type's one-value form used everywhere else.
1033/// A type may declare both — expansion wins at the fn boundaries where it is
1034/// declared; the conversion serves every other position. The method names
1035/// differ deliberately: converters are direction-things ([`input`](ConvertDecl::input)
1036/// also serves callback returns, [`output`](ConvertDecl::output) also serves
1037/// callback arguments), while expansion decls are position-things.
1038/// One direction's conversion **source** — where the conversion code comes
1039/// from, the lowered form of a [`ConvertSourceDecl`].
1040// large_enum_variant: a handful of these exist per binding, held once in the
1041// builder — boxing the syn payloads would only complicate the decl arms.
1042#[allow(clippy::large_enum_variant)]
1043#[derive(Clone)]
1044pub enum ConvertSpec {
1045    /// A `#[prebindgen]` fn (flat or helper crate): the representable type
1046    /// and fallibility are read from its registry signature at lookup time.
1047    PrebindgenFn(syn::Ident),
1048    /// A `core::convert` trait impl; the representable type is stated
1049    /// explicitly (there is no signature to read). `fallible` selects
1050    /// `TryInto` (the associated `Error` routes to the caller's error
1051    /// handler) vs `Into`.
1052    Trait { repr: syn::Type, fallible: bool },
1053}
1054
1055impl ConvertSpec {
1056    /// One-line human description of the source kind (report use).
1057    pub fn describe(&self) -> String {
1058        match self {
1059            ConvertSpec::PrebindgenFn(f) => format!("`#[prebindgen]` fn `{f}`"),
1060            ConvertSpec::Trait {
1061                repr,
1062                fallible: false,
1063            } => format!("`Into` ⇄ `{}`", repr.to_token_stream()),
1064            ConvertSpec::Trait {
1065                repr,
1066                fallible: true,
1067            } => format!("`TryInto` ⇄ `{}`", repr.to_token_stream()),
1068        }
1069    }
1070}
1071
1072/// Which direction a [`ConvertSourceDecl`] was built for. The constructor
1073/// macro states it (`from!`/`try_from!` = into-Rust, `into!`/`try_into!` =
1074/// out-of-Rust) and the acceptor cross-checks it, so a chain like
1075/// `.output(from!(i32))` is a hard error instead of a silent misread.
1076///
1077/// Stated as an [`Direction`], the one type naming this axis: a `from!` builds
1078/// the Rust value out of what crossed, an `into!` takes one apart.
1079fn convert_macros(direction: Direction) -> &'static str {
1080    match direction {
1081        Direction::Construct => "from!/try_from!",
1082        Direction::Deconstruct => "into!/try_into!",
1083    }
1084}

1086/// One conversion source, accepted by [`ConvertDecl::input`] /
1087/// [`ConvertDecl::output`]. Built by [`fun!`](crate::fun) — a
1088/// `#[prebindgen]` conversion fn (bare ident, signature read from the
1089/// registry) or a **binding-local** one (`fun!(crate::f)` +
1090/// [`.sig(sig!(…))`](FunctionDecl::sig), the one vocabulary for locally
1091/// defined callables; a `Result<_, E>` return states the error channel) —
1092/// or by the direction-stating macros [`from!`](crate::from) /
1093/// [`try_from!`](crate::try_from) / [`into!`](crate::into) /
1094/// [`try_into!`](crate::try_into) (a `core::convert` **trait** conversion
1095/// with a stated representation type).
1096#[derive(Clone)]
1097pub struct ConvertSourceDecl {
1098    kind: ConvertSourceKind,
1099}

1101// large_enum_variant: a handful of these exist per binding, held transiently
1102// while a decl is built — boxing the syn payloads would only complicate the
1103// arms (same trade-off as `ConvertSpec`).
1104#[allow(clippy::large_enum_variant)]
1105#[derive(Clone)]
1106pub(crate) enum ConvertSourceKind {
1107    /// `fun!(f)` / `fun!(crate::f).sig(…)` — a conversion fn; representable
1108    /// type and fallibility are read from its signature (registry, or the
1109    /// stated one carried in `local` and synthesized before scanning).
1110    Fun {
1111        ident: syn::Ident,
1112        local: Option<(syn::Path, syn::Signature)>,
1113    },
1114    /// `from!`/`try_from!`/`into!`/`try_into!` — a stated representation
1115    /// type, converted via the `core::convert` trait.
1116    Repr {
1117        direction: Direction,
1118        fallible: bool,
1119        ty: syn::Type,
1120    },
1121}
1122
1123impl ConvertSourceDecl {
1124    fn repr(direction: Direction, fallible: bool, ty: syn::Type) -> Self {
1125        Self {
1126            kind: ConvertSourceKind::Repr {
1127                direction,
1128                fallible,
1129                ty,
1130            },
1131        }
1132    }
1133    /// `from!(T)` — input via `T: Into<Self>`.
1134    pub fn from_type(ty: syn::Type) -> Self {
1135        Self::repr(Direction::Construct, false, ty)
1136    }
1137    /// `try_from!(T)` — input via `T: TryInto<Self>`.
1138    pub fn try_from_type(ty: syn::Type) -> Self {
1139        Self::repr(Direction::Construct, true, ty)
1140    }
1141    /// `into!(T)` — output via `Self: Into<T>`.
1142    pub fn into_type(ty: syn::Type) -> Self {
1143        Self::repr(Direction::Deconstruct, false, ty)
1144    }
1145    /// `try_into!(T)` — output via `Self: TryInto<T>`.
1146    pub fn try_into_type(ty: syn::Type) -> Self {
1147        Self::repr(Direction::Deconstruct, true, ty)
1148    }
1149}
1150
1151impl From<FunctionDecl> for ConvertSourceDecl {
1152    fn from(decl: FunctionDecl) -> Self {
1153        assert!(
1154            decl.name_override.is_none()
1155                && decl.param_expands.is_empty()
1156                && decl.return_expand.is_none(),
1157            "fun!({}) as a conversion source: a conversion fn is never surfaced in \
1158             the target language — .name()/expand overrides don't apply",
1159            decl.rust_ident
1160        );
1161        let local = decl.local.map(|(path, sig)| {
1162            let Some(sig) = sig else {
1163                panic!(
1164                    "fun!({p}) as a conversion source: a binding-local fn states its \
1165                     signature — chain .sig(sig!((params) -> Ret))",
1166                    p = quote::quote!(#path)
1167                );
1168            };
1169            (path, sig)
1170        });
1171        Self {
1172            kind: ConvertSourceKind::Fun {
1173                ident: decl.rust_ident,
1174                local,
1175            },
1176        }
1177    }
1178}

1180#[derive(Clone)]
1181pub struct ConvertDecl {
1182    key: TypeKey,
1183    /// The type this declaration was **written with** — the `X` the macro
1184    /// received. Kept because the declaration is where it came from: recovering
1185    /// it later *from* the key was reasoning backwards from an identity (#291).
1186    rust_type: Origin<syn::Type>,
1187    input: Option<ConvertSpec>,
1188    output: Option<ConvertSpec>,
1189    domain: Option<crate::RepresentationDomain>,
1190    /// Binding-local fn sources declared on this convert (`fun!(crate::f)
1191    /// .sig(…)`): drained into [`Declarations::local_fns`] at acceptance so the
1192    /// synthesis pre-pass covers them.
1193    locals: Vec<(syn::Ident, syn::Path, syn::Signature)>,
1194}
1195
1196impl ConvertDecl {
1197    /// `: input …, output …` suffix for the report's conversions section.
1198    pub fn describe_sources(&self) -> String {
1199        let mut parts = Vec::new();
1200        if let Some(i) = &self.input {
1201            parts.push(format!("input {}", i.describe()));
1202        }
1203        if let Some(o) = &self.output {
1204            parts.push(format!("output {}", o.describe()));
1205        }
1206        if parts.is_empty() {
1207            String::new()
1208        } else {
1209            format!(": {}", parts.join(", "))
1210        }
1211    }
1212
1213    pub fn new(rust_type: syn::Type) -> Self {
1214        reject_builtin_convert_type(&TypeKey::from_type(&rust_type));
1215        Self {
1216            key: TypeKey::from_type(&rust_type),
1217            rust_type: declared_origin(rust_type),
1218            input: None,
1219            output: None,
1220            domain: None,
1221            locals: Vec::new(),
1222        }
1223    }
1224
1225    /// The type identity this declaration is registered under.
1226    pub fn key(&self) -> &TypeKey {
1227        &self.key
1228    }
1229
1230    /// The type this declaration was written with, as originally parsed.
1231    pub fn rust_type(&self) -> &Origin<syn::Type> {
1232        &self.rust_type
1233    }
1234
1235    /// The declared **into-Rust** conversion source, if any. Named
1236    /// `input_spec` rather than `input` — that name is already the builder
1237    /// method that declares it ([`Self::input`]).
1238    pub fn input_spec(&self) -> &Option<ConvertSpec> {
1239        &self.input
1240    }
1241
1242    /// The declared **out-of-Rust** conversion source, if any. Named
1243    /// `output_spec` rather than `output` — that name is already the builder
1244    /// method that declares it ([`Self::output`]).
1245    pub fn output_spec(&self) -> &Option<ConvertSpec> {
1246        &self.output
1247    }
1248
1249    /// The declared representation-domain restriction, if any.
1250    pub fn domain(&self) -> &Option<crate::RepresentationDomain> {
1251        &self.domain
1252    }
1253
1254    /// Binding-local fn sources declared on this convert, drained into the
1255    /// synthesis pre-pass at acceptance.
1256    pub fn locals(&self) -> &[(syn::Ident, syn::Path, syn::Signature)] {
1257        &self.locals
1258    }
1259
1260    /// Mutable access for draining binding-local fn sources into the
1261    /// synthesis pre-pass at acceptance (`Vec::append`).
1262    pub fn locals_mut(&mut self) -> &mut Vec<(syn::Ident, syn::Path, syn::Signature)> {
1263        &mut self.locals
1264    }
1265
1266    fn set_input(mut self, spec: ConvertSpec) -> Self {
1267        assert!(
1268            self.input.is_none(),
1269            "convert!({}): the input conversion is already declared — \
1270             declare each direction's conversion in ONE .input()/.output() call",
1271            self.key.as_str()
1272        );
1273        self.input = Some(spec);
1274        self
1275    }
1276
1277    fn set_output(mut self, spec: ConvertSpec) -> Self {
1278        assert!(
1279            self.output.is_none(),
1280            "convert!({}): the output conversion is already declared — \
1281             declare each direction's conversion in ONE .input()/.output() call",
1282            self.key.as_str()
1283        );
1284        self.output = Some(spec);
1285        self
1286    }
1287
1288    fn check_repr(&self, method: &str, repr: &syn::Type) {
1289        assert!(
1290            TypeKey::from_type(repr) != self.key,
1291            "convert!({k}).{method}: the representable type must differ from `{k}` itself",
1292            k = self.key.as_str()
1293        );
1294    }
1295
1296    /// Lower an accepted [`ConvertSourceDecl`] to the internal spec,
1297    /// cross-checking the source's stated direction against the acceptor. A
1298    /// binding-local fn source records its `(ident, path, sig)` in
1299    /// [`Self::locals`] for the synthesis pre-pass — after which it lowers
1300    /// exactly like a `#[prebindgen]` fn source.
1301    fn spec_of(
1302        &mut self,
1303        direction: Direction,
1304        method: &str,
1305        src: ConvertSourceDecl,
1306    ) -> ConvertSpec {
1307        match src.kind {
1308            ConvertSourceKind::Fun { ident, local } => {
1309                if let Some((path, sig)) = local {
1310                    self.locals.push((ident.clone(), path, sig));
1311                }
1312                ConvertSpec::PrebindgenFn(ident)
1313            }
1314            ConvertSourceKind::Repr {
1315                direction: stated,
1316                fallible,
1317                ty,
1318            } => {
1319                assert!(
1320                    stated == direction,
1321                    "convert!({k}).{method}(...): the source was built with {got} — \
1322                     an {method} conversion is built with {want}",
1323                    k = self.key.as_str(),
1324                    got = convert_macros(stated),
1325                    want = convert_macros(direction),
1326                );
1327                self.check_repr(method, &ty);
1328                ConvertSpec::Trait { repr: ty, fallible }
1329            }
1330        }
1331    }
1332
1333    /// The **into-Rust** conversion (parameters, callback returns): how a
1334    /// value of this type is built from its representation. Accepts
1335    /// [`fun!`](crate::fun) (a `#[prebindgen]` `fn(U) -> T` /
1336    /// `fn(U) -> Result<T, E>`) or [`from!`](crate::from) /
1337    /// [`try_from!`](crate::try_from) (`Repr: Into<T>` / `TryInto`, or a
1338    /// binding-local callable via `.with(...)`).
1339    pub fn input(mut self, src: impl Into<ConvertSourceDecl>) -> Self {
1340        let spec = self.spec_of(Direction::Construct, "input", src.into());
1341        self.set_input(spec)
1342    }
1343
1344    /// The **out-of-Rust** conversion (returns, callback arguments): how a
1345    /// value of this type is turned into its representation. Accepts
1346    /// [`fun!`](crate::fun) (a `#[prebindgen]` `fn(&T) -> U` / `fn(T) -> U`)
1347    /// or [`into!`](crate::into) / [`try_into!`](crate::try_into)
1348    /// (`T: Into<Repr>` / `TryInto`, or a binding-local callable via
1349    /// `.with(...)`).
1350    pub fn output(mut self, src: impl Into<ConvertSourceDecl>) -> Self {
1351        let spec = self.spec_of(Direction::Deconstruct, "output", src.into());
1352        self.set_output(spec)
1353    }
1354
1355    /// Restrict the scalar representation to a numeric range. Values outside
1356    /// the range are rejected and may be reused by wrappers such as `Option`.
1357    /// Floating-point ranges reject every NaN; use [`Self::valid_values`] when
1358    /// exact raw IEEE values (including a specific NaN payload) are intended.
1359    pub fn valid_range<T, R>(mut self, range: R) -> Self
1360    where
1361        T: crate::DomainScalar,
1362        R: ::core::ops::RangeBounds<T>,
1363    {
1364        assert!(
1365            self.domain.is_none(),
1366            "convert!({}): the representation domain is already declared",
1367            self.key.as_str()
1368        );
1369        self.domain = Some(crate::RepresentationDomain::range(range));
1370        self
1371    }
1372
1373    /// Restrict the scalar representation to a finite valid-value set. Float
1374    /// membership uses raw IEEE bits, so `0.0`, `-0.0`, and NaN payloads remain
1375    /// distinct.
1376    pub fn valid_values<T>(mut self, values: impl IntoIterator<Item = T>) -> Self
1377    where
1378        T: crate::DomainScalar,
1379    {
1380        assert!(
1381            self.domain.is_none(),
1382            "convert!({}): the representation domain is already declared",
1383            self.key.as_str()
1384        );
1385        self.domain = Some(crate::RepresentationDomain::values(values));
1386        self
1387    }
1388
1389    /// Remove finite values from the previously declared base domain. Float
1390    /// exclusions use raw IEEE bits.
1391    pub fn exclude_values<T>(mut self, values: impl IntoIterator<Item = T>) -> Self
1392    where
1393        T: crate::DomainScalar,
1394    {
1395        self.domain
1396            .as_mut()
1397            .unwrap_or_else(|| {
1398                panic!(
1399                    "convert!({}): .exclude_values(...) requires .valid_range(...) \
1400                     or .valid_values(...) first",
1401                    self.key.as_str()
1402                )
1403            })
1404            .exclude(values);
1405        self
1406    }
1407}

1409/// Rejects a `convert!` declaration on a Rust **builtin** type: builtins
1410/// already have their own converters, and the generated calls would try to
1411/// qualify the builtin with a crate path. Wrap the builtin in a source-crate
1412/// newtype (like `Millis(u64)`) instead.
1413fn reject_builtin_convert_type(key: &TypeKey) {
1414    const BUILTINS: &[&str] = &[
1415        "usize", "isize", "u8", "u16", "u32", "u64", "u128", "i8", "i16", "i32", "i64", "i128",
1416        "f32", "f64", "bool", "char", "str", "String",
1417    ];
1418    assert!(
1419        !BUILTINS.contains(&key.as_str()),
1420        "convert!({}): builtins already have converters — wrap the builtin in a newtype instead",
1421        key.as_str()
1422    );
1423}

1425/// Bare-ident type `__JniErr` — the generated file's alias for the
1426/// `prebindgen-jni` crate's `JniBindingError` framework type. Built-in
1427/// converters use this as their `Result<…, _>` error type so their bodies'
1428/// `<__JniErr as From<String>>::from(...)` calls keep compiling. A
1429/// `Result<T, E>` return instead binds its own raw `E` (see
1430/// `JniGenBuilder::lookup_output`); the extern's `Err` arm funnels both to the
1431/// per-call `signal_error` sink via `E: Display`.
1432/// The origin-module prefix of a binding-local fn's declared path
1433/// (`crate::sub::f` → `"crate::sub"`). Paths are validated ≥2 segments at
1434/// decl time (`fun!` path arm / `FieldDecl::with`), so the prefix is
1435/// always non-empty.
1436pub fn local_path_prefix(path: &syn::Path) -> String {
1437    path.segments
1438        .iter()
1439        .take(path.segments.len() - 1)
1440        .map(|s| s.ident.to_string())
1441        .collect::<Vec<_>>()
1442        .join("::")
1443}
