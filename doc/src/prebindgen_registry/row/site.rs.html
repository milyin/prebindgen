1//! Which row a given place in the generated API uses.
2//!
3//! [`Rows`] says what rows a crossing has; this says which of them the
4//! second parameter of `z_put`, or the `Err` arm of a return, or one part of
5//! another row actually takes. A declaration states that through
6//! [`BindingsBuilder::bind`], and [`Bindings::resolve`] answers it for one
7//! site.
8//!
9//! A site nobody binds uses its crossing's default row, so the common case is
10//! declared nowhere.
11

19/// One place in the generated API where a value crosses.
20#[derive(Clone, Debug, PartialEq, Eq, Hash)]
21pub struct Site {
22    /// The item the site belongs to: the exported function for a parameter or
23    /// a return, the crossed type for a part.
24    pub owner: syn::Ident,
25    /// Which place within that item.
26    pub role: Role,
27}
28
29impl Site {
30    /// The site of one part of `of`'s row `recipe`.
31    ///
32    /// The one `Site` an adapter must be able to build **exactly**, because the
33    /// driver builds it too and the two have to meet: a per-part binding is
34    /// found by this key or not at all. Its `owner` is derived from the crossed
35    /// type rather than chosen, so composing one by hand is a guess — this is
36    /// the answer instead.
37    pub fn part(of: &Crossing, recipe: &RowId, index: usize) -> Self {
38        Self::arm_part(of, recipe, None, index)
39    }
40
41    /// The site of one part of `of`'s row `recipe`, inside alternative `arm`.
42    ///
43    /// [`Self::part`] with the alternative stated. A part of a
44    /// [`Shape::Choice`](super::Shape::Choice) row needs it, because every arm
45    /// numbers its parts from zero.
46    pub fn arm_part(of: &Crossing, recipe: &RowId, arm: Option<usize>, index: usize) -> Self {
47        Self {
48            owner: of
49                .value()
50                .stripped_key()
51                .ident()
52                .unwrap_or_else(|| syn::Ident::new("_", proc_macro2::Span::call_site())),
53            role: Role::Part {
54                of: of.key(),
55                recipe: recipe.clone(),
56                arm,
57                index,
58            },
59        }
60    }
61}
62
63impl fmt::Display for Site {
64    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
65        write!(f, "{}'s {}", self.owner, self.role)
66    }
67}

69/// Which place within an item a site is.
70#[derive(Clone, Debug, PartialEq, Eq, Hash)]
71pub enum Role {
72    /// One parameter of an exported function.
73    ///
74    /// Identified by position and not by name: a `Site` is a map key, so
75    /// anything on it has to be something two callers cannot spell differently.
76    /// The parameter's name is the model's, reached through `owner`.
77    Param {
78        /// Position in the parameter list.
79        index: usize,
80    },
81    /// The value a method is called on.
82    Receiver,
83    /// What the function returns.
84    Return,
85    /// The `Err` arm of a `Result`.
86    Error,
87    /// One argument the Rust side passes out through a callback.
88    ///
89    /// Swaps jobs: Rust holds the value and pushes it out, so the argument is
90    /// deconstructed even though it sits in a parameter list.
91    ///
92    /// A **root** role, like [`Return`](Self::Return) and
93    /// [`Param`](Self::Param): it names an argument of the callback in one
94    /// exported function, so it is what an adapter passes to
95    /// [`Compiler::site`](super::Compiler::site) when it compiles that argument
96    /// itself. It is deliberately not what the driver asks at while compiling a
97    /// callback row — a row is shared by every function whose callback has the
98    /// same signature, so a per-function answer could not apply to it. Inside a
99    /// row the driver asks at [`Part`](Self::Part), which is keyed by the row.
100    CallbackArg {
101        /// Which parameter of the exported function is the callback.
102        param: usize,
103        /// Which argument of that callback.
104        arg: usize,
105    },
106    /// One part of another crossing's recipe.
107    Part {
108        /// The crossing whose row names this part.
109        of: CrossingKey,
110        /// Which of that crossing's rows.
111        recipe: RowId,
112        /// Which alternative, for a part inside a [`Shape::Choice`](super::Shape::Choice)
113        /// arm; `None` for a product's own part.
114        ///
115        /// Load-bearing rather than decorative: **every arm numbers its parts
116        /// from zero**, so `part 0` of a two-armed sum names two different
117        /// parts of two different types. Without the alternative there is no
118        /// key that tells them apart, and a binding written for one silently
119        /// collides with the other.
120        arm: Option<usize>,
121        /// The part's position within the row — or within its arm.
122        index: usize,
123    },
124    /// A `#[prebindgen]` constant's value.
125    Const,
126}
127
128impl fmt::Display for Role {
129    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
130        match self {
131            Role::Param { index } => write!(f, "parameter {index}"),
132            Role::Receiver => f.write_str("receiver"),
133            Role::Return => f.write_str("return value"),
134            Role::Error => f.write_str("error arm"),
135            Role::CallbackArg { param, arg } => {
136                write!(f, "argument {arg} of the callback in parameter {param}")
137            }
138            Role::Part {
139                of,
140                recipe,
141                arm,
142                index,
143            } => match arm {
144                Some(arm) => write!(f, "part {index} of arm {arm} of row `{recipe}` of {of}"),
145                None => write!(f, "part {index} of row `{recipe}` of {of}"),
146            },
147            Role::Const => f.write_str("value"),
148        }
149    }
150}

152/// What a declaration asks for at one site.
153#[derive(Clone, Debug, PartialEq, Eq)]
154pub enum Ask {
155    /// The crossing's default row.
156    Default,
157    /// This named row of the crossing.
158    Row(RowId),
159    /// This site contributes nothing.
160    Omit,
161}
162
163/// Which declaration made the ask.
164///
165/// Ordered highest precedence first, so the winner of two asks for one site is
166/// the smaller of the two.
167///
168/// Not to be confused with [`flat::Origin`](crate::flat::Origin), which is a
169/// captured item's own syntax.
170#[derive(Copy, Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
171pub enum Origin {
172    /// A per-function override, written against one exported function.
173    Function,
174    /// A per-part declaration, written against one part of one row.
175    Part,
176    /// The type's own default, written against the type.
177    Type,
178    /// What the adapter would otherwise have picked.
179    Adapter,
180}
181
182impl fmt::Display for Origin {
183    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
184        f.write_str(match self {
185            Origin::Function => "a per-function declaration",
186            Origin::Part => "a per-part declaration",
187            Origin::Type => "the type's own declaration",
188            Origin::Adapter => "the adapter",
189        })
190    }
191}

193/// What the table hands an adapter for one site. Immutable.
194#[derive(Clone, Debug)]
195pub struct Bound {
196    /// Where the value crosses.
197    pub site: Site,
198    /// What crosses, and which way.
199    pub crossing: Crossing,
200    /// Which of the crossing's rows this site takes.
201    pub recipe: RowId,
202    /// Which declaration won, so a diagnostic can say why this row applies.
203    pub origin: Origin,
204}
205
206#[derive(Clone, Debug)]
207struct Binding {
208    crossing: Crossing,
209    ask: Ask,
210    origin: Origin,
211}

213/// Describes the [`Bindings`] of one binding crate.
214#[derive(Default)]
215pub struct BindingsBuilder {
216    asks: HashMap<Site, Binding>,
217    /// Two declarations of equal precedence that disagree, reported by
218    /// [`Self::build`] rather than resolved by declaration order. One entry per
219    /// site, so a site declared three ways is reported once.
220    conflicts: BTreeMap<String, (Site, Origin)>,
221}
222
223impl BindingsBuilder {
224    /// State which row one site takes.
225    ///
226    /// Where two declarations bind one site, the higher-precedence `origin`
227    /// wins — with its `crossing` and its `ask` together, since the two are one
228    /// declaration's answer. Two declarations of equal precedence asking for
229    /// different rows is an error, and two asking for the same row is not.
230    pub fn bind(&mut self, site: Site, crossing: Crossing, ask: Ask, origin: Origin) -> &mut Self {
231        let binding = Binding {
232            crossing,
233            ask,
234            origin,
235        };
236        match self.asks.get(&site) {
237            // A weaker declaration never displaces a stronger one.
238            Some(held) if held.origin < binding.origin => {}
239            Some(held) if held.origin == binding.origin => {
240                // The whole answer has to match, not only the ask: two
241                // declarations naming different crossings disagree just as much
242                // as two naming different rows.
243                if held.ask != binding.ask || held.crossing.key() != binding.crossing.key() {
244                    self.conflicts.insert(site.to_string(), (site, origin));
245                }
246            }
247            _ => {
248                self.asks.insert(site, binding);
249            }
250        }
251        self
252    }
253
254    /// Check every ask against the table and freeze the result.
255    ///
256    /// The one check here is that a site naming a row asks for one the crossing
257    /// has. Whether a crossing with several rows says which of them is the
258    /// default is [`RowsBuilder::build`](super::RowsBuilder::build)'s.
259    pub fn build(self, recipes: &Rows) -> Result<Bindings, Vec<RowError>> {
260        let mut errors: Vec<RowError> = self
261            .conflicts
262            .into_values()
263            .map(|(site, origin)| RowError::Rebound { site, origin })
264            .collect();
265        let mut bound = HashMap::new();
266        for (site, binding) in self.asks {
267            let key = binding.crossing.key();
268            let recipe = match &binding.ask {
269                Ask::Omit => {
270                    bound.insert(site, None);
271                    continue;
272                }
273                Ask::Default => recipes.row(&binding.crossing).0,
274                Ask::Row(id) => {
275                    if recipes.get(&key, id).is_none() {
276                        errors.push(RowError::UnknownRow {
277                            site,
278                            crossing: key,
279                            recipe: id.clone(),
280                        });
281                        continue;
282                    }
283                    id.clone()
284                }
285            };
286            bound.insert(
287                site.clone(),
288                Some(Bound {
289                    site,
290                    crossing: binding.crossing,
291                    recipe,
292                    origin: binding.origin,
293                }),
294            );
295        }
296        if errors.is_empty() {
297            Ok(Bindings { bound })
298        } else {
299            Err(errors)
300        }
301    }
302}

304/// Which row every declared site takes. Built by [`BindingsBuilder`], checked
305/// once, then immutable.
306#[derive(Debug, Default)]
307pub struct Bindings {
308    /// `None` for a site a declaration omitted, which is a different answer
309    /// from a site nobody bound.
310    bound: HashMap<Site, Option<Bound>>,
311}
312
313impl Bindings {
314    /// Start describing the sites of one binding crate.
315    pub fn builder() -> BindingsBuilder {
316        BindingsBuilder::default()
317    }
318
319    /// The row this site takes.
320    ///
321    /// `None` when a declaration bound the site to [`Ask::Omit`]. A site nobody
322    /// bound takes its crossing's default row, attributed to
323    /// [`Origin::Adapter`], so an adapter asks here for every site rather than
324    /// checking first whether one was declared.
325    pub fn resolve(&self, site: &Site, crossing: &Crossing, recipes: &Rows) -> Option<Bound> {
326        match self.bound.get(site) {
327            Some(bound) => {
328                // A site is one place, so the crossing a caller asks with must
329                // be the one the declaration bound. Getting a plausible answer
330                // back for the wrong type would hide the caller's bug.
331                debug_assert!(
332                    bound
333                        .as_ref()
334                        .is_none_or(|b| b.crossing.key() == crossing.key()),
335                    "{site} was bound as {} and is being resolved as {}",
336                    bound
337                        .as_ref()
338                        .map(|b| b.crossing.key().to_string())
339                        .unwrap_or_default(),
340                    crossing.key(),
341                );
342                bound.clone()
343            }
344            None => Some(Bound {
345                site: site.clone(),
346                recipe: recipes.row(crossing).0,
347                crossing: crossing.clone(),
348                origin: Origin::Adapter,
349            }),
350        }
351    }
352
353    /// Whether any declaration named this site.
354    pub fn is_declared(&self, site: &Site) -> bool {
355        self.bound.contains_key(site)
356    }
