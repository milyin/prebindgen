1//! Generating Rust field groups from explicit Flat shape facts.
2//!
3//! No helper here reads an origin or retained `syn` node. The frontend records
4//! delimiter kind as [`FieldShape`], and final generation switches only on
5//! that model fact.
6
7use proc_macro2::TokenStream;
8use quote::{quote, ToTokens};
9
10use super::element::{Field, FieldShape};
11
12/// Spell a field group: `head`, `head(parts…)` or `head { parts… }`, following
13/// the delimiters the source wrote.
14///
15/// The one place those delimiters are chosen — for match patterns and
16/// constructors alike, in either direction, for a struct and a variant alike.
17/// `B()` carries no payload and still must be written `E::B()` wherever Rust
18/// names it.
19///
20/// `head` is the type's or variant's path, and each part is an already-rendered
21/// [`Field::bind`].
22pub(crate) fn fields(shape: FieldShape, head: TokenStream, parts: &[TokenStream]) -> TokenStream {
23    match shape {
24        FieldShape::Unit => head,
25        FieldShape::Tuple => quote!(#head(#(#parts),*)),
26        FieldShape::Named => quote!(#head { #(#parts),* }),
27    }
28}
29
30impl Field {
31    /// How the field is addressed in a pattern or an initializer: by name when
32    /// it has one, else by position.
33    ///
34    /// Ungated, and the reason is what it reads: [`name`](Self::name) and
35    /// [`index`](Self::index), both model facts. No captured syntax is
36    /// involved.
37    pub fn member(&self) -> syn::Member {
38        match &self.name {
39            Some(id) => syn::Member::Named(id.clone()),
40            None => syn::Member::Unnamed(syn::Index::from(self.index)),
41        }
42    }
43
44    /// The field bound to `bind`, shaped for whichever address it uses —
45    /// `id: __f0` for a named field, `__f0` for a positional one. The part a
46    /// spelled fields list is built from.
47    pub fn bind(&self, bind: &impl ToTokens) -> TokenStream {
48        match &self.name {
49            Some(id) => quote!(#id: #bind),
50            None => quote!(#bind),
51        }
52    }
53}
