1//! Extended `#[prebindgen]` surface primarily used to exercise language-binding
2//! generator features that the lean performance surface in [`crate`] does not
3//! need. Most items exist so a *coverage* binding (e.g.
4//! `examples/covertest-kotlin`) can map one flat library through **every**
5//! adapter feature and assert the result; the `ObjectBoundary*` family also
6//! supports the flattened-vs-`JObject` JNI input micro-benchmark.
7//!
8//! Everything here is re-exported at the crate root (`pub use ext::*`), so a
9//! single `source_module = perftest_flat` reaches both the perf surface and this
10//! one. The items extend the same Storage/Payload domain with light
11//! "introspection / analytics" helpers:
12//!
13//! * [`Priority`] — a `#[repr(i32)]` enum (→ Kotlin `enum class`).
14//! * [`Stamp`] — a small `Copy` value crossing as its scalar fields (→ Kotlin
15//!   `data class`); `Vec<Stamp>` surfaces as `List<Stamp>`.
16//! * [`StorageError`] — the `E` of a fallible `Result` (→ the `onError` channel).
17//! * [`Summary`] — an opaque handle whose fields decompose at the boundary
18//!   (→ flatten-input / flatten-output).
19//! * [`Millis`] — a newtype crossing as a plain `Long` via a custom
20//!   input/output wrapper.
21//! * [`Duration`] — the standard-library semantic type crossing as bounded
22//!   milliseconds, with `Option<Duration>` using an invalid representation as
23//!   an allocation-free niche.

135/// Coarse importance bucket derived from a payload's `value`. A C-like
136/// `#[repr(i32)]` enum with explicit discriminants, mapped by the binding to a
137/// Kotlin `enum class` (and a C enum).
138#[prebindgen]
139#[repr(i32)]
140#[derive(Clone, Copy, Debug, PartialEq, Eq)]
141pub enum Priority {
142    Low = 0,
143    Normal = 1,
144    High = 2,
145}
146
147/// Classify a payload by magnitude of its `value` (enum **return**).
148#[prebindgen]
149pub fn payload_priority(p: &Payload) -> Priority {
150    let v = p.value.abs();
151    if v < 10.0 {
152        Priority::Low
153    } else if v < 100.0 {
154        Priority::High
155    } else {
156        Priority::Normal
157    }
158}
159
160/// Numeric weight of a priority (enum **by-value parameter**).
161#[prebindgen]
162pub fn priority_weight(p: Priority) -> i32 {
163    match p {
164        Priority::Low => 1,
165        Priority::Normal => 5,
166        Priority::High => 10,
167    }
168}

170/// Resolve an optional priority against a fallback (`Option<enum>` parameter).
171#[prebindgen]
172pub fn priority_or(p: Option<Priority>, fallback: Priority) -> Priority {
173    p.unwrap_or(fallback)
174}

204/// A sensor reading: exactly **one** of these alternatives is live, and it
205/// carries that alternative's payload. Written as plain Rust — the "exactly
206/// one of" invariant is in the type, not in a doc comment on a struct of
207/// optional fields.
208///
209/// All four variant shapes a sum can take are here: a payload-less variant, a
210/// single-payload tuple variant, a multi-field named variant, and a tuple
211/// variant whose payloads include a declared `enum_class`. The binding maps it
212/// to a Kotlin `sealed interface` with the variants nested inside
213/// (`lang::JniGenBuilder` `sealed_class!`).
214#[prebindgen]
215#[derive(Clone, Debug, PartialEq)]
216pub enum Reading {
217    /// No reading — the empty payload group; only the tag is live.
218    Missing,
219    /// An exact value (single-payload tuple variant).
220    Exact(i64),
221    /// A bounded interval (multi-field named variant).
222    Range { low: i64, high: i64 },
223    /// A described reading: a `String` beside a declared `enum_class` payload.
224    Labeled(String, Priority),
225    /// A variant whose name collides with the Kotlin **companion object** the
226    /// binding emits to hold `fromParts`. The source crate keeps the name it
227    /// wants: `Companion` is not reserved by Kotlin, it is the generator's own
228    /// default, so the generator renames *its* companion instead.
229    Companion(i64),
230}

232/// A data class carrying a **sum** as a field — the position where "exactly one
233/// of" composes with ordinary product data.
234///
235/// `reading` is required and `fallback` optional, so the binding has to gate a
236/// tag *and* a present flag independently; both sit beside already-flattened
237/// siblings (`id`, `note`) so the tag-gated groups must interleave correctly
238/// with ordinary leaves rather than only working in isolation. `Reading`'s
239/// `Labeled` arm carries a `String`, which is the payload that proves an inert
240/// group's object slot is wire-defaulted to null and therefore nullable in the
241/// generated `fromParts`.
242#[prebindgen]
243#[derive(Clone, Debug, PartialEq)]
244pub struct Observation {
245    pub id: i64,
246    pub reading: Reading,
247    pub fallback: Option<Reading>,
248    pub note: String,
249}
250
251/// The `Reading` alternative selected by `which` (declaration order, so it is
252/// the same numbering as the generated tag).
253fn reading_for(which: i32) -> Reading {
254    match which {
255        0 => Reading::Missing,
256        1 => Reading::Exact(42),
257        2 => Reading::Range { low: 1, high: 9 },
258        3 => Reading::Labeled("warm".to_string(), Priority::High),
259        _ => Reading::Companion(5),
260    }
261}

263/// Build an [`Observation`] carrying the selected alternative, optionally with
264/// a `fallback` (the next alternative round-robin) — a **sum as a struct
265/// field** crossing Rust → Kotlin, required and optional in one value.
266#[prebindgen]
267pub fn observation_new(which: i32, with_fallback: bool) -> Observation {
268    Observation {
269        id: 7,
270        reading: reading_for(which),
271        fallback: with_fallback.then(|| reading_for((which + 1) % 5)),
272        note: "obs".to_string(),
273    }
274}

365/// A sum whose alternatives are a **payload-less** variant and one carrying an
366/// opaque **handle** — the shape a real lookup/reply result takes, and the one
367/// that proves a tag-gated group can own a native resource: the live group
368/// hands over a fresh handle, the inert group's slot stays a null pointer that
369/// is never wrapped.
370#[prebindgen]
371#[derive(Clone)]
372pub enum Lookup {
373    /// Nothing matched — only the tag is live.
374    Absent,
375    /// What matched, as a handle the caller owns (and must close).
376    Found(Summary),
377    /// Why the lookup could not run — a `String` beside the handle group, so an
378    /// inert object slot is exercised alongside an inert primitive one.
379    Failed(String),
380}
381
382/// Build a [`Lookup`]: `count < 0` is a failure, `count == 0` is absent,
383/// anything else is found.
384#[prebindgen]
385pub fn lookup_of(count: i64, total: f64) -> Lookup {
386    match count {
387        c if c < 0 => Lookup::Failed("negative count".to_string()),
388        0 => Lookup::Absent,
389        c => Lookup::Found(summary_new(c, total)),
390    }
391}

393/// The payload shapes whose **layers** a sum's raw builder has to carry.
394///
395/// Between a wire slot and the property there can be a collection, an `Option`,
396/// and the leaf the wrap knows about, and the builder applied the leaf's
397/// conversion straight to the slot — so a null was dropped, a handle was minted
398/// from a nullable slot, and an element conversion ran on the list itself
399/// (#429). Every one of those is a Kotlin compile error; the Rust half is
400/// identical either way, which is why only a compiled harness can hold the
401/// line.
402///
403/// The last two alternatives are the controls, and they matter as much as the
404/// first three: distributing over a collection is right only when its elements
405/// convert. `Vec<u8>` surfaces as a Kotlin `ByteArray`, and mapping the
406/// identity over one produces a `List<Byte>` — the property's type replaced
407/// rather than preserved.
408#[prebindgen]
409#[derive(Clone)]
410pub enum Layered {
411    /// `Option<u64>` — JVM null is the absent case, so the unsigned conversion
412    /// has to be null-safe rather than applied through it.
413    Count(Option<u64>),
414    /// `Option<Summary>` — the same, over a handle that must be **minted** in
415    /// the present case and left alone in the absent one.
416    Held(Option<Summary>),
417    /// `Vec<Option<u64>>` — the absences are *inside* the list, so the slot is
418    /// un-inerted and the conversion runs per element.
419    Many(Vec<Option<u64>>),
420    /// A run under an `Option` — the two layers in the other order. Looking for
421    /// the run without peeling the `Option` first finds none, and the element
422    /// conversion lands on the list.
423    Values(Option<Vec<Option<u64>>>),
424    /// Layers nest: the element of this run is another run, so the leaf's
425    /// conversion belongs two levels down and a walk that unrolls a fixed two
426    /// applies it one level too high.
427    Nested(Vec<Vec<Option<u64>>>),
428    /// A control: a run of values that needs no element conversion, and whose
429    /// Kotlin type is not a `List` at all.
430    Blob(Vec<u8>),
431    /// A control: no layer between the slot and the property.
432    Plain(i64),
433}
434
435/// Build a [`Layered`], one `which` per alternative and per case within it:
436/// `0` absent count, `1` present count, `2` absent handle, `3` present handle,
437/// `4` a list mixing present and absent, `5` an absent run, `6` a present run,
438/// `7` a run of runs, `8` a byte run, anything else `Plain`.
439#[prebindgen]
440pub fn layered_of(which: i32) -> Layered {
441    match which {
442        0 => Layered::Count(None),
443        1 => Layered::Count(Some(4)),
444        2 => Layered::Held(None),
445        3 => Layered::Held(Some(summary_new(4, 8.0))),
446        4 => Layered::Many(vec![Some(1), None, Some(3)]),
447        5 => Layered::Values(None),
448        6 => Layered::Values(Some(vec![Some(5), None])),
449        7 => Layered::Nested(vec![vec![Some(6), None], vec![]]),
450        8 => Layered::Blob(vec![1, 2, 3]),
451        _ => Layered::Plain(7),
452    }
453}

494/// A handle-carrying sum as a **callback argument** — the same `Lookup` that
495/// [`lookup_of`] returns, arriving through `impl Fn` instead. Alternatives are
496/// delivered in `count` order starting at `-1`, so `n >= 3` covers all three:

505    for i in 0..n {
506        sink(lookup_of(i - 1, total));
507    }

602/// A value whose equality is **array-backed** on the JVM side: a byte-array
603/// field beside a nested data class.
604///
605/// Kotlin arrays compare by identity, so the `Vec<u8>` field would make two
606/// equal-content values compare unequal unless the binding emits content-based
607/// operators. This mirrors the shape that broke downstream (a `Vec<u8>` struct
608/// field), which nothing else here exercised. Two fields are enough: `id` is
609/// array-backed and the nested [`Stamp`] — which itself crosses as its scalar
610/// fields — is not, so both comparison branches are covered, and a third of
611/// either kind would only repeat an emitted form.
612///
613/// Field ORDER is deliberate: the array-backed fields come after `stamp`, so
614/// the generated `hashCode` folds them as `31 * result + id.contentHashCode()`
615/// rather than seeding the accumulator with them. That is the shape a real
616/// value takes (`Timestamp(ntp64, id)`), and it is a different emitted form
617/// from the array-first one.
618#[prebindgen]
619#[derive(Clone, Debug, PartialEq, Eq)]
620pub struct BlobValue {

2377/// `match` belongs inside the arm that binds the report.
2378#[prebindgen]
2379pub fn ledger_each(n: i64, sink: impl Fn(Ledger) + Send + Sync + 'static) {
2380    for i in 0..n {
2381        sink(ledger_new(i));
2382    }
2383}
