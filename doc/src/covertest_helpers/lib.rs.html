1//! Binding-side conversion helpers for `covertest-kotlin` — a **separate
2//! `#[prebindgen]` source crate** layered on top of [`perftest_flat`].
3//!
4//! This crate exists to prove the multi-source model: a binding crate's
5//! `build.rs` chains SEVERAL prebindgen source streams into one registry
6//! (`Flat::builder().items(flat.items_all()).items(helpers.items_all())`) and
7//! the generated Rust qualifies each function with its origin crate.
8//! covertest-kotlin additionally RENAMES this dependency
9//! (`cov_helpers = { package = "covertest-helpers", .. }`) and overrides the
10//! stamped origin via `Source::builder(..).crate_name("cov_helpers")` — so
11//! generated calls read `perftest_flat::…` vs `cov_helpers::…`, proving the
12//! per-source rename escape hatch.
13//!
14//! Why a separate crate at all: `#[prebindgen]` markers are captured during a
15//! crate's own compilation, which happens *after* its `build.rs` runs — so a
16//! binding crate can never feed its own markers into its own generation.
17//! Helper conversions therefore live either in the flat crate itself or in a
18//! small helper crate like this one, consumed as both a normal and a build
19//! dependency (exactly like the flat crate).
20
21use perftest_flat::{summary_total, Millis, Summary};
22use prebindgen_proc_macro::{features, prebindgen};
23
24/// Output directory with the prebindgen data of this crate.
25pub const PREBINDGEN_OUT_DIR: &str = prebindgen_proc_macro::prebindgen_out_dir!();
26
27/// Enabled Cargo features of this crate (referenced by the generated
28/// `konst` feature guard, like every prebindgen source crate).
29pub const FEATURES: &str = features!();

31/// Canonical input conversion for [`Millis`]: build it from the raw
32/// millisecond count. Referenced by covertest's
33/// `convert!(Millis).input(fun!(millis_from_long))`.
34#[prebindgen]
35pub fn millis_from_long(v: i64) -> Millis {
36    Millis(v as u64)
37}
38
39/// Canonical output conversion for [`Millis`]: read the raw millisecond
40/// count. Referenced by covertest's
41/// `convert!(Millis).output(fun!(millis_value))`.
42#[prebindgen]
43pub fn millis_value(m: &Millis) -> i64 {
44    m.0 as i64
45}
46
47/// Optionally-supplied summary total — exercises the **Optional
48/// combined-selector expansion**: `Summary`'s dual-arm type default (build
49/// from `(count, total)` OR pass a handle) applies to this `Option<&Summary>`
50/// parameter, so the selector also encodes absence (`-1` = `None`). Returns
51/// `-1.0` when absent.
52#[prebindgen]
53pub fn summary_total_opt(s: Option<&Summary>) -> f64 {
54    s.map(summary_total).unwrap_or(-1.0)
55}
56
57/// Two-`Summary`-parameter function exercising #52's **cartesian-product**
58/// overloads: covertest declares `.split_on_param("primary")
59/// .split_on_param("fallback")`, so each parameter is independently split into
60/// its (count, total)-build / handle arms and the generator emits the 2×2
61/// product (all four combinations distinct). Returns `1` when `primary` has the
62/// larger total, else `0`.
63#[prebindgen]
64pub fn summary_prefer(primary: Summary, fallback: Summary) -> i64 {
65    if summary_total(&primary) >= summary_total(&fallback) {
66        1
67    } else {
68        0
69    }
70}
