1//! The flat surface under test: one field per **Rust spelling** of a shape the
2//! model already reads representation-agnostically.
3//!
4//! This module is ordinary compiled Rust, and `build.rs` parses this very file
5//! to build the flat model — so a shape added here is emitted, compiled, and
6//! called against the same definitions, with nothing restated. Bodies are
7//! `unimplemented!()`: the question is whether the *emitted* code type-checks
8//! against these signatures, not what they compute.
9//!
10//! The generated bindings qualify every call as `myflat::…` (build.rs stamps
11//! that crate name onto each item, exactly as the unit-test fixtures do), which
12//! is why `lib.rs` mounts this file under that name.
13
14// The shapes are chosen for the generator, not for idiomatic Rust — nobody
15// writes a `Box<Option<T>>` field by hand.
16#![allow(clippy::box_collection, clippy::redundant_allocation)]
17
18use std::borrow::Cow;
19
20/// An opaque handle. Every value-form field below reaches one of these, or a
21/// leaf.
22///
23/// The field is not decoration: a handle crosses as a `jlong` whose bit 0 is
24/// the closed tag, so the generated code asserts `align_of::<T>() >= 2` and a
25/// zero-sized handle fails that at compile time.
26pub struct ZKeyExpr(pub u64);
27
28/// The subject whose value form carries the spellings.
29pub struct ZSample(pub u64);
30
31/// The child boundary: a `ZKeyExpr` field must still cross through *this*,
32/// because the field's own type decides how it crosses — not the wrapper the
33/// value form happens to spell it with.
34pub fn z_keyexpr_as_str(k: &ZKeyExpr) -> &str {
35    let _ = k;
36    unimplemented!()
37}

39/// The value form, one field per spelling. `prebindgen-flat`'s `acceptance.rs`
40/// pins that each pair below *reads* the same; the emitted access for each one
41/// is what this crate compiles.
42///
43/// The pairs, in the order the model's own acceptance test lists them:
44///
45/// | shape    | plain            | wrapped                |
46/// |----------|------------------|------------------------|
47/// | optional | `Option<T>`      | `Box<Option<T>>` (#268)|
48/// | sequence | `Vec<u8>`        | `Cow<'_, [u8]>`        |
49/// | `Str`    | `String`         | `Box<String>`, `Cow<'_, str>` |
50///
51/// `Box<Option<T>>` is the shape of #268 specifically: match ergonomics does
52/// not deref a `Box`, so an access that destructures the raw place is `E0308`
53/// here and compiles one field up.
54pub struct ZSampleStruct {
55    pub opt_plain: Option<ZKeyExpr>,
56    pub opt_boxed: Box<Option<ZKeyExpr>>,
57    pub seq_plain: Vec<u8>,
58    pub seq_cow: Cow<'static, [u8]>,
59    pub text_plain: String,
60    pub text_boxed: Box<String>,
61    pub text_cow: Cow<'static, str>,
62}
63
64pub fn z_sample_to_struct(s: &ZSample) -> ZSampleStruct {
65    let _ = s;
66    unimplemented!()
67}

69// The fourth spelling the model's acceptance test pins — a run of values
70// spelled **bare**, `[T]` — is deliberately absent, and cannot be added. It is
71// unsized, so it is not a struct field, not a return, and not a by-value
72// callback argument: there is no signature carrying it that is itself valid
73// Rust. A spelling that no compilable source can hold has no emitted access to
74// compile, which is why `prebindgen-flat`'s acceptance test is the only place
75// it can be pinned.
