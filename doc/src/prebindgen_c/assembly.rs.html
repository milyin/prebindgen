1//! What the generated C-facing Rust file is assembled from.
2//!
3//! Each final artifact is planned when the C generation plan is frozen and
4//! rendered at the final writing boundary, so nothing it needs is looked up in
5//! a live registry while the file is assembled. A wrapper — the exported
6//! `extern "C"` function one declared `#[prebindgen]` function becomes — reads
7//! its own frozen state plus the boundary sites of its own function, which the
8//! same frozen plan holds.
9
10use prebindgen_registry::{
11    generation::ArtifactId,
12    write::{ArtifactKey, RustArtifact},
13};

17/// One final artifact of the generated Rust file.
18#[derive(Clone)]
19pub(crate) enum CFinalArtifact {
20    /// A private converter, carrying one value across the boundary.
21    Converter(Box<crate::chain::CFunction>),
22    /// The exported wrapper for one declared `#[prebindgen]` function.
23    Wrapper(Box<CWrapper>),
24    /// One captured constant, re-stated as an alias to its source.
25    Const(Box<CConst>),
26    /// The memory helpers the generated layer hands `char*` and array blocks
27    /// to C through.
28    Memory(Box<CMemory>),
29    /// The array builder, which copies a `Vec<T>` into a C block.
30    ArrayBuilder,
31    /// One `#[repr(C)]` mirror of a declared data struct.
32    DataStruct(Box<CDataStruct>),
33    /// One `#[repr(C)]` mirror of a declared fieldless enum.
34    Enum(Box<CEnum>),
35    /// One reserved representation value a generated sum-type ABI uses.
36    DomainConstant(Box<CDomainConstant>),
37    /// One opaque handle: its `#[repr(C)]` struct and its typed drop.
38    ///
39    /// These four carry the [`ArtifactId`] they are planned under, because an
40    /// artifact's key is its identity in the plan and the payload alone cannot
41    /// say it — the id is built from the declared type key, which is not a field
42    /// of what gets rendered.
43    OpaqueHandle(ArtifactId, Box<crate::chain::OpaqueHandleArtifact>),
44    /// One value-opaque: a declared type C sees only as bytes.
45    ValueOpaque(ArtifactId, Box<crate::chain::ValueOpaqueArtifact>),
46    /// One tagged union: its `#[repr(C)]` struct and its reader.
47    TaggedUnion(ArtifactId, Box<crate::chain::TaggedUnionArtifact>),
48    /// One callback: the trampoline C calls back through.
49    Callback(ArtifactId, Box<crate::chain::CallbackArtifact>),
50}
51
52impl RustArtifact for CFinalArtifact {
53    fn key(&self) -> ArtifactKey {
54        match self {
55            Self::Converter(converter) => converter.key(),
56            Self::Wrapper(wrapper) => wrapper.key(),
57            Self::Const(constant) => constant.key(),
58            Self::Memory(memory) => memory.key(),
59            Self::ArrayBuilder => array_builder_key(),
60            Self::DataStruct(item) => item.key(),
61            Self::Enum(item) => item.key(),
62            Self::DomainConstant(item) => item.key(),
63            Self::OpaqueHandle(id, _)
64            | Self::ValueOpaque(id, _)
65            | Self::TaggedUnion(id, _)
66            | Self::Callback(id, _) => ArtifactKey::Artifact(id.clone()),
67        }
68    }
69
70    fn provides(&self) -> Vec<ArtifactKey> {
71        let mut provided = vec![self.key()];
72        // A callback renders the Invoke helper of its converter operation, whose
73        // fragment deliberately carries no artifact of its own.
74        if let Self::Callback(_, callback) = self {
75            provided.push(ArtifactKey::Operation(callback.invoke.operation.clone()));
76        }
77        provided
78    }
79
80    fn reachable(&self) -> bool {
81        // Every C artifact reaches the file. Unreached converters are pruned
82        // from the generation plan before the assembly sees them, and every
83        // other kind is planned only when something needs it — the memory
84        // helpers because an artifact says it calls them, a mirror because its
85        // type crosses. Stated rather than inherited: an artifact that needs a
86        // different answer should be a deliberate change here.
87        true
88    }
89
90    fn calls(&self) -> Vec<ArtifactKey> {
91        match self {
92            Self::Converter(converter) => converter.calls(),
93            Self::Wrapper(wrapper) => wrapper.calls(),
94            Self::Const(constant) => constant.calls(),
95            Self::Memory(memory) => memory.calls(),
96            // The block it fills is malloc'd, and freed by the universal freer.
97            Self::ArrayBuilder => vec![memory_key()],
98            Self::DataStruct(item) => item.calls(),
99            Self::Enum(item) => item.calls(),
100            Self::DomainConstant(item) => item.calls(),
101            // The Invoke helper converts each argument on its way out.
102            Self::Callback(_, callback) => {
103                let mut calls = Vec::new();
104                for argument in &callback.arguments {
105                    argument.value.calls(&mut calls);
106                }
107                calls
108            }
109            // A tagged union's typed destructor frees what its live arm owns,
110            // which reaches a nested union's destructor and the memory helpers.
111            Self::TaggedUnion(_, union) => {
112                let mut calls = Vec::new();
113                for field in union.arms.iter().flat_map(|arm| &arm.fields) {
114                    if let Some(cleanup) = &field.cleanup {
115                        cleanup.calls(&mut calls);
116                    }
117                }
118                calls
119            }
120            // A handle is a pointer and a value-opaque is reinterpreted: neither
121            // converts nor frees anything of its own.
122            Self::OpaqueHandle(..) | Self::ValueOpaque(..) => Vec::new(),
123        }
124    }
125
126    fn render(&self, emit: &prebindgen_registry::RustWriter) -> Vec<syn::Item> {
127        match self {
128            Self::Converter(converter) => converter.render(emit),
129            Self::Wrapper(wrapper) => wrapper.render(emit),
130            Self::Const(constant) => constant.render(emit),
131            Self::Memory(memory) => memory.render(emit),
132            Self::ArrayBuilder => CMemory::render_array_builder(),
133            Self::DataStruct(item) => item.render(emit),
134            Self::Enum(item) => item.render(emit),
135            Self::DomainConstant(item) => item.render(emit),
136            Self::Callback(_, callback) => callback.render(emit),
137            Self::OpaqueHandle(_, handle) => handle.render(emit),
138            Self::TaggedUnion(_, union) => union.render(emit),
139            Self::ValueOpaque(_, value) => value.render(emit),
140        }
141    }
142}

144/// One declared function's exported wrapper, frozen at the end of resolution.
145#[derive(Clone)]
146pub(crate) struct CWrapper {
147    /// This function's own boundary sites, as the registry planned them. All a
148    /// wrapper reads of the plan, and taken by value so a wrapper can be stated
149    /// before the plan is built.
150    sites: Vec<prebindgen_registry::generation::SitePlan<crate::compile::CRepresentation>>,
151    /// The `#[prebindgen]` function this wrapper exports.
152    function: prebindgen_registry::flat::Function,
153    /// The `#[no_mangle]` symbol the wrapper is exported under.
154    symbol: syn::Ident,
155    /// Path the wrapper calls the source function by.
156    call_path: syn::Path,
157    /// Whether `.panic()` was declared, which is what allows a fallible
158    /// conversion in a function that returns no `Result`.
159    panic: bool,
160    /// Per parameter, the resource its argument names and how the call uses
161    /// it — `None` for a parameter that names no single owned resource. The
162    /// alias preflight compares these; declaring which types are opaque is a
163    /// binding question, so it is answered while planning.
164    alias_slots: Vec<Option<(TypeKey, AliasAccess)>>,
165}
166
167impl CWrapper {
168    /// Plan the wrapper for one declared function.
169    /// `sites` are the boundary sites the registry planned for this function,
170    /// which is everything of the plan a wrapper reads. Taking them rather than
171    /// the plan is what lets a wrapper be stated as an `ArtifactPlan` before the
172    /// plan is built, since the sites exist first.
173    pub(crate) fn new(
174        decls: &CbindgenBuilder,
175        sites: &[prebindgen_registry::generation::SitePlan<crate::compile::CRepresentation>],
176        function: &prebindgen_registry::flat::Function,
177    ) -> Self {
178        let name = &function.name;
179        if let Some((_, err)) = function.ret.fallible_parts() {
180            let err_key = err.key();
181            assert!(
182                decls.error.contains(&err_key),
183                "Cbindgen: function `{}` returns `Result<_, {}>` but `{}` is not a \
184                 declared error type — add `.data_struct({}).error()`",
185                name,
186                err_key,
187                err_key,
188                err_key,
189            );
190        }
191        Self {
192            sites: sites
193                .iter()
194                .filter(|site| site.id().site().owner == *name)
195                .cloned()
196                .collect(),
197            function: function.clone(),
198            symbol: decls.fn_symbol(name),
199            call_path: decls.src_fn(name),
200            panic: decls.functions.get(name).is_some_and(|cfg| cfg.panic),
201            alias_slots: function
202                .params
203                .iter()
204                .map(|param| decls.alias_slot_of(&param.ty))
205                .collect(),
206        }
207    }
208
209    /// One boundary site of this wrapper's own function, when the function has
210    /// one in that role. A void return and an infallible function have no
211    /// return or error site.
212    fn site_if_planned(
213        &self,
214        role: prebindgen_registry::recipe::Role,
215    ) -> Option<&prebindgen_registry::generation::SitePlan<crate::compile::CRepresentation>> {
216        self.sites.iter().find(|site| site.id().site().role == role)
217    }
218
219    /// One boundary site of this wrapper's own function.
220    fn site(
221        &self,
222        role: prebindgen_registry::recipe::Role,
223    ) -> &prebindgen_registry::generation::SitePlan<crate::compile::CRepresentation> {
224        let name = self.function.name.clone();
225        let described = role.to_string();
226        let site = self
227            .site_if_planned(role)
228            .unwrap_or_else(|| panic!("C generation plan has no {described} of `{name}`"));
229        assert!(
230            matches!(
231                site.cleanup(),
232                prebindgen_registry::generation::Cleanup::None
233            ),
234            "ordinary C sites cannot carry a deferred cleanup operation"
235        );
236        site
237    }
238
239    /// Assemble this wrapper's `#[no_mangle] extern "C"` function.
240    fn render_fn(&self, emit: &prebindgen_registry::RustWriter) -> syn::ItemFn {
241        let f = &self.function;
242        let orig = &f.name;
243        let call_path = &self.call_path;
244        let sym = &self.symbol;
245
246        // The ELEMENT: a signature is a parameter list and a return, both
247        // already classified. An elided return is `TypeKind::Unit`, which is
248        // the `ReturnType::Default` arm this used to write.
249
250        let has_fallible_input = f.params.iter().enumerate().any(|(index, _)| {
251            self.site(prebindgen_registry::recipe::Role::Param { index })
252                .abi()
253                .payload()
254                .failure()
255                == prebindgen_registry::generation::Failure::Fallible
256        });
257
258        // Peel an outer `Result<_, E>`; `value_ty` is the success/return value.
259        // Off `TypeKind::Fallible`, where `result_parts` found the `Result` in a
260        // path first — and both sides come back as readings, so everything
261        // downstream of here reads too.
262        let (value_ty, err_reading) = match f.ret.fallible_parts() {
263            Some((ok, e)) => (ok, Some(e)),
264            None => (&f.ret, None),
265        };
266        let value_site = (!matches!(value_ty.kind(), TypeKind::Unit))
267            .then(|| self.site(prebindgen_registry::recipe::Role::Return));
268        let has_fallible_output = value_site.is_some_and(|site| {
269            site.abi().payload().failure() == prebindgen_registry::generation::Failure::Fallible
270        });
271
272        // Error wiring: that the error type is declared via `.error()` was
273        // checked when the wrapper was planned.
274        let err_bits = err_reading.map(|err| {
275            let site = self.site(prebindgen_registry::recipe::Role::Error);
276            let crate::compile::CValue::Direct {
277                wire, converter, ..
278            } = site.abi().payload()
279            else {
280                panic!("C error site must have one direct wire");
281            };
282            (
283                wire.clone(),
284                converter.ident(emit),
285                emit.emit_source_type(err),
286            )
287        });
288
289        // No `Result` channel ⇒ a fallible input must be declared `.panic()`.
290        if err_reading.is_none() {
291            assert!(
292                !(has_fallible_input || has_fallible_output) || self.panic,
293                "Cbindgen: function `{}` has a fallible binding conversion but does not \
294                 return `Result`; add \
295                 `.panic()` after its `.function(...)` declaration to allow aborting \
296                 on the internal error, or change its signature",
297                orig,
298            );
299        }
300
301        // Structural lowering of the (present/ok) value, then the null-niche rule:
302        //   * Result + a free pointer niche  → NULL marks `Err` (value in-band);
303        //   * Result without a free niche     → `bool` status, value to out-params;
304        //   * no Result                       → field 0 is the C return, rest out.
305        let shape = value_site.map_or(
306            FrozenValueLayout {
307                fields: Vec::new(),
308                niches: Niches::empty(),
309            },
310            |site| FrozenValueLayout {
311                fields: site.abi().payload().fields(),
312                niches: site.abi().payload().effective_niches(),
313            },
314        );
315        let result_slot = shape.niches.clone().carve().map(|(slot, _)| slot);
316        let result_in_band = err_reading.is_some() && result_slot.is_some();
317        let field0_is_return = result_in_band || err_reading.is_none();
318
319        // Partition fields into the (optional) C return value + out-parameters,
320        // and pick C names for the out-params (see `out_param_name`).
321        let mut targets: Vec<TokenStream> = Vec::new();
322        let mut out_fields: Vec<&WireField> = Vec::new();
323        // `field0_wire` is the wire of the value's primary field when that field
324        // is carried by the C return slot (modes A/D); `None` for mode B and unit.
325        let field0_wire: Option<syn::Type> = if field0_is_return {
326            shape.fields.first().map(|f| f.wire.clone())
327        } else {
328            None
329        };
330        if field0_is_return {
331            if !shape.fields.is_empty() {
332                targets.push(quote!(__ret));
333                out_fields.extend(shape.fields[1..].iter());
334            }
335        } else {
336            out_fields.extend(shape.fields.iter());
337        }
338        let prefixed = out_fields.iter().any(|wf| wf.suffix.is_empty());
339        let out_names: Vec<syn::Ident> = out_fields
340            .iter()
341            .map(|wf| out_param_name(wf.suffix, prefixed))
342            .collect();
343        for name in &out_names {
344            targets.push(quote!(*#name));
345        }
346        let out_param_decls: Vec<TokenStream> = out_fields
347            .iter()
348            .zip(&out_names)
349            .map(|(wf, name)| {
350                let wire = &wf.wire;
351                quote!(#name: *mut #wire)
352            })
353            .collect();
354
355        // C wrapper return type: the payload's field 0 (modes A/D), `bool` status
356        // (mode B), or `void` (a unit value with no `Result`).
357        let c_return: Option<syn::Type> = if field0_is_return {
358            field0_wire.clone()
359        } else {
360            Some(syn::parse_quote!(bool))
361        };
362
363        // Input decode: route a fallible-input failure to the error out-param
364        // (with the wrapper's fail value) when there is a `Result`, else panic.
365        let fail_return = if result_in_band {
366            let slot = result_slot.as_ref().expect("in-band result has a niche");
367            let value = &slot.value;
368            quote!(#value)
369        } else {
370            quote!(false)
371        };
372        let mut planned_routes = f
373            .params
374            .iter()
375            .enumerate()
376            .filter_map(|(index, _)| {
377                self.site(prebindgen_registry::recipe::Role::Param { index })
378                    .failure_route()
379            })
380            .chain(
381                value_site
382                    .into_iter()
383                    .filter_map(|site| site.failure_route()),
384            );
385        let planned_route = planned_routes.next();
386        assert!(
387            planned_routes.all(|route| Some(route) == planned_route),
388            "C function sites disagree on their frozen failure route"
389        );
390        match planned_route {
391            Some(crate::compile::CFailureRoute::Panic) => {
392                assert!(
393                    err_bits.is_none(),
394                    "panic route attached to a Result function"
395                );
396            }
397            Some(crate::compile::CFailureRoute::Error(error_site)) => {
398                assert!(
399                    err_bits.is_some(),
400                    "error route attached without a Result channel"
401                );
402                let expected = prebindgen_registry::generation::SiteId::new(
403                    prebindgen_registry::recipe::Site {
404                        owner: orig.clone(),
405                        role: prebindgen_registry::recipe::Role::Error,
406                    },
407                );
408                assert_eq!(
409                    error_site, &expected,
410                    "failure route names the wrong error site"
411                );
412            }
413            None => {}
414        }
415        let input_route = match &err_bits {
416            Some((_, e_conv, e_ty_src)) => ErrRoute::Result {
417                e_conv,
418                e_ty_src: e_ty_src.clone(),
419                fail_return: fail_return.clone(),
420            },
421            None => ErrRoute::Panic,
422        };
423        let (in_params, decodes, call_args) = self.planned_inputs(&input_route, emit);
424        let call = quote!(#call_path(#(#call_args),*));
425
426        let e_param = err_bits
427            .as_ref()
428            .map(|(err_wire, _, _)| quote!(e: *mut #err_wire));
429        let ret_arrow = c_return.as_ref().map(|w| quote!(-> #w));
430
431        // Assemble the body per the three structural modes.
432        let body = match (&err_bits, field0_is_return) {
433            // No `Result`: straight-line. `void` when there are no fields.
434            (None, _) => {
435                if let Some(field0_wire) = field0_wire.as_ref() {
436                    let enc = value_site.map_or_else(TokenStream::new, |site| {
437                        site.abi()
438                            .payload()
439                            .encode(quote!(__v), &targets, &input_route, emit)
440                    });
441                    quote!(
442                        #(#decodes)*
443                        let __v = #call;
444                        let __ret: #field0_wire;
445                        #enc
446                        __ret
447                    )
448                } else {
449                    quote!( #(#decodes)* #call; )
450                }
451            }
452            // `Result` with a free niche: value in-band, NULL marks `Err`.
453            (Some((_, e_conv, _)), true) => {
454                let field0_wire = field0_wire.as_ref().expect("in-band ⇒ pointer return");
455                let null = &result_slot
456                    .as_ref()
457                    .expect("in-band result has a niche")
458                    .value;
459                let enc = value_site.map_or_else(TokenStream::new, |site| {
460                    site.abi()
461                        .payload()
462                        .encode(quote!(__v), &targets, &input_route, emit)
463                });
464                quote!(
465                    #(#decodes)*
466                    match #call {
467                        ::core::result::Result::Ok(__v) => { let __ret: #field0_wire; #enc __ret }
468                        ::core::result::Result::Err(__err) => {
469                            if !e.is_null() { *e = #e_conv(__err); }
470                            #null
471                        }
472                    }
473                )
474            }
475            // `Result` without a free niche: `bool` status, value to out-params.
476            (Some((_, e_conv, _)), false) => {
477                let enc = value_site.map_or_else(TokenStream::new, |site| {
478                    site.abi()
479                        .payload()
480                        .encode(quote!(__v), &targets, &input_route, emit)
481                });
482                quote!(
483                    #(#decodes)*
484                    match #call {
485                        ::core::result::Result::Ok(__v) => { #enc true }
486                        ::core::result::Result::Err(__err) => {
487                            if !e.is_null() { *e = #e_conv(__err); }
488                            false
489                        }
490                    }
491                )
492            }
493        };
494
495        syn::parse_quote! {
496            #[no_mangle]
497            #[allow(non_snake_case, unused_mut, unused_variables, unused_unsafe, dead_code)]
498            pub unsafe extern "C" fn #sym(
499                #(#in_params,)*
500                #(#out_param_decls,)*
501                #e_param
502            ) #ret_arrow {
503                #body
504            }
505        }
506    }

508    /// The runtime **alias preflight**: reject a call whose arguments name the
509    /// same resource in a combination that would reconstruct or invalidate it
510    /// twice, *before* any conversion runs.
511    ///
512    /// `z_combine(primary: ZThing, fallback: ZThing)` is a supported
513    /// declaration; called as `z_combine(x, x)` it reaches `Box::from_raw`
514    /// twice on one allocation. So is `f(a: ZThing, b: &ZThing)` — the borrow
515    /// dangles the moment the consume takes ownership — and `f(a: &mut T, b:
516    /// &T)`, where the exclusive reference is not exclusive at all. Rejecting
517    /// the *declaration* is not an option: these are shapes that ship today, so
518    /// removing them would be a regression that a later stage would have to
519    /// undo.
520    ///
521    /// Emitted whenever a call has **at least one `Consume` or
522    /// `ExclusiveBorrow`** and **any other active access in the same resource
523    /// domain**. Stated that way rather than as "two or more consumed
524    /// parameters", which would skip exactly the two mixed cases above.
525    ///
526    /// A NULL pointer names no resource — two NULLs are not an alias — and is
527    /// rejected by each converter's own null check, so the comparison skips it.
528    ///
529    /// Shared/shared is *not* rejected: two `&T` to one resource is legal Rust
530    /// and legal C.
531    fn alias_preflight(&self, route: &ErrRoute) -> Option<TokenStream> {
532        let slots: Vec<(syn::Ident, TypeKey, AliasAccess)> = self
533            .function
534            .params
535            .iter()
536            .zip(&self.alias_slots)
537            .filter_map(|(param, slot)| {
538                slot.as_ref()
539                    .map(|(key, access)| (param.name.clone(), key.clone(), *access))
540            })
541            .collect();
542
543        let on_err = route_message(route);
544        let mut checks: Vec<TokenStream> = Vec::new();
545        for i in 0..slots.len() {
546            for j in (i + 1)..slots.len() {
547                let (a, a_key, a_access) = &slots[i];
548                let (b, b_key, b_access) = &slots[j];
549                if a_key != b_key {
550                    continue;
551                }
552                // At least one side must be exclusive about the resource.
553                if matches!(a_access, AliasAccess::Shared)
554                    && matches!(b_access, AliasAccess::Shared)
555                {
556                    continue;
557                }
558                let msg = format!(
559                    "aliasing arguments: `{a}` ({}) and `{b}` ({}) are the same `{}` — a \
560                     consumed or exclusively-borrowed resource may not be named twice in one call",
561                    a_access.describe(),
562                    b_access.describe(),
563                    a_key.as_str(),
564                );
565                checks.push(quote!(
566                    if !(#a as *const ()).is_null() && (#a as *const ()) == (#b as *const ()) {
567                        let __msg = ::std::string::String::from(#msg);
568                        #on_err
569                    }
570                ));
571            }
572        }
573        (!checks.is_empty()).then(|| quote!(#(#checks)*))
574    }

576    /// Build the wire param list, per-input decode statements, and call-site
577    /// argument expressions. Fallible inputs (converter returns `Result<_,
578    /// String>`) route their `Err(msg)` per `route`; infallible inputs decode
579    /// directly.
580    fn planned_inputs(
581        &self,
582        route: &ErrRoute<'_>,
583        emit: &prebindgen_registry::RustWriter,
584    ) -> (Vec<TokenStream>, Vec<TokenStream>, Vec<TokenStream>) {
585        let mut params = Vec::new();
586        let mut decodes: Vec<TokenStream> = self.alias_preflight(route).into_iter().collect();
587        let mut call_args = Vec::new();
588
589        for (index, param) in self.function.params.iter().enumerate() {
590            let ident = &param.name;
591            let site = self.site(prebindgen_registry::recipe::Role::Param { index });
592            match site.abi().payload() {
593                crate::compile::CValue::BorrowedInput {
594                    element,
595                    wire,
596                    reinterpret,
597                } => {
598                    let len_id = format_ident!("{}_len", ident);
599                    let source = emit.emit_source_type(element);
600                    params.push(quote!(#ident: #wire));
601                    params.push(quote!(#len_id: usize));
602                    let from_parts = if *reinterpret {
603                        quote!(::core::slice::from_raw_parts(
604                            #ident as *const #source,
605                            #len_id,
606                        ))
607                    } else {
608                        quote!(::core::slice::from_raw_parts(#ident, #len_id))
609                    };
610                    decodes.push(quote!(
611                        let #ident: &[#source] = if #ident.is_null() {
612                            &[]
613                        } else {
614                            #from_parts
615                        };
616                    ));
617                }
618                crate::compile::CValue::Direct {
619                    wire, converter, ..
620                } => {
621                    let conv = converter.ident(emit);
622                    params.push(quote!(#ident: #wire));
623                    if converter.fallible() {
624                        let on_err = route_message(route);
625                        decodes.push(quote!(
626                            let #ident = match #conv(#ident) {
627                                ::core::result::Result::Ok(__v) => __v,
628                                ::core::result::Result::Err(__msg) => { #on_err }
629                            };
630                        ));
631                    } else {
632                        decodes.push(quote!(let #ident = #conv(#ident);));
633                    }
634                }
635                _ => panic!("C input site has an output-only ABI plan"),
636            }
637            call_args.push(quote!(#ident));
638        }
639
640        (params, decodes, call_args)
641    }
642}

749/// The memory helpers: the C allocator the generated layer calls, the raw
750/// C-string block builder, the universal freer C calls back, and — when a
751/// `Vec<T>` return hands out a block — the array builder.
752///
753/// Planned only when the layer actually hands memory to C, which is also when
754/// a declared `.free_memory_function` becomes required.
755#[derive(Clone)]
756pub(crate) struct CMemory {
757    /// The declared freer's exported symbol.
758    /// `None` when the binding declared no free function.
759    ///
760    /// Optional because these helpers are planned whether or not one is
761    /// declared, and kept only when something requires them. Whether that
762    /// happened is the plan's answer, and `build_with` reads it there — so a
763    /// rendered memory artifact always has a declaration.
764    free_ident: Option<syn::Ident>,
765}
766
767impl CMemory {
768    /// Plan the memory helpers for a binding that hands memory to C.
769    ///
770    /// Whether it does is not asked here: the artifacts that allocate say so
771    /// through their own dependencies, and the plan keeps this only when one of
772    /// them requires it.
773    ///
774    /// So a binding with no free function still PLANS these helpers — it just
775    /// does not keep them. Whether it kept them is what `build_with` asks the
776    /// finished plan, and it refuses a binding that did without declaring a free
777    /// function. The refusal stays in the build phase: `write_rust` renders an
778    /// assembly that is already valid.
779    pub(crate) fn new(decls: &CbindgenBuilder) -> Self {
780        Self {
781            free_ident: decls.free_fn.as_ref().map(|name| format_ident!("{name}")),
782        }
783    }
784
785    /// Copy a `Vec<W>` into a C-`malloc`'d block of `W` and return
786    /// `(ptr, len)` (empty ⇒ `(NULL, 0)`). The block is freed C-side via the
787    /// `z_free_array` macro (per-element drop + the universal freer).
788    fn render_array_builder() -> Vec<syn::Item> {
789        vec![syn::parse_quote!(
790            #[allow(non_snake_case, dead_code)]
791            pub(crate) unsafe fn __cbg_alloc_array<W>(v: ::std::vec::Vec<W>) -> (*mut W, usize) {
792                let n = v.len();
793                if n == 0 {
794                    return (::core::ptr::null_mut(), 0);
795                }
796                let p = malloc(n.wrapping_mul(::core::mem::size_of::<W>())) as *mut W;
797                if p.is_null() {
798                    return (::core::ptr::null_mut(), 0);
799                }
800                for (i, e) in v.into_iter().enumerate() {
801                    ::core::ptr::write(p.add(i), e);
802                }
803                (p, n)
804            }
805        )]
806    }
807}
808
809impl RustArtifact for CMemory {
810    fn calls(&self) -> Vec<ArtifactKey> {
811        Vec::new()
812    }
813
814    fn key(&self) -> ArtifactKey {
815        memory_key()
816    }
817
818    fn render(&self, _emit: &prebindgen_registry::RustWriter) -> Vec<syn::Item> {
819        // `build_with` rejected a binding that keeps these helpers without a
820        // free function, so by the time an assembly is rendered there is one.
821        let free_ident = self
822            .free_ident
823            .as_ref()
824            .expect("a retained memory artifact was validated when the generator was built");
825        // C allocator (linked from the C runtime; no crate dependency).
826        let items: Vec<syn::Item> = vec![
827            syn::parse_quote!(
828                extern "C" {
829                    fn malloc(size: usize) -> *mut ::core::ffi::c_void;
830                    fn free(ptr: *mut ::core::ffi::c_void);
831                }
832            ),
833            // Raw, destructor-free C-string block. `CString::new` drops interior
834            // NULs so the terminator marks the true end for C consumers.
835            syn::parse_quote!(
836                #[allow(non_snake_case, dead_code)]
837                pub(crate) fn __cbg_alloc_cstr(
838                    s: ::std::string::String,
839                ) -> *mut ::core::ffi::c_char {
840                    let c = ::std::ffi::CString::new(s).unwrap_or_default();
841                    let bytes = c.as_bytes_with_nul();
842                    unsafe {
843                        let p = malloc(bytes.len()) as *mut u8;
844                        if p.is_null() {
845                            return ::core::ptr::null_mut();
846                        }
847                        ::core::ptr::copy_nonoverlapping(bytes.as_ptr(), p, bytes.len());
848                        p as *mut ::core::ffi::c_char
849                    }
850                }
851            ),
852            // Universal raw memory freer: type-agnostic C `free`, no length, no
853            // destructor (NULL-safe via C `free`).
854            syn::parse_quote!(
855                #[no_mangle]
856                #[allow(non_snake_case, unused_variables)]
857                pub unsafe extern "C" fn #free_ident(p: *mut ::core::ffi::c_void) {
858                    free(p);
859                }
860            ),
861        ];
862        items
863    }
864}

866/// One declared data struct's `#[repr(C)]` mirror.
867///
868/// The mirror is a layout fact: each field is stated in its wire form, which
869/// is decided while planning, when the binding's declarations and the model
870/// are both available.
871#[derive(Clone)]
872pub(crate) struct CDataStruct {
873    /// The C-facing struct name.
874    c_struct: syn::Ident,
875    /// Field name and wire type, in source order.
876    fields: Vec<(syn::Ident, syn::Type)>,
877}
878
879impl CDataStruct {
880    /// Plan the mirrors of every declared data struct that crosses.
881    pub(crate) fn all(decls: &CbindgenBuilder, registry: &Registry) -> Vec<Self> {
882        let mut mirrors = Vec::new();
883        for (key, _cfg) in sorted_by_key(&decls.data) {
884            let Some(reading) = registry.reading(key) else {
885                continue;
886            };
887            if decls.in_frag(&reading).is_none() && decls.out_frag(&reading).is_none() {
888                continue;
889            }
890            let Some(fields) = decls.struct_fields(registry, &reading.key()) else {
891                continue;
892            };
893            mirrors.push(Self {
894                c_struct: decls.c_type_ident(&reading.key()),
895                fields: fields
896                    .iter()
897                    .map(|(name, ty)| {
898                        let wire = decls.data_field_wire(ty).unwrap_or_else(|| {
899                            panic!(
900                                "Cbindgen: field `{}` of data struct `{}` has unsupported type `{}`",
901                                name,
902                                type_short(&reading.key()),
903                                ty
904                            )
905                        });
906                        (name.clone(), wire)
907                    })
908                    .collect(),
909            });
910        }
911        mirrors
912    }
913}

915impl RustArtifact for CDataStruct {
916    fn calls(&self) -> Vec<ArtifactKey> {
917        Vec::new()
918    }
919
920    fn key(&self) -> ArtifactKey {
921        artifact_id("c-data-struct", self.c_struct.to_string())
922    }
923
924    fn render(&self, _emit: &prebindgen_registry::RustWriter) -> Vec<syn::Item> {
925        let c_struct = &self.c_struct;
926        let fields = self
927            .fields
928            .iter()
929            .map(|(name, wire)| quote!(pub #name: #wire));
930        vec![syn::parse_quote!(
931            #[repr(C)]
932            #[allow(non_camel_case_types)]
933            pub struct #c_struct {
934                #(#fields,)*
935            }
936        )]
937    }

940/// One declared fieldless enum's `#[repr(C)]` mirror.
941///
942/// Each discriminant is re-stated **as written** — `= 0x07` stays `0x07` —
943/// which is what keeps every value C already accepted: a `const` or
944/// `cfg`-driven expression, and anything the source's own `repr` admits.
945/// Resolving each to a number would narrow that to what `i64` and a literal
946/// can express, for no gain, since cbindgen re-reads this as Rust source.
947///
948/// That is why the model's own values are retained here and spelled by the
949/// writer, rather than the mirror being spelled while planning.
950#[derive(Clone)]
951pub(crate) struct CEnum {
952    /// The C-facing enum name.
953    c_name: syn::Ident,
954    /// The source enum's values, in declaration order.
955    values: Vec<prebindgen_registry::flat::EnumValue>,
956}
957
958impl CEnum {
959    /// Plan the mirrors of every declared fieldless enum that crosses.
960    pub(crate) fn all(decls: &CbindgenBuilder, registry: &Registry) -> Vec<Self> {
961        let mut mirrors = Vec::new();
962        for (key, _cfg) in sorted_by_key(&decls.enums) {
963            let Some(reading) = registry.reading(key) else {
964                continue;
965            };
966            if decls.in_frag(&reading).is_none() && decls.out_frag(&reading).is_none() {
967                continue;
968            }
969            let Some(item) = unit_enum(registry, &reading.key()) else {
970                continue;
971            };
972            mirrors.push(Self {
973                c_name: decls.c_type_ident(&reading.key()),
974                values: item.values.clone(),
975            });
976        }
977        mirrors
978    }
979}

981impl RustArtifact for CEnum {
982    fn calls(&self) -> Vec<ArtifactKey> {
983        Vec::new()
984    }
985
986    fn key(&self) -> ArtifactKey {
987        artifact_id("c-enum", self.c_name.to_string())
988    }
989
990    fn render(&self, emit: &prebindgen_registry::RustWriter) -> Vec<syn::Item> {
991        let c_name = &self.c_name;
992        let variants = self.values.iter().map(|value| {
993            let id = &value.name;
994            match emit.discriminant(value) {
995                Some(expr) => quote!(#id = #expr),
996                None => quote!(#id),
997            }
998        });
999        vec![syn::parse_quote!(
1000            #[repr(C)]
1001            #[derive(Copy, Clone, Debug, Eq, PartialEq)]
1002            #[allow(non_camel_case_types)]
1003            pub enum #c_name {
1004                #(#variants),*
1005            }
1006        )]
1007    }
