1//! Flat, FFI-friendly example library — a miniature in the style of `zenoh-flat`.
2//!
3//! Every public function is annotated with `#[prebindgen]`, so `prebindgen`
4//! captures this surface and a language adapter (here `prebindgen::lang::CbindgenBuilder`,
5//! driven by `example-cbindgen`) generates the FFI layer — no hand-written
6//! `extern "C"` glue, and **no `#[repr(C)]`** in this crate.
7//!
8//! The API is plain idiomatic Rust:
9//!
10//! - [`Calculator`] is an opaque handle returned **by value**; the adapter boxes it
11//!   and emits a typed `calculator_drop`.
12//! - [`Error`] is a boxed `std` error rendered to a message by [`error_get_message`];
13//!   fallible calls return `Result<T, Error>`.
14//! - [`Operation`] is a primitive-repr enum (`#[repr(i32)]`, like zenoh-flat's
15//!   `Priority`).
16//! - Items are delivered to a C closure through an `impl Fn(..)` callback
17//!   ([`calculator_for_each`]).
18//!
19//! Function names encode their receiver and role: `calculator_new*` construct,
20//! `calculator_get_*` read, `calculator_to_string` converts.
21

31/// Boxed error type, mirroring zenoh-flat's `Error`. It is the `E` of every
32/// fallible `Result` and never crosses the FFI boundary as a value; the adapter
33/// marshals it to C as a `char*` message obtained from [`error_get_message`].
34///
35/// Marked, because that is how a type whose contents do not cross gets a name in
36/// the flat API: the alias declares `Error` as an opaque handle, which is what
37/// lets every `Result<_, Error>` below resolve.
38#[prebindgen]
39pub type Error = Box<dyn std::error::Error + Send + Sync>;

41/// Render an error as its display string. Wired into the C adapter as the

43#[prebindgen]
44pub fn error_get_message(e: &Error) -> String {
45    e.to_string()
46}

60/// An enum whose discriminants **skip zero** — the case an all-zero fill cannot
61/// be checked against.
62///
63/// A declared `enum_type`'s discriminants are the source's own, re-emitted
64/// verbatim, so the wire is this very enum and zero need not name a variant at
65/// all. Anything that fabricates a value for an absent slot builds an invalid
66/// one here, which is undefined behaviour whether or not the C side reads it
67/// (#428 review). `Option<f64>` cannot show that: every bit pattern is a legal
68/// `f64`.
69#[prebindgen]
70#[repr(i32)]
71#[derive(Debug, Clone, Copy, PartialEq, Eq)]
72pub enum Grade {
73    Low = 1,
74    High = 2,
75}

77/// A **data-carrying enum** (a sum type): exactly one alternative is live, and it
78/// carries that alternative's payload. Written as plain Rust — the invariant is in
79/// the type, not in a doc comment on a struct of optional fields.
80///
81/// The four variant shapes a sum can take are all here: a unit variant, a
82/// single-payload tuple variant, a multi-field named variant, and a tuple variant
83/// with an **owning** payload (a `String`, which crosses to C as a malloc'd
84/// `char *`) beside a payload that is itself a declared enum. The C adapter
85/// lowers the whole thing to a `#[repr(C)]` enum, which cbindgen renders as the
86/// idiomatic tag + `union`. (`lang::CbindgenBuilder` `.tagged_union`.)
87#[prebindgen]
88#[derive(Debug, Clone, PartialEq)]
89pub enum Shape {
90    /// Unit variant: the empty payload group — only the tag is live.
91    Empty,
92    /// Single-payload tuple variant.
93    Circle(f64),
94    /// Multi-field named variant.
95    Rect { width: f64, height: f64 },
96    /// Owning payload: the `String` is handed to C as a malloc'd `char *` that the
97    /// generated `shape_drop` releases, beside a declared-`enum_type` payload.
98    Labeled(String, Operation),
99}

101/// A by-value data struct used as a **union payload** — the shape zenoh-flat's
102/// `ReplyResult` needs, where an alternative carries a whole record rather than a
103/// scalar. Its `String` field owns memory, so the union's typed drop has to reach
104/// through the payload and release it.
105///
106/// The `bool` field is the second half of #170: a `data_struct` field crosses by
107/// value through a per-field wire, so a byte C wrote has to be normalised there
108/// too — and because this struct is also a union payload, that is what makes the
109/// tagged union's "no invalid value is ever materialised" invariant hold
110/// *transitively*, one level down.
111#[prebindgen]
112#[derive(Debug, Clone, PartialEq)]
113pub struct Caption {
114    pub id: u64,
115    pub text: String,
116    pub emphatic: bool,
117}
118
119/// Build a [`Caption`].

121pub fn caption_new(id: u64, text: &str, emphatic: bool) -> Caption {
122    Caption {
123        id,
124        text: text.to_string(),
125        emphatic,
126    }
127}

129/// A sum whose payloads are the shapes the zero-copy mirror policy could not
130/// express: a nested `data_struct` (by value, owning a `char *`) and a
131/// **converted leaf** whose wire is its converter's destination rather than its
132/// own layout.
133#[prebindgen]
134#[derive(Debug, Clone, PartialEq)]
135pub enum Note {
136    /// No payload — only the tag.
137    Silent,
138    /// A whole record by value, owning a `char *`.
139    Titled(Caption),
140    /// A converted leaf: `Millis` crosses as the `u64` its conversion produces.
141    After(Millis),
142    /// A `bool` payload — the one scalar whose domain is restricted (`0`/`1`),
143    /// so it crosses behind `MaybeUninit` like a declared enum does and a byte
144    /// C wrote is normalised, never materialised as a Rust `bool`.
145    Flagged(bool),
146    /// A record whose own field is ANOTHER union with an owning arm. The
147    /// payload crosses by value, so `note_drop` has to reach through it and
148    /// release the nested union's active arm — nothing else can: a union arm
149    /// is not a top-level struct field the C caller drops by hand.
150    Sketched(Drawing),
151}

263/// Area of a shape — the sum in **parameter** position, consumed by value. Every
264/// alternative is handled; there is no "both set" or "neither set" case to guard.
265#[prebindgen]
266pub fn shape_area(s: Shape) -> f64 {
267    match s {
268        Shape::Empty => 0.0,
269        Shape::Circle(r) => std::f64::consts::PI * r * r,
270        Shape::Rect { width, height } => width * height,
271        Shape::Labeled(_, _) => f64::NAN,
272    }
273}

275/// Area of a shape, reporting the alternative that has none through the error
276/// channel — the sum in parameter position on a **fallible** function. The C
277/// binding routes a rejected tag (a discriminant no variant has, which a C
278/// caller can always supply) to this same `char **e`, so the boundary check is
279/// observable instead of aborting.
280#[prebindgen]
281pub fn shape_try_area(s: Shape) -> Result<f64, Error> {
282    match s {
283        Shape::Labeled(_, _) => Err("a labeled shape has no area".to_string().into()),
284        other => Ok(shape_area(other)),
285    }
286}

309/// A stateful accumulator. This is a plain Rust type used as an opaque handle:
310/// the binding holds it behind a pointer and frees it with `calculator_drop`.
311///
312/// The definition lives in a private module and the flat API exports a marked
313/// alias to it. That is how a handle whose contents never cross gets a name here
314/// — the same shape zenoh-flat uses for the Zenoh types it re-exports — and the
315/// alias is transparent, so everything below still says `Calculator`.
316mod calculator {
317    pub struct Calculator {
318        pub(super) value: f64,
319        pub(super) history: Vec<f64>,
320    }
321}
322
323#[prebindgen]
324pub type Calculator = calculator::Calculator;

326/// Build a fresh accumulator initialized to zero.
327#[prebindgen]
328pub fn calculator_new() -> Calculator {
329    Calculator {
330        value: 0.0,
331        history: Vec::new(),
332    }
333}

335/// Parse an initial value from a string, returning an error on bad input

337#[prebindgen]

340    Ok(Calculator {

356/// Fold one accumulator into another, consuming **both**.
357///
358/// Two consumed handles of the same type is a supported shape, and it is the
359/// aliasing hazard: called as `calculator_merge(x, x)` a C caller would have
360/// one allocation reconstructed twice, so the generated wrapper runs a
361/// pointer-identity **preflight** before either conversion and reports the
362/// alias through this `Result`'s error channel.
363#[prebindgen]
364pub fn calculator_merge(a: Calculator, b: Calculator) -> Result<Calculator, Error> {
365    let mut history = a.history;
366    history.extend(b.history);
367    Ok(Calculator {
368        value: a.value + b.value,
369        history,
370    })
371}
372
373/// Consume one accumulator while **borrowing** another — the mixed case the
374/// "two or more consumed parameters" reading of the rule would have skipped.
375/// The borrow dangles the moment the consume takes ownership, so the same
376/// preflight covers it.
377#[prebindgen]
378pub fn calculator_absorb(a: Calculator, b: &Calculator) -> Result<f64, Error> {
379    Ok(a.value + b.value)
380}

382/// Apply `op` with `operand`, updating the accumulator and returning the new
383/// value. Division by zero returns an error (its fallible `&mut` input routes
384/// through the error channel of the `Result`).
385#[prebindgen]
386pub fn calculator_apply(c: &mut Calculator, op: Operation, operand: f64) -> Result<f64, Error> {
387    let next = match op {
388        Operation::Add => c.value + operand,
389        Operation::Sub => c.value - operand,
390        Operation::Mul => c.value * operand,
391        Operation::Div => {
392            if operand == 0.0 {
393                return Err("division by zero".to_string().into());
394            }
395            c.value / operand
396        }
397    };
398    c.value = next;
399    c.history.push(next);
400    Ok(next)
401}

434/// Invoke `f` once per recorded value in application order — replays the history
435/// into a C closure (demonstrates callback / closure-struct generation).
436#[prebindgen]
437pub fn calculator_for_each(c: &Calculator, f: impl Fn(f64) + Send + Sync + 'static) {
438    for v in &c.history {
439        f(*v);
440    }
441}

443/// The same delivery with an **optional** argument — a composite in argument
444/// position.
445///
446/// `Option`/`Vec`/`Cow` have no converter of their own: each resolves to a
447/// marker whose destination is `()`, and the real ABI is the shape it lowers
448/// to. The return path has always lowered them; the callback-argument path
449/// called that marker as if it were a converter, and a marker takes no
450/// arguments — so this shape emitted a binding that did not build (#428).
451///
452/// Fires once with the last recorded value and once with `None`, so both arms
453/// of the lowering reach C from a single call: an `Option<f64>` has no spare
454/// bit pattern, so it crosses as a `bool` beside the value.
455#[prebindgen]
456pub fn calculator_last_or_none(c: &Calculator, f: impl Fn(Option<f64>) + Send + Sync + 'static) {
457    f(c.history.last().copied());
458    f(None);
459}
460
461/// Deliver the whole history as an owned array — the composite whose lowering
462/// **allocates**.
463///
464/// A `Vec<T>` argument crosses as a malloc'd `(ptr, len)` pair the C side owns,
465/// so a closure whose `call` is NULL must not convert at all: nobody would
466/// receive that block, and nobody could free it (#428 review). Its sibling
467/// callbacks allocate nothing, so only this one can say whether the encode is
468/// inside the guard.
469#[prebindgen]
470pub fn calculator_history_batch(c: &Calculator, f: impl Fn(Vec<f64>) + Send + Sync + 'static) {
471    f(c.history.clone());
472}

474/// The same shape over a [`Grade`], whose discriminants skip zero.
475///
476/// The absent arm must leave the value slot alone rather than fill it: the wire
477/// IS the Rust enum, so a fabricated zero would be an invalid value of it. Fires
478/// present then absent, like its sibling.
479#[prebindgen]
480pub fn calculator_grade_or_none(c: &Calculator, f: impl Fn(Option<Grade>) + Send + Sync + 'static) {
481    f((c.value > 0.0).then_some(if c.value > 10.0 {
482        Grade::High
483    } else {
484        Grade::Low
485    }));
486    f(None);
487}

498// ─────────────────────────────────────────────────────────────────────────────
499// Multi-target cfg demonstration.
500//
501// These items show that `#[prebindgen]` captures per-target `cfg` and that the C
502// binding crate (`example-cbindgen`) then generates *different* code per target:
503// `InsideFoo`'s discriminants and `Foo`'s field set change with `target_arch`
504// (and `Foo` also varies by feature). Build for x86_64 vs aarch64 to get two
505// different `inside_foo_t` / `foo_t` in the generated header.
506// ─────────────────────────────────────────────────────────────────────────────
507
508/// A fieldless enum whose **discriminants differ by target architecture**. The two
509/// definitions are mutually exclusive — the `#[prebindgen(cfg = ...)]` macro emits a
510/// matching real `#[cfg]`, so each target compiles exactly one and the generated C
511/// enum carries that target's values. (`lang::CbindgenBuilder` `.enum_type`.)
512#[prebindgen("structs", cfg = "target_arch = \"x86_64\"")]
513#[repr(i32)]
514#[derive(Clone, Copy, Debug, PartialEq, Eq)]
515pub enum InsideFoo {
516    DouddleDee = 42,
517    DouddleDum = 24,
518}
519#[prebindgen("structs", cfg = "target_arch = \"aarch64\"")]
520#[repr(i32)]
521#[derive(Clone, Copy, Debug, PartialEq, Eq)]
522pub enum InsideFoo {
523    DouddleDee = 14,
524    DouddleDum = 88,
525}
526
527/// A by-value data struct whose **field set varies by target architecture and by
528/// feature**. `#[prebindgen]` records every `cfg`-gated field; the binding crate
529/// keeps only those matching the build target, so the generated `#[repr(C)] foo_t`
530/// differs per target. (`lang::CbindgenBuilder` `.data_struct`.)
531#[prebindgen("structs")]
532#[derive(Clone, Copy, Debug, Default, PartialEq)]
533pub struct Foo {
534    /// Always present.
535    pub id: u64,
536    #[cfg(target_arch = "x86_64")]
537    pub x86_64_field: u64,
538    #[cfg(target_arch = "aarch64")]
539    pub aarch64_field: u64,
540    #[cfg(feature = "unstable")]
541    pub unstable_field: u64,
542    #[cfg(not(feature = "unstable"))]
543    pub stable_field: u64,
544}

552    }
553}
